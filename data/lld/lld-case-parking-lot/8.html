Full code

Complete, runnable, and the same files that sit in lld-code/lld-case-parking-lot/. Java is the primary version (four files, no packages, so javac *.java just works). Python mirrors it with a unittest file. The C++14 version is single-threaded because the MinGW g++ 6.3 used to build it has no std::thread; the comments mark where the mutexes go.

Read in this order (the files are long; this is the 10-minute path):

  1. Main.main (Main.java): run it first and match the output to the lot map from the story.
  2. ParkingLot.park and unpark (ParkingLot.java, about 40 lines): the facade and the heart of the design. Note the plate claim with putIfAbsent and its undo.
  3. Level.tryTake / release: one atomic pollFirst() per spot claim.
  4. NearestFirstAllocation and HourlyFee (Strategies.java): the two strategies; then PeakHourFee.
  5. Skip on first read: the enums, records and exceptions in Model.java.

From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

The longer version is OOP for C programmers.

Model.java - enums, value objects, the vehicle factory, exceptions.

import java.time.LocalDateTime;
import java.util.List;

// ---------- enums: the fixed vocabulary of the problem ----------
enum SpotSize { SMALL, MEDIUM, LARGE }

enum VehicleType {
    // which spot sizes a vehicle may use, smallest first
    MOTORCYCLE(List.of(SpotSize.SMALL, SpotSize.MEDIUM, SpotSize.LARGE)),
    CAR(List.of(SpotSize.MEDIUM, SpotSize.LARGE)),
    TRUCK(List.of(SpotSize.LARGE));

    private final List<SpotSize> fits;
    VehicleType(List<SpotSize> fits) { this.fits = fits; }
    List<SpotSize> fits() { return fits; }
}

// ---------- value objects (immutable) ----------
record Vehicle(String plate, VehicleType type) {}

record Ticket(String id, Vehicle vehicle, ParkingSpot spot, LocalDateTime entry) {}

record Receipt(Ticket ticket, LocalDateTime exit, long minutes, long feeRupees) {}

// ---------- Factory: turns raw input ("car", "ka01ab1234") into a valid Vehicle ----------
final class VehicleFactory {
    private VehicleFactory() {}

    static Vehicle create(String type, String plate) {
        if (plate == null || plate.isBlank()) throw new IllegalArgumentException("plate is required");
        if (type == null) throw new IllegalArgumentException("vehicle type is required");
        VehicleType t;
        try {
            t = VehicleType.valueOf(type.trim().toUpperCase());
        } catch (IllegalArgumentException e) {
            throw new IllegalArgumentException("unknown vehicle type: " + type);
        }
        return new Vehicle(plate.trim().toUpperCase(), t);
    }
}

// ---------- a spot is identity + size; "free or taken" lives in Level's free sets ----------
final class ParkingSpot implements Comparable<ParkingSpot> {
    private final int level;
    private final int number;
    private final SpotSize size;

    ParkingSpot(int level, int number, SpotSize size) {
        this.level = level;
        this.number = number;
        this.size = size;
    }

    int level() { return level; }
    int number() { return number; }
    SpotSize size() { return size; }
    String id() { return "L" + level + "-" + number; }

    // lower number = nearer the entry gate of that level
    @Override public int compareTo(ParkingSpot o) { return Integer.compare(number, o.number); }
    @Override public String toString() { return id() + "(" + size + ")"; }
}

// ---------- domain exceptions: callers can tell "full" from "bad ticket" ----------
class LotFullException extends RuntimeException {
    LotFullException(String msg) { super(msg); }
}

class InvalidTicketException extends RuntimeException {
    InvalidTicketException(String msg) { super(msg); }
}

Strategies.java - allocation and pricing strategies, the pricing factory, the listener interface.

import java.time.Duration;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;
import java.util.Optional;

// ---------- Strategy 1: which spot does a vehicle get? ----------
interface SpotAllocationStrategy {
    String name();
    Optional<ParkingSpot> allocate(List<Level> levels, Vehicle vehicle);
}

// lowest level first; inside a level, the smallest size that fits
final class NearestFirstAllocation implements SpotAllocationStrategy {
    public String name() { return "nearest-first"; }

    public Optional<ParkingSpot> allocate(List<Level> levels, Vehicle v) {
        for (Level level : levels)
            for (SpotSize size : v.type().fits()) {
                Optional<ParkingSpot> spot = level.tryTake(size);
                if (spot.isPresent()) return spot;
            }
        return Optional.empty();
    }
}

// smallest size that fits anywhere in the lot first, so big spots stay free for trucks
final class BestFitAllocation implements SpotAllocationStrategy {
    public String name() { return "best-fit"; }

    public Optional<ParkingSpot> allocate(List<Level> levels, Vehicle v) {
        for (SpotSize size : v.type().fits())
            for (Level level : levels) {
                Optional<ParkingSpot> spot = level.tryTake(size);
                if (spot.isPresent()) return spot;
            }
        return Optional.empty();
    }
}

// ---------- Strategy 2: how much does a stay cost? ----------
interface FeeStrategy {
    String name();
    long fee(VehicleType type, LocalDateTime entry, LocalDateTime exit);

    // every started hour is billed, minimum one hour
    static long billableHours(LocalDateTime entry, LocalDateTime exit) {
        long minutes = Duration.between(entry, exit).toMinutes();
        return Math.max(1, (minutes + 59) / 60);
    }
}

final class HourlyFee implements FeeStrategy {
    private final Map<VehicleType, Long> ratePerHour;
    HourlyFee(Map<VehicleType, Long> ratePerHour) { this.ratePerHour = Map.copyOf(ratePerHour); }

    public String name() { return "hourly"; }

    public long fee(VehicleType type, LocalDateTime entry, LocalDateTime exit) {
        return FeeStrategy.billableHours(entry, exit) * ratePerHour.get(type);
    }
}

// each billed hour that STARTS inside [peakStart, peakEnd) costs peakPercent% of the base rate
final class PeakHourFee implements FeeStrategy {
    private final Map<VehicleType, Long> ratePerHour;
    private final int peakStartHour, peakEndHour, peakPercent;

    PeakHourFee(Map<VehicleType, Long> ratePerHour, int peakStartHour, int peakEndHour, int peakPercent) {
        this.ratePerHour = Map.copyOf(ratePerHour);
        this.peakStartHour = peakStartHour;
        this.peakEndHour = peakEndHour;
        this.peakPercent = peakPercent;
    }

    public String name() { return "peak " + peakStartHour + "-" + peakEndHour + "h x" + peakPercent + "%"; }

    public long fee(VehicleType type, LocalDateTime entry, LocalDateTime exit) {
        long hours = FeeStrategy.billableHours(entry, exit);
        long base = ratePerHour.get(type), total = 0;
        for (long h = 0; h < hours; h++) {
            int hourOfDay = entry.plusHours(h).getHour();
            boolean peak = hourOfDay >= peakStartHour && hourOfDay < peakEndHour;
            total += peak ? base * peakPercent / 100 : base;
        }
        return total;
    }
}

// ---------- Factory: build a pricing strategy from a config string ----------
final class FeeStrategyFactory {
    private FeeStrategyFactory() {}

    static FeeStrategy create(String name, Map<VehicleType, Long> rates) {
        return switch (name.trim().toLowerCase()) {
            case "hourly" -> new HourlyFee(rates);
            case "peak" -> new PeakHourFee(rates, 17, 21, 150);
            default -> throw new IllegalArgumentException("unknown pricing: " + name);
        };
    }
}

// ---------- Observer: anyone who wants to hear about entries and exits ----------
interface ParkingEventListener {
    void onParked(Ticket ticket);
    void onLeft(Receipt receipt);
}

ParkingLot.java - Level (free sets), ParkingLot (the service), DisplayBoard (an observer).

import java.time.Duration;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentSkipListSet;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicLong;

// One floor. Free spots are kept per size in a concurrent sorted set:
// pollFirst() removes the nearest free spot atomically, so two threads can never get the same spot.
final class Level {
    private final int number;
    private final Map<SpotSize, ConcurrentSkipListSet<ParkingSpot>> free = new EnumMap<>(SpotSize.class);
    private final Map<SpotSize, Integer> capacity = new EnumMap<>(SpotSize.class);

    Level(int number, int small, int medium, int large) {
        this.number = number;
        for (SpotSize s : SpotSize.values()) free.put(s, new ConcurrentSkipListSet<>());
        int next = 1;
        next = addSpots(SpotSize.SMALL, small, next);
        next = addSpots(SpotSize.MEDIUM, medium, next);
        addSpots(SpotSize.LARGE, large, next);
    }

    private int addSpots(SpotSize size, int count, int next) {
        capacity.put(size, count);
        for (int i = 0; i < count; i++) free.get(size).add(new ParkingSpot(number, next++, size));
        return next;
    }

    int number() { return number; }
    Optional<ParkingSpot> tryTake(SpotSize size) { return Optional.ofNullable(free.get(size).pollFirst()); }

    void release(ParkingSpot spot) {
        if (!free.get(spot.size()).add(spot)) throw new IllegalStateException("spot released twice: " + spot.id());
    }

    int freeCount(SpotSize size) { return free.get(size).size(); }
    int capacity(SpotSize size) { return capacity.get(size); }
}

// The service the gates talk to. Plain class, created once by main and passed around (no global Singleton).
final class ParkingLot {
    private final List<Level> levels;
    private volatile SpotAllocationStrategy allocation;
    private volatile FeeStrategy pricing;
    private final ConcurrentHashMap<String, Ticket> activeTickets = new ConcurrentHashMap<>();
    private final ConcurrentHashMap<String, String> ticketByPlate = new ConcurrentHashMap<>();
    private final AtomicLong ticketSeq = new AtomicLong();
    private final List<ParkingEventListener> listeners = new CopyOnWriteArrayList<>();

    ParkingLot(List<Level> levels, SpotAllocationStrategy allocation, FeeStrategy pricing) {
        this.levels = List.copyOf(levels);
        this.allocation = allocation;
        this.pricing = pricing;
    }

    void setAllocation(SpotAllocationStrategy s) { this.allocation = s; }
    void setPricing(FeeStrategy p) { this.pricing = p; }
    void addListener(ParkingEventListener l) { listeners.add(l); }

    Ticket park(Vehicle vehicle, LocalDateTime at) {
        String ticketId = "T" + ticketSeq.incrementAndGet();
        // claim the plate first: the same car cannot be inside twice
        if (ticketByPlate.putIfAbsent(vehicle.plate(), ticketId) != null)
            throw new IllegalStateException(vehicle.plate() + " is already parked");
        Optional<ParkingSpot> spot = allocation.allocate(levels, vehicle);
        if (spot.isEmpty()) {
            ticketByPlate.remove(vehicle.plate(), ticketId);   // undo the claim
            throw new LotFullException("no free spot for " + vehicle.type() + " " + vehicle.plate());
        }
        Ticket ticket = new Ticket(ticketId, vehicle, spot.get(), at);
        activeTickets.put(ticketId, ticket);
        listeners.forEach(l -> l.onParked(ticket));
        return ticket;
    }

    Receipt unpark(String ticketId, LocalDateTime at) {
        Ticket ticket = activeTickets.get(ticketId);
        if (ticket == null) throw new InvalidTicketException("unknown or already used ticket " + ticketId);
        if (at.isBefore(ticket.entry())) throw new IllegalArgumentException("exit time is before entry time");
        // remove(key, value) is atomic: if two exits race with one ticket, only one wins
        if (!activeTickets.remove(ticketId, ticket))
            throw new InvalidTicketException("ticket " + ticketId + " was already used");
        long fee = pricing.fee(ticket.vehicle().type(), ticket.entry(), at);
        long minutes = Duration.between(ticket.entry(), at).toMinutes();
        levels.get(ticket.spot().level() - 1).release(ticket.spot());
        ticketByPlate.remove(ticket.vehicle().plate(), ticketId);
        Receipt receipt = new Receipt(ticket, at, minutes, fee);
        listeners.forEach(l -> l.onLeft(receipt));
        return receipt;
    }

    Map<SpotSize, Integer> freeSpots() {
        Map<SpotSize, Integer> out = new EnumMap<>(SpotSize.class);
        for (SpotSize s : SpotSize.values()) {
            int n = 0;
            for (Level l : levels) n += l.freeCount(s);
            out.put(s, n);
        }
        return out;
    }

    List<Ticket> activeTickets() {
        List<Ticket> list = new ArrayList<>(activeTickets.values());
        list.sort((a, b) -> a.spot().id().compareTo(b.spot().id()));
        return Collections.unmodifiableList(list);
    }

    String allocationName() { return allocation.name(); }
    String pricingName() { return pricing.name(); }
}

// Observer example: an entrance display that shows free spots per size
final class DisplayBoard implements ParkingEventListener {
    private final ParkingLot lot;
    private volatile String text = "";
    DisplayBoard(ParkingLot lot) { this.lot = lot; }

    public void onParked(Ticket t) { refresh(); }
    public void onLeft(Receipt r) { refresh(); }
    private void refresh() { text = "FREE " + lot.freeSpots(); }
    String text() { return text; }
}

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static void expectThrows(Class<? extends RuntimeException> type, Runnable r, String what) {
        try {
            r.run();
        } catch (RuntimeException e) {
            check(type.isInstance(e), what + " (got " + e.getClass().getSimpleName() + ")");
            System.out.println("  rejected as expected: " + e.getMessage());
            return;
        }
        throw new AssertionError("FAILED: expected " + type.getSimpleName() + " for " + what);
    }

    static LocalDateTime at(int h, int m) { return LocalDateTime.of(2026, 11, 23, h, m); }

    public static void main(String[] args) throws Exception {
        Map<VehicleType, Long> rates = Map.of(VehicleType.MOTORCYCLE, 10L, VehicleType.CAR, 20L, VehicleType.TRUCK, 50L);
        // L1: spots 1 small, 2-3 medium, 4 large.  L2: 1-2 small, 3-4 medium, 5 large.
        List<Level> levels = List.of(new Level(1, 1, 2, 1), new Level(2, 2, 2, 1));
        ParkingLot lot = new ParkingLot(levels, new NearestFirstAllocation(), FeeStrategyFactory.create("hourly", rates));
        DisplayBoard board = new DisplayBoard(lot);
        lot.addListener(board);

        System.out.println("== 1. park with " + lot.allocationName() + " ==");
        Ticket bike = lot.park(VehicleFactory.create("motorcycle", "ka01-b1"), at(9, 0));
        Ticket c1 = lot.park(VehicleFactory.create("car", "ka01-c1"), at(9, 10));
        Ticket c2 = lot.park(VehicleFactory.create("car", "ka01-c2"), at(9, 15));
        Ticket c3 = lot.park(VehicleFactory.create("car", "ka01-c3"), at(9, 20));
        for (Ticket t : List.of(bike, c1, c2, c3)) System.out.println("  " + t.id() + " " + t.vehicle().plate() + " -> " + t.spot());
        check(bike.spot().id().equals("L1-1"), "bike takes the small spot on L1");
        check(c3.spot().id().equals("L1-4"), "nearest-first puts the 3rd car in L1's LARGE spot");
        System.out.println("  board: " + board.text());

        System.out.println("== 2. switch allocation to best-fit ==");
        lot.setAllocation(new BestFitAllocation());
        Ticket c4 = lot.park(VehicleFactory.create("car", "ka01-c4"), at(9, 30));
        Ticket truck = lot.park(VehicleFactory.create("truck", "ka01-t1"), at(9, 40));
        System.out.println("  " + c4.id() + " " + c4.vehicle().plate() + " -> " + c4.spot());
        System.out.println("  " + truck.id() + " " + truck.vehicle().plate() + " -> " + truck.spot());
        check(c4.spot().id().equals("L2-3"), "best-fit gives the car a MEDIUM spot on L2");
        check(truck.spot().id().equals("L2-5"), "truck gets the only free LARGE spot");

        System.out.println("== 3. invalid operations ==");
        expectThrows(LotFullException.class, () -> lot.park(VehicleFactory.create("truck", "ka01-t2"), at(9, 45)), "no large spot left");
        expectThrows(IllegalStateException.class, () -> lot.park(VehicleFactory.create("car", "KA01-C1"), at(9, 50)), "same plate twice");
        expectThrows(IllegalArgumentException.class, () -> VehicleFactory.create("rocket", "x1"), "unknown vehicle type");
        expectThrows(IllegalArgumentException.class, () -> lot.unpark(c2.id(), at(9, 0)), "exit before entry");

        System.out.println("== 4. exit with " + lot.pricingName() + " pricing ==");
        Receipt r1 = lot.unpark(c1.id(), at(11, 30));   // 2h20m -> 3 billed hours x 20
        System.out.println("  " + r1.ticket().vehicle().plate() + " stayed " + r1.minutes() + " min, pays Rs " + r1.feeRupees());
        check(r1.feeRupees() == 60, "2h20m car = 3 hours x Rs 20");
        expectThrows(InvalidTicketException.class, () -> lot.unpark(c1.id(), at(11, 31)), "same ticket used twice");
        Receipt r2 = lot.unpark(bike.id(), at(9, 5));   // 5 minutes -> minimum 1 hour
        check(r2.feeRupees() == 10, "minimum one billed hour");

        System.out.println("== 5. peak pricing (17:00-21:00 at 150%) ==");
        lot.setPricing(FeeStrategyFactory.create("peak", rates));
        Ticket c5 = lot.park(VehicleFactory.create("car", "ka01-c5"), at(16, 30));
        Receipt r3 = lot.unpark(c5.id(), at(19, 10));   // hours start 16:30, 17:30, 18:30 -> 20 + 30 + 30
        System.out.println("  " + r3.ticket().vehicle().plate() + " stayed " + r3.minutes() + " min, pays Rs " + r3.feeRupees());
        check(r3.feeRupees() == 80, "one normal hour + two peak hours = 80");
        System.out.println("  board: " + board.text());
        check(lot.freeSpots().get(SpotSize.MEDIUM) == 2, "two MEDIUM spots free after exits");

        System.out.println("== 6. concurrency: 200 cars race for 50 spots on 8 threads ==");
        concurrencyTest();

        System.out.println("All " + checks + " checks passed");
    }

    static void concurrencyTest() throws Exception {
        ParkingLot lot = new ParkingLot(List.of(new Level(1, 0, 50, 0)), new NearestFirstAllocation(),
                new HourlyFee(Map.of(VehicleType.MOTORCYCLE, 10L, VehicleType.CAR, 20L, VehicleType.TRUCK, 50L)));
        ExecutorService pool = Executors.newFixedThreadPool(8);
        CountDownLatch start = new CountDownLatch(1);
        ConcurrentLinkedQueue<Ticket> issued = new ConcurrentLinkedQueue<>();
        AtomicInteger full = new AtomicInteger();
        for (int i = 0; i < 200; i++) {
            final int n = i;
            pool.submit(() -> {
                try {
                    start.await();
                    issued.add(lot.park(new Vehicle("CAR-" + n, VehicleType.CAR), at(10, 0)));
                } catch (LotFullException e) {
                    full.incrementAndGet();
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
        }
        start.countDown();   // release all threads at once
        pool.shutdown();
        check(pool.awaitTermination(10, TimeUnit.SECONDS), "threads finished");
        Set<String> spotIds = new HashSet<>();
        for (Ticket t : issued) spotIds.add(t.spot().id());
        System.out.println("  parked=" + issued.size() + " rejected=" + full.get() + " distinct spots=" + spotIds.size());
        check(issued.size() == 50 && full.get() == 150, "exactly 50 cars parked");
        check(spotIds.size() == 50, "no spot was given to two cars");

        // everyone leaves at once; each ticket must be accepted exactly once
        ExecutorService exitPool = Executors.newFixedThreadPool(8);
        AtomicInteger exits = new AtomicInteger(), rejectedExits = new AtomicInteger();
        List<Ticket> tickets = new ArrayList<>(issued);
        for (Ticket t : tickets)
            for (int k = 0; k < 2; k++)   // every ticket is presented twice
                exitPool.submit(() -> {
                    try { lot.unpark(t.id(), at(11, 0)); exits.incrementAndGet(); }
                    catch (InvalidTicketException e) { rejectedExits.incrementAndGet(); }
                });
        exitPool.shutdown();
        check(exitPool.awaitTermination(10, TimeUnit.SECONDS), "exit threads finished");
        System.out.println("  exits=" + exits.get() + " duplicate exits rejected=" + rejectedExits.get() + " free=" + lot.freeSpots().get(SpotSize.MEDIUM));
        check(exits.get() == 50 && rejectedExits.get() == 50, "each ticket exits once");
        check(lot.freeSpots().get(SpotSize.MEDIUM) == 50, "all 50 spots free again");
    }
}

parking_lot.py - the whole model. One lock per level guards its heaps; one lock guards the ticket maps.

"""Parking lot - machine-coding reference (Python 3.11). Same design as the Java version."""
from __future__ import annotations

import heapq
import itertools
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import datetime, timedelta
from enum import Enum


class SpotSize(Enum):
    SMALL = 1
    MEDIUM = 2
    LARGE = 3


class VehicleType(Enum):
    MOTORCYCLE = (SpotSize.SMALL, SpotSize.MEDIUM, SpotSize.LARGE)   # smallest fit first
    CAR = (SpotSize.MEDIUM, SpotSize.LARGE)
    TRUCK = (SpotSize.LARGE,)

    @property
    def fits(self) -> tuple[SpotSize, ...]:
        return self.value


@dataclass(frozen=True)
class Vehicle:
    plate: str
    type: VehicleType


@dataclass(frozen=True, order=True)
class ParkingSpot:
    level: int
    number: int
    size: SpotSize

    @property
    def id(self) -> str:
        return f"L{self.level}-{self.number}"

    def __str__(self) -> str:
        return f"{self.id}({self.size.name})"


@dataclass(frozen=True)
class Ticket:
    id: str
    vehicle: Vehicle
    spot: ParkingSpot
    entry: datetime


@dataclass(frozen=True)
class Receipt:
    ticket: Ticket
    exit: datetime
    minutes: int
    fee_rupees: int


class LotFullError(Exception):
    pass


class InvalidTicketError(Exception):
    pass


class VehicleFactory:
    @staticmethod
    def create(kind: str, plate: str) -> Vehicle:
        if not plate or not plate.strip():
            raise ValueError("plate is required")
        try:
            vtype = VehicleType[kind.strip().upper()]
        except KeyError:
            raise ValueError(f"unknown vehicle type: {kind}") from None
        return Vehicle(plate.strip().upper(), vtype)


class Level:
    """One floor. A min-heap of free spots per size; one lock per level guards the heaps."""

    def __init__(self, number: int, small: int, medium: int, large: int) -> None:
        self.number = number
        self._lock = threading.Lock()
        self._free: dict[SpotSize, list[ParkingSpot]] = {s: [] for s in SpotSize}
        n = itertools.count(1)
        for size, count in ((SpotSize.SMALL, small), (SpotSize.MEDIUM, medium), (SpotSize.LARGE, large)):
            for _ in range(count):
                self._free[size].append(ParkingSpot(number, next(n), size))
            heapq.heapify(self._free[size])

    def try_take(self, size: SpotSize) -> ParkingSpot | None:
        with self._lock:                      # check-then-take must be one critical section
            heap = self._free[size]
            return heapq.heappop(heap) if heap else None

    def release(self, spot: ParkingSpot) -> None:
        with self._lock:
            if spot in self._free[spot.size]:
                raise RuntimeError(f"spot released twice: {spot.id}")
            heapq.heappush(self._free[spot.size], spot)

    def free_count(self, size: SpotSize) -> int:
        with self._lock:
            return len(self._free[size])


# ---------- Strategy: allocation ----------
class SpotAllocationStrategy(ABC):
    name = "?"

    @abstractmethod
    def allocate(self, levels: list[Level], vehicle: Vehicle) -> ParkingSpot | None: ...


class NearestFirstAllocation(SpotAllocationStrategy):
    name = "nearest-first"

    def allocate(self, levels, vehicle):
        for level in levels:
            for size in vehicle.type.fits:
                if spot := level.try_take(size):
                    return spot
        return None


class BestFitAllocation(SpotAllocationStrategy):
    name = "best-fit"

    def allocate(self, levels, vehicle):
        for size in vehicle.type.fits:
            for level in levels:
                if spot := level.try_take(size):
                    return spot
        return None


# ---------- Strategy: pricing ----------
def billable_hours(entry: datetime, exit_: datetime) -> int:
    minutes = int((exit_ - entry).total_seconds() // 60)
    return max(1, (minutes + 59) // 60)


class FeeStrategy(ABC):
    name = "?"

    @abstractmethod
    def fee(self, vtype: VehicleType, entry: datetime, exit_: datetime) -> int: ...


class HourlyFee(FeeStrategy):
    name = "hourly"

    def __init__(self, rates: dict[VehicleType, int]) -> None:
        self.rates = dict(rates)

    def fee(self, vtype, entry, exit_):
        return billable_hours(entry, exit_) * self.rates[vtype]


class PeakHourFee(FeeStrategy):
    """Each billed hour that starts inside [start, end) costs percent% of the base rate."""

    def __init__(self, rates, start_hour: int, end_hour: int, percent: int) -> None:
        self.rates, self.start, self.end, self.percent = dict(rates), start_hour, end_hour, percent
        self.name = f"peak {start_hour}-{end_hour}h x{percent}%"

    def fee(self, vtype, entry, exit_):
        base, total = self.rates[vtype], 0
        for h in range(billable_hours(entry, exit_)):
            hour = (entry + timedelta(hours=h)).hour
            total += base * self.percent // 100 if self.start <= hour < self.end else base
        return total


class FeeStrategyFactory:
    @staticmethod
    def create(name: str, rates: dict[VehicleType, int]) -> FeeStrategy:
        match name.strip().lower():
            case "hourly":
                return HourlyFee(rates)
            case "peak":
                return PeakHourFee(rates, 17, 21, 150)
        raise ValueError(f"unknown pricing: {name}")


# ---------- Observer ----------
class ParkingEventListener(ABC):
    @abstractmethod
    def on_parked(self, ticket: Ticket) -> None: ...

    @abstractmethod
    def on_left(self, receipt: Receipt) -> None: ...


class ParkingLot:
    def __init__(self, levels: list[Level], allocation: SpotAllocationStrategy, pricing: FeeStrategy) -> None:
        self._levels = list(levels)
        self.allocation = allocation
        self.pricing = pricing
        self._lock = threading.Lock()          # guards the two ticket maps
        self._active: dict[str, Ticket] = {}
        self._ticket_by_plate: dict[str, str] = {}
        self._seq = itertools.count(1)
        self._listeners: list[ParkingEventListener] = []

    def add_listener(self, listener: ParkingEventListener) -> None:
        self._listeners.append(listener)

    def park(self, vehicle: Vehicle, at: datetime) -> Ticket:
        with self._lock:
            if vehicle.plate in self._ticket_by_plate:
                raise ValueError(f"{vehicle.plate} is already parked")
            ticket_id = f"T{next(self._seq)}"
            self._ticket_by_plate[vehicle.plate] = ticket_id   # claim the plate
        spot = self.allocation.allocate(self._levels, vehicle)  # level locks, not the lot lock
        if spot is None:
            with self._lock:
                del self._ticket_by_plate[vehicle.plate]
            raise LotFullError(f"no free spot for {vehicle.type.name} {vehicle.plate}")
        ticket = Ticket(ticket_id, vehicle, spot, at)
        with self._lock:
            self._active[ticket_id] = ticket
        for listener in self._listeners:
            listener.on_parked(ticket)
        return ticket

    def unpark(self, ticket_id: str, at: datetime) -> Receipt:
        with self._lock:
            ticket = self._active.get(ticket_id)
            if ticket is None:
                raise InvalidTicketError(f"unknown or already used ticket {ticket_id}")
            if at < ticket.entry:
                raise ValueError("exit time is before entry time")
            del self._active[ticket_id]                      # only one exit can get here
            del self._ticket_by_plate[ticket.vehicle.plate]
        fee = self.pricing.fee(ticket.vehicle.type, ticket.entry, at)
        self._levels[ticket.spot.level - 1].release(ticket.spot)
        minutes = int((at - ticket.entry).total_seconds() // 60)
        receipt = Receipt(ticket, at, minutes, fee)
        for listener in self._listeners:
            listener.on_left(receipt)
        return receipt

    def free_spots(self) -> dict[str, int]:
        return {s.name: sum(level.free_count(s) for level in self._levels) for s in SpotSize}


class DisplayBoard(ParkingEventListener):
    def __init__(self, lot: ParkingLot) -> None:
        self.lot, self.text = lot, ""

    def on_parked(self, ticket):
        self.text = f"FREE {self.lot.free_spots()}"

    def on_left(self, receipt):
        self.text = f"FREE {self.lot.free_spots()}"

// Parking lot in C++14: same design, single-threaded demo.
// Build: g++ -std=c++14 -Wall -O2 main.cpp -o parking && ./parking
// In a multi-threaded build, give each Level a std::mutex around takeSpot/release
// and guard the ticket map with another one (the lock points are marked LOCK).
#include <cassert>
#include <iostream>
#include <map>
#include <memory>
#include <set>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>

enum class SpotSize { SMALL, MEDIUM, LARGE };
enum class VehicleType { MOTORCYCLE, CAR, TRUCK };

static const char* sizeName(SpotSize s) {
    switch (s) { case SpotSize::SMALL: return "SMALL"; case SpotSize::MEDIUM: return "MEDIUM"; default: return "LARGE"; }
}

static std::vector<SpotSize> fits(VehicleType t) {   // smallest first
    switch (t) {
        case VehicleType::MOTORCYCLE: return {SpotSize::SMALL, SpotSize::MEDIUM, SpotSize::LARGE};
        case VehicleType::CAR: return {SpotSize::MEDIUM, SpotSize::LARGE};
        default: return {SpotSize::LARGE};
    }
}

struct Vehicle { std::string plate; VehicleType type; };

struct ParkingSpot {
    int level, number; SpotSize size;
    std::string id() const { return "L" + std::to_string(level) + "-" + std::to_string(number); }
    bool operator<(const ParkingSpot& o) const { return number < o.number; }
};

struct Ticket { std::string id; Vehicle vehicle; ParkingSpot spot; int entryMin; };   // minutes since midnight

struct LotFull : std::runtime_error { using std::runtime_error::runtime_error; };
struct InvalidTicket : std::runtime_error { using std::runtime_error::runtime_error; };

class Level {
    int number_;
    std::map<SpotSize, std::set<ParkingSpot>> free_;   // ordered: begin() = nearest spot
public:
    Level(int number, int small, int medium, int large) : number_(number) {
        int n = 1;
        for (int i = 0; i < small; i++) free_[SpotSize::SMALL].insert({number, n++, SpotSize::SMALL});
        for (int i = 0; i < medium; i++) free_[SpotSize::MEDIUM].insert({number, n++, SpotSize::MEDIUM});
        for (int i = 0; i < large; i++) free_[SpotSize::LARGE].insert({number, n++, SpotSize::LARGE});
    }
    bool takeSpot(SpotSize s, ParkingSpot& out) {      // LOCK: check-then-take is one critical section
        auto& set = free_[s];
        if (set.empty()) return false;
        out = *set.begin();
        set.erase(set.begin());
        return true;
    }
    void release(const ParkingSpot& p) {               // LOCK
        if (!free_[p.size].insert(p).second) throw std::logic_error("spot released twice: " + p.id());
    }
    int freeCount(SpotSize s) { return (int)free_[s].size(); }
};

// ---------- Strategy interfaces: an abstract class = a vtable, like struct file_operations ----------
struct SpotAllocationStrategy {
    virtual ~SpotAllocationStrategy() {}
    virtual bool allocate(std::vector<Level>& levels, const Vehicle& v, ParkingSpot& out) = 0;
};

struct NearestFirst : SpotAllocationStrategy {
    bool allocate(std::vector<Level>& levels, const Vehicle& v, ParkingSpot& out) override {
        for (auto& level : levels)
            for (SpotSize s : fits(v.type))
                if (level.takeSpot(s, out)) return true;
        return false;
    }
};

struct BestFit : SpotAllocationStrategy {
    bool allocate(std::vector<Level>& levels, const Vehicle& v, ParkingSpot& out) override {
        for (SpotSize s : fits(v.type))
            for (auto& level : levels)
                if (level.takeSpot(s, out)) return true;
        return false;
    }
};

struct FeeStrategy {
    virtual ~FeeStrategy() {}
    virtual long fee(VehicleType t, int entryMin, int exitMin) const = 0;
    static long billableHours(int entryMin, int exitMin) {
        long m = exitMin - entryMin;
        return m <= 0 ? 1 : (m + 59) / 60;
    }
};

struct HourlyFee : FeeStrategy {
    std::map<VehicleType, long> rate;
    explicit HourlyFee(std::map<VehicleType, long> r) : rate(std::move(r)) {}
    long fee(VehicleType t, int in, int out) const override { return billableHours(in, out) * rate.at(t); }
};

struct PeakHourFee : FeeStrategy {
    std::map<VehicleType, long> rate; int startH, endH, pct;
    PeakHourFee(std::map<VehicleType, long> r, int s, int e, int p) : rate(std::move(r)), startH(s), endH(e), pct(p) {}
    long fee(VehicleType t, int in, int out) const override {
        long total = 0, base = rate.at(t);
        for (long h = 0; h < billableHours(in, out); h++) {
            int hourOfDay = (int)((in + h * 60) / 60) % 24;
            total += (hourOfDay >= startH && hourOfDay < endH) ? base * pct / 100 : base;
        }
        return total;
    }
};

class ParkingLot {
    std::vector<Level> levels_;
    std::unique_ptr<SpotAllocationStrategy> alloc_;
    std::unique_ptr<FeeStrategy> pricing_;
    std::map<std::string, Ticket> active_;          // LOCK (lot mutex)
    std::map<std::string, std::string> byPlate_;   // LOCK (lot mutex)
    long seq_ = 0;
public:
    ParkingLot(std::vector<Level> levels, std::unique_ptr<SpotAllocationStrategy> a, std::unique_ptr<FeeStrategy> f)
        : levels_(std::move(levels)), alloc_(std::move(a)), pricing_(std::move(f)) {}
    void setAllocation(std::unique_ptr<SpotAllocationStrategy> a) { alloc_ = std::move(a); }
    void setPricing(std::unique_ptr<FeeStrategy> f) { pricing_ = std::move(f); }

    Ticket park(const Vehicle& v, int atMin) {
        if (byPlate_.count(v.plate)) throw std::invalid_argument(v.plate + " is already parked");
        ParkingSpot spot{};
        if (!alloc_->allocate(levels_, v, spot)) throw LotFull("no free spot for " + v.plate);
        Ticket t{"T" + std::to_string(++seq_), v, spot, atMin};
        active_[t.id] = t;
        byPlate_[v.plate] = t.id;
        return t;
    }

    long unpark(const std::string& ticketId, int atMin) {
        auto it = active_.find(ticketId);
        if (it == active_.end()) throw InvalidTicket("unknown or already used ticket " + ticketId);
        Ticket t = it->second;
        if (atMin < t.entryMin) throw std::invalid_argument("exit time is before entry time");
        active_.erase(it);
        byPlate_.erase(t.vehicle.plate);
        levels_[t.spot.level - 1].release(t.spot);
        return pricing_->fee(t.vehicle.type, t.entryMin, atMin);
    }

    std::string freeSpots() {
        std::ostringstream os;
        for (SpotSize s : {SpotSize::SMALL, SpotSize::MEDIUM, SpotSize::LARGE}) {
            int n = 0;
            for (auto& l : levels_) n += l.freeCount(s);
            os << sizeName(s) << "=" << n << " ";
        }
        return os.str();
    }
};

static int hm(int h, int m) { return h * 60 + m; }

int main() {
