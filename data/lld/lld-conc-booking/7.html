The code, run

Java races 8 threads on one seat with every strategy: the naive version gives 8 winners (a barrier forces the bad interleaving every time), per-seat locks, striped locks and CAS give exactly one winner in 200 out of 200 rounds, multi-seat booking with sorted locks never deadlocks (which of Alice and Bob wins A2 varies per run), and the TTL hold rejects a late payment. Python shows the three database guards on SQLite. C++14 shows CAS on a packed (version, owner) word, including the ABA case. Files: lld-code/lld-conc-booking/.

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.locks.ReentrantLock;
import java.util.function.LongSupplier;

// Four ways to stop two users booking the same seat, raced by 8 threads.
public class BookingDemo {

    interface SeatBooker { boolean book(List<String> seats, String user); String owner(String seat); }

    // ---------- 0. naive check-then-act (BROKEN). The barrier forces the bad interleaving.
    static final class NaiveBooker implements SeatBooker {
        private final Map<String, String> owner = new HashMap<>();
        private final CyclicBarrier gap;
        NaiveBooker(CyclicBarrier gap) { this.gap = gap; }
        public boolean book(List<String> seats, String user) {
            String seat = seats.get(0);
            if (owner.get(seat) != null) return false;          // check
            try { gap.await(); } catch (Exception e) { return false; } // everyone has checked "free"
            synchronized (owner) { owner.put(seat, user); }     // act (lock here is too late!)
            return true;
        }
        public String owner(String seat) { synchronized (owner) { return owner.get(seat); } }
    }

    // ---------- 1. pessimistic: one lock per seat, multi-seat in sorted order (no deadlock)
    static final class PessimisticBooker implements SeatBooker {
        private final ConcurrentHashMap<String, ReentrantLock> locks = new ConcurrentHashMap<>();
        private final ConcurrentHashMap<String, String> owner = new ConcurrentHashMap<>();
        public boolean book(List<String> seats, String user) {
            List<String> sorted = new ArrayList<>(seats);
            Collections.sort(sorted);                            // global lock order = seat id
            List<ReentrantLock> held = new ArrayList<>();
            try {
                for (String s : sorted) {
                    ReentrantLock l = locks.computeIfAbsent(s, k -> new ReentrantLock());
                    l.lock();
                    held.add(l);
                }
                for (String s : sorted) if (owner.containsKey(s)) return false;  // all or nothing
                for (String s : sorted) owner.put(s, user);
                return true;
            } finally {
                for (int i = held.size() - 1; i >= 0; i--) held.get(i).unlock();
            }
        }
        public String owner(String seat) { return owner.get(seat); }
    }

    // ---------- 1b. lock striping: a fixed array of locks, seat -> stripe by hash
    static final class StripedBooker implements SeatBooker {
        private final ReentrantLock[] stripes = new ReentrantLock[16];
        private final ConcurrentHashMap<String, String> owner = new ConcurrentHashMap<>();
        StripedBooker() { for (int i = 0; i < stripes.length; i++) stripes[i] = new ReentrantLock(); }
        private int stripe(String seat) { return Math.floorMod(seat.hashCode(), stripes.length); }
        public boolean book(List<String> seats, String user) {
            int[] idx = seats.stream().mapToInt(this::stripe).distinct().sorted().toArray(); // order by stripe
            for (int i : idx) stripes[i].lock();
            try {
                for (String s : seats) if (owner.containsKey(s)) return false;
                for (String s : seats) owner.put(s, user);
                return true;
            } finally {
                for (int i = idx.length - 1; i >= 0; i--) stripes[idx[i]].unlock();
            }
        }
        public String owner(String seat) { return owner.get(seat); }
    }

    // ---------- 2. optimistic: immutable state + version, compare-and-set, no lock held
    record SeatState(long version, String owner) {}
    static final class OptimisticBooker implements SeatBooker {
        final ConcurrentHashMap<String, SeatState> seats = new ConcurrentHashMap<>();
        final AtomicInteger conflicts = new AtomicInteger();
        OptimisticBooker(Collection<String> ids) { for (String id : ids) seats.put(id, new SeatState(0, null)); }
        public boolean book(List<String> ids, String user) {          // single seat for clarity
            String id = ids.get(0);
            while (true) {
                SeatState cur = seats.get(id);                             // 1. read (with version)
                if (cur.owner() != null) return false;                     // 2. business check
                SeatState next = new SeatState(cur.version() + 1, user);   // 3. compute new state
                if (seats.replace(id, cur, next)) return true;             // 4. CAS: only if unchanged
                conflicts.incrementAndGet();                               // someone won; re-read
            }
        }
        public String owner(String seat) { return seats.get(seat).owner(); }
    }

    // ---------- 3. temporary holds with TTL (the BookMyShow / IRCTC flow)
    record Hold(String id, String user, List<String> seats, long expiresAt) {}
    static final class HoldService {
        private final LongSupplier clock;                // injected: tests control time
        private final long ttl;
        private final Map<String, Hold> holdBySeat = new HashMap<>();
        private final Map<String, Hold> holds = new HashMap<>();
        private final Set<String> booked = new HashSet<>();
        private final AtomicLong ids = new AtomicLong();
        HoldService(LongSupplier clock, long ttl) { this.clock = clock; this.ttl = ttl; }

        private boolean free(String seat, long now) {
            if (booked.contains(seat)) return false;
            Hold h = holdBySeat.get(seat);
            return h == null || h.expiresAt() <= now;    // expired hold = free (lazy expiry)
        }
        private void dropIfExpired(String seat, long now) {   // lazy cleanup of the seats we touch
            Hold h = holdBySeat.get(seat);
            if (h != null && h.expiresAt() <= now) {
                holds.remove(h.id());
                for (String s : h.seats()) holdBySeat.remove(s, h);
            }
        }
        /** Eager cleanup for abandoned checkouts: call from a ScheduledExecutorService
         *  (in a database: DELETE FROM holds WHERE expires_at < now). Returns holds removed. */
        synchronized int sweep() {
            long now = clock.getAsLong();
            int n = 0;
            for (Iterator<Hold> it = holds.values().iterator(); it.hasNext(); ) {
                Hold h = it.next();
                if (h.expiresAt() <= now) {
                    it.remove(); n++;
                    for (String s : h.seats()) holdBySeat.remove(s, h);
                }
            }
            return n;
        }
        synchronized int liveHolds() { return holds.size(); }
        synchronized Optional<Hold> hold(List<String> seats, String user) {   // one lock per show
            long now = clock.getAsLong();
            for (String s : seats) dropIfExpired(s, now);
            for (String s : seats) if (!free(s, now)) return Optional.empty();
            Hold h = new Hold("H" + ids.incrementAndGet(), user, List.copyOf(seats), now + ttl);
            for (String s : seats) holdBySeat.put(s, h);
            holds.put(h.id(), h);
            return Optional.of(h);
        }
        synchronized String confirm(String holdId) {     // called after payment succeeds
            long now = clock.getAsLong();
            Hold h = holds.remove(holdId);
            if (h == null) return "REJECTED (no live hold: expired and cleaned up, or unknown; refund payment)";
            for (String s : h.seats()) holdBySeat.remove(s, h);   // drop our own entries first
            if (h.expiresAt() <= now) return "REJECTED (hold expired, refund payment)";
            for (String s : h.seats()) booked.add(s);
            return "BOOKED " + h.seats() + " for " + h.user();
        }
    }

    // ---------- race harness: 8 users try to book the same seat at the same instant
    static int race(SeatBooker b, List<String> seats, int users) throws Exception {
        ExecutorService pool = Executors.newFixedThreadPool(users);
        CountDownLatch start = new CountDownLatch(1);
        List<Future<Boolean>> fs = new ArrayList<>();
        for (int u = 0; u < users; u++) {
            String user = "user" + u;
            fs.add(pool.submit(() -> { start.await(); return b.book(seats, user); }));
        }
        start.countDown();
        int winners = 0;
        for (Future<Boolean> f : fs) if (f.get(5, TimeUnit.SECONDS)) winners++;
        pool.shutdown();
        return winners;
    }

    public static void main(String[] args) throws Exception {
        int users = 8;
        System.out.println("0. naive check-then-act : winners for seat A1 = "
                + race(new NaiveBooker(new CyclicBarrier(users)), List.of("A1"), users) + "  <- DOUBLE BOOKING");

        int rounds = 200, bad = 0;
        for (int r = 0; r < rounds; r++) if (race(new PessimisticBooker(), List.of("A1"), users) != 1) bad++;
        System.out.println("1. per-seat lock        : " + rounds + " rounds, rounds with != 1 winner = " + bad);

        PessimisticBooker multi = new PessimisticBooker();
        ExecutorService p = Executors.newFixedThreadPool(2);
        Future<Boolean> x = p.submit(() -> multi.book(List.of("A3", "A1", "A2"), "alice"));
        Future<Boolean> y = p.submit(() -> multi.book(List.of("A2", "A4"), "bob"));
        boolean ax = x.get(5, TimeUnit.SECONDS), by = y.get(5, TimeUnit.SECONDS);
        p.shutdown();
        System.out.println("   multi-seat sorted    : alice=" + ax + " bob=" + by + " (exactly one gets A2, no deadlock) A2 -> " + multi.owner("A2"));

        bad = 0;
        for (int r = 0; r < rounds; r++) if (race(new StripedBooker(), List.of("A1"), users) != 1) bad++;
        System.out.println("1b. striped locks (16)  : " + rounds + " rounds, rounds with != 1 winner = " + bad);

        bad = 0; int conflicts = 0;
        for (int r = 0; r < rounds; r++) {
            OptimisticBooker ob = new OptimisticBooker(List.of("A1"));
            if (race(ob, List.of("A1"), users) != 1) bad++;
            conflicts += ob.conflicts.get();
        }
        System.out.println("2. optimistic CAS       : " + rounds + " rounds, rounds with != 1 winner = " + bad
                + ", CAS retries seen = " + (conflicts > 0 ? "yes" : "none this run"));

        long[] now = {0};
        HoldService hs = new HoldService(() -> now[0], 300);              // 300 s hold
        Hold a = hs.hold(List.of("B5", "B6"), "alice").orElseThrow();
        System.out.println("3. t=0    alice holds B5,B6 until t=" + a.expiresAt());
        now[0] = 10;
        System.out.println("   t=10   bob tries B6       -> " + (hs.hold(List.of("B6"), "bob").isPresent() ? "held" : "seat not available"));
        now[0] = 301;
        Optional<Hold> b = hs.hold(List.of("B6"), "bob");
        System.out.println("   t=301  alice's hold expired; bob tries B6 -> " + (b.isPresent() ? "held as " + b.get().id() : "no"));
        now[0] = 302;
        System.out.println("   t=302  alice pays late, confirm(" + a.id() + ") -> " + hs.confirm(a.id()));
        System.out.println("   t=302  bob pays, confirm(" + b.get().id() + ")   -> " + hs.confirm(b.get().id()));
        for (int i = 0; i < 5; i++) hs.hold(List.of("C" + i), "u" + i);    // five checkouts, all abandoned
        now[0] = 700;
        System.out.println("   t=700  5 abandoned holds; sweep() removed " + hs.sweep() + ", live holds left = " + hs.liveHolds());
    }
}

"""Three database-level guards against double booking, shown with SQLite.
Two app servers (sessions) read the same free seat, then both try to book it.
(SQLite has no SELECT ... FOR UPDATE; the pessimistic SQL for PostgreSQL/MySQL is on the page.)"""
import sqlite3

db = sqlite3.connect(":memory:", isolation_level=None)   # autocommit; we use explicit statements
db.executescript("""
CREATE TABLE show_seat (
    show_id  INTEGER NOT NULL,
    seat_id  TEXT    NOT NULL,
    status   TEXT    NOT NULL DEFAULT 'FREE',     -- FREE | HELD | BOOKED
    user_id  TEXT,
    version  INTEGER NOT NULL DEFAULT 0,
    PRIMARY KEY (show_id, seat_id)
);
CREATE TABLE booking (
    show_id INTEGER NOT NULL,
    seat_id TEXT    NOT NULL,
    user_id TEXT    NOT NULL,
    UNIQUE (show_id, seat_id)                      -- guard 3: the database refuses a second row
);
INSERT INTO show_seat (show_id, seat_id) VALUES (7, 'A1'), (7, 'A2');
""")


def read(seat):
    return db.execute("SELECT status, version FROM show_seat WHERE show_id=7 AND seat_id=?", (seat,)).fetchone()


# ---- guard 1: conditional update ("only if still FREE"): the check and the write are ONE statement
print("guard 1: conditional UPDATE ... WHERE status='FREE'")
for user in ("alice", "bob"):
    cur = db.execute("UPDATE show_seat SET status='BOOKED', user_id=? "
                     "WHERE show_id=7 AND seat_id='A1' AND status='FREE'", (user,))
    print(f"  {user:5}: rows changed = {cur.rowcount} -> {'booked' if cur.rowcount == 1 else 'seat taken'}")

# ---- guard 2: optimistic locking with a version column
print("guard 2: version column (optimistic)")
alice_view = read("A2")          # both servers read version 0 before either writes
bob_view = read("A2")
print(f"  alice read {alice_view}, bob read {bob_view}")


def book_with_version(user, seen, retries=2):
    status, version = seen
    for attempt in range(retries + 1):
        if status != "FREE":
            return f"{user}: seat no longer free (attempt {attempt + 1}), show next best seats"
        cur = db.execute("UPDATE show_seat SET status='BOOKED', user_id=?, version=version+1 "
                         "WHERE show_id=7 AND seat_id='A2' AND version=?", (user, version))
        if cur.rowcount == 1:
            return f"{user}: booked with version {version} -> {version + 1}"
        status, version = read("A2")  # conflict: someone changed the row; re-read and re-check
    return f"{user}: gave up after {retries} retries"


print("  " + book_with_version("alice", alice_view))
print("  " + book_with_version("bob", bob_view))

# ---- guard 3: unique constraint as the last line of defence
print("guard 3: UNIQUE (show_id, seat_id) on booking")
for user in ("alice", "bob"):
    try:
        db.execute("INSERT INTO booking (show_id, seat_id, user_id) VALUES (7, 'A1', ?)", (user,))
        print(f"  {user:5}: insert ok")
    except sqlite3.IntegrityError as e:
        print(f"  {user:5}: rejected by database ({e})")

print("final:", db.execute("SELECT seat_id, status, user_id, version FROM show_seat ORDER BY seat_id").fetchall())

// Optimistic seat booking with one 64-bit word per seat: [version:32 | owner:32].
// compare_exchange_strong is the C++ spelling of cmpxchg. The interleaving is written out
// by hand (no std::thread in g++ 6.3 MinGW) so every step is visible.
#include <atomic>
#include <cstdint>
#include <iostream>

static uint64_t pack(uint32_t version, uint32_t owner) { return (uint64_t(version) << 32) | owner; }
static uint32_t version_of(uint64_t w) { return uint32_t(w >> 32); }
static uint32_t owner_of(uint64_t w) { return uint32_t(w); }
const uint32_t FREE = 0;

// Try to move the seat from the state we SAW to "owned by user". Fails if anyone changed it.
bool try_book(std::atomic<uint64_t>& seat, uint64_t seen, uint32_t user) {
    if (owner_of(seen) != FREE) return false;
    uint64_t want = pack(version_of(seen) + 1, user);
    return seat.compare_exchange_strong(seen, want);   // on failure, 'seen' gets the current value
}

int main() {
    std::atomic<uint64_t> seat(pack(0, FREE));
    uint64_t alice_saw = seat.load();                    // both read version 0, owner FREE
    uint64_t bob_saw = seat.load();

    bool a = try_book(seat, alice_saw, 101);             // alice (user 101) commits first
    std::cout << "alice CAS v0 -> v1: " << (a ? "won" : "lost") << "\n";
    bool b = try_book(seat, bob_saw, 202);               // bob still holds the stale v0 view
    std::cout << "bob   CAS v0 -> v1: " << (b ? "won" : "lost, seat changed under him") << "\n";

    uint64_t now = seat.load();                          // bob re-reads and re-checks
    std::cout << "bob re-reads: version " << version_of(now) << ", owner " << owner_of(now)
              << " -> " << (try_book(seat, now, 202) ? "booked" : "not free, show other seats") << "\n";

    // ABA guard: if alice cancels (owner back to FREE) the version still moves forward,
    // so a third client holding the very first snapshot (v0, FREE) cannot sneak in.
    uint64_t cur = seat.load();
    seat.compare_exchange_strong(cur, pack(version_of(cur) + 1, FREE));
    std::cout << "alice cancels: version " << version_of(seat.load()) << ", owner FREE\n";
    std::cout << "carol with old v0 snapshot: " << (try_book(seat, pack(0, FREE), 303) ? "won (ABA bug!)" : "lost (version protects us)") << "\n";
    return 0;
}
