Code: editor text styles, per-flow QoS profiles, and the caches you already use

Each tab is a complete program that was compiled and run; the output shown is real. Java pools text styles with a thread-safe factory and then shows the JDK's own flyweights. Python measures the memory difference with tracemalloc. C++14 shares a 64 KB QoS profile table across 100,000 network flows. Files live in lld-code/lld-pat-flyweight/.

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

public class FlyweightDemo {

    // ---- Flyweight: the INTRINSIC state. Immutable, shared by many characters. ----
    record CharStyle(String font, int sizePt, boolean bold, String color) {
        // Extrinsic state (the character and where it sits) is passed in, never stored here.
        void draw(char c, int row, int col) {
            System.out.printf("  draw '%c' at row %d col %d with %s %dpt%s %s%n",
                    c, row, col, font, sizePt, bold ? " bold" : "", color);
        }
    }

    // ---- Flyweight factory: hands out one shared instance per distinct style. ----
    static final class StyleFactory {
        private final Map<CharStyle, CharStyle> pool = new ConcurrentHashMap<>();

        CharStyle get(String font, int sizePt, boolean bold, String color) {
            CharStyle wanted = new CharStyle(font, sizePt, bold, color);   // short-lived lookup key
            // <-- the pattern: one shared immutable object per key
            return pool.computeIfAbsent(wanted, k -> k);                    // atomic: one winner per key
        }
        int pooled() { return pool.size(); }
    }

    // ---- Context / client: holds the EXTRINSIC state (the char, its position) plus a reference. ----
    static final class Document {
        private final StringBuilder text = new StringBuilder();
        private final List<CharStyle> styleOf = new ArrayList<>();   // one reference per char, not one object

        void type(String s, CharStyle style) {
            for (char c : s.toCharArray()) { text.append(c); styleOf.add(style); }
        }
        void render(int maxChars, int lineWidth) {
            for (int i = 0; i < Math.min(maxChars, text.length()); i++)
                styleOf.get(i).draw(text.charAt(i), i / lineWidth, i % lineWidth);   // position computed now
        }
        long distinctStyleObjects() {
            Set<CharStyle> ids = Collections.newSetFromMap(new IdentityHashMap<>());
            ids.addAll(styleOf);
            return ids.size();
        }
        int length() { return text.length(); }
    }

    public static void main(String[] args) {
        StyleFactory styles = new StyleFactory();
        Document doc = new Document();

        String[] words = {"Hello ", "kernel ", "hackers ", "and ", "SDE ", "candidates. "};
        for (int i = 0; i < 2_000; i++) {
            String w = words[i % words.length];
            CharStyle s = switch (i % 3) {
                case 0 -> styles.get("Inter", 12, false, "black");
                case 1 -> styles.get("Inter", 12, true, "black");      // bold words
                default -> styles.get("JetBrains Mono", 11, false, "teal"); // code words
            };
            doc.type(w, s);
        }

        System.out.println("characters in document : " + doc.length());
        System.out.println("style objects in pool  : " + styles.pooled());
        System.out.println("distinct style objects : " + doc.distinctStyleObjects());
        System.out.println("without flyweight      : " + doc.length() + " style objects (one per char)");
        doc.render(3, 80);

        // ---- Flyweights the JDK gives you for free ----
        Integer a = 127, b = 127, c = 128, d = 128;          // autoboxing calls Integer.valueOf
        System.out.println("Integer 127 == 127 ? " + (a == b) + "   (cached, same object)");
        System.out.println("Integer 128 == 128 ? " + (c == d) + "  (outside default cache -128..127)");
        System.out.println("Integer 128 equals ? " + c.equals(d) + "   (always compare values with equals)");

        String s1 = new String("route-42");
        String s2 = "route-42";
        System.out.println("new String == literal ? " + (s1 == s2) + ",  after intern() ? " + (s1.intern() == s2));
        System.out.println("Boolean.valueOf(true) shared ? " + (Boolean.valueOf(true) == Boolean.TRUE));
    }
}

A Java record is a perfect flyweight: fields are final, and equals/hashCode come from the fields, so the record itself can be the pool key. computeIfAbsent on a ConcurrentHashMap makes "check then create" atomic, so two threads never create two copies of one style.

"""Flyweight: share immutable intrinsic state; keep extrinsic state outside. Measured with tracemalloc."""
from dataclasses import dataclass
import sys
import tracemalloc


@dataclass(frozen=True, slots=True)
class CharStyle:                      # Flyweight: intrinsic, immutable
    font: str
    size_pt: int
    bold: bool
    color: str

    def draw(self, ch: str, row: int, col: int) -> None:   # extrinsic state comes in as arguments
        print(f"  draw {ch!r} at row {row} col {col} with {self.font} {self.size_pt}pt"
              f"{' bold' if self.bold else ''} {self.color}")


class StyleFactory:                   # Flyweight factory
    def __init__(self):
        self._pool: dict[tuple, CharStyle] = {}

    def get(self, font: str, size_pt: int, bold: bool, color: str) -> CharStyle:
        key = (font, size_pt, bold, color)
        style = self._pool.get(key)
        if style is None:
            style = self._pool[key] = CharStyle(*key)
        return style

    def __len__(self):
        return len(self._pool)


SPECS = [("Inter", 12, False, "black"), ("Inter", 12, True, "black"), ("JetBrains Mono", 11, False, "teal")]
N = 100_000


def build_naive():                    # one style object per character
    return [CharStyle(*SPECS[i % 3]) for i in range(N)]


def build_flyweight(factory):         # one reference per character, 3 objects in total
    return [factory.get(*SPECS[i % 3]) for i in range(N)]


def measure(fn, *args):
    tracemalloc.start()
    data = fn(*args)
    current, _peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    return data, current


if __name__ == "__main__":
    naive, naive_bytes = measure(build_naive)
    factory = StyleFactory()
    shared, fly_bytes = measure(build_flyweight, factory)

    print(f"characters                   : {N}")
    print(f"naive: distinct style objects : {len({id(s) for s in naive})}")
    print(f"flyweight: pooled styles      : {len(factory)}")
    print(f"naive memory     (tracemalloc): {naive_bytes / 1024:,.0f} KiB")
    print(f"flyweight memory (tracemalloc): {fly_bytes / 1024:,.0f} KiB  (mostly the list of references)")

    text = "Hi!"
    for i, ch in enumerate(text):
        shared[i].draw(ch, row=0, col=i)

    # Flyweights CPython already uses
    x, y = int("256"), int("256")     # int() at runtime, so the compiler cannot fold constants
    p, q = int("257"), int("257")
    print("256 is 256 ?", x is y, "  257 is 257 ?", p is q, "  (small-int cache, an implementation detail)")
    k1 = sys.intern("".join(["user", "_id"]))
    k2 = sys.intern("user_id")
    print("sys.intern gives the same object ?", k1 is k2)

About 9x less memory for the same document, measured, not guessed. The remaining 783 KiB is the list itself: 100,000 pointers. That is the floor: Flyweight shrinks each item to "one reference plus its extrinsic state", it cannot make the items disappear. CPython's small-int cache covers -5 to 256 in 3.11 and 3.12; the newest docs list a wider range, which is exactly why you must never rely on is for numbers.

// Flyweight in C++14: 100,000 network flows sharing a few immutable QoS profiles.
// Build: g++ -std=c++14 -Wall -O2 flyweight.cpp -o flyweight
#include <iostream>
#include <map>
#include <memory>
#include <string>
#include <tuple>
#include <vector>

// Flyweight: intrinsic state. Big (the classifier/shaper table), immutable, shared.
class QosProfile {
public:
    QosProfile(std::string cls, int dscp, std::size_t tableBytes)
        : cls_(std::move(cls)), dscp_(dscp), table_(tableBytes, 'x') {}
    void shape(unsigned short sport, unsigned short dport) const {   // extrinsic state passed in
        std::cout << "  flow " << sport << "->" << dport << " uses " << cls_ << " (dscp " << dscp_
                  << "), table " << table_.size() << " bytes shared\n";
    }
    std::size_t bytes() const { return sizeof(*this) + table_.capacity(); }
private:
    const std::string cls_;
    const int dscp_;
    const std::string table_;                            // stands in for a shaper / classifier table
};

// Flyweight factory: one QosProfile per (class, dscp).
class QosProfileFactory {
public:
    const QosProfile* get(const std::string& cls, int dscp) {
        auto key = std::make_tuple(cls, dscp);
        auto it = pool_.find(key);
        if (it == pool_.end())
            it = pool_.emplace(key, std::make_unique<QosProfile>(cls, dscp, 64 * 1024)).first;
        return it->second.get();                         // pointer to const: callers cannot mutate
    }
    std::size_t size() const { return pool_.size(); }
    std::size_t bytes() const {
        std::size_t total = 0;
        for (const auto& kv : pool_) total += kv.second->bytes();
        return total;
    }
private:
    std::map<std::tuple<std::string, int>, std::unique_ptr<QosProfile>> pool_;
};

// Context: extrinsic state (the ports) + a pointer to the shared flyweight. Small and cheap.
struct Flow {
    unsigned short sport, dport;
    const QosProfile* qos;                               // <-- the pattern: share, do not copy
    void shape() const { qos->shape(sport, dport); }
};

int main() {
    QosProfileFactory factory;
    std::vector<Flow> flows;
    struct { const char* cls; int dscp; } kinds[] = {{"voice", 46}, {"video", 34}, {"best-effort", 0}};

    for (int i = 0; i < 100000; ++i) {
        const auto& k = kinds[i % 3];
        flows.push_back(Flow{static_cast<unsigned short>(1024 + i % 60000),
                             static_cast<unsigned short>(i % 3 == 0 ? 5060 : 443),
                             factory.get(k.cls, k.dscp)});
    }

    std::size_t contexts = flows.size() * sizeof(Flow);
    std::size_t oneProfile = factory.bytes() / factory.size();
    std::cout << "flows: " << flows.size() << ", QoS profiles: " << factory.size() << '\n';
    std::cout << "sizeof(Flow) = " << sizeof(Flow) << " bytes, one QosProfile ~ " << oneProfile << " bytes\n";
    std::cout << "with flyweight   : " << contexts / 1024 << " KiB (contexts) + "
              << factory.bytes() / 1024 << " KiB (shared profiles)\n";
    std::cout << "without flyweight: about " << static_cast<unsigned long long>(flows.size()) * oneProfile / (1024 * 1024)
              << " MiB (computed: a table copy per flow)\n";
    for (int i = 0; i < 3; ++i) flows[i].shape();
    return 0;
}

sizeof(Flow) is 8 bytes on this 32-bit build (two 16-bit ports plus a 4-byte pointer); a 64-bit build would print 16 (the pointer grows and padding follows). It is the same shape as per-flow state in a router or a conntrack-style table that points at a shared, read-only QoS or route profile. The "without" line is arithmetic, not a measurement: we did not try to allocate 6 GB. A small bug from writing this demo is worth remembering: the first version computed forest.size() * oneType in a 32-bit size_t and silently overflowed, printing 2160 MiB. Widen before you multiply sizes.
