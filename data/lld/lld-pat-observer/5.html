Code: inventory (Java), price ticker (Python), notifier chain (C++14)

All three programs were compiled and run; the output is real. Java shows push events, subscription handles, copy-on-write iteration, exception isolation and the java.util.concurrent.Flow API. Python shows the pull model and weak references against leaks. C++14 mimics a Linux notifier chain with priorities and NOTIFY_STOP. Files live in lld-code/lld-pat-observer/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Java interface: a struct of function pointers with no data. A class that implements it must fill every slot.
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • Anonymous class new Foo() { ... }: defines and creates a one-off implementation in place, like a static ops table filled inline.
  • Generics Foo<T>: a type parameter chosen by the user of the class. Think of a macro that generates one typed struct per element type, but checked by the compiler.
  • std::function<R(Args)>: holds any callable: a plain function pointer, a lambda with captured state, or an object with operator().
  • C++ lambda [&](int x) { ... }: an inline function; the brackets list what it captures from the enclosing scope (& = by reference, = = by copy).

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;
import java.util.concurrent.*;

public class ObserverDemo {

    // ---- The event (push model: the subject sends what changed) ----
    record StockEvent(String sku, int oldQty, int newQty) { }

    // ---- Observer interface ----
    interface StockListener {
        void onStockChanged(StockEvent e);
    }

    // ---- Subject: knows only the interface, never the concrete observers ----
    static final class Inventory {
        // Copy-on-write: publish() iterates a snapshot, so listeners may unsubscribe
        // (or new ones subscribe) while a notification is in progress.
        // Only the listener LIST is thread-safe. qty is single-threaded in this demo; for concurrent
        // setQty, guard it with a lock or use ConcurrentHashMap.compute.
        private final List<StockListener> listeners = new CopyOnWriteArrayList<>();
        private final Map<String, Integer> qty = new HashMap<>();

        /** Returns a handle; closing it unsubscribes. No handle kept = easy leak to spot in review. */
        AutoCloseable subscribe(StockListener l) {
            listeners.add(l);
            return () -> listeners.remove(l);
        }

        void setQty(String sku, int newQty) {
            int old = qty.getOrDefault(sku, 0);
            if (old == newQty) return;                         // no change, no noise
            qty.put(sku, newQty);
            publish(new StockEvent(sku, old, newQty));
        }

        // Named publish, not notifyAll: notifyAll() already exists on every Object (wait/notifyAll).
        private void publish(StockEvent e) {
            for (StockListener l : listeners) {
                try {
                    l.onStockChanged(e);
                } catch (RuntimeException ex) {                // one bad observer must not starve the rest
                    System.out.println("  [inventory] listener failed: " + ex.getMessage());
                }
            }
        }
        int listenerCount() { return listeners.size(); }
    }

    // ---- Concrete observers ----
    static final class BackInStockNotifier implements StockListener {
        private final String user, sku;
        private AutoCloseable handle;
        BackInStockNotifier(String user, String sku) { this.user = user; this.sku = sku; }
        void attach(Inventory inv) { handle = inv.subscribe(this); }
        public void onStockChanged(StockEvent e) {
            if (!e.sku().equals(sku) || e.oldQty() != 0 || e.newQty() == 0) return;
            System.out.println("  SMS to " + user + ": " + sku + " is back in stock");
            try { handle.close(); } catch (Exception ignored) { }   // one-shot: unsubscribe itself
        }
    }

    static final class LowStockAlert implements StockListener {
        public void onStockChanged(StockEvent e) {
            if (e.newQty() > 0 && e.newQty() < 3) System.out.println("  ALERT ops: " + e.sku() + " low (" + e.newQty() + ")");
        }
    }

    public static void main(String[] args) throws Exception {
        Inventory inv = new Inventory();
        new BackInStockNotifier("asha", "PS5").attach(inv);
        new BackInStockNotifier("ravi", "PS5").attach(inv);
        inv.subscribe(new LowStockAlert());
        AutoCloseable audit = inv.subscribe(e -> System.out.println("  audit: " + e));      // a lambda observer
        inv.subscribe(e -> { throw new IllegalStateException("analytics down"); });

        System.out.println("PS5 restocked to 2 (listeners: " + inv.listenerCount() + ")");
        inv.setQty("PS5", 2);
        System.out.println("PS5 sold down to 1 (listeners: " + inv.listenerCount() + ")");
        audit.close();                                                                     // explicit unsubscribe
        inv.setQty("PS5", 1);

        // ---- java.util.concurrent.Flow: async observer with back-pressure ----
        System.out.println("Flow API: async delivery, subscriber asks for 1 item at a time");
        CountDownLatch done = new CountDownLatch(1);
        try (SubmissionPublisher<StockEvent> pub = new SubmissionPublisher<>()) {
            pub.subscribe(new Flow.Subscriber<>() {
                private Flow.Subscription sub;
                public void onSubscribe(Flow.Subscription s) { sub = s; s.request(1); }
                public void onNext(StockEvent e) {
                    System.out.println("  async got " + e.sku() + " -> " + e.newQty() + " on " +
                            (Thread.currentThread().getName().contains("ForkJoinPool") ? "pool thread" : "other thread"));
                    sub.request(1);                                  // pull the next one when ready
                }
                public void onError(Throwable t) { done.countDown(); }
                public void onComplete() { System.out.println("  async complete"); done.countDown(); }
            });
            pub.submit(new StockEvent("TV", 0, 5));
            pub.submit(new StockEvent("TV", 5, 4));
        }                                                            // close() -> onComplete
        done.await(5, TimeUnit.SECONDS);
    }
}

Count the listeners: 5 before the restock, 3 after, because the two one-shot notifiers unsubscribed themselves during the walk. The failing lambda is reported on each change but never blocks the others. In the Flow part the subscriber calls request(1) each time: that is back-pressure, the subscriber telling the publisher how much it can take, like a TCP receive window.

"""Observer in Python: a price ticker with weak references, so forgotten observers do not leak.

Pull model here: the subject only says "I changed"; observers read what they need.
"""
import gc
import weakref


class Ticker:                                         # Subject
    def __init__(self, symbol):
        self.symbol, self._price = symbol, 0.0
        self._observers = weakref.WeakSet()           # does not keep observers alive

    def attach(self, obs):
        self._observers.add(obs)

    def detach(self, obs):
        self._observers.discard(obs)

    @property
    def price(self):
        return self._price

    def set_price(self, p):
        self._price = p
        for obs in list(self._observers):             # copy: observers may detach while we loop
            obs.update(self)                          # pull: pass the subject, not the data


class PriceAlert:                                     # Concrete observer
    def __init__(self, name, above):
        self.name, self.above = name, above

    def update(self, ticker):
        if ticker.price > self.above:
            print(f"  {self.name}: {ticker.symbol} at {ticker.price} > {self.above}")


class Chart:
    def __init__(self):
        self.points = []

    def update(self, ticker):
        self.points.append(ticker.price)


if __name__ == "__main__":
    infy = Ticker("INFY")
    chart, alert = Chart(), PriceAlert("asha", 1500)
    infy.attach(chart)
    infy.attach(alert)

    def screen_opened():                              # an observer that nobody detaches
        temp = PriceAlert("temp-screen", 0)
        infy.attach(temp)
        print(f"  observers while screen open: {len(infy._observers)}")

    screen_opened()
    gc.collect()                                      # CPython frees it at once; collect() makes it explicit
    print(f"  observers after screen closed: {len(infy._observers)} (weak ref let it go)")

    for p in (1490, 1510, 1525):
        infy.set_price(p)
    infy.detach(alert)
    infy.set_price(1600)                              # alert no longer hears this
    print(f"  chart saw {chart.points}")
$ python observer.py
  observers while screen open: 3
  observers after screen closed: 2 (weak ref let it go)
  asha: INFY at 1510 > 1500
  asha: INFY at 1525 > 1500
  chart saw [1490, 1510, 1525, 1600]

The temporary screen's observer disappeared on its own because the subject holds it only through a WeakSet. The trade-off: an observer that nobody else references vanishes immediately, so infy.attach(PriceAlert("x", 0)) would silently do nothing. For bound methods use weakref.WeakMethod, since a bound method object is created fresh on every attribute access.

// Observer in C++14, shaped like a Linux notifier chain:
// a priority-ordered list of callbacks; a callback can stop the walk.
// Build: g++ -std=c++14 -Wall observer.cpp -o observer
#include <functional>
#include <iostream>
#include <list>
#include <string>

enum NetEvent { NETDEV_UP, NETDEV_DOWN, NETDEV_CHANGEMTU };
const char* evName(NetEvent e) {
    return e == NETDEV_UP ? "NETDEV_UP" : e == NETDEV_DOWN ? "NETDEV_DOWN" : "NETDEV_CHANGEMTU";
}

enum { NOTIFY_DONE = 0x0000, NOTIFY_OK = 0x0001, NOTIFY_STOP_MASK = 0x8000,
       NOTIFY_STOP = NOTIFY_OK | NOTIFY_STOP_MASK };          // same values as linux/notifier.h

struct NotifierBlock {                                    // like struct notifier_block
    std::string name;
    int priority;                                         // higher runs first
    std::function<int(NetEvent, const std::string&)> call;
};

class NotifierChain {                                     // the Subject
    std::list<NotifierBlock*> chain_;                     // caller owns the blocks, as in the kernel
public:
    void reg(NotifierBlock* nb) {                         // insert sorted by priority
        auto it = chain_.begin();
        while (it != chain_.end() && (*it)->priority >= nb->priority) ++it;
        chain_.insert(it, nb);
    }
    void unreg(NotifierBlock* nb) { chain_.remove(nb); }
    int call(NetEvent ev, const std::string& dev) {      // like call_netdevice_notifiers()
        int ret = NOTIFY_DONE;
        for (auto it = chain_.begin(); it != chain_.end();) {
            NotifierBlock* nb = *it++;                    // advance first: callee may unregister itself
            ret = nb->call(ev, dev);
            if (ret & NOTIFY_STOP_MASK) {
                std::cout << "    (" << nb->name << " stopped the chain)\n";
                break;
            }
        }
        return ret;
    }
};

int main() {
    NotifierChain netdev_chain;

    NotifierBlock routing{"routing", 10, [](NetEvent e, const std::string& d) {
        std::cout << "  routing : " << evName(e) << " " << d
                  << (e == NETDEV_DOWN ? " -> flush routes via " + d : "") << "\n";
        return NOTIFY_OK;
    }};
    NotifierBlock bonding{"bonding", 20, [](NetEvent e, const std::string& d) {
        std::cout << "  bonding : " << evName(e) << " " << d << "\n";
        if (e == NETDEV_CHANGEMTU && d == "bond0") return NOTIFY_STOP;   // veto: slaves must match
        return NOTIFY_OK;
    }};
    NotifierBlock logger{"logger", 0, [](NetEvent e, const std::string& d) {
        std::cout << "  logger  : " << evName(e) << " " << d << "\n";
        return NOTIFY_DONE;                                // "not interested", keep going
    }};

    netdev_chain.reg(&routing);
    netdev_chain.reg(&logger);
    netdev_chain.reg(&bonding);                            // registered last, runs first (priority 20)

    std::cout << "eth0 goes down:\n";
    netdev_chain.call(NETDEV_DOWN, "eth0");
    std::cout << "MTU change on bond0:\n";
    netdev_chain.call(NETDEV_CHANGEMTU, "bond0");

    netdev_chain.unreg(&bonding);                          // module unload MUST unregister first
    std::cout << "eth0 up, bonding unloaded:\n";
    netdev_chain.call(NETDEV_UP, "eth0");
    return 0;
}
$ g++ -std=c++14 -Wall observer.cpp -o observer && ./observer
eth0 goes down:
  bonding : NETDEV_DOWN eth0
  routing : NETDEV_DOWN eth0 -> flush routes via eth0
  logger  : NETDEV_DOWN eth0
MTU change on bond0:
  bonding : NETDEV_CHANGEMTU bond0
    (bonding stopped the chain)
eth0 up, bonding unloaded:
  routing : NETDEV_UP eth0
  logger  : NETDEV_UP eth0

Bonding registered last but ran first, because the chain is ordered by priority. When it returned NOTIFY_STOP for the MTU change, the lower-priority listeners never heard of it. The walk advances the iterator before the call, so a callback may unregister itself, which is the same care the kernel takes when walking its chains.
