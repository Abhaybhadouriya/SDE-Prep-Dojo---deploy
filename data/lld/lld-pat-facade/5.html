Code: an order facade, a backup facade, a NIC bring-up facade

Each tab is a complete program that was compiled and run. Java orchestrates four services with rollback on failure. Python wraps five standard library modules behind backup() and verify(), then shows that shutil is itself a facade. C++14 brings a network device up and down, the way a driver's open and stop callbacks hide PHY, DMA and IRQ setup. Files live in lld-code/lld-pat-facade/.

import java.util.*;

public class FacadeDemo {

    // ---------------- Subsystem classes: each does one job, none knows the facade ----------------
    static final class Inventory {
        private final Map<String, Integer> stock = new HashMap<>(Map.of("book", 3, "pen", 0));
        boolean reserve(String sku, int qty) {
            int have = stock.getOrDefault(sku, 0);
            if (have < qty) return false;
            stock.put(sku, have - qty);
            System.out.println("  inventory: reserved " + qty + " x " + sku);
            return true;
        }
        void release(String sku, int qty) {
            stock.merge(sku, qty, Integer::sum);
            System.out.println("  inventory: released " + qty + " x " + sku);
        }
    }

    static final class PaymentService {
        String charge(String user, long paise) {
            if (paise > 100_000) throw new IllegalStateException("card limit exceeded");
            System.out.println("  payment: charged " + user + " Rs " + paise / 100);
            return "PAY-" + Math.abs(user.hashCode() % 1000);
        }
        void refund(String paymentId) { System.out.println("  payment: refunded " + paymentId); }
    }

    static final class Shipping {
        String book(String address, String sku) {
            System.out.println("  shipping: pickup booked for " + sku + " to " + address);
            return "AWB-" + sku.length() + address.length();
        }
    }

    static final class Notifier {
        void send(String user, String msg) { System.out.println("  notify " + user + ": " + msg); }
    }

    // ---------------- Facade: one simple call for the common use case ----------------
    record OrderResult(boolean ok, String message) { }

    static final class OrderFacade {
        private final Inventory inventory;
        private final PaymentService payments;
        private final Shipping shipping;
        private final Notifier notifier;

        OrderFacade(Inventory i, PaymentService p, Shipping s, Notifier n) {   // injected, not new'd
            inventory = i; payments = p; shipping = s; notifier = n;
        }

        OrderResult placeOrder(String user, String sku, int qty, long unitPaise, String address) {
            // <-- the pattern: one call orchestrates the subsystems in order
            if (!inventory.reserve(sku, qty)) return new OrderResult(false, "out of stock: " + sku);
            String paymentId;
            try {
                paymentId = payments.charge(user, unitPaise * qty);
            } catch (IllegalStateException e) {
                inventory.release(sku, qty);                       // undo step 1
                return new OrderResult(false, "payment failed: " + e.getMessage());
            }
            String awb = shipping.book(address, sku);
            notifier.send(user, "order confirmed, tracking " + awb);
            return new OrderResult(true, paymentId + " / " + awb);
        }
    }

    public static void main(String[] args) {
        Inventory inv = new Inventory();
        PaymentService pay = new PaymentService();
        OrderFacade shop = new OrderFacade(inv, pay, new Shipping(), new Notifier());

        System.out.println("1) " + shop.placeOrder("asha", "book", 1, 49_900, "Pune"));
        System.out.println("2) " + shop.placeOrder("ravi", "pen", 2, 2_000, "Delhi"));
        System.out.println("3) " + shop.placeOrder("meera", "book", 2, 60_000, "Goa"));

        // The subsystem is still public: an admin tool can use it directly.
        pay.refund("PAY-ADMIN");
    }
}

"""Facade: one backup() call over pathlib, zipfile, hashlib, tempfile and json."""
import hashlib
import json
import tempfile
import zipfile
from pathlib import Path


class BackupFacade:
    """The simple entry point. Callers never touch zip, hashing or manifests."""

    def __init__(self, dest_dir: Path):
        self.dest_dir = dest_dir
        dest_dir.mkdir(parents=True, exist_ok=True)

    def backup(self, src: Path, name: str) -> Path:
        files = sorted(p for p in src.rglob("*") if p.is_file())
        archive = self.dest_dir / f"{name}.zip"
        with zipfile.ZipFile(archive, "w", zipfile.ZIP_DEFLATED) as z:
            for f in files:
                z.write(f, f.relative_to(src).as_posix())
        manifest = {"archive": archive.name,
                    "files": len(files),
                    "sha256": hashlib.sha256(archive.read_bytes()).hexdigest()[:16]}
        (self.dest_dir / f"{name}.json").write_text(json.dumps(manifest))
        return archive

    def verify(self, name: str) -> bool:
        manifest = json.loads((self.dest_dir / f"{name}.json").read_text())
        digest = hashlib.sha256((self.dest_dir / manifest["archive"]).read_bytes()).hexdigest()[:16]
        with zipfile.ZipFile(self.dest_dir / manifest["archive"]) as z:
            return digest == manifest["sha256"] and z.testzip() is None


if __name__ == "__main__":
    with tempfile.TemporaryDirectory() as tmp:
        src = Path(tmp, "project")
        (src / "net").mkdir(parents=True)
        (src / "net" / "tcp.c").write_text("int tcp_v4_rcv(void);\n" * 50)
        (src / "README").write_text("hello\n")

        backups = BackupFacade(Path(tmp, "backups"))      # client code: two calls total
        archive = backups.backup(src, "nightly")
        print("archive:", archive.name, "| files listed:",
              json.loads(Path(tmp, "backups", "nightly.json").read_text())["files"])
        print("verified:", backups.verify("nightly"))

        # The standard library's own facades: shutil.make_archive does a similar job in one call.
        import shutil
        made = shutil.make_archive(str(Path(tmp, "quick")), "zip", src)
        print("shutil facade made:", Path(made).name)

The client makes two calls. Walking a tree, zipping, hashing and writing a manifest are hidden. Python's standard library is full of facades like this: shutil over os, subprocess.run() over Popen and pipes, urllib.request.urlopen() over HTTP handlers.

// Facade in C++14: bringing a NIC up touches PHY, DMA rings, IRQs and the MAC filter.
// The driver core exposes one call, like ndo_open() behind "ip link set eth0 up".
// Build: g++ -std=c++14 -Wall facade.cpp -o facade
#include <iostream>
#include <stdexcept>
#include <string>

// ---- Subsystem classes (would be separate files in a real driver) ----
class Phy {
public:
    bool linkUp(int mbps) { std::cout << "  phy: autoneg -> " << mbps << " Mb/s full duplex\n"; return true; }
    void powerDown() { std::cout << "  phy: power down\n"; }
};

class DmaRings {
public:
    void allocate(int rx, int tx) {
        if (rx > 4096) throw std::runtime_error("ring too large");
        std::cout << "  dma: rx ring " << rx << ", tx ring " << tx << " descriptors\n";
    }
    void release() { std::cout << "  dma: rings freed\n"; }
};

class IrqController {
public:
    void request(int vector) { std::cout << "  irq: vector " << vector << " requested\n"; }
    void free(int vector) { std::cout << "  irq: vector " << vector << " freed\n"; }
};

class MacFilter {
public:
    void program(const std::string& mac, bool promisc) {
        std::cout << "  mac: filter " << mac << (promisc ? " (promiscuous)" : "") << "\n";
    }
};

// ---- Facade: the right order, the right defaults, and cleanup on failure ----
class NetDevice {
public:
    explicit NetDevice(std::string mac) : mac_(std::move(mac)) {}

    bool open(int rxRing = 512) {
        if (!phy_.linkUp(1000)) return false;
        try {
            dma_.allocate(rxRing, 512);
        } catch (const std::exception& e) {          // unwind in reverse order
            std::cout << "  open failed: " << e.what() << "\n";
            phy_.powerDown();
            return false;
        }
        irq_.request(kVector);
        mac_filter_.program(mac_, false);
        up_ = true;
        return true;
    }

    void stop() {                                     // teardown is the mirror image
        if (!up_) return;
        irq_.free(kVector);
        dma_.release();
        phy_.powerDown();
        up_ = false;
    }

    bool isUp() const { return up_; }

private:
    static constexpr int kVector = 42;
    std::string mac_;
    bool up_ = false;
    Phy phy_;
    DmaRings dma_;
    IrqController irq_;
    MacFilter mac_filter_;
};

int main() {
    NetDevice eth0("02:00:5e:10:00:01");
    std::cout << "ip link set eth0 up\n";
    std::cout << "up = " << std::boolalpha << eth0.open() << "\n";
    std::cout << "ip link set eth0 down\n";
    eth0.stop();

    NetDevice eth1("02:00:5e:10:00:02");
    std::cout << "ethtool -G eth1 rx 8192, then up\n";
    std::cout << "up = " << eth1.open(8192) << "\n";
    return 0;
}

This is the shape of a real driver's ndo_open / ndo_stop: the networking core calls one function, and the driver sets up hardware in order and unwinds in reverse on failure (the C version is the familiar goto err_free_rings; ladder). Nothing outside needs to know there is a PHY.
