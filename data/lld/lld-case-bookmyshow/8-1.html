"""Movie ticket booking (BookMyShow) - machine-coding reference, Python 3.11. Same design as the Java version."""
from __future__ import annotations

import itertools
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum


class SeatType(Enum):
    REGULAR = "REGULAR"
    PREMIUM = "PREMIUM"
    RECLINER = "RECLINER"


@dataclass(frozen=True)
class Seat:
    id: str
    type: SeatType


@dataclass(frozen=True)
class Screen:
    id: str
    theatre: str
    city: str
    seats: tuple[Seat, ...]


@dataclass(frozen=True)
class Show:
    id: str
    title: str
    screen: Screen
    start: datetime


# ---- seat state for one show: exactly one of these ----
@dataclass(frozen=True)
class Free:
    pass


@dataclass(frozen=True)
class Held:
    hold_id: str
    user_id: str
    expires_at: datetime


@dataclass(frozen=True)
class Booked:
    booking_id: str


@dataclass(frozen=True)
class SeatHold:
    id: str
    show_id: str
    user_id: str
    seat_ids: tuple[str, ...]
    expires_at: datetime


class BookingStatus(Enum):
    PENDING = "PENDING"
    CONFIRMED = "CONFIRMED"
    CANCELLED = "CANCELLED"
    EXPIRED = "EXPIRED"
    FAILED = "FAILED"


ALLOWED = {
    BookingStatus.PENDING: {BookingStatus.CONFIRMED, BookingStatus.EXPIRED, BookingStatus.FAILED},
    BookingStatus.CONFIRMED: {BookingStatus.CANCELLED},
}


@dataclass
class Booking:
    id: str
    user_id: str
    show: Show
    hold: SeatHold
    amount: int
    status: BookingStatus = BookingStatus.PENDING
    lock: threading.Lock = field(default_factory=threading.Lock, repr=False, compare=False)

    def move_to(self, nxt: BookingStatus) -> None:
        if nxt not in ALLOWED.get(self.status, set()):
            raise ValueError(f"booking {self.id}: {self.status.name} -> {nxt.name} not allowed")
        self.status = nxt

    def __str__(self) -> str:
        return f"{self.id} {self.user_id} {list(self.hold.seat_ids)} Rs {self.amount} {self.status.name}"


class SeatUnavailableError(Exception):
    pass


class HoldExpiredError(Exception):
    pass


class ManualClock:
    def __init__(self, start: datetime) -> None:
        self._now = start

    def now(self) -> datetime:
        return self._now

    def advance(self, seconds: int) -> None:
        self._now += timedelta(seconds=seconds)


# ---------------- inventory: the Strategy the service depends on ----------------
class SeatInventory(ABC):
    def __init__(self, clock: ManualClock, ttl: timedelta) -> None:
        self.clock, self.ttl = clock, ttl
        self._hold_seq = itertools.count(1)
        self._seq_lock = threading.Lock()

    def _next_hold_id(self) -> str:
        with self._seq_lock:
            return f"H{next(self._hold_seq)}"

    @staticmethod
    def _free_now(state, now: datetime) -> bool:
        return isinstance(state, Free) or (isinstance(state, Held) and now >= state.expires_at)

    @staticmethod
    def _sorted(seat_ids: list[str]) -> tuple[str, ...]:
        if not seat_ids:
            raise ValueError("choose at least one seat")
        if len(set(seat_ids)) != len(seat_ids):
            raise ValueError("duplicate seat in request")
        return tuple(sorted(seat_ids))

    @abstractmethod
    def register(self, show: Show) -> None: ...
    @abstractmethod
    def hold(self, show: Show, seat_ids: list[str], user_id: str) -> SeatHold: ...
    @abstractmethod
    def confirm(self, hold: SeatHold, booking_id: str) -> None: ...
    @abstractmethod
    def release(self, hold: SeatHold) -> None: ...
    @abstractmethod
    def cancel(self, show: Show, seat_ids: tuple[str, ...], booking_id: str) -> None: ...
    @abstractmethod
    def available(self, show: Show) -> list[str]: ...

    def occupancy(self, show: Show) -> float:
        return 1 - len(self.available(show)) / len(show.screen.seats)


class PessimisticSeatInventory(SeatInventory):
    """One lock per show: check all seats and write all seats inside the same critical section."""
    name = "pessimistic (lock per show)"

    def __init__(self, clock, ttl) -> None:
        super().__init__(clock, ttl)
        self._shows: dict[str, tuple[threading.Lock, dict[str, object]]] = {}

    def register(self, show):
        self._shows[show.id] = (threading.Lock(), {s.id: Free() for s in show.screen.seats})

    def _of(self, show_id):
        if show_id not in self._shows:
            raise ValueError(f"unknown show {show_id}")
        return self._shows[show_id]

    def hold(self, show, seat_ids, user_id):
        ids = self._sorted(seat_ids)
        lock, state = self._of(show.id)
        with lock:
            now = self.clock.now()
            for sid in ids:
                if sid not in state:
                    raise ValueError(f"no seat {sid}")
                if not self._free_now(state[sid], now):
                    raise SeatUnavailableError(f"seat {sid} is not available")
            hold = SeatHold(self._next_hold_id(), show.id, user_id, ids, now + self.ttl)
            for sid in ids:
                state[sid] = Held(hold.id, user_id, hold.expires_at)
            return hold

    def confirm(self, hold, booking_id):
        lock, state = self._of(hold.show_id)
        with lock:
            now = self.clock.now()
            for sid in hold.seat_ids:
                st = state[sid]
                if not (isinstance(st, Held) and st.hold_id == hold.id and now < st.expires_at):
                    raise HoldExpiredError(f"hold {hold.id} expired or lost")
            for sid in hold.seat_ids:
                state[sid] = Booked(booking_id)

    def release(self, hold):
        lock, state = self._of(hold.show_id)
        with lock:
            for sid in hold.seat_ids:
                st = state[sid]
                if isinstance(st, Held) and st.hold_id == hold.id:
                    state[sid] = Free()

    def cancel(self, show, seat_ids, booking_id):
        lock, state = self._of(show.id)
        with lock:
            for sid in seat_ids:
                if state[sid] == Booked(booking_id):
                    state[sid] = Free()

    def available(self, show):
        lock, state = self._of(show.id)
        with lock:
            now = self.clock.now()
            return sorted(sid for sid, st in state.items() if self._free_now(st, now))


class _Cell:
    """One seat with a version number. compare_and_set mimics a hardware CAS (Python has none)."""

    def __init__(self) -> None:
        self.state, self.version = Free(), 0
        self._guard = threading.Lock()      # held for a few instructions only, never across seats

    def read(self):
        with self._guard:
            return self.state, self.version

    def compare_and_set(self, seen_version: int, new_state) -> bool:
        with self._guard:
            if self.version != seen_version:
                return False
            self.state, self.version = new_state, self.version + 1
            return True


class OptimisticSeatInventory(SeatInventory):
    """No show lock: read a seat's version, then write only if the version is unchanged; roll back on failure."""
    name = "optimistic (version per seat)"

    def __init__(self, clock, ttl) -> None:
        super().__init__(clock, ttl)
        self._shows: dict[str, dict[str, _Cell]] = {}

    def register(self, show):
        self._shows[show.id] = {s.id: _Cell() for s in show.screen.seats}

    def _cell(self, show_id, sid):
        if show_id not in self._shows:
            raise ValueError(f"unknown show {show_id}")
        if sid not in self._shows[show_id]:
            raise ValueError(f"no seat {sid}")
        return self._shows[show_id][sid]

    def _rollback(self, done, expected_type, key, value):
        for cell in done:
            st, ver = cell.read()
            if isinstance(st, expected_type) and getattr(st, key) == value:
                cell.compare_and_set(ver, Free())

    def hold(self, show, seat_ids, user_id):
        ids = self._sorted(seat_ids)
        now = self.clock.now()
        hold = SeatHold(self._next_hold_id(), show.id, user_id, ids, now + self.ttl)
        mine = Held(hold.id, user_id, hold.expires_at)
        done = []
        for sid in ids:
            cell = self._cell(show.id, sid)
            st, ver = cell.read()
            if not self._free_now(st, now) or not cell.compare_and_set(ver, mine):
                self._rollback(done, Held, "hold_id", hold.id)
                raise SeatUnavailableError(f"seat {sid} is not available")
            done.append(cell)
        return hold

    def confirm(self, hold, booking_id):
        if self.clock.now() >= hold.expires_at:
            raise HoldExpiredError(f"hold {hold.id} expired")
        done = []
        for sid in hold.seat_ids:
            cell = self._cell(hold.show_id, sid)
            st, ver = cell.read()
            if not (isinstance(st, Held) and st.hold_id == hold.id) or not cell.compare_and_set(ver, Booked(booking_id)):
                self._rollback(done, Booked, "booking_id", booking_id)
                raise HoldExpiredError(f"hold {hold.id} expired or lost")
            done.append(cell)

    def release(self, hold):
        for sid in hold.seat_ids:
            cell = self._cell(hold.show_id, sid)
            st, ver = cell.read()
            if isinstance(st, Held) and st.hold_id == hold.id:
                cell.compare_and_set(ver, Free())

    def cancel(self, show, seat_ids, booking_id):
        for sid in seat_ids:
            cell = self._cell(show.id, sid)
            st, ver = cell.read()
            if st == Booked(booking_id):
                cell.compare_and_set(ver, Free())

    def available(self, show):
        now = self.clock.now()
        return sorted(sid for sid, c in self._shows[show.id].items() if self._free_now(c.read()[0], now))


# ---------------- pricing and payment strategies ----------------
class PricingStrategy(ABC):
    @abstractmethod
    def price(self, show: Show, seat: Seat, occupancy: float) -> int: ...


class SeatTypePricing(PricingStrategy):
    def __init__(self, base: dict[SeatType, int]) -> None:
        self.base = dict(base)

    def price(self, show, seat, occupancy):
        return self.base[seat.type]


class DemandPricing(PricingStrategy):
    """Decorator: surcharge once the show is at least `threshold` full."""

    def __init__(self, inner: PricingStrategy, threshold: float, percent: int) -> None:
        self.inner, self.threshold, self.percent = inner, threshold, percent

    def price(self, show, seat, occupancy):
        p = self.inner.price(show, seat, occupancy)
        return p + p * self.percent // 100 if occupancy >= self.threshold else p


class PaymentStrategy(ABC):
    @abstractmethod
    def pay(self, booking_id: str, amount: int) -> bool: ...


class UpiPayment(PaymentStrategy):
    def pay(self, booking_id, amount):
        return amount > 0


class CardPayment(PaymentStrategy):
    def __init__(self, declined: bool = False) -> None:
        self.declined = declined

    def pay(self, booking_id, amount):
        return not self.declined


# ---------------- the service ----------------
class BookingService:
    def __init__(self, inventory: SeatInventory, pricing: PricingStrategy, clock: ManualClock) -> None:
        self.inventory, self.pricing, self.clock = inventory, pricing, clock
        self._shows: dict[str, Show] = {}
        self._bookings: dict[str, Booking] = {}
        self._seq = itertools.count(1)
        self._lock = threading.Lock()
        self.listeners: list = []          # callables(booking): the Observer hook

    def add_show(self, show: Show) -> None:
        self._shows[show.id] = show
        self.inventory.register(show)

    def _show(self, show_id: str) -> Show:
        if show_id not in self._shows:
            raise ValueError(f"unknown show {show_id}")
        return self._shows[show_id]

    def _publish(self, b: Booking) -> None:
        for listener in self.listeners:
            listener(b)

    def start(self, user_id: str, show_id: str, seat_ids: list[str]) -> Booking:
        show = self._show(show_id)
        occupancy = self.inventory.occupancy(show)
        hold = self.inventory.hold(show, seat_ids, user_id)
        seats = {s.id: s for s in show.screen.seats}
        amount = sum(self.pricing.price(show, seats[sid], occupancy) for sid in hold.seat_ids)
        with self._lock:
            b = Booking(f"B{next(self._seq)}", user_id, show, hold, amount)
            self._bookings[b.id] = b
        return b

    def get(self, booking_id: str) -> Booking:
        if booking_id not in self._bookings:
            raise ValueError(f"unknown booking {booking_id}")
        return self._bookings[booking_id]

    def pay(self, booking_id: str, payment: PaymentStrategy) -> Booking:
        b = self.get(booking_id)
        with b.lock:
            if b.status is not BookingStatus.PENDING:
                raise ValueError(f"booking {booking_id} is {b.status.name}")
            if self.clock.now() >= b.hold.expires_at:          # check before charging
                self.inventory.release(b.hold)
                b.move_to(BookingStatus.EXPIRED)
                self._publish(b)
                raise HoldExpiredError(f"hold expired for {booking_id}")
            if not payment.pay(b.id, b.amount):
                self.inventory.release(b.hold)
                b.move_to(BookingStatus.FAILED)
                self._publish(b)
                return b
            try:
                self.inventory.confirm(b.hold, b.id)
            except HoldExpiredError:
                b.move_to(BookingStatus.EXPIRED)               # refund would be triggered here
                self._publish(b)
                raise
            b.move_to(BookingStatus.CONFIRMED)
            self._publish(b)
            return b

    def cancel(self, booking_id: str) -> Booking:
        b = self.get(booking_id)
        with b.lock:
            b.move_to(BookingStatus.CANCELLED)
            self.inventory.cancel(b.show, b.hold.seat_ids, b.id)
            self._publish(b)
            return b

    def available_seats(self, show_id: str) -> list[str]:
        return self.inventory.available(self._show(show_id))

"""Demo driver: python main.py   (tests: python -m unittest test_booking)"""
from datetime import datetime, timedelta

from booking import (BookingService, BookingStatus, CardPayment, DemandPricing, HoldExpiredError, ManualClock,
                     OptimisticSeatInventory, PessimisticSeatInventory, Screen, Seat, SeatType, SeatTypePricing,
                     SeatUnavailableError, Show, UpiPayment)

BASE = {SeatType.REGULAR: 200, SeatType.PREMIUM: 350, SeatType.RECLINER: 600}


def screen() -> Screen:
    seats = [Seat(f"A{i}", SeatType.REGULAR) for i in range(1, 5)] + [Seat(f"B{i}", SeatType.PREMIUM) for i in range(1, 5)]
    return Screen("S1", "Dojo Cinemas", "Bengaluru", tuple(seats))


def demo(inventory_cls) -> None:
    clock = ManualClock(datetime(2026, 11, 23, 12, 0))
    inv = inventory_cls(clock, timedelta(minutes=5))
    svc = BookingService(inv, SeatTypePricing(BASE), clock)
    svc.add_show(Show("SH1", "Packet Storm", screen(), datetime(2026, 11, 23, 18, 30)))
    print(f"######## {inv.name} ########")

    alice = svc.start("alice", "SH1", ["A1", "A2"])
    print(f"  {alice}")
    try:
        svc.start("bob", "SH1", ["A2", "A3"])
        raise AssertionError("A2 should be held")
    except SeatUnavailableError as e:
        print(f"  rejected as expected: {e}")
    bob = svc.start("bob", "SH1", ["A3", "A4"])
    svc.pay(alice.id, UpiPayment())
    print(f"  {alice} | free now: {svc.available_seats('SH1')}")
    assert alice.status is BookingStatus.CONFIRMED

    clock.advance(6 * 60)                       # TTL is 5 minutes
    try:
        svc.pay(bob.id, CardPayment())
        raise AssertionError("bob's hold should have expired")
    except HoldExpiredError as e:
        print(f"  rejected as expected: {e}")
    assert bob.status is BookingStatus.EXPIRED

    dave = svc.start("dave", "SH1", ["B1", "B2"])
    svc.pay(dave.id, CardPayment(declined=True))
    print(f"  {dave}")
    assert {"B1", "B2"} <= set(svc.available_seats("SH1"))

    svc.cancel(alice.id)
    svc.pricing = DemandPricing(SeatTypePricing(BASE), 0.5, 20)
    erin = svc.start("erin", "SH1", ["A1", "A2", "A3", "A4"])
    svc.pay(erin.id, UpiPayment())
    frank = svc.start("frank", "SH1", ["B3"])
    print(f"  {erin}\n  {frank}")
    assert (erin.amount, frank.amount) == (800, 420)
    print("  demo OK")


if __name__ == "__main__":
    demo(PessimisticSeatInventory)
    demo(OptimisticSeatInventory)

class BookingTest(unittest.TestCase):
    def each(self):
        return (PessimisticSeatInventory, OptimisticSeatInventory)

    def test_failed_multi_seat_hold_is_all_or_nothing(self):
        for cls in self.each():
            svc, _ = service(cls)
            svc.start("a", "SH1", ["A2"])
            with self.assertRaises(SeatUnavailableError):
                svc.start("b", "SH1", ["A1", "A2"])
            self.assertIn("A1", svc.available_seats("SH1"), cls.name)

    def test_expired_hold_is_free_again(self):
        for cls in self.each():
            svc, clock = service(cls)
            svc.start("a", "SH1", ["A1"])
            clock.advance(301)
            self.assertEqual(svc.start("b", "SH1", ["A1"]).hold.seat_ids, ("A1",))

    def test_cancel_only_after_confirm(self):
        for cls in self.each():
            svc, _ = service(cls)
            b = svc.start("a", "SH1", ["A1"])
            with self.assertRaises(ValueError):
                svc.cancel(b.id)
            svc.pay(b.id, UpiPayment())
            self.assertIs(svc.cancel(b.id).status, BookingStatus.CANCELLED)
            self.assertIn("A1", svc.available_seats("SH1"))

        for cls in self.each():
            svc, _ = service(cls)

            def grab(n):
                pair = [["B2", "B3"], ["A1", "B2"], ["A1", "A2"], ["A2", "B3"]][n % 4]
                try:
                    b = svc.start(f"u{n}", "SH1", pair)

                    return None

            with ThreadPoolExecutor(max_workers=8) as pool:
                done = [b for b in pool.map(grab, range(300)) if b and b.status is BookingStatus.CONFIRMED]
            sold = Counter(s for b in done for s in b.hold.seat_ids)
            self.assertTrue(done, cls.name)
            self.assertTrue(all(c == 1 for c in sold.values()), f"{cls.name}: {sold}")

main.cpp - the hold/confirm/release core with a tagged-union seat state and a pricing vtable.

// Movie ticket booking in C++14: seat holds with a TTL, all-or-nothing, single-threaded demo.
// Build: g++ -std=c++14 -Wall -O2 main.cpp -o booking && ./booking
// Multi-threaded version: one std::mutex per Show around hold/confirm/release (marked LOCK),
// exactly like the Java PessimisticSeatInventory.
#include <algorithm>
#include <cassert>
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

enum class SeatType { REGULAR, PREMIUM };
enum class Kind { FREE, HELD, BOOKED };

struct SeatState {                 // a tagged union, the C way: kind says which fields matter
    Kind kind = Kind::FREE;
    std::string holdId, userId, bookingId;
    long expiresAt = 0;            // seconds
};

struct Hold { std::string id, showId, userId; std::vector<std::string> seats; long expiresAt; };

struct SeatUnavailable : std::runtime_error { using std::runtime_error::runtime_error; };
struct HoldExpired : std::runtime_error { using std::runtime_error::runtime_error; };

struct PricingStrategy {           // the vtable: one function pointer, many implementations
    virtual ~PricingStrategy() {}
    virtual long price(SeatType t, double occupancy) const = 0;
};
struct SeatTypePricing : PricingStrategy {
    long price(SeatType t, double) const override { return t == SeatType::REGULAR ? 200 : 350; }
};
struct DemandPricing : PricingStrategy {   // decorator: wraps another strategy
    std::unique_ptr<PricingStrategy> inner; double threshold; int pct;
    DemandPricing(std::unique_ptr<PricingStrategy> in, double th, int p) : inner(std::move(in)), threshold(th), pct(p) {}
    long price(SeatType t, double occ) const override {
        long p = inner->price(t, occ);
        return occ >= threshold ? p + p * pct / 100 : p;
    }
};

class Show {
public:
    std::string id;
    std::map<std::string, SeatType> types;
    std::map<std::string, SeatState> state;          // LOCK: guarded by the show's mutex
    Show(std::string i, std::map<std::string, SeatType> t) : id(std::move(i)), types(std::move(t)) {
        for (auto& kv : types) state[kv.first] = SeatState{};
    }
    static bool freeNow(const SeatState& s, long now) {
        return s.kind == Kind::FREE || (s.kind == Kind::HELD && now >= s.expiresAt);
    }
    int freeCount(long now) const {
        int n = 0;
        for (auto& kv : state) n += freeNow(kv.second, now) ? 1 : 0;
        return n;
    }
};

class Inventory {
    long ttl_; long seq_ = 0;
public:
    explicit Inventory(long ttlSeconds) : ttl_(ttlSeconds) {}

    Hold hold(Show& show, std::vector<std::string> seats, const std::string& user, long now) {   // LOCK
        std::sort(seats.begin(), seats.end());
        for (auto& id : seats) {                                       // 1. check every seat
            auto it = show.state.find(id);
            if (it == show.state.end()) throw std::invalid_argument("no seat " + id);
            if (!Show::freeNow(it->second, now)) throw SeatUnavailable("seat " + id + " is not available");
        }
        Hold h{"H" + std::to_string(++seq_), show.id, user, seats, now + ttl_};
        for (auto& id : seats) {                                       // 2. then write every seat
            SeatState s; s.kind = Kind::HELD; s.holdId = h.id; s.userId = user; s.expiresAt = h.expiresAt;
            show.state[id] = s;
        }
        return h;
    }

    void confirm(Show& show, const Hold& h, const std::string& bookingId, long now) {         // LOCK
        for (auto& id : h.seats) {
            const SeatState& s = show.state[id];
            if (s.kind != Kind::HELD || s.holdId != h.id || now >= s.expiresAt)
                throw HoldExpired("hold " + h.id + " expired or lost");
        }
        for (auto& id : h.seats) { SeatState s; s.kind = Kind::BOOKED; s.bookingId = bookingId; show.state[id] = s; }
    }

    void release(Show& show, const Hold& h) {                                                 // LOCK
        for (auto& id : h.seats)
            if (show.state[id].kind == Kind::HELD && show.state[id].holdId == h.id) show.state[id] = SeatState{};
    }
};

int main() {
    std::map<std::string, SeatType> seats;
    for (int i = 1; i <= 4; i++) { seats["A" + std::to_string(i)] = SeatType::REGULAR; seats["B" + std::to_string(i)] = SeatType::PREMIUM; }
    Show show("SH1", seats);
    Inventory inv(300);                                   // 5-minute holds
    std::unique_ptr<PricingStrategy> pricing(new SeatTypePricing());
    long now = 0;

    Hold alice = inv.hold(show, {"A2", "A1"}, "alice", now);
    long amount = 0;
    for (auto& id : alice.seats) amount += pricing->price(show.types[id], 0.0);
    std::cout << "alice holds A1,A2 for Rs " << amount << " until t=" << alice.expiresAt << "s\n";
    assert(amount == 400);

    try { inv.hold(show, {"A2", "A3"}, "bob", now); assert(false); }
    catch (const SeatUnavailable& e) { std::cout << "rejected as expected: " << e.what() << "\n"; }
    assert(show.state["A3"].kind == Kind::FREE);          // all or nothing

    Hold bob = inv.hold(show, {"A3", "A4"}, "bob", now);
    inv.confirm(show, alice, "B1", now + 60);             // alice pays after 1 minute
    std::cout << "alice confirmed; free seats: " << show.freeCount(now + 60) << "\n";

    now += 360;                                           // 6 minutes later
    try { inv.confirm(show, bob, "B2", now); assert(false); }
    catch (const HoldExpired& e) { std::cout << "rejected as expected: " << e.what() << "\n"; }
    Hold carol = inv.hold(show, {"A3"}, "carol", now);    // expired hold counts as free
    std::cout << "carol holds A3 with " << carol.id << "\n";

    pricing.reset(new DemandPricing(std::unique_ptr<PricingStrategy>(new SeatTypePricing()), 0.25, 20));
    double occ = 1.0 - show.freeCount(now) / 8.0;         // A1, A2 booked + A3 held = 3/8
    std::cout << "occupancy " << occ << ", premium seat now costs Rs " << pricing->price(SeatType::PREMIUM, occ) << "\n";
    assert(pricing->price(SeatType::PREMIUM, occ) == 420);
    inv.release(show, carol);
    assert(show.state["A3"].kind == Kind::FREE);
    std::cout << "C++ demo OK\n";
    return 0;
}
