The same files live in lld-code/lld-case-splitwise/ (java, python, cpp) in this repo; each was compiled and run, and the output below is the real output.

Read in this order (the files are long; this is the 10-minute path):

  1. Main.main: the SHOW and SIMPLIFY lines.
  2. ExpenseService.addExpense: validate users, pick the split strategy, record into the balance sheet.
  3. EqualSplit, ExactSplit, PercentSplit: the strategies, and how each puts the rounding paise somewhere.
  4. ExpenseService.simplify with BalanceSheet.netBalances: the greedy debt simplification (about 20 lines, the part interviewers probe).
  5. Skip on first read: the small value classes User, Expense, Transfer.

From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

import java.util.*;

// ---------- model ----------
enum SplitType { EQUAL, EXACT, PERCENT }

final class User {
    final String id, name;
    User(String id, String name) { this.id = id; this.name = name; }
}

final class Expense {
    final String id, paidBy, description;
    final long totalPaise;
    final SplitType type;
    final Map<String, Long> shares;              // userId -> paise that user consumed
    Expense(String id, String paidBy, long totalPaise, SplitType type,
            Map<String, Long> shares, String description) {
        this.id = id; this.paidBy = paidBy; this.totalPaise = totalPaise;
        this.type = type; this.shares = Collections.unmodifiableMap(shares); this.description = description;
    }
}

final class Transfer {                           // "from pays to amountPaise"
    final String from, to; final long amountPaise;
    Transfer(String from, String to, long amountPaise) { this.from = from; this.to = to; this.amountPaise = amountPaise; }
}

class InvalidSplitException extends IllegalArgumentException {
    InvalidSplitException(String msg) { super(msg); }
}

// ---------- strategy: how one total becomes per-user shares ----------
interface SplitStrategy {
    /** values: unused for EQUAL, rupee amounts for EXACT, percentages for PERCENT. */
    Map<String, Long> split(long totalPaise, List<String> users, List<Double> values);
}

final class EqualSplit implements SplitStrategy {
    public Map<String, Long> split(long total, List<String> users, List<Double> values) {
        if (values != null && !values.isEmpty()) throw new InvalidSplitException("EQUAL takes no values");
        long base = total / users.size(), extra = total % users.size();
        Map<String, Long> out = new LinkedHashMap<>();
        for (int i = 0; i < users.size(); i++)
            out.merge(users.get(i), base + (i < extra ? 1 : 0), Long::sum);   // first users absorb the leftover paise
        return out;
    }
}

final class ExactSplit implements SplitStrategy {
    public Map<String, Long> split(long total, List<String> users, List<Double> values) {
        if (values == null || values.size() != users.size()) throw new InvalidSplitException("EXACT needs one amount per user");
        Map<String, Long> out = new LinkedHashMap<>();
        long sum = 0;
        for (int i = 0; i < users.size(); i++) {
            long p = Math.round(values.get(i) * 100);
            if (p < 0) throw new InvalidSplitException("negative amount");
            out.merge(users.get(i), p, Long::sum);
            sum += p;
        }
        if (sum != total) throw new InvalidSplitException("EXACT amounts add to " + sum + " paise, total is " + total);
        return out;
    }
}

final class PercentSplit implements SplitStrategy {
    public Map<String, Long> split(long total, List<String> users, List<Double> values) {
        if (values == null || values.size() != users.size()) throw new InvalidSplitException("PERCENT needs one value per user");
        long[] bp = new long[users.size()];       // basis points: 12.5% -> 1250
        long bpSum = 0;
        for (int i = 0; i < bp.length; i++) {
            bp[i] = Math.round(values.get(i) * 100);
            if (bp[i] < 0) throw new InvalidSplitException("negative percent");
            bpSum += bp[i];
        }
        if (bpSum != 10_000) throw new InvalidSplitException("percentages add to " + bpSum / 100.0 + ", not 100");
        Map<String, Long> out = new LinkedHashMap<>();
        long given = 0;
        for (int i = 0; i < bp.length; i++) {
            long p = total * bp[i] / 10_000;
            out.merge(users.get(i), p, Long::sum);
            given += p;
        }
        out.merge(users.get(0), total - given, Long::sum);   // rounding leftover goes to the first user
        return out;
    }
}

final class SplitStrategyFactory {
    private static final Map<SplitType, SplitStrategy> STRATEGIES = new EnumMap<>(SplitType.class);
    static {
        STRATEGIES.put(SplitType.EQUAL, new EqualSplit());
        STRATEGIES.put(SplitType.EXACT, new ExactSplit());
        STRATEGIES.put(SplitType.PERCENT, new PercentSplit());
    }
    static SplitStrategy of(SplitType t) { return STRATEGIES.get(t); }
}

// ---------- balance sheet: who owes whom ----------
final class BalanceSheet {
    // owed.get(a).get(b) > 0  means  b owes a that many paise. Always kept antisymmetric.
    private final Map<String, Map<String, Long>> owed = new HashMap<>();

    void record(Expense e) {
        for (Map.Entry<String, Long> s : e.shares.entrySet()) {
            String debtor = s.getKey();
            if (debtor.equals(e.paidBy) || s.getValue() == 0) continue;
            add(e.paidBy, debtor, s.getValue());
            add(debtor, e.paidBy, -s.getValue());
        }
    }
    private void add(String creditor, String debtor, long paise) {
        Map<String, Long> row = owed.computeIfAbsent(creditor, k -> new TreeMap<>());
        long v = row.getOrDefault(debtor, 0L) + paise;
        if (v == 0) row.remove(debtor); else row.put(debtor, v);
    }
    long owes(String debtor, String creditor) {                 // > 0 means debtor owes creditor
        return owed.getOrDefault(creditor, Collections.emptyMap()).getOrDefault(debtor, 0L);
    }
    /** Every pair with a positive debt, as "debtor owes creditor". */
    List<Transfer> debts(String onlyUser) {
        List<Transfer> out = new ArrayList<>();
        for (String creditor : new TreeSet<>(owed.keySet()))
            for (Map.Entry<String, Long> d : owed.get(creditor).entrySet())
                if (d.getValue() > 0 && (onlyUser == null || onlyUser.equals(creditor) || onlyUser.equals(d.getKey())))
                    out.add(new Transfer(d.getKey(), creditor, d.getValue()));
        return out;
    }
    Map<String, Long> netBalances() {                           // + means the group owes you
        Map<String, Long> net = new TreeMap<>();
        for (Map.Entry<String, Map<String, Long>> row : owed.entrySet())
            for (long v : row.getValue().values()) net.merge(row.getKey(), v, Long::sum);
        return net;
    }
}

// ---------- service (the facade the driver talks to) ----------
final class ExpenseService {
    private final Map<String, User> users = new LinkedHashMap<>();
    private final List<Expense> expenses = new ArrayList<>();
    private final BalanceSheet sheet = new BalanceSheet();
    private int nextId = 1;

    synchronized void addUser(String id, String name) {
        if (users.putIfAbsent(id, new User(id, name)) != null) throw new IllegalArgumentException("duplicate user " + id);
    }

    synchronized Expense addExpense(String paidBy, double amount, SplitType type,
                                    List<String> participants, List<Double> values, String desc) {
        requireUser(paidBy);
        if (participants == null || participants.isEmpty()) throw new InvalidSplitException("no participants");
        participants.forEach(this::requireUser);
        long total = Math.round(amount * 100);
        if (total <= 0) throw new InvalidSplitException("amount must be positive");
        Map<String, Long> shares = SplitStrategyFactory.of(type).split(total, participants, values);
        Expense e = new Expense("E" + nextId++, paidBy, total, type, shares, desc);
        expenses.add(e);           // validate fully BEFORE touching the sheet: all-or-nothing
        sheet.record(e);
        return e;
    }

    synchronized List<String> show(String userId) {
        if (userId != null) requireUser(userId);
        List<String> lines = new ArrayList<>();
        for (Transfer t : sheet.debts(userId))
            lines.add(name(t.from) + " owes " + name(t.to) + ": " + rupees(t.amountPaise));
        if (lines.isEmpty()) lines.add("No balances");
        return lines;
    }

    synchronized long owes(String debtor, String creditor) { return sheet.owes(debtor, creditor); }

    /** Greedy min-cash-flow: biggest debtor pays biggest creditor until both lists are empty. */
    synchronized List<Transfer> simplify() {
        PriorityQueue<long[]> give = new PriorityQueue<>((a, b) -> Long.compare(b[1], a[1]));
        PriorityQueue<long[]> take = new PriorityQueue<>((a, b) -> Long.compare(b[1], a[1]));
        List<String> ids = new ArrayList<>();
        for (Map.Entry<String, Long> n : sheet.netBalances().entrySet()) {
            ids.add(n.getKey());
            long idx = ids.size() - 1;
            if (n.getValue() > 0) take.add(new long[]{idx, n.getValue()});
            else if (n.getValue() < 0) give.add(new long[]{idx, -n.getValue()});
        }
        List<Transfer> out = new ArrayList<>();
        while (!give.isEmpty() && !take.isEmpty()) {
            long[] d = give.poll(), c = take.poll();
            long x = Math.min(d[1], c[1]);
            out.add(new Transfer(ids.get((int) d[0]), ids.get((int) c[0]), x));
            if (d[1] > x) give.add(new long[]{d[0], d[1] - x});
            if (c[1] > x) take.add(new long[]{c[0], c[1] - x});
        }
        return out;
    }

    synchronized int expenseCount() { return expenses.size(); }
    String name(String id) { return users.get(id).name; }
    static String rupees(long p) { return String.format("%d.%02d", p / 100, p % 100); }
    private void requireUser(String id) {
        if (!users.containsKey(id)) throw new IllegalArgumentException("unknown user " + id);
    }
}

public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }
    static void print(List<String> lines) { lines.forEach(l -> System.out.println("  " + l)); }

    public static void main(String[] args) {
        ExpenseService s = new ExpenseService();
        for (int i = 1; i <= 4; i++) s.addUser("u" + i, "User" + i);
        List<String> all = Arrays.asList("u1", "u2", "u3", "u4");

        System.out.println("SHOW");
        print(s.show(null));
        check(s.show(null).get(0).equals("No balances"), "empty sheet");

        System.out.println("EXPENSE u1 1000 EQUAL u1 u2 u3 u4");
        s.addExpense("u1", 1000, SplitType.EQUAL, all, null, "dinner");
        print(s.show("u1"));
        check(s.owes("u2", "u1") == 25000, "u2 owes u1 250");

        System.out.println("EXPENSE u1 1250 EXACT u2=370 u3=880");
        s.addExpense("u1", 1250, SplitType.EXACT, Arrays.asList("u2", "u3"), Arrays.asList(370.0, 880.0), "tickets");
        check(s.owes("u3", "u1") == 25000 + 88000, "u3 owes u1 1130");

        System.out.println("EXPENSE u4 1200 PERCENT u1=40 u2=20 u3=20 u4=20");
        s.addExpense("u4", 1200, SplitType.PERCENT, all, Arrays.asList(40.0, 20.0, 20.0, 20.0), "cab");
        check(s.owes("u1", "u4") == 48000 - 25000, "u1 owes u4 230 after netting");
        System.out.println("SHOW");
        print(s.show(null));

        System.out.println("EXPENSE u2 100 EQUAL u1 u2 u3  (100/3 does not divide)");
        Expense odd = s.addExpense("u2", 100, SplitType.EQUAL, Arrays.asList("u1", "u2", "u3"), null, "tea");
        check(odd.shares.get("u1") == 3334 && odd.shares.get("u3") == 3333, "33.34 / 33.33 / 33.33");
        long sum = 0; for (long v : odd.shares.values()) sum += v;
        check(sum == 10000, "shares add up to the total");

        int before = s.expenseCount();
        try {
            s.addExpense("u1", 500, SplitType.PERCENT, all, Arrays.asList(50.0, 20.0, 20.0, 5.0), "bad");
            check(false, "95% must be rejected");
        } catch (InvalidSplitException ex) {
            System.out.println("rejected: " + ex.getMessage());
        }
        check(s.expenseCount() == before, "rejected expense left no trace");

        System.out.println("SIMPLIFY");
        List<Transfer> plan = s.simplify();
        for (Transfer t : plan) System.out.println("  " + s.name(t.from) + " pays " + s.name(t.to) + " " + ExpenseService.rupees(t.amountPaise));
        check(plan.size() <= 3, "n users need at most n-1 transfers");

        System.out.println("All " + passed + " checks passed");
    }
}
"""Splitwise / expense sharing - machine-coding solution (in-memory, money in integer paise)."""
import heapq
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from enum import Enum

class SplitType(Enum):
    EQUAL = "EQUAL"
    EXACT = "EXACT"
    PERCENT = "PERCENT"

class InvalidSplitError(ValueError):
    pass

@dataclass(frozen=True)
class User:
    id: str
    name: str

@dataclass(frozen=True)
class Expense:
    id: str
    paid_by: str
    total_paise: int
    type: SplitType
    shares: dict = field(hash=False)      # user id -> paise that user consumed
    description: str = ""

@dataclass(frozen=True)
class Transfer:
    frm: str
    to: str
    amount_paise: int

# ---------- strategy: how one total becomes per-user shares ----------
class SplitStrategy(ABC):
    @abstractmethod
    def split(self, total: int, users: list, values: list) -> dict: ...

class EqualSplit(SplitStrategy):
    def split(self, total, users, values):
        if values:
            raise InvalidSplitError("EQUAL takes no values")
        base, extra = divmod(total, len(users))
        out = {}
        for i, u in enumerate(users):                 # first users absorb the leftover paise
            out[u] = out.get(u, 0) + base + (1 if i < extra else 0)
        return out

class ExactSplit(SplitStrategy):
    def split(self, total, users, values):
        if not values or len(values) != len(users):
            raise InvalidSplitError("EXACT needs one amount per user")
        out, s = {}, 0
        for u, v in zip(users, values):
            p = round(v * 100)
            if p < 0:
                raise InvalidSplitError("negative amount")
            out[u] = out.get(u, 0) + p
            s += p
        if s != total:
            raise InvalidSplitError(f"EXACT amounts add to {s} paise, total is {total}")
        return out

class PercentSplit(SplitStrategy):
    def split(self, total, users, values):
        if not values or len(values) != len(users):
            raise InvalidSplitError("PERCENT needs one value per user")
        bps = [round(v * 100) for v in values]        # basis points: 12.5% -> 1250
        if any(b < 0 for b in bps):
            raise InvalidSplitError("negative percent")
        if sum(bps) != 10_000:
            raise InvalidSplitError(f"percentages add to {sum(bps) / 100}, not 100")
        out, given = {}, 0
        for u, b in zip(users, bps):
            p = total * b // 10_000
            out[u] = out.get(u, 0) + p
            given += p
        out[users[0]] += total - given                # rounding leftover goes to the first user
        return out

STRATEGIES = {SplitType.EQUAL: EqualSplit(), SplitType.EXACT: ExactSplit(), SplitType.PERCENT: PercentSplit()}

# ---------- balance sheet ----------
class BalanceSheet:
    def __init__(self):
        self._owed = {}    # owed[a][b] > 0 means b owes a; kept antisymmetric

    def record(self, e: Expense):
        for debtor, amt in e.shares.items():
            if debtor == e.paid_by or amt == 0:
                continue
            self._add(e.paid_by, debtor, amt)
            self._add(debtor, e.paid_by, -amt)

    def _add(self, creditor, debtor, paise):
        row = self._owed.setdefault(creditor, {})
        v = row.get(debtor, 0) + paise
        if v == 0:
            row.pop(debtor, None)
        else:
            row[debtor] = v

    def owes(self, debtor, creditor):
        return self._owed.get(creditor, {}).get(debtor, 0)

    def debts(self, only_user=None):
        out = []
        for creditor in sorted(self._owed):
            for debtor in sorted(self._owed[creditor]):
                v = self._owed[creditor][debtor]
                if v > 0 and (only_user is None or only_user in (creditor, debtor)):
                    out.append(Transfer(debtor, creditor, v))
        return out

    def net_balances(self):
        return {u: sum(row.values()) for u, row in sorted(self._owed.items())}

# ---------- service (facade) ----------
class ExpenseService:
    def __init__(self):
        self._users = {}
        self._expenses = []
        self._sheet = BalanceSheet()
        self._lock = threading.Lock()
        self._next = 1

    def add_user(self, uid, name):
        with self._lock:
            if uid in self._users:
                raise ValueError(f"duplicate user {uid}")
            self._users[uid] = User(uid, name)

    def add_expense(self, paid_by, amount, type_, participants, values=None, desc=""):
        with self._lock:
            self._require(paid_by)
            if not participants:
                raise InvalidSplitError("no participants")
            for p in participants:
                self._require(p)
            total = round(amount * 100)
            if total <= 0:
                raise InvalidSplitError("amount must be positive")
            shares = STRATEGIES[type_].split(total, participants, values)
            e = Expense(f"E{self._next}", paid_by, total, type_, shares, desc)
            self._next += 1
            self._expenses.append(e)    # validated fully before the sheet changes
            self._sheet.record(e)
            return e

    def show(self, uid=None):
        with self._lock:
            if uid is not None:
                self._require(uid)
            lines = [f"{self.name(t.frm)} owes {self.name(t.to)}: {rupees(t.amount_paise)}"
                     for t in self._sheet.debts(uid)]
            return lines or ["No balances"]

    def owes(self, debtor, creditor):
        with self._lock:
            return self._sheet.owes(debtor, creditor)

    def simplify(self):
        """Greedy min-cash-flow: biggest debtor pays biggest creditor."""
        with self._lock:
            give, take = [], []
            for u, v in self._sheet.net_balances().items():
                if v > 0:
                    heapq.heappush(take, (-v, u))
                elif v < 0:
                    heapq.heappush(give, (v, u))
            out = []
            while give and take:
                dv, d = heapq.heappop(give)
                cv, c = heapq.heappop(take)
                x = min(-dv, -cv)
                out.append(Transfer(d, c, x))
                if -dv > x:
                    heapq.heappush(give, (dv + x, d))
                if -cv > x:
                    heapq.heappush(take, (cv + x, c))
            return out

    def expense_count(self):
        return len(self._expenses)

    def name(self, uid):
        return self._users[uid].name

    def _require(self, uid):
        if uid not in self._users:
            raise ValueError(f"unknown user {uid}")

def rupees(p):
    return f"{p // 100}.{p % 100:02d}"

def demo():
    s = ExpenseService()
    for i in range(1, 5):
        s.add_user(f"u{i}", f"User{i}")
    everyone = ["u1", "u2", "u3", "u4"]
    print("SHOW")
    print("\n".join("  " + l for l in s.show()))
    s.add_expense("u1", 1000, SplitType.EQUAL, everyone, desc="dinner")
    s.add_expense("u1", 1250, SplitType.EXACT, ["u2", "u3"], [370, 880], "tickets")
    s.add_expense("u4", 1200, SplitType.PERCENT, everyone, [40, 20, 20, 20], "cab")
    print("SHOW after 3 expenses")
    print("\n".join("  " + l for l in s.show()))
    s.add_expense("u2", 100, SplitType.EQUAL, ["u1", "u2", "u3"], desc="tea")
    try:
        s.add_expense("u1", 500, SplitType.PERCENT, everyone, [50, 20, 20, 5])
    except InvalidSplitError as ex:
        print("rejected:", ex)
    print("SIMPLIFY")
    for t in s.simplify():
        print(f"  {s.name(t.frm)} pays {s.name(t.to)} {rupees(t.amount_paise)}")
    return s

if __name__ == "__main__":
    demo()

import threading
import unittest

from main import ExpenseService, InvalidSplitError, SplitType

def fresh(n=4):
    s = ExpenseService()
    for i in range(1, n + 1):
        s.add_user(f"u{i}", f"User{i}")
    return s

class SplitwiseTest(unittest.TestCase):
    def test_empty(self):
        self.assertEqual(fresh().show(), ["No balances"])

    def test_equal_exact_percent_and_netting(self):
        s = fresh()
        s.add_expense("u1", 1000, SplitType.EQUAL, ["u1", "u2", "u3", "u4"])
        self.assertEqual(s.owes("u2", "u1"), 25000)
        s.add_expense("u1", 1250, SplitType.EXACT, ["u2", "u3"], [370, 880])
        self.assertEqual(s.owes("u3", "u1"), 113000)
        s.add_expense("u4", 1200, SplitType.PERCENT, ["u1", "u2", "u3", "u4"], [40, 20, 20, 20])
        self.assertEqual(s.owes("u1", "u4"), 23000)       # 480 - 250 after netting
        self.assertEqual(s.owes("u4", "u1"), -23000)      # antisymmetric

    def test_rounding_keeps_total(self):
        e = fresh().add_expense("u2", 100, SplitType.EQUAL, ["u1", "u2", "u3"])
        self.assertEqual(sum(e.shares.values()), 10000)
        self.assertEqual(e.shares["u1"], 3334)

    def test_invalid_is_all_or_nothing(self):
        s = fresh()
        with self.assertRaises(InvalidSplitError):
            s.add_expense("u1", 500, SplitType.PERCENT, ["u1", "u2"], [50, 40])
        with self.assertRaises(InvalidSplitError):
            s.add_expense("u1", 500, SplitType.EXACT, ["u1", "u2"], [100, 100])
        with self.assertRaises(ValueError):
            s.add_expense("u1", 500, SplitType.EQUAL, ["u1", "nobody"])
        self.assertEqual(s.expense_count(), 0)
        self.assertEqual(s.show(), ["No balances"])

    def test_simplify_settles_everything(self):
        s = fresh()
        s.add_expense("u1", 300, SplitType.EQUAL, ["u1", "u2", "u3"])
        s.add_expense("u2", 300, SplitType.EQUAL, ["u1", "u2", "u3"])
        plan = s.simplify()
        # u1 and u2 cancel each other out; only u3 still owes 100 to each of them
        self.assertEqual(len(plan), 2)
        self.assertTrue(all(t.frm == "u3" for t in plan))
        self.assertEqual(sum(t.amount_paise for t in plan), 20000)

    def test_concurrent_adds(self):
        s = fresh()
        def work():
            for _ in range(200):
                s.add_expense("u1", 4, SplitType.EQUAL, ["u1", "u2", "u3", "u4"])
        ts = [threading.Thread(target=work) for _ in range(4)]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
        self.assertEqual(s.expense_count(), 800)
        self.assertEqual(s.owes("u2", "u1"), 800 * 100)

#include <sstream>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>

using std::map; using std::string; using std::vector;
typedef long long paise_t;

enum class SplitType { EQUAL, EXACT, PERCENT };

struct InvalidSplit : std::invalid_argument {
    explicit InvalidSplit(const string& m) : std::invalid_argument(m) {}
};

struct Expense {
    string id, paidBy;
    paise_t total;
    SplitType type;
    map<string, paise_t> shares;   // user -> paise consumed
};

// ---------- strategy ----------
struct SplitStrategy {                       // the C "ops table", with a virtual destructor
    virtual ~SplitStrategy() {}
    virtual map<string, paise_t> split(paise_t total, const vector<string>& users,
                                       const vector<double>& values) const = 0;
};

static paise_t toPaise(double v) { return (paise_t)(v * 100 + (v >= 0 ? 0.5 : -0.5)); }

struct EqualSplit : SplitStrategy {
    map<string, paise_t> split(paise_t total, const vector<string>& users, const vector<double>& values) const override {
        if (!values.empty()) throw InvalidSplit("EQUAL takes no values");
        paise_t n = (paise_t)users.size(), base = total / n, extra = total % n;
        map<string, paise_t> out;
        for (size_t i = 0; i < users.size(); ++i) out[users[i]] += base + ((paise_t)i < extra ? 1 : 0);
        return out;
    }
};

struct ExactSplit : SplitStrategy {
    map<string, paise_t> split(paise_t total, const vector<string>& users, const vector<double>& values) const override {
        if (values.size() != users.size()) throw InvalidSplit("EXACT needs one amount per user");
        map<string, paise_t> out; paise_t sum = 0;
        for (size_t i = 0; i < users.size(); ++i) {
            paise_t p = toPaise(values[i]);
            if (p < 0) throw InvalidSplit("negative amount");
            out[users[i]] += p; sum += p;
        }
        if (sum != total) throw InvalidSplit("EXACT amounts do not add up to the total");
        return out;
    }
};

struct PercentSplit : SplitStrategy {
    map<string, paise_t> split(paise_t total, const vector<string>& users, const vector<double>& values) const override {
        if (values.size() != users.size()) throw InvalidSplit("PERCENT needs one value per user");
        vector<paise_t> bp; paise_t bpSum = 0;
        for (double v : values) { paise_t b = toPaise(v); if (b < 0) throw InvalidSplit("negative percent"); bp.push_back(b); bpSum += b; }
        if (bpSum != 10000) throw InvalidSplit("percentages do not add up to 100");
        map<string, paise_t> out; paise_t given = 0;
        for (size_t i = 0; i < users.size(); ++i) { paise_t p = total * bp[i] / 10000; out[users[i]] += p; given += p; }
        out[users[0]] += total - given;
        return out;
    }
};

// ---------- balance sheet ----------
class BalanceSheet {
    map<string, map<string, paise_t>> owed_;   // owed_[a][b] > 0: b owes a
    void add(const string& cr, const string& dr, paise_t p) {
        paise_t v = (owed_[cr][dr] += p);
        if (v == 0) owed_[cr].erase(dr);
    }
public:
    void record(const Expense& e) {
        for (const auto& s : e.shares) {
            if (s.first == e.paidBy || s.second == 0) continue;
            add(e.paidBy, s.first, s.second);
            add(s.first, e.paidBy, -s.second);
        }
    }
    paise_t owes(const string& debtor, const string& creditor) const {
        auto r = owed_.find(creditor);
        if (r == owed_.end()) return 0;
        auto c = r->second.find(debtor);
        return c == r->second.end() ? 0 : c->second;
    }
    vector<std::pair<string, std::pair<string, paise_t>>> debts() const {   // (debtor, (creditor, amount))
        vector<std::pair<string, std::pair<string, paise_t>>> out;
        for (const auto& row : owed_)
            for (const auto& d : row.second)
                if (d.second > 0) out.push_back({d.first, {row.first, d.second}});
        return out;
    }
    map<string, paise_t> net() const {
        map<string, paise_t> n;
        for (const auto& row : owed_) for (const auto& d : row.second) n[row.first] += d.second;
        return n;
    }
};

// ---------- service ----------
class ExpenseService {
    map<string, string> users_;                          // id -> name
    vector<Expense> expenses_;
    BalanceSheet sheet_;
    map<SplitType, std::unique_ptr<SplitStrategy>> strategies_;
    int next_ = 1;
    void require(const string& id) const { if (!users_.count(id)) throw std::invalid_argument("unknown user " + id); }
public:
    ExpenseService() {
        strategies_[SplitType::EQUAL].reset(new EqualSplit());
        strategies_[SplitType::EXACT].reset(new ExactSplit());
        strategies_[SplitType::PERCENT].reset(new PercentSplit());
    }
    void addUser(const string& id, const string& name) { users_[id] = name; }
    const Expense& addExpense(const string& paidBy, double amount, SplitType t,
                              const vector<string>& who, const vector<double>& values) {
        require(paidBy);
        if (who.empty()) throw InvalidSplit("no participants");
        for (const auto& u : who) require(u);
        paise_t total = toPaise(amount);
        if (total <= 0) throw InvalidSplit("amount must be positive");
        Expense e{"E" + std::to_string(next_++), paidBy, total, t, strategies_[t]->split(total, who, values)};
        expenses_.push_back(e);
        sheet_.record(expenses_.back());
        return expenses_.back();
    }
    paise_t owes(const string& d, const string& c) const { return sheet_.owes(d, c); }
    size_t count() const { return expenses_.size(); }
    const string& name(const string& id) const { return users_.at(id); }
    static string rupees(paise_t p) {
        std::ostringstream o; o << p / 100 << '.' << std::setw(2) << std::setfill('0') << p % 100; return o.str();
    }
    void show() const {
        auto d = sheet_.debts();
        if (d.empty()) std::cout << "  No balances\n";
        for (const auto& x : d) std::cout << "  " << name(x.first) << " owes " << name(x.second.first) << ": " << rupees(x.second.second) << "\n";
    }
    vector<std::pair<std::pair<string, string>, paise_t>> simplify() const {
        typedef std::pair<paise_t, string> P;
        std::priority_queue<P> give, take;                    // max-heaps
        for (const auto& n : sheet_.net()) {
            if (n.second > 0) take.push({n.second, n.first});
            else if (n.second < 0) give.push({-n.second, n.first});
        }
        vector<std::pair<std::pair<string, string>, paise_t>> out;
        while (!give.empty() && !take.empty()) {
            P d = give.top(); give.pop(); P c = take.top(); take.pop();
            paise_t x = std::min(d.first, c.first);
            out.push_back({{d.second, c.second}, x});
            if (d.first > x) give.push({d.first - x, d.second});
            if (c.first > x) take.push({c.first - x, c.second});
        }
        return out;
    }
};

static int passed = 0;
static void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); }
    ++passed;
}

int main() {
    ExpenseService s;
    for (int i = 1; i <= 4; ++i) { string id = "u" + std::to_string(i); s.addUser(id, "User" + std::to_string(i)); }
    vector<string> all = {"u1", "u2", "u3", "u4"};
    s.addExpense("u1", 1000, SplitType::EQUAL, all, {});
    s.addExpense("u1", 1250, SplitType::EXACT, {"u2", "u3"}, {370, 880});
    s.addExpense("u4", 1200, SplitType::PERCENT, all, {40, 20, 20, 20});
    std::cout << "SHOW\n"; s.show();
    check(s.owes("u2", "u1") == 62000, "u2 owes u1 620");
    check(s.owes("u1", "u4") == 23000, "u1 owes u4 230");
    const Expense& tea = s.addExpense("u2", 100, SplitType::EQUAL, {"u1", "u2", "u3"}, {});
    check(tea.shares.at("u1") == 3334 && tea.shares.at("u3") == 3333, "33.34/33.33/33.33");
    size_t before = s.count();
    try { s.addExpense("u1", 500, SplitType::PERCENT, all, {50, 20, 20, 5}); check(false, "95% accepted"); }
    catch (const InvalidSplit& e) { std::cout << "rejected: " << e.what() << "\n"; }
    check(s.count() == before, "rejected expense left no trace");
    std::cout << "SIMPLIFY\n";
    for (const auto& t : s.simplify())
        std::cout << "  " << s.name(t.first.first) << " pays " << s.name(t.first.second) << " " << ExpenseService::rupees(t.second) << "\n";
    std::cout << "All " << passed << " checks passed\n";
    return 0;
}
