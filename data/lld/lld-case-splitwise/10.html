CREATE TABLE users    (id VARCHAR(32) PRIMARY KEY, name VARCHAR(100) NOT NULL);
CREATE TABLE expenses (id BIGINT PRIMARY KEY, group_id BIGINT, paid_by VARCHAR(32) REFERENCES users(id),
                       total_paise BIGINT NOT NULL CHECK (total_paise > 0), split_type VARCHAR(10),
                       request_id VARCHAR(64) UNIQUE, created_at TIMESTAMP NOT NULL);
CREATE TABLE expense_shares (expense_id BIGINT REFERENCES expenses(id), user_id VARCHAR(32) REFERENCES users(id),
                             share_paise BIGINT NOT NULL, PRIMARY KEY (expense_id, user_id));

-- what does each user owe the payer? (balance derived from facts, no update race)
SELECT s.user_id AS debtor, e.paid_by AS creditor, SUM(s.share_paise) AS paise
FROM expense_shares s JOIN expenses e ON e.id = s.expense_id
WHERE s.user_id <> e.paid_by AND e.group_id = ?
GROUP BY s.user_id, e.paid_by;

Net the two directions in code (or with a self-join). If reads must be fast, add a balances(user_lo, user_hi, paise) table with the pair stored in sorted order, updated in the same transaction as the expense.
