/** What kind of delivery it was. */
enum BallType { NORMAL, WIDE, NO_BALL, WICKET }

/** One delivery. Immutable value object, created only through parse(). */
final class Ball {
    final BallType type;
    final int batRuns;   // runs credited to the batter (and run between the wickets)
    final int extras;    // penalty runs (1 for a wide or a no-ball)

    private Ball(BallType type, int batRuns, int extras) {
        this.type = type; this.batRuns = batRuns; this.extras = extras;
    }

    /** Tokens: "0".."6", "W", "Wd", "Nb" or "Nb<runs>" such as "Nb4". Case does not matter. */
    static Ball parse(String token) {
        String t = token.trim().toUpperCase();
        if (t.equals("W")) return new Ball(BallType.WICKET, 0, 0);
        if (t.equals("WD")) return new Ball(BallType.WIDE, 0, 1);
        if (t.startsWith("NB")) return new Ball(BallType.NO_BALL, runs(t.substring(2), token, true), 1);
        return new Ball(BallType.NORMAL, runs(t, token, false), 0);
    }

    private static int runs(String digits, String token, boolean emptyIsZero) {
        if (digits.isEmpty() && emptyIsZero) return 0;
        if (!digits.matches("[0-6]")) throw new IllegalArgumentException("bad ball token: " + token);
        return Integer.parseInt(digits);
    }

    boolean isLegal()  { return type != BallType.WIDE && type != BallType.NO_BALL; }
    boolean isWicket() { return type == BallType.WICKET; }
    int totalRuns()    { return batRuns + extras; }
    /** Batters cross when they run an odd number. A wide or no-ball penalty is not run. */
    boolean crossesStrike() { return batRuns % 2 == 1; }
}

        throw new IllegalArgumentException(name + " is not in " + this.name);
    }
}

/** One batter's line in one innings. */
final class BattingStats {
    int runs, balls, fours, sixes;
    boolean out;
    double strikeRate() { return balls == 0 ? 0 : 100.0 * runs / balls; }
}

/** One bowler's figures in one innings. */
final class BowlingStats {
    int legalBalls, runsConceded, wickets;
    String overs() { return (legalBalls / 6) + "." + (legalBalls % 6); }
    double economy() { return legalBalls == 0 ? 0 : runsConceded * 6.0 / legalBalls; }
}
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/** One team batting once. Owns every scoring rule; knows nothing about printing. */
final class Innings {
    static final int BALLS_PER_OVER = 6;

    final Team batting, bowling;
    final int maxOvers;
    final Integer target;                       // null in the first innings
    private final List<Player> order;
    private final Map<Player, BattingStats> batCard = new LinkedHashMap<>();
    private final Map<Player, BowlingStats> bowlCard = new LinkedHashMap<>();
    private int runs, wickets, extras, legalBalls, nextBatter;
    private Player striker, nonStriker, bowler, lastOverBowler;

    Innings(Team batting, Team bowling, int maxOvers, Integer target) {
        this.batting = batting; this.bowling = bowling; this.maxOvers = maxOvers; this.target = target;
        this.order = batting.battingOrder();
        for (Player p : order) batCard.put(p, new BattingStats());
        striker = order.get(0); nonStriker = order.get(1); nextBatter = 2;
    }

    void startOver(Player next) {
        if (isComplete()) throw new IllegalStateException("innings is over");
        if (bowler != null) throw new IllegalStateException("over in progress, bowled by " + bowler);
        if (next == lastOverBowler) throw new IllegalArgumentException(next + " cannot bowl two overs in a row");
        bowler = next;
        bowlCard.putIfAbsent(next, new BowlingStats());
    }

    /** Applies one delivery. Returns true when it completed an over. */
    boolean bowl(Ball b) {
        if (isComplete()) throw new IllegalStateException("innings is over");
        if (bowler == null) throw new IllegalStateException("call startOver(bowler) first");
        BattingStats bat = batCard.get(striker);
        BowlingStats bowl = bowlCard.get(bowler);

        runs += b.totalRuns();
        extras += b.extras;
        bowl.runsConceded += b.totalRuns();     // wides and no-balls count against the bowler
        if (b.type != BallType.WIDE) bat.balls++; // a no-ball is still a ball faced
        bat.runs += b.batRuns;
        if (b.batRuns == 4) bat.fours++;
        if (b.batRuns == 6) bat.sixes++;
        if (b.isLegal()) { legalBalls++; bowl.legalBalls++; }

        if (b.isWicket()) {
            wickets++; bowl.wickets++; bat.out = true;
            striker = nextBatter < order.size() ? order.get(nextBatter++) : null;  // new batter takes strike
        } else if (b.crossesStrike()) {
            swapStrike();
        }
        boolean overEnded = b.isLegal() && legalBalls % BALLS_PER_OVER == 0;
        if (overEnded) { swapStrike(); lastOverBowler = bowler; bowler = null; }
        return overEnded;
    }

    private void swapStrike() { Player t = striker; striker = nonStriker; nonStriker = t; }

    boolean isComplete() {
        return wickets >= order.size() - 1                      // all out: last batter has no partner
            || legalBalls >= maxOvers * BALLS_PER_OVER
            || (target != null && runs >= target);
    }

    boolean atCrease(Player p) { return p == striker || p == nonStriker; }   // the "*" on a scorecard
    int runs()        { return runs; }
    int wickets()     { return wickets; }
    int extras()      { return extras; }
    int legalBalls()  { return legalBalls; }
    Player striker()  { return striker; }
    String overs()    { return legalBalls / 6 + (legalBalls % 6 == 0 ? "" : "." + legalBalls % 6); }
    Map<Player, BattingStats> battingCard() { return Collections.unmodifiableMap(batCard); }
    Map<Player, BowlingStats> bowlingCard() { return Collections.unmodifiableMap(bowlCard); }
}
import java.util.ArrayList;
import java.util.List;

/** Observer: anyone who wants live updates. Default no-ops, so a listener overrides only what it needs. */
interface MatchListener {
    default void onBall(Innings in, Ball b) {}   // C: default method = an ops-table slot with a generic body
    default void onOverEnd(Innings in) {}
    default void onInningsEnd(Innings in) {}
    default void onMatchEnd(Match m, String result) {}
}

/** The orchestrator: two innings, the match state machine, and event fan-out. */
final class Match {
    enum State { FIRST_INNINGS, SECOND_INNINGS, FINISHED }

    final Team team1, team2;
    final int overs;
    private final List<MatchListener> listeners = new ArrayList<>();
    private final Innings first;
    private Innings second;
    private State state = State.FIRST_INNINGS;
    private String result;

    Match(Team team1, Team team2, int overs) {
        if (overs < 1) throw new IllegalArgumentException("overs must be >= 1");
        if (team1.battingOrder().size() != team2.battingOrder().size())
            throw new IllegalArgumentException("teams must have the same number of players");
        this.team1 = team1; this.team2 = team2; this.overs = overs;
        this.first = new Innings(team1, team2, overs, null);
    }

    void addListener(MatchListener l) { listeners.add(l); }
    State state() { return state; }
    String result() { return result; }

    Innings current() {
        if (state == State.FINISHED) throw new IllegalStateException("match is finished");
        return state == State.FIRST_INNINGS ? first : second;
    }

    void startOver(String bowlerName) {
        Innings in = current();
        in.startOver(in.bowling.find(bowlerName));
    }

    /** Scorer input: one token per delivery, e.g. "1", "4", "W", "Wd", "Nb4". */
    void bowl(String token) {
        Innings in = current();
        Ball b = Ball.parse(token);               // validate before changing anything
        boolean overEnded = in.bowl(b);
        for (MatchListener l : listeners) l.onBall(in, b);
        if (overEnded) for (MatchListener l : listeners) l.onOverEnd(in);
        if (in.isComplete()) endInnings(in);
    }

    private void endInnings(Innings in) {
        for (MatchListener l : listeners) l.onInningsEnd(in);
        if (state == State.FIRST_INNINGS) {
            second = new Innings(team2, team1, overs, first.runs() + 1);
            state = State.SECOND_INNINGS;
        } else {
            state = State.FINISHED;
            result = decide();
            for (MatchListener l : listeners) l.onMatchEnd(this, result);
        }
    }

    private String decide() {
        int maxWickets = team2.battingOrder().size() - 1;
        if (second.runs() > first.runs())
            return team2.name + " won the match by " + plural(maxWickets - second.wickets(), "wicket");
        if (second.runs() == first.runs()) return "Match tied";
        return team1.name + " won the match by " + plural(first.runs() - second.runs(), "run");
    }

    private static String plural(int n, String word) { return n + " " + word + (n == 1 ? "" : "s"); }
}

/** Observer 1: prints the Udaan-style scorecard at the end of every over and of every innings. */
final class ScorecardPrinter implements MatchListener {
    private boolean dirty;   // a ball was bowled since the last printed card

    @Override public void onBall(Innings in, Ball b) { dirty = true; }
    @Override public void onOverEnd(Innings in) { print(in); }
    @Override public void onInningsEnd(Innings in) {
        if (dirty) print(in);                       // innings ended mid-over
        System.out.print("Bowling:");
        for (Map.Entry<Player, BowlingStats> e : in.bowlingCard().entrySet()) {
            BowlingStats s = e.getValue();
            System.out.printf("  %s %s ov %d runs %d wkts econ %.2f |", e.getKey(), s.overs(), s.runsConceded, s.wickets, s.economy());
        }
        System.out.println();
    }
    @Override public void onMatchEnd(Match m, String result) { System.out.println("\nResult: " + result); }

    private void print(Innings in) {
        dirty = false;
        System.out.println("\nScorecard for " + in.batting.name + ":");
        System.out.printf("%-8s%6s%4s%4s%7s%8s%n", "Player", "Score", "4s", "6s", "Balls", "SR");
        for (Map.Entry<Player, BattingStats> e : in.battingCard().entrySet()) {
            BattingStats s = e.getValue();
            String name = e.getKey().name + (in.atCrease(e.getKey()) ? "*" : "");
            System.out.printf("%-8s%6d%4d%4d%7d%8.1f%n", name, s.runs, s.fours, s.sixes, s.balls, s.strikeRate());
        }
        System.out.println("Total: " + in.runs() + "/" + in.wickets() + "  Extras: " + in.extras());
        System.out.println("Overs: " + in.overs());
    }
}

"""Cricket scoreboard (Cricbuzz-style machine-coding case). Same design as the Java version."""
from __future__ import annotations

import re
import threading
from dataclasses import dataclass
from enum import Enum

BALLS_PER_OVER = 6


class BallType(Enum):
    NORMAL = "normal"
    WIDE = "wide"
    NO_BALL = "no_ball"
    WICKET = "wicket"


@dataclass(frozen=True)
class Ball:
    """One delivery. Immutable value object; build it with Ball.parse()."""
    type: BallType
    bat_runs: int = 0
    extras: int = 0

    @staticmethod
    def parse(token: str) -> "Ball":
        t = token.strip().upper()
        if t == "W":
            return Ball(BallType.WICKET)
        if t == "WD":
            return Ball(BallType.WIDE, 0, 1)
        if t.startswith("NB"):
            rest = t[2:]
            if rest and not re.fullmatch(r"[0-6]", rest):
                raise ValueError(f"bad ball token: {token}")
            return Ball(BallType.NO_BALL, int(rest or 0), 1)
        if not re.fullmatch(r"[0-6]", t):
            raise ValueError(f"bad ball token: {token}")
        return Ball(BallType.NORMAL, int(t), 0)

    @property
    def legal(self) -> bool:
        return self.type not in (BallType.WIDE, BallType.NO_BALL)

    @property
    def total(self) -> int:
        return self.bat_runs + self.extras


class Player:
    def __init__(self, name: str) -> None:
        self.name = name

    def __repr__(self) -> str:
        return self.name


class Team:
    def __init__(self, name: str, order: list[str]) -> None:
        if len(order) < 2:
            raise ValueError("a team needs at least 2 players")
        self.name = name
        self.batting_order = tuple(Player(n) for n in order)

    def find(self, name: str) -> Player:
        for p in self.batting_order:
            if p.name == name:
                return p
        raise ValueError(f"{name} is not in {self.name}")

class Innings:
    """One team batting once. Owns the scoring rules; never prints."""

    def __init__(self, batting: Team, bowling: Team, max_overs: int, target: int | None) -> None:
        self.batting, self.bowling, self.max_overs, self.target = batting, bowling, max_overs, target
        self._order = batting.batting_order
        self.batting_card = {p: BattingStats() for p in self._order}   # dicts keep insertion order
        self.bowling_card: dict[Player, BowlingStats] = {}
        self.runs = self.wickets = self.extras = self.legal_balls = 0
        self.striker: Player | None = self._order[0]
        self.non_striker: Player | None = self._order[1]
        self._next = 2
        self._bowler: Player | None = None
        self._last_bowler: Player | None = None

    def start_over(self, bowler: Player) -> None:
        if self.complete:
            raise RuntimeError("innings is over")
        if self._bowler is not None:
            raise RuntimeError(f"over in progress, bowled by {self._bowler}")
        if bowler is self._last_bowler:
            raise ValueError(f"{bowler} cannot bowl two overs in a row")
        self._bowler = bowler
        self.bowling_card.setdefault(bowler, BowlingStats())

    def bowl(self, b: Ball) -> bool:
        """Apply one delivery; return True if it completed an over."""
        if self.complete:
            raise RuntimeError("innings is over")
        if self._bowler is None:
            raise RuntimeError("call start_over(bowler) first")
        bat = self.batting_card[self.striker]
        bowl = self.bowling_card[self._bowler]
        self.runs += b.total
        self.extras += b.extras
        bowl.runs += b.total
        if b.type is not BallType.WIDE:
            bat.balls += 1
        bat.runs += b.bat_runs
        bat.fours += b.bat_runs == 4
        bat.sixes += b.bat_runs == 6
        if b.legal:
            self.legal_balls += 1
            bowl.legal_balls += 1
        if b.type is BallType.WICKET:
            self.wickets += 1
            bowl.wickets += 1
            bat.out = True
            self.striker = self._order[self._next] if self._next < len(self._order) else None
            self._next += 1
        elif b.bat_runs % 2 == 1:
            self._swap()
        over_ended = b.legal and self.legal_balls % BALLS_PER_OVER == 0
        if over_ended:
            self._swap()
            self._last_bowler, self._bowler = self._bowler, None
        return over_ended

    def _swap(self) -> None:
        self.striker, self.non_striker = self.non_striker, self.striker

    @property
    def complete(self) -> bool:
        return (self.wickets >= len(self._order) - 1
                or self.legal_balls >= self.max_overs * BALLS_PER_OVER
                or (self.target is not None and self.runs >= self.target))

    def at_crease(self, p: Player) -> bool:
        return p is self.striker or p is self.non_striker

    @property
    def overs(self) -> str:
        whole, part = divmod(self.legal_balls, BALLS_PER_OVER)
        return f"{whole}.{part}" if part else str(whole)


class MatchListener:
    """Observer base class: override only the events you need."""
    def on_ball(self, innings: Innings, ball: Ball) -> None: ...
    def on_over_end(self, innings: Innings) -> None: ...
    def on_innings_end(self, innings: Innings) -> None: ...
    def on_match_end(self, match: "Match", result: str) -> None: ...


class MatchState(Enum):
    FIRST_INNINGS = 1
    SECOND_INNINGS = 2
    FINISHED = 3


def _plural(n: int, word: str) -> str:
    return f"{n} {word}" + ("" if n == 1 else "s")


class Match:
    def __init__(self, team1: Team, team2: Team, overs: int) -> None:
        if overs < 1:
            raise ValueError("overs must be >= 1")
        if len(team1.batting_order) != len(team2.batting_order):
            raise ValueError("teams must have the same number of players")
        self.team1, self.team2, self.overs = team1, team2, overs
        self._listeners: list[MatchListener] = []
        self.first = Innings(team1, team2, overs, None)
        self.second: Innings | None = None
        self.state = MatchState.FIRST_INNINGS
        self.result: str | None = None

    def add_listener(self, listener: MatchListener) -> None:
        self._listeners.append(listener)

    @property
    def current(self) -> Innings:
        if self.state is MatchState.FINISHED:
            raise RuntimeError("match is finished")
        return self.first if self.state is MatchState.FIRST_INNINGS else self.second

    def start_over(self, bowler_name: str) -> None:
        inn = self.current
        inn.start_over(inn.bowling.find(bowler_name))

    def bowl(self, token: str) -> None:
        inn = self.current
        ball = Ball.parse(token)               # validate before changing anything
        over_ended = inn.bowl(ball)
        for l in self._listeners:
            l.on_ball(inn, ball)
        if over_ended:
            for l in self._listeners:
                l.on_over_end(inn)
        if inn.complete:
            self._end_innings(inn)

    def _end_innings(self, inn: Innings) -> None:
        for l in self._listeners:
            l.on_innings_end(inn)
        if self.state is MatchState.FIRST_INNINGS:
            self.second = Innings(self.team2, self.team1, self.overs, self.first.runs + 1)
            self.state = MatchState.SECOND_INNINGS
        else:
            self.state = MatchState.FINISHED
            self.result = self._decide()
            for l in self._listeners:
                l.on_match_end(self, self.result)

    def _decide(self) -> str:
        a, b = self.first, self.second
        if b.runs > a.runs:
            left = len(self.team2.batting_order) - 1 - b.wickets
            return f"{self.team2.name} won the match by {_plural(left, 'wicket')}"
        if b.runs == a.runs:
            return "Match tied"
        return f"{self.team1.name} won the match by {_plural(a.runs - b.runs, 'run')}"


class ScorecardPrinter(MatchListener):
    def __init__(self) -> None:
        self._dirty = False

    def on_ball(self, innings, ball):
        self._dirty = True

    def on_over_end(self, innings):
        self._print(innings)

    def on_innings_end(self, innings):
        if self._dirty:
            self._print(innings)
        parts = [f"{p} {s.overs} ov {s.runs} runs {s.wickets} wkts econ {s.economy:.2f}"
                 for p, s in innings.bowling_card.items()]
        print("Bowling:  " + " |  ".join(parts) + " |")

    def on_match_end(self, match, result):
        print(f"\nResult: {result}")

    def _print(self, inn: Innings) -> None:
        self._dirty = False
        print(f"\nScorecard for {inn.batting.name}:")
        print(f"{'Player':<8}{'Score':>6}{'4s':>4}{'6s':>4}{'Balls':>7}{'SR':>8}")
        for p, s in inn.batting_card.items():
            name = p.name + ("*" if inn.at_crease(p) else "")
            print(f"{name:<8}{s.runs:>6}{s.fours:>4}{s.sixes:>4}{s.balls:>7}{s.strike_rate:>8.1f}")
        print(f"Total: {inn.runs}/{inn.wickets}  Extras: {inn.extras}")
        print(f"Overs: {inn.overs}")


class LiveScore(MatchListener):
    """Live feed: the scorer thread replaces one immutable string; readers just read it."""

    def __init__(self) -> None:
        self._lock = threading.Lock()

    def latest(self) -> str:

        self._set(result)

    match.start_over(bowler)
    for t in balls.split():
        if match.state is MatchState.FINISHED:
            return
        match.bowl(t)

// Build: g++ -std=c++14 -Wall -Wextra -O2 main.cpp -o cricket
#include <cassert>
#include <cctype>
#include <cstdio>
#include <memory>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>
#include <mutex>
// Locking: one facade lock, a real std::recursive_mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). Recursive because public methods call each other,
// just as Java's synchronized is re-entrant. MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and
// -pthread fails to link there, so the default build uses a no-op Mutex with the same lock()/unlock():
// the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::recursive_mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

static const int BALLS_PER_OVER = 6;

enum class BallType { Normal, Wide, NoBall, Wicket };

struct Ball {                                    // immutable value object
    BallType type; int batRuns; int extras;
    bool legal() const { return type != BallType::Wide && type != BallType::NoBall; }
    int total() const { return batRuns + extras; }

    static int digit(const std::string& d, const std::string& tok) {
        if (d.size() != 1 || d[0] < '0' || d[0] > '6') throw std::invalid_argument("bad ball token: " + tok);
        return d[0] - '0';
    }
    static Ball parse(const std::string& tok) {
        std::string t;
        for (char c : tok) t += static_cast<char>(toupper(static_cast<unsigned char>(c)));
        if (t == "W") return Ball{BallType::Wicket, 0, 0};
        if (t == "WD") return Ball{BallType::Wide, 0, 1};
        if (t.compare(0, 2, "NB") == 0) return Ball{BallType::NoBall, t.size() == 2 ? 0 : digit(t.substr(2), tok), 1};
        return Ball{BallType::Normal, digit(t, tok), 0};
    }
};

struct BattingStats { int runs = 0, balls = 0, fours = 0, sixes = 0; bool out = false;
    double strikeRate() const { return balls ? 100.0 * runs / balls : 0.0; } };
struct BowlingStats { int legalBalls = 0, runs = 0, wickets = 0;
    double economy() const { return legalBalls ? runs * 6.0 / legalBalls : 0.0; } };

struct Team {
    std::string name; std::vector<std::string> order;   // players are identified by index in order
    Team(std::string n, std::vector<std::string> o) : name(std::move(n)), order(std::move(o)) {
        if (order.size() < 2) throw std::invalid_argument("a team needs at least 2 players");
    }
    int find(const std::string& p) const {
        for (size_t i = 0; i < order.size(); ++i) if (order[i] == p) return static_cast<int>(i);
        throw std::invalid_argument(p + " is not in " + name);
    }
};

class Innings {
public:
    const Team& batting; const Team& bowling; const int maxOvers; const int target;   // target -1 = first innings
    std::vector<BattingStats> bat;                         // index = batting order
    std::vector<int> bowlOrder; std::vector<BowlingStats> bowl;   // bowlers in the order they first bowled

    Innings(const Team& b, const Team& w, int overs, int tgt)
        : batting(b), bowling(w), maxOvers(overs), target(tgt), bat(b.order.size()) {}

    void startOver(int bowler) {
        if (complete()) throw std::logic_error("innings is over");
        if (bowler_ >= 0) throw std::logic_error("over in progress");
        if (bowler == lastBowler_) throw std::invalid_argument(bowling.order[bowler] + " cannot bowl two overs in a row");
        bowler_ = bowler;
        if (slot(bowler) < 0) { bowlOrder.push_back(bowler); bowl.push_back(BowlingStats{}); }
    }

    bool bowlBall(const Ball& b) {                         // true when the ball completed an over
        if (complete()) throw std::logic_error("innings is over");
        if (bowler_ < 0) throw std::logic_error("call startOver(bowler) first");
        BattingStats& s = bat[striker_];
        BowlingStats& w = bowl[slot(bowler_)];
        runs_ += b.total(); extras_ += b.extras; w.runs += b.total();
        if (b.type != BallType::Wide) s.balls++;
        s.runs += b.batRuns;
        if (b.batRuns == 4) s.fours++;
        if (b.batRuns == 6) s.sixes++;
        if (b.legal()) { legal_++; w.legalBalls++; }
        if (b.type == BallType::Wicket) {
            wickets_++; w.wickets++; s.out = true;
            striker_ = next_ < static_cast<int>(bat.size()) ? next_++ : -1;
        } else if (b.batRuns % 2 == 1) {
            std::swap(striker_, nonStriker_);
        }
        bool overEnded = b.legal() && legal_ % BALLS_PER_OVER == 0;
        if (overEnded) { std::swap(striker_, nonStriker_); lastBowler_ = bowler_; bowler_ = -1; }
        return overEnded;
    }

    bool complete() const {
        return wickets_ >= static_cast<int>(bat.size()) - 1 || legal_ >= maxOvers * BALLS_PER_OVER
            || (target >= 0 && runs_ >= target);
    }
    bool atCrease(int p) const { return p == striker_ || p == nonStriker_; }
    int runs() const { return runs_; }
    int wickets() const { return wickets_; }
    int extras() const { return extras_; }
    int legalBalls() const { return legal_; }
    int striker() const { return striker_; }
    std::string overs() const {
        std::string s = std::to_string(legal_ / 6);
        return legal_ % 6 ? s + "." + std::to_string(legal_ % 6) : s;
    }

private:
    int slot(int bowler) const {
        for (size_t i = 0; i < bowlOrder.size(); ++i) if (bowlOrder[i] == bowler) return static_cast<int>(i);
        return -1;
    }
    int runs_ = 0, wickets_ = 0, extras_ = 0, legal_ = 0;
    int striker_ = 0, nonStriker_ = 1, next_ = 2, bowler_ = -1, lastBowler_ = -1;
};

struct MatchListener {                                   // Observer: a vtable of event hooks
    virtual ~MatchListener() = default;
    virtual void onBall(const Innings&, const Ball&) {}
    virtual void onOverEnd(const Innings&) {}
    virtual void onInningsEnd(const Innings&) {}
    virtual void onMatchEnd(const std::string&) {}
};

class Match {
    mutable Mutex mu_;   // the facade lock: every public call holds it
public:
    enum class State { FirstInnings, SecondInnings, Finished };
    Match(Team a, Team b, int overs) : t1_(std::move(a)), t2_(std::move(b)), overs_(overs) {
        if (overs < 1) throw std::invalid_argument("overs must be >= 1");
        if (t1_.order.size() != t2_.order.size()) throw std::invalid_argument("teams must have the same size");
        first_.reset(new Innings(t1_, t2_, overs_, -1));
    }
    void addListener(MatchListener* l) { LockGuard g(mu_); listeners_.push_back(l); }   // not owned
    State state() const { LockGuard g(mu_); return state_; }
    const std::string& result() const { LockGuard g(mu_); return result_; }
    Innings& current() { LockGuard g(mu_);
        if (state_ == State::Finished) throw std::logic_error("match is finished");
        return state_ == State::FirstInnings ? *first_ : *second_;
    }
    void startOver(const std::string& bowler) { LockGuard g(mu_); Innings& in = current(); in.startOver(in.bowling.find(bowler)); }
    void bowl(const std::string& token) { LockGuard g(mu_);
        Innings& in = current();
        Ball b = Ball::parse(token);
        bool overEnded = in.bowlBall(b);
        for (auto* l : listeners_) l->onBall(in, b);
        if (overEnded) for (auto* l : listeners_) l->onOverEnd(in);
        if (in.complete()) endInnings(in);
    }

private:
    static std::string plural(int n, const std::string& w) { return std::to_string(n) + " " + w + (n == 1 ? "" : "s"); }
    void endInnings(Innings& in) {
        for (auto* l : listeners_) l->onInningsEnd(in);
        if (state_ == State::FirstInnings) {
            second_.reset(new Innings(t2_, t1_, overs_, first_->runs() + 1));
            state_ = State::SecondInnings;
        } else {
            state_ = State::Finished;
            int a = first_->runs(), b = second_->runs();
            if (b > a) result_ = t2_.name + " won the match by " + plural(static_cast<int>(t2_.order.size()) - 1 - second_->wickets(), "wicket");
            else if (a == b) result_ = "Match tied";
            else result_ = t1_.name + " won the match by " + plural(a - b, "run");
            for (auto* l : listeners_) l->onMatchEnd(result_);
        }
    }
    Team t1_, t2_; int overs_;
    std::unique_ptr<Innings> first_, second_;
    std::vector<MatchListener*> listeners_;
    State state_ = State::FirstInnings;
    std::string result_;
};

struct ScorecardPrinter : MatchListener {
    bool dirty = false;
    void onBall(const Innings&, const Ball&) override { dirty = true; }

        for (size_t i = 0; i < in.bowlOrder.size(); ++i) {
            const BowlingStats& s = in.bowl[i];
            std::printf("  %s %d.%d ov %d runs %d wkts econ %.2f |", in.bowling.order[in.bowlOrder[i]].c_str(),
                        s.legalBalls / 6, s.legalBalls % 6, s.runs, s.wickets, s.economy());
        }
        std::printf("\n");
    }
    void onMatchEnd(const std::string& r) override { std::printf("\nResult: %s\n", r.c_str()); }
    void print(const Innings& in) {
        dirty = false;
        std::printf("\nScorecard for %s:\n%-8s%6s%4s%4s%7s%8s\n", in.batting.name.c_str(), "Player", "Score", "4s", "6s", "Balls", "SR");
        for (size_t i = 0; i < in.bat.size(); ++i) {
            const BattingStats& s = in.bat[i];
            std::string n = in.batting.order[i] + (in.atCrease(static_cast<int>(i)) ? "*" : "");
            std::printf("%-8s%6d%4d%4d%7d%8.1f\n", n.c_str(), s.runs, s.fours, s.sixes, s.balls, s.strikeRate());
        }
        std::printf("Total: %d/%d  Extras: %d\nOvers: %s\n", in.runs(), in.wickets(), in.extras(), in.overs().c_str());
    }
};

static void over(Match& m, const std::string& bowler, const std::string& balls) {
    m.startOver(bowler);
    std::istringstream in(balls);
    std::string t;
    while (in >> t && m.state() != Match::State::Finished) m.bowl(t);
}

static void tests() {

    Innings& first = m.current();
    assert(first.runs() == 7 && first.striker() == 0);
    over(m, "P9", "W 4 4 Wd W 1 6");
    assert(first.runs() == 23 && first.wickets() == 2 && first.legalBalls() == 12 && first.bat[1].runs == 10);

In the C++ version MatchListener is a struct of virtual functions with empty default bodies: a vtable where unused slots point at no-op stubs, the same trick as a file_operations table where a driver leaves .mmap NULL and the core falls back to a default. The match stores raw MatchListener* pointers it does not own, so the caller controls the listeners' lifetime, exactly like a notifier block that a module must unregister before it unloads.
