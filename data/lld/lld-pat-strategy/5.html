Code: parking fees (Java), bill splitting (Python), congestion control (C++14)

All three programs were compiled and run; the output under each is real. Java shows a strategy interface, three strategies (one wraps another), a registry, a runtime swap and Comparator. Python shows that a plain function can be a strategy. The C++14 tab is a toy of Linux congestion control so you can map it onto tcp_congestion_ops. Files live in lld-code/lld-pat-strategy/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Java interface: a struct of function pointers with no data. A class that implements it must fill every slot.
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • Method reference Foo::new, obj::method: shorthand for a lambda that just calls that constructor or method.
  • Supplier<T>, Consumer<T>, Function<A,B>: ready-made one-slot interfaces: "returns a T", "takes a T", "maps A to B". A lambda can fill any of them.
  • Map.of(k1, v1, ...), List.of(...): build a small read-only map or list in one expression.
  • Anonymous class new Foo() { ... }: defines and creates a one-off implementation in place, like a static ops table filled inline.
  • static { ... }: a block that runs once when the class is loaded, like a module init function.
  • Comparator.comparingLong(f).reversed(): builds a compare function (like the one you pass to qsort) from a key extractor, then flips its order.
  • Python Protocol: an interface checked by shape: any class with the right method names fits, without inheriting from it.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • std::unique_ptr<T>: an owning pointer that frees the object when it goes out of scope (no manual free); std::move hands that ownership to someone else.
  • std::function<R(Args)>: holds any callable: a plain function pointer, a lambda with captured state, or an object with operator().

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;
import java.util.function.*;

public class StrategyDemo {

    enum VehicleType { BIKE, CAR, TRUCK }

    record Stay(VehicleType type, int minutes, boolean weekend) { }

    // ---- Strategy: one job ("how much does this stay cost?"), many ways to do it ----
    interface FeeStrategy {
        String name();
        long feeRupees(Stay s);
    }

    // Concrete strategy 1: per started hour, rate depends on the vehicle
    static final class HourlyFee implements FeeStrategy {
        private final Map<VehicleType, Long> ratePerHour;
        HourlyFee(Map<VehicleType, Long> ratePerHour) { this.ratePerHour = ratePerHour; }
        public String name() { return "hourly"; }
        public long feeRupees(Stay s) {
            long hours = Math.max(1, (s.minutes() + 59) / 60);         // round up, minimum 1 hour
            return hours * ratePerHour.get(s.type());
        }
    }

    // Concrete strategy 2: first 2 hours flat, then hourly
    static final class FirstHoursFlatFee implements FeeStrategy {
        public String name() { return "flat-2h-then-hourly"; }
        public long feeRupees(Stay s) {
            long hours = Math.max(1, (s.minutes() + 59) / 60);
            return 50 + Math.max(0, hours - 2) * 30;
        }
    }

    // Concrete strategy 3: a decorator-like strategy that wraps another one
    static final class WeekendSurcharge implements FeeStrategy {
        private final FeeStrategy base;
        WeekendSurcharge(FeeStrategy base) { this.base = base; }
        public String name() { return base.name() + "+weekend"; }
        public long feeRupees(Stay s) {
            long fee = base.feeRupees(s);
            return s.weekend() ? fee + fee / 4 : fee;                  // +25% on weekends
        }
    }

    // ---- Context: knows WHEN to charge, not HOW the price is computed ----
    static final class ExitGate {
        private FeeStrategy pricing;
        ExitGate(FeeStrategy pricing) { this.pricing = Objects.requireNonNull(pricing); }
        void setPricing(FeeStrategy p) { this.pricing = Objects.requireNonNull(p); }  // swap at runtime
        String checkout(String ticket, Stay s) {
            long fee = pricing.feeRupees(s);                           // the one delegated call
            return String.format("%-6s %-5s %4d min  %-28s Rs %d", ticket, s.type(), s.minutes(), pricing.name(), fee);
        }
    }

    // ---- Registry: picks a strategy by config key, so callers do not need "new" ----
    static final Map<String, Supplier<FeeStrategy>> REGISTRY = new LinkedHashMap<>();
    static {
        REGISTRY.put("hourly", () -> new HourlyFee(Map.of(VehicleType.BIKE, 10L, VehicleType.CAR, 40L, VehicleType.TRUCK, 100L)));
        REGISTRY.put("mall", FirstHoursFlatFee::new);
        REGISTRY.put("mall-weekend", () -> new WeekendSurcharge(new FirstHoursFlatFee()));
    }
    static FeeStrategy pricingFor(String key) {
        Supplier<FeeStrategy> s = REGISTRY.get(key);
        if (s == null) throw new IllegalArgumentException("unknown pricing: " + key);
        return s.get();
    }

    public static void main(String[] args) {
        List<Stay> stays = List.of(
                new Stay(VehicleType.CAR, 45, false),
                new Stay(VehicleType.CAR, 200, true),
                new Stay(VehicleType.TRUCK, 130, false));

        for (String key : REGISTRY.keySet()) {
            ExitGate gate = new ExitGate(pricingFor(key));
            int i = 1;
            for (Stay s : stays) System.out.println(gate.checkout("T-" + i++, s));
        }

        // A one-off strategy needs no named class (with a single-method interface it could be a lambda).
        FeeStrategy festival = new FeeStrategy() {
            public String name() { return "festival-free"; }
            public long feeRupees(Stay s) { return 0; }
        };
        ExitGate gate = new ExitGate(pricingFor("hourly"));
        gate.setPricing(festival);                                     // runtime swap, no if/else
        System.out.println(gate.checkout("T-9", stays.get(1)));

        // The JDK's most famous strategy: Comparator passed to sort()
        List<Stay> byCost = new ArrayList<>(stays);
        FeeStrategy hourly = pricingFor("hourly");
        byCost.sort(Comparator.comparingLong(hourly::feeRupees).reversed());
        System.out.println("Most expensive first (hourly): " + byCost.stream().map(s -> s.type() + "/" + s.minutes()).toList());
    }
}

Look at the T-2 row of each block (the 200-minute car): Rs 160, Rs 110 or Rs 137, depending only on the strategy object. ExitGate.checkout was written once. WeekendSurcharge wraps another strategy, so "weekend" is combined with any base tariff instead of being copied into each.

"""Strategy in Python: split-expense strategies (Splitwise style).

Two flavours: classes behind a Protocol, and plain functions (a function IS a strategy).
"""
from typing import Callable, Dict, List, Protocol


class SplitStrategy(Protocol):
    def split(self, amount: int, people: List[str]) -> Dict[str, int]: ...


class EqualSplit:
    def split(self, amount, people):
        share, extra = divmod(amount, len(people))
        # the first `extra` people pay one more paisa so the total is exact
        return {p: share + (1 if i < extra else 0) for i, p in enumerate(people)}


class PercentSplit:
    def __init__(self, percents: Dict[str, int]):
        if sum(percents.values()) != 100:
            raise ValueError("percentages must add up to 100")
        self.percents = percents

    def split(self, amount, people):
        out = {p: amount * self.percents[p] // 100 for p in people}
        out[people[0]] += amount - sum(out.values())   # rounding remainder to the payer
        return out


class Expense:                                        # the Context
    def __init__(self, paid_by: str, amount: int, people: List[str], strategy: SplitStrategy):
        self.paid_by, self.amount, self.people, self.strategy = paid_by, amount, people, strategy

    def shares(self) -> Dict[str, int]:
        return self.strategy.split(self.amount, self.people)


def exact(shares: Dict[str, int]) -> Callable:
    """A strategy built from a closure: no class needed."""
    def split(amount, people):
        if sum(shares.values()) != amount:
            raise ValueError("exact shares must add up to the amount")
        return dict(shares)
    return split


class FunctionStrategy:                               # adapts a plain function to the interface
    def __init__(self, fn):
        self.split = fn


if __name__ == "__main__":
    people = ["asha", "ravi", "meera"]
    for strategy in (EqualSplit(),
                     PercentSplit({"asha": 50, "ravi": 30, "meera": 20}),
                     FunctionStrategy(exact({"asha": 400, "ravi": 300, "meera": 300}))):
        e = Expense("asha", 1000, people, strategy)
        print(f"{type(strategy).__name__:<16} {e.shares()}")

    # The stdlib's everyday strategy: the key function passed to sorted()
    bills = [("dinner", 1000), ("cab", 350), ("movie", 900)]
    print(sorted(bills, key=lambda b: b[1], reverse=True))

In Python, any callable with the right signature is a strategy, so a whole class is often too much. exact() returns a closure, and the key= argument of sorted() is the standard library's everyday strategy. Classes still help when a strategy has configuration and validation (PercentSplit checks that percentages add to 100).

// Strategy in C++14: a toy of Linux TCP congestion control.
// Each algorithm is a strategy object; the socket (context) calls it on ACK and on loss.
// Build: g++ -std=c++14 -Wall strategy.cpp -o strategy
#include <algorithm>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <string>

// Strategy interface: like struct tcp_congestion_ops (name, cong_avoid, ssthresh ...)
class CongestionControl {
public:
    virtual ~CongestionControl() = default;
    virtual const char* name() const = 0;
    virtual void onAck(int& cwnd, int ssthresh) = 0;      // grow the window
    virtual int onLoss(int cwnd) = 0;                      // return the new ssthresh
};

class Reno : public CongestionControl {                   // slow start, then +1 per RTT
    int acked_ = 0;
public:
    const char* name() const override { return "reno"; }
    void onAck(int& cwnd, int ssthresh) override {
        if (cwnd < ssthresh) { cwnd += 1; return; }        // slow start: doubles per RTT
        if (++acked_ >= cwnd) { cwnd += 1; acked_ = 0; }   // congestion avoidance
    }
    int onLoss(int cwnd) override { return std::max(cwnd / 2, 2); }
};

class Gentle : public CongestionControl {                 // made-up: never slow starts, small backoff
public:
    const char* name() const override { return "gentle"; }
    void onAck(int& cwnd, int) override { cwnd += 1; }
    int onLoss(int cwnd) override { return std::max(cwnd * 3 / 4, 2); }
};

// Registry: like tcp_register_congestion_control() + a lookup by name
std::map<std::string, std::function<std::unique_ptr<CongestionControl>()>>& registry() {
    static std::map<std::string, std::function<std::unique_ptr<CongestionControl>()>> r;
    return r;
}
bool registerCC(const std::string& n, std::function<std::unique_ptr<CongestionControl>()> f) {
    return registry().emplace(n, std::move(f)).second;
}

// Context: the socket owns the state (cwnd, ssthresh) and the algorithm choice.
class Socket {
    std::unique_ptr<CongestionControl> cc_;
    int cwnd_ = 1, ssthresh_ = 8;
public:
    explicit Socket(const std::string& algo) { setCongestion(algo); }
    bool setCongestion(const std::string& algo) {          // like setsockopt(TCP_CONGESTION)
        auto it = registry().find(algo);
        if (it == registry().end()) return false;          // like -ENOENT
        cc_ = it->second();
        return true;
    }
    void ack() { cc_->onAck(cwnd_, ssthresh_); }
    void loss() { ssthresh_ = cc_->onLoss(cwnd_); cwnd_ = ssthresh_; }
    std::string trace(const char* ev) const {
        return std::string(cc_->name()) + " " + ev + " cwnd=" + std::to_string(cwnd_) +
               " ssthresh=" + std::to_string(ssthresh_);
    }
};

int main() {
    registerCC("reno", [] { return std::unique_ptr<CongestionControl>(new Reno); });
    registerCC("gentle", [] { return std::unique_ptr<CongestionControl>(new Gentle); });

    for (const char* algo : {"reno", "gentle"}) {
        Socket s(algo);
        for (int i = 0; i < 10; ++i) s.ack();
        std::cout << s.trace("after 10 acks:") << "\n";
        s.loss();
        std::cout << s.trace("after loss:   ") << "\n";
    }
    Socket s("reno");
    std::cout << "set bbr: " << (s.setCongestion("bbr") ? "ok" : "ENOENT (not registered)") << "\n";
    return 0;
}

The socket owns cwnd and ssthresh; the algorithm only decides how they change. That split of ownership (context holds state, strategy holds the rule) is the same in the kernel, where congestion state lives in struct tcp_sock and the per-algorithm private data sits in a small area the algorithm can use.
