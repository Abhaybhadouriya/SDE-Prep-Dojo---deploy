"""python -m unittest -v test_chess"""
import unittest

from chess import Board, Color, Game, GameStatus, Player, Position

def game(*rows, to_move=Color.WHITE):
    w, b = Player("W", Color.WHITE), Player("B", Color.BLACK)
    return Game(w, b, Board.from_rows(*rows) if rows else None, to_move)

class ChessTest(unittest.TestCase):
    def test_start_position_has_20_moves(self):
        self.assertEqual(len(game().all_legal_moves()), 20)

    def test_knight_jumps_over_pieces(self):
        targets = sorted(str(m.to) for m in game().legal_moves(Position.of("g1")))
        self.assertEqual(targets, ["f3", "h3"])

    def test_cannot_castle_through_check(self):
        ok = game("r...k...", "........", "........", "........",
                  "........", "........", "........", "....K..R")   # rook on a8 does not touch f1/g1
        bad = game("....kr..", "........", "........", "........",
                   "........", "........", "........", "....K..R")  # rook on f8 attacks f1
        self.assertIn("g1", [str(m.to) for m in ok.legal_moves(Position.of("e1"))])
        self.assertNotIn("g1", [str(m.to) for m in bad.legal_moves(Position.of("e1"))])

    def test_undo_restores_everything(self):
        g = game()
        before = g.board.render()
        for m in "e2e4 d7d5 e4d5 d8d5 b1c3".split():
            g.move(m)
        for _ in range(5):
            g.undo()
        self.assertEqual(g.board.render(), before)
        self.assertEqual(len(g.all_legal_moves()), 20)

    def test_fools_mate(self):
        g = game()
        for m in "f2f3 e7e5 g2g4 d8h4".split():
            g.move(m)
        self.assertIs(g.status, GameStatus.CHECKMATE)
        self.assertEqual(g.winner.name, "B")

    def test_resign(self):
        g = game()
        g.resign(Color.WHITE)
        self.assertIs(g.status, GameStatus.RESIGNED)
        with self.assertRaises(RuntimeError):
            g.move("e2e4")

    def test_underpromotion_to_knight(self):
        g = game(".......k", "P.......", "........", "........",
                 "........", "........", "........", "....K...")
        g.move("a7a8n")
        self.assertEqual(g.board.at(Position.of("a8")).symbol(), "N")

if __name__ == "__main__":
    unittest.main()
