Zepto asked for APIs and a database schema too. A thin REST layer over the same domain:

EndpointBody / responseNotes
POST /games{whiteId, blackId} -> {gameId}creates the start position
GET /games/{id}-> {fen, turn, status, moves[]}FEN is the standard one-line position text
POST /games/{id}/moves{uci, expectedPly} -> {ply, status}409 if expectedPly is stale (two tabs, double click)
POST /games/{id}/takeback-> {ply}usually needs the opponent's consent in online play
POST /games/{id}/resign{playerId}idempotent
 
CREATE TABLE players (id BIGINT PRIMARY KEY, name VARCHAR(64) NOT NULL, rating INT DEFAULT 1200);
CREATE TABLE games (
  id BIGINT PRIMARY KEY, white_id BIGINT REFERENCES players(id), black_id BIGINT REFERENCES players(id),
  status VARCHAR(16) NOT NULL, turn CHAR(1) NOT NULL, fen VARCHAR(100) NOT NULL,
  ply INT NOT NULL DEFAULT 0,          -- optimistic-lock version: UPDATE ... WHERE ply = :expected
  winner_id BIGINT NULL, created_at TIMESTAMP, updated_at TIMESTAMP);
CREATE TABLE moves (
  game_id BIGINT REFERENCES games(id), ply INT, uci VARCHAR(5) NOT NULL, fen_after VARCHAR(100),
  created_at TIMESTAMP, PRIMARY KEY (game_id, ply));
