import java.io.*;
import java.util.*;

public class NullObjectDemo {

    // ---- 1. A logger that may be "switched off" ----
    interface Logger { void log(String msg); }

    static final class ConsoleLogger implements Logger {
        public void log(String msg) { System.out.println("  [log] " + msg); }
    }
    // Null Object: same interface, does nothing, stateless -> one shared instance is enough.
    static final class NullLogger implements Logger {
        // <-- the pattern: a do-nothing implementation, shared
        static final NullLogger INSTANCE = new NullLogger();
        private NullLogger() { }
        public void log(String msg) { /* intentionally nothing */ }
    }

    // ---- 2. A discount that may not exist ----
    interface DiscountPolicy { long discountPaise(long subtotalPaise); String label(); }

    record PercentOff(int percent) implements DiscountPolicy {
        public long discountPaise(long s) { return s * percent / 100; }
        public String label() { return percent + "% off"; }
    }
    enum NoDiscount implements DiscountPolicy {          // enum = a ready-made singleton
        INSTANCE;
        public long discountPaise(long s) { return 0; }  // the neutral value
        public String label() { return "no discount"; }
    }

    // ---- 3. A customer who may be unknown ----
    interface Customer { String greeting(); DiscountPolicy discount(); boolean isGuest(); }

    record Member(String name, DiscountPolicy discount) implements Customer {
        public String greeting() { return "Welcome back, " + name; }
        public boolean isGuest() { return false; }
    }
    static final class GuestCustomer implements Customer {   // "special case" for "not logged in"
        static final GuestCustomer INSTANCE = new GuestCustomer();
        public String greeting() { return "Hello, guest"; }
        public DiscountPolicy discount() { return NoDiscount.INSTANCE; }
        public boolean isGuest() { return true; }               // kept for the few places that must know
    }

    static final class CustomerDirectory {
        private final Map<String, Customer> byId = new HashMap<>();
        void add(String id, Customer c) { byId.put(id, c); }
        Customer findOrGuest(String id) {                        // never returns null
            return id == null ? GuestCustomer.INSTANCE : byId.getOrDefault(id, GuestCustomer.INSTANCE);
        }
        Optional<Customer> findForRefund(String id) {            // absence is an ERROR here: make caller decide
            return Optional.ofNullable(id == null ? null : byId.get(id));
        }
    }

    // ---- Client: no null checks anywhere ----
    static final class Checkout {
        private final Logger log;
        private final CustomerDirectory dir;
        Checkout(Logger log, CustomerDirectory dir) { this.log = log; this.dir = dir; }

        long total(String customerId, long subtotalPaise) {
            Customer c = dir.findOrGuest(customerId);
            long off = c.discount().discountPaise(subtotalPaise);
            log.log(c.greeting() + " | " + c.discount().label() + " | -" + off);
            return subtotalPaise - off;
        }
    }

    public static void main(String[] args) throws IOException {
        CustomerDirectory dir = new CustomerDirectory();
        dir.add("C1", new Member("Asha", new PercentOff(10)));
        dir.add("C2", new Member("Ravi", NoDiscount.INSTANCE));

        Checkout verbose = new Checkout(new ConsoleLogger(), dir);
        System.out.println("C1    total=" + verbose.total("C1", 50_000));
        System.out.println("C2    total=" + verbose.total("C2", 50_000));
        System.out.println("guest total=" + verbose.total(null, 50_000));
        System.out.println("C9    total=" + verbose.total("C9", 50_000));

        Checkout quiet = new Checkout(NullLogger.INSTANCE, dir);   // logging off, zero if-statements
        System.out.println("quiet total=" + quiet.total("C1", 20_000));

        // When absence must not be silent, Optional forces a decision:
        try {
            dir.findForRefund("C9").orElseThrow(() -> new IllegalArgumentException("unknown customer C9"));
        } catch (IllegalArgumentException e) { System.out.println("refund: " + e.getMessage()); }

        // JDK null objects:
        List<String> none = Collections.emptyList();
        for (String s : none) System.out.println(s);               // loop runs zero times, no NPE
        try (Writer w = Writer.nullWriter(); OutputStream os = OutputStream.nullOutputStream()) {
            w.write("dropped"); os.write(42);                       // like writing to /dev/null
        }
        System.out.println("JDK: emptyList size=" + none.size() + ", nullWriter/nullOutputStream ok");
    }
}

Look at C9: an unknown id became a guest at checkout (fine: anyone may buy), but the same unknown id is an error for a refund (not fine: money must go to a real account). Same directory, two methods, two different answers to "what does absence mean". That is the judgement this pattern needs.

"""Null Object: do-nothing implementations instead of None checks."""
import contextlib
import logging
from dataclasses import dataclass
from typing import Protocol


class Notifier(Protocol):
    def send(self, user: str, text: str) -> None: ...


class SmsNotifier:
    def send(self, user, text):
        print(f"  [sms to {user}] {text}")


class NullNotifier:
    """Same interface, does nothing. Used when the user opted out."""
    def send(self, user, text):
        pass

    def __bool__(self):          # optional: lets the rare caller ask "is anything there?"
        return False


class Discount(Protocol):
    def off(self, subtotal: int) -> int: ...


@dataclass(frozen=True)
class PercentOff:
    percent: int

    def off(self, subtotal):
        return subtotal * self.percent // 100


class NoDiscount:
    def off(self, subtotal):
        return 0


NO_DISCOUNT = NoDiscount()       # one shared instance (stateless)
NULL_NOTIFIER = NullNotifier()


@dataclass
class Order:
    user: str
    subtotal: int
    discount: Discount = NO_DISCOUNT     # default is the null object, never None
    notifier: Notifier = NULL_NOTIFIER

    def place(self):
        total = self.subtotal - self.discount.off(self.subtotal)
        self.notifier.send(self.user, f"order placed, pay {total}")
        return total


def maybe_lock(lock=None):
    # stdlib null object: nullcontext() is a context manager that does nothing
    return lock if lock is not None else contextlib.nullcontext()


if __name__ == "__main__":
    print("asha:", Order("asha", 50_000, PercentOff(10), SmsNotifier()).place())
    print("ravi:", Order("ravi", 50_000).place())               # no discount, no SMS, no ifs

    with maybe_lock():
        print("ran inside nullcontext (no lock needed)")

    # Libraries add NullHandler so "no logging configured" stays silent and safe.
    lib_log = logging.getLogger("mylib")
    lib_log.addHandler(logging.NullHandler())
    lib_log.propagate = False
    lib_log.warning("this goes nowhere")
    print("logging.NullHandler swallowed the warning")
    print("bool(NULL_NOTIFIER) =", bool(NULL_NOTIFIER))

Using a null object as the default argument (discount: Discount = NO_DISCOUNT) means None never enters the object. It is safe here because the null objects are immutable and stateless; never use a mutable default.

// Null Object in C++14: a do-nothing ops implementation instead of NULL function pointers.
// Build: g++ -std=c++14 -Wall null_object.cpp -o null_object
#include <cstdio>
#include <iostream>
#include <string>

// ---- C style: NULL means "not provided", so every caller must check ----
struct packet_hooks {
    void (*on_rx)(const char* pkt);   // may be NULL
};
static void count_rx(const char* pkt) { std::printf("  [c hook] rx %s\n", pkt); }

static void deliver_c(const packet_hooks* h, const char* pkt) {
    if (h && h->on_rx) h->on_rx(pkt);  // the check every call site must remember
}

// ---- C++ with a Null Object: always a valid object, no checks ----
class RxObserver {
public:
    virtual ~RxObserver() = default;
    virtual void onRx(const std::string& pkt) = 0;
};
class CountingObserver : public RxObserver {
    int n_ = 0;
public:
    void onRx(const std::string& pkt) override { std::cout << "  [counter] rx #" << ++n_ << " " << pkt << "\n"; }
};
class NullObserver : public RxObserver {
public:
    void onRx(const std::string&) override {}                  // does nothing, safely
    static NullObserver& instance() { static NullObserver n; return n; }
};

class NetDevice {
    std::string name_;
    RxObserver* obs_;                                           // never null by construction
public:
    explicit NetDevice(std::string name, RxObserver& obs = NullObserver::instance())
        : name_(std::move(name)), obs_(&obs) {}
    void setObserver(RxObserver& o) { obs_ = &o; }
    void clearObserver() { obs_ = &NullObserver::instance(); }  // "remove" = put the null object back
    void receive(const std::string& pkt) {
        obs_->onRx(pkt);                                         // no if
        std::cout << name_ << " delivered " << pkt << "\n";
    }
};

int main() {
    packet_hooks none = {nullptr}, some = {count_rx};
    deliver_c(&none, "p0");
    deliver_c(&some, "p1");

    CountingObserver counter;
    NetDevice eth0("eth0");                // default: null observer
    eth0.receive("p2");
    eth0.setObserver(counter);
    eth0.receive("p3");
    eth0.receive("p4");
    eth0.clearObserver();
    eth0.receive("p5");
    return 0;
}

"Remove the observer" puts the null object back instead of setting the pointer to nullptr, so the invariant "obs_ always points at something valid" holds for the whole life of the device. A function-local static gives one shared null instance, created on first use.
