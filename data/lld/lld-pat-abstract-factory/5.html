Code: a runnable payment-gateway family

Each tab is a complete program. They print the same thing. Sources live in lld-code/lld-pat-abstract-factory/. The gateways are made up; their "signatures" are toy checks, not real crypto.

Map before you read: which class plays which role in the Java tab (GatewayDemo.java). The Python and C++ tabs use the same roles under their own names. Key lines carry a // <-- the pattern comment.

// Abstract Factory demo: one factory object hands out a matching FAMILY of products.
// Run: javac GatewayDemo.java && java GatewayDemo
import java.util.Map;
import java.util.function.Supplier;

// ---- abstract products (one interface per KIND of product) ----
interface Charger         { String charge(String orderId, long paise); }
interface Refunder        { String refund(String chargeId, long paise); }
interface WebhookVerifier { boolean verify(String body, String signature); }

// ---- the abstract factory: one method per kind ----
interface GatewayFactory {
    Charger charger();
    Refunder refunder();
    WebhookVerifier verifier();
}

// ---- family 1: AcmePay (fictional) ----
final class AcmeCharger implements Charger {
    public String charge(String orderId, long paise) { return "acme_ch_" + orderId + " (" + paise + " paise)"; }
}
final class AcmeRefunder implements Refunder {
    public String refund(String chargeId, long paise) { return "acme_rf_for_" + chargeId; }
}
final class AcmeVerifier implements WebhookVerifier {
    public boolean verify(String body, String sig) { return sig.equals("acme:" + body.length()); }
}
final class AcmePayFactory implements GatewayFactory {
    public Charger charger()          { return new AcmeCharger(); }
    public Refunder refunder()        { return new AcmeRefunder(); }
    public WebhookVerifier verifier() { return new AcmeVerifier(); }
}

// ---- family 2: ZenPay (fictional) ----
final class ZenCharger implements Charger {
    public String charge(String orderId, long paise) { return "zen-" + orderId + "-" + paise; }
}
final class ZenRefunder implements Refunder {
    public String refund(String chargeId, long paise) { return "zen-refund-" + chargeId; }
}
final class ZenVerifier implements WebhookVerifier {
    public boolean verify(String body, String sig) { return sig.equals("zen#" + Integer.toHexString(body.hashCode())); }
}
final class ZenPayFactory implements GatewayFactory {
    public Charger charger()          { return new ZenCharger(); }
    public Refunder refunder()        { return new ZenRefunder(); }
    public WebhookVerifier verifier() { return new ZenVerifier(); }
}

// ---- client: knows ONLY the interfaces ----
final class CheckoutService {
    private final Charger charger;
    private final Refunder refunder;
    private final WebhookVerifier verifier;

    CheckoutService(GatewayFactory f) {          // the whole family comes from one object
        // <-- the pattern: the whole family comes from one factory object
        this.charger = f.charger();
        this.refunder = f.refunder();
        this.verifier = f.verifier();
    }
    void run(String orderId, String webhookBody, String sig) {
        String ch = charger.charge(orderId, 49_900);
        System.out.println("  charged : " + ch);
        System.out.println("  webhook : " + (verifier.verify(webhookBody, sig) ? "valid" : "REJECTED"));
        System.out.println("  refunded: " + refunder.refund(ch, 49_900));
    }
}

public class GatewayDemo {
    // registry: adding a new FAMILY = one new factory class + one line here
    static final Map<String, Supplier<GatewayFactory>> FACTORIES = Map.of(
        "acme", AcmePayFactory::new,
        "zen",  ZenPayFactory::new);

    public static void main(String[] args) {
        String body = "{\"order\":\"A1\"}";
        for (String name : new String[]{"acme", "zen"}) {
            GatewayFactory f = FACTORIES.get(name).get();   // chosen once, from config
            System.out.println("gateway = " + name);
            String sig = name.equals("acme") ? "acme:" + body.length()
                                             : "zen#" + Integer.toHexString(body.hashCode());
            new CheckoutService(f).run("A1", body, sig);
        }
    }
}
"""Abstract Factory demo: one factory object hands out a matching FAMILY of products.
Run: python gateway_demo.py
"""
from abc import ABC, abstractmethod


# ---- abstract products ----
class Charger(ABC):
    @abstractmethod
    def charge(self, order_id: str, paise: int) -> str: ...


class Refunder(ABC):
    @abstractmethod
    def refund(self, charge_id: str, paise: int) -> str: ...


class WebhookVerifier(ABC):
    @abstractmethod
    def verify(self, body: str, signature: str) -> bool: ...


# ---- the abstract factory ----
class GatewayFactory(ABC):
    @abstractmethod
    def charger(self) -> Charger: ...
    @abstractmethod
    def refunder(self) -> Refunder: ...
    @abstractmethod
    def verifier(self) -> WebhookVerifier: ...


# ---- family 1: AcmePay (fictional) ----
class AcmeCharger(Charger):
    def charge(self, order_id, paise):
        return f"acme_ch_{order_id} ({paise} paise)"


class AcmeRefunder(Refunder):
    def refund(self, charge_id, paise):
        return f"acme_rf_for_{charge_id}"


class AcmeVerifier(WebhookVerifier):
    def verify(self, body, signature):
        return signature == f"acme:{len(body)}"


class AcmePayFactory(GatewayFactory):
    def charger(self): return AcmeCharger()
    def refunder(self): return AcmeRefunder()
    def verifier(self): return AcmeVerifier()


# ---- family 2: ZenPay (fictional) ----
class ZenCharger(Charger):
    def charge(self, order_id, paise):
        return f"zen-{order_id}-{paise}"


class ZenRefunder(Refunder):
    def refund(self, charge_id, paise):
        return f"zen-refund-{charge_id}"


class ZenVerifier(WebhookVerifier):
    def verify(self, body, signature):
        return signature == f"zen#{sum(map(ord, body)):x}"


class ZenPayFactory(GatewayFactory):
    def charger(self): return ZenCharger()
    def refunder(self): return ZenRefunder()
    def verifier(self): return ZenVerifier()


# ---- client: knows only the abstract types ----
class CheckoutService:
    def __init__(self, factory: GatewayFactory):
        self._charger = factory.charger()
        self._refunder = factory.refunder()
        self._verifier = factory.verifier()

    def run(self, order_id, body, sig):
        ch = self._charger.charge(order_id, 49_900)
        print("  charged :", ch)
        print("  webhook :", "valid" if self._verifier.verify(body, sig) else "REJECTED")
        print("  refunded:", self._refunder.refund(ch, 49_900))


FACTORIES = {"acme": AcmePayFactory, "zen": ZenPayFactory}   # new family = one line

if __name__ == "__main__":
    body = '{"order":"A1"}'
    sigs = {"acme": f"acme:{len(body)}", "zen": f"zen#{sum(map(ord, body)):x}"}
    for name in ("acme", "zen"):
        print("gateway =", name)
        CheckoutService(FACTORIES[name]()).run("A1", body, sigs[name])
// Abstract Factory demo (C++14): one factory object hands out a matching FAMILY.
// Build: g++ -std=c++14 -Wall -o gateway_demo gateway_demo.cpp && ./gateway_demo
#include <iostream>
#include <map>
#include <memory>
#include <string>

struct Charger         { virtual ~Charger() = default;  virtual std::string charge(const std::string& id, long paise) = 0; };
struct Refunder        { virtual ~Refunder() = default; virtual std::string refund(const std::string& chargeId) = 0; };
struct WebhookVerifier { virtual ~WebhookVerifier() = default; virtual bool verify(const std::string& body, const std::string& sig) = 0; };

struct GatewayFactory {                       // like a struct of constructor function pointers
    virtual ~GatewayFactory() = default;
    virtual std::unique_ptr<Charger> charger() = 0;
    virtual std::unique_ptr<Refunder> refunder() = 0;
    virtual std::unique_ptr<WebhookVerifier> verifier() = 0;
};

// ---- family 1: AcmePay (fictional) ----
struct AcmeCharger : Charger {
    std::string charge(const std::string& id, long paise) override { return "acme_ch_" + id + " (" + std::to_string(paise) + " paise)"; }
};
struct AcmeRefunder : Refunder {
    std::string refund(const std::string& c) override { return "acme_rf_for_" + c; }
};
struct AcmeVerifier : WebhookVerifier {
    bool verify(const std::string& b, const std::string& s) override { return s == "acme:" + std::to_string(b.size()); }
};
struct AcmePayFactory : GatewayFactory {
    std::unique_ptr<Charger> charger() override { return std::make_unique<AcmeCharger>(); }
    std::unique_ptr<Refunder> refunder() override { return std::make_unique<AcmeRefunder>(); }
    std::unique_ptr<WebhookVerifier> verifier() override { return std::make_unique<AcmeVerifier>(); }
};

// ---- family 2: ZenPay (fictional) ----
struct ZenCharger : Charger {
    std::string charge(const std::string& id, long paise) override { return "zen-" + id + "-" + std::to_string(paise); }
};
struct ZenRefunder : Refunder {
    std::string refund(const std::string& c) override { return "zen-refund-" + c; }
};
struct ZenVerifier : WebhookVerifier {
    bool verify(const std::string& b, const std::string& s) override { return s == "zen#" + std::to_string(b.size() * 7); }
};
struct ZenPayFactory : GatewayFactory {
    std::unique_ptr<Charger> charger() override { return std::make_unique<ZenCharger>(); }
    std::unique_ptr<Refunder> refunder() override { return std::make_unique<ZenRefunder>(); }
    std::unique_ptr<WebhookVerifier> verifier() override { return std::make_unique<ZenVerifier>(); }
};

class CheckoutService {                        // client: only abstract types
    std::unique_ptr<Charger> charger_;
    std::unique_ptr<Refunder> refunder_;
    std::unique_ptr<WebhookVerifier> verifier_;
public:
    explicit CheckoutService(GatewayFactory& f)
        : charger_(f.charger()), refunder_(f.refunder()), verifier_(f.verifier()) {}
    void run(const std::string& id, const std::string& body, const std::string& sig) {
        std::string ch = charger_->charge(id, 49900);
        std::cout << "  charged : " << ch << "\n";
        std::cout << "  webhook : " << (verifier_->verify(body, sig) ? "valid" : "REJECTED") << "\n";
        std::cout << "  refunded: " << refunder_->refund(ch) << "\n";
    }
};

int main() {
    std::map<std::string, std::unique_ptr<GatewayFactory>> factories;
    factories["acme"] = std::make_unique<AcmePayFactory>();
    factories["zen"]  = std::make_unique<ZenPayFactory>();

    const std::string body = "{\"order\":\"A1\"}";
    std::map<std::string, std::string> sigs = {
        {"acme", "acme:" + std::to_string(body.size())},
        {"zen",  "zen#" + std::to_string(body.size() * 7)}};
    for (const char* name : {"acme", "zen"}) {
        std::cout << "gateway = " << name << "\n";
        CheckoutService svc(*factories[name]);
        svc.run("A1", body, sigs[name]);
    }
    return 0;
}

What to notice:

  • CheckoutService has no if on the vendor and imports no vendor class.
  • The map FACTORIES is the one place that knows names. Adding "PayNova" = write four classes (factory + 3 products) and add one map entry. Nothing else changes.
  • In Python the factory is often just a class or a module. Python's duck typing means the ABCs are optional, but they make the contract explicit and fail fast if a family forgets a method.
  • In C++ the factory returns std::unique_ptr so ownership of each product is clear. That is the C++ version of "the caller must free it".
