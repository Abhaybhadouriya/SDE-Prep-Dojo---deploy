Code: editor with undo, redo and macros (Java), remote control (Python), workqueue (C++14)

All three programs were compiled and run; the output under each one is real. Files live in lld-code/lld-pat-command/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Java interface: a struct of function pointers with no data. A class that implements it must fill every slot.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • Python ABC + @abstractmethod: a base class that refuses to be instantiated until a subclass fills the marked methods.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • std::unique_ptr<T>: an owning pointer that frees the object when it goes out of scope (no manual free); std::move hands that ownership to someone else.
  • std::function<R(Args)>: holds any callable: a plain function pointer, a lambda with captured state, or an object with operator().

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

/** Command pattern: a tiny text editor with undo, redo, macros and a deferred job queue.
 *  Build and run: javac -d out CommandDemo.java && java -cp out CommandDemo */
public class CommandDemo {

    // ---- Receiver: the object that really does the work. Knows nothing about commands. ----
    static final class Document {
        private final StringBuilder text = new StringBuilder();
        void insert(int pos, String s) { text.insert(pos, s); }
        String delete(int pos, int len) {                 // returns what was removed
            String gone = text.substring(pos, pos + len);
            text.delete(pos, pos + len);
            return gone;
        }
        int length() { return text.length(); }
        @Override public String toString() { return "\"" + text + "\""; }
    }

    // ---- Command interface: a request turned into an object ----
    interface Command {
        void execute();
        void undo();
        String name();
    }

    // ---- Concrete commands: each stores the receiver AND the arguments it needs ----
    static final class InsertText implements Command {
        private final Document doc; private final int pos; private final String s;
        InsertText(Document doc, int pos, String s) { this.doc = doc; this.pos = pos; this.s = s; }
        public void execute() { doc.insert(pos, s); }
        public void undo()    { doc.delete(pos, s.length()); }
        public String name()  { return "insert '" + s + "' @" + pos; }
    }

    static final class DeleteText implements Command {
        private final Document doc; private final int pos, len;
        private String removed;                           // state captured at execute time, needed for undo
        DeleteText(Document doc, int pos, int len) { this.doc = doc; this.pos = pos; this.len = len; }
        public void execute() { removed = doc.delete(pos, len); }
        public void undo()    { doc.insert(pos, removed); }
        public String name()  { return "delete " + len + " @" + pos; }
    }

    /** Composite command: many steps that undo as one. */
    static final class Macro implements Command {
        private final String label; private final List<Command> steps;
        Macro(String label, List<Command> steps) { this.label = label; this.steps = steps; }
        public void execute() { for (Command c : steps) c.execute(); }
        public void undo() {                              // reverse order!
            for (int i = steps.size() - 1; i >= 0; i--) steps.get(i).undo();
        }
        public String name() { return "macro " + label; }
    }

    // ---- Invoker: runs commands and keeps history. Never looks inside them. ----
    static final class Editor {
        private final Deque<Command> undoStack = new ArrayDeque<>();
        private final Deque<Command> redoStack = new ArrayDeque<>();
        void run(Command c) {
            c.execute();
            undoStack.push(c);
            redoStack.clear();                            // a new action kills the redo branch
            System.out.println("  do   " + c.name());
        }
        void undo() {
            if (undoStack.isEmpty()) { System.out.println("  undo (nothing)"); return; }
            Command c = undoStack.pop(); c.undo(); redoStack.push(c);
            System.out.println("  undo " + c.name());
        }
        void redo() {
            if (redoStack.isEmpty()) { System.out.println("  redo (nothing)"); return; }
            Command c = redoStack.pop(); c.execute(); undoStack.push(c);
            System.out.println("  redo " + c.name());
        }
    }

    public static void main(String[] args) {
        Document doc = new Document();
        Editor ed = new Editor();

        System.out.println("1) Undo and redo with two stacks:");
        ed.run(new InsertText(doc, 0, "hello"));
        ed.run(new InsertText(doc, 5, " world"));
        ed.run(new DeleteText(doc, 0, 6));
        System.out.println("  doc = " + doc);
        ed.undo();
        ed.undo();
        System.out.println("  doc = " + doc);
        ed.redo();
        System.out.println("  doc = " + doc);
        ed.run(new InsertText(doc, doc.length(), "!"));   // clears redo stack
        ed.redo();
        System.out.println("  doc = " + doc);

        System.out.println("2) A macro undoes as one step:");
        ed.run(new Macro("sign-off", List.of(
                new InsertText(doc, doc.length(), "\n--"),
                new InsertText(doc, doc.length() + 3, "\nRavi"))));
        System.out.println("  doc = " + doc.toString().replace("\n", "\\n"));
        ed.undo();
        System.out.println("  doc = " + doc);

        System.out.println("3) Commands queued now, run later (like a job queue):");
        Queue<Runnable> jobs = new ArrayDeque<>();        // Runnable is the JDK's command interface
        jobs.add(() -> System.out.println("  job: send email to ravi@example.com"));
        jobs.add(() -> System.out.println("  job: resize avatar 42.png"));
        System.out.println("  queued " + jobs.size() + " jobs, caller returns immediately");
        while (!jobs.isEmpty()) jobs.poll().run();        // the worker drains the queue
    }
}

Three things to notice. DeleteText saves the removed text at execute() time, because only then does it know what was there. Macro is a command made of commands (a Composite) and undoes its steps in reverse order. And part 3 uses Runnable: Java's built-in command interface, with a lambda as the concrete command.

"""Command pattern in Python: light switch remote with undo, and a job queue of callables.
Run: python command_demo.py
"""
from abc import ABC, abstractmethod
from collections import deque


class Light:                                  # Receiver
    def __init__(self, room):
        self.room, self.level = room, 0

    def set(self, level):
        self.level = level
        print(f"  {self.room} light -> {level}%")


class Command(ABC):
    @abstractmethod
    def execute(self): ...

    @abstractmethod
    def undo(self): ...


class SetBrightness(Command):                 # Concrete command
    def __init__(self, light, level):
        self.light, self.level, self.prev = light, level, None

    def execute(self):
        self.prev = self.light.level          # remember what undo needs
        self.light.set(self.level)

    def undo(self):
        self.light.set(self.prev)


class Remote:                                 # Invoker: buttons hold commands
    def __init__(self):
        self.buttons, self.history = {}, []

    def bind(self, button, command):
        self.buttons[button] = command

    def press(self, button):
        cmd = self.buttons[button]
        cmd.execute()
        self.history.append(cmd)

    def undo(self):
        if self.history:
            self.history.pop().undo()


if __name__ == "__main__":
    hall, bed = Light("hall"), Light("bedroom")
    r = Remote()
    r.bind("A", SetBrightness(hall, 100))
    r.bind("B", SetBrightness(bed, 30))
    r.bind("C", SetBrightness(hall, 0))       # same command class, different arguments
    print("press A, B, C:")
    for b in "ABC":
        r.press(b)
    print("undo, undo:")
    r.undo()
    r.undo()

    print("a queue of callables (Python's lightest command):")
    jobs = deque()
    jobs.append(lambda: print("  job: email ravi@example.com"))
    jobs.append(lambda: print("  job: resize 42.png"))
    while jobs:
        jobs.popleft()()

Python functions are already objects, so the lightest command is a plain callable (the last part). You need a class only when the command must carry extra behaviour such as undo() or a name for the history list.

// Command pattern in C++14, kernel flavour: a mini workqueue.
// A "work item" = function + context, queued now and run later by a worker.
// Build: g++ -std=c++14 -Wall workqueue.cpp -o workqueue
#include <deque>
#include <functional>
#include <iostream>
#include <memory>
#include <string>

// 1) C style, exactly the work_struct idea: a function pointer plus context.
struct work {
    void (*func)(struct work *w);     // what to do
    int irq;                           // context the deferred half needs
};

static void handle_rx(struct work *w) {
    std::cout << "  [C-style] bottom half: drain RX ring for irq " << w->irq << "\n";
}

// 2) Object style: Command interface with a virtual execute().
class Command {
public:
    virtual ~Command() = default;
    virtual void execute() = 0;
    virtual std::string name() const = 0;
};

class FlushStats : public Command {
    std::string dev_;
public:
    explicit FlushStats(std::string dev) : dev_(std::move(dev)) {}
    void execute() override { std::cout << "  flush stats for " << dev_ << "\n"; }
    std::string name() const override { return "flush-stats"; }
};

class LinkUp : public Command {
    std::string dev_; int speed_;
public:
    LinkUp(std::string dev, int speed) : dev_(std::move(dev)), speed_(speed) {}
    void execute() override { std::cout << "  " << dev_ << " link up at " << speed_ << " Mb/s\n"; }
    std::string name() const override { return "link-up"; }
};

// 3) Lambda style: std::function is the C++ "command object" for free.
class WorkQueue {                       // Invoker: it never knows what the work does
    std::deque<std::unique_ptr<Command>> q_;
    std::deque<std::function<void()>> fq_;
public:
    void queue(std::unique_ptr<Command> c) {
        std::cout << "queue_work(" << c->name() << ")\n";
        q_.push_back(std::move(c));
    }
    void queue(std::function<void()> f) { std::cout << "queue_work(lambda)\n"; fq_.push_back(std::move(f)); }
    void run_worker() {                 // like a kworker thread draining the queue
        std::cout << "worker wakes up, runs " << q_.size() + fq_.size() << " item(s):\n";
        while (!q_.empty()) { q_.front()->execute(); q_.pop_front(); }
        while (!fq_.empty()) { fq_.front()(); fq_.pop_front(); }
    }
};

int main() {
    struct work w = {handle_rx, 33};    // "interrupt" schedules work, returns fast
    std::cout << "top half: irq 33 acked, deferring\n";

    WorkQueue wq;
    wq.queue(std::unique_ptr<Command>(new LinkUp("eth0", 10000)));
    wq.queue(std::unique_ptr<Command>(new FlushStats("eth0")));
    wq.queue([&w] { w.func(&w); });     // the C-style item rides in the same queue
    wq.run_worker();
    return 0;
}

queue_work(link-up)
queue_work(flush-stats)
queue_work(lambda)
worker wakes up, runs 3 item(s):
  eth0 link up at 10000 Mb/s
  flush stats for eth0
  [C-style] bottom half: drain RX ring for irq 33

Three spellings of the same idea in one queue: a C struct with a function pointer (the work_struct shape), a class with a virtual execute(), and a std::function lambda. The worker treats all of them the same way: "run the next item".

A subtle bug to avoid. In the Python remote, each button holds one command object. If you press the same button twice, both history entries point at the same object, and the second press overwrites prev, so the first undo restores the wrong level. The fix: push a fresh copy of the command into history on every press (or let the button hold a factory that creates a new command each time). The Java editor avoids this because every run() receives a new object.
