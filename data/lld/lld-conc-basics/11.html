Python: the GIL does not make your code thread-safe

CPython's Global Interpreter Lock lets only one thread run Python bytecode at a time, which surprises people into thinking races cannot happen. They can. The GIL can switch threads between bytecodes, and it is released during I/O and sleeps. counter["n"] += 1 compiles to separate load, add and store bytecodes (BINARY_SUBSCR, BINARY_OP, STORE_SUBSCR), and any check-then-act that waits on I/O has a wide-open window:

"""The GIL does not make your code thread-safe: check-then-act still races."""
import threading
import time


class UnsafeShow:
    def __init__(self):
        self.seats_left = 1

    def book(self, user, results):
        if self.seats_left > 0:              # check
            time.sleep(0.01)                 # the GIL is released here (I/O, sleep, long C calls)
            self.seats_left -= 1             # act on a stale check
            results.append(user)


class SafeShow(UnsafeShow):
    def __init__(self):
        super().__init__()
        self._lock = threading.Lock()

    def book(self, user, results):
        with self._lock:                     # check + act under one lock
            super().book(user, results)


def run(show):
    results = []
    threads = [threading.Thread(target=show.book, args=(u, results)) for u in ("asha", "ravi")]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    return results, show.seats_left


if __name__ == "__main__":
    print("unsafe:", run(UnsafeShow()))
    print("safe  :", run(SafeShow()))

    # Read-modify-write on a shared counter: correct only because of the lock.
    counter = {"n": 0}
    lock = threading.Lock()

    def add(k):
        for _ in range(k):
            with lock:
                counter["n"] += 1            # += is LOAD, ADD, STORE bytecodes: a switch can land between them

    ts = [threading.Thread(target=add, args=(50_000,)) for _ in range(4)]
    for t in ts:
        t.start()
    for t in ts:
        t.join()
    print("locked counter:", counter["n"], "(expected 200000)")
$ python races.py
unsafe: (['asha', 'ravi'], -1)
safe  : (['asha'], 0)
locked counter: 200000 (expected 200000)
