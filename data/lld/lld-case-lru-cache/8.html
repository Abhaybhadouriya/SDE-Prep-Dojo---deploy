  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import java.util.concurrent.locks.*;

// ---------- eviction strategy ----------
interface EvictionPolicy<K> {
    void onPut(K key);          // a new key entered the cache
    void onAccess(K key);       // an existing key was read or overwritten
    void onRemove(K key);       // a key left for another reason (remove, expiry)
    K evict();                  // choose a victim, forget it, return it
}

/** LRU: hash map + hand-written doubly linked list. Head = most recent, tail = victim. All O(1). */
final class LruPolicy<K> implements EvictionPolicy<K> {
    private static final class Node<K> { K key; Node<K> prev, next; Node(K k) { key = k; } }
    private final Map<K, Node<K>> nodes = new HashMap<>();
    private final Node<K> head = new Node<>(null), tail = new Node<>(null);   // sentinels: no null checks
    LruPolicy() { head.next = tail; tail.prev = head; }

    private void unlink(Node<K> n) { n.prev.next = n.next; n.next.prev = n.prev; }
    private void pushFront(Node<K> n) { n.next = head.next; n.prev = head; head.next.prev = n; head.next = n; }

    public void onPut(K k) { Node<K> n = new Node<>(k); nodes.put(k, n); pushFront(n); }
    public void onAccess(K k) { Node<K> n = nodes.get(k); if (n != null) { unlink(n); pushFront(n); } }
    public void onRemove(K k) { Node<K> n = nodes.remove(k); if (n != null) unlink(n); }
    public K evict() {
        Node<K> n = tail.prev;
        if (n == head) throw new IllegalStateException("nothing to evict");
        unlink(n); nodes.remove(n.key);
        return n.key;
    }
}

/** LFU: freq per key + one insertion-ordered bucket per frequency + minFreq. Ties broken by LRU. */
final class LfuPolicy<K> implements EvictionPolicy<K> {
    private final Map<K, Integer> freq = new HashMap<>();
    private final Map<Integer, LinkedHashSet<K>> buckets = new HashMap<>();
    private int minFreq = 0;

    private void addTo(int f, K k) { buckets.computeIfAbsent(f, x -> new LinkedHashSet<>()).add(k); }
    private void dropFrom(int f, K k) {
        LinkedHashSet<K> b = buckets.get(f);
        b.remove(k);
        if (b.isEmpty()) buckets.remove(f);
    }
    public void onPut(K k) { freq.put(k, 1); addTo(1, k); minFreq = 1; }
    public void onAccess(K k) {
        Integer f = freq.get(k);
        if (f == null) return;
        dropFrom(f, k);
        if (f == minFreq && !buckets.containsKey(f)) minFreq = f + 1;
        freq.put(k, f + 1); addTo(f + 1, k);
    }
    public void onRemove(K k) {
        Integer f = freq.remove(k);
        if (f != null) dropFrom(f, k);          // minFreq may now be stale; evict() repairs it
    }
    public K evict() {
        if (freq.isEmpty()) throw new IllegalStateException("nothing to evict");
        if (!buckets.containsKey(minFreq)) minFreq = Collections.min(buckets.keySet());   // rare path after onRemove
        K victim = buckets.get(minFreq).iterator().next();   // oldest key at the lowest frequency
        dropFrom(minFreq, victim); freq.remove(victim);
        return victim;
    }
}

/** FIFO: evict in insertion order, ignore reads. Shows how cheap a new policy is. */
final class FifoPolicy<K> implements EvictionPolicy<K> {
    private final LinkedHashSet<K> order = new LinkedHashSet<>();
    public void onPut(K k) { order.add(k); }
    public void onAccess(K k) { }
    public void onRemove(K k) { order.remove(k); }
    public K evict() { Iterator<K> it = order.iterator(); K k = it.next(); it.remove(); return k; }
}

interface DataStore<K, V> { V load(K key); void save(K key, V value); void delete(K key); }

final class FakeDb<K, V> implements DataStore<K, V> {
    final Map<K, V> rows = new ConcurrentHashMap<>();
    final AtomicInteger loads = new AtomicInteger(), saves = new AtomicInteger();
    public V load(K k) { loads.incrementAndGet(); return rows.get(k); }
    public void save(K k, V v) { saves.incrementAndGet(); rows.put(k, v); }
    public void delete(K k) { rows.remove(k); }
}

enum WritePolicy { WRITE_THROUGH, WRITE_BACK }

// ---------- the cache ----------
final class Cache<K, V> implements DataStore<K, V> {        // a cache is also a store: that is how levels chain
    private static final class Entry<V> {
        V value; long expiresAt; boolean dirty;
        Entry(V v, long exp) { value = v; expiresAt = exp; }
    }
    private final int capacity;
    private final EvictionPolicy<K> policy;
    private final DataStore<K, V> store;                    // may be null: pure in-memory cache
    private final WritePolicy writePolicy;
    private final long ttlMillis;                           // 0 = never expire
    private final Clock clock;
    private final Map<K, Entry<V>> map = new HashMap<>();
    private final ReentrantLock lock = new ReentrantLock(); // even get() mutates LRU order: one lock, not a RW lock
    long hits, misses, evictions;

    Cache(int capacity, EvictionPolicy<K> policy, DataStore<K, V> store, WritePolicy wp, long ttlMillis, Clock clock) {
        if (capacity <= 0) throw new IllegalArgumentException("capacity must be > 0");
        this.capacity = capacity; this.policy = policy; this.store = store;
        this.writePolicy = wp; this.ttlMillis = ttlMillis; this.clock = clock;
    }
    static <K, V> Cache<K, V> inMemory(int capacity, EvictionPolicy<K> p) {
        return new Cache<>(capacity, p, null, WritePolicy.WRITE_THROUGH, 0, System::currentTimeMillis);
    }

    public V get(K key) {
        lock.lock();
        try {
            Entry<V> e = live(key);
            if (e != null) { hits++; policy.onAccess(key); return e.value; }
            misses++;
            if (store == null) return null;
        } finally { lock.unlock(); }
        V loaded = store.load(key);                         // slow I/O outside the lock
        if (loaded == null) return null;
        lock.lock();
        try {
            Entry<V> e = live(key);
            if (e != null) return e.value;                  // another thread filled it meanwhile: keep theirs
            insert(key, loaded, false);
            return loaded;
        } finally { lock.unlock(); }
    }

    public void put(K key, V value) {
        lock.lock();
        try {
            Entry<V> e = live(key);
            boolean back = writePolicy == WritePolicy.WRITE_BACK && store != null;
            if (e != null) {
                e.value = value; e.expiresAt = expiry(); e.dirty |= back;
                policy.onAccess(key);
            } else {
                insert(key, value, back);
            }
            if (!back && store != null) store.save(key, value);   // write-through: store before returning
        } finally { lock.unlock(); }
    }

    public void delete(K key) { remove(key); }
    public V load(K key) { return get(key); }
    public void save(K key, V value) { put(key, value); }

    public boolean remove(K key) {
        lock.lock();
        try {
            Entry<V> e = map.remove(key);
            if (e == null) return false;
            policy.onRemove(key);
            if (store != null) store.delete(key);
            return true;
        } finally { lock.unlock(); }
    }

    /** Write every dirty entry to the store (write-back). Call on shutdown or from a timer. */
    public void flush() {
        lock.lock();
        try {
            for (Map.Entry<K, Entry<V>> me : map.entrySet())
                if (me.getValue().dirty) { store.save(me.getKey(), me.getValue().value); me.getValue().dirty = false; }
        } finally { lock.unlock(); }
    }

    /** Active expiry: a background sweeper can call this; get() also expires lazily. */
    public int purgeExpired() {
        lock.lock();
        try {
            List<K> dead = new ArrayList<>();
            for (Map.Entry<K, Entry<V>> me : map.entrySet()) if (expired(me.getValue())) dead.add(me.getKey());
            for (K k : dead) drop(k);
            return dead.size();
        } finally { lock.unlock(); }
    }

    public int size() { lock.lock(); try { return map.size(); } finally { lock.unlock(); } }
    public String stats() { lock.lock(); try { return "hits=" + hits + " misses=" + misses + " evictions=" + evictions + " size=" + map.size(); } finally { lock.unlock(); } }

    // ----- helpers, called with the lock held -----
    private long expiry() { return ttlMillis == 0 ? Long.MAX_VALUE : clock.nowMillis() + ttlMillis; }
    private boolean expired(Entry<V> e) { return clock.nowMillis() >= e.expiresAt; }
    private Entry<V> live(K key) {
        Entry<V> e = map.get(key);
        if (e != null && expired(e)) { drop(key); return null; }   // lazy expiry
        return e;
    }
    private void drop(K key) {
        Entry<V> e = map.remove(key);
        policy.onRemove(key);
        if (e.dirty) store.save(key, e.value);              // never lose a write-back value
    }
    private void insert(K key, V value, boolean dirty) {
        if (map.size() >= capacity) {
            K victim = policy.evict();
            Entry<V> old = map.remove(victim);
            if (old.dirty) store.save(victim, old.value);
            evictions++;
        }
        Entry<V> e = new Entry<>(value, expiry());
        e.dirty = dirty;
        map.put(key, e);
        policy.onPut(key);
    }
}

    public static void main(String[] args) throws Exception {

        check(lru.get("b") == null && lru.get("a") == 1, "LRU evicts b");

        System.out.println("LFU  c evicted: " + (lfu.get("c") == null) + ", d evicted: " + (lfu.get("d") == null));
        check(lfu.get("c") == null && lfu.get("d") == null && lfu.get("a") == 1, "LFU evicts c then d");

        Cache<String, Integer> fifo = Cache.inMemory(3, new FifoPolicy<>());
        fifo.put("a", 1); fifo.put("b", 2); fifo.put("c", 3); fifo.get("a"); fifo.put("d", 4);

"""Cache with pluggable eviction (LRU / LFU / FIFO), TTL, read-through, write-through / write-back, levels."""
import threading
import time
from abc import ABC, abstractmethod
from collections import OrderedDict
from enum import Enum

# ---------- eviction strategy ----------
class EvictionPolicy(ABC):
    @abstractmethod
    def on_put(self, key): ...
    @abstractmethod
    def on_access(self, key): ...
    @abstractmethod
    def on_remove(self, key): ...
    @abstractmethod
    def evict(self): ...

class _Node:
    __slots__ = ("key", "prev", "next")

    def __init__(self, key=None):
        self.key, self.prev, self.next = key, None, None

class LruPolicy(EvictionPolicy):
    """Dict + hand-written doubly linked list with sentinels. head.next = most recent."""

    def __init__(self):
        self._nodes = {}
        self._head, self._tail = _Node(), _Node()
        self._head.next, self._tail.prev = self._tail, self._head

    def _unlink(self, n):
        n.prev.next, n.next.prev = n.next, n.prev

    def _push_front(self, n):
        n.next, n.prev = self._head.next, self._head
        self._head.next.prev = n
        self._head.next = n

    def on_put(self, key):
        n = _Node(key)
        self._nodes[key] = n
        self._push_front(n)

    def on_access(self, key):
        n = self._nodes.get(key)
        if n:
            self._unlink(n)
            self._push_front(n)

    def on_remove(self, key):
        n = self._nodes.pop(key, None)
        if n:
            self._unlink(n)

    def evict(self):
        n = self._tail.prev
        if n is self._head:
            raise LookupError("nothing to evict")
        self._unlink(n)
        del self._nodes[n.key]
        return n.key

class LfuPolicy(EvictionPolicy):
    """freq per key, one insertion-ordered bucket per frequency, min_freq. Ties broken by LRU."""

    def __init__(self):
        self._freq = {}
        self._buckets = {}          # freq -> OrderedDict used as an ordered set
        self._min = 0

    def _add(self, f, key):
        self._buckets.setdefault(f, OrderedDict())[key] = None

    def _drop(self, f, key):
        b = self._buckets[f]
        del b[key]
        if not b:
            del self._buckets[f]

    def on_put(self, key):
        self._freq[key] = 1
        self._add(1, key)
        self._min = 1

    def on_access(self, key):
        f = self._freq.get(key)
        if f is None:
            return
        self._drop(f, key)
        if f == self._min and f not in self._buckets:
            self._min = f + 1
        self._freq[key] = f + 1
        self._add(f + 1, key)

    def on_remove(self, key):
        f = self._freq.pop(key, None)
        if f is not None:
            self._drop(f, key)

    def evict(self):
        if not self._freq:
            raise LookupError("nothing to evict")
        if self._min not in self._buckets:          # rare: repair after on_remove
            self._min = min(self._buckets)
        victim = next(iter(self._buckets[self._min]))
        self._drop(self._min, victim)
        del self._freq[victim]
        return victim

class FifoPolicy(EvictionPolicy):
    def __init__(self):
        self._order = OrderedDict()

    def on_put(self, key): self._order[key] = None
    def on_access(self, key): pass
    def on_remove(self, key): self._order.pop(key, None)
    def evict(self): return self._order.popitem(last=False)[0]

# ---------- collaborators ----------
class ManualClock:
    def __init__(self): self.now = 0.0
    def __call__(self): return self.now
    def advance(self, s): self.now += s

class FakeDb:
    def __init__(self):
        self.rows, self.loads, self.saves = {}, 0, 0

    def load(self, k):
        self.loads += 1
        return self.rows.get(k)

    def save(self, k, v):
        self.saves += 1
        self.rows[k] = v

    def delete(self, k):
        self.rows.pop(k, None)

class WritePolicy(Enum):
    WRITE_THROUGH = 1
    WRITE_BACK = 2

class _Entry:
    __slots__ = ("value", "expires_at", "dirty")

    def __init__(self, value, expires_at, dirty=False):
        self.value, self.expires_at, self.dirty = value, expires_at, dirty

class Cache:
    """A cache is also a store (load/save/delete), so levels chain: L1 -> L2 -> db."""

    def __init__(self, capacity, policy, store=None, write_policy=WritePolicy.WRITE_THROUGH, ttl=0, clock=time.monotonic):
        if capacity <= 0:
            raise ValueError("capacity must be > 0")
        self.capacity, self.policy, self.store = capacity, policy, store
        self.write_policy, self.ttl, self.clock = write_policy, ttl, clock
        self._map = {}
        self._lock = threading.Lock()       # get() also mutates order: one lock, not a reader-writer lock
        self.hits = self.misses = self.evictions = 0

    def get(self, key):
        with self._lock:
            e = self._live(key)
            if e is not None:
                self.hits += 1
                self.policy.on_access(key)
                return e.value
            self.misses += 1
            if self.store is None:
                return None
        loaded = self.store.load(key)        # slow I/O outside the lock
        if loaded is None:
            return None
        with self._lock:
            e = self._live(key)
            if e is not None:
                return e.value
            self._insert(key, loaded, False)
            return loaded

    def put(self, key, value):
        with self._lock:
            back = self.write_policy is WritePolicy.WRITE_BACK and self.store is not None
            e = self._live(key)
            if e is not None:
                e.value, e.expires_at, e.dirty = value, self._expiry(), e.dirty or back
                self.policy.on_access(key)
            else:
                self._insert(key, value, back)
            if not back and self.store is not None:
                self.store.save(key, value)

    def remove(self, key):
        with self._lock:
            if self._map.pop(key, None) is None:
                return False
            self.policy.on_remove(key)
            if self.store is not None:
                self.store.delete(key)
            return True

    # store interface, so a Cache can sit behind another Cache
    load = get
    save = put
    delete = remove

    def flush(self):
        with self._lock:
            for k, e in self._map.items():
                if e.dirty:
                    self.store.save(k, e.value)
                    e.dirty = False

    def purge_expired(self):
        with self._lock:
            dead = [k for k, e in self._map.items() if self._expired(e)]
            for k in dead:
                self._drop(k)
            return len(dead)

    def __len__(self):
        with self._lock:
            return len(self._map)

    # ----- helpers, lock held -----
    def _expiry(self):
        return float("inf") if self.ttl == 0 else self.clock() + self.ttl

    def _expired(self, e):
        return self.clock() >= e.expires_at

    def _live(self, key):
        e = self._map.get(key)
        if e is not None and self._expired(e):
            self._drop(key)
            return None
        return e

    def _drop(self, key):
        e = self._map.pop(key)
        self.policy.on_remove(key)
        if e.dirty:
            self.store.save(key, e.value)

    def _insert(self, key, value, dirty):
        if len(self._map) >= self.capacity:
            victim = self.policy.evict()
            old = self._map.pop(victim)
            if old.dirty:
                self.store.save(victim, old.value)
            self.evictions += 1
        self._map[key] = _Entry(value, self._expiry(), dirty)
        self.policy.on_put(key)

// Cache with pluggable eviction - C++14 core (LRU, LFU, FIFO). TTL and write policies are in the Java/Python versions.
#include <cstdlib>
#include <iostream>
#include <list>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <unordered_map>

template <typename K>
struct EvictionPolicy {                       // the "ops table"
    virtual ~EvictionPolicy() {}
    virtual void onPut(const K& k) = 0;
    virtual void onAccess(const K& k) = 0;
    virtual void onRemove(const K& k) = 0;
    virtual K evict() = 0;
};

// LRU: std::list keeps recency order, the map points at list nodes; splice() moves a node in O(1)
template <typename K>
class LruPolicy : public EvictionPolicy<K> {
    std::list<K> order_;                                          // front = most recent
    std::unordered_map<K, typename std::list<K>::iterator> pos_;
public:
    void onPut(const K& k) override { order_.push_front(k); pos_[k] = order_.begin(); }
    void onAccess(const K& k) override {
        auto it = pos_.find(k);
        if (it != pos_.end()) order_.splice(order_.begin(), order_, it->second);
    }
    void onRemove(const K& k) override {
        auto it = pos_.find(k);
        if (it != pos_.end()) { order_.erase(it->second); pos_.erase(it); }
    }
    K evict() override {
        if (order_.empty()) throw std::logic_error("nothing to evict");
        K k = order_.back(); order_.pop_back(); pos_.erase(k); return k;
    }
};

// LFU: frequency buckets, each an LRU list; minFreq points at the bucket to evict from
template <typename K>
class LfuPolicy : public EvictionPolicy<K> {
    struct Info { int freq; typename std::list<K>::iterator it; };
    std::unordered_map<K, Info> info_;
    std::map<int, std::list<K>> buckets_;                         // freq -> keys, front = newest
    int minFreq_ = 0;
    void drop(const Info& in) {
        auto b = buckets_.find(in.freq);
        b->second.erase(in.it);
        if (b->second.empty()) buckets_.erase(b);
    }
public:
    void onPut(const K& k) override {
        buckets_[1].push_front(k); info_[k] = Info{1, buckets_[1].begin()}; minFreq_ = 1;
    }
    void onAccess(const K& k) override {
        auto it = info_.find(k);
        if (it == info_.end()) return;
        int f = it->second.freq;
        drop(it->second);
        if (f == minFreq_ && !buckets_.count(f)) minFreq_ = f + 1;
        buckets_[f + 1].push_front(k);
        it->second = Info{f + 1, buckets_[f + 1].begin()};
    }
    void onRemove(const K& k) override {
        auto it = info_.find(k);
        if (it == info_.end()) return;
        drop(it->second); info_.erase(it);
    }
    K evict() override {
        if (info_.empty()) throw std::logic_error("nothing to evict");
        if (!buckets_.count(minFreq_)) minFreq_ = buckets_.begin()->first;   // repair after onRemove
        std::list<K>& b = buckets_[minFreq_];
        K k = b.back();                                           // oldest at the lowest frequency
        b.pop_back();
        if (b.empty()) buckets_.erase(minFreq_);
        info_.erase(k);
        return k;
    }
};

template <typename K>
class FifoPolicy : public EvictionPolicy<K> {
    std::list<K> order_;
    std::unordered_map<K, typename std::list<K>::iterator> pos_;
public:
    void onPut(const K& k) override { order_.push_back(k); pos_[k] = std::prev(order_.end()); }
    void onAccess(const K&) override {}
    void onRemove(const K& k) override { auto it = pos_.find(k); if (it != pos_.end()) { order_.erase(it->second); pos_.erase(it); } }
    K evict() override { K k = order_.front(); order_.pop_front(); pos_.erase(k); return k; }
};

template <typename K, typename V>
class Cache {
    size_t capacity_;
    std::unique_ptr<EvictionPolicy<K>> policy_;
    std::unordered_map<K, V> map_;
public:
    long hits = 0, misses = 0, evictions = 0;
    Cache(size_t cap, std::unique_ptr<EvictionPolicy<K>> p) : capacity_(cap), policy_(std::move(p)) {
        if (cap == 0) throw std::invalid_argument("capacity must be > 0");
    }
    bool get(const K& k, V& out) {
        auto it = map_.find(k);
        if (it == map_.end()) { ++misses; return false; }
        ++hits; policy_->onAccess(k); out = it->second; return true;
    }
    void put(const K& k, const V& v) {
        auto it = map_.find(k);
        if (it != map_.end()) { it->second = v; policy_->onAccess(k); return; }
        if (map_.size() >= capacity_) { map_.erase(policy_->evict()); ++evictions; }
        map_[k] = v; policy_->onPut(k);
    }
    bool remove(const K& k) {
        if (!map_.erase(k)) return false;
        policy_->onRemove(k); return true;
    }
    bool contains(const K& k) const { return map_.count(k) != 0; }
    size_t size() const { return map_.size(); }
};

static int passed = 0;
static void check(bool ok, const char* what) { if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); } ++passed; }

int main() {
    int v = 0;
    Cache<std::string, int> lru(3, std::unique_ptr<EvictionPolicy<std::string>>(new LruPolicy<std::string>()));
    lru.put("a", 1); lru.put("b", 2); lru.put("c", 3); lru.get("a", v); lru.put("d", 4);
    std::cout << "LRU  b evicted: " << std::boolalpha << !lru.contains("b") << "\n";
    check(!lru.contains("b") && lru.contains("a"), "LRU evicts b");

    Cache<std::string, int> lfu(3, std::unique_ptr<EvictionPolicy<std::string>>(new LfuPolicy<std::string>()));
    lfu.put("a", 1); lfu.put("b", 2); lfu.put("c", 3);
    lfu.get("a", v); lfu.get("a", v); lfu.get("b", v);
    lfu.put("d", 4); lfu.put("e", 5);
    std::cout << "LFU  c and d evicted: " << (!lfu.contains("c") && !lfu.contains("d")) << "\n";
    check(!lfu.contains("c") && !lfu.contains("d") && lfu.contains("a"), "LFU evicts c then d");

    lfu.remove("a");                                   // exercise the minFreq repair path
    lfu.put("f", 6); lfu.put("g", 7);
    check(lfu.size() == 3, "LFU size stays at capacity after remove + puts");

    Cache<std::string, int> fifo(3, std::unique_ptr<EvictionPolicy<std::string>>(new FifoPolicy<std::string>()));
    fifo.put("a", 1); fifo.put("b", 2); fifo.put("c", 3); fifo.get("a", v); fifo.put("d", 4);
    std::cout << "FIFO a evicted despite the read: " << !fifo.contains("a") << "\n";
    check(!fifo.contains("a"), "FIFO evicts a");

    Cache<int, int> big(100, std::unique_ptr<EvictionPolicy<int>>(new LruPolicy<int>()));
    unsigned seed = 7; long gets = 0;
    for (int i = 0; i < 100000; ++i) {
        seed = seed * 1103515245u + 12345u;
        int k = (int)((seed >> 8) % 300);
        if ((seed >> 16) & 1) big.put(k, k); else { big.get(k, v); ++gets; }
    }
    std::cout << "STRESS hits=" << big.hits << " misses=" << big.misses << " evictions=" << big.evictions << " size=" << big.size() << "\n";
    check(big.size() <= 100 && big.hits + big.misses == gets, "capacity and stats hold");
    std::cout << "All " << passed << " checks passed\n";
}
