Code: vending machine (Java), trip lifecycle table (Python), TCP (C++14)

All three programs were compiled and run; the output is real. Java is the classic State pattern with stateless shared state objects. Python deliberately shows the other approach, an enum plus a transition table, for a ride's lifecycle, so you can compare. C++14 is a slice of the TCP state machine, one class per state. Files live in lld-code/lld-pat-state/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Java interface: a struct of function pointers with no data. A class that implements it must fill every slot.
  • default method in an interface: a slot with a fallback implementation, like an ops table where NULL means "use the generic helper".
  • Anonymous class new Foo() { ... }: defines and creates a one-off implementation in place, like a static ops table filled inline.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • Method reference Foo::new, obj::method: shorthand for a lambda that just calls that constructor or method.
  • Map.of(k1, v1, ...), List.of(...): build a small read-only map or list in one expression.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • Python lambda x: ...: a one-expression anonymous function.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

public class StateDemo {

    // ---- State interface: one method per event the machine can receive ----
    interface State {
        String name();
        default void insertCoin(VendingMachine m, int rupees) { m.say("ignored coin in " + name() + ", returned Rs " + rupees); }
        default void select(VendingMachine m, String code)    { m.say("select ignored in " + name()); }
        default void cancel(VendingMachine m)                 { m.say("nothing to cancel in " + name()); }
        default void dispensed(VendingMachine m)              { throw new IllegalStateException("motor event in " + name()); }
        default void refill(VendingMachine m, String code, int n) { m.addStock(code, n); }
    }

    // ---- Concrete states: stateless, so one shared instance each ----
    static final State IDLE = new State() {
        public String name() { return "IDLE"; }
        public void insertCoin(VendingMachine m, int rupees) {
            m.addBalance(rupees);
            m.setState(HAS_MONEY);                                   // the state picks the next state
        }
    };

    static final State HAS_MONEY = new State() {
        public String name() { return "HAS_MONEY"; }
        public void insertCoin(VendingMachine m, int rupees) { m.addBalance(rupees); }
        public void select(VendingMachine m, String code) {
            if (m.stock(code) == 0) { m.say(code + " is empty, pick another"); return; }
            if (m.balance() < m.price(code)) { m.say("need Rs " + (m.price(code) - m.balance()) + " more for " + code); return; }
            m.setState(DISPENSING);
            m.dispense(code);                                        // DISPENSING decides where to go next
        }
        public void cancel(VendingMachine m) {
            m.refund();
            m.setState(IDLE);
        }
    };

    static final State DISPENSING = new State() {
        public String name() { return "DISPENSING"; }
        public void insertCoin(VendingMachine m, int rupees) { m.say("busy, returned Rs " + rupees); }
        public void cancel(VendingMachine m) { m.say("too late to cancel, item is dropping"); }
        public void refill(VendingMachine m, String code, int n) { m.say("busy, refill later"); }
        public void dispensed(VendingMachine m) {                   // the motor finished
            if (m.balance() > 0) m.refund();
            m.setState(m.hasAnyStock() ? IDLE : SOLD_OUT);
        }
    };

    static final State SOLD_OUT = new State() {
        public String name() { return "SOLD_OUT"; }
        public void refill(VendingMachine m, String code, int n) {
            m.addStock(code, n);
            if (n > 0) m.setState(IDLE);
        }
    };

    // ---- Context: owns the data, forwards every event to the current state ----
    static final class VendingMachine {
        private State state = SOLD_OUT;
        private int balance;
        private final Map<String, Integer> stock = new TreeMap<>();
        private final Map<String, Integer> price = Map.of("A1", 20, "B2", 35);

        // public API: no if/switch on state anywhere
        void insertCoin(int rupees) { state.insertCoin(this, rupees); }
        void select(String code)    { state.select(this, code); }
        void cancel()               { state.cancel(this); }
        void refill(String code, int n) { state.refill(this, code, n); }

        // package-private helpers used by the states
        void setState(State next) { say(state.name() + " -> " + next.name()); state = next; }
        void say(String s) { System.out.println("  " + s); }
        void addBalance(int r) { balance += r; say("balance Rs " + balance); }
        int balance() { return balance; }
        int price(String code) { return price.getOrDefault(code, Integer.MAX_VALUE); }
        int stock(String code) { return stock.getOrDefault(code, 0); }
        boolean hasAnyStock() { return stock.values().stream().anyMatch(q -> q > 0); }
        void addStock(String code, int n) { stock.merge(code, n, Integer::sum); say("stock " + stock); }
        void refund() { say("refund Rs " + balance); balance = 0; }
        void dispense(String code) {
            balance -= price(code);
            stock.merge(code, -1, Integer::sum);
            say("dropped " + code);
            state.dispensed(this);                                   // in real hardware: a sensor interrupt
        }
    }

    public static void main(String[] args) {
        VendingMachine vm = new VendingMachine();
        System.out.println("coin while empty:");   vm.insertCoin(10);
        System.out.println("refill:");             vm.refill("A1", 1); vm.refill("B2", 2);
        System.out.println("buy A1 with Rs 50:");  vm.insertCoin(20); vm.select("B2"); vm.insertCoin(30); vm.select("A1");
        System.out.println("cancel half way:");    vm.insertCoin(10); vm.cancel();
        System.out.println("buy B2 twice:");
        vm.insertCoin(50); vm.select("A1"); vm.select("B2");
        vm.insertCoin(35); vm.select("B2");
    }
}

Read the output against the stepper: every arrow printed (IDLE -> HAS_MONEY) was decided by the state that was current. The public methods of VendingMachine are one line each, with no switch. Adding a MAINTENANCE state is one new object plus the transitions into it; existing states that do not lead there are untouched.

"""The other way to write a state machine: an enum plus a transition table.

A ride's lifecycle (ride hailing LLD). Compare with the State-classes version in StateDemo.java.
Good when states have little behaviour of their own and you want every legal move in one place.
"""
from enum import Enum, auto


class Trip(Enum):
    REQUESTED = auto()
    ACCEPTED = auto()
    ARRIVED = auto()
    IN_PROGRESS = auto()
    COMPLETED = auto()
    CANCELLED = auto()


# (current state, event) -> next state. Anything missing is illegal.
TRANSITIONS = {
    (Trip.REQUESTED, "driver_accepts"): Trip.ACCEPTED,
    (Trip.REQUESTED, "rider_cancels"): Trip.CANCELLED,
    (Trip.ACCEPTED, "driver_arrives"): Trip.ARRIVED,
    (Trip.ACCEPTED, "rider_cancels"): Trip.CANCELLED,
    (Trip.ACCEPTED, "driver_cancels"): Trip.REQUESTED,     # back to matching
    (Trip.ARRIVED, "start_trip"): Trip.IN_PROGRESS,
    (Trip.ARRIVED, "rider_cancels"): Trip.CANCELLED,       # a fee may apply here
    (Trip.IN_PROGRESS, "end_trip"): Trip.COMPLETED,
}

# entry actions: run once when a state is entered
ON_ENTER = {
    Trip.CANCELLED: lambda ride, prev: print(f"    fee: Rs {50 if prev is Trip.ARRIVED else 0}"),
    Trip.COMPLETED: lambda ride, prev: print("    charge rider, pay driver"),
}


class IllegalTransition(Exception):
    pass


class Ride:
    def __init__(self, ride_id):
        self.ride_id, self.state, self.history = ride_id, Trip.REQUESTED, [Trip.REQUESTED]

    def fire(self, event):
        nxt = TRANSITIONS.get((self.state, event))
        if nxt is None:
            raise IllegalTransition(f"{event} not allowed in {self.state.name}")
        prev, self.state = self.state, nxt
        self.history.append(nxt)
        print(f"  {self.ride_id}: {prev.name} --{event}--> {nxt.name}")
        if nxt in ON_ENTER:
            ON_ENTER[nxt](self, prev)


if __name__ == "__main__":
    r1 = Ride("R1")
    for ev in ("driver_accepts", "driver_cancels", "driver_accepts", "driver_arrives", "start_trip", "end_trip"):
        r1.fire(ev)

    r2 = Ride("R2")
    r2.fire("driver_accepts")
    r2.fire("driver_arrives")
    r2.fire("rider_cancels")
    try:
        r2.fire("start_trip")
    except IllegalTransition as e:
        print(f"  rejected: {e}")

    print("  states with no way out:", sorted({s.name for s in Trip} - {s.name for s, _ in TRANSITIONS}))

Here the states have almost no behaviour; what matters is which moves are legal. A dictionary keyed by (state, event) shows the whole lifecycle in eight lines, rejects illegal moves in one place, and can be checked mechanically (the last line finds terminal states). Many production systems (order, payment and ticket workflows) use this table form, often stored in configuration or a database.

// State in C++14: the passive-open and passive-close half of the TCP state machine,
// one class per state. Compare with the big switch (sk->sk_state) in tcp_rcv_state_process().
// Build: g++ -std=c++14 -Wall tcp_state.cpp -o tcp_state
#include <iostream>
#include <string>

class Connection;

class TcpState {                                  // one virtual per event
public:
    virtual ~TcpState() = default;
    virtual const char* name() const = 0;
    virtual void onSyn(Connection& c) const;
    virtual void onAck(Connection& c) const;
    virtual void onFin(Connection& c) const;
    virtual void appClose(Connection& c) const;
};

class Connection {                                // the Context
    const TcpState* state_;
public:
    explicit Connection(const TcpState& s) : state_(&s) {}
    void setState(const TcpState& next) {
        std::cout << "    " << state_->name() << " -> " << next.name() << "\n";
        state_ = &next;
    }
    void send(const char* seg) { std::cout << "    send " << seg << "\n"; }
    const char* stateName() const { return state_->name(); }
    // the public API just forwards to the current state: no switch here
    void rcvSyn()  { state_->onSyn(*this); }
    void rcvAck()  { state_->onAck(*this); }
    void rcvFin()  { state_->onFin(*this); }
    void close()   { state_->appClose(*this); }
};

// Stateless state objects: one shared instance each (like the kernel's TCP_* constants).
struct Closed : TcpState {
    const char* name() const override { return "CLOSED"; }
};
struct Listen : TcpState {
    const char* name() const override { return "LISTEN"; }
    void onSyn(Connection& c) const override;
    void onAck(Connection& c) const override { c.send("RST"); std::cout << "    (stray ACK: reset it, keep listening)\n"; }
};
struct SynRcvd : TcpState {
    const char* name() const override { return "SYN_RECV"; }
    void onAck(Connection& c) const override;
};
struct Established : TcpState {
    const char* name() const override { return "ESTABLISHED"; }
    void onAck(Connection&) const override { std::cout << "    data acked, stay\n"; }
    void onFin(Connection& c) const override;
};
struct CloseWait : TcpState {
    const char* name() const override { return "CLOSE_WAIT"; }
    void appClose(Connection& c) const override;
};
struct LastAck : TcpState {
    const char* name() const override { return "LAST_ACK"; }
    void onAck(Connection& c) const override;
};

const Closed CLOSED{};
const Listen LISTEN{};
const SynRcvd SYN_RECV{};
const Established ESTABLISHED{};
const CloseWait CLOSE_WAIT{};
const LastAck LAST_ACK{};

// Default for an unexpected event: answer with RST and drop to CLOSED.
void TcpState::onSyn(Connection& c) const    { std::cout << "    unexpected SYN in " << name() << "\n"; c.send("RST"); c.setState(CLOSED); }
void TcpState::onAck(Connection& c) const    { std::cout << "    unexpected ACK in " << name() << "\n"; c.send("RST"); c.setState(CLOSED); }
void TcpState::onFin(Connection& c) const    { std::cout << "    unexpected FIN in " << name() << ", ignored\n"; }
void TcpState::appClose(Connection& c) const { c.setState(CLOSED); }

// Each state decides the next state: transitions live next to the behaviour.
void Listen::onSyn(Connection& c) const      { c.send("SYN+ACK"); c.setState(SYN_RECV); }
void SynRcvd::onAck(Connection& c) const     { c.setState(ESTABLISHED); }
void Established::onFin(Connection& c) const { c.send("ACK"); c.setState(CLOSE_WAIT); }
void CloseWait::appClose(Connection& c) const{ c.send("FIN"); c.setState(LAST_ACK); }
void LastAck::onAck(Connection& c) const     { c.setState(CLOSED); }

int main() {
    Connection c(LISTEN);
    std::cout << "client connects:\n";      c.rcvSyn(); c.rcvAck();
    std::cout << "data flows:\n";           c.rcvAck();
    std::cout << "client closes:\n";        c.rcvFin();
    std::cout << "server app closes:\n";    c.close(); c.rcvAck();
    std::cout << "final state: " << c.stateName() << "\n";

    Connection d(LISTEN);
    std::cout << "stray ACK on a listener:\n"; d.rcvAck();
    return 0;
}

The base class gives every unexpected event a default (reset, or ignore). Each state overrides only the segments it expects. Compare Listen::onAck (a stray ACK: send RST, stay listening) with the base onAck (reset and drop to CLOSED): the same event has different meanings in different states, which is the whole point of the pattern.
