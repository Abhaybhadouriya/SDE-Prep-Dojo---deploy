  • main.cpp: one file, g++ -std=c++14 main.cpp.

Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was C++ demo OK.

// In-memory file system in C++14: same design as the Java version, single-threaded demo.
// Ownership: a Directory owns its children (shared_ptr); a child points up with weak_ptr, so there is
// no reference cycle. A Shell holds a shared_ptr to its cwd, which keeps a removed directory alive
// (like a dentry reference from dget) until the shell moves away.
// Thread safety: one reader-writer lock for the tree (readers shared, writers exclusive), like Java's
// ReentrantReadWriteLock. Windows (MinGW g++ 6.3): g++ -std=c++14 -Wall main.cpp -o fs
// Linux (real lock + 8-thread append): g++ -std=c++14 -Wall -pthread -DHAVE_STD_MUTEX main.cpp -o fs
#include <algorithm>
#include <cassert>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>

using namespace std;

// MinGW.org g++ 6.3 (win32 thread model) has no std::shared_timed_mutex/std::thread and -pthread fails to
// link there, so without HAVE_STD_MUTEX RwLock is a no-op with the same interface: the SAME locking code
// compiles both ways and runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
#include <shared_mutex>
#include <thread>
typedef std::shared_timed_mutex RwLock;               // C++14; C++17 has std::shared_mutex
#else
struct RwLock { void lock() {} void unlock() {} void lock_shared() {} void unlock_shared() {} };
#endif
struct ReadGuard {                                     // like down_read()/up_read()
    RwLock& m; explicit ReadGuard(RwLock& x) : m(x) { m.lock_shared(); } ~ReadGuard() { m.unlock_shared(); }
};
struct WriteGuard {                                    // like down_write()/up_write()
    RwLock& m; explicit WriteGuard(RwLock& x) : m(x) { m.lock(); } ~WriteGuard() { m.unlock(); }
};

struct FsError : runtime_error {
    explicit FsError(const string& m) : runtime_error(m) {}
};

struct Directory;

// ---------- Composite ----------
struct Node {
    string name;
    weak_ptr<Directory> parent;                  // empty for root and for removed nodes
    explicit Node(string n) : name(move(n)) {}
    virtual ~Node() {}
    virtual bool isDir() const = 0;
    virtual long size() const = 0;
    string path() const;
};

struct FileNode : Node {
    string content;
    explicit FileNode(string n) : Node(move(n)) {}
    bool isDir() const override { return false; }
    long size() const override { return (long)content.size(); }
};

struct Directory : Node, enable_shared_from_this<Directory> {
    map<string, shared_ptr<Node>> children;      // sorted = ls order
    explicit Directory(string n) : Node(move(n)) {}
    bool isDir() const override { return true; }
    long size() const override {
        long total = 0;
        for (const auto& kv : children) total += kv.second->size();
        return total;
    }
    shared_ptr<Directory> up() {                 // '..' of root is root
        auto p = parent.lock();
        return p ? p : shared_from_this();
    }
    vector<shared_ptr<Directory>> childDirs() const {
        vector<shared_ptr<Directory>> out;
        for (const auto& kv : children)
            if (kv.second->isDir()) out.push_back(static_pointer_cast<Directory>(kv.second));
        return out;
    }
};

string Node::path() const {
    auto p = parent.lock();
    if (!p) return "/";
    string pp = p->path();
    return pp == "/" ? "/" + name : pp + "/" + name;
}

// ---------- glob ----------
bool hasWildcard(const string& s) { return s.find_first_of("*?") != string::npos; }

bool globMatch(const string& pat, const string& text) {
    size_t p = 0, t = 0, mark = 0;
    long star = -1;
    while (t < text.size()) {
        if (p < pat.size() && (pat[p] == '?' || pat[p] == text[t])) { ++p; ++t; }
        else if (p < pat.size() && pat[p] == '*') { star = (long)p++; mark = t; }
        else if (star >= 0) { p = (size_t)star + 1; t = ++mark; }
        else return false;
    }
    while (p < pat.size() && pat[p] == '*') ++p;
    return p == pat.size();
}

// ---------- Specification ----------
struct Spec {
    function<bool(const Node&)> fn;
    bool test(const Node& n) const { return fn(n); }
    Spec And(Spec o) const { auto a = fn; return Spec{[a, o](const Node& n) { return a(n) && o.test(n); }}; }
    Spec Or(Spec o) const { auto a = fn; return Spec{[a, o](const Node& n) { return a(n) || o.test(n); }}; }
    Spec Not() const { auto a = fn; return Spec{[a](const Node& n) { return !a(n); }}; }
};
Spec nameSpec(string g) { return Spec{[g](const Node& n) { return globMatch(g, n.name); }}; }
Spec extSpec(string e) {
    return Spec{[e](const Node& n) {
        string suf = "." + e;
        return !n.isDir() && n.name.size() >= suf.size() &&
               n.name.compare(n.name.size() - suf.size(), suf.size(), suf) == 0;
    }};
}
Spec minSize(long b) { return Spec{[b](const Node& n) { return !n.isDir() && n.size() >= b; }}; }
Spec isFile() { return Spec{[](const Node& n) { return !n.isDir(); }}; }
Spec isDir() { return Spec{[](const Node& n) { return n.isDir(); }}; }

// ---------- the shared tree ----------
using DirPtr = shared_ptr<Directory>;
using NodePtr = shared_ptr<Node>;

vector<string> segments(const string& path) {
    vector<string> out;
    string cur;
    for (char c : path) {
        if (c == '/') { if (!cur.empty()) out.push_back(cur); cur.clear(); }
        else cur += c;
    }
    if (!cur.empty()) out.push_back(cur);
    return out;
}

class FileSystem {
public:
    DirPtr root = make_shared<Directory>("");

    DirPtr resolveDir(const DirPtr& cwd, const string& path) {
        ReadGuard g(rw_);
        NodePtr n = mustExist(cwd, path);
        if (!n->isDir()) throw FsError("not a directory: " + path);
        return static_pointer_cast<Directory>(n);
    }

    vector<DirPtr> resolveDirs(const DirPtr& cwd, const string& pattern) {
        ReadGuard g(rw_);
        vector<DirPtr> frontier{start(cwd, pattern)};
        for (const string& seg : segments(pattern)) {
            set<Directory*> seen;
            vector<DirPtr> next;
            for (const DirPtr& d : frontier) {
                vector<DirPtr> cands;
                if (seg == ".") cands.push_back(d);
                else if (seg == "..") cands.push_back(d->up());
                else if (seg == "*") {
                    cands.push_back(d);
                    cands.push_back(d->up());
                    for (auto& c : d->childDirs()) cands.push_back(c);
                } else if (hasWildcard(seg)) {
                    for (auto& c : d->childDirs()) if (globMatch(seg, c->name)) cands.push_back(c);
                } else {
                    auto it = d->children.find(seg);
                    if (it != d->children.end() && it->second->isDir())
                        cands.push_back(static_pointer_cast<Directory>(it->second));
                }
                for (auto& c : cands) if (seen.insert(c.get()).second) next.push_back(c);
            }
            frontier = next;
            if (frontier.empty()) break;
        }
        sort(frontier.begin(), frontier.end(),
             [](const DirPtr& a, const DirPtr& b) { return a->path() < b->path(); });
        return frontier;
    }

    void mkdir(const DirPtr& cwd, const string& path, bool parents) {
        WriteGuard g(rw_);
        vector<string> segs = segments(path);
        if (segs.empty()) throw FsError("mkdir: missing name");
        DirPtr cur = start(cwd, path);
        for (size_t i = 0; i < segs.size(); ++i) {
            const string& s = segs[i];
            bool last = i + 1 == segs.size();
            if (s == ".") continue;
            if (s == "..") { cur = cur->up(); continue; }
            auto it = cur->children.find(s);
            if (it == cur->children.end()) {
                if (!last && !parents) throw FsError("no such directory: " + s);
                checkName(s);
                auto d = make_shared<Directory>(s);
                d->parent = cur;
                cur->children[s] = d;
                cur = d;
            } else if (!it->second->isDir()) {
                throw FsError("not a directory: " + it->second->path());
            } else if (last && !parents) {
                throw FsError("already exists: " + it->second->path());
            } else {
                cur = static_pointer_cast<Directory>(it->second);
            }
        }
    }

    vector<string> ls(const DirPtr& cwd, const string& path) {
        ReadGuard g(rw_);
        vector<string> segs = segments(path), out;
        if (!segs.empty() && hasWildcard(segs.back())) {
            for (auto& kv : parentOf(cwd, path, segs)->children)
                if (globMatch(segs.back(), kv.first)) out.push_back(kv.first);
            return out;
        }
        NodePtr n = mustExist(cwd, path);
        if (!n->isDir()) return {n->name};
        for (auto& kv : static_pointer_cast<Directory>(n)->children) out.push_back(kv.first);
        return out;
    }

    void writeFile(const DirPtr& cwd, const string& path, const string& text, bool append) {
        WriteGuard g(rw_);
        vector<string> segs = segments(path);
        if (segs.empty()) throw FsError("not a file: " + path);
        DirPtr d = parentOf(cwd, path, segs);
        auto it = d->children.find(segs.back());
        shared_ptr<FileNode> f;
        if (it == d->children.end()) {
            checkName(segs.back());
            f = make_shared<FileNode>(segs.back());
            f->parent = d;
            d->children[segs.back()] = f;
        } else if (it->second->isDir()) {
            throw FsError("is a directory: " + it->second->path());
        } else {
            f = static_pointer_cast<FileNode>(it->second);
        }
        if (!append) f->content.clear();
        f->content += text;
    }

    string readFile(const DirPtr& cwd, const string& path) {
        ReadGuard g(rw_);
        NodePtr n = mustExist(cwd, path);
        if (n->isDir()) throw FsError("is a directory: " + path);
        return static_pointer_cast<FileNode>(n)->content;
    }

    void remove(const DirPtr& cwd, const string& path, bool recursive) {
        WriteGuard g(rw_);
        NodePtr n = mustExist(cwd, path);
        if (n == root) throw FsError("cannot remove /");
        if (n->isDir() && !recursive && !static_pointer_cast<Directory>(n)->children.empty())
            throw FsError("directory not empty: " + n->path());
        DirPtr p = n->parent.lock();
        n->parent.reset();                       // detach first: stale holders see "not attached"
        p->children.erase(n->name);
    }

    long du(const DirPtr& cwd, const string& path) { ReadGuard g(rw_); return mustExist(cwd, path)->size(); }

    vector<string> find(const DirPtr& cwd, const string& path, const Spec& spec) {
        ReadGuard g(rw_);
        vector<string> out;
        vector<NodePtr> stack{mustExist(cwd, path)};
        while (!stack.empty()) {
            NodePtr n = stack.back();
            stack.pop_back();
            if (spec.test(*n)) out.push_back(n->path());
            if (n->isDir()) {
                auto& kids = static_pointer_cast<Directory>(n)->children;
                for (auto it = kids.rbegin(); it != kids.rend(); ++it) stack.push_back(it->second);
            }
        }
        return out;
    }

    bool isAttached(const DirPtr& d) {
        ReadGuard g(rw_);
        DirPtr cur = d;
        while (auto p = cur->parent.lock()) cur = p;
        return cur == root;
    }

private:
    RwLock rw_;                                        // guards the whole tree; private helpers assume it is held
    DirPtr start(const DirPtr& cwd, const string& path) {
        if (path.empty()) throw FsError("empty path");
        return path[0] == '/' ? root : cwd;
    }
    NodePtr walk(const DirPtr& from, const vector<string>& segs, size_t count) {
        NodePtr cur = from;
        for (size_t i = 0; i < count; ++i) {
            const string& s = segs[i];
            if (!cur->isDir()) throw FsError("not a directory: " + cur->path());
            DirPtr d = static_pointer_cast<Directory>(cur);
            if (s == ".") continue;
            if (s == "..") { cur = d->up(); continue; }
            auto it = d->children.find(s);
            if (it == d->children.end()) return nullptr;
            cur = it->second;
        }
        return cur;
    }
    NodePtr mustExist(const DirPtr& cwd, const string& path) {
        vector<string> segs = segments(path);
        NodePtr n = walk(start(cwd, path), segs, segs.size());
        if (!n) throw FsError("no such file or directory: " + path);
        return n;
    }
    DirPtr parentOf(const DirPtr& cwd, const string& path, const vector<string>& segs) {
        NodePtr p = walk(start(cwd, path), segs, segs.size() - 1);
        if (!p) throw FsError("no such directory: " + path);
        if (!p->isDir()) throw FsError("not a directory: " + p->path());
        return static_pointer_cast<Directory>(p);
    }
    static void checkName(const string& s) {
        if (s == "." || s == ".." || hasWildcard(s)) throw FsError("invalid name: " + s);
    }
};

class Shell {
    FileSystem& fs;
    DirPtr cwd;
    DirPtr here() {
        if (!fs.isAttached(cwd)) cwd = fs.root;
        return cwd;
    }
public:
    explicit Shell(FileSystem& f) : fs(f), cwd(f.root) {}
    string pwd() { return here()->path(); }
    void cd(const string& path) {
        if (!hasWildcard(path)) { cwd = fs.resolveDir(here(), path); return; }
        auto m = fs.resolveDirs(here(), path);
        if (m.empty()) throw FsError("no such directory: " + path);
        cwd = m.front();
    }
    void mkdir(const string& p, bool parents = false) { fs.mkdir(here(), p, parents); }
    vector<string> ls(const string& p) { return fs.ls(here(), p); }
    void write(const string& p, const string& t) { fs.writeFile(here(), p, t, false); }
    void append(const string& p, const string& t) { fs.writeFile(here(), p, t, true); }
    string cat(const string& p) { return fs.readFile(here(), p); }
    void rm(const string& p, bool recursive = false) { fs.remove(here(), p, recursive); }
    long du(const string& p) { return fs.du(here(), p); }
    vector<string> find(const string& p, const Spec& s) { return fs.find(here(), p, s); }
};

string join(const vector<string>& v) {
    string s = "[";
    for (size_t i = 0; i < v.size(); ++i) s += (i ? ", " : "") + v[i];
    return s + "]";
}
