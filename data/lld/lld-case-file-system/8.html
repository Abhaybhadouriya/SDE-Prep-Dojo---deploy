/** Composite "Component": every entry in the tree, file or directory, answers these questions. */
abstract class Node {
    private final String name;
    Directory parent;                     // null for root and for removed (detached) nodes

    Node(String name, Directory parent) {
        this.name = name;
        this.parent = parent;
    }

    String name() { return name; }

    abstract boolean isDirectory();

    abstract long size();

    String path() {
        if (parent == null) return "/";
        String p = parent.path();
        return p.equals("/") ? "/" + name : p + "/" + name;
    }
}

/** Composite "Leaf". */
final class FileNode extends Node {
    private final StringBuilder content = new StringBuilder();

    FileNode(String name, Directory parent) { super(name, parent); }

    @Override boolean isDirectory() { return false; }

    @Override long size() { return content.length(); }

    void write(String text, boolean append) {
        if (!append) content.setLength(0);
        content.append(text);
    }

    String read() { return content.toString(); }
}

/** Composite "Composite": holds children of the Component type, sorted by name (ls order). */
final class Directory extends Node {
    private final TreeMap<String, Node> children = new TreeMap<>();

    Directory(String name, Directory parent) { super(name, parent); }

    @Override boolean isDirectory() { return true; }

    @Override long size() {                         // recursion falls out of the Composite shape
        long total = 0;
        for (Node child : children.values()) total += child.size();
        return total;
    }

    Node child(String name) { return children.get(name); }

    Collection<Node> children() { return children.values(); }

    List<Directory> childDirs() {
        List<Directory> dirs = new ArrayList<>();
        for (Node n : children.values()) if (n.isDirectory()) dirs.add((Directory) n);
        return dirs;
    }

    void add(Node node) { children.put(node.name(), node); }

    void remove(String name) {
        Node n = children.remove(name);
        if (n != null) n.parent = null;             // detach so stale handles can tell
    }

    boolean isEmpty() { return children.isEmpty(); }

    /** ".." of root is root, as in Unix. */
    Directory up() { return parent == null ? this : parent; }
}

/** Shell-style glob: '*' = any run of characters (also empty), '?' = exactly one character. */
final class Glob {
    private Glob() {}

    static boolean hasWildcard(String s) {
        return s.indexOf('*') >= 0 || s.indexOf('?') >= 0;
    }

    /** Greedy two-pointer match with backtracking to the last '*'. O(len(p) * len(t)) worst case. */
    static boolean matches(String pattern, String text) {
        int p = 0, t = 0, star = -1, mark = 0;
        while (t < text.length()) {
            if (p < pattern.length() && (pattern.charAt(p) == '?' || pattern.charAt(p) == text.charAt(t))) {
                p++;
                t++;
            } else if (p < pattern.length() && pattern.charAt(p) == '*') {
                star = p++;
                mark = t;
            } else if (star >= 0) {
                p = star + 1;                       // let the last '*' swallow one more character
                t = ++mark;
            } else {
                return false;
            }
        }
        while (p < pattern.length() && pattern.charAt(p) == '*') p++;
        return p == pattern.length();
    }
}

interface Spec {

    default Spec or(Spec other) { return n -> test(n) || other.test(n); }

    default Spec not() { return n -> !test(n); }
}

/**
 * The shared tree. Thread-safe with one read-write lock: many readers (ls, cat, find, du) or one writer.
 * It never stores a current directory: that belongs to each Shell (session), like cwd belongs to a process.
 */
final class FileSystem {
    private final Directory root = new Directory("", null);
    private final ReentrantReadWriteLock lock = new ReentrantReadWriteLock();

    Directory root() { return root; }

    // ---------- path parsing ----------
    static List<String> segments(String path) {
        List<String> out = new ArrayList<>();
        for (String s : path.split("/")) if (!s.isEmpty()) out.add(s);
        return out;
    }

    private Directory start(Directory cwd, String path) {
        if (path.isEmpty()) throw new FsException("empty path");
        return path.startsWith("/") ? root : cwd;
    }

    /** Walk a literal path (no wildcards). Returns null if a component is missing. */
    private Node walk(Directory from, List<String> segs) {
        Node cur = from;
        for (String s : segs) {
            if (!cur.isDirectory()) throw new FsException("not a directory: " + cur.path());
            Directory d = (Directory) cur;
            if (s.equals(".")) continue;
            if (s.equals("..")) { cur = d.up(); continue; }
            cur = d.child(s);
            if (cur == null) return null;
        }
        return cur;
    }

    private Node mustExist(Directory cwd, String path) {
        Node n = walk(start(cwd, path), segments(path));
        if (n == null) throw new FsException("no such file or directory: " + path);
        return n;
    }

    private Directory mustBeDir(Directory cwd, String path) {
        Node n = mustExist(cwd, path);
        if (!n.isDirectory()) throw new FsException("not a directory: " + path);
        return (Directory) n;
    }

    /** Parent directory of the last segment, for create / glob operations. */
    private Directory parentOf(Directory cwd, String path, List<String> segs) {
        Node p = walk(start(cwd, path), segs.subList(0, segs.size() - 1));
        if (p == null) throw new FsException("no such directory: " + path);
        if (!p.isDirectory()) throw new FsException("not a directory: " + p.path());
        return (Directory) p;
    }

    private static void checkName(String name) {
        if (name.equals(".") || name.equals("..") || Glob.hasWildcard(name))
            throw new FsException("invalid name: " + name);
    }

    // ---------- commands ----------
    Directory resolveDir(Directory cwd, String path) {
        lock.readLock().lock();
        try { return mustBeDir(cwd, path); } finally { lock.readLock().unlock(); }   // C: finally runs on every exit path: the goto out_unlock label
    }

    /**
     * Resolve a path whose segments may hold wildcards. A bare "*" matches ".", ".." or any child
     * directory (the Uber variant); a segment like "a*" matches child directories by glob.
     * Returns every distinct match, sorted by absolute path.
     */
    List<Directory> resolveDirs(Directory cwd, String pattern) {
        lock.readLock().lock();
        try {
            List<Directory> frontier = List.of(start(cwd, pattern));   // C: immutable literal collection
            for (String seg : segments(pattern)) {
                Set<Directory> seen = Collections.newSetFromMap(new IdentityHashMap<>());
                List<Directory> next = new ArrayList<>();
                for (Directory d : frontier) {
                    List<Directory> cands = new ArrayList<>();
                    if (seg.equals(".")) cands.add(d);
                    else if (seg.equals("..")) cands.add(d.up());
                    else if (seg.equals("*")) { cands.add(d); cands.add(d.up()); cands.addAll(d.childDirs()); }
                    else if (Glob.hasWildcard(seg)) {
                        for (Directory c : d.childDirs()) if (Glob.matches(seg, c.name())) cands.add(c);
                    } else {
                        Node c = d.child(seg);
                        if (c != null && c.isDirectory()) cands.add((Directory) c);
                    }
                    for (Directory c : cands) if (seen.add(c)) next.add(c);
                }
                frontier = next;
                if (frontier.isEmpty()) break;
            }
            List<Directory> out = new ArrayList<>(frontier);
            out.sort(Comparator.comparing(Node::path));   // C: Type::method = a pointer to an existing function
            return out;
        } finally { lock.readLock().unlock(); }
    }

    void mkdir(Directory cwd, String path, boolean parents) {
        lock.writeLock().lock();
        try {
            List<String> segs = segments(path);
            if (segs.isEmpty()) throw new FsException("mkdir: missing name");
            Directory cur = start(cwd, path);
            for (int i = 0; i < segs.size(); i++) {
                String s = segs.get(i);
                boolean last = i == segs.size() - 1;
                if (s.equals(".")) continue;
                if (s.equals("..")) { cur = cur.up(); continue; }
                Node c = cur.child(s);
                if (c == null) {
                    if (!last && !parents) throw new FsException("no such directory: " + s);
                    checkName(s);
                    Directory d = new Directory(s, cur);
                    cur.add(d);
                    cur = d;
                } else if (!c.isDirectory()) {
                    throw new FsException("not a directory: " + c.path());
                } else if (last && !parents) {
                    throw new FsException("already exists: " + c.path());
                } else {
                    cur = (Directory) c;
                }
            }
        } finally { lock.writeLock().unlock(); }
    }

    /** ls: a file lists itself; a directory lists its children; a glob in the last part filters names. */
    List<String> ls(Directory cwd, String path) {
        lock.readLock().lock();
        try {
            List<String> segs = segments(path);
            List<String> out = new ArrayList<>();
            if (!segs.isEmpty() && Glob.hasWildcard(segs.get(segs.size() - 1))) {
                String glob = segs.get(segs.size() - 1);
                for (Node n : parentOf(cwd, path, segs).children())
                    if (Glob.matches(glob, n.name())) out.add(n.name());
                return out;
            }
            Node n = mustExist(cwd, path);
            if (!n.isDirectory()) return List.of(n.name());
            for (Node c : ((Directory) n).children()) out.add(c.name());
            return out;
        } finally { lock.readLock().unlock(); }
    }

    void writeFile(Directory cwd, String path, String text, boolean append) {
        lock.writeLock().lock();
        try {
            List<String> segs = segments(path);
            if (segs.isEmpty()) throw new FsException("not a file: " + path);
            Directory dir = parentOf(cwd, path, segs);
            String name = segs.get(segs.size() - 1);
            Node n = dir.child(name);
            if (n == null) {
                checkName(name);
                n = new FileNode(name, dir);
                dir.add(n);
            } else if (n.isDirectory()) {
                throw new FsException("is a directory: " + n.path());
            }
            ((FileNode) n).write(text, append);
        } finally { lock.writeLock().unlock(); }
    }

    String readFile(Directory cwd, String path) {
        lock.readLock().lock();
        try {
            Node n = mustExist(cwd, path);
            if (n.isDirectory()) throw new FsException("is a directory: " + path);
            return ((FileNode) n).read();
        } finally { lock.readLock().unlock(); }
    }

    void remove(Directory cwd, String path, boolean recursive) {
        lock.writeLock().lock();
        try {
            Node n = mustExist(cwd, path);
            if (n == root) throw new FsException("cannot remove /");
            if (n.isDirectory() && !recursive && !((Directory) n).isEmpty())
                throw new FsException("directory not empty: " + n.path());
            n.parent.remove(n.name());
        } finally { lock.writeLock().unlock(); }
    }

    long du(Directory cwd, String path) {
        lock.readLock().lock();
        try { return mustExist(cwd, path).size(); } finally { lock.readLock().unlock(); }
    }

    /** Unix find: pre-order walk with an explicit stack (no recursion depth limit), filter by a Spec. */
    List<String> find(Directory cwd, String path, Spec spec) {
        lock.readLock().lock();
        try {
            List<String> out = new ArrayList<>();
            Deque<Node> stack = new ArrayDeque<>();
            stack.push(mustExist(cwd, path));
            while (!stack.isEmpty()) {
                Node n = stack.pop();
                if (spec.test(n)) out.add(n.path());
                if (n.isDirectory()) {
                    List<Node> kids = new ArrayList<>(((Directory) n).children());
                    for (int i = kids.size() - 1; i >= 0; i--) stack.push(kids.get(i));  // keep name order
                }
            }
            return out;
        } finally { lock.readLock().unlock(); }
    }

    boolean isAttached(Node n) {
        lock.readLock().lock();
        try {
            Node cur = n;
            while (cur.parent != null) cur = cur.parent;
            return cur == root;
        } finally { lock.readLock().unlock(); }
    }
}
import java.util.List;

/** One user's session: owns the current directory, delegates everything else to the shared FileSystem. */
final class Shell {
    private final FileSystem fs;
    private Directory cwd;

    Shell(FileSystem fs) {
        this.fs = fs;
        this.cwd = fs.root();
    }

    private Directory here() {
        if (!fs.isAttached(cwd)) cwd = fs.root();   // our directory was removed by someone: fall back to /
        return cwd;
    }

    String pwd() { return here().path(); }

    /** cd with wildcards: if several directories match, pick the smallest absolute path (agreed rule). */
    void cd(String path) {
        if (!Glob.hasWildcard(path)) { cwd = fs.resolveDir(here(), path); return; }
        List<Directory> matches = fs.resolveDirs(here(), path);
        if (matches.isEmpty()) throw new FsException("no such directory: " + path);
        cwd = matches.get(0);
    }

    void mkdir(String path) { fs.mkdir(here(), path, false); }

    void mkdirP(String path) { fs.mkdir(here(), path, true); }

    List<String> ls(String path) { return fs.ls(here(), path); }

    void write(String path, String text) { fs.writeFile(here(), path, text, false); }

    void append(String path, String text) { fs.writeFile(here(), path, text, true); }

    String cat(String path) { return fs.readFile(here(), path); }

    void rm(String path) { fs.remove(here(), path, false); }

    void rmR(String path) { fs.remove(here(), path, true); }

    long du(String path) { return fs.du(here(), path); }

    List<String> find(String path, Spec spec) { return fs.find(here(), path, spec); }
}

        Shell other = new Shell(fs);
        other.cd("/var/log/nginx");

        System.out.println("  other shell pwd -> " + other.pwd());
        check(other.pwd().equals("/"), "detached cwd resets");
        check(sh.ls("/var").isEmpty(), "subtree gone");

        System.out.println("== 7. 8 threads x 200 appends under the read-write lock ==");
        ExecutorService pool = Executors.newFixedThreadPool(8);

"""In-memory file system: Composite tree, path parsing, wildcards, find with Specification."""
from __future__ import annotations

import threading
from abc import ABC, abstractmethod
from typing import Callable, Dict, List, Optional


class FsError(Exception):
    pass


# ---------- Composite ----------
class Node(ABC):
    def __init__(self, name: str, parent: Optional["Directory"]) -> None:
        self.name = name
        self.parent = parent          # None for root and for removed (detached) nodes

    @abstractmethod
    def is_dir(self) -> bool: ...

    @abstractmethod
    def size(self) -> int: ...

    def path(self) -> str:
        if self.parent is None:
            return "/"
        p = self.parent.path()
        return "/" + self.name if p == "/" else p + "/" + self.name


class FileNode(Node):
    def __init__(self, name: str, parent: "Directory") -> None:
        super().__init__(name, parent)
        self._parts: List[str] = []

    def is_dir(self) -> bool:
        return False

    def size(self) -> int:
        return sum(len(p) for p in self._parts)

    def write(self, text: str, append: bool) -> None:
        if not append:
            self._parts.clear()
        self._parts.append(text)

    def read(self) -> str:
        return "".join(self._parts)


class Directory(Node):
    def __init__(self, name: str, parent: Optional["Directory"]) -> None:
        super().__init__(name, parent)
        self._children: Dict[str, Node] = {}

    def is_dir(self) -> bool:
        return True

    def size(self) -> int:
        return sum(c.size() for c in self._children.values())

    def child(self, name: str) -> Optional[Node]:
        return self._children.get(name)

    def children(self) -> List[Node]:              # sorted by name = ls order
        return [self._children[k] for k in sorted(self._children)]

    def child_dirs(self) -> List["Directory"]:
        return [c for c in self.children() if isinstance(c, Directory)]

    def add(self, node: Node) -> None:
        self._children[node.name] = node

    def remove(self, name: str) -> None:
        node = self._children.pop(name, None)
        if node is not None:
            node.parent = None

    def is_empty(self) -> bool:
        return not self._children

    def up(self) -> "Directory":                   # '..' of root is root
        return self.parent if self.parent is not None else self


# ---------- glob ----------
def has_wildcard(s: str) -> bool:
    return "*" in s or "?" in s


def glob_match(pattern: str, text: str) -> bool:
    p = t = 0
    star, mark = -1, 0
    while t < len(text):
        if p < len(pattern) and pattern[p] in ("?", text[t]):
            p += 1
            t += 1
        elif p < len(pattern) and pattern[p] == "*":
            star, mark = p, t
            p += 1
        elif star >= 0:
            p = star + 1
            mark += 1
            t = mark
        else:
            return False
    while p < len(pattern) and pattern[p] == "*":
        p += 1
    return p == len(pattern)


# ---------- Specification ----------
class Spec:
    def __init__(self, fn: Callable[[Node], bool]) -> None:
        self._fn = fn

    def test(self, node: Node) -> bool:
        return self._fn(node)

    def __and__(self, other: "Spec") -> "Spec":
        return Spec(lambda n: self.test(n) and other.test(n))

    def __or__(self, other: "Spec") -> "Spec":
        return Spec(lambda n: self.test(n) or other.test(n))

    def __invert__(self) -> "Spec":
        return Spec(lambda n: not self.test(n))


def name_spec(glob: str) -> Spec:
    return Spec(lambda n: glob_match(glob, n.name))


def ext_spec(ext: str) -> Spec:
    return Spec(lambda n: not n.is_dir() and n.name.endswith("." + ext))


def min_size(b: int) -> Spec:
    return Spec(lambda n: not n.is_dir() and n.size() >= b)


IS_FILE = Spec(lambda n: not n.is_dir())
IS_DIR = Spec(lambda n: n.is_dir())


# ---------- the shared tree ----------
def segments(path: str) -> List[str]:
    return [s for s in path.split("/") if s]


class FileSystem:
    """Shared tree. One lock (Python has no stdlib RW lock); a Shell owns the cwd."""

    def __init__(self) -> None:
        self.root = Directory("", None)
        self._lock = threading.RLock()

    def _start(self, cwd: Directory, path: str) -> Directory:
        if not path:
            raise FsError("empty path")
        return self.root if path.startswith("/") else cwd

    def _walk(self, start: Directory, segs: List[str]) -> Optional[Node]:
        cur: Node = start
        for s in segs:
            if not isinstance(cur, Directory):
                raise FsError("not a directory: " + cur.path())
            if s == ".":
                continue
            if s == "..":
                cur = cur.up()
                continue
            nxt = cur.child(s)
            if nxt is None:
                return None
            cur = nxt
        return cur

    def _must_exist(self, cwd: Directory, path: str) -> Node:
        n = self._walk(self._start(cwd, path), segments(path))
        if n is None:
            raise FsError("no such file or directory: " + path)
        return n

    def _parent_of(self, cwd: Directory, path: str, segs: List[str]) -> Directory:
        p = self._walk(self._start(cwd, path), segs[:-1])
        if p is None:
            raise FsError("no such directory: " + path)
        if not isinstance(p, Directory):
            raise FsError("not a directory: " + p.path())
        return p

    @staticmethod
    def _check_name(name: str) -> None:
        if name in (".", "..") or has_wildcard(name):
            raise FsError("invalid name: " + name)

    def resolve_dir(self, cwd: Directory, path: str) -> Directory:
        with self._lock:
            n = self._must_exist(cwd, path)
            if not isinstance(n, Directory):
                raise FsError("not a directory: " + path)
            return n

    def resolve_dirs(self, cwd: Directory, pattern: str) -> List[Directory]:
        """Bare '*' = '.', '..' or any child dir; 'a*' = child dirs by glob. All matches, sorted."""
        with self._lock:
            frontier = [self._start(cwd, pattern)]
            for seg in segments(pattern):
                seen, nxt = set(), []
                for d in frontier:
                    if seg == ".":
                        cands = [d]
                    elif seg == "..":
                        cands = [d.up()]
                    elif seg == "*":
                        cands = [d, d.up()] + d.child_dirs()
                    elif has_wildcard(seg):
                        cands = [c for c in d.child_dirs() if glob_match(seg, c.name)]
                    else:
                        c = d.child(seg)
                        cands = [c] if isinstance(c, Directory) else []
                    for c in cands:
                        if id(c) not in seen:
                            seen.add(id(c))
                            nxt.append(c)
                frontier = nxt
                if not frontier:
                    break
            return sorted(frontier, key=lambda d: d.path())

    def mkdir(self, cwd: Directory, path: str, parents: bool = False) -> None:
        with self._lock:
            segs = segments(path)
            if not segs:
                raise FsError("mkdir: missing name")
            cur = self._start(cwd, path)
            for i, s in enumerate(segs):
                last = i == len(segs) - 1
                if s == ".":
                    continue
                if s == "..":
                    cur = cur.up()
                    continue
                c = cur.child(s)
                if c is None:
                    if not last and not parents:
                        raise FsError("no such directory: " + s)
                    self._check_name(s)
                    d = Directory(s, cur)
                    cur.add(d)
                    cur = d
                elif not isinstance(c, Directory):
                    raise FsError("not a directory: " + c.path())
                elif last and not parents:
                    raise FsError("already exists: " + c.path())
                else:
                    cur = c

    def ls(self, cwd: Directory, path: str) -> List[str]:
        with self._lock:
            segs = segments(path)
            if segs and has_wildcard(segs[-1]):
                return [n.name for n in self._parent_of(cwd, path, segs).children()
                        if glob_match(segs[-1], n.name)]
            n = self._must_exist(cwd, path)
            if not isinstance(n, Directory):
                return [n.name]
            return [c.name for c in n.children()]

    def write_file(self, cwd: Directory, path: str, text: str, append: bool) -> None:
        with self._lock:
            segs = segments(path)
            if not segs:
                raise FsError("not a file: " + path)
            d = self._parent_of(cwd, path, segs)
            n = d.child(segs[-1])
            if n is None:
                self._check_name(segs[-1])
                n = FileNode(segs[-1], d)
                d.add(n)
            elif isinstance(n, Directory):
                raise FsError("is a directory: " + n.path())
            n.write(text, append)

    def read_file(self, cwd: Directory, path: str) -> str:
        with self._lock:
            n = self._must_exist(cwd, path)
            if isinstance(n, Directory):
                raise FsError("is a directory: " + path)
            return n.read()

    def remove(self, cwd: Directory, path: str, recursive: bool = False) -> None:
        with self._lock:
            n = self._must_exist(cwd, path)
            if n is self.root:
                raise FsError("cannot remove /")
            if isinstance(n, Directory) and not recursive and not n.is_empty():
                raise FsError("directory not empty: " + n.path())
            n.parent.remove(n.name)

    def du(self, cwd: Directory, path: str) -> int:
        with self._lock:
            return self._must_exist(cwd, path).size()

    def find(self, cwd: Directory, path: str, spec: Spec) -> List[str]:
        with self._lock:
            out, stack = [], [self._must_exist(cwd, path)]
            while stack:
                n = stack.pop()
                if spec.test(n):
                    out.append(n.path())
                if isinstance(n, Directory):
                    stack.extend(reversed(n.children()))
            return out

    def is_attached(self, n: Node) -> bool:
        with self._lock:
            while n.parent is not None:
                n = n.parent
            return n is self.root


class Shell:
    """One session: owns cwd, delegates to the shared FileSystem."""

    def __init__(self, fs: FileSystem) -> None:
        self.fs = fs
        self.cwd = fs.root

    def _here(self) -> Directory:
        if not self.fs.is_attached(self.cwd):
            self.cwd = self.fs.root
        return self.cwd

    def pwd(self) -> str:
        return self._here().path()

    def cd(self, path: str) -> None:
        if not has_wildcard(path):
            self.cwd = self.fs.resolve_dir(self._here(), path)
            return
        matches = self.fs.resolve_dirs(self._here(), path)
        if not matches:
            raise FsError("no such directory: " + path)
        self.cwd = matches[0]

    def mkdir(self, path: str, parents: bool = False) -> None:
        self.fs.mkdir(self._here(), path, parents)

    def ls(self, path: str = ".") -> List[str]:
        return self.fs.ls(self._here(), path)

    def write(self, path: str, text: str) -> None:
        self.fs.write_file(self._here(), path, text, False)

    def append(self, path: str, text: str) -> None:
        self.fs.write_file(self._here(), path, text, True)

    def cat(self, path: str) -> str:
        return self.fs.read_file(self._here(), path)

    def rm(self, path: str, recursive: bool = False) -> None:
        self.fs.remove(self._here(), path, recursive)

    def du(self, path: str) -> int:
        return self.fs.du(self._here(), path)

    def find(self, path: str, spec: Spec) -> List[str]:
        return self.fs.find(self._here(), path, spec)

import unittest

from fs import FileSystem, FsError, Shell, IS_DIR, IS_FILE, ext_spec, glob_match, min_size, name_spec


class FileSystemTest(unittest.TestCase):
    def setUp(self) -> None:
        self.fs = FileSystem()
        self.sh = Shell(self.fs)
        self.sh.mkdir("/a/b/c", parents=True)
        self.sh.mkdir("/a/bx", parents=True)
        self.sh.write("/a/b/f.log", "12345")
        self.sh.write("/a/g.txt", "1")

    def test_glob(self) -> None:
        self.assertTrue(glob_match("*.log", "app.log"))
        self.assertTrue(glob_match("a?c", "abc"))
        self.assertTrue(glob_match("*", ""))
        self.assertFalse(glob_match("*.log", "app.txt"))
        self.assertTrue(glob_match("a*b*c", "aXbYbZc"))

    def test_pwd_and_relative_cd(self) -> None:
        self.sh.cd("/a/b/c")
        self.sh.cd("../../bx")
        self.assertEqual(self.sh.pwd(), "/a/bx")
        self.sh.cd("/..")                     # '..' of root is root
        self.assertEqual(self.sh.pwd(), "/")

    def test_ls_sorted_and_file(self) -> None:
        self.assertEqual(self.sh.ls("/a"), ["b", "bx", "g.txt"])
        self.assertEqual(self.sh.ls("/a/g.txt"), ["g.txt"])
        self.assertEqual(self.sh.ls("/a/b*"), ["b", "bx"])

    def test_cd_star_matches_dot_dotdot_child(self) -> None:
        self.sh.cd("/a/b")
        got = [d.path() for d in self.fs.resolve_dirs(self.sh.cwd, "*")]
        self.assertEqual(got, ["/a", "/a/b", "/a/b/c"])
        self.sh.cd("*/bx")                    # only '..' leads on to bx
        self.assertEqual(self.sh.pwd(), "/a/bx")

    def test_cd_no_match(self) -> None:

    def test_mkdir_errors(self) -> None:

            self.sh.mkdir("/a")
        with self.assertRaises(FsError):
            self.sh.mkdir("/q/r")
        with self.assertRaises(FsError):
            self.sh.mkdir("/a/g.txt/x", parents=True)

    def test_files(self) -> None:
