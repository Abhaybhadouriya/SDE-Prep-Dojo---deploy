A runnable demo of every idea above: instances and a static counter, aliasing, pass-by-value of references, identity vs equality, a correct hashCode, boxed-integer comparison and null. The C++ tab shows stack vs heap objects and the three ways to pass a parameter; watch when each destructor runs. Sources are in lld-code/lld-oop-classes/.

// lld-oop-classes: blueprint vs instance, stack vs heap, references, identity vs equality, null.
// Run: java MemoryDemo.java
import java.util.HashSet;
import java.util.Objects;
import java.util.Set;

public class MemoryDemo {
    public static void main(String[] args) {
        // 1. One blueprint, many instances. Each 'new' makes a separate heap object.
        Ticket a = new Ticket("T-1", "A12");
        Ticket b = new Ticket("T-2", "B07");
        System.out.println("tickets created: " + Ticket.created());           // static: 2

        // 2. References are copied, objects are not.
        Ticket alias = a;                 // second reference, same heap object
        alias.moveTo("C01");
        System.out.println("a.seat after alias.moveTo: " + a.seat());       // C01

        // 3. Java passes the reference BY VALUE.
        reassign(a);
        System.out.println("after reassign(a): " + a.id());                 // still T-1
        mutate(a);
        System.out.println("after mutate(a): " + a.seat());                 // D99

        // 4. Identity (==) vs equality (equals).
        Ticket same = new Ticket("T-1", "D99");
        System.out.println("a == same      : " + (a == same));              // false: two objects
        System.out.println("a.equals(same) : " + a.equals(same));           // true: same id

        // 5. equals without hashCode breaks hash collections; ours has both.
        Set<Ticket> set = new HashSet<>();
        set.add(a);
        System.out.println("set.contains(same): " + set.contains(same));    // true

        // 6. Primitives live in the variable itself; boxed Integers are objects.
        Integer x = 1000, y = 1000;
        System.out.println("Integer 1000 == 1000 : " + (x == y) + "  equals: " + x.equals(y));

        // 7. null: a reference that points at nothing.
        Ticket none = null;
        try { none.seat(); } catch (NullPointerException e) { System.out.println("NPE caught: calling a method on null"); }
        System.out.println("Objects.equals(none, a): " + Objects.equals(none, a)); // null-safe
        b = null;   // the T-2 object is now unreachable; the GC may reclaim it later
    }

    static void reassign(Ticket t) { t = new Ticket("T-9", "Z00"); }  // changes the local copy only
    static void mutate(Ticket t)   { t.moveTo("D99"); }              // changes the shared object
}

final class Ticket {
    private static int created = 0;      // one copy per class, not per object
    private final String id;             // identity in the domain
    private String seat;                 // state that may change

    Ticket(String id, String seat) { this.id = id; this.seat = seat; created++; }
    String id()   { return id; }
    String seat() { return seat; }
    void moveTo(String newSeat) { this.seat = newSeat; }
    static int created() { return created; }

    @Override public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Ticket other)) return false;   // Java 16+ pattern matching
        return id.equals(other.id);                       // equal = same ticket id
    }
    @Override public int hashCode() { return id.hashCode(); }
    @Override public String toString() { return "Ticket(" + id + "," + seat + ")"; }
}

// lld-oop-classes: where objects live in C++14 (stack, heap, static), pointers vs references, == vs identity.
// Build: g++ -std=c++14 -Wall -o memory memory.cpp && ./memory
#include <iostream>
#include <memory>
#include <string>

struct Point {
    int x, y;
    static int created;                         // static storage, one per class
    Point(int x_, int y_) : x(x_), y(y_) { ++created; }
    ~Point() { std::cout << "  ~Point(" << x << "," << y << ")\n"; }
    bool operator==(const Point& o) const { return x == o.x && y == o.y; }   // equality
};
int Point::created = 0;

void byValue(Point p)      { p.x = 100; }       // works on a copy
void byRef(Point& p)       { p.x = 200; }       // works on the caller's object
void byPtr(Point* p)       { if (p) p->x = 300; } // may be null, so check

int main() {
    {
        Point s(1, 2);                          // stack: freed at the closing brace
        byValue(s); std::cout << "after byValue: " << s.x << "\n";   // 1
        byRef(s);   std::cout << "after byRef:   " << s.x << "\n";   // 200
        byPtr(&s);  std::cout << "after byPtr:   " << s.x << "\n";   // 300
        byPtr(nullptr);
        Point t(300, 2);
        std::cout << "s == t (equal values): " << (s == t) << ", &s == &t (same object): " << (&s == &t) << "\n";
        std::cout << "leaving block...\n";
    }
    auto h = std::make_unique<Point>(7, 8);    // heap, owned by a smart pointer
    Point* raw = new Point(9, 9);              // heap, owned by nobody: you must delete
    std::cout << "created so far: " << Point::created << "\n";
    delete raw;                                // forget this and it leaks
    std::cout << "end of main...\n";
    return 0;                                  // h's destructor frees (7,8)
}

# lld-oop-classes: Python names are references; 'is' = identity, '==' = equality.
# Run: python memory_demo.py
class Ticket:
    created = 0                                   # class attribute (shared)

    def __init__(self, tid, seat):
        self.tid, self.seat = tid, seat           # instance attributes
        Ticket.created += 1

    def __eq__(self, other):
        return isinstance(other, Ticket) and self.tid == other.tid

    def __hash__(self):                           # keep hash consistent with __eq__
        return hash(self.tid)

a = Ticket("T-1", "A12")

print("a is same:", a is same, "| a == same:", a == same)   # False | True
print("in set:", same in {a})                     # True
print("created:", Ticket.created)                 # 2
none = None

Two things in the C++ output are worth a second look. First, ~Point(100,2) prints before "after byValue": that is the copy made for the by-value parameter being destroyed when byValue returns. Second, "created so far" is 4, not 5: the copy was made by the compiler-generated copy constructor, which does not run our counting constructor. That is the doorway to the rule of three, covered on Constructors & Copy Semantics.
