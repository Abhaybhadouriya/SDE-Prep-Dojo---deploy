// Coffee machine LLD: recipes, ingredient inventory, add-on Decorator, Factory, N concurrent outlets.
// Build and run:  javac -d out Main.java && java -cp out Main
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.function.UnaryOperator;

// ---------- model ----------
/** Immutable recipe: ingredient name -> quantity (ml or g). */
final class Recipe {
    private final Map<String, Integer> parts;
    Recipe(Map<String, Integer> parts) {
        for (Map.Entry<String, Integer> e : parts.entrySet())
            if (e.getValue() <= 0) throw new IllegalArgumentException("bad quantity for " + e.getKey());
        this.parts = Collections.unmodifiableMap(new TreeMap<>(parts));
    }
    Map<String, Integer> parts() { return parts; }
    /** A new recipe with the extra quantities added (used by add-ons). */
    Recipe plus(Map<String, Integer> extra) {
        Map<String, Integer> m = new TreeMap<>(parts);
        extra.forEach((k, v) -> m.merge(k, v, Integer::sum));
        return new Recipe(m);
    }
    @Override public String toString() { return parts.toString(); }
}

/** What the customer gets. Base drinks and add-on wrappers both implement it. */
interface Beverage {
    String name();
    int price();          // whole rupees, never floating point
    Recipe recipe();
}

final class BaseBeverage implements Beverage {
    private final String name; private final int price; private final Recipe recipe;
    BaseBeverage(String name, int price, Recipe recipe) { this.name = name; this.price = price; this.recipe = recipe; }
    public String name() { return name; }
    public int price() { return price; }
    public Recipe recipe() { return recipe; }
}

/** Decorator: wraps any Beverage, adds a label, a price and extra ingredients. */
class AddOn implements Beverage {
    private final Beverage inner; private final String label; private final int extraPrice;
    private final Map<String, Integer> extra;
    AddOn(Beverage inner, String label, int extraPrice, Map<String, Integer> extra) {
        this.inner = inner; this.label = label; this.extraPrice = extraPrice; this.extra = extra;
    }
    public String name() { return inner.name() + " + " + label; }
    public int price() { return inner.price() + extraPrice; }
    public Recipe recipe() { return inner.recipe().plus(extra); }
}

final class ExtraShot extends AddOn {
    ExtraShot(Beverage b) { super(b, "extra shot", 40, Map.of("coffee", 9, "water", 15)); }
}
final class ExtraMilk extends AddOn {
    ExtraMilk(Beverage b) { super(b, "extra milk", 20, Map.of("milk", 50)); }
}
final class Sugar extends AddOn {
    Sugar(Beverage b) { super(b, "sugar", 0, Map.of("sugar", 5)); }
}
final class Caramel extends AddOn {
    Caramel(Beverage b) { super(b, "caramel", 30, Map.of("caramel", 15)); }
}

class CoffeeException extends RuntimeException {
    private static final long serialVersionUID = 1L;
    CoffeeException(String msg) { super(msg); }
}

// ---------- factory ----------
/** Factory + registry: menu items and add-ons are data, so a new drink is one register() call. */
final class BeverageFactory {
    private final Map<String, Beverage> menu = new LinkedHashMap<>();
    private final Map<String, UnaryOperator<Beverage>> addOns = new LinkedHashMap<>();

    void registerDrink(String name, int price, Map<String, Integer> recipe) {
        menu.put(name, new BaseBeverage(name, price, new Recipe(recipe)));
    }
    void registerAddOn(String name, UnaryOperator<Beverage> wrap) { addOns.put(name, wrap); }

    Beverage create(String drink, List<String> extras) {
        Beverage b = menu.get(drink);
        if (b == null) throw new CoffeeException("unknown drink " + drink);
        for (String x : extras) {
            UnaryOperator<Beverage> wrap = addOns.get(x);
            if (wrap == null) throw new CoffeeException("unknown add-on " + x);
            b = wrap.apply(b);                       // decorate, one layer per add-on
        }
        return b;
    }
    Set<String> drinks() { return menu.keySet(); }

    static BeverageFactory standard() {
        BeverageFactory f = new BeverageFactory();
        f.registerDrink("espresso", 120, Map.of("water", 30, "coffee", 18));
        f.registerDrink("americano", 140, Map.of("water", 150, "coffee", 18));
        f.registerDrink("latte", 180, Map.of("water", 30, "coffee", 18, "milk", 150));
        f.registerDrink("cappuccino", 170, Map.of("water", 30, "coffee", 18, "milk", 100));
        f.registerAddOn("shot", ExtraShot::new);
        f.registerAddOn("milk", ExtraMilk::new);
        f.registerAddOn("sugar", Sugar::new);
        f.registerAddOn("caramel", Caramel::new);
        return f;
    }
}

// ---------- inventory ----------
interface LowStockListener { void onLow(String ingredient, int left); }

/** All-or-nothing reservation under one lock; low-stock events fire outside the lock. */
final class Inventory {
    private final Map<String, Integer> stock = new TreeMap<>();
    private final Map<String, Integer> lowMark = new HashMap<>();
    private final List<LowStockListener> listeners = new CopyOnWriteArrayList<>();

    synchronized void add(String ingredient, int qty, int lowAt) {
        if (qty < 0) throw new IllegalArgumentException("negative refill");
        stock.merge(ingredient, qty, Integer::sum);
        lowMark.put(ingredient, lowAt);
    }
    void refill(String ingredient, int qty) {
        synchronized (this) {
            if (!stock.containsKey(ingredient)) throw new CoffeeException("unknown ingredient " + ingredient);
            if (qty <= 0) throw new IllegalArgumentException("refill must be positive");
            stock.merge(ingredient, qty, Integer::sum);
        }
    }
    void subscribe(LowStockListener l) { listeners.add(l); }

    /** Reserve every ingredient of the recipe, or none. Returns an empty list on success, else the reasons. */
    List<String> tryReserve(Recipe r) {
        List<String> problems = new ArrayList<>();
        List<String[]> crossed = new ArrayList<>();
        synchronized (this) {
            for (Map.Entry<String, Integer> e : r.parts().entrySet()) {
                Integer have = stock.get(e.getKey());
                if (have == null || have == 0) problems.add(e.getKey() + " is not available");
                else if (have < e.getValue()) problems.add(e.getKey() + " is not sufficient (need " + e.getValue() + ", have " + have + ")");
            }
            if (!problems.isEmpty()) return problems;          // nothing was touched
            for (Map.Entry<String, Integer> e : r.parts().entrySet()) {
                int before = stock.get(e.getKey()), after = before - e.getValue();
                stock.put(e.getKey(), after);
                int mark = lowMark.getOrDefault(e.getKey(), 0);
                if (before >= mark && after < mark) crossed.add(new String[]{e.getKey(), String.valueOf(after)});
            }
        }
        for (String[] c : crossed)                           // never call listeners while holding the lock
            for (LowStockListener l : listeners) l.onLow(c[0], Integer.parseInt(c[1]));
        return problems;
    }
    /** Put a reservation back, e.g. when brewing fails before anything was poured. */
    synchronized void release(Recipe r) { r.parts().forEach((k, v) -> stock.merge(k, v, Integer::sum)); }
    synchronized int level(String ingredient) { return stock.getOrDefault(ingredient, 0); }
    synchronized List<String> lowIngredients() {
        List<String> low = new ArrayList<>();
        stock.forEach((k, v) -> { if (v < lowMark.getOrDefault(k, 0)) low.add(k + "=" + v); });
        return low;
    }
    synchronized Map<String, Integer> snapshot() { return new TreeMap<>(stock); }
}

// ---------- outlets ----------
/** The hardware that pours a drink; swapped for a fake in tests. */
interface Brewer { void brew(Beverage b) throws Exception; }

final class Result {
    final String drink; final boolean served; final String detail;
    Result(String drink, boolean served, String detail) { this.drink = drink; this.served = served; this.detail = detail; }
    @Override public String toString() { return served ? drink + " is prepared (Rs" + detail + ")" : drink + " cannot be prepared: " + detail; }
}

final class CoffeeMachine implements AutoCloseable {
    private final BeverageFactory factory;
    private final Inventory inventory;
    private final Brewer brewer;
    private final ExecutorService outlets;            // N threads = N outlets = at most N in parallel
    private final AtomicInteger busy = new AtomicInteger(), peakBusy = new AtomicInteger();
    private final AtomicInteger revenue = new AtomicInteger();
    private final AtomicInteger pending = new AtomicInteger();   // reserved, not yet poured
    private volatile boolean closed;

    CoffeeMachine(int outletCount, BeverageFactory factory, Inventory inventory, Brewer brewer) {
        if (outletCount <= 0) throw new IllegalArgumentException("need at least one outlet");
        this.factory = factory; this.inventory = inventory; this.brewer = brewer;
        this.outlets = Executors.newFixedThreadPool(outletCount);
    }

    /** Validate and reserve now (caller's thread), brew later on a free outlet. */
    Future<Result> order(String drink, String... extras) {
        final Beverage b;
        try { b = factory.create(drink, Arrays.asList(extras)); }
        catch (CoffeeException e) { return CompletableFuture.completedFuture(new Result(drink, false, e.getMessage())); }
        if (closed) return CompletableFuture.completedFuture(new Result(b.name(), false, "machine closed"));
        List<String> problems = inventory.tryReserve(b.recipe());
        if (!problems.isEmpty())
            return CompletableFuture.completedFuture(new Result(b.name(), false, String.join(", ", problems)));
        pending.incrementAndGet();
        try {
            return outlets.submit(() -> pour(b));
        } catch (RejectedExecutionException e) {      // close() won the race after our closed check
            pending.decrementAndGet();
            inventory.release(b.recipe());              // give the reservation back: no stock leak
            return CompletableFuture.completedFuture(new Result(b.name(), false, "machine closed"));
        }
    }

    private Result pour(Beverage b) {
        int now = busy.incrementAndGet();
        peakBusy.accumulateAndGet(now, Math::max);
        try {
            brewer.brew(b);
            revenue.addAndGet(b.price());
            return new Result(b.name(), true, String.valueOf(b.price()));
        } catch (Exception e) {
            inventory.release(b.recipe());              // simple policy: a failed brew wasted nothing
            return new Result(b.name(), false, "outlet error: " + e.getMessage());
        } finally {
            busy.decrementAndGet();
            pending.decrementAndGet();
        }
    }

    int peakParallel() { return peakBusy.get(); }
    int revenue() { return revenue.get(); }
    Inventory inventory() { return inventory; }
    @Override public void close() {                   // drain the outlets: every accepted (reserved) drink is poured
        closed = true;                                  // new orders are refused before they reserve anything
        outlets.shutdown();
        try {
            while (!outlets.awaitTermination(10, TimeUnit.SECONDS))
                System.err.println("close(): still draining " + pending.get() + " accepted order(s)");
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            System.err.println("close() interrupted with " + pending.get() + " accepted order(s) not drained");
        }
    }
}

// ---------- demo driver with checks ----------
public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }

    static Inventory stocked() {
        Inventory inv = new Inventory();
        inv.add("water", 1000, 200);
        inv.add("coffee", 120, 40);
        inv.add("milk", 500, 150);
        inv.add("sugar", 30, 10);
        inv.add("caramel", 0, 15);          // known to the machine, but empty
        return inv;
    }

    public static void main(String[] args) throws Exception {
        BeverageFactory menu = BeverageFactory.standard();
        Brewer slow = b -> Thread.sleep(50);    // every drink takes the same time

        System.out.println("1) decorators stack price and recipe");
        Beverage fancy = menu.create("latte", List.of("shot", "milk", "sugar"));
        System.out.println("  " + fancy.name() + " = Rs" + fancy.price());
        System.out.println("  recipe " + fancy.recipe());
        check(fancy.price() == 180 + 40 + 20 + 0, "decorated price");
        check(fancy.recipe().parts().get("milk") == 200 && fancy.recipe().parts().get("coffee") == 27, "decorated recipe");

        Inventory inv = stocked();
        List<String> alerts = new CopyOnWriteArrayList<>();
        inv.subscribe((ing, left) -> alerts.add(ing + " low (" + left + " left)"));
        try (CoffeeMachine m = new CoffeeMachine(3, menu, inv, slow)) {
            System.out.println("2) six orders on 3 outlets");
            List<Future<Result>> fs = new ArrayList<>();
            String[] drinks = {"espresso", "latte", "cappuccino", "americano", "espresso", "latte"};
            for (String d : drinks) fs.add(m.order(d));
            for (Future<Result> f : fs) System.out.println("  " + f.get());
            System.out.println("  peak parallel brews = " + m.peakParallel());
            check(m.peakParallel() <= 3, "never more than N in parallel");
            check(inv.level("coffee") == 120 - 6 * 18, "coffee consumed");

            System.out.println("3) not enough coffee: nothing is taken");
            Map<String, Integer> before = inv.snapshot();
            Result r = m.order("latte", "shot").get();
            System.out.println("  " + r);
            check(!r.served && inv.snapshot().equals(before), "all-or-nothing reservation");

            System.out.println("4) missing and unknown things");
            System.out.println("  " + m.order("espresso", "caramel").get());
            System.out.println("  " + m.order("mocha").get());
            System.out.println("  " + m.order("espresso", "whipped cream").get());

            System.out.println("5) low indicator, refill, retry");
            System.out.println("  alerts " + alerts);
            System.out.println("  low now " + inv.lowIngredients());
            check(alerts.contains("coffee low (30 left)") && alerts.contains("milk low (100 left)"), "low alerts fired once");
            inv.refill("coffee", 200);
            inv.refill("caramel", 60);
            System.out.println("  " + m.order("espresso", "shot", "caramel").get());
            System.out.println("  low now " + inv.lowIngredients());
            check(m.revenue() == 2 * 120 + 2 * 180 + 170 + 140 + 190, "revenue matches served drinks");
        }

        System.out.println("6) 8 threads x 25 espresso orders race for 20 shots of coffee");
        Inventory race = new Inventory();
        race.add("water", 10_000, 0);
        race.add("coffee", 20 * 18, 0);
        try (CoffeeMachine m = new CoffeeMachine(4, menu, race, b -> Thread.sleep(1))) {
            ExecutorService customers = Executors.newFixedThreadPool(8);
            List<Future<Future<Result>>> calls = new ArrayList<>();
            for (int i = 0; i < 200; i++) calls.add(customers.submit(() -> m.order("espresso")));
            int served = 0, refused = 0;
            for (Future<Future<Result>> c : calls) { if (c.get().get().served) served++; else refused++; }
            customers.shutdown();
            System.out.println("  served " + served + ", refused " + refused + ", coffee left " + race.level("coffee")
                    + ", peak parallel " + m.peakParallel());
            check(served == 20 && refused == 180 && race.level("coffee") == 0, "never oversold");
            check(m.peakParallel() <= 4, "outlet limit under load");
        }

        System.out.println("7) a failing outlet gives the ingredients back");
        Inventory inv2 = stocked();
        try (CoffeeMachine m = new CoffeeMachine(1, menu, inv2, b -> { throw new IllegalStateException("jam"); })) {
            Result r = m.order("espresso").get();
            System.out.println("  " + r + "; coffee " + inv2.level("coffee"));
            check(!r.served && inv2.level("coffee") == 120, "released on failure");
        }

        System.out.println("8) ordering after close() reserves nothing");
        Inventory inv3 = stocked();
        CoffeeMachine shut = new CoffeeMachine(1, menu, inv3, b -> { });
        shut.close();
        Map<String, Integer> stock = inv3.snapshot();
        Result late = shut.order("latte").get();
        System.out.println("  " + late + "; stock unchanged " + inv3.snapshot().equals(stock));
        check(!late.served && inv3.snapshot().equals(stock), "closed machine does not leak stock");
        System.out.println("All " + passed + " checks passed");
    }
}
"""Coffee machine LLD: recipes, ingredient inventory, add-on Decorator, Factory, N concurrent outlets.

Run the demo:   python main.py
Run the tests:  python -m unittest test_main
"""
from __future__ import annotations

import threading
import time
from concurrent.futures import Future, ThreadPoolExecutor
from dataclasses import dataclass
from types import MappingProxyType
from typing import Callable, Dict, List, Mapping, Optional


def check(cond, msg="check failed"):
    """Like assert, but never stripped by python -O: the call inside it always runs."""
    if not cond:
        raise AssertionError(msg)


class CoffeeError(Exception):
    pass


# ---------- model ----------
class Recipe:
    """Immutable: ingredient name -> quantity (ml or g)."""

    def __init__(self, parts: Mapping[str, int]):
        for k, v in parts.items():
            if v <= 0:
                raise ValueError(f"bad quantity for {k}")
        self._parts = MappingProxyType(dict(sorted(parts.items())))

    @property
    def parts(self) -> Mapping[str, int]:
        return self._parts

    def plus(self, extra: Mapping[str, int]) -> "Recipe":
        merged = dict(self._parts)
        for k, v in extra.items():
            merged[k] = merged.get(k, 0) + v
        return Recipe(merged)

    def __repr__(self) -> str:
        return str(dict(self._parts))


class Beverage:
    """What the customer gets. Base drinks and add-on wrappers share this interface."""

    def name(self) -> str: raise NotImplementedError
    def price(self) -> int: raise NotImplementedError       # whole rupees
    def recipe(self) -> Recipe: raise NotImplementedError


class BaseBeverage(Beverage):
    def __init__(self, name: str, price: int, recipe: Recipe):
        self._name, self._price, self._recipe = name, price, recipe

    def name(self) -> str: return self._name
    def price(self) -> int: return self._price
    def recipe(self) -> Recipe: return self._recipe


class AddOn(Beverage):
    """Decorator: wraps any Beverage, adds a label, a price and extra ingredients."""
    label = ""
    extra_price = 0
    extra: Mapping[str, int] = {}

    def __init__(self, inner: Beverage):
        self._inner = inner

    def name(self) -> str: return f"{self._inner.name()} + {self.label}"
    def price(self) -> int: return self._inner.price() + self.extra_price
    def recipe(self) -> Recipe: return self._inner.recipe().plus(self.extra)


class ExtraShot(AddOn):
    label, extra_price, extra = "extra shot", 40, {"coffee": 9, "water": 15}


class ExtraMilk(AddOn):
    label, extra_price, extra = "extra milk", 20, {"milk": 50}


class Sugar(AddOn):
    label, extra_price, extra = "sugar", 0, {"sugar": 5}


class Caramel(AddOn):
    label, extra_price, extra = "caramel", 30, {"caramel": 15}


# ---------- factory ----------
class BeverageFactory:
    """Factory + registry: a new drink or add-on is one register call, no if-else chain."""

    def __init__(self) -> None:
        self._menu: Dict[str, Beverage] = {}
        self._addons: Dict[str, Callable[[Beverage], Beverage]] = {}

    def register_drink(self, name: str, price: int, recipe: Mapping[str, int]) -> None:
        self._menu[name] = BaseBeverage(name, price, Recipe(recipe))

    def register_addon(self, name: str, wrap: Callable[[Beverage], Beverage]) -> None:
        self._addons[name] = wrap

    def create(self, drink: str, extras: List[str] = ()) -> Beverage:
        b = self._menu.get(drink)
        if b is None:
            raise CoffeeError(f"unknown drink {drink}")
        for x in extras:
            wrap = self._addons.get(x)
            if wrap is None:
                raise CoffeeError(f"unknown add-on {x}")
            b = wrap(b)                        # one decorator layer per add-on
        return b

    @staticmethod
    def standard() -> "BeverageFactory":
        f = BeverageFactory()
        f.register_drink("espresso", 120, {"water": 30, "coffee": 18})
        f.register_drink("americano", 140, {"water": 150, "coffee": 18})
        f.register_drink("latte", 180, {"water": 30, "coffee": 18, "milk": 150})
        f.register_drink("cappuccino", 170, {"water": 30, "coffee": 18, "milk": 100})
        f.register_addon("shot", ExtraShot)
        f.register_addon("milk", ExtraMilk)
        f.register_addon("sugar", Sugar)
        f.register_addon("caramel", Caramel)
        return f


# ---------- inventory ----------
class Inventory:
    """All-or-nothing reservation under one lock; low-stock listeners run outside the lock."""

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._stock: Dict[str, int] = {}
        self._low_mark: Dict[str, int] = {}
        self._listeners: List[Callable[[str, int], None]] = []

    def add(self, ingredient: str, qty: int, low_at: int) -> None:
        with self._lock:
            self._stock[ingredient] = self._stock.get(ingredient, 0) + qty
            self._low_mark[ingredient] = low_at

    def refill(self, ingredient: str, qty: int) -> None:
        with self._lock:
            if ingredient not in self._stock:
                raise CoffeeError(f"unknown ingredient {ingredient}")
            if qty <= 0:
                raise ValueError("refill must be positive")
            self._stock[ingredient] += qty

    def subscribe(self, listener: Callable[[str, int], None]) -> None:
        self._listeners.append(listener)

    def try_reserve(self, recipe: Recipe) -> List[str]:
        """Take every ingredient or none. Returns [] on success, else the reasons."""
        crossed = []
        with self._lock:
            problems = []
            for k, need in recipe.parts.items():
                have = self._stock.get(k, 0)
                if have == 0:
                    problems.append(f"{k} is not available")
                elif have < need:
                    problems.append(f"{k} is not sufficient (need {need}, have {have})")
            if problems:
                return problems                     # nothing was touched
            for k, need in recipe.parts.items():
                before = self._stock[k]
                after = before - need
                self._stock[k] = after
                mark = self._low_mark.get(k, 0)
                if before >= mark > after:
                    crossed.append((k, after))
        for k, left in crossed:                     # never call out while holding the lock
            for listener in self._listeners:
                listener(k, left)
        return []

    def release(self, recipe: Recipe) -> None:
        with self._lock:
            for k, v in recipe.parts.items():
                self._stock[k] = self._stock.get(k, 0) + v

    def level(self, ingredient: str) -> int:
        with self._lock:
            return self._stock.get(ingredient, 0)

    def low_ingredients(self) -> List[str]:
        with self._lock:
            return [f"{k}={v}" for k, v in sorted(self._stock.items()) if v < self._low_mark.get(k, 0)]

    def snapshot(self) -> Dict[str, int]:
        with self._lock:
            return dict(self._stock)


# ---------- outlets ----------
@dataclass(frozen=True)
class Result:
    drink: str
    served: bool
    detail: str

    def __str__(self) -> str:
        if self.served:
            return f"{self.drink} is prepared (Rs{self.detail})"
        return f"{self.drink} cannot be prepared: {self.detail}"


def _done(result: Result) -> "Future[Result]":
    f: Future = Future()
    f.set_result(result)
    return f


class CoffeeMachine:
    """N worker threads = N outlets, so at most N drinks brew in parallel."""

    def __init__(self, outlets: int, factory: BeverageFactory, inventory: Inventory,
                 brewer: Callable[[Beverage], None]):
        if outlets <= 0:
            raise ValueError("need at least one outlet")
        self.factory, self.inventory, self._brewer = factory, inventory, brewer
        self._pool = ThreadPoolExecutor(max_workers=outlets, thread_name_prefix="outlet")
        self._lock = threading.Lock()
        self._busy = 0
        self.peak_parallel = 0
        self.revenue = 0
        self._closed = False

    def order(self, drink: str, *extras: str) -> "Future[Result]":
        """Validate and reserve now (caller's thread), brew later on a free outlet."""
        try:
            b = self.factory.create(drink, list(extras))
        except CoffeeError as e:
            return _done(Result(drink, False, str(e)))
        if self._closed:                            # refuse before reserving anything
            return _done(Result(b.name(), False, "machine closed"))
        problems = self.inventory.try_reserve(b.recipe())
        if problems:
            return _done(Result(b.name(), False, ", ".join(problems)))
        try:
            return self._pool.submit(self._pour, b)
        except RuntimeError:                        # pool shut down after our check: give the stock back
            self.inventory.release(b.recipe())
            return _done(Result(b.name(), False, "machine closed"))

    def _pour(self, b: Beverage) -> Result:
        with self._lock:
            self._busy += 1
            self.peak_parallel = max(self.peak_parallel, self._busy)
        try:
            self._brewer(b)
            with self._lock:
                self.revenue += b.price()
            return Result(b.name(), True, str(b.price()))
        except Exception as e:                      # hardware fault: give the ingredients back
            self.inventory.release(b.recipe())
            return Result(b.name(), False, f"outlet error: {e}")
        finally:
            with self._lock:
                self._busy -= 1

    def close(self) -> None:
        self._closed = True
        self._pool.shutdown(wait=True)              # every accepted (reserved) drink is still poured

    def __enter__(self) -> "CoffeeMachine":
        return self

    def __exit__(self, *exc) -> None:
        self.close()

// Build: g++ -std=c++14 -Wall -Wextra main.cpp -o main && ./main
// Single-threaded on purpose (this MinGW toolchain has no std::thread and no pthread.h); outlets are
// modelled as a counted resource. The comments marked THREADED BUILD are a sketch of the pthread code a POSIX
// build would add: a sketch, not compiled, and it needs a posix-thread toolchain (Linux/WSL g++ -pthread).

using Parts = std::map<std::string, int>;   // ingredient -> quantity

struct CoffeeError : std::runtime_error {
    explicit CoffeeError(const std::string& m) : std::runtime_error(m) {}
};

class Recipe {
    Parts parts_;
public:
    explicit Recipe(Parts p) : parts_(std::move(p)) {
        for (const auto& kv : parts_)
            if (kv.second <= 0) throw std::invalid_argument("bad quantity for " + kv.first);
    }
    const Parts& parts() const { return parts_; }
    Recipe plus(const Parts& extra) const {
        Parts m = parts_;
        for (const auto& kv : extra) m[kv.first] += kv.second;
        return Recipe(m);
    }
};

std::string show(const Parts& p) {
    std::ostringstream os; os << "{";
    bool first = true;
    for (const auto& kv : p) { os << (first ? "" : ", ") << kv.first << "=" << kv.second; first = false; }
    return os.str() + "}";
}

// The component interface: like a struct of function pointers that every layer fills in.
class Beverage {
public:
    virtual ~Beverage() = default;
    virtual std::string name() const = 0;
    virtual int price() const = 0;
    virtual Recipe recipe() const = 0;
};
using BevPtr = std::shared_ptr<const Beverage>;

class BaseBeverage : public Beverage {
    std::string name_; int price_; Recipe recipe_;
public:
    BaseBeverage(std::string n, int p, Recipe r) : name_(std::move(n)), price_(p), recipe_(std::move(r)) {}
    std::string name() const override { return name_; }
    int price() const override { return price_; }
    Recipe recipe() const override { return recipe_; }
};

// Decorator: holds the inner beverage and forwards, adding its own part.
class AddOn : public Beverage {
    BevPtr inner_; std::string label_; int extraPrice_; Parts extra_;
public:
    AddOn(BevPtr inner, std::string label, int extraPrice, Parts extra)
        : inner_(std::move(inner)), label_(std::move(label)), extraPrice_(extraPrice), extra_(std::move(extra)) {}
    std::string name() const override { return inner_->name() + " + " + label_; }
    int price() const override { return inner_->price() + extraPrice_; }
    Recipe recipe() const override { return inner_->recipe().plus(extra_); }
};

class BeverageFactory {
    std::map<std::string, BevPtr> menu_;
    std::map<std::string, std::function<BevPtr(BevPtr)>> addOns_;
public:
    void registerDrink(const std::string& n, int price, Parts r) {
        menu_[n] = std::make_shared<BaseBeverage>(n, price, Recipe(std::move(r)));
    }
    void registerAddOn(const std::string& n, const std::string& label, int price, Parts extra) {
        addOns_[n] = [label, price, extra](BevPtr b) -> BevPtr {
            return std::make_shared<AddOn>(std::move(b), label, price, extra);
        };
    }
    BevPtr create(const std::string& drink, const std::vector<std::string>& extras) const {
        auto it = menu_.find(drink);
        if (it == menu_.end()) throw CoffeeError("unknown drink " + drink);
        BevPtr b = it->second;
        for (const auto& x : extras) {
            auto a = addOns_.find(x);
            if (a == addOns_.end()) throw CoffeeError("unknown add-on " + x);
            b = a->second(b);
        }
        return b;
    }
    static BeverageFactory standard() {
        BeverageFactory f;
        f.registerDrink("espresso", 120, {{"water", 30}, {"coffee", 18}});
        f.registerDrink("americano", 140, {{"water", 150}, {"coffee", 18}});
        f.registerDrink("latte", 180, {{"water", 30}, {"coffee", 18}, {"milk", 150}});
        f.registerDrink("cappuccino", 170, {{"water", 30}, {"coffee", 18}, {"milk", 100}});
        f.registerAddOn("shot", "extra shot", 40, {{"coffee", 9}, {"water", 15}});
        f.registerAddOn("milk", "extra milk", 20, {{"milk", 50}});
        f.registerAddOn("sugar", "sugar", 0, {{"sugar", 5}});
        f.registerAddOn("caramel", "caramel", 30, {{"caramel", 15}});
        return f;
    }
};

class Inventory {
    Parts stock_, lowMark_;
    std::vector<std::function<void(const std::string&, int)>> listeners_;
    // THREADED BUILD: one mutex guards stock_ and lowMark_ (the Java synchronized methods).
    //   pthread_mutex_t mu_ = PTHREAD_MUTEX_INITIALIZER;
    // tryReserve: pthread_mutex_lock(&mu_) before the check loop; after the subtract loop copy `crossed`,
    //   pthread_mutex_unlock(&mu_), and only then call the listeners (never call out holding the lock).
    // release / refill / level / snapshot: lock, touch stock_, unlock (a few map operations each).
public:
    void add(const std::string& ing, int qty, int lowAt) { stock_[ing] += qty; lowMark_[ing] = lowAt; }
    void refill(const std::string& ing, int qty) {
        if (!stock_.count(ing)) throw CoffeeError("unknown ingredient " + ing);
        if (qty <= 0) throw std::invalid_argument("refill must be positive");
        stock_[ing] += qty;
    }
    void subscribe(std::function<void(const std::string&, int)> l) { listeners_.push_back(std::move(l)); }
    // All or nothing: check every ingredient first, then subtract.
    std::vector<std::string> tryReserve(const Recipe& r) {
        std::vector<std::string> problems;
        for (const auto& kv : r.parts()) {
            auto it = stock_.find(kv.first);
            int have = it == stock_.end() ? 0 : it->second;
            if (have == 0) problems.push_back(kv.first + " is not available");
            else if (have < kv.second)
                problems.push_back(kv.first + " is not sufficient (need " + std::to_string(kv.second) +
                                   ", have " + std::to_string(have) + ")");
        }
        if (!problems.empty()) return problems;
        std::vector<std::pair<std::string, int>> crossed;
        for (const auto& kv : r.parts()) {
            int before = stock_[kv.first], after = before - kv.second;
            stock_[kv.first] = after;
            int mark = lowMark_[kv.first];
            if (before >= mark && after < mark) crossed.emplace_back(kv.first, after);
        }
        for (const auto& c : crossed) for (auto& l : listeners_) l(c.first, c.second);
        return problems;
    }
    void release(const Recipe& r) { for (const auto& kv : r.parts()) stock_[kv.first] += kv.second; }
    int level(const std::string& ing) const { auto it = stock_.find(ing); return it == stock_.end() ? 0 : it->second; }
    Parts snapshot() const { return stock_; }
};

struct Result { std::string drink; bool served; std::string detail; };
std::ostream& operator<<(std::ostream& os, const Result& r) {
    if (r.served && r.detail == "brewing") return os << r.drink << " is brewing on an outlet";
    if (r.served) return os << r.drink << " is prepared (Rs" << r.detail << ")";
    return os << r.drink << " cannot be prepared: " << r.detail;
}

// Outlets as a counted resource: start() takes one, finish() gives it back.
class CoffeeMachine {
    const BeverageFactory& factory_; Inventory& inv_; int free_; int revenue_ = 0; bool closed_ = false;
    // THREADED BUILD: free_ and revenue_ become std::atomic<int> (atomic_fetch_add in C), closed_ a
    // std::atomic<bool>; the all-or-nothing reservation stays inside Inventory's mutex.
public:
    void close() { closed_ = true; }               // later orders are refused before they reserve anything
    CoffeeMachine(int outlets, const BeverageFactory& f, Inventory& inv) : factory_(f), inv_(inv), free_(outlets) {
        if (outlets <= 0) throw std::invalid_argument("need at least one outlet");
    }
    // Returns the beverage now brewing, or a Result explaining the refusal.
    Result start(const std::string& drink, const std::vector<std::string>& extras, BevPtr& brewing) {
        BevPtr b;
        try { b = factory_.create(drink, extras); }
        catch (const CoffeeError& e) { return {drink, false, e.what()}; }
        if (closed_) return {b->name(), false, "machine closed"};
        if (free_ == 0) return {b->name(), false, "all outlets busy"};
        auto problems = inv_.tryReserve(b->recipe());
        if (!problems.empty()) {
            std::string s;
            for (size_t i = 0; i < problems.size(); ++i) s += (i ? ", " : "") + problems[i];
            return {b->name(), false, s};
        }
        --free_;
        brewing = b;
        return {b->name(), true, "brewing"};
    }
    Result finish(const BevPtr& b) {
        ++free_;
        revenue_ += b->price();
        return {b->name(), true, std::to_string(b->price())};
    }
    int freeOutlets() const { return free_; }
    int revenue() const { return revenue_; }
};

    ++passed;
}
