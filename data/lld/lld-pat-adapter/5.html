Each tab is a complete program that was compiled and run. Java shows two object adapters, one class adapter, and two JDK adapters. Python uses a Protocol (structural typing) as the Target. C++14 wraps a C-style vendor function with an object adapter and shows a class adapter using private inheritance. Files live in lld-code/lld-pat-adapter/.

import java.util.*;

public class AdapterDemo {

    // ---- Target: the interface OUR checkout code is written against ----
    enum Status { SUCCESS, DECLINED, ERROR }
    record PaymentResult(Status status, String txnId, String message) { }

    interface PaymentGateway {
        PaymentResult charge(String orderId, long amountPaise, String currency);
    }

    // ---- Adaptee 1: an old bank SDK we cannot change (rupees as double, int codes) ----
    static class OldBankSdk {                   // not final: the class adapter below extends it
        int makePayment(String reference, double rupees) {
            if (rupees > 10_000) return 51;                 // 51 = insufficient funds
            System.out.printf("  [OldBank] debit %.2f INR ref=%s%n", rupees, reference);
            return 0;                                        // 0 = OK
        }
        private int seq = 1000;
        String lastTxn() { return "OB-" + (++seq); }
    }

    // ---- Adaptee 2: a newer vendor client with a map-based API and string statuses ----
    static final class AcmePayClient {
        Map<String, String> createCharge(Map<String, Object> req) {
            System.out.println("  [AcmePay] createCharge " + req);
            Map<String, String> resp = new HashMap<>();
            boolean ok = !"USD".equals(req.get("currency_code"));
            resp.put("state", ok ? "captured" : "failed");
            resp.put("id", ok ? "ac_" + req.get("idempotency_key") : "");
            resp.put("reason", ok ? "" : "currency not enabled");
            return resp;
        }
    }

    // ---- Object adapter: implements Target, HOLDS an adaptee (composition) ----
    static final class OldBankAdapter implements PaymentGateway {
        private final OldBankSdk bank;
        OldBankAdapter(OldBankSdk bank) { this.bank = bank; }

        @Override public PaymentResult charge(String orderId, long amountPaise, String currency) {
            if (!"INR".equals(currency))
                return new PaymentResult(Status.ERROR, null, "OldBank supports INR only");
            double rupees = amountPaise / 100.0;             // unit conversion
            int code = bank.makePayment(orderId, rupees);    // call translation
            return switch (code) {                           // result translation
                case 0 -> new PaymentResult(Status.SUCCESS, bank.lastTxn(), "ok");
                case 51 -> new PaymentResult(Status.DECLINED, null, "insufficient funds");
                default -> new PaymentResult(Status.ERROR, null, "bank code " + code);
            };
        }
    }

    static final class AcmePayAdapter implements PaymentGateway {
        private final AcmePayClient client;
        AcmePayAdapter(AcmePayClient client) { this.client = client; }

        @Override public PaymentResult charge(String orderId, long amountPaise, String currency) {
            Map<String, Object> req = new LinkedHashMap<>();
            req.put("amount_minor", amountPaise);
            req.put("currency_code", currency);
            req.put("idempotency_key", orderId);
            // <-- the pattern: translate our call into the adaptee's API
            Map<String, String> resp = client.createCharge(req);
            return "captured".equals(resp.get("state"))
                    ? new PaymentResult(Status.SUCCESS, resp.get("id"), "ok")
                    : new PaymentResult(Status.DECLINED, null, resp.get("reason"));
        }
    }

    // ---- Class adapter: EXTENDS the adaptee and implements Target (inheritance) ----
    static final class OldBankClassAdapter extends OldBankSdk implements PaymentGateway {
        @Override public PaymentResult charge(String orderId, long amountPaise, String currency) {
            int code = makePayment(orderId, amountPaise / 100.0);   // inherited method
            return code == 0 ? new PaymentResult(Status.SUCCESS, lastTxn(), "ok")
                             : new PaymentResult(Status.DECLINED, null, "code " + code);
        }
    }

    // ---- Client: knows only PaymentGateway ----
    static final class Checkout {
        private final Map<String, PaymentGateway> gateways;
        Checkout(Map<String, PaymentGateway> gateways) { this.gateways = gateways; }
        void pay(String via, String orderId, long paise, String currency) {
            PaymentResult r = gateways.get(via).charge(orderId, paise, currency);
            System.out.println(via + " -> " + r);
        }
    }

    public static void main(String[] args) {
        Map<String, PaymentGateway> gw = new LinkedHashMap<>();
        gw.put("oldbank", new OldBankAdapter(new OldBankSdk()));
        gw.put("acme", new AcmePayAdapter(new AcmePayClient()));
        gw.put("oldbank-class", new OldBankClassAdapter());
        Checkout checkout = new Checkout(gw);

        checkout.pay("oldbank", "ORD-1", 49_900, "INR");     // Rs 499.00
        checkout.pay("oldbank", "ORD-2", 2_000_000, "INR");  // Rs 20,000 -> declined
        checkout.pay("acme", "ORD-3", 49_900, "INR");
        checkout.pay("acme", "ORD-4", 1_000, "USD");
        checkout.pay("oldbank-class", "ORD-5", 10_000, "INR");

        // The JDK ships adapters too:
        List<String> list = Arrays.asList("a", "b", "c");          // array seen as a List
        Enumeration<String> e = Collections.enumeration(list);      // List seen as an old Enumeration
        System.out.println("JDK adapters: " + list + " first=" + e.nextElement());
    }
}

Note that OldBankSdk had to be non-final for the class adapter to extend it. Real vendor classes are often final, one more reason the object adapter is the default.

"""Adapter: make two incompatible payment SDKs fit one PaymentGateway interface."""
from dataclasses import dataclass
from typing import Protocol
import io


@dataclass(frozen=True)
class PaymentResult:
    status: str            # "SUCCESS" | "DECLINED" | "ERROR"
    txn_id: str | None
    message: str


class PaymentGateway(Protocol):               # Target
    def charge(self, order_id: str, amount_paise: int, currency: str) -> PaymentResult: ...


class OldBankSdk:                             # Adaptee 1 (cannot change it)
    def __init__(self):
        self._seq = 1000

    def make_payment(self, reference: str, rupees: float) -> int:
        if rupees > 10_000:
            return 51                         # insufficient funds
        print(f"  [OldBank] debit {rupees:.2f} INR ref={reference}")
        self._seq += 1
        return 0

    def last_txn(self) -> str:
        return f"OB-{self._seq}"


class AcmePayClient:                          # Adaptee 2
    def create_charge(self, req: dict) -> dict:
        print("  [AcmePay] create_charge", req)
        ok = req["currency_code"] != "USD"
        return {"state": "captured" if ok else "failed",
                "id": f"ac_{req['idempotency_key']}" if ok else "",
                "reason": "" if ok else "currency not enabled"}


class OldBankAdapter:                         # object adapter: wraps, translates
    _codes = {0: ("SUCCESS", "ok"), 51: ("DECLINED", "insufficient funds")}

    def __init__(self, bank: OldBankSdk):
        self._bank = bank

    def charge(self, order_id, amount_paise, currency):
        if currency != "INR":
            return PaymentResult("ERROR", None, "OldBank supports INR only")
        code = self._bank.make_payment(order_id, amount_paise / 100)
        status, msg = self._codes.get(code, ("ERROR", f"bank code {code}"))
        return PaymentResult(status, self._bank.last_txn() if code == 0 else None, msg)


class AcmePayAdapter:
    def __init__(self, client: AcmePayClient):
        self._client = client

    def charge(self, order_id, amount_paise, currency):
        resp = self._client.create_charge({"amount_minor": amount_paise,
                                           "currency_code": currency,
                                           "idempotency_key": order_id})
        if resp["state"] == "captured":
            return PaymentResult("SUCCESS", resp["id"], "ok")
        return PaymentResult("DECLINED", None, resp["reason"])


def checkout(gateway: PaymentGateway, order_id: str, paise: int, currency: str = "INR"):
    print(type(gateway).__name__, "->", gateway.charge(order_id, paise, currency))


if __name__ == "__main__":
    checkout(OldBankAdapter(OldBankSdk()), "ORD-1", 49_900)
    checkout(OldBankAdapter(OldBankSdk()), "ORD-2", 2_000_000)
    checkout(AcmePayAdapter(AcmePayClient()), "ORD-3", 49_900)
    checkout(AcmePayAdapter(AcmePayClient()), "ORD-4", 1_000, "USD")

    # The standard library's own adapter: a bytes stream seen as a text stream.
    raw = io.BytesIO("namaste\n".encode("utf-8"))
    text = io.TextIOWrapper(raw, encoding="utf-8")
    print("stdlib adapter read:", text.readline().strip())

Python has duck typing, so an adapter does not have to inherit anything. The Protocol only lets a type checker (mypy, pyright) confirm that both adapters really match the Target.

// Adapter in C++14: object adapter (composition) and class adapter (private inheritance).
// Build: g++ -std=c++14 -Wall adapter.cpp -o adapter
#include <cstdio>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

// Target
struct PaymentResult { bool ok; std::string txn; std::string msg; };
class PaymentGateway {
public:
    virtual ~PaymentGateway() = default;
    virtual PaymentResult charge(const std::string& order, long paise) = 0;
};

// Adaptee: a C-style vendor library we cannot change.
extern "C" int oldbank_pay(const char* ref, double rupees, char* txn_out, int txn_len) {
    if (rupees > 10000.0) return 51;
    std::snprintf(txn_out, txn_len, "OB-%s", ref);
    std::printf("  [oldbank] debit %.2f ref=%s\n", rupees, ref);
    return 0;
}

class LegacyLedger {                      // an adaptee class, for the class adapter
public:
    bool post(double rupees) { std::printf("  [ledger] posted %.2f\n", rupees); return true; }
};

// Object adapter: wraps the C function behind the C++ interface.
class OldBankAdapter : public PaymentGateway {
public:
    PaymentResult charge(const std::string& order, long paise) override {
        char txn[32] = {0};
        int rc = oldbank_pay(order.c_str(), paise / 100.0, txn, sizeof txn);
        if (rc == 0) return {true, txn, "ok"};
        return {false, "", rc == 51 ? "insufficient funds" : "bank error"};
    }
};

// Class adapter: inherits the adaptee privately (implementation only),
// inherits the target publicly (interface).
class LedgerAdapter : public PaymentGateway, private LegacyLedger {
public:
    PaymentResult charge(const std::string& order, long paise) override {
        bool ok = post(paise / 100.0);    // LegacyLedger::post, reachable only inside
        return {ok, "LG-" + order, ok ? "ok" : "ledger error"};
    }
};

int main() {
    std::vector<std::unique_ptr<PaymentGateway>> gws;
    gws.push_back(std::make_unique<OldBankAdapter>());
    gws.push_back(std::make_unique<LedgerAdapter>());

    const long amounts[] = {49900, 2000000};
    for (auto& g : gws)
        for (long paise : amounts) {
            PaymentResult r = g->charge("ORD" + std::to_string(paise), paise);
            std::cout << (r.ok ? "SUCCESS " : "DECLINED ") << r.txn << " " << r.msg << "\n";
        }
    return 0;
}

Wrapping a C library (char buffers, int return codes) in a small C++ class with clean types is the most common adapter a systems engineer writes. private inheritance is "implemented in terms of": LedgerAdapter can call post(), but callers cannot.

The same idea in C

Adapter in plain C: a legacy function with the wrong units and return convention, fitted under the ops table the rest of the code expects. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-adapter/adapter_sketch.c).

/* Adapter in C: fit an old API under the ops table the rest of the code expects.
   Illustrative, not real kernel code (compat ioctl and old-driver shims do this).
   Build: gcc -std=c99 -Wall adapter_sketch.c -o adapter_sketch */
#include <stdio.h>

/* Target: what our checkout code calls */
struct pay_ops { int (*charge)(void *ctx, const char *order, long paise); };

/* Adaptee: a legacy SDK we cannot change (rupees as double, 1 = success) */
static int oldbank_debit(double rupees, const char *ref)
{
    printf("  oldbank: debit Rs %.2f for %s\n", rupees, ref);
    return rupees < 10000.0 ? 1 : 0;
}

/* Adapter: translates units, argument order and return convention */
static int oldbank_charge(void *ctx, const char *order, long paise)
{
    (void)ctx;
    return oldbank_debit(paise / 100.0, order) == 1 ? 0 : -1;   /* 0 = ok, -1 = declined */
}
static const struct pay_ops oldbank_adapter = { oldbank_charge };

static void checkout(const struct pay_ops *gw, const char *order, long paise)
{
    printf("%s: %s\n", order, gw->charge(NULL, order, paise) == 0 ? "PAID" : "DECLINED");
}

int main(void)
{
    checkout(&oldbank_adapter, "ORD-1", 49900);       /* checkout never sees oldbank_debit */
    checkout(&oldbank_adapter, "ORD-2", 2500000);
    return 0;
}
