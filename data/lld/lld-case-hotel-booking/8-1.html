"""Hotel booking: inventory by night, holds with a TTL, pricing Strategy + Decorators, cancellation policies."""
from __future__ import annotations

import itertools
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from datetime import date, datetime, time, timedelta
from enum import Enum
from typing import Callable, Dict, List


class RoomType(Enum):
    SINGLE = "SINGLE"
    DOUBLE = "DOUBLE"
    SUITE = "SUITE"


class BookingStatus(Enum):
    HELD = "HELD"
    CONFIRMED = "CONFIRMED"
    CANCELLED = "CANCELLED"
    EXPIRED = "EXPIRED"


ALLOWED = {
    BookingStatus.HELD: {BookingStatus.CONFIRMED, BookingStatus.CANCELLED, BookingStatus.EXPIRED},
    BookingStatus.CONFIRMED: {BookingStatus.CANCELLED},
    BookingStatus.CANCELLED: set(),                 # terminal
    BookingStatus.EXPIRED: set(),                   # terminal
}


class BookingError(Exception):
    pass


def money(paise: int) -> str:
    return f"{paise // 100}.{paise % 100:02d}"


@dataclass(frozen=True)
class DateRange:
    """Half-open stay [check_in, check_out): the guest sleeps check_in .. check_out - 1."""
    check_in: date
    check_out: date

    def __post_init__(self) -> None:
        if self.check_out <= self.check_in:
            raise BookingError("check-out must be after check-in")

    def nights(self) -> int:
        return (self.check_out - self.check_in).days

    def each_night(self) -> List[date]:
        return [self.check_in + timedelta(days=i) for i in range(self.nights())]

    def __str__(self) -> str:
        return f"{self.check_in}..{self.check_out}"


# ---------- pricing: Strategy, with Decorators that stack rules ----------

class PricingStrategy(ABC):
    @abstractmethod
    def price(self, rate: int, stay: DateRange, rooms: int) -> int: ...


class BaseRatePricing(PricingStrategy):
    def price(self, rate: int, stay: DateRange, rooms: int) -> int:
        return rate * stay.nights() * rooms


class WeekendSurcharge(PricingStrategy):
    """Adds `percent` of the nightly rate for every Friday and Saturday night."""

    def __init__(self, inner: PricingStrategy, percent: int) -> None:
        self.inner, self.percent = inner, percent

    def price(self, rate: int, stay: DateRange, rooms: int) -> int:
        extra = sum(rate * self.percent // 100 for n in stay.each_night() if n.weekday() in (4, 5))
        return self.inner.price(rate, stay, rooms) + extra * rooms


class LongStayDiscount(PricingStrategy):
    """Takes `percent` off the whole stay when it is at least `min_nights` long."""

    def __init__(self, inner: PricingStrategy, min_nights: int, percent: int) -> None:
        self.inner, self.min_nights, self.percent = inner, min_nights, percent

    def price(self, rate: int, stay: DateRange, rooms: int) -> int:
        p = self.inner.price(rate, stay, rooms)
        return p * (100 - self.percent) // 100 if stay.nights() >= self.min_nights else p


# ---------- cancellation: another Strategy, chosen per hotel ----------

class CancellationPolicy(ABC):
    @abstractmethod
    def refund(self, booking: "Booking", now: datetime) -> int: ...


class FlexibleCancellation(CancellationPolicy):
    """Free until `hours_before` hours before check-in (14:00); after that the first night is kept."""
    CHECK_IN_TIME = time(14, 0)

    def __init__(self, hours_before: int) -> None:
        self.hours_before = hours_before

    def refund(self, booking: "Booking", now: datetime) -> int:
        deadline = datetime.combine(booking.stay.check_in, self.CHECK_IN_TIME) - timedelta(hours=self.hours_before)
        if now < deadline:
            return booking.price
        return booking.price - booking.price // booking.stay.nights()


class NonRefundable(CancellationPolicy):
    def refund(self, booking: "Booking", now: datetime) -> int:
        return 0


# ---------- entities ----------

class Inventory:
    """Counts per (room type, night). The owning Hotel's lock guards it."""

    def __init__(self) -> None:
        self.total: Dict[RoomType, int] = {}
        self.taken: Dict[RoomType, Dict[date, int]] = {}

    def add_room(self, t: RoomType) -> None:
        self.total[t] = self.total.get(t, 0) + 1

    def free(self, t: RoomType, stay: DateRange) -> int:
        cap = self.total.get(t, 0)
        per_night = self.taken.get(t, {})
        return min(cap - per_night.get(n, 0) for n in stay.each_night())

    def take(self, t: RoomType, stay: DateRange, rooms: int) -> None:
        per_night = self.taken.setdefault(t, {})
        for n in stay.each_night():
            per_night[n] = per_night.get(n, 0) + rooms

    def release(self, t: RoomType, stay: DateRange, rooms: int) -> None:
        per_night = self.taken[t]
        for n in stay.each_night():
            per_night[n] -= rooms
            if per_night[n] == 0:
                del per_night[n]


@dataclass(frozen=True)
class Room:
    number: str
    type: RoomType


class Hotel:
    def __init__(self, id: str, name: str, city: str, pricing: PricingStrategy,
                 cancellation: CancellationPolicy) -> None:
        self.id, self.name, self.city = id, name, city
        self.pricing, self.cancellation = pricing, cancellation
        self.rooms: List[Room] = []
        self.rates: Dict[RoomType, int] = {}
        self.inventory = Inventory()
        self.lock = threading.Lock()                 # one lock per hotel

    def add_room(self, number: str, t: RoomType) -> "Hotel":
        self.rooms.append(Room(number, t))
        self.inventory.add_room(t)
        return self

    def rate(self, t: RoomType, paise_per_night: int) -> "Hotel":
        self.rates[t] = paise_per_night
        return self


@dataclass
class Booking:
    id: str
    guest: str
    hotel: Hotel
    type: RoomType
    stay: DateRange
    rooms: int
    price: int
    hold_expires_at: datetime
    status: BookingStatus = BookingStatus.HELD     # guarded by hotel.lock
    payment_ref: str = ""

    def move_to(self, nxt: BookingStatus) -> None:
        if nxt not in ALLOWED[self.status]:
            raise BookingError(f"{self.id}: cannot go from {self.status.value} to {nxt.value}")
        self.status = nxt

    def __str__(self) -> str:
        return (f"{self.id} {self.status.value} {self.hotel.id} {self.type.value} x{self.rooms} "
                f"{self.stay} price={money(self.price)}")


@dataclass(frozen=True)
class Offer:
    hotel: Hotel
    type: RoomType
    free: int
    price: int

    def __str__(self) -> str:
        return f"{self.hotel.id} {self.hotel.name:<10} {self.type.value} free={self.free} price={money(self.price)}"


class FakeClock:
    def __init__(self, start: datetime) -> None:
        self.t = start

    def now(self) -> datetime:
        return self.t

    def advance_minutes(self, m: int) -> None:
        self.t += timedelta(minutes=m)


# ---------- the facade ----------

class BookingService:
    HOLD_TTL = timedelta(minutes=15)

    def __init__(self, clock) -> None:
        self.clock = clock
        self.hotels: Dict[str, Hotel] = {}
        self.bookings: Dict[str, Booking] = {}
        self.listeners: List[Callable[[Booking], None]] = []
        self._seq = itertools.count(1)
        self._registry = threading.Lock()            # guards the two dicts and the id counter

    def add_listener(self, fn: Callable[[Booking], None]) -> None:
        self.listeners.append(fn)

    def add_hotel(self, h: Hotel) -> Hotel:
        with self._registry:
            if h.id in self.hotels:
                raise BookingError(f"duplicate hotel {h.id}")
            self.hotels[h.id] = h
        return h

    def search(self, city: str, t: RoomType, stay: DateRange, rooms: int) -> List[Offer]:
        self.expire_holds()
        out = []
        for h in list(self.hotels.values()):
            if h.city.lower() != city.lower() or t not in h.rates:
                continue
            with h.lock:
                free = h.inventory.free(t, stay)
            if free >= rooms:
                out.append(Offer(h, t, free, h.pricing.price(h.rates[t], stay, rooms)))
        return sorted(out, key=lambda o: (o.price, o.hotel.id))

    def hold(self, guest: str, hotel_id: str, t: RoomType, stay: DateRange, rooms: int) -> Booking:
        if rooms < 1:
            raise BookingError("rooms must be >= 1")
        h = self._hotel(hotel_id)
        if t not in h.rates:
            raise BookingError(f"{hotel_id} has no {t.value} rooms")
        self.expire_holds()
        with h.lock:                                 # check and take under one lock
            free = h.inventory.free(t, stay)
            if free < rooms:
                raise BookingError(f"{hotel_id} has {free} {t.value} free for {stay}, wanted {rooms}")
            h.inventory.take(t, stay, rooms)
            with self._registry:
                b = Booking(f"B{next(self._seq)}", guest, h, t, stay, rooms,
                            h.pricing.price(h.rates[t], stay, rooms), self.clock.now() + self.HOLD_TTL)
                self.bookings[b.id] = b
            return b

    def confirm(self, booking_id: str, payment_ref: str) -> Booking:
        """Called after the payment gateway says OK. Idempotent for the same payment reference."""
        b = self.booking(booking_id)
        expired_now = False
        with b.hotel.lock:
            if b.status is BookingStatus.CONFIRMED and b.payment_ref == payment_ref:
                return b
            if b.status is BookingStatus.HELD and self.clock.now() >= b.hold_expires_at:
                self._expire_locked(b)               # the sweep has not run yet: expire it now
                expired_now = True
            else:
                b.move_to(BookingStatus.CONFIRMED)   # raises unless HELD
                b.payment_ref = payment_ref
        self._publish(b)
        if expired_now:
            raise BookingError(f"{b.id}: hold expired, refund the payment")
        return b

    def cancel(self, booking_id: str) -> int:
        """Returns the refund in paise (0 for an unpaid hold)."""
        b = self.booking(booking_id)
        with b.hotel.lock:
            paid = b.status is BookingStatus.CONFIRMED
            refund = b.hotel.cancellation.refund(b, self.clock.now()) if paid else 0
            b.move_to(BookingStatus.CANCELLED)       # raises if already CANCELLED or EXPIRED
            b.hotel.inventory.release(b.type, b.stay, b.rooms)
        self._publish(b)
        return refund

    def expire_holds(self) -> int:
        """Lazy sweep: release every hold whose 15 minutes are over."""
        now = self.clock.now()
        expired = []
        with self._registry:
            candidates = list(self.bookings.values())
        for b in candidates:
            if b.status is not BookingStatus.HELD or now < b.hold_expires_at:
                continue
            with b.hotel.lock:
                if b.status is BookingStatus.HELD:   # re-check under the lock
                    self._expire_locked(b)
                    expired.append(b)
        for b in sorted(expired, key=lambda x: int(x.id[1:])):
            self._publish(b)
        return len(expired)

    def _expire_locked(self, b: Booking) -> None:
        b.move_to(BookingStatus.EXPIRED)
        b.hotel.inventory.release(b.type, b.stay, b.rooms)

    def _publish(self, b: Booking) -> None:          # always called outside the hotel lock
        for fn in self.listeners:
            fn(b)

    def _hotel(self, hotel_id: str) -> Hotel:
        h = self.hotels.get(hotel_id)
        if h is None:
            raise BookingError(f"unknown hotel {hotel_id}")
        return h

    def booking(self, booking_id: str) -> Booking:
        b = self.bookings.get(booking_id)
        if b is None:
            raise BookingError(f"unknown booking {booking_id}")
        return b
"""Demo driver: python main.py   (tests: python -m unittest test_hotel_booking)"""
import threading
from datetime import date, datetime

from hotel_booking import (BaseRatePricing, BookingError, BookingService, BookingStatus, DateRange, FakeClock,
                           FlexibleCancellation, Hotel, LongStayDiscount, NonRefundable, RoomType, WeekendSurcharge,
                           money)

passed = 0


def check(ok: bool, what: str) -> None:
    global passed
    if not ok:
        raise AssertionError("FAILED: " + what)
    passed += 1


def stay(a: str, b: str) -> DateRange:
    return DateRange(date.fromisoformat(a), date.fromisoformat(b))


def show(offers) -> None:
    for o in offers:
        print(f"  {o}")


def expect_reject(fn) -> None:
    global passed
    try:
        fn()
    except BookingError as e:
        print(f"  rejected: {e}")
        passed += 1
        return
    check(False, "should have been rejected")


def main() -> None:
    clock = FakeClock(datetime(2026, 11, 23, 10, 0))
    svc = BookingService(clock)
    svc.add_listener(lambda b: print(f"  [notify] {b.id} {b.status.value}"))

    svc.add_hotel(Hotel("H1", "Sea View", "Goa", WeekendSurcharge(BaseRatePricing(), 20), FlexibleCancellation(48))
                  .add_room("101", RoomType.DOUBLE).add_room("102", RoomType.DOUBLE).add_room("201", RoomType.SUITE)
                  .rate(RoomType.DOUBLE, 400000).rate(RoomType.SUITE, 900000))
    svc.add_hotel(Hotel("H2", "Hill Top", "Goa", LongStayDiscount(BaseRatePricing(), 7, 10), NonRefundable())
                  .add_room("1", RoomType.DOUBLE).rate(RoomType.DOUBLE, 350000))

    xmas = stay("2026-12-18", "2026-12-21")          # Fri, Sat, Sun nights
    print(f"== 1. Search Goa, DOUBLE x1, {xmas} ({xmas.nights()} nights)")
    offers = svc.search("Goa", RoomType.DOUBLE, xmas, 1)
    show(offers)
    check(len(offers) == 2 and offers[0].hotel.id == "H2", "cheapest first")
    check(offers[1].price == 1360000, "H1: 3 x 4000 + 20% on Fri and Sat = 13600")

    print("== 2. Alice holds 2 DOUBLE at H1")
    alice = svc.hold("alice", "H1", RoomType.DOUBLE, xmas, 2)
    print(f"  {alice} hold until {alice.hold_expires_at:%Y-%m-%d %H:%M}")
    check(alice.price == 2720000, "two rooms cost twice")

    print("== 3. Search again: H1 has no DOUBLE left for those nights")
    offers = svc.search("Goa", RoomType.DOUBLE, xmas, 1)
    show(offers)
    check(len(offers) == 1, "held rooms are not on sale")

    print("== 4. Bob wants H1 DOUBLE 2026-12-20..2026-12-22 (overlaps one night)")
    expect_reject(lambda: svc.hold("bob", "H1", RoomType.DOUBLE, stay("2026-12-20", "2026-12-22"), 1))

    print("== 5. Bob books H1 DOUBLE 2026-12-21..2026-12-23 (starts on Alice's check-out day)")
    bob = svc.hold("bob", "H1", RoomType.DOUBLE, stay("2026-12-21", "2026-12-23"), 1)
    svc.confirm(bob.id, "PAY-501")
    print(f"  {bob}")
    check(bob.status is BookingStatus.CONFIRMED, "half-open ranges: back-to-back stays fit")

    print("== 6. Alice pays")
    svc.confirm(alice.id, "PAY-778")
    print(f"  {alice} payment={alice.payment_ref}")
    svc.confirm(alice.id, "PAY-778")
    print(f"  confirm again with the same payment ref: {alice.id} {alice.status.value} (idempotent)")
    check(alice.status is BookingStatus.CONFIRMED, "retry is a no-op")

    print("== 7. Carol holds the SUITE and walks away")
    carol = svc.hold("carol", "H1", RoomType.SUITE, xmas, 1)
    print(f"  {carol}")
    check(carol.price == 3060000, "suite 3 x 9000 + 2 x 1800")
    clock.advance_minutes(16)
    print("  ... 16 minutes later")
    n = svc.expire_holds()
    print(f"  expired holds: {n}")
    check(n == 1 and carol.status is BookingStatus.EXPIRED, "hold released after 15 minutes")
    expect_reject(lambda: svc.confirm(carol.id, "PAY-900"))
    check(len(svc.search("Goa", RoomType.SUITE, xmas, 1)) == 1, "suite is back on sale")

    print("== 8. Alice cancels three weeks early (flexible policy)")
    refund = svc.cancel(alice.id)
    print(f"  refund={money(refund)}")
    check(refund == alice.price, "full refund before the deadline")
    offers = svc.search("Goa", RoomType.DOUBLE, xmas, 2)
    show(offers)
    check(len(offers) == 1 and offers[0].free == 2, "both rooms free again")

    print("== 9. Dev books H1 for tomorrow and cancels late")
    dev = svc.hold("dev", "H1", RoomType.DOUBLE, stay("2026-11-24", "2026-11-26"), 1)
    svc.confirm(dev.id, "PAY-610")
    refund = svc.cancel(dev.id)
    print(f"  {dev} refund={money(refund)}")
    check(refund == 400000, "inside 48 h: first night kept")

    print("== 10. Eve books H2 for 7 nights (long-stay discount, non-refundable)")
    eve = svc.hold("eve", "H2", RoomType.DOUBLE, stay("2027-01-01", "2027-01-08"), 1)
    svc.confirm(eve.id, "PAY-720")
    refund = svc.cancel(eve.id)
    print(f"  {eve} refund={money(refund)}")
    check(eve.price == 2205000 and refund == 0, "7 x 3500 - 10%, nothing back")
    expect_reject(lambda: svc.cancel(eve.id))

    print("== 11. Bad input")
    expect_reject(lambda: stay("2026-12-21", "2026-12-18"))
    expect_reject(lambda: svc.hold("x", "H1", RoomType.DOUBLE, xmas, 0))
    expect_reject(lambda: svc.hold("x", "H9", RoomType.DOUBLE, xmas, 1))
    expect_reject(lambda: svc.hold("x", "H2", RoomType.SUITE, xmas, 1))

    print("== 12. 8 guests race for H2's only DOUBLE")
    race = stay("2027-02-05", "2027-02-07")
    go = threading.Event()
    wins, losses, count_lock = [0], [0], threading.Lock()

    def attempt(guest: str) -> None:
        go.wait()
        try:
            svc.hold(guest, "H2", RoomType.DOUBLE, race, 1)
            with count_lock:
                wins[0] += 1
        except BookingError:
            with count_lock:
                losses[0] += 1

    threads = [threading.Thread(target=attempt, args=(f"g{i}",)) for i in range(8)]
    for t in threads:
        t.start()
    go.set()
    for t in threads:
        t.join()
    print(f"  winners={wins[0]} rejected={losses[0]}")
    check(wins[0] == 1 and losses[0] == 7, "exactly one hold wins")

    print(f"All {passed} checks passed")


if __name__ == "__main__":
    main()
import threading
import unittest
from datetime import date, datetime

from hotel_booking import (BaseRatePricing, BookingError, BookingService, BookingStatus, DateRange, FakeClock,
                           FlexibleCancellation, Hotel, LongStayDiscount, NonRefundable, RoomType, WeekendSurcharge)


def stay(a, b):
    return DateRange(date.fromisoformat(a), date.fromisoformat(b))


def fresh(doubles=2):
    clock = FakeClock(datetime(2026, 11, 23, 10, 0))
    svc = BookingService(clock)
    h = Hotel("H1", "Sea View", "Goa", WeekendSurcharge(BaseRatePricing(), 20), FlexibleCancellation(48))
    for i in range(doubles):
        h.add_room(str(100 + i), RoomType.DOUBLE)
    svc.add_hotel(h.rate(RoomType.DOUBLE, 400000))
    return svc, clock


class HotelBookingTest(unittest.TestCase):
    def test_half_open_ranges_share_the_changeover_day(self):
        svc, _ = fresh(doubles=1)
        svc.hold("a", "H1", RoomType.DOUBLE, stay("2026-12-18", "2026-12-21"), 1)
        svc.hold("b", "H1", RoomType.DOUBLE, stay("2026-12-21", "2026-12-23"), 1)   # fits
        with self.assertRaises(BookingError):
            svc.hold("c", "H1", RoomType.DOUBLE, stay("2026-12-20", "2026-12-22"), 1)

    def test_free_is_minimum_over_nights(self):
        svc, _ = fresh(doubles=2)
        svc.hold("a", "H1", RoomType.DOUBLE, stay("2026-12-19", "2026-12-20"), 1)    # one busy night
        offers = svc.search("goa", RoomType.DOUBLE, stay("2026-12-18", "2026-12-22"), 1)
        self.assertEqual(offers[0].free, 1)

    def test_pricing_decorators(self):
        r = stay("2026-12-18", "2026-12-21")                                       # Fri Sat Sun
        self.assertEqual(WeekendSurcharge(BaseRatePricing(), 20).price(400000, r, 1), 1360000)
        week = stay("2027-01-01", "2027-01-08")
        self.assertEqual(LongStayDiscount(BaseRatePricing(), 7, 10).price(350000, week, 1), 2205000)
        both = LongStayDiscount(WeekendSurcharge(BaseRatePricing(), 20), 7, 10)    # decorators stack
        self.assertEqual(both.price(100000, week, 1), (700000 + 2 * 20000) * 90 // 100)

    def test_hold_expires_and_releases(self):
        svc, clock = fresh(doubles=1)
        r = stay("2026-12-18", "2026-12-21")
        b = svc.hold("a", "H1", RoomType.DOUBLE, r, 1)
        clock.advance_minutes(14)
        self.assertEqual(svc.expire_holds(), 0)
        clock.advance_minutes(1)                                                   # exactly 15 min: expired
        self.assertEqual(svc.expire_holds(), 1)
        self.assertIs(b.status, BookingStatus.EXPIRED)
        svc.hold("b", "H1", RoomType.DOUBLE, r, 1)                                 # room is back

    def test_confirm_after_ttl_without_sweep_is_rejected(self):
        svc, clock = fresh(doubles=1)
        b = svc.hold("a", "H1", RoomType.DOUBLE, stay("2026-12-18", "2026-12-21"), 1)
        clock.advance_minutes(20)
        with self.assertRaises(BookingError):
            svc.confirm(b.id, "PAY-1")
        self.assertIs(b.status, BookingStatus.EXPIRED)

    def test_confirm_is_idempotent_per_payment_ref(self):
        svc, _ = fresh()
        b = svc.hold("a", "H1", RoomType.DOUBLE, stay("2026-12-18", "2026-12-21"), 1)
        events = []
        svc.add_listener(lambda x: events.append(x.status))
        svc.confirm(b.id, "PAY-1")
        svc.confirm(b.id, "PAY-1")
        self.assertEqual(events, [BookingStatus.CONFIRMED])                        # notified once
        with self.assertRaises(BookingError):
            svc.confirm(b.id, "PAY-2")                                             # different payment: error

    def test_cancellation_policies(self):
        svc, clock = fresh()
        early = svc.hold("a", "H1", RoomType.DOUBLE, stay("2026-12-18", "2026-12-21"), 1)
        svc.confirm(early.id, "P1")
        self.assertEqual(svc.cancel(early.id), early.price)
        late = svc.hold("b", "H1", RoomType.DOUBLE, stay("2026-11-24", "2026-11-26"), 1)
        svc.confirm(late.id, "P2")
        self.assertEqual(svc.cancel(late.id), 400000)                              # first night kept
        unpaid = svc.hold("c", "H1", RoomType.DOUBLE, stay("2026-12-01", "2026-12-02"), 1)
        self.assertEqual(svc.cancel(unpaid.id), 0)
        with self.assertRaises(BookingError):
            svc.cancel(unpaid.id)                                                  # terminal state

    def test_non_refundable(self):
        clock = FakeClock(datetime(2026, 11, 23, 10, 0))
        svc = BookingService(clock)
        svc.add_hotel(Hotel("H2", "Hill Top", "Goa", BaseRatePricing(), NonRefundable())
                      .add_room("1", RoomType.DOUBLE).rate(RoomType.DOUBLE, 350000))
        b = svc.hold("a", "H2", RoomType.DOUBLE, stay("2027-01-01", "2027-01-03"), 1)

        self.assertEqual(svc.cancel(b.id), 0)

// ---------- dates: a day number (days since 1970-01-01) and minutes since the epoch ----------
static long daysFromCivil(int y, int m, int d) {          // proleptic Gregorian calendar
    y -= m <= 2;
    long era = (y >= 0 ? y : y - 399) / 400;
    long yoe = y - era * 400;
    long doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
    long doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    return era * 146097 + doe - 719468;
}
static void civilFromDays(long z, int& y, int& m, int& d) {
    z += 719468;
    long era = (z >= 0 ? z : z - 146096) / 146097;
    long doe = z - era * 146097;
    long yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
    long doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    long mp = (5 * doy + 2) / 153;
    d = (int)(doy - (153 * mp + 2) / 5 + 1);
    m = (int)(mp < 10 ? mp + 3 : mp - 9);
    y = (int)(yoe + era * 400 + (m <= 2));
}

}

}

struct WeekendSurcharge : PricingStrategy {

class Inventory {
    std::map<RoomType, int> total_;
    std::map<RoomType, std::map<long, int>> taken_;
public:
    void addRoom(RoomType t) { total_[t]++; }
    int free(RoomType t, const DateRange& s) const {
        auto it = total_.find(t);
        int cap = it == total_.end() ? 0 : it->second, mn = cap;
        auto tk = taken_.find(t);
        for (long n = s.checkIn; n < s.checkOut; ++n) {
            int used = 0;
            if (tk != taken_.end()) { auto u = tk->second.find(n); if (u != tk->second.end()) used = u->second; }
            mn = std::min(mn, cap - used);
        }
        return mn;
    }
    void take(RoomType t, const DateRange& s, int rooms) { for (long n = s.checkIn; n < s.checkOut; ++n) taken_[t][n] += rooms; }
    void release(RoomType t, const DateRange& s, int rooms) {
        auto& per = taken_[t];
        for (long n = s.checkIn; n < s.checkOut; ++n) if ((per[n] -= rooms) == 0) per.erase(n);
    }
};

    Booking(std::string i, std::string g, Hotel* h, RoomType t, DateRange s, int r, long long p, Minutes exp)

class BookingService {
    std::map<std::string, std::unique_ptr<Hotel>> hotels_;
    std::map<std::string, std::unique_ptr<Booking>> bookings_;
    std::vector<std::function<void(const Booking&)>> listeners_;
    int seq_ = 0;
    const Minutes& clock_;
    static const Minutes HOLD_TTL = 15;
    Mutex reg_;                                              // hotels_, bookings_, listeners_, seq_

    void publish(const Booking& b) {                         // called with no lock held
        std::vector<std::function<void(const Booking&)>> ls;
        { LockGuard g(reg_); ls = listeners_; }
        for (auto& l : ls) l(b);
    }
    // caller holds b.hotel->mu
    void expireLocked(Booking& b) { b.moveTo(Status::EXPIRED); b.hotel->inventory.release(b.type, b.stay, b.rooms); }
    Hotel& hotel(const std::string& id) {
        LockGuard g(reg_);
        auto it = hotels_.find(id);
        if (it == hotels_.end()) throw BookingError("unknown hotel " + id);
        return *it->second;
    }
public:
    explicit BookingService(const Minutes& clock) : clock_(clock) {}
    void addListener(std::function<void(const Booking&)> l) { LockGuard g(reg_); listeners_.push_back(std::move(l)); }
    Hotel& addHotel(std::unique_ptr<Hotel> h) {
        LockGuard g(reg_);
        if (hotels_.count(h->id)) throw BookingError("duplicate hotel " + h->id);
        Hotel& ref = *h; hotels_[h->id] = std::move(h); return ref;
    }
    Booking& booking(const std::string& id) {
        LockGuard g(reg_);
        auto it = bookings_.find(id);
        if (it == bookings_.end()) throw BookingError("unknown booking " + id);
        return *it->second;
    }

    std::vector<Offer> search(const std::string& city, RoomType t, const DateRange& s, int rooms) {
        expireHolds();
        std::vector<Offer> out;
        std::vector<Hotel*> hs;
        { LockGuard g(reg_); for (auto& kv : hotels_) hs.push_back(kv.second.get()); }
        for (Hotel* hp : hs) {
            Hotel& h = *hp;
            if (h.city != city || !h.rates.count(t)) continue;
            LockGuard g(h.mu);                                 // a consistent read of this hotel's counts
            int free = h.inventory.free(t, s);
            if (free >= rooms) out.push_back({&h, t, free, h.pricing->price(h.rates[t], s, rooms)});
        }
        std::sort(out.begin(), out.end(), [](const Offer& a, const Offer& b) {
            return a.price != b.price ? a.price < b.price : a.hotel->id < b.hotel->id; });
        return out;
    }

    Booking& hold(const std::string& guest, const std::string& hotelId, RoomType t, const DateRange& s, int rooms) {
        if (rooms < 1) throw BookingError("rooms must be >= 1");
        Hotel& h = hotel(hotelId);
        if (!h.rates.count(t)) throw BookingError(hotelId + " has no " + name(t) + " rooms");
        expireHolds();
        LockGuard g(h.mu);                                     // check and take in one critical section
        int free = h.inventory.free(t, s);
        if (free < rooms)
            throw BookingError(hotelId + " has " + std::to_string(free) + " " + name(t) + " free for " + s.str()
                               + ", wanted " + std::to_string(rooms));
        h.inventory.take(t, s, rooms);
        LockGuard r(reg_);                                     // hotel -> registry: reg_ is always the inner lock
        std::string id = "B" + std::to_string(++seq_);
        bookings_[id].reset(new Booking(id, guest, &h, t, s, rooms, h.pricing->price(h.rates[t], s, rooms), clock_ + HOLD_TTL));
        return *bookings_[id];
    }

    Booking& confirm(const std::string& id, const std::string& paymentRef) {
        Booking& b = booking(id);
        bool expiredNow = false;
        {
            LockGuard g(b.hotel->mu);                           // one winner: confirm vs expiry vs cancel
            if (b.status == Status::CONFIRMED && b.paymentRef == paymentRef) return b;
            if (b.status == Status::HELD && clock_ >= b.holdExpiresAt) { expireLocked(b); expiredNow = true; }
            else { b.moveTo(Status::CONFIRMED); b.paymentRef = paymentRef; }
        }
        publish(b);
        if (expiredNow) throw BookingError(b.id + ": hold expired, refund the payment");
        return b;
    }

    long long cancel(const std::string& id) {
        Booking& b = booking(id);
        long long refund;
        {
            LockGuard g(b.hotel->mu);
            refund = b.status == Status::CONFIRMED ? b.hotel->cancellation->refund(b, clock_) : 0;
            b.moveTo(Status::CANCELLED);
            b.hotel->inventory.release(b.type, b.stay, b.rooms);
        }
        publish(b);
        return refund;
    }

    int expireHolds() {
        std::vector<Booking*> all, expired;
        { LockGuard g(reg_); for (auto& kv : bookings_) all.push_back(kv.second.get()); }
        for (Booking* bp : all) {
            Booking& b = *bp;
            LockGuard g(b.hotel->mu);
            if (b.status == Status::HELD && clock_ >= b.holdExpiresAt) { expireLocked(b); expired.push_back(&b); }
        }
        std::sort(expired.begin(), expired.end(), [](Booking* a, Booking* b) {
            return std::stoi(a->id.substr(1)) < std::stoi(b->id.substr(1)); });
        for (Booking* b : expired) publish(*b);
        return (int)expired.size();
    }
};

    svc.confirm(bob.id, "PAY-501");
    std::cout << "  " << bob.str() << "\n";
    check(bob.status == Status::CONFIRMED, "back-to-back stays fit");

    std::cout << "== 6. Alice pays\n";
    svc.confirm(alice.id, "PAY-778");
    std::cout << "  " << alice.str() << " payment=" << alice.paymentRef << "\n";
    svc.confirm(alice.id, "PAY-778");
    std::cout << "  confirm again with the same payment ref: " << alice.id << " " << name(alice.status) << " (idempotent)\n";
    check(alice.status == Status::CONFIRMED, "retry is a no-op");

    std::cout << "== 7. Carol holds the SUITE and walks away\n";
    Booking& carol = svc.hold("carol", "H1", RoomType::SUITE, xmas, 1);
    std::cout << "  " << carol.str() << "\n";
    check(carol.price == 3060000, "suite price");
    clock += 16;
    std::cout << "  ... 16 minutes later\n";
    int n = svc.expireHolds();
    std::cout << "  expired holds: " << n << "\n";
    check(n == 1 && carol.status == Status::EXPIRED, "hold released");
    expectReject([&] { svc.confirm(carol.id, "PAY-900"); });
    check(svc.search("Goa", RoomType::SUITE, xmas, 1).size() == 1, "suite back on sale");

    std::cout << "== 8. Alice cancels three weeks early (flexible policy)\n";
    long long refund = svc.cancel(alice.id);
    std::cout << "  refund=" << money(refund) << "\n";
    check(refund == alice.price, "full refund");

    show(offers);
    check(offers.size() == 1 && offers[0].free == 2, "both rooms free again");

    std::cout << "== 9. Dev books H1 for tomorrow and cancels late\n";
    Booking& dev = svc.hold("dev", "H1", RoomType::DOUBLE, stay("2026-11-24", "2026-11-26"), 1);
    svc.confirm(dev.id, "PAY-610");
    refund = svc.cancel(dev.id);
    std::cout << "  " << dev.str() << " refund=" << money(refund) << "\n";
    check(refund == 400000, "first night kept");

    std::cout << "== 10. Eve books H2 for 7 nights (long-stay discount, non-refundable)\n";
    Booking& eve = svc.hold("eve", "H2", RoomType::DOUBLE, stay("2027-01-01", "2027-01-08"), 1);
    svc.confirm(eve.id, "PAY-720");
    refund = svc.cancel(eve.id);
    std::cout << "  " << eve.str() << " refund=" << money(refund) << "\n";
    check(eve.price == 2205000 && refund == 0, "long stay, non-refundable");
    expectReject([&] { svc.cancel(eve.id); });

    std::cout << "== 11. Bad input\n";
    expectReject([&] { stay("2026-12-21", "2026-12-18"); });
    expectReject([&] { svc.hold("x", "H1", RoomType::DOUBLE, xmas, 0); });
    expectReject([&] { svc.hold("x", "H9", RoomType::DOUBLE, xmas, 1); });
    expectReject([&] { svc.hold("x", "H2", RoomType::SUITE, xmas, 1); });

    std::cout << "== 12. 8 guests ask for H2's only DOUBLE (one after another in this build)\n";
    DateRange race = stay("2027-02-05", "2027-02-07");
    int wins = 0, losses = 0;
    for (int i = 0; i < 8; ++i) {
        try { svc.hold("g" + std::to_string(i), "H2", RoomType::DOUBLE, race, 1); ++wins; }
        catch (const BookingError&) { ++losses; }
    }
    std::cout << "  winners=" << wins << " rejected=" << losses << "\n";
    check(wins == 1 && losses == 7, "exactly one hold wins");

    std::cout << "All " << passed << " checks passed\n";
    return 0;

What this C++ version leaves out: nothing in behaviour: the same 21 checks and the same 43-line demo as Java, measured. On MinGW the per-hotel lock is a no-op, so its 8-thread race runs sequentially there; the Linux -DHAVE_STD_MUTEX build runs it on real threads (compiled and run, same output).

Read the demo against the calendar picture at the top. Step 4 rejects Bob for 20..22 because the night of the 20th is Alice's; step 5 accepts 21..23 because the 21st is her check-out morning. Step 1's price is the decorator at work: 3 nights at 4000 plus 20% on Friday and Saturday is 13600. Step 9 is the flexible policy after its deadline: the stay starts tomorrow, the 48-hour window has passed, so the hotel keeps the first night (4000 of 8000).
