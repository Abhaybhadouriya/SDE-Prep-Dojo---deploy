  1. Enums and data classes. BookingStatus with its transition table, and the half-open DateRange.
  2. One interface, its decorators. PricingStrategy prices a stay; PricingDecorator wraps another strategy and WeekendSurcharge adds one rule.
  3. The facade's core method. hold takes the hotel's lock, checks every night and takes every night as one unit.

How to read the full code

Reading order: Read Model.java (DateRange, Inventory, Booking), then BookingService.hold, confirm and cancel, then Policies.java. Skip FakeClock and the listener in Main on a first read.

Java for C readers: the idioms this page's code uses (open if a line looks strange)
You will seeRead it as
x -> x.price (lambda)An anonymous function that can capture local variables: a callback and its void *ctx in one value. (a, b) -> a.x - b.x is the cmp you would hand to qsort.
Type::method (method reference)A pointer to a function that already exists, written without a wrapper: Post::score is &post_score.
map.computeIfAbsent(k, f)Lookup-or-create in one call: find k; if it is missing, call f, insert the result and return it.
map.merge(k, 1, Integer::sum)counts[k] += 1, inserting the first value when k is absent.
Comparator.comparing(f).reversed().thenComparing(g)Builds the cmp() function field by field: compare by f, negate the result, break ties with g.
enum with fields and methodsA Java enum is a fixed set of singleton objects: think of a static const array of structs that can carry function pointers.
lock(); try { ... } finally { unlock(); }The finally block runs on every exit path, normal return or exception: it is the goto out_unlock; label of kernel C.
throw new XException(..)Unwinds to the nearest catch: an error return that a caller cannot silently ignore.

/** A stay is half-open: [checkIn, checkOut). The guest sleeps the nights checkIn .. checkOut-1. */
final class DateRange {
    final LocalDate checkIn, checkOut;

    DateRange(LocalDate checkIn, LocalDate checkOut) {
        if (!checkOut.isAfter(checkIn)) throw new BookingException("check-out must be after check-in");
        this.checkIn = checkIn;
        this.checkOut = checkOut;
    }

    int nights() { return (int) ChronoUnit.DAYS.between(checkIn, checkOut); }

    List<LocalDate> eachNight() {
        List<LocalDate> out = new ArrayList<>();
        for (LocalDate d = checkIn; d.isBefore(checkOut); d = d.plusDays(1)) out.add(d);
        return out;
    }

    @Override public String toString() { return checkIn + ".." + checkOut; }
}

final class Room {
    final String number;
    final RoomType type;
    Room(String number, RoomType type) { this.number = number; this.type = type; }
}

/** Counts per (room type, night). Not thread safe by itself: the owning Hotel's lock guards it. */
final class Inventory {
    private final Map<RoomType, Integer> total = new EnumMap<>(RoomType.class);
    private final Map<RoomType, Map<LocalDate, Integer>> taken = new EnumMap<>(RoomType.class);

    void addRoom(RoomType t) { total.merge(t, 1, Integer::sum); }   // C: Type::method = a pointer to an existing function

    /** Rooms of this type free on EVERY night of the stay (the minimum over the nights). */
    int free(RoomType t, DateRange stay) {
        int cap = total.getOrDefault(t, 0);
        Map<LocalDate, Integer> perNight = taken.getOrDefault(t, new HashMap<>());
        int min = cap;
        for (LocalDate night : stay.eachNight()) min = Math.min(min, cap - perNight.getOrDefault(night, 0));
        return min;
    }

    void take(RoomType t, DateRange stay, int rooms) {
        Map<LocalDate, Integer> perNight = taken.computeIfAbsent(t, k -> new HashMap<>());   // C: a lambda = a function pointer plus its captured context
        for (LocalDate night : stay.eachNight()) perNight.merge(night, rooms, Integer::sum);
    }

    void release(RoomType t, DateRange stay, int rooms) {
        Map<LocalDate, Integer> perNight = taken.get(t);
        for (LocalDate night : stay.eachNight()) {
            int left = perNight.get(night) - rooms;
            if (left == 0) perNight.remove(night); else perNight.put(night, left);
        }
    }
}

final class Hotel {
    final String id, name, city;
    final PricingStrategy pricing;
    final CancellationPolicy cancellation;
    final List<Room> rooms = new ArrayList<>();
    final Map<RoomType, Long> nightlyRatePaise = new EnumMap<>(RoomType.class);
    final Inventory inventory = new Inventory();
    final ReentrantLock lock = new ReentrantLock();      // one lock per hotel: hotels never block each other

    Hotel(String id, String name, String city, PricingStrategy pricing, CancellationPolicy cancellation) {
        this.id = id; this.name = name; this.city = city;
        this.pricing = pricing; this.cancellation = cancellation;
    }

    Hotel addRoom(String number, RoomType type) {
        rooms.add(new Room(number, type));
        inventory.addRoom(type);
        return this;
    }

    Hotel rate(RoomType type, long paisePerNight) { nightlyRatePaise.put(type, paisePerNight); return this; }
}

final class Booking {
    final String id, guest;
    final Hotel hotel;
    final RoomType type;
    final DateRange stay;
    final int rooms;
    final long pricePaise;
    final LocalDateTime holdExpiresAt;
    private BookingStatus status = BookingStatus.HELD;   // guarded by hotel.lock
    private String paymentRef;

    Booking(String id, String guest, Hotel hotel, RoomType type, DateRange stay, int rooms,
            long pricePaise, LocalDateTime holdExpiresAt) {
        this.id = id; this.guest = guest; this.hotel = hotel; this.type = type; this.stay = stay;
        this.rooms = rooms; this.pricePaise = pricePaise; this.holdExpiresAt = holdExpiresAt;
    }

    BookingStatus status() { return status; }
    String paymentRef() { return paymentRef; }

    void moveTo(BookingStatus next) {
        if (!status.canMoveTo(next)) throw new BookingException(id + ": cannot go from " + status + " to " + next);
        status = next;
    }

    void markPaid(String ref) { moveTo(BookingStatus.CONFIRMED); paymentRef = ref; }

    @Override public String toString() {
        return id + " " + status + " " + hotel.id + " " + type + " x" + rooms + " " + stay
                + " price=" + Money.fmt(pricePaise);
    }
}

// ---------- pricing: Strategy, with Decorators that stack rules ----------

interface PricingStrategy {
    /** Total price in paise for `rooms` rooms of one type over the stay. */
    long price(long nightlyRatePaise, DateRange stay, int rooms);
}

final class BaseRatePricing implements PricingStrategy {
    public long price(long rate, DateRange stay, int rooms) { return rate * stay.nights() * rooms; }
}

abstract class PricingDecorator implements PricingStrategy {
    protected final PricingStrategy inner;
    PricingDecorator(PricingStrategy inner) { this.inner = inner; }
}

/** Adds `percent` of the nightly rate for every Friday and Saturday night. */
final class WeekendSurcharge extends PricingDecorator {
    private final int percent;
    WeekendSurcharge(PricingStrategy inner, int percent) { super(inner); this.percent = percent; }

    public long price(long rate, DateRange stay, int rooms) {
        long extra = 0;
        for (LocalDate night : stay.eachNight()) {
            DayOfWeek d = night.getDayOfWeek();
            if (d == DayOfWeek.FRIDAY || d == DayOfWeek.SATURDAY) extra += rate * percent / 100;
        }
        return inner.price(rate, stay, rooms) + extra * rooms;
    }
}

/** Takes `percent` off the whole stay when it is at least `minNights` long. */
final class LongStayDiscount extends PricingDecorator {
    private final int minNights, percent;
    LongStayDiscount(PricingStrategy inner, int minNights, int percent) {
        super(inner); this.minNights = minNights; this.percent = percent;
    }

    public long price(long rate, DateRange stay, int rooms) {
        long p = inner.price(rate, stay, rooms);
        return stay.nights() >= minNights ? p * (100 - percent) / 100 : p;
    }
}

// ---------- cancellation: another Strategy, chosen per hotel ----------

interface CancellationPolicy {
    /** Refund in paise for a CONFIRMED booking cancelled at `now`. */
    long refund(Booking b, LocalDateTime now);
}

/** Free until `hoursBefore` hours before check-in (14:00); after that the first night is kept. */
final class FlexibleCancellation implements CancellationPolicy {
    static final LocalTime CHECK_IN_TIME = LocalTime.of(14, 0);
    private final int hoursBefore;
    FlexibleCancellation(int hoursBefore) { this.hoursBefore = hoursBefore; }

    public long refund(Booking b, LocalDateTime now) {
        LocalDateTime deadline = b.stay.checkIn.atTime(CHECK_IN_TIME).minusHours(hoursBefore);
        if (now.isBefore(deadline)) return b.pricePaise;
        return b.pricePaise - b.pricePaise / b.stay.nights();
    }
}

final class NonRefundable implements CancellationPolicy {
    public long refund(Booking b, LocalDateTime now) { return 0; }
}

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;

/** Facade: every caller (driver, CLI parser, REST controller) goes through these methods. */
final class BookingService {
    static final Duration HOLD_TTL = Duration.ofMinutes(15);

    private final Map<String, Hotel> hotels = new ConcurrentHashMap<>();
    private final Map<String, Booking> bookings = new ConcurrentHashMap<>();
    private final List<BookingListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicInteger seq = new AtomicInteger();
    private final TimeSource clock;

    BookingService(TimeSource clock) { this.clock = clock; }

    void addListener(BookingListener l) { listeners.add(l); }

    Hotel addHotel(Hotel h) {
        if (hotels.putIfAbsent(h.id, h) != null) throw new BookingException("duplicate hotel " + h.id);
        return h;
    }

    List<Offer> search(String city, RoomType type, DateRange stay, int rooms) {
        expireHolds();
        List<Offer> out = new ArrayList<>();
        for (Hotel h : hotels.values()) {
            Long rate = h.nightlyRatePaise.get(type);
            if (!h.city.equalsIgnoreCase(city) || rate == null) continue;
            int free;
            h.lock.lock();
            try { free = h.inventory.free(type, stay); } finally { h.lock.unlock(); }   // C: finally runs on every exit path: the goto out_unlock label
            if (free >= rooms) out.add(new Offer(h, type, free, h.pricing.price(rate, stay, rooms)));
        }
        out.sort(Comparator.comparingLong((Offer o) -> o.pricePaise).thenComparing(o -> o.hotel.id));   // C: builds the cmp() function you would pass to qsort
        return out;
    }

    /** Check and take inventory atomically, then start a 15-minute hold. */
    Booking hold(String guest, String hotelId, RoomType type, DateRange stay, int rooms) {
        if (rooms < 1) throw new BookingException("rooms must be >= 1");
        Hotel h = hotel(hotelId);
        Long rate = h.nightlyRatePaise.get(type);
        if (rate == null) throw new BookingException(hotelId + " has no " + type + " rooms");
        expireHolds();
        h.lock.lock();
        try {
            int free = h.inventory.free(type, stay);
            if (free < rooms)
                throw new BookingException(hotelId + " has " + free + " " + type + " free for " + stay + ", wanted " + rooms);
            h.inventory.take(type, stay, rooms);
            Booking b = new Booking("B" + seq.incrementAndGet(), guest, h, type, stay, rooms,
                    h.pricing.price(rate, stay, rooms), clock.now().plus(HOLD_TTL));
            bookings.put(b.id, b);
            return b;
        } finally {
            h.lock.unlock();
        }
    }

    /** Called after the payment gateway says OK. Idempotent for the same payment reference. */
    Booking confirm(String bookingId, String paymentRef) {
        Booking b = booking(bookingId);
        boolean expiredNow = false;
        b.hotel.lock.lock();
        try {
            if (b.status() == BookingStatus.CONFIRMED && paymentRef.equals(b.paymentRef())) return b;
            if (b.status() == BookingStatus.HELD && !clock.now().isBefore(b.holdExpiresAt)) {
                expireLocked(b);                          // the sweep has not run yet: expire it now
                expiredNow = true;
            } else {
                b.markPaid(paymentRef);                   // throws unless HELD
            }
        } finally {
            b.hotel.lock.unlock();
        }
        publish(b);
        if (expiredNow) throw new BookingException(b.id + ": hold expired, refund the payment");
        return b;
    }

    /** Returns the refund in paise (0 for an unpaid hold). */
    long cancel(String bookingId) {
        Booking b = booking(bookingId);
        long refund;
        b.hotel.lock.lock();
        try {
            refund = b.status() == BookingStatus.CONFIRMED ? b.hotel.cancellation.refund(b, clock.now()) : 0;
            b.moveTo(BookingStatus.CANCELLED);            // throws if already CANCELLED or EXPIRED
            b.hotel.inventory.release(b.type, b.stay, b.rooms);
        } finally {
            b.hotel.lock.unlock();
        }
        publish(b);
        return refund;
    }

    /** Lazy sweep: release every hold whose 15 minutes are over. Returns how many expired. */
    int expireHolds() {
        LocalDateTime now = clock.now();
        List<Booking> expired = new ArrayList<>();
        for (Booking b : bookings.values()) {
            if (b.status() != BookingStatus.HELD || now.isBefore(b.holdExpiresAt)) continue;
            b.hotel.lock.lock();
            try {
                if (b.status() == BookingStatus.HELD) { expireLocked(b); expired.add(b); }   // re-check under the lock
            } finally {
                b.hotel.lock.unlock();
            }
        }
        expired.sort(Comparator.comparingInt(b -> Integer.parseInt(b.id.substring(1))));
        for (Booking b : expired) publish(b);
        return expired.size();
    }

    private void expireLocked(Booking b) {
        b.moveTo(BookingStatus.EXPIRED);
        b.hotel.inventory.release(b.type, b.stay, b.rooms);
    }

    private void publish(Booking b) { for (BookingListener l : listeners) l.onChange(b); }  // outside the lock

    private Hotel hotel(String id) {
        Hotel h = hotels.get(id);
        if (h == null) throw new BookingException("unknown hotel " + id);
        return h;
    }

    Booking booking(String id) {
        Booking b = bookings.get(id);
        if (b == null) throw new BookingException("unknown booking " + id);
        return b;
    }
}

public class Main {
    static int passed = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }

    static DateRange stay(String in, String out) { return new DateRange(LocalDate.parse(in), LocalDate.parse(out)); }

    static void show(List<Offer> offers) { for (Offer o : offers) System.out.println("  " + o); }

        try { r.run(); check(false, "should have been rejected"); }
        catch (BookingException e) { System.out.println("  rejected: " + e.getMessage()); passed++; }

    public static void main(String[] args) throws Exception {
        FakeClock clock = new FakeClock(LocalDateTime.of(2026, 11, 23, 10, 0));
        BookingService svc = new BookingService(clock);
        svc.addListener(b -> System.out.println("  [notify] " + b.id + " " + b.status()));

        DateRange xmas = stay("2026-12-18", "2026-12-21");          // Fri, Sat, Sun nights
        System.out.println("== 1. Search Goa, DOUBLE x1, " + xmas + " (" + xmas.nights() + " nights)");
        List<Offer> offers = svc.search("Goa", RoomType.DOUBLE, xmas, 1);
        show(offers);
        check(offers.size() == 2 && offers.get(0).hotel.id.equals("H2"), "cheapest first");
        check(offers.get(1).pricePaise == 1360000, "H1: 3 x 4000 + 20% on Fri and Sat = 13600");

        System.out.println("== 2. Alice holds 2 DOUBLE at H1");
        Booking alice = svc.hold("alice", "H1", RoomType.DOUBLE, xmas, 2);
        System.out.println("  " + alice + " hold until "
                + alice.holdExpiresAt.format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm")));
        check(alice.pricePaise == 2720000, "two rooms cost twice");

        System.out.println("== 3. Search again: H1 has no DOUBLE left for those nights");
        offers = svc.search("Goa", RoomType.DOUBLE, xmas, 1);
        show(offers);
        check(offers.size() == 1, "held rooms are not on sale");

        System.out.println("== 4. Bob wants H1 DOUBLE 2026-12-20..2026-12-22 (overlaps one night)");
        expectReject(() -> svc.hold("bob", "H1", RoomType.DOUBLE, stay("2026-12-20", "2026-12-22"), 1));

        System.out.println("== 5. Bob books H1 DOUBLE 2026-12-21..2026-12-23 (starts on Alice's check-out day)");
        Booking bob = svc.hold("bob", "H1", RoomType.DOUBLE, stay("2026-12-21", "2026-12-23"), 1);
        svc.confirm(bob.id, "PAY-501");
        System.out.println("  " + bob);
        check(bob.status() == BookingStatus.CONFIRMED, "half-open ranges: back-to-back stays fit");

        System.out.println("== 6. Alice pays");
        svc.confirm(alice.id, "PAY-778");
        System.out.println("  " + alice + " payment=" + alice.paymentRef());
        svc.confirm(alice.id, "PAY-778");
        System.out.println("  confirm again with the same payment ref: " + alice.id + " " + alice.status() + " (idempotent)");
        check(alice.status() == BookingStatus.CONFIRMED, "retry is a no-op");

        System.out.println("== 7. Carol holds the SUITE and walks away");
        Booking carol = svc.hold("carol", "H1", RoomType.SUITE, xmas, 1);
        System.out.println("  " + carol);
        check(carol.pricePaise == 3060000, "suite 3 x 9000 + 2 x 1800");
        clock.advanceMinutes(16);
        System.out.println("  ... 16 minutes later");
        int n = svc.expireHolds();
        System.out.println("  expired holds: " + n);
        check(n == 1 && carol.status() == BookingStatus.EXPIRED, "hold released after 15 minutes");
        expectReject(() -> svc.confirm(carol.id, "PAY-900"));
        check(svc.search("Goa", RoomType.SUITE, xmas, 1).size() == 1, "suite is back on sale");

        System.out.println("== 8. Alice cancels three weeks early (flexible policy)");
        long refund = svc.cancel(alice.id);
        System.out.println("  refund=" + Money.fmt(refund));
        check(refund == alice.pricePaise, "full refund before the deadline");
        offers = svc.search("Goa", RoomType.DOUBLE, xmas, 2);
        show(offers);
        check(offers.size() == 1 && offers.get(0).free == 2, "both rooms free again");

        System.out.println("== 9. Dev books H1 for tomorrow and cancels late");
        Booking dev = svc.hold("dev", "H1", RoomType.DOUBLE, stay("2026-11-24", "2026-11-26"), 1);
        svc.confirm(dev.id, "PAY-610");
        refund = svc.cancel(dev.id);
        System.out.println("  " + dev + " refund=" + Money.fmt(refund));
        check(refund == 400000, "inside 48 h: first night kept");

        System.out.println("== 10. Eve books H2 for 7 nights (long-stay discount, non-refundable)");
        Booking eve = svc.hold("eve", "H2", RoomType.DOUBLE, stay("2027-01-01", "2027-01-08"), 1);
        svc.confirm(eve.id, "PAY-720");
        refund = svc.cancel(eve.id);
        System.out.println("  " + eve + " refund=" + Money.fmt(refund));
        check(eve.pricePaise == 2205000 && refund == 0, "7 x 3500 - 10%, nothing back");
        expectReject(() -> svc.cancel(eve.id));

        System.out.println("== 11. Bad input");
        expectReject(() -> stay("2026-12-21", "2026-12-18"));
        expectReject(() -> svc.hold("x", "H1", RoomType.DOUBLE, xmas, 0));
        expectReject(() -> svc.hold("x", "H9", RoomType.DOUBLE, xmas, 1));
        expectReject(() -> svc.hold("x", "H2", RoomType.SUITE, xmas, 1));

        System.out.println("== 12. 8 guests race for H2's only DOUBLE");
        DateRange race = stay("2027-02-05", "2027-02-07");
        ExecutorService pool = Executors.newFixedThreadPool(8);
        CountDownLatch go = new CountDownLatch(1);
        AtomicInteger wins = new AtomicInteger(), losses = new AtomicInteger();
        for (int i = 0; i < 8; i++) {
            String guest = "g" + i;
            pool.submit(() -> {
                try { go.await(); svc.hold(guest, "H2", RoomType.DOUBLE, race, 1); wins.incrementAndGet(); }
                catch (BookingException e) { losses.incrementAndGet(); }
                catch (InterruptedException e) { Thread.currentThread().interrupt(); }
            });
        }
        go.countDown();
        pool.shutdown();
        pool.awaitTermination(10, TimeUnit.SECONDS);
        System.out.println("  winners=" + wins.get() + " rejected=" + losses.get());
        check(wins.get() == 1 && losses.get() == 7, "exactly one hold wins");

        System.out.println("All " + passed + " checks passed");
    }
}
