Java for C readers: the idioms this page's code uses (open if a line looks strange)
You will seeRead it as
x -> x.price (lambda)An anonymous function that can capture local variables: a callback and its void *ctx in one value. (a, b) -> a.x - b.x is the cmp you would hand to qsort.
Type::method (method reference)A pointer to a function that already exists, written without a wrapper: Post::score is &post_score.
Optional<T>A pointer that may be NULL, made explicit in the type, so the caller has to say what happens when it is empty (orElse, isPresent, map).
.stream().filter(..).map(..).collect(..)A for-loop written as a pipeline: filter is if (!pred(x)) continue;, map transforms each element, collect/toList appends the survivors to a new array.
map.merge(k, 1, Integer::sum)counts[k] += 1, inserting the first value when k is absent.
Comparator.comparing(f).reversed().thenComparing(g)Builds the cmp() function field by field: compare by f, negate the result, break ties with g.
enum with fields and methodsA Java enum is a fixed set of singleton objects: think of a static const array of structs that can carry function pointers.
synchronizedTake the object's built-in mutex on entry and release it on every exit. It is reentrant: the owning thread may enter again.
throw new XException(..)Unwinds to the nearest catch: an error return that a caller cannot silently ignore.

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

// Food delivery (FoodKart style) - in-memory, thread-safe. Build: javac -d out Main.java && java -cp out Main

// ---------- order state: one enum = the whole transition table ----------
enum OrderStatus {
    PLACED, ACCEPTED, READY, PICKED_UP, DELIVERED, CANCELLED;

    boolean canMoveTo(OrderStatus next) {
        switch (this) {
            case PLACED:    return next == ACCEPTED || next == CANCELLED;
            case ACCEPTED:  return next == READY || next == CANCELLED;
            case READY:     return next == PICKED_UP;
            case PICKED_UP: return next == DELIVERED;
            default:        return false;              // DELIVERED and CANCELLED are terminal
        }
    }
}

class FoodException extends RuntimeException {
    FoodException(String msg) { super(msg); }
}

// ---------- model ----------
final class Restaurant {
    final String name;
    private final int capacity;                                   // items cooking at once
    private final Map<String, Integer> menu = new ConcurrentHashMap<>(); // item -> price (rupees)
    private int inUse;                                            // guarded by this
    private final Map<String, Integer> served = new TreeMap<>();  // guarded by this

    Restaurant(String name, Map<String, Integer> menu, int capacity) {
        if (capacity <= 0) throw new FoodException("capacity must be > 0");
        this.name = name; this.capacity = capacity;
        menu.forEach(this::setPrice);   // C: Type::method = a pointer to an existing function
    }

    void setPrice(String item, int price) {
        if (price <= 0) throw new FoodException("price must be > 0: " + item);
        menu.put(item, price);
    }

    /** Total price for all items, or -1 if any item is not on the menu. */
    int quote(Map<String, Integer> items) {
        int total = 0;
        for (Map.Entry<String, Integer> e : items.entrySet()) {
            Integer p = menu.get(e.getKey());
            if (p == null) return -1;
            total += p * e.getValue();
        }
        return total;
    }

    synchronized int freeCapacity() { return capacity - inUse; }

    /** Check-and-claim in one critical section: no other thread can sneak in between. */
    synchronized boolean tryReserve(int qty) {
        if (inUse + qty > capacity) return false;
        inUse += qty;
        return true;
    }

    synchronized void release(int qty) { inUse -= qty; }

    synchronized void recordServed(Map<String, Integer> items) {
        items.forEach((k, v) -> served.merge(k, v, Integer::sum));   // C: counts[k] += v, inserting v if k is absent
    }

    synchronized Map<String, Integer> served() { return new TreeMap<>(served); }

    synchronized String describe() {
        return name + " menu=" + new TreeMap<>(menu) + " capacity=" + capacity + " inUse=" + inUse;
    }
}

final class DeliveryPartner {
    final String id;
    private final AtomicInteger deliveries = new AtomicInteger();
    DeliveryPartner(String id) { this.id = id; }
    int deliveries() { return deliveries.get(); }
    void delivered() { deliveries.incrementAndGet(); }
}

final class Order {
    final String id, customer;
    final Map<String, Integer> items;      // item -> quantity
    final Restaurant restaurant;
    final int total;                       // price snapshot taken at order time
    private OrderStatus status = OrderStatus.PLACED;
    private DeliveryPartner partner;

    Order(String id, String customer, Map<String, Integer> items, Restaurant r, int total) {
        this.id = id; this.customer = customer; this.items = Collections.unmodifiableMap(new TreeMap<>(items));
        this.restaurant = r; this.total = total;
    }

    int quantity() { int q = 0; for (int v : items.values()) q += v; return q; }
    synchronized OrderStatus status() { return status; }
    synchronized DeliveryPartner partner() { return partner; }
    synchronized void setPartner(DeliveryPartner p) { partner = p; }

    /** Atomic guarded transition; returns the old status. Exactly one racing caller wins. */
    synchronized OrderStatus moveTo(OrderStatus next) {
        if (!status.canMoveTo(next))
            throw new FoodException("order " + id + ": " + status + " -> " + next + " not allowed");
        OrderStatus old = status;
        status = next;
        return old;
    }
}

// ---------- strategies ----------
interface RestaurantSelector {
    Optional<Restaurant> pick(List<Restaurant> candidates, Map<String, Integer> items);   // C: Optional = a pointer that may be NULL, made explicit
}

final class LowestPriceSelector implements RestaurantSelector {
    public Optional<Restaurant> pick(List<Restaurant> c, Map<String, Integer> items) {
        return c.stream().min(Comparator.comparingInt((Restaurant r) -> r.quote(items)).thenComparing(r -> r.name));   // C: stream() = a for-loop written as filter/map/collect
    }
}

final class MaxCapacitySelector implements RestaurantSelector {
    public Optional<Restaurant> pick(List<Restaurant> c, Map<String, Integer> items) {
        return c.stream().min(Comparator.comparingInt((Restaurant r) -> -r.freeCapacity()).thenComparing(r -> r.name));
    }
}

interface PartnerAssigner { DeliveryPartner pick(List<DeliveryPartner> free); }

final class LeastDeliveriesAssigner implements PartnerAssigner {
    public DeliveryPartner pick(List<DeliveryPartner> free) {
        return Collections.min(free, Comparator.comparingInt(DeliveryPartner::deliveries).thenComparing(p -> p.id));
    }
}

// ---------- observer ----------
interface OrderListener { void onStatus(Order o, OrderStatus from, OrderStatus to); }

final class EventLog implements OrderListener {
    final List<String> lines = Collections.synchronizedList(new ArrayList<>());
    final boolean echo;
    EventLog(boolean echo) { this.echo = echo; }
    public void onStatus(Order o, OrderStatus from, OrderStatus to) {
        String s = "  [event] " + o.id + " " + from + " -> " + to;
        lines.add(s);
        if (echo) System.out.println(s);
    }
}

// ---------- dispatcher: free partners + FIFO of orders waiting for one ----------
final class Dispatcher {
    private final List<DeliveryPartner> partners = new ArrayList<>();
    private final Set<String> busy = new HashSet<>();
    private final Deque<Order> waiting = new ArrayDeque<>();
    private final PartnerAssigner assigner;
    Dispatcher(PartnerAssigner assigner) { this.assigner = assigner; }

    synchronized void add(DeliveryPartner p) { partners.add(p); }

    /** A free partner for this order, or null after queueing it. */
    synchronized DeliveryPartner assignOrQueue(Order o) {
        List<DeliveryPartner> free = new ArrayList<>();
        for (DeliveryPartner p : partners) if (!busy.contains(p.id)) free.add(p);
        if (free.isEmpty()) { waiting.addLast(o); return null; }
        DeliveryPartner p = assigner.pick(free);
        busy.add(p.id);
        return p;
    }

    /** Partner finished: hand it the oldest waiting order (it stays busy) or mark it free. */
    synchronized Order next(DeliveryPartner p) {
        Order o = waiting.pollFirst();
        if (o == null) busy.remove(p.id);
        return o;
    }

    synchronized int waitingCount() { return waiting.size(); }
}

// ---------- service facade ----------
final class FoodService {
    private final Map<String, Restaurant> restaurants = new ConcurrentHashMap<>();
    private final Map<String, Order> orders = new ConcurrentHashMap<>();
    private final Map<String, DeliveryPartner> partners = new ConcurrentHashMap<>();
    private final List<OrderListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicInteger seq = new AtomicInteger();
    private final Dispatcher dispatcher;
    private volatile RestaurantSelector selector;

    FoodService(RestaurantSelector selector, PartnerAssigner assigner) {
        this.selector = selector; this.dispatcher = new Dispatcher(assigner);
    }

    void setSelector(RestaurantSelector s) { selector = s; }
    void subscribe(OrderListener l) { listeners.add(l); }

    Restaurant onboard(String name, Map<String, Integer> menu, int capacity) {
        Restaurant r = new Restaurant(name, menu, capacity);
        if (restaurants.putIfAbsent(name, r) != null) throw new FoodException("duplicate restaurant " + name);
        return r;
    }

    void updatePrice(String restaurant, String item, int price) { get(restaurants, restaurant, "restaurant").setPrice(item, price); }

    void addPartner(String id) {
        DeliveryPartner p = new DeliveryPartner(id);
        if (partners.putIfAbsent(id, p) != null) throw new FoodException("duplicate partner " + id);
        dispatcher.add(p);
    }

    Order placeOrder(String customer, Map<String, Integer> items) {
        if (items.isEmpty() || items.values().stream().anyMatch(q -> q <= 0)) throw new FoodException("bad items " + items);
        int qty = items.values().stream().mapToInt(Integer::intValue).sum();
        List<Restaurant> candidates = new ArrayList<>();
        for (Restaurant r : restaurants.values())
            if (r.quote(items) >= 0 && r.freeCapacity() >= qty) candidates.add(r);
        while (true) {                                       // pick, then claim; retry if we lost a race
            Optional<Restaurant> pick = selector.pick(candidates, items);
            if (!pick.isPresent()) throw new FoodException("no restaurant can serve " + items + " now");
            Restaurant r = pick.get();
            int total = r.quote(items);
            if (total >= 0 && r.tryReserve(qty)) {
                Order o = new Order("o" + seq.incrementAndGet(), customer, items, r, total);
                orders.put(o.id, o);
                move(o, OrderStatus.ACCEPTED);
                return o;
            }
            candidates.remove(r);                            // full (or menu changed) since we looked
        }
    }

    /** Kitchen finished the order: free its capacity, count items, find a rider. */
    void markReady(String orderId) {
        Order o = get(orders, orderId, "order");
        move(o, OrderStatus.READY);
        o.restaurant.release(o.quantity());
        o.restaurant.recordServed(o.items);
        DeliveryPartner p = dispatcher.assignOrQueue(o);
        if (p != null) pickUp(o, p);
    }

    void deliver(String orderId) {
        Order o = get(orders, orderId, "order");
        move(o, OrderStatus.DELIVERED);
        DeliveryPartner p = o.partner();
        p.delivered();
        Order next = dispatcher.next(p);
        if (next != null) pickUp(next, p);
    }

    void cancel(String orderId) {
        Order o = get(orders, orderId, "order");
        move(o, OrderStatus.CANCELLED);                      // throws once food is READY or later
        o.restaurant.release(o.quantity());                  // only the winning transition gets here
    }

    Order order(String id) { return get(orders, id, "order"); }
    Restaurant restaurant(String name) { return get(restaurants, name, "restaurant"); }
    DeliveryPartner partner(String id) { return get(partners, id, "partner"); }
    int waitingForPartner() { return dispatcher.waitingCount(); }

    private void pickUp(Order o, DeliveryPartner p) {
        o.setPartner(p);
        move(o, OrderStatus.PICKED_UP);
    }

    private void move(Order o, OrderStatus to) {
        OrderStatus from = o.moveTo(to);
        for (OrderListener l : listeners) l.onStatus(o, from, to);   // outside the order lock
    }

    private static <T> T get(Map<String, T> m, String id, String what) {
        T v = m.get(id);
        if (v == null) throw new FoodException("unknown " + what + " " + id);
        return v;
    }
}

public class Main {
    static int checks = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }
    static Map<String, Integer> m(Object... kv) {
        Map<String, Integer> out = new LinkedHashMap<>();
        for (int i = 0; i < kv.length; i += 2) out.put((String) kv[i], (Integer) kv[i + 1]);
        return out;
    }
    static void expectFail(Runnable r, String what) {
        try { r.run(); check(false, what + " should fail"); }
        catch (FoodException e) { System.out.println("rejected: " + e.getMessage()); checks++; }
    }

    public static void main(String[] args) throws Exception {
        FoodService s = new FoodService(new LowestPriceSelector(), new LeastDeliveriesAssigner());
        s.subscribe(new EventLog(true));
        s.onboard("resta1", m("king_burger", 10, "samosa_pizza", 20, "alu_pasta", 19), 15);
        s.onboard("resta2", m("bendi_macaroni", 12, "samosa_pizza", 25, "alu_pasta", 17), 12);
        s.updatePrice("resta1", "bendi_macaroni", 8);
        s.updatePrice("resta1", "king_burger", 15);
        System.out.println(s.restaurant("resta1").describe());
        System.out.println(s.restaurant("resta2").describe());
        s.addPartner("rider1");

        System.out.println("BOOK cust1 bendi_macaroni x3, samosa_pizza x2 (lowest price)");
        Order o1 = s.placeOrder("cust1", m("bendi_macaroni", 3, "samosa_pizza", 2));
        System.out.println("  -> " + o1.id + " at " + o1.restaurant.name + " total=" + o1.total);
        check(o1.restaurant.name.equals("resta1") && o1.total == 64, "o1 at resta1 for 64");

        System.out.println("BOOK cust2 alu_pasta x3 (lowest price)");
        Order o2 = s.placeOrder("cust2", m("alu_pasta", 3));
        System.out.println("  -> " + o2.id + " at " + o2.restaurant.name + " total=" + o2.total);
        check(o2.restaurant.name.equals("resta2") && o2.total == 51, "o2 at resta2 for 51");

        s.setSelector(new MaxCapacitySelector());
        System.out.println("BOOK cust3 alu_pasta x3 (max capacity left)");
        Order o3 = s.placeOrder("cust3", m("alu_pasta", 3));
        System.out.println("  -> " + o3.id + " at " + o3.restaurant.name + " total=" + o3.total);
        check(o3.restaurant.name.equals("resta1"), "resta1 has 10 free vs 9");

        expectFail(() -> s.placeOrder("cust4", m("samosa_pizza", 13)), "13 items > any free capacity");
        expectFail(() -> s.placeOrder("cust4", m("dosa", 1)), "item on no menu");

        System.out.println("READY o1, READY o2 (only one rider)");
        s.markReady("o1");
        s.markReady("o2");
        check(s.order("o2").status() == OrderStatus.READY && s.waitingForPartner() == 1, "o2 waits for a rider");
        System.out.println("DELIVER o1 -> rider1 takes the waiting o2");
        s.deliver("o1");
        check(s.order("o2").status() == OrderStatus.PICKED_UP, "o2 auto-assigned");
        s.deliver("o2");
        check(s.partner("rider1").deliveries() == 2, "rider1 did 2 deliveries");

        System.out.println("CANCEL o3 (still cooking)");
        s.cancel("o3");
        check(s.restaurant("resta1").freeCapacity() == 15, "o3 capacity returned");
        expectFail(() -> s.cancel("o1"), "cancel after delivery");
        System.out.println("served resta1 = " + s.restaurant("resta1").served());
        check(s.restaurant("resta1").served().equals(m("bendi_macaroni", 3, "samosa_pizza", 2)), "served counts");

        System.out.println("RACE 16 threads x 1 item, one restaurant with capacity 4");
        FoodService race = new FoodService(new LowestPriceSelector(), new LeastDeliveriesAssigner());
        race.onboard("tiny", m("idli", 30), 4);
        ExecutorService pool = Executors.newFixedThreadPool(8);
        CountDownLatch go = new CountDownLatch(1);
        AtomicInteger ok = new AtomicInteger(), no = new AtomicInteger();
        List<Future<?>> fs = new ArrayList<>();
        for (int i = 0; i < 16; i++) {
            final String c = "c" + i;
            fs.add(pool.submit(() -> {
                go.await();
                try { race.placeOrder(c, m("idli", 1)); ok.incrementAndGet(); }
                catch (FoodException e) { no.incrementAndGet(); }
                return null;
            }));
        }
        go.countDown();
        for (Future<?> f : fs) f.get();
        pool.shutdown();
        System.out.println("  accepted=" + ok.get() + " rejected=" + no.get() + " free=" + race.restaurant("tiny").freeCapacity());
        check(ok.get() == 4 && no.get() == 12 && race.restaurant("tiny").freeCapacity() == 0, "never over capacity");

        System.out.println("ALL " + checks + " CHECKS PASSED");
    }

"""Food delivery (FoodKart style) - in-memory, thread-safe. Run: python main.py ; tests: python -m unittest"""
from __future__ import annotations

import itertools
import threading
from collections import deque
from enum import Enum
from typing import Callable, Dict, List, Optional, Protocol


class FoodError(Exception):
    pass


class OrderStatus(Enum):
    PLACED = "PLACED"
    ACCEPTED = "ACCEPTED"
    READY = "READY"
    PICKED_UP = "PICKED_UP"
    DELIVERED = "DELIVERED"
    CANCELLED = "CANCELLED"

    def can_move_to(self, nxt: "OrderStatus") -> bool:
        return nxt in _ALLOWED[self]


_ALLOWED = {
    OrderStatus.PLACED: {OrderStatus.ACCEPTED, OrderStatus.CANCELLED},
    OrderStatus.ACCEPTED: {OrderStatus.READY, OrderStatus.CANCELLED},
    OrderStatus.READY: {OrderStatus.PICKED_UP},
    OrderStatus.PICKED_UP: {OrderStatus.DELIVERED},
    OrderStatus.DELIVERED: set(),
    OrderStatus.CANCELLED: set(),
}


class Restaurant:
    def __init__(self, name: str, menu: Dict[str, int], capacity: int):
        if capacity <= 0:
            raise FoodError("capacity must be > 0")
        self.name = name
        self._capacity = capacity
        self._menu: Dict[str, int] = {}
        self._in_use = 0
        self._served: Dict[str, int] = {}
        self._lock = threading.Lock()
        for item, price in menu.items():
            self.set_price(item, price)

    def set_price(self, item: str, price: int) -> None:
        if price <= 0:
            raise FoodError(f"price must be > 0: {item}")
        with self._lock:
            self._menu[item] = price

    def quote(self, items: Dict[str, int]) -> int:
        """Total price, or -1 if any item is missing."""
        with self._lock:
            if any(i not in self._menu for i in items):
                return -1
            return sum(self._menu[i] * q for i, q in items.items())

    def free_capacity(self) -> int:
        with self._lock:
            return self._capacity - self._in_use

    def try_reserve(self, qty: int) -> bool:
        with self._lock:                      # check and claim under one lock
            if self._in_use + qty > self._capacity:
                return False
            self._in_use += qty
            return True

    def release(self, qty: int) -> None:
        with self._lock:
            self._in_use -= qty

    def record_served(self, items: Dict[str, int]) -> None:
        with self._lock:
            for i, q in items.items():
                self._served[i] = self._served.get(i, 0) + q

    def served(self) -> Dict[str, int]:
        with self._lock:
            return dict(sorted(self._served.items()))


class DeliveryPartner:
    def __init__(self, pid: str):
        self.id = pid
        self.deliveries = 0


class Order:
    def __init__(self, oid: str, customer: str, items: Dict[str, int], restaurant: Restaurant, total: int):
        self.id, self.customer = oid, customer
        self.items = dict(sorted(items.items()))
        self.restaurant, self.total = restaurant, total
        self.partner: Optional[DeliveryPartner] = None
        self._status = OrderStatus.PLACED
        self._lock = threading.Lock()

    @property
    def status(self) -> OrderStatus:
        with self._lock:
            return self._status

    def quantity(self) -> int:
        return sum(self.items.values())

    def move_to(self, nxt: OrderStatus) -> OrderStatus:
        with self._lock:
            if not self._status.can_move_to(nxt):
                raise FoodError(f"order {self.id}: {self._status.value} -> {nxt.value} not allowed")
            old, self._status = self._status, nxt
            return old


# ---------- strategies ----------
class RestaurantSelector(Protocol):
    def pick(self, candidates: List[Restaurant], items: Dict[str, int]) -> Optional[Restaurant]: ...


class LowestPriceSelector:
    def pick(self, candidates, items):
        return min(candidates, key=lambda r: (r.quote(items), r.name), default=None)


class MaxCapacitySelector:
    def pick(self, candidates, items):
        return min(candidates, key=lambda r: (-r.free_capacity(), r.name), default=None)


def least_deliveries(free: List[DeliveryPartner]) -> DeliveryPartner:
    return min(free, key=lambda p: (p.deliveries, p.id))


Listener = Callable[[Order, OrderStatus, OrderStatus], None]


class Dispatcher:
    def __init__(self, assigner: Callable[[List[DeliveryPartner]], DeliveryPartner]):
        self._partners: List[DeliveryPartner] = []
        self._busy: set = set()
        self._waiting: deque = deque()
        self._assign = assigner
        self._lock = threading.Lock()

    def add(self, p: DeliveryPartner) -> None:
        with self._lock:
            self._partners.append(p)

    def assign_or_queue(self, order: Order) -> Optional[DeliveryPartner]:
        with self._lock:
            free = [p for p in self._partners if p.id not in self._busy]
            if not free:
                self._waiting.append(order)
                return None
            p = self._assign(free)
            self._busy.add(p.id)
            return p

    def next(self, p: DeliveryPartner) -> Optional[Order]:
        with self._lock:
            if self._waiting:
                return self._waiting.popleft()
            self._busy.discard(p.id)
            return None

    def waiting_count(self) -> int:
        with self._lock:
            return len(self._waiting)


class FoodService:
    def __init__(self, selector, assigner=least_deliveries):
        self.selector = selector
        self._restaurants: Dict[str, Restaurant] = {}
        self._orders: Dict[str, Order] = {}
        self._partners: Dict[str, DeliveryPartner] = {}
        self._listeners: List[Listener] = []
        self._dispatcher = Dispatcher(assigner)
        self._seq = itertools.count(1)
        self._reg = threading.Lock()          # guards the three registries

    def subscribe(self, fn: Listener) -> None:
        self._listeners.append(fn)

    def onboard(self, name: str, menu: Dict[str, int], capacity: int) -> Restaurant:
        r = Restaurant(name, menu, capacity)
        with self._reg:
            if name in self._restaurants:
                raise FoodError(f"duplicate restaurant {name}")
            self._restaurants[name] = r
        return r

    def update_price(self, restaurant: str, item: str, price: int) -> None:
        self.restaurant(restaurant).set_price(item, price)

    def add_partner(self, pid: str) -> None:
        p = DeliveryPartner(pid)
        with self._reg:
            if pid in self._partners:
                raise FoodError(f"duplicate partner {pid}")
            self._partners[pid] = p
        self._dispatcher.add(p)

    def place_order(self, customer: str, items: Dict[str, int]) -> Order:
        if not items or any(q <= 0 for q in items.values()):
            raise FoodError(f"bad items {items}")
        qty = sum(items.values())
        with self._reg:
            all_r = list(self._restaurants.values())
        candidates = [r for r in all_r if r.quote(items) >= 0 and r.free_capacity() >= qty]
        while True:                              # pick, then claim; retry if we lost a race
            r = self.selector.pick(candidates, items)
            if r is None:
                raise FoodError(f"no restaurant can serve {items} now")
            total = r.quote(items)
            if total >= 0 and r.try_reserve(qty):
                o = Order(f"o{next(self._seq)}", customer, items, r, total)
                with self._reg:
                    self._orders[o.id] = o
                self._move(o, OrderStatus.ACCEPTED)
                return o
            candidates.remove(r)

    def mark_ready(self, oid: str) -> None:
        o = self.order(oid)
        self._move(o, OrderStatus.READY)
        o.restaurant.release(o.quantity())
        o.restaurant.record_served(o.items)
        p = self._dispatcher.assign_or_queue(o)
        if p is not None:
            self._pick_up(o, p)

    def deliver(self, oid: str) -> None:
        o = self.order(oid)
        self._move(o, OrderStatus.DELIVERED)
        p = o.partner
        p.deliveries += 1
        nxt = self._dispatcher.next(p)
        if nxt is not None:
            self._pick_up(nxt, p)

    def cancel(self, oid: str) -> None:
        o = self.order(oid)
        self._move(o, OrderStatus.CANCELLED)
        o.restaurant.release(o.quantity())

    def order(self, oid: str) -> Order:
        return self._get(self._orders, oid, "order")

    def restaurant(self, name: str) -> Restaurant:
        return self._get(self._restaurants, name, "restaurant")

    def partner(self, pid: str) -> DeliveryPartner:
        return self._get(self._partners, pid, "partner")

    def waiting_for_partner(self) -> int:
        return self._dispatcher.waiting_count()

    def _pick_up(self, o: Order, p: DeliveryPartner) -> None:
        o.partner = p
        self._move(o, OrderStatus.PICKED_UP)

    def _move(self, o: Order, to: OrderStatus) -> None:
        frm = o.move_to(to)
        for fn in self._listeners:
            fn(o, frm, to)

    def _get(self, d, key, what):
        with self._reg:
            v = d.get(key)
        if v is None:
            raise FoodError(f"unknown {what} {key}")
        return v

enum class Status { PLACED, ACCEPTED, READY, PICKED_UP, DELIVERED, CANCELLED };
const char* name(Status s) {
    static const char* n[] = {"PLACED", "ACCEPTED", "READY", "PICKED_UP", "DELIVERED", "CANCELLED"};
    return n[static_cast<int>(s)];

bool canMove(Status from, Status to) {                // the whole state machine in one table
    switch (from) {
        case Status::PLACED:    return to == Status::ACCEPTED || to == Status::CANCELLED;
        case Status::ACCEPTED:  return to == Status::READY || to == Status::CANCELLED;
        case Status::READY:     return to == Status::PICKED_UP;
        case Status::PICKED_UP: return to == Status::DELIVERED;
        default:                return false;
    }
}

struct FoodError : std::runtime_error { explicit FoodError(const string& m) : std::runtime_error(m) {} };

class Restaurant {
public:
    const string name;
    Restaurant(string n, const Items& menu, int cap) : name(std::move(n)), capacity_(cap) {
        if (cap <= 0) throw FoodError("capacity must be > 0");
        for (const auto& kv : menu) setPrice(kv.first, kv.second);
    }
    void setPrice(const string& item, int price) {
        if (price <= 0) throw FoodError("price must be > 0: " + item);
        LockGuard g(mu_);
        menu_[item] = price;
    }
    int quote(const Items& items) const {             // -1 if any item is missing
        LockGuard g(mu_);
        int total = 0;
        for (const auto& kv : items) {
            auto it = menu_.find(kv.first);
            if (it == menu_.end()) return -1;
            total += it->second * kv.second;
        }
        return total;
    }
    int freeCapacity() const { LockGuard g(mu_); return capacity_ - inUse_; }
    bool tryReserve(int qty) {                        // check and claim under ONE lock: no oversell
        LockGuard g(mu_);
        if (inUse_ + qty > capacity_) return false;
        inUse_ += qty;
        return true;
    }
    void release(int qty) { LockGuard g(mu_); inUse_ -= qty; }
    void recordServed(const Items& items) { LockGuard g(mu_); for (const auto& kv : items) served_[kv.first] += kv.second; }
    Items served() const { LockGuard g(mu_); return served_; }   // a copy, safe to read later
private:
    mutable Mutex mu_;
    int capacity_, inUse_ = 0;
    Items menu_, served_;
};

struct Partner { string id; int deliveries = 0; };

class Order {
public:
    const string id, customer;
    const Items items;
    Restaurant* const restaurant;                     // non-owning; the service owns restaurants
    const int total;
    Partner* partner = nullptr;
    Order(string i, string c, Items it, Restaurant* r, int t)
        : id(std::move(i)), customer(std::move(c)), items(std::move(it)), restaurant(r), total(t) {}
    Status status() const { LockGuard g(mu_); return status_; }
    int quantity() const { int q = 0; for (const auto& kv : items) q += kv.second; return q; }
    Status moveTo(Status next) {                      // per-order lock: one winner per transition
        LockGuard g(mu_);
        if (!canMove(status_, next))
            throw FoodError("order " + id + ": " + ::name(status_) + " -> " + ::name(next) + " not allowed");
        Status old = status_;
        status_ = next;
        return old;
    }
private:
    mutable Mutex mu_;
    Status status_ = Status::PLACED;
};

Restaurant* lowestPrice(const vector<Restaurant*>& c, const Items& items) {
    Restaurant* best = nullptr;
    for (Restaurant* r : c)
        if (!best || r->quote(items) < best->quote(items) ||
            (r->quote(items) == best->quote(items) && r->name < best->name)) best = r;
    return best;
}
Restaurant* maxCapacity(const vector<Restaurant*>& c, const Items&) {
    Restaurant* best = nullptr;
    for (Restaurant* r : c)
        if (!best || r->freeCapacity() > best->freeCapacity() ||
            (r->freeCapacity() == best->freeCapacity() && r->name < best->name)) best = r;
    return best;
}

typedef std::function<void(const Order&, Status, Status)> Listener;   // Observer

class FoodService {
public:
    explicit FoodService(Selector s) : selector_(std::move(s)) {}
    void setSelector(Selector s) { LockGuard g(regMu_); selector_ = std::move(s); }
    void subscribe(Listener l) { LockGuard g(regMu_); listeners_.push_back(std::move(l)); }

    Restaurant& onboard(const string& n, const Items& menu, int cap) {
        LockGuard g(regMu_);
        if (restaurants_.count(n)) throw FoodError("duplicate restaurant " + n);
        restaurants_[n].reset(new Restaurant(n, menu, cap));
        return *restaurants_[n];
    }
    void updatePrice(const string& r, const string& item, int price) { restaurant(r).setPrice(item, price); }
    void addPartner(const string& id) {
        LockGuard g(regMu_);
        if (partners_.count(id)) throw FoodError("duplicate partner " + id);
        partners_[id].reset(new Partner{id});
    }

    Order& placeOrder(const string& customer, const Items& items) {
        if (items.empty()) throw FoodError("empty order");
        int qty = 0;
        for (const auto& kv : items) { if (kv.second <= 0) throw FoodError("bad quantity"); qty += kv.second; }
        vector<Restaurant*> cand;
        Selector pick;
        {
            LockGuard g(regMu_);
            pick = selector_;                         // read the strategy once (Java: a volatile field)
            for (auto& kv : restaurants_)
                if (kv.second->quote(items) >= 0 && kv.second->freeCapacity() >= qty) cand.push_back(kv.second.get());
        }
        for (;;) {                                    // pick, claim, retry on a lost race
            Restaurant* r = cand.empty() ? nullptr : pick(cand, items);
            if (!r) throw FoodError("no restaurant can serve the order now");
            int total = r->quote(items);
            if (total >= 0 && r->tryReserve(qty)) {   // the real check, under the restaurant lock
                Order* op;
                {
                    LockGuard g(regMu_);
                    string id = "o" + std::to_string(++seq_);
                    orders_[id].reset(new Order(id, customer, items, r, total));
                    op = orders_[id].get();
                }
                Order& o = *op;
                move(o, Status::ACCEPTED);
                return o;
            }
            cand.erase(std::remove(cand.begin(), cand.end(), r), cand.end());
        }
    }

    void markReady(const string& id) {
        Order& o = order(id);
        move(o, Status::READY);
        o.restaurant->release(o.quantity());
        o.restaurant->recordServed(o.items);
        LockGuard g(dispatchMu_);                     // partners and the waiting queue
        Partner* p = assignOrQueue(o);
        if (p) pickUp(o, *p);
    }
    void deliver(const string& id) {
        Order& o = order(id);
        move(o, Status::DELIVERED);
        LockGuard g(dispatchMu_);
        Partner* p = o.partner;
        p->deliveries++;
        if (!waiting_.empty()) { Order* next = waiting_.front(); waiting_.pop_front(); pickUp(*next, *p); }
        else busy_.erase(p->id);
    }
    void cancel(const string& id) {
        Order& o = order(id);
        move(o, Status::CANCELLED);
        o.restaurant->release(o.quantity());
    }

    Order& order(const string& id) { LockGuard g(regMu_); return *find(orders_, id, "order"); }
    Restaurant& restaurant(const string& n) { LockGuard g(regMu_); return *find(restaurants_, n, "restaurant"); }
    Partner& partner(const string& id) { LockGuard g(regMu_); return *find(partners_, id, "partner"); }
    size_t waiting() { LockGuard g(dispatchMu_); return waiting_.size(); }

private:
    Mutex regMu_;                                     // maps, selector, listeners, seq_
    Mutex dispatchMu_;                                // busy_, waiting_, partner counters
    Selector selector_;
    vector<Listener> listeners_;
    map<string, std::unique_ptr<Restaurant>> restaurants_;
    map<string, std::unique_ptr<Order>> orders_;
    map<string, std::unique_ptr<Partner>> partners_;
    std::set<string> busy_;
    std::deque<Order*> waiting_;
    int seq_ = 0;

    Partner* assignOrQueue(Order& o) {                // caller holds dispatchMu_
        Partner* best = nullptr;                      // least deliveries, then id (map is id-sorted)
        LockGuard g(regMu_);                          // partners_ map (dispatch -> registry is the only nesting)
        for (auto& kv : partners_)
            if (!busy_.count(kv.first) && (!best || kv.second->deliveries < best->deliveries)) best = kv.second.get();
        if (!best) { waiting_.push_back(&o); return nullptr; }
        busy_.insert(best->id);
        return best;
    }
    void pickUp(Order& o, Partner& p) { o.partner = &p; move(o, Status::PICKED_UP); }
    void move(Order& o, Status to) {
        Status from = o.moveTo(to);
        vector<Listener> ls;
        { LockGuard g(regMu_); ls = listeners_; }     // never call out while holding a lock
        for (auto& l : ls) l(o, from, to);
    }
    template <class T>
    static T* find(map<string, std::unique_ptr<T>>& m, const string& k, const char* what) {
        auto it = m.find(k);
        if (it == m.end()) throw FoodError(string("unknown ") + what + " " + k);
        return it->second.get();
    }
};
