APIs and interfaces

// Facade used by the driver and the tests
Restaurant onboard(String name, Map<String,Integer> menu, int capacity);
void       updatePrice(String restaurant, String item, int price);
void       addPartner(String partnerId);
Order      placeOrder(String customer, Map<String,Integer> items);   // throws FoodException
void       markReady(String orderId);      // kitchen dispatched the food
void       deliver(String orderId);
void       cancel(String orderId);         // only PLACED / ACCEPTED
void       setSelector(RestaurantSelector s);
void       subscribe(OrderListener l);

// Extension points
interface RestaurantSelector { Optional<Restaurant> pick(List<Restaurant> candidates, Map<String,Integer> items); }
interface PartnerAssigner    { DeliveryPartner pick(List<DeliveryPartner> free); }
interface OrderListener      { void onStatus(Order o, OrderStatus from, OrderStatus to); }
