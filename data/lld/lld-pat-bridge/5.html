Code: notifications (Java, Python) and a mini VFS (C++14)

All three programs were compiled and run. Java builds all 9 combinations from 3 + 3 classes and shows the runtime switch. Python is the same idea in fewer lines. The C++14 tab is a tiny VFS so you can map it straight onto struct file and file_operations. Files live in lld-code/lld-pat-bridge/.

Map before you read: which class plays which role in the Java tab (BridgeDemo.java). The Python and C++ tabs use the same roles under their own names. Key lines carry a // <-- the pattern comment.

import java.util.*;

public class BridgeDemo {

    record User(String name, String phone, String email, String deviceToken) { }

    // ---- Implementor: HOW a message physically leaves the system ----
    interface Channel {
        String name();
        int maxChars();
        boolean deliver(User to, String title, String body);
    }

    static final class SmsChannel implements Channel {
        public String name() { return "SMS"; }
        public int maxChars() { return 160; }
        public boolean deliver(User to, String title, String body) {
            System.out.println("  SMS   -> " + to.phone() + " : " + body);
            return true;
        }
    }

    static final class EmailChannel implements Channel {
        public String name() { return "Email"; }
        public int maxChars() { return 10_000; }
        public boolean deliver(User to, String title, String body) {
            System.out.println("  EMAIL -> " + to.email() + " [" + title + "] " + body);
            return true;
        }
    }

    static final class PushChannel implements Channel {
        public String name() { return "Push"; }
        public int maxChars() { return 120; }
        public boolean deliver(User to, String title, String body) {
            if (to.deviceToken() == null) return false;      // app not installed
            System.out.println("  PUSH  -> " + to.deviceToken() + " " + title + ": " + body);
            return true;
        }
    }

    // ---- Abstraction: WHAT we are telling the user. Holds the bridge reference. ----
    abstract static class Notification {
        protected Channel channel;                             // the bridge
        Notification(Channel channel) { this.channel = channel; }
        void switchChannel(Channel c) { this.channel = c; }    // swap at runtime

        protected String fit(String text) {                    // uses only the Implementor API
            int max = channel.maxChars();
            return text.length() <= max ? text : text.substring(0, max - 3) + "...";
        }
        abstract boolean send(User u);
    }

    // ---- Refined abstractions: each kind of message has its own rules ----
    static final class OtpNotification extends Notification {
        private final String code;
        OtpNotification(Channel c, String code) { super(c); this.code = code; }
        boolean send(User u) {
            // <-- the pattern: the abstraction hands delivery to the implementor
            return channel.deliver(u, "Your OTP", fit(code + " is your OTP. Valid 5 min. Do not share."));
        }
    }

    static final class OrderUpdate extends Notification {
        private final String orderId, status;
        OrderUpdate(Channel c, String orderId, String status) { super(c); this.orderId = orderId; this.status = status; }
        boolean send(User u) {
            return channel.deliver(u, "Order " + orderId, fit("Hi " + u.name() + ", order " + orderId + " is " + status + "."));
        }
    }

    static final class PromoNotification extends Notification {
        private final String offer;
        private final Set<String> optedOut;
        PromoNotification(Channel c, String offer, Set<String> optedOut) { super(c); this.offer = offer; this.optedOut = optedOut; }
        boolean send(User u) {
            if (optedOut.contains(u.name())) { System.out.println("  (promo skipped: " + u.name() + " opted out)"); return false; }
            return channel.deliver(u, "Offer for you", fit(offer + " Reply STOP to opt out."));
        }
    }

    public static void main(String[] args) {
        Channel sms = new SmsChannel(), email = new EmailChannel(), push = new PushChannel();
        User asha = new User("Asha", "+91-98xxxxxx01", "asha@example.com", "tok-asha");
        User ravi = new User("Ravi", "+91-98xxxxxx02", "ravi@example.com", null);

        System.out.println("3 notification kinds x 3 channels = 9 combinations, built from 3 + 3 classes:");
        for (Channel c : List.of(sms, email, push)) {
            new OtpNotification(c, "482913").send(asha);
            new OrderUpdate(c, "ORD-77", "out for delivery").send(asha);
            new PromoNotification(c, "Flat 20% off on groceries this weekend, use code FRESH20 at checkout. "
                    + "Valid on orders above Rs 499 in selected cities only, while stocks last.", Set.of()).send(asha);
        }

        System.out.println("Runtime switch: push fails for Ravi, fall back to SMS:");
        Notification n = new OrderUpdate(push, "ORD-78", "delivered");
        if (!n.send(ravi)) { n.switchChannel(sms); n.send(ravi); }

        new PromoNotification(email, "Weekend sale!", Set.of("Ravi")).send(ravi);
    }
}

Look at the promo line: SMS cut it at 160 characters, push at 120, email sent it whole. That rule lives once, in Notification.fit(), on the abstraction side.

"""Bridge: notification kinds (abstraction) x delivery channels (implementation)."""
from abc import ABC, abstractmethod


class Channel(ABC):                                   # Implementor
    max_chars = 10_000

    @abstractmethod
    def deliver(self, to: dict, title: str, body: str) -> bool: ...


class Sms(Channel):
    max_chars = 160

    def deliver(self, to, title, body):
        print(f"  SMS   -> {to['phone']}: {body}")
        return True


class Email(Channel):
    def deliver(self, to, title, body):
        print(f"  EMAIL -> {to['email']} [{title}] {body}")
        return True


class Push(Channel):
    max_chars = 120

    def deliver(self, to, title, body):
        if not to.get("token"):
            return False
        print(f"  PUSH  -> {to['token']} {title}: {body}")
        return True


class Notification(ABC):                              # Abstraction
    def __init__(self, channel: Channel):
        self.channel = channel                        # the bridge

    def fit(self, text: str) -> str:
        m = self.channel.max_chars
        return text if len(text) <= m else text[: m - 3] + "..."

    @abstractmethod
    def send(self, user: dict) -> bool: ...


class Otp(Notification):                              # Refined abstractions
    def __init__(self, channel, code):
        super().__init__(channel)
        self.code = code

    def send(self, user):
        return self.channel.deliver(user, "Your OTP", self.fit(f"{self.code} is your OTP. Valid 5 min."))


class OrderUpdate(Notification):
    def __init__(self, channel, order_id, status):
        super().__init__(channel)
        self.order_id, self.status = order_id, status

    def send(self, user):
        return self.channel.deliver(user, f"Order {self.order_id}",
                                    self.fit(f"Hi {user['name']}, order {self.order_id} is {self.status}."))


if __name__ == "__main__":
    asha = {"name": "Asha", "phone": "+91-98xxxxxx01", "email": "asha@example.com", "token": "tok-asha"}

    for ch in (Sms(), Email(), Push()):               # 2 kinds x 3 channels, 5 classes
        Otp(ch, "482913").send(asha)

    n = OrderUpdate(Push(), "ORD-78", "delivered")
    if not n.send(ravi):                              # swap the implementation at runtime
        n.channel = Sms()
        n.send(ravi)

// Bridge in C++14: a tiny VFS. File (abstraction) x FileOps (implementation).
// Build: g++ -std=c++14 -Wall bridge.cpp -o bridge
#include <iostream>
#include <memory>
#include <string>

// Implementor: what every "filesystem" or driver must provide,
// like struct file_operations in the kernel.
class FileOps {
public:
    virtual ~FileOps() = default;
    virtual std::string name() const = 0;
    virtual long read(std::string& out, long pos, long len) = 0;
    virtual long write(const std::string& data, long pos) = 0;
};

class RamFs : public FileOps {                // like a tmpfs file
    std::string data_;
public:
    std::string name() const override { return "ramfs"; }
    long read(std::string& out, long pos, long len) override {
        if (pos >= (long)data_.size()) return 0;               // EOF
        out = data_.substr(pos, len);
        return (long)out.size();
    }
    long write(const std::string& d, long pos) override {
        if ((long)data_.size() < pos + (long)d.size()) data_.resize(pos + d.size());
        data_.replace(pos, d.size(), d);
        return (long)d.size();
    }
};

class NullDev : public FileOps {              // like /dev/null
public:
    std::string name() const override { return "null"; }
    long read(std::string& out, long, long) override { out.clear(); return 0; }
    long write(const std::string& d, long) override { return (long)d.size(); }   // swallow
};

// Abstraction: what user code sees. Owns the position, holds the bridge.
class File {
protected:
    std::shared_ptr<FileOps> ops_;             // the bridge (like file->f_op)
    long pos_ = 0;
public:
    explicit File(std::shared_ptr<FileOps> ops) : ops_(std::move(ops)) {}
    virtual ~File() = default;
    virtual long write(const std::string& d) { long n = ops_->write(d, pos_); pos_ += n; return n; }
    std::string read(long len) { std::string s; pos_ += ops_->read(s, pos_, len); return s; }
    void seek(long p) { pos_ = p; }
    std::string backend() const { return ops_->name(); }
};

// Refined abstractions: new behaviour on the abstraction side only.
class AppendOnlyFile : public File {          // like O_APPEND
public:
    using File::File;
    long write(const std::string& d) override {
        std::string chunk;
        long end = 0;
        while (ops_->read(chunk, end, 64) > 0) end += (long)chunk.size();   // find the end
        long n = ops_->write(d, end);
        pos_ = end + n;
        return n;
    }
};

class ReadOnlyFile : public File {            // like opening with O_RDONLY
public:
    using File::File;
    long write(const std::string&) override { std::cout << "  EBADF: read-only\n"; return -1; }
};

int main() {
    auto ram = std::make_shared<RamFs>();
    File f(ram);
    f.write("hello kernel");
    f.seek(0);
    std::cout << "File/" << f.backend() << " read: " << f.read(5) << "\n";

    AppendOnlyFile log(ram);                   // same backend, different abstraction
    log.seek(0);
    log.write(" + appended");
    ReadOnlyFile ro(ram);
    ro.write("oops");
    std::cout << "ReadOnly/" << ro.backend() << " read: " << ro.read(100) << "\n";

    AppendOnlyFile sink(std::make_shared<NullDev>());   // same abstraction, different backend
    std::cout << "Append/" << sink.backend() << " wrote " << sink.write("discard me")
              << " bytes, read back '" << sink.read(10) << "'\n";
    return 0;
}

Same backend (RamFs) under three abstractions (File, AppendOnlyFile, ReadOnlyFile), and the same abstraction (AppendOnlyFile) over two backends. ops_ plays the role of file->f_op; pos_ plays file->f_pos, which the VFS, not the filesystem, owns.
