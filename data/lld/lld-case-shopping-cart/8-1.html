Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was All 4 checks passed.

// Shopping cart with coupons - C++14 version.
// Specification (Rule), Strategy (DiscountStrategy), Decorator (PriceCalculator layers).
// The MinGW 6.3 toolchain here has no std::thread / std::mutex, so this version is single-threaded;
// the Java and Python tabs show the locking.
#include <algorithm>
#include <cassert>
#include <cstdio>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
#include <mutex>
// Locking: one facade lock, a real std::recursive_mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). Recursive because public methods call each other,
// just as Java's synchronized is re-entrant. MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and
// -pthread fails to link there, so the default build uses a no-op Mutex with the same lock()/unlock():
// the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::recursive_mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

typedef long long Paise;

std::string fmt(Paise p) {
    std::string cents = std::to_string(p % 100);       // MinGW printf has no %lld
    return std::to_string(p / 100) + "." + (cents.size() < 2 ? "0" : "") + cents;
}
Paise rupees(long long r) { return r * 100; }

struct ShopError : std::runtime_error {
    explicit ShopError(const std::string& m) : std::runtime_error(m) {}
};

enum class Category { ELECTRONICS, FASHION, GROCERY };
const char* name(Category c) {
    switch (c) {
        case Category::ELECTRONICS: return "ELECTRONICS";
        case Category::FASHION: return "FASHION";
        default: return "GROCERY";
    }
}

struct Product {
    std::string id, name;
    Category category;
    Paise price;
};

struct CartLine {
    const Product* product;
    int qty;
    Paise total() const { return product->price * qty; }
};

// Dates as yyyymmdd integers keep the demo free of <ctime> quirks.
struct PricingContext {
    std::string user;
    int today;
    int pastOrders;
    std::map<std::string, int> couponUses;
    std::vector<CartLine> lines;
    Paise subtotal() const {
        Paise s = 0;
        for (const auto& l : lines) s += l.total();
        return s;
    }
    Paise subtotalOf(Category c) const {
        Paise s = 0;
        for (const auto& l : lines) if (l.product->category == c) s += l.total();
        return s;
    }
};

// ---------- Specification: "" = satisfied, otherwise the reason ----------
struct Rule {
    std::function<std::string(const PricingContext&)> violation;
    Rule operator&&(const Rule& other) const {
        auto a = violation, b = other.violation;
        return Rule{[a, b](const PricingContext& c) { std::string v = a(c); return v.empty() ? b(c) : v; }};
    }
};
Rule minSubtotal(Paise p) {
    return Rule{[p](const PricingContext& c) {
        return c.subtotal() >= p ? std::string() : "needs a cart value of at least " + fmt(p);
    }};
}
Rule hasCategory(Category cat) {
    return Rule{[cat](const PricingContext& c) {
        return c.subtotalOf(cat) > 0 ? std::string() : std::string("needs a ") + name(cat) + " item in the cart";
    }};
}
Rule firstOrderOnly() {
    return Rule{[](const PricingContext& c) { return c.pastOrders == 0 ? std::string() : std::string("is for the first order only"); }};
}
Rule validUntil(int yyyymmdd) {
    return Rule{[yyyymmdd](const PricingContext& c) {
        if (c.today <= yyyymmdd) return std::string();
        char buf[32];
        std::snprintf(buf, sizeof buf, "expired on %04d-%02d-%02d", yyyymmdd / 10000, yyyymmdd / 100 % 100, yyyymmdd % 100);
        return std::string(buf);
    }};
}

// ---------- Strategy ----------
struct DiscountStrategy {
    virtual ~DiscountStrategy() {}
    virtual Paise discount(const PricingContext& ctx, Paise running) const = 0;
};
struct FlatOff : DiscountStrategy {
    Paise amount;
    explicit FlatOff(Paise a) : amount(a) {}
    Paise discount(const PricingContext&, Paise running) const override { return std::min(amount, running); }
};
struct PercentOff : DiscountStrategy {
    long long bp; Paise cap;
    PercentOff(long long b, Paise c) : bp(b), cap(c) {}
    Paise discount(const PricingContext&, Paise running) const override {
        Paise d = running * bp / 10000;
        return cap > 0 ? std::min(d, cap) : d;
    }
};
struct CategoryPercentOff : DiscountStrategy {
    Category cat; long long bp;
    CategoryPercentOff(Category c, long long b) : cat(c), bp(b) {}
    Paise discount(const PricingContext& ctx, Paise running) const override {
        return std::min(ctx.subtotalOf(cat) * bp / 10000, running);
    }
};

struct Coupon {
    std::string code;
    Rule rule;
    std::shared_ptr<DiscountStrategy> strategy;
    int priority;
    bool stackable;
    int maxUsesPerUser;
    std::string whyNot(const PricingContext& ctx) const {
        auto it = ctx.couponUses.find(code);
        if (it != ctx.couponUses.end() && it->second >= maxUsesPerUser) return "already used";
        return rule.violation(ctx);
    }
};

struct Bill {
    Paise subtotal = 0;
    std::vector<std::pair<std::string, Paise>> discounts;
    std::vector<std::string> notes;
    Paise total() const {
        Paise t = subtotal;
        for (const auto& d : discounts) t -= d.second;
        return t;
    }
};

// ---------- Decorator ----------
struct PriceCalculator {
    virtual ~PriceCalculator() {}
    virtual Bill calculate(const PricingContext& ctx) const = 0;
};
struct BasePrice : PriceCalculator {
    Bill calculate(const PricingContext& ctx) const override { Bill b; b.subtotal = ctx.subtotal(); return b; }
};
struct CouponLayer : PriceCalculator {
    std::unique_ptr<PriceCalculator> inner;
    const Coupon* coupon;
    CouponLayer(std::unique_ptr<PriceCalculator> in, const Coupon* c) : inner(std::move(in)), coupon(c) {}
    Bill calculate(const PricingContext& ctx) const override {
        Bill b = inner->calculate(ctx);
        std::string why = coupon->whyNot(ctx);
        if (!why.empty()) { b.notes.push_back(coupon->code + " not applied: " + why); return b; }
        Paise d = std::min(coupon->strategy->discount(ctx, b.total()), b.total());
        if (d > 0) b.discounts.push_back(std::make_pair(coupon->code, d));
        else b.notes.push_back(coupon->code + " gives nothing on this cart");
        return b;
    }
};

struct Cart {
    std::vector<CartLine> lines;       // keeps insertion order
    std::vector<std::string> coupons;
    int qtyOf(const std::string& pid) const {
        for (const auto& l : lines) if (l.product->id == pid) return l.qty;
        return 0;
    }
    void setQty(const Product* p, int qty) {
        if (qty < 0) throw ShopError("quantity cannot be negative");
        for (auto it = lines.begin(); it != lines.end(); ++it)
            if (it->product->id == p->id) {
                if (qty == 0) lines.erase(it); else it->qty = qty;
                return;
            }
        if (qty > 0) lines.push_back(CartLine{p, qty});
    }
};

struct Order { std::string id; Bill bill; };

class ShopService {
    mutable Mutex mu_;   // the facade lock: every public call holds it
    std::map<std::string, Product> catalog;
    std::map<std::string, int> stock;
    std::map<std::string, Coupon> coupons;
    std::map<std::string, Cart> carts;
    std::map<std::string, std::map<std::string, int>> uses;
    std::map<std::string, int> orders;
    int seq = 0;

public:
    int today;
    explicit ShopService(int t) : today(t) {}

    void addProduct(const Product& p, int qty) { LockGuard g(mu_); catalog[p.id] = p; stock[p.id] += qty; }
    void addCoupon(const Coupon& c) { LockGuard g(mu_); coupons[c.code] = c; }
    int stockOf(const std::string& pid) const { LockGuard g(mu_); auto it = stock.find(pid); return it == stock.end() ? 0 : it->second; }

    const Product* product(const std::string& id) const { LockGuard g(mu_);
        auto it = catalog.find(id);
        if (it == catalog.end()) throw ShopError("unknown product " + id);
        return &it->second;
    }
    void addItem(const std::string& u, const std::string& pid, int qty) { LockGuard g(mu_);
        if (qty <= 0) throw ShopError("quantity must be positive");
        const Product* p = product(pid);
        Cart& c = carts[u];
        int want = c.qtyOf(pid) + qty;
        if (want > stockOf(pid)) throw ShopError("only " + std::to_string(stockOf(pid)) + " " + p->name + " left");
        c.setQty(p, want);
    }
    void removeItem(const std::string& u, const std::string& pid) { LockGuard g(mu_); carts[u].setQty(product(pid), 0); }

    PricingContext context(const std::string& u) { LockGuard g(mu_);
        return PricingContext{u, today, orders[u], uses[u], carts[u].lines};
    }
    void applyCoupon(const std::string& u, const std::string& code) { LockGuard g(mu_);
        auto it = coupons.find(code);
        if (it == coupons.end()) throw ShopError("unknown coupon " + code);
        const Coupon& k = it->second;
        Cart& c = carts[u];
        if (std::find(c.coupons.begin(), c.coupons.end(), code) != c.coupons.end()) throw ShopError(code + " is already applied");
        bool exclusiveThere = false;
        std::string list;
        for (const auto& x : c.coupons) {
            if (!coupons[x].stackable) exclusiveThere = true;
            list += (list.empty() ? "" : ",") + x;
        }
        if (!c.coupons.empty() && (!k.stackable || exclusiveThere)) throw ShopError(code + " cannot be combined with " + list);
        std::string why = k.whyNot(context(u));
        if (!why.empty()) throw ShopError(code + " " + why);
        c.coupons.push_back(code);
    }
    Bill price(const std::string& u) { LockGuard g(mu_);
        std::vector<const Coupon*> layers;
        for (const auto& x : carts[u].coupons) layers.push_back(&coupons[x]);
        std::sort(layers.begin(), layers.end(), [](const Coupon* a, const Coupon* b) {
            return a->priority != b->priority ? a->priority < b->priority : a->code < b->code;
        });
        std::unique_ptr<PriceCalculator> calc(new BasePrice());
        for (const Coupon* k : layers) calc.reset(new CouponLayer(std::move(calc), k));
        return calc->calculate(context(u));
    }
    Order checkout(const std::string& u) { LockGuard g(mu_);
        Cart& c = carts[u];
        if (c.lines.empty()) throw ShopError("cart is empty");
        Bill bill = price(u);
        for (const auto& l : c.lines)                         // check all first
            if (stockOf(l.product->id) < l.qty) throw ShopError("out of stock: " + l.product->name);
        for (const auto& l : c.lines) stock[l.product->id] -= l.qty;   // then apply
        for (const auto& d : bill.discounts) uses[u][d.first]++;
        orders[u]++;
        c = Cart();
        return Order{"O" + std::to_string(++seq), bill};
    }
};

static int checks = 0;
void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "check failed: " << what << "\n"; std::exit(1); }
    ++checks;
}
void show(ShopService& s, const std::string& u) {
    Bill b = s.price(u);
    std::printf("  %-22s %10s\n", "Subtotal", fmt(b.subtotal).c_str());
    for (const auto& d : b.discounts) std::printf("  %-22s %10s\n", ("- " + d.first).c_str(), fmt(d.second).c_str());
    for (const auto& n : b.notes) std::printf("  note: %s\n", n.c_str());
    std::printf("  %-22s %10s\n", "Total", fmt(b.total()).c_str());
}
void tryApply(ShopService& s, const std::string& u, const std::string& code) {
    try { s.applyCoupon(u, code); std::cout << "apply " << code << " -> ok\n"; }
    catch (const ShopError& e) { std::cout << "apply " << code << " -> rejected: " << e.what() << "\n"; }
}

int main() {
    ShopService shop(20261123);
    shop.addProduct(Product{"P1", "Laptop", Category::ELECTRONICS, rupees(50000)}, 2);
    shop.addProduct(Product{"P2", "Headphones", Category::ELECTRONICS, rupees(2000)}, 5);
    shop.addProduct(Product{"P3", "T-shirt", Category::FASHION, rupees(799)}, 10);
    shop.addCoupon(Coupon{"FASHION20", hasCategory(Category::FASHION),
                          std::make_shared<CategoryPercentOff>(Category::FASHION, 2000), 10, true, 99});
    shop.addCoupon(Coupon{"SAVE10", validUntil(20261130), std::make_shared<PercentOff>(1000, rupees(500)), 20, true, 99});
    shop.addCoupon(Coupon{"FLAT200", minSubtotal(rupees(1000)), std::make_shared<FlatOff>(rupees(200)), 30, false, 99});
    shop.addCoupon(Coupon{"NEWUSER", firstOrderOnly() && validUntil(20261231), std::make_shared<FlatOff>(rupees(100)), 30, true, 1});

    std::cout << "== u1: Laptop x1, T-shirt x2\n";
    shop.addItem("u1", "P1", 1);
    shop.addItem("u1", "P3", 2);
    tryApply(shop, "u1", "FASHION20");
    tryApply(shop, "u1", "SAVE10");
    tryApply(shop, "u1", "FLAT200");
    show(shop, "u1");
    Order o1 = shop.checkout("u1");
    std::cout << "checkout u1 -> " << o1.id << " paid " << fmt(o1.bill.total()) << "\n";
    check(o1.bill.total() == 5077840, "u1 total");
    check(shop.stockOf("P1") == 1 && shop.stockOf("P3") == 8, "stock decremented");

    std::cout << "== u2: small cart, then the cart changes after a coupon\n";
    shop.addItem("u2", "P3", 1);
    tryApply(shop, "u2", "FLAT200");
    tryApply(shop, "u2", "FASHION20");
    tryApply(shop, "u2", "NEWUSER");
    shop.addItem("u2", "P2", 1);
    shop.removeItem("u2", "P3");
    show(shop, "u2");
    Order o2 = shop.checkout("u2");
    std::cout << "checkout u2 -> " << o2.id << " paid " << fmt(o2.bill.total()) << "\n";
    check(o2.bill.total() == rupees(1900), "FASHION20 dropped, NEWUSER kept");
    shop.addItem("u2", "P3", 1);
    tryApply(shop, "u2", "NEWUSER");

    std::cout << "== last laptop, two buyers (sequential here)\n";
    shop.addItem("u3", "P1", 1);
    shop.addItem("u4", "P1", 1);
    shop.checkout("u3");
    try { shop.checkout("u4"); } catch (const ShopError& e) { std::cout << "u4 -> " << e.what() << "\n"; }
    check(shop.stockOf("P1") == 0, "no oversell");
    std::cout << "All " << checks << " checks passed\n";
    return 0;
}

Read the u2 block closely: FASHION20 was valid when it was applied (a T-shirt was in the cart), then the T-shirt was removed. The coupon stays attached, gives nothing, and the bill explains why. That is the "cart changed after applying" edge case most candidates never test.
