  1. Enums and data classes. Category, the Cart with its lines and coupon codes (every method locks the cart).
  2. One interface, one implementation. DiscountStrategy answers "how much off"; PercentOff is one answer, capped.
  3. The facade's core method. checkout holds the cart lock, prices the cart, then asks Inventory.reserveAll for all lines at once (lock order: cart, then inventory).
  4. The driver. main runs the u1 and u2 carts, bad input and the last-laptop race.

How to read the full code

Reading order: Read Product, Cart and Inventory first, then Rule and DiscountStrategy, then Coupon, then ShopService.price and checkout. Skip Bill's formatting and the PriceCalculator layers on a first read.

Java for C readers: the idioms this page's code uses (open if a line looks strange)
You will seeRead it as
x -> x.price (lambda)An anonymous function that can capture local variables: a callback and its void *ctx in one value. (a, b) -> a.x - b.x is the cmp you would hand to qsort.
Type::method (method reference)A pointer to a function that already exists, written without a wrapper: Post::score is &post_score.
Optional<T>A pointer that may be NULL, made explicit in the type, so the caller has to say what happens when it is empty (orElse, isPresent, map).
.stream().filter(..).map(..).collect(..)A for-loop written as a pipeline: filter is if (!pred(x)) continue;, map transforms each element, collect/toList appends the survivors to a new array.
map.computeIfAbsent(k, f)Lookup-or-create in one call: find k; if it is missing, call f, insert the result and return it.
map.merge(k, 1, Integer::sum)counts[k] += 1, inserting the first value when k is absent.
Comparator.comparing(f).reversed().thenComparing(g)Builds the cmp() function field by field: compare by f, negate the result, break ties with g.
default method in an interfaceAn ops-table slot pre-filled with a generic implementation; an implementing class overrides only the slots it cares about.
synchronizedTake the object's built-in mutex on entry and release it on every exit. It is reentrant: the owning thread may enter again.
throw new XException(..)Unwinds to the nearest catch: an error return that a caller cannot silently ignore.

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

// ---------- money: integer paise, formatted only at the edge ----------
final class Money {
    private Money() {}
    static long of(String rupees) { return new BigDecimal(rupees).movePointRight(2).longValueExact(); }
    static String fmt(long paise) { return String.format("%d.%02d", paise / 100, paise % 100); }
}

enum Category { ELECTRONICS, FASHION, GROCERY }

final class Product {
    final String id, name;
    final Category category;
    final long pricePaise;
    Product(String id, String name, Category category, long pricePaise) {
        this.id = id; this.name = name; this.category = category; this.pricePaise = pricePaise;
    }
}

class ShopException extends RuntimeException {
    private static final long serialVersionUID = 1L;
    ShopException(String msg) { super(msg); }
}

// ---------- inventory: all-or-nothing reservation under one lock ----------
final class Inventory {
    private final Map<String, Integer> stock = new HashMap<>();
    synchronized void add(String productId, int qty) { stock.merge(productId, qty, Integer::sum); }   // C: Type::method = a pointer to an existing function
    synchronized int available(String productId) { return stock.getOrDefault(productId, 0); }
    /** Check every line first, then decrement: a failed checkout leaves stock untouched. */
    synchronized void reserveAll(Map<Product, Integer> want) {
        for (Map.Entry<Product, Integer> e : want.entrySet())
            if (available(e.getKey().id) < e.getValue())
                throw new ShopException("out of stock: " + e.getKey().name);
        for (Map.Entry<Product, Integer> e : want.entrySet())
            stock.merge(e.getKey().id, -e.getValue(), Integer::sum);
    }
}

// ---------- cart ----------
final class CartLine {
    final Product product;
    final int qty;
    CartLine(Product product, int qty) { this.product = product; this.qty = qty; }
    long total() { return product.pricePaise * qty; }
}

final class Cart {

    synchronized void clear() { lines.clear(); coupons.clear(); }
}

// ---------- what a rule or a discount may look at ----------
final class PricingContext {
    final String userId;
    final LocalDate today;
    final int pastOrders;
    final Map<String, Integer> couponUses;   // code -> times this user already used it
    final List<CartLine> lines;
    final long subtotal;
    PricingContext(String userId, LocalDate today, int pastOrders, Map<String, Integer> couponUses, List<CartLine> lines) {
        this.userId = userId; this.today = today; this.pastOrders = pastOrders;
        this.couponUses = couponUses; this.lines = lines;
        long s = 0;
        for (CartLine l : lines) s += l.total();
        this.subtotal = s;
    }
    long subtotalOf(Category c) {
        long s = 0;
        for (CartLine l : lines) if (l.product.category == c) s += l.total();
        return s;
    }
}

// ---------- Specification: composable eligibility rules ----------
interface Rule {
    /** empty = satisfied; otherwise the reason it is not. */
    Optional<String> violation(PricingContext ctx);   // C: Optional = a pointer that may be NULL, made explicit
    default boolean isSatisfiedBy(PricingContext ctx) { return !violation(ctx).isPresent(); }   // C: default method = an ops-table slot with a generic body
    default Rule and(Rule other) {
        return ctx -> { Optional<String> v = violation(ctx); return v.isPresent() ? v : other.violation(ctx); };
    }
    static Rule always() { return ctx -> Optional.empty(); }
    static Rule minSubtotal(long paise) {
        return ctx -> ctx.subtotal >= paise ? Optional.empty()
                : Optional.of("needs a cart value of at least " + Money.fmt(paise));
    }
    static Rule hasCategory(Category c) {
        return ctx -> ctx.subtotalOf(c) > 0 ? Optional.empty() : Optional.of("needs a " + c + " item in the cart");
    }
    static Rule firstOrderOnly() {
        return ctx -> ctx.pastOrders == 0 ? Optional.empty() : Optional.of("is for the first order only");
    }
    static Rule validUntil(LocalDate last) {
        return ctx -> !ctx.today.isAfter(last) ? Optional.empty() : Optional.of("expired on " + last);
    }
}

// ---------- Strategy: how much a coupon takes off ----------
interface DiscountStrategy {
    long discount(PricingContext ctx, long runningTotal);
}

final class FlatOff implements DiscountStrategy {
    private final long paise;
    FlatOff(long paise) { this.paise = paise; }
    public long discount(PricingContext ctx, long running) { return Math.min(paise, running); }
}

final class PercentOff implements DiscountStrategy {
    private final long basisPoints, capPaise;            // 1000 bp = 10%; cap 0 = no cap
    PercentOff(long basisPoints, long capPaise) { this.basisPoints = basisPoints; this.capPaise = capPaise; }
    public long discount(PricingContext ctx, long running) {
        long d = running * basisPoints / 10000;
        return capPaise > 0 ? Math.min(d, capPaise) : d;
    }
}

final class CategoryPercentOff implements DiscountStrategy {
    private final Category category;
    private final long basisPoints;
    CategoryPercentOff(Category category, long basisPoints) { this.category = category; this.basisPoints = basisPoints; }
    public long discount(PricingContext ctx, long running) {
        return Math.min(ctx.subtotalOf(category) * basisPoints / 10000, running);
    }
}

final class Coupon {
    final String code;
    final Rule rule;
    final DiscountStrategy strategy;
    final int priority;          // lower runs first: item-level before cart-level
    final boolean stackable;
    final int maxUsesPerUser;
    Coupon(String code, Rule rule, DiscountStrategy strategy, int priority, boolean stackable, int maxUsesPerUser) {
        this.code = code; this.rule = rule; this.strategy = strategy;
        this.priority = priority; this.stackable = stackable; this.maxUsesPerUser = maxUsesPerUser;
    }
    Optional<String> whyNot(PricingContext ctx) {
        if (ctx.couponUses.getOrDefault(code, 0) >= maxUsesPerUser) return Optional.of("already used");
        return rule.violation(ctx);
    }
}

// ---------- the bill: an immutable value that each layer extends ----------
final class Bill {
    final long subtotal;
    final List<String> discountCodes;
    final List<Long> discountAmounts;
    final List<String> notes;
    Bill(long subtotal) { this(subtotal, new ArrayList<>(), new ArrayList<>(), new ArrayList<>()); }
    private Bill(long subtotal, List<String> codes, List<Long> amounts, List<String> notes) {
        this.subtotal = subtotal;
        this.discountCodes = Collections.unmodifiableList(codes);
        this.discountAmounts = Collections.unmodifiableList(amounts);
        this.notes = Collections.unmodifiableList(notes);
    }
    long total() { long t = subtotal; for (long d : discountAmounts) t -= d; return t; }
    Bill withDiscount(String code, long amount) {
        List<String> c = new ArrayList<>(discountCodes); c.add(code);
        List<Long> a = new ArrayList<>(discountAmounts); a.add(amount);
        return new Bill(subtotal, c, a, new ArrayList<>(notes));
    }
    Bill withNote(String note) {
        List<String> n = new ArrayList<>(notes); n.add(note);
        return new Bill(subtotal, new ArrayList<>(discountCodes), new ArrayList<>(discountAmounts), n);
    }
}

// ---------- Decorator: price = base wrapped by one layer per coupon ----------
interface PriceCalculator { Bill calculate(PricingContext ctx); }

final class BasePrice implements PriceCalculator {
    public Bill calculate(PricingContext ctx) { return new Bill(ctx.subtotal); }
}

final class CouponLayer implements PriceCalculator {
    private final PriceCalculator inner;
    private final Coupon coupon;
    CouponLayer(PriceCalculator inner, Coupon coupon) { this.inner = inner; this.coupon = coupon; }
    public Bill calculate(PricingContext ctx) {
        Bill bill = inner.calculate(ctx);
        Optional<String> why = coupon.whyNot(ctx);
        if (why.isPresent()) return bill.withNote(coupon.code + " not applied: " + why.get());
        long d = Math.min(coupon.strategy.discount(ctx, bill.total()), bill.total());
        return d > 0 ? bill.withDiscount(coupon.code, d) : bill.withNote(coupon.code + " gives nothing on this cart");
    }
}

final class Order {
    final String id, userId;
    final List<CartLine> lines;
    final Bill bill;
    Order(String id, String userId, List<CartLine> lines, Bill bill) {
        this.id = id; this.userId = userId; this.lines = Collections.unmodifiableList(lines); this.bill = bill;
    }
}

// ---------- facade ----------
final class ShopService {
    private final Map<String, Product> catalog = new ConcurrentHashMap<>();
    private final Map<String, Coupon> coupons = new ConcurrentHashMap<>();
    private final Map<String, Cart> carts = new ConcurrentHashMap<>();
    private final Map<String, Map<String, Integer>> uses = new ConcurrentHashMap<>();
    private final Map<String, Integer> orderCount = new ConcurrentHashMap<>();
    private final Inventory inventory = new Inventory();
    private final AtomicInteger orderSeq = new AtomicInteger();
    private volatile LocalDate today;

    ShopService(LocalDate today) { this.today = today; }
    void setToday(LocalDate d) { today = d; }

    void addProduct(Product p, int stock) { catalog.put(p.id, p); inventory.add(p.id, stock); }
    void addCoupon(Coupon c) { coupons.put(c.code, c); }
    int stockOf(String productId) { return inventory.available(productId); }

    private Cart cart(String userId) { return carts.computeIfAbsent(userId, Cart::new); }   // C: look up k; if missing, allocate, insert and return it
    private Product product(String id) {
        Product p = catalog.get(id);
        if (p == null) throw new ShopException("unknown product " + id);
        return p;
    }

    void addItem(String userId, String productId, int qty) {
        if (qty <= 0) throw new ShopException("quantity must be positive");
        Product p = product(productId);
        Cart c = cart(userId);
        synchronized (c) {
            int want = c.qtyOf(productId) + qty;
            if (want > inventory.available(productId))       // soft check; checkout re-checks
                throw new ShopException("only " + inventory.available(productId) + " " + p.name + " left");
            c.setQty(p, want);
        }
    }
    void updateQty(String userId, String productId, int qty) { cart(userId).setQty(product(productId), qty); }
    void removeItem(String userId, String productId) { updateQty(userId, productId, 0); }

    private PricingContext context(String userId, Cart c) {
        return new PricingContext(userId, today, orderCount.getOrDefault(userId, 0),
                new HashMap<>(uses.getOrDefault(userId, Collections.emptyMap())), c.lines());
    }

    void applyCoupon(String userId, String code) {
        Coupon coupon = coupons.get(code);
        if (coupon == null) throw new ShopException("unknown coupon " + code);
        Cart c = cart(userId);
        synchronized (c) {
            List<String> applied = c.coupons();
            if (applied.contains(code)) throw new ShopException(code + " is already applied");
            boolean exclusiveThere = applied.stream().anyMatch(x -> !coupons.get(x).stackable);   // C: stream() = a for-loop written as filter/map/collect
            if (!applied.isEmpty() && (!coupon.stackable || exclusiveThere))
                throw new ShopException(code + " cannot be combined with " + String.join(",", applied));
            Optional<String> why = coupon.whyNot(context(userId, c));
            if (why.isPresent()) throw new ShopException(code + " " + why.get());
            c.addCoupon(code);
        }
    }
    void removeCoupon(String userId, String code) { cart(userId).removeCoupon(code); }

    Bill price(String userId) {
        Cart c = cart(userId);
        synchronized (c) { return priceLocked(userId, c); }
    }

    private Bill priceLocked(String userId, Cart c) {
        List<Coupon> list = new ArrayList<>();
        for (String code : c.coupons()) list.add(coupons.get(code));
        list.sort(Comparator.comparingInt((Coupon x) -> x.priority).thenComparing(x -> x.code));   // C: builds the cmp() function you would pass to qsort
        PriceCalculator calc = new BasePrice();
        for (Coupon k : list) calc = new CouponLayer(calc, k);
        return calc.calculate(context(userId, c));
    }

    Order checkout(String userId) {
        Cart c = cart(userId);
        synchronized (c) {                                     // lock order: cart, then inventory
            List<CartLine> lines = c.lines();
            if (lines.isEmpty()) throw new ShopException("cart is empty");
            Bill bill = priceLocked(userId, c);
            Map<Product, Integer> want = new LinkedHashMap<>();
            for (CartLine l : lines) want.put(l.product, l.qty);
            inventory.reserveAll(want);                        // throws before anything changes
            Map<String, Integer> mine = uses.computeIfAbsent(userId, k -> new ConcurrentHashMap<>());
            for (String code : bill.discountCodes) mine.merge(code, 1, Integer::sum);
            orderCount.merge(userId, 1, Integer::sum);
            c.clear();
            return new Order("O" + orderSeq.incrementAndGet(), userId, lines, bill);
        }
    }
}

public class Main {
    static int checks = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("check failed: " + what);
        checks++;
    }
    static void print(ShopService s, String user) {
        Bill b = s.price(user);
        System.out.printf("  %-22s %10s%n", "Subtotal", Money.fmt(b.subtotal));
        for (int i = 0; i < b.discountCodes.size(); i++)
            System.out.printf("  %-22s %10s%n", "- " + b.discountCodes.get(i), Money.fmt(b.discountAmounts.get(i)));
        for (String n : b.notes) System.out.println("  note: " + n);
        System.out.printf("  %-22s %10s%n", "Total", Money.fmt(b.total()));
    }
    static void tryApply(ShopService s, String user, String code) {
        try { s.applyCoupon(user, code); System.out.println("apply " + code + " -> ok"); }
        catch (ShopException e) { System.out.println("apply " + code + " -> rejected: " + e.getMessage()); }
    }

    public static void main(String[] args) throws Exception {
        ShopService shop = new ShopService(LocalDate.of(2026, 11, 23));
        shop.addProduct(new Product("P1", "Laptop", Category.ELECTRONICS, Money.of("50000")), 2);
        shop.addProduct(new Product("P2", "Headphones", Category.ELECTRONICS, Money.of("2000")), 5);
        shop.addProduct(new Product("P3", "T-shirt", Category.FASHION, Money.of("799")), 10);

        shop.addCoupon(new Coupon("FASHION20", Rule.hasCategory(Category.FASHION),
                new CategoryPercentOff(Category.FASHION, 2000), 10, true, 99));
        shop.addCoupon(new Coupon("SAVE10", Rule.validUntil(LocalDate.of(2026, 11, 30)),
                new PercentOff(1000, Money.of("500")), 20, true, 99));
        shop.addCoupon(new Coupon("FLAT200", Rule.minSubtotal(Money.of("1000")),
                new FlatOff(Money.of("200")), 30, false, 99));
        shop.addCoupon(new Coupon("NEWUSER", Rule.firstOrderOnly().and(Rule.validUntil(LocalDate.of(2026, 12, 31))),
                new FlatOff(Money.of("100")), 30, true, 1));

        System.out.println("== u1: Laptop x1, T-shirt x2");
        shop.addItem("u1", "P1", 1);
        shop.addItem("u1", "P3", 2);
        tryApply(shop, "u1", "FASHION20");
        tryApply(shop, "u1", "SAVE10");
        tryApply(shop, "u1", "FLAT200");
        print(shop, "u1");
        Bill b1 = shop.price("u1");
        check(b1.subtotal == Money.of("51598"), "subtotal");
        check(b1.discountAmounts.get(0) == Money.of("319.60"), "20% of fashion only");
        check(b1.discountAmounts.get(1) == Money.of("500"), "10% capped at 500");
        Order o1 = shop.checkout("u1");
        System.out.println("checkout u1 -> " + o1.id + " paid " + Money.fmt(o1.bill.total()));
        check(o1.bill.total() == Money.of("50778.40"), "u1 total");
        check(shop.stockOf("P1") == 1 && shop.stockOf("P3") == 8, "stock decremented");

        System.out.println("== u2: small cart, then the cart changes after a coupon");
        shop.addItem("u2", "P3", 1);
        tryApply(shop, "u2", "FLAT200");
        tryApply(shop, "u2", "FASHION20");
        tryApply(shop, "u2", "NEWUSER");
        shop.addItem("u2", "P2", 1);
        shop.removeItem("u2", "P3");
        print(shop, "u2");
        Order o2 = shop.checkout("u2");
        System.out.println("checkout u2 -> " + o2.id + " paid " + Money.fmt(o2.bill.total()));
        check(o2.bill.total() == Money.of("1900"), "FASHION20 dropped, NEWUSER kept");
        shop.addItem("u2", "P3", 1);
        tryApply(shop, "u2", "NEWUSER");
        check(shop.price("u2").discountCodes.isEmpty(), "NEWUSER once only");

        System.out.println("== bad input");
        try { shop.addItem("u5", "P2", 9); } catch (ShopException e) { System.out.println("add 9 Headphones -> " + e.getMessage()); }
        try { shop.checkout("u6"); } catch (ShopException e) { System.out.println("checkout empty cart -> " + e.getMessage()); }
        shop.setToday(LocalDate.of(2026, 12, 1));
        shop.addItem("u5", "P2", 1);
        tryApply(shop, "u5", "SAVE10");
        check(shop.price("u5").total() == Money.of("2000"), "expired coupon not applied");

        System.out.println("== race: u3 and u4 both buy the last laptop");
        shop.addItem("u3", "P1", 1);
        shop.addItem("u4", "P1", 1);
        ExecutorService pool = Executors.newFixedThreadPool(2);
        CountDownLatch start = new CountDownLatch(1);
        List<Future<Boolean>> results = new ArrayList<>();
        for (String u : new String[]{"u3", "u4"})
            results.add(pool.submit(() -> {
                start.await();
                try { shop.checkout(u); return true; } catch (ShopException e) { return false; }
            }));
        start.countDown();
        int ok = 0;
        for (Future<Boolean> f : results) if (f.get()) ok++;
        pool.shutdown();

"""Shopping cart with coupons: Specification rules, Strategy discounts, Decorator pricing."""
from __future__ import annotations

import threading
from dataclasses import dataclass, field
from datetime import date
from decimal import Decimal
from enum import Enum
from typing import Callable, Dict, List, Optional


def money(rupees: str) -> int:
    """Rupees as text -> integer paise. Never float."""
    return int(Decimal(rupees) * 100)


def fmt(paise: int) -> str:
    return f"{paise // 100}.{paise % 100:02d}"


class Category(Enum):
    ELECTRONICS = "ELECTRONICS"
    FASHION = "FASHION"
    GROCERY = "GROCERY"


class ShopError(Exception):
    pass


@dataclass(frozen=True)
class Product:
    id: str
    name: str
    category: Category
    price: int  # paise


class Inventory:
    """All-or-nothing reservation under one lock."""

    def __init__(self) -> None:
        self._stock: Dict[str, int] = {}
        self._lock = threading.Lock()

    def add(self, product_id: str, qty: int) -> None:
        with self._lock:
            self._stock[product_id] = self._stock.get(product_id, 0) + qty

    def available(self, product_id: str) -> int:
        with self._lock:
            return self._stock.get(product_id, 0)

    def reserve_all(self, want: Dict[Product, int]) -> None:
        with self._lock:
            for p, q in want.items():  # check everything first
                if self._stock.get(p.id, 0) < q:
                    raise ShopError(f"out of stock: {p.name}")
            for p, q in want.items():  # then apply
                self._stock[p.id] -= q


@dataclass(frozen=True)
class CartLine:
    product: Product
    qty: int

    def total(self) -> int:
        return self.product.price * self.qty


class Cart:
    def __init__(self, user_id: str) -> None:
        self.user_id = user_id
        self.lock = threading.RLock()
        self._lines: Dict[str, CartLine] = {}  # insertion ordered
        self._coupons: List[str] = []

    def set_qty(self, p: Product, qty: int) -> None:
        if qty < 0:
            raise ShopError("quantity cannot be negative")
        with self.lock:
            if qty == 0:
                self._lines.pop(p.id, None)
            else:
                self._lines[p.id] = CartLine(p, qty)

    def qty_of(self, product_id: str) -> int:
        line = self._lines.get(product_id)
        return line.qty if line else 0

    def lines(self) -> List[CartLine]:
        with self.lock:
            return list(self._lines.values())

    def coupons(self) -> List[str]:
        with self.lock:
            return list(self._coupons)

    def add_coupon(self, code: str) -> None:
        with self.lock:
            self._coupons.append(code)

    def remove_coupon(self, code: str) -> None:
        with self.lock:
            if code in self._coupons:
                self._coupons.remove(code)

    def clear(self) -> None:
        with self.lock:
            self._lines.clear()
            self._coupons.clear()


@dataclass(frozen=True)
class PricingContext:
    user_id: str
    today: date
    past_orders: int
    coupon_uses: Dict[str, int]
    lines: List[CartLine]

    @property
    def subtotal(self) -> int:
        return sum(l.total() for l in self.lines)

    def subtotal_of(self, c: Category) -> int:
        return sum(l.total() for l in self.lines if l.product.category == c)


# ---------- Specification: a rule returns None when satisfied, else the reason ----------
class Rule:
    def __init__(self, check: Callable[[PricingContext], Optional[str]]) -> None:
        self._check = check

    def violation(self, ctx: PricingContext) -> Optional[str]:
        return self._check(ctx)

    def is_satisfied_by(self, ctx: PricingContext) -> bool:
        return self.violation(ctx) is None

    def and_(self, other: "Rule") -> "Rule":
        return Rule(lambda ctx: self.violation(ctx) or other.violation(ctx))

    @staticmethod
    def always() -> "Rule":
        return Rule(lambda ctx: None)

    @staticmethod
    def min_subtotal(paise: int) -> "Rule":
        return Rule(lambda ctx: None if ctx.subtotal >= paise else f"needs a cart value of at least {fmt(paise)}")

    @staticmethod
    def has_category(c: Category) -> "Rule":
        return Rule(lambda ctx: None if ctx.subtotal_of(c) > 0 else f"needs a {c.value} item in the cart")

    @staticmethod
    def first_order_only() -> "Rule":
        return Rule(lambda ctx: None if ctx.past_orders == 0 else "is for the first order only")

    @staticmethod
    def valid_until(last: date) -> "Rule":
        return Rule(lambda ctx: None if ctx.today <= last else f"expired on {last.isoformat()}")


# ---------- Strategy: how much a coupon takes off ----------
class DiscountStrategy:
    def discount(self, ctx: PricingContext, running: int) -> int:
        raise NotImplementedError


class FlatOff(DiscountStrategy):
    def __init__(self, paise: int) -> None:
        self.paise = paise

    def discount(self, ctx: PricingContext, running: int) -> int:
        return min(self.paise, running)


class PercentOff(DiscountStrategy):
    def __init__(self, basis_points: int, cap: int = 0) -> None:
        self.bp, self.cap = basis_points, cap

    def discount(self, ctx: PricingContext, running: int) -> int:
        d = running * self.bp // 10000
        return min(d, self.cap) if self.cap else d


class CategoryPercentOff(DiscountStrategy):
    def __init__(self, category: Category, basis_points: int) -> None:
        self.category, self.bp = category, basis_points

    def discount(self, ctx: PricingContext, running: int) -> int:
        return min(ctx.subtotal_of(self.category) * self.bp // 10000, running)


@dataclass(frozen=True)
class Coupon:
    code: str
    rule: Rule
    strategy: DiscountStrategy
    priority: int  # lower runs first
    stackable: bool
    max_uses_per_user: int

    def why_not(self, ctx: PricingContext) -> Optional[str]:
        if ctx.coupon_uses.get(self.code, 0) >= self.max_uses_per_user:
            return "already used"
        return self.rule.violation(ctx)


@dataclass(frozen=True)
class Bill:
    subtotal: int
    discounts: tuple = ()  # ((code, paise), ...)
    notes: tuple = ()

    def total(self) -> int:
        return self.subtotal - sum(a for _, a in self.discounts)

    def with_discount(self, code: str, amount: int) -> "Bill":
        return Bill(self.subtotal, self.discounts + ((code, amount),), self.notes)

    def with_note(self, note: str) -> "Bill":
        return Bill(self.subtotal, self.discounts, self.notes + (note,))


# ---------- Decorator: base price wrapped by one layer per coupon ----------
class PriceCalculator:
    def calculate(self, ctx: PricingContext) -> Bill:
        raise NotImplementedError


class BasePrice(PriceCalculator):
    def calculate(self, ctx: PricingContext) -> Bill:
        return Bill(ctx.subtotal)


class CouponLayer(PriceCalculator):
    def __init__(self, inner: PriceCalculator, coupon: Coupon) -> None:
        self.inner, self.coupon = inner, coupon

    def calculate(self, ctx: PricingContext) -> Bill:
        bill = self.inner.calculate(ctx)
        why = self.coupon.why_not(ctx)
        if why:
            return bill.with_note(f"{self.coupon.code} not applied: {why}")
        d = min(self.coupon.strategy.discount(ctx, bill.total()), bill.total())
        if d <= 0:
            return bill.with_note(f"{self.coupon.code} gives nothing on this cart")
        return bill.with_discount(self.coupon.code, d)


@dataclass(frozen=True)
class Order:
    id: str
    user_id: str
    lines: tuple
    bill: Bill


class ShopService:
    """Facade. Lock order: cart lock, then inventory lock."""

    def __init__(self, today: date) -> None:
        self.today = today
        self._catalog: Dict[str, Product] = {}
        self._coupons: Dict[str, Coupon] = {}
        self._carts: Dict[str, Cart] = {}
        self._uses: Dict[str, Dict[str, int]] = {}
        self._orders: Dict[str, int] = {}
        self._inventory = Inventory()
        self._meta = threading.Lock()
        self._seq = 0

    def add_product(self, p: Product, stock: int) -> None:
        self._catalog[p.id] = p
        self._inventory.add(p.id, stock)

    def add_coupon(self, c: Coupon) -> None:
        self._coupons[c.code] = c

    def stock_of(self, product_id: str) -> int:
        return self._inventory.available(product_id)

    def _cart(self, user_id: str) -> Cart:
        with self._meta:
            return self._carts.setdefault(user_id, Cart(user_id))

    def _product(self, pid: str) -> Product:
        if pid not in self._catalog:
            raise ShopError(f"unknown product {pid}")
        return self._catalog[pid]

    def add_item(self, user_id: str, product_id: str, qty: int) -> None:
        if qty <= 0:
            raise ShopError("quantity must be positive")
        p = self._product(product_id)
        c = self._cart(user_id)
        with c.lock:
            want = c.qty_of(product_id) + qty
            left = self._inventory.available(product_id)
            if want > left:  # soft check; checkout re-checks
                raise ShopError(f"only {left} {p.name} left")
            c.set_qty(p, want)

    def update_qty(self, user_id: str, product_id: str, qty: int) -> None:
        self._cart(user_id).set_qty(self._product(product_id), qty)

    def remove_item(self, user_id: str, product_id: str) -> None:
        self.update_qty(user_id, product_id, 0)

    def _context(self, user_id: str, c: Cart) -> PricingContext:
        with self._meta:
            uses = dict(self._uses.get(user_id, {}))
            past = self._orders.get(user_id, 0)
        return PricingContext(user_id, self.today, past, uses, c.lines())

    def apply_coupon(self, user_id: str, code: str) -> None:
        coupon = self._coupons.get(code)
        if coupon is None:
            raise ShopError(f"unknown coupon {code}")
        c = self._cart(user_id)
        with c.lock:
            applied = c.coupons()
            if code in applied:
                raise ShopError(f"{code} is already applied")
            exclusive_there = any(not self._coupons[x].stackable for x in applied)
            if applied and (not coupon.stackable or exclusive_there):
                raise ShopError(f"{code} cannot be combined with {','.join(applied)}")
            why = coupon.why_not(self._context(user_id, c))
            if why:
                raise ShopError(f"{code} {why}")
            c.add_coupon(code)

    def remove_coupon(self, user_id: str, code: str) -> None:
        self._cart(user_id).remove_coupon(code)

    def price(self, user_id: str) -> Bill:
        c = self._cart(user_id)
        with c.lock:
            return self._price_locked(user_id, c)

    def _price_locked(self, user_id: str, c: Cart) -> Bill:
        layers = sorted((self._coupons[x] for x in c.coupons()), key=lambda k: (k.priority, k.code))
        calc: PriceCalculator = BasePrice()
        for k in layers:
            calc = CouponLayer(calc, k)
        return calc.calculate(self._context(user_id, c))

    def checkout(self, user_id: str) -> Order:
        c = self._cart(user_id)
        with c.lock:
            lines = c.lines()
            if not lines:
                raise ShopError("cart is empty")
            bill = self._price_locked(user_id, c)
            self._inventory.reserve_all({l.product: l.qty for l in lines})  # raises before any change
            with self._meta:
                mine = self._uses.setdefault(user_id, {})
                for code, _ in bill.discounts:
                    mine[code] = mine.get(code, 0) + 1
                self._orders[user_id] = self._orders.get(user_id, 0) + 1
                self._seq += 1
                order_id = f"O{self._seq}"
            c.clear()
            return Order(order_id, user_id, tuple(lines), bill)


def build_shop() -> ShopService:
    shop = ShopService(date(2026, 11, 23))
    shop.add_product(Product("P1", "Laptop", Category.ELECTRONICS, money("50000")), 2)
    shop.add_product(Product("P2", "Headphones", Category.ELECTRONICS, money("2000")), 5)
    shop.add_product(Product("P3", "T-shirt", Category.FASHION, money("799")), 10)
    shop.add_coupon(Coupon("FASHION20", Rule.has_category(Category.FASHION),
                           CategoryPercentOff(Category.FASHION, 2000), 10, True, 99))
    shop.add_coupon(Coupon("SAVE10", Rule.valid_until(date(2026, 11, 30)),
                           PercentOff(1000, money("500")), 20, True, 99))
    shop.add_coupon(Coupon("FLAT200", Rule.min_subtotal(money("1000")), FlatOff(money("200")), 30, False, 99))
    shop.add_coupon(Coupon("NEWUSER", Rule.first_order_only().and_(Rule.valid_until(date(2026, 12, 31))),
                           FlatOff(money("100")), 30, True, 1))
    return shop


def show(shop: ShopService, user: str) -> None:
    b = shop.price(user)
    print(f"  {'Subtotal':<22} {fmt(b.subtotal):>10}")
    for code, amt in b.discounts:
        print(f"  {'- ' + code:<22} {fmt(amt):>10}")
    for n in b.notes:
        print(f"  note: {n}")
    print(f"  {'Total':<22} {fmt(b.total()):>10}")


def try_apply(shop: ShopService, user: str, code: str) -> None:
    try:
        shop.apply_coupon(user, code)
        print(f"apply {code} -> ok")
    except ShopError as e:
        print(f"apply {code} -> rejected: {e}")


def main() -> None:
    shop = build_shop()
    print("== u1: Laptop x1, T-shirt x2")
    shop.add_item("u1", "P1", 1)
    shop.add_item("u1", "P3", 2)
    for code in ("FASHION20", "SAVE10", "FLAT200"):
        try_apply(shop, "u1", code)
    show(shop, "u1")
    o1 = shop.checkout("u1")
    print(f"checkout u1 -> {o1.id} paid {fmt(o1.bill.total())}")
    assert o1.bill.total() == money("50778.40")

    print("== u2: small cart, then the cart changes after a coupon")
    shop.add_item("u2", "P3", 1)
    for code in ("FLAT200", "FASHION20", "NEWUSER"):
        try_apply(shop, "u2", code)
    shop.add_item("u2", "P2", 1)
    shop.remove_item("u2", "P3")
    show(shop, "u2")
    o2 = shop.checkout("u2")

    try_apply(shop, "u2", "NEWUSER")

    print("== race: u3 and u4 both buy the last laptop")
    shop.add_item("u3", "P1", 1)
    shop.add_item("u4", "P1", 1)

            pass

    for t in ts:
        t.start()
    for t in ts:
        t.join()
    print(f"succeeded: {len(wins)}, out of stock: {2 - len(wins)}, laptops left: {shop.stock_of('P1')}")
    assert len(wins) == 1 and shop.stock_of("P1") == 0
    print("demo OK")

class ShoppingCartTest(unittest.TestCase):
    def setUp(self):
        self.shop = build_shop()

    def test_stacked_coupons_run_in_priority_order(self):
        self.shop.add_item("u1", "P1", 1)
        self.shop.add_item("u1", "P3", 2)
        self.shop.apply_coupon("u1", "SAVE10")      # applied first, but runs second
        self.shop.apply_coupon("u1", "FASHION20")
        bill = self.shop.price("u1")
        self.assertEqual([c for c, _ in bill.discounts], ["FASHION20", "SAVE10"])
        self.assertEqual(bill.discounts[0][1], money("319.60"))
        self.assertEqual(bill.discounts[1][1], money("500"))   # 10% would be 5127.84, capped
        self.assertEqual(bill.total(), money("50778.40"))

    def test_exclusive_coupon_cannot_stack(self):
        self.shop.add_item("u1", "P2", 1)
        self.shop.apply_coupon("u1", "FLAT200")
        with self.assertRaisesRegex(ShopError, "cannot be combined"):
            self.shop.apply_coupon("u1", "SAVE10")

    def test_rule_rejects_with_reason(self):
        self.shop.add_item("u1", "P3", 1)
        with self.assertRaisesRegex(ShopError, "at least 1000.00"):
            self.shop.apply_coupon("u1", "FLAT200")
        self.shop.today = date(2026, 12, 1)
        with self.assertRaisesRegex(ShopError, "expired on 2026-11-30"):
            self.shop.apply_coupon("u1", "SAVE10")

    def test_coupon_dropped_when_cart_changes(self):
        self.shop.add_item("u1", "P3", 1)
        self.shop.apply_coupon("u1", "FASHION20")
        self.shop.add_item("u1", "P2", 1)
        self.shop.remove_item("u1", "P3")
        bill = self.shop.price("u1")
        self.assertEqual(bill.discounts, ())
        self.assertIn("FASHION20 not applied", bill.notes[0])

    def test_discount_never_below_zero(self):
        self.shop.add_item("u1", "P3", 1)              # 799
        self.shop.apply_coupon("u1", "FASHION20")       # -159.80
        self.shop.apply_coupon("u1", "NEWUSER")         # -100
        self.assertEqual(self.shop.price("u1").total(), money("539.20"))

    def test_first_order_coupon_once(self):
        self.shop.add_item("u1", "P2", 1)
        self.shop.apply_coupon("u1", "NEWUSER")
        self.shop.checkout("u1")
        self.shop.add_item("u1", "P2", 1)
        with self.assertRaisesRegex(ShopError, "already used"):
            self.shop.apply_coupon("u1", "NEWUSER")

    def test_failed_checkout_changes_nothing(self):
        self.shop.add_item("u1", "P1", 2)
        self.shop.add_item("u2", "P1", 1)
        self.shop.checkout("u1")                       # takes both laptops
        with self.assertRaisesRegex(ShopError, "out of stock"):
            self.shop.checkout("u2")
        self.assertEqual(self.shop.stock_of("P1"), 0)
        self.assertEqual(len(self.shop.price("u2").discounts), 0)

    def test_concurrent_checkouts_never_oversell(self):
        users = [f"x{i}" for i in range(8)]
        for u in users:
            self.shop.add_item(u, "P2", 1)             # 5 in stock, 8 buyers
        wins = []
        gate = threading.Barrier(len(users))

        def buy(u):
            gate.wait()
            try:
                self.shop.checkout(u)
                wins.append(u)
            except ShopError:
                pass

        ts = [threading.Thread(target=buy, args=(u,)) for u in users]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
        self.assertEqual(len(wins), 5)
        self.assertEqual(self.shop.stock_of("P2"), 0)
