-- schema for the design-shape round (Swiggy asked for it)
CREATE TABLE products  (id VARCHAR PRIMARY KEY, name VARCHAR, category VARCHAR, price_paise BIGINT, stock INT CHECK (stock >= 0));
CREATE TABLE carts     (user_id VARCHAR PRIMARY KEY, updated_at TIMESTAMP);
CREATE TABLE cart_items(user_id VARCHAR, product_id VARCHAR, qty INT, PRIMARY KEY (user_id, product_id));
CREATE TABLE coupons   (code VARCHAR PRIMARY KEY, type VARCHAR, value BIGINT, cap_paise BIGINT,
                        priority INT, stackable BOOLEAN, max_per_user INT, remaining INT NULL,
                        rules JSON, valid_until DATE);
CREATE TABLE cart_coupons(user_id VARCHAR, code VARCHAR, PRIMARY KEY (user_id, code));
CREATE TABLE orders    (id VARCHAR PRIMARY KEY, user_id VARCHAR, total_paise BIGINT,
                        idempotency_key VARCHAR UNIQUE, created_at TIMESTAMP);
CREATE TABLE redemptions(order_id VARCHAR, code VARCHAR, user_id VARCHAR, amount_paise BIGINT,
                        PRIMARY KEY (order_id, code));

-- checkout, in one transaction: conditional decrement never oversells
UPDATE products SET stock = stock - :qty WHERE id = :pid AND stock >= :qty;   -- 0 rows = out of stock, roll back

For speed, cache the active coupons per category in memory or Redis (they change rarely), and precompute "coupons this cart could use" asynchronously for the UI, which is what the Swiggy candidate offered as optimisations.
