Trap 2: Bird and Penguin

The second classic: Bird has fly(), and Penguin extends Bird overrides it to throw UnsupportedOperationException. A migrate(List<Bird>) function crashes the moment a penguin is in the flock (see the output above). Leaving fly() empty is no better: the penguin silently stays behind, and the caller thinks it moved.

The cause is a base type that promised too much. Not all birds fly, so Bird must not promise flight. Move the promise to where it is true:

import java.util.List;
import java.util.function.Supplier;

// AFTER: model behaviour, not taxonomy. Every subtype keeps every promise of its supertype.
public class LspAfter {
    // ---- 1. Shapes: no shared mutators, so nothing for Square to break ----
    interface Shape { int area(); }

    record Rectangle(int w, int h) implements Shape {
        Rectangle { if (w <= 0 || h <= 0) throw new IllegalArgumentException("sides must be > 0"); }
        public int area() { return w * h; }
        Rectangle withWidth(int nw) { return new Rectangle(nw, h); }    // "change" = new value
    }
    record Square(int side) implements Shape {
        Square { if (side <= 0) throw new IllegalArgumentException("side must be > 0"); }
        public int area() { return side * side; }
    }

    // ---- 2. Birds: only flyers promise fly() ----
    interface Bird { String name(); default String eat() { return name() + " eats"; } }
    interface FlyingBird extends Bird { String fly(); }
    interface SwimmingBird extends Bird { String swim(); }

    record Sparrow() implements FlyingBird {
        public String name() { return "Sparrow"; }
        public String fly()  { return "Sparrow flies south"; }
    }
    record Penguin() implements SwimmingBird {
        public String name() { return "Penguin"; }
        public String swim() { return "Penguin swims"; }
    }

    static void migrate(List<? extends FlyingBird> flock) {       // the type system now refuses penguins
        for (FlyingBird b : flock) System.out.println(b.fly());
    }

    // ---- 3. A contract test: run the SAME checks against every implementation ----
    static void shapeContract(String label, Supplier<Shape> make) {
        Shape s = make.get();
        if (s.area() <= 0) throw new AssertionError(label + ": area must be positive");
        if (s.area() != make.get().area()) throw new AssertionError(label + ": area must be deterministic");
        System.out.println("contract ok: " + label + " area=" + s.area());
    }

    public static void main(String[] args) {
        Rectangle r = new Rectangle(5, 4);
        System.out.println("rect 5x4 area=" + r.area() + ", widened=" + r.withWidth(6).area());
        List<Shape> shapes = List.of(r, new Square(4));
        int total = 0;
        for (Shape s : shapes) total += s.area();                     // any Shape works here
        System.out.println("total area=" + total);

        migrate(List.of(new Sparrow()));
        // migrate(List.of(new Penguin()));   // does not compile: the bug is caught by javac, not by users
        System.out.println(new Penguin().swim() + "; " + new Penguin().eat());

        shapeContract("Rectangle", () -> new Rectangle(3, 2));
        shapeContract("Square", () -> new Square(3));
    }
}
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Iterable


class Shape(ABC):
    @abstractmethod
    def area(self) -> int: ...


@dataclass(frozen=True)          # immutable: no setters for a subtype to break
class Rectangle(Shape):
    w: int
    h: int

    def __post_init__(self):
        if self.w <= 0 or self.h <= 0:
            raise ValueError("sides must be > 0")

    def area(self) -> int:
        return self.w * self.h

    def with_width(self, nw: int) -> "Rectangle":
        return Rectangle(nw, self.h)


@dataclass(frozen=True)
class Square(Shape):
    side: int

    def area(self) -> int:
        return self.side * self.side


class Bird(ABC):
    @abstractmethod
    def name(self) -> str: ...


class FlyingBird(Bird):
    @abstractmethod
    def fly(self) -> str: ...


class Sparrow(FlyingBird):
    def name(self) -> str: return "Sparrow"
    def fly(self) -> str: return "Sparrow flies south"


class Penguin(Bird):
    def name(self) -> str: return "Penguin"
    def swim(self) -> str: return "Penguin swims"


def migrate(flock: Iterable[FlyingBird]) -> None:
    for b in flock:
        print(b.fly())


def shape_contract(make) -> None:
    """Same test for every Shape subtype: the practical way to enforce LSP."""
    s = make()
    assert s.area() > 0, f"{type(s).__name__}: area must be positive"
    assert s.area() == make().area(), "area must be deterministic"
    print(f"contract ok: {type(s).__name__} area={s.area()}")


if __name__ == "__main__":
    print("total area =", sum(s.area() for s in [Rectangle(5, 4), Square(4)]))
    migrate([Sparrow()])          # a type checker (mypy) rejects migrate([Penguin()])
    print(Penguin().swim())
    for make in (lambda: Rectangle(3, 2), lambda: Square(3)):
        shape_contract(make)

Look at the commented-out line in main. Passing a penguin to migrate no longer compiles (javac reports "incompatible types"). The bug moved from "crash in production" to "error at compile time". That is the real payoff of LSP-friendly design: the type system starts protecting you. Splitting Bird into small role interfaces is also the single-responsibility idea applied to interfaces, which the Interface Segregation Principle (the I in SOLID) makes explicit.

A real interview variant (reported for a Cars24 SDE role on LeetCode Discuss, post): you are shown code that violates LSP and asked to "introduce an interface so that it follows LSP". The answer is exactly this move: find the method that some subtypes cannot honour, and pull it out into a narrower interface that only the capable types implement.
