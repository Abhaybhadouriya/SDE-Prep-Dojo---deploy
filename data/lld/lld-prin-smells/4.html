Ten key refactorings (1-5)

1. Extract Method - cures Long Method, Duplicate Code, Comments

Smell: a method that computes a subtotal, a discount, tax and formats a bill. Its parts are marked with blank lines or comments. Recipe: take each part, move it into a method named for what it does, pass in what it reads, return what it produces. Name magic numbers while you are there.

static String print(String customer, List<Line> lines, boolean member) {
    StringBuilder sb = new StringBuilder("Bill for " + customer + "\n");
    long subtotal = 0;
    for (Line l : lines) {
        long amt = l.qty() * l.unitPaise();
        subtotal += amt;
        sb.append(String.format("  %-8s x%d %8d%n", l.item(), l.qty(), amt));
    }
    long discount = 0;
    if (member && subtotal > 50000) discount = subtotal * 10 / 100;   // what are 50000 and 10?
    long tax = (subtotal - discount) * 18 / 100;
    sb.append(String.format("  total %d%n", subtotal - discount + tax));
    return sb.toString();
}
static final long MEMBER_THRESHOLD = 50000;
static final int MEMBER_PCT = 10, GST_PCT = 18;

static String print(String customer, List<Line> lines, boolean member) {
    long subtotal = subtotal(lines);
    long discount = memberDiscount(subtotal, member);
    return header(customer) + lineItems(lines) + footer(subtotal - discount + tax(subtotal - discount));
}
static long subtotal(List<Line> lines) { return lines.stream().mapToLong(l -> l.qty() * l.unitPaise()).sum(); }
static long memberDiscount(long subtotal, boolean member) {
    return member && subtotal > MEMBER_THRESHOLD ? subtotal * MEMBER_PCT / 100 : 0;
}
static long tax(long taxable) { return taxable * GST_PCT / 100; }
// header(), lineItems(), footer() are three short formatting methods

The top method now reads like its own summary. Each small method can be tested alone, and tax() can be reused by the refund code instead of copy-pasted.

2. Extract Class - cures Large Class, Data Clumps, Divergent Change

Smell: guestName, guestEmail, guestPhone always travel together, and Booking contains rules about phone numbers that have nothing to do with rooms. Recipe: create a new class, move the related fields and the methods that use them, and have the old class hold one instance.

final class Booking {
    String guestName, guestEmail, guestPhone;   // a data clump
    int room;
    String contactLine()     { return guestName + " <" + guestEmail + "> " + guestPhone; }
    boolean canNotifyBySms() { return guestPhone != null && guestPhone.startsWith("+91"); }
}
record Contact(String name, String email, String phone) {
    String line()          { return name + " <" + email + "> " + phone; }
    boolean smsReachable() { return phone != null && phone.startsWith("+91"); }
}
record Booking(Contact guest, int room) {}   // Booking now changes only for room reasons

3. Introduce Parameter Object - cures Long Parameter List, Data Clumps

Smell: search(city, fromDay, toDay, minPaise, maxPaise). Two ints and two longs side by side are easy to swap, and every caller repeats the same validation. Recipe: group parameters that belong together into small classes, and move the logic that uses them into those classes.

static List<Room> search(String city, int fromDay, int toDay, long minPaise, long maxPaise) {
    return ROOMS.stream().filter(r -> r.city().equals(city)
            && r.day() >= fromDay && r.day() <= toDay
            && r.pricePaise() >= minPaise && r.pricePaise() <= maxPaise).toList();
}
search("Pune", 1, 10, 0, 280000);    // which number is which?
record DayRange(int from, int to) {
    DayRange { if (from > to) throw new IllegalArgumentException("from > to"); }   // validated once
    boolean contains(int d) { return d >= from && d <= to; }
}
record PriceRange(long min, long max) { boolean contains(long p) { return p >= min && p <= max; } }
record RoomQuery(String city, DayRange days, PriceRange price) {
    boolean matches(Room r) { return r.city().equals(city) && days.contains(r.day()) && price.contains(r.pricePaise()); }
}
static List<Room> search(RoomQuery q) { return ROOMS.stream().filter(q::matches).toList(); }

search(new RoomQuery("Pune", new DayRange(1, 10), new PriceRange(0, 280000)));

A nice side effect: the new classes attract behaviour (contains, overlaps, validation). When a query object grows many optional fields, give it a Builder.

4. Replace Data Value with Object - cures Primitive Obsession

Smell: money as double. Split Rs 100 between three friends and charge each the rounded Rs 33.33: you collect Rs 99.99, and one paisa disappears. Multiply that by a million transactions. Recipe: make a small immutable class for the value, store it in the smallest exact unit, and put the rules inside it.

static String split(double amountRupees, int people) {
    double each = Math.round(amountRupees / people * 100) / 100.0;           // 33.33
    return String.format("%.2f each, %.2f collected", each, each * people);  // 99.99 collected
}
record Money(long paise) {
    Money { if (paise < 0) throw new IllegalArgumentException("negative money"); }
    static Money rupees(long r) { return new Money(r * 100); }
    Money plus(Money o) { return new Money(paise + o.paise); }
    List<Money> splitEvenly(int n) {          // remainder goes to the first shares: nothing lost
        List<Money> out = new ArrayList<>();
        for (int i = 0; i < n; i++) out.add(new Money(paise / n + (i < paise % n ? 1 : 0)));
        return out;
    }
}
// Money.rupees(100).splitEvenly(3) -> [33.34, 33.33, 33.33], sum 100.00
from dataclasses import dataclass

@dataclass(frozen=True)
class Money:
    paise: int
    def __post_init__(self):
        if self.paise < 0:
            raise ValueError("negative money")
    @staticmethod
    def rupees(r: int) -> "Money":
        return Money(r * 100)
    def __add__(self, other: "Money") -> "Money":
        return Money(self.paise + other.paise)
    def split_evenly(self, n: int) -> list["Money"]:
        q, r = divmod(self.paise, n)
        return [Money(q + (1 if i < r else 0)) for i in range(n)]

assert sum(Money.rupees(100).split_evenly(3), Money(0)) == Money(10000)

The same refactoring applies to Email, PhoneNumber, SeatId, DateRange. In an interview, a Money class in your Splitwise or wallet design is one of the cheapest ways to show care. Splitwise is the classic place it matters.

5. Replace Conditional with Polymorphism - cures Switch Statements

Smell: a switch (channel) in send(), another in cost(), another in retryPolicy(). Adding WhatsApp means finding all of them. Recipe: create an interface with the varying method, one class per case, move each branch into its class, and replace the switch with a call (choose the object once, via a map or a factory).

static String notify(String channel, String to, String msg) {
    switch (channel) {
        case "EMAIL": return "mail " + to + ": " + msg;
        case "SMS":   return "sms " + to + ": " + (msg.length() > 20 ? msg.substring(0, 20) : msg);
        case "PUSH":  return "push " + to + ": " + msg.toUpperCase();
        default: throw new IllegalArgumentException(channel);
    }
}
interface Channel { String send(String to, String msg); }
final class Email implements Channel { public String send(String to, String m) { return "mail " + to + ": " + m; } }
final class Sms implements Channel {
    public String send(String to, String m) { return "sms " + to + ": " + (m.length() > 20 ? m.substring(0, 20) : m); }
}
final class Push implements Channel { public String send(String to, String m) { return "push " + to + ": " + m.toUpperCase(); } }

static final Map<String, Channel> CHANNELS = Map.of("EMAIL", new Email(), "SMS", new Sms(), "PUSH", new Push());
// caller: CHANNELS.get(name).send(to, msg)   - the one remaining "switch" is the map lookup
from typing import Protocol

class Channel(Protocol):
    def send(self, to: str, msg: str) -> str: ...

class Email:
    def send(self, to, msg): return f"mail {to}: {msg}"
class Sms:
    def send(self, to, msg): return f"sms {to}: {msg[:20]}"
class Push:
    def send(self, to, msg): return f"push {to}: {msg.upper()}"

CHANNELS: dict[str, Channel] = {"EMAIL": Email(), "SMS": Sms(), "PUSH": Push()}
print(CHANNELS["SMS"].send("u1", "Your order has shipped today"))

Not every switch is a smell. One switch, in one factory, that turns input into an object is fine - somewhere the decision must be made once. The smell is the same decision repeated across the codebase.
