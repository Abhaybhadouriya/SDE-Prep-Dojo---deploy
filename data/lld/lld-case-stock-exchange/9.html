Java is the main version: exchange + broker + ticker in one file, with a demo that checks itself, including a 4-thread stress test that proves cash and shares are conserved and the book is never left crossed. Python mirrors it with a unittest suite. The C++14 tab is the matching engine alone (single-threaded, which is how a per-symbol engine runs anyway).

From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import java.util.concurrent.locks.*;

// ================= exchange side =================
enum Side { BUY, SELL }
enum OrderType { LIMIT, MARKET }
enum OrderStatus { OPEN, PARTIALLY_FILLED, FILLED, CANCELLED, REJECTED }

final class Order {
    final String id, userId, symbol;
    final Side side;
    final OrderType type;
    final long limitPrice;          // paise; ignored for MARKET
    final int qty;
    final long seq;                 // arrival order = time priority
    private volatile int filled;                     // written under the book lock, read anywhere
    private volatile OrderStatus status = OrderStatus.OPEN;

    Order(String id, String userId, String symbol, Side side, OrderType type, long limitPrice, int qty, long seq) {
        this.id = id; this.userId = userId; this.symbol = symbol; this.side = side;
        this.type = type; this.limitPrice = limitPrice; this.qty = qty; this.seq = seq;
    }
    int filled() { return filled; }
    int remaining() { return qty - filled; }
    OrderStatus status() { return status; }
    void fill(int q) {
        filled += q;
        status = remaining() == 0 ? OrderStatus.FILLED : OrderStatus.PARTIALLY_FILLED;
    }
    void cancel() { status = OrderStatus.CANCELLED; }
    void reject() { status = OrderStatus.REJECTED; }
    boolean isLive() { return status == OrderStatus.OPEN || status == OrderStatus.PARTIALLY_FILLED; }
    public String toString() {
        return id + "(" + side + " " + qty + (type == OrderType.MARKET ? " MKT" : " @" + Money.fmt(limitPrice))
                + ", filled " + filled + ", " + status + ")";
    }
}

final class Trade {
    final long id; final String symbol, buyOrderId, sellOrderId, buyer, seller;
    final long price; final int qty;
    Trade(long id, String symbol, Order buy, Order sell, long price, int qty) {
        this.id = id; this.symbol = symbol; this.buyOrderId = buy.id; this.sellOrderId = sell.id;
        this.buyer = buy.userId; this.seller = sell.userId; this.price = price; this.qty = qty;
    }
    public String toString() { return "T" + id + " " + symbol + " " + qty + " @ " + Money.fmt(price) + " (" + buyOrderId + " x " + sellOrderId + ")"; }
}

final class Money {
    static String fmt(long paise) { return String.format("%d.%02d", paise / 100, paise % 100); }
}

// Observer: anyone interested in trades (broker settlement, price ticker, alerts)
interface TradeListener { void onTrade(Trade t); }

// One book per symbol. Price-time priority: best price first, then earliest seq (FIFO queue per level).
final class OrderBook {
    final String symbol;
    private final TreeMap<Long, ArrayDeque<Order>> bids = new TreeMap<>(Comparator.reverseOrder()); // highest first
    private final TreeMap<Long, ArrayDeque<Order>> asks = new TreeMap<>();                          // lowest first
    private final Map<String, Order> resting = new HashMap<>();
    private long lastPrice;
    final ReentrantLock lock = new ReentrantLock(true);   // one writer per symbol

    OrderBook(String symbol, long openPrice) { this.symbol = symbol; this.lastPrice = openPrice; }
    long lastPrice() { return lastPrice; }

    /** Match the incoming order against the opposite side; rest the remainder if it is a LIMIT order. */
    List<Trade> match(Order in, AtomicLong tradeIds) {
        List<Trade> trades = new ArrayList<>();
        TreeMap<Long, ArrayDeque<Order>> opposite = in.side == Side.BUY ? asks : bids;
        while (in.remaining() > 0 && !opposite.isEmpty()) {
            Map.Entry<Long, ArrayDeque<Order>> best = opposite.firstEntry();
            long price = best.getKey();
            if (in.type == OrderType.LIMIT && !crosses(in, price)) break;
            ArrayDeque<Order> level = best.getValue();
            Order maker = level.peekFirst();                 // earliest order at the best price
            int q = Math.min(in.remaining(), maker.remaining());
            in.fill(q); maker.fill(q);
            lastPrice = price;                               // trade happens at the resting (maker) price
            Order buy = in.side == Side.BUY ? in : maker, sell = in.side == Side.BUY ? maker : in;
            trades.add(new Trade(tradeIds.incrementAndGet(), symbol, buy, sell, price, q));
            if (maker.remaining() == 0) { level.pollFirst(); resting.remove(maker.id); }
            if (level.isEmpty()) opposite.remove(price);
        }
        if (in.remaining() > 0) {
            if (in.type == OrderType.LIMIT) {
                (in.side == Side.BUY ? bids : asks).computeIfAbsent(in.limitPrice, p -> new ArrayDeque<>()).addLast(in);
                resting.put(in.id, in);
            } else {
                in.cancel();                                 // market remainder: immediate-or-cancel
            }
        }
        return trades;
    }
    private static boolean crosses(Order in, long price) {
        return in.side == Side.BUY ? in.limitPrice >= price : in.limitPrice <= price;
    }
    Order cancel(String orderId) {
        Order o = resting.remove(orderId);
        if (o == null) return null;
        TreeMap<Long, ArrayDeque<Order>> side = o.side == Side.BUY ? bids : asks;
        ArrayDeque<Order> level = side.get(o.limitPrice);
        level.remove(o);
        if (level.isEmpty()) side.remove(o.limitPrice);
        o.cancel();
        return o;
    }
    Long bestBid() { return bids.isEmpty() ? null : bids.firstKey(); }
    Long bestAsk() { return asks.isEmpty() ? null : asks.firstKey(); }
    String depth() {
        StringBuilder sb = new StringBuilder(symbol + " asks:");
        asks.descendingMap().forEach((p, q) -> sb.append(' ').append(Money.fmt(p)).append('x').append(sum(q)));
        sb.append(" | bids:");
        bids.forEach((p, q) -> sb.append(' ').append(Money.fmt(p)).append('x').append(sum(q)));
        return sb.toString();
    }
    private static int sum(ArrayDeque<Order> q) { int s = 0; for (Order o : q) s += o.remaining(); return s; }
}

final class Exchange {
    private final ConcurrentHashMap<String, OrderBook> books = new ConcurrentHashMap<>();
    private final List<TradeListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicLong seq = new AtomicLong(), tradeIds = new AtomicLong();

    void listStock(String symbol, long openPrice) {
        if (books.putIfAbsent(symbol, new OrderBook(symbol, openPrice)) != null)
            throw new IllegalArgumentException("already listed: " + symbol);
    }
    void subscribe(TradeListener l) { listeners.add(l); }
    long nextSeq() { return seq.incrementAndGet(); }
    OrderBook book(String symbol) {
        OrderBook b = books.get(symbol);
        if (b == null) throw new IllegalArgumentException("unknown symbol " + symbol);
        return b;
    }
    long lastPrice(String symbol) { return book(symbol).lastPrice(); }

    List<Trade> submit(Order o) {
        OrderBook b = book(o.symbol);
        b.lock.lock();
        try {
            List<Trade> trades = b.match(o, tradeIds);
            for (Trade t : trades) for (TradeListener l : listeners) l.onTrade(t);   // in trade order, under the book lock
            return trades;
        } finally { b.lock.unlock(); }
    }
    Order cancel(String symbol, String orderId) {
        OrderBook b = book(symbol);
        b.lock.lock();
        try { return b.cancel(orderId); } finally { b.lock.unlock(); }
    }
}

// ================= trading client (broker) side =================
final class Account {
    final String userId;
    long cash, blockedCash;                                  // paise
    final Map<String, Integer> shares = new HashMap<>(), blockedShares = new HashMap<>();
    Account(String userId, long cash) { this.userId = userId; this.cash = cash; }
    int free(String s) { return shares.getOrDefault(s, 0); }
    public synchronized String toString() {
        return userId + " cash=" + Money.fmt(cash) + " blocked=" + Money.fmt(blockedCash) + " shares=" + shares + " blockedShares=" + blockedShares;
    }
}

class OrderRejected extends RuntimeException { OrderRejected(String m) { super(m); } }

// Pre-trade risk checks: small strategies run in order (a simple chain of rules).
interface OrderRule { void check(Order o, Account a, long lastPrice, long blockPerUnit); }

final class QuantityRule implements OrderRule {
    public void check(Order o, Account a, long last, long block) {
        if (o.qty <= 0) throw new OrderRejected("quantity must be positive");
    }
}
final class PriceBandRule implements OrderRule {                 // limit price within +-band% of last trade
    private final int bandPercent;
    PriceBandRule(int bandPercent) { this.bandPercent = bandPercent; }
    public void check(Order o, Account a, long last, long block) {
        if (o.type == OrderType.MARKET) return;
        long lo = last * (100 - bandPercent) / 100, hi = last * (100 + bandPercent) / 100;
        if (o.limitPrice < lo || o.limitPrice > hi)
            throw new OrderRejected("price " + Money.fmt(o.limitPrice) + " outside band " + Money.fmt(lo) + "-" + Money.fmt(hi));
    }
}
final class FundsRule implements OrderRule {
    public void check(Order o, Account a, long last, long block) {
        if (o.side == Side.BUY && a.cash < block * o.qty) throw new OrderRejected("insufficient funds");
        if (o.side == Side.SELL && a.free(o.symbol) < o.qty) throw new OrderRejected("insufficient shares");
    }
}

/** The trading client: users, wallets, holdings, order history; settles trades it hears about. */
final class Broker implements TradeListener {
    private final Exchange ex;
    private final int bandPercent;
    private final Map<String, Account> accounts = new ConcurrentHashMap<>();
    private final Map<String, Order> orders = new ConcurrentHashMap<>();
    private final Map<String, Long> blockPerUnit = new ConcurrentHashMap<>();
    private final Map<String, List<Trade>> history = new ConcurrentHashMap<>();
    private final List<OrderRule> rules;
    private final AtomicLong ids = new AtomicLong();

    Broker(Exchange ex, int bandPercent) {
        this.ex = ex; this.bandPercent = bandPercent;
        this.rules = List.of(new QuantityRule(), new PriceBandRule(bandPercent), new FundsRule());
        ex.subscribe(this);
    }
    void register(String userId, long cash) { accounts.put(userId, new Account(userId, cash)); }
    void addMoney(String userId, long paise) {
        if (paise <= 0) throw new IllegalArgumentException("amount must be positive");
        Account a = acct(userId); synchronized (a) { a.cash += paise; }
    }
    void depositShares(String userId, String symbol, int qty) {
        Account a = acct(userId); synchronized (a) { a.shares.merge(symbol, qty, Integer::sum); }
    }
    Account acct(String id) {
        Account a = accounts.get(id);
        if (a == null) throw new IllegalArgumentException("unknown user " + id);
        return a;
    }

    Order place(String userId, String symbol, Side side, OrderType type, long limitPrice, int qty) {
        Account a = acct(userId);
        long last = ex.lastPrice(symbol);
        // what to block per share: the limit price, or the top of the band for a market buy
        long block = type == OrderType.LIMIT ? limitPrice : last * (100 + bandPercent) / 100;
        Order o = new Order("O" + ids.incrementAndGet(), userId, symbol, side, type, limitPrice, qty, ex.nextSeq());
        orders.put(o.id, o);
        synchronized (a) {
            try {
                for (OrderRule r : rules) r.check(o, a, last, block);
            } catch (OrderRejected e) { o.reject(); throw e; }
            if (side == Side.BUY) { a.cash -= block * qty; a.blockedCash += block * qty; }
            else { a.shares.merge(symbol, -qty, Integer::sum); a.blockedShares.merge(symbol, qty, Integer::sum); }
        }
        blockPerUnit.put(o.id, block);
        ex.submit(o);                                   // trades settle through onTrade()
        if (type == OrderType.MARKET && o.status() == OrderStatus.CANCELLED) release(o);  // IOC remainder
        return o;
    }

    boolean cancel(String userId, String orderId) {
        Order o = orders.get(orderId);
        if (o == null || !o.userId.equals(userId)) return false;
        Order c = ex.cancel(o.symbol, orderId);
        if (c == null) return false;                    // already filled or cancelled
        release(c);
        return true;
    }

    private void release(Order o) {                     // give back what is still blocked
        Account a = acct(o.userId);
        int rem = o.remaining();
        synchronized (a) {
            if (o.side == Side.BUY) { long amt = blockPerUnit.get(o.id) * rem; a.blockedCash -= amt; a.cash += amt; }
            else { a.blockedShares.merge(o.symbol, -rem, Integer::sum); a.shares.merge(o.symbol, rem, Integer::sum); }
        }
    }

    @Override public void onTrade(Trade t) {            // settlement: one account lock at a time, never nested
        Account buyer = acct(t.buyer), seller = acct(t.seller);
        long blocked = blockPerUnit.get(t.buyOrderId);
        synchronized (buyer) {
            buyer.blockedCash -= blocked * t.qty;
            buyer.cash += (blocked - t.price) * t.qty;  // refund the gap if it traded cheaper than blocked
            buyer.shares.merge(t.symbol, t.qty, Integer::sum);
        }
        synchronized (seller) {
            seller.blockedShares.merge(t.symbol, -t.qty, Integer::sum);
            seller.cash += t.price * t.qty;
        }
        history.computeIfAbsent(t.buyer, k -> new CopyOnWriteArrayList<>()).add(t);
        history.computeIfAbsent(t.seller, k -> new CopyOnWriteArrayList<>()).add(t);
    }
    List<Trade> tradesOf(String userId) { return history.getOrDefault(userId, List.of()); }
    Order order(String id) { return orders.get(id); }
    Collection<Account> accounts() { return accounts.values(); }
}

/** Observer #2: a market-data ticker that remembers the last price per symbol and fires alerts. */
final class PriceTicker implements TradeListener {
    final Map<String, Long> last = new ConcurrentHashMap<>();
    final List<String> alerts = new CopyOnWriteArrayList<>();
    private final Map<String, Long> alertAbove = new ConcurrentHashMap<>();
    void alertWhenAbove(String symbol, long price) { alertAbove.put(symbol, price); }
    public void onTrade(Trade t) {
        last.put(t.symbol, t.price);
        Long lim = alertAbove.get(t.symbol);
        if (lim != null && t.price > lim && alertAbove.remove(t.symbol, lim))
            alerts.add(t.symbol + " crossed " + Money.fmt(lim) + " at " + Money.fmt(t.price));
    }
}

public class Main {
    static int checks = 0;
    static void check(boolean ok, String what) { if (!ok) throw new AssertionError("FAILED: " + what); checks++; }
    static long rs(double rupees) { return Math.round(rupees * 100); }

    public static void main(String[] args) throws Exception {
        Exchange ex = new Exchange();
        ex.listStock("INFY", rs(1500));
        Broker zerodha = new Broker(ex, 20);
        PriceTicker ticker = new PriceTicker();
        ex.subscribe(ticker);
        ticker.alertWhenAbove("INFY", rs(1504));

        zerodha.register("asha", rs(100_000));
        zerodha.register("bala", rs(10_000));
        zerodha.register("chen", rs(10_000));
        zerodha.depositShares("bala", "INFY", 50);
        zerodha.depositShares("chen", "INFY", 50);

        System.out.println("== 1. resting sells, FIFO at the same price ==");
        Order s1 = zerodha.place("bala", "INFY", Side.SELL, OrderType.LIMIT, rs(1500), 10);
        Order s2 = zerodha.place("chen", "INFY", Side.SELL, OrderType.LIMIT, rs(1505), 5);
        Order s3 = zerodha.place("bala", "INFY", Side.SELL, OrderType.LIMIT, rs(1505), 5);
        System.out.println(ex.book("INFY").depth());

        System.out.println("== 2. buy 12 @ 1510 sweeps two levels, partial fill ==");
        Order b1 = zerodha.place("asha", "INFY", Side.BUY, OrderType.LIMIT, rs(1510), 12);
        zerodha.tradesOf("asha").forEach(System.out::println);
        check(b1.status() == OrderStatus.FILLED, "buy filled");
        check(s1.status() == OrderStatus.FILLED, "best price first");
        check(s2.status() == OrderStatus.PARTIALLY_FILLED && s2.filled() == 2, "earlier order at 1505 goes first");
        check(s3.filled() == 0, "later order at same price waits");
        check(ex.lastPrice("INFY") == rs(1505), "last price = last trade price");
        Account asha = zerodha.acct("asha");
        check(asha.cash == rs(100_000) - rs(1500) * 10 - rs(1505) * 2, "asha paid trade prices, not her limit");
        check(asha.blockedCash == 0, "nothing left blocked");
        System.out.println(asha);
        System.out.println(ex.book("INFY").depth());

        System.out.println("== 3. market buy 10: takes 3 + 5, rest cancelled ==");
        Order m = zerodha.place("asha", "INFY", Side.BUY, OrderType.MARKET, 0, 10);
        System.out.println(m);
        check(m.filled() == 8 && m.status() == OrderStatus.CANCELLED, "IOC remainder cancelled");
        check(asha.blockedCash == 0, "market block released");

        System.out.println("== 4. rejections ==");
        try { zerodha.place("asha", "INFY", Side.BUY, OrderType.LIMIT, rs(1900), 1); check(false, "band"); }
        catch (OrderRejected e) { System.out.println("rejected: " + e.getMessage()); checks++; }
        try { zerodha.place("bala", "INFY", Side.BUY, OrderType.LIMIT, rs(1500), 100); check(false, "funds"); }
        catch (OrderRejected e) { System.out.println("rejected: " + e.getMessage()); checks++; }
        try { zerodha.place("chen", "INFY", Side.SELL, OrderType.LIMIT, rs(1500), 999); check(false, "shares"); }
        catch (OrderRejected e) { System.out.println("rejected: " + e.getMessage()); checks++; }

        System.out.println("== 5. resting bid, then cancel releases the block ==");
        long before = zerodha.acct("bala").cash;
        Order bid = zerodha.place("bala", "INFY", Side.BUY, OrderType.LIMIT, rs(1490), 4);
        check(zerodha.acct("bala").blockedCash == rs(1490) * 4, "cash blocked");
        check(zerodha.cancel("bala", bid.id), "cancel ok");
        check(!zerodha.cancel("bala", bid.id), "second cancel is a no-op");
        check(zerodha.acct("bala").cash == before && zerodha.acct("bala").blockedCash == 0, "block released");
        System.out.println(bid);

        System.out.println("== 6. observer: ticker and price alert ==");
        System.out.println("ticker last INFY = " + Money.fmt(ticker.last.get("INFY")) + ", alerts = " + ticker.alerts);
        check(ticker.alerts.size() == 1, "alert fired once");

        System.out.println("== 7. 4 threads x 2000 random orders: conservation + uncrossed book ==");
        Exchange ex2 = new Exchange();
        ex2.listStock("TCS", rs(4000));
        Broker br = new Broker(ex2, 20);
        for (int i = 0; i < 8; i++) { br.register("u" + i, rs(10_000_000)); br.depositShares("u" + i, "TCS", 10_000); }
        long cash0 = totalCash(br), shares0 = totalShares(br, "TCS");
        ExecutorService pool = Executors.newFixedThreadPool(4);
        AtomicInteger rejected = new AtomicInteger();
        for (int t = 0; t < 4; t++) {
            final int seed = t;
            pool.submit(() -> {
                Random r = new Random(seed);
                for (int i = 0; i < 2000; i++) {
                    String u = "u" + r.nextInt(8);
                    Side side = r.nextBoolean() ? Side.BUY : Side.SELL;
                    long px = rs(3990) + r.nextInt(21) * 100;           // 3990..4010
                    try {
                        Order o = br.place(u, "TCS", side, OrderType.LIMIT, px, 1 + r.nextInt(20));
                        if (r.nextInt(5) == 0) br.cancel(u, o.id);
                    } catch (OrderRejected e) { rejected.incrementAndGet(); }
                }
            });
        }
        pool.shutdown();
        check(pool.awaitTermination(30, TimeUnit.SECONDS), "threads done");
        OrderBook tcs = ex2.book("TCS");
        check(tcs.bestBid() == null || tcs.bestAsk() == null || tcs.bestBid() < tcs.bestAsk(), "book never left crossed");
        check(totalCash(br) == cash0, "cash conserved (free + blocked)");
        check(totalShares(br, "TCS") == shares0, "shares conserved (free + blocked)");
        System.out.println("cash and shares conserved; book uncrossed: " + (tcs.bestBid() == null || tcs.bestAsk() == null || tcs.bestBid() < tcs.bestAsk()));
        System.out.println("All " + checks + " checks passed");
    }

    static long totalCash(Broker b) {
        long s = 0; for (Account a : b.accounts()) synchronized (a) { s += a.cash + a.blockedCash; } return s;
    }
    static long totalShares(Broker b, String sym) {
        long s = 0;
        for (Account a : b.accounts()) synchronized (a) { s += a.shares.getOrDefault(sym, 0) + a.blockedShares.getOrDefault(sym, 0); }
        return s;
    }
}

"""Stock exchange order book (price-time priority) + a broker / trading client. Money in paise."""
from __future__ import annotations

import bisect
import enum
import itertools
import threading
from collections import deque
from dataclasses import dataclass, field
from typing import Callable, Deque, Dict, List, Optional


class Side(enum.Enum):
    BUY = "BUY"
    SELL = "SELL"


class OrderType(enum.Enum):
    LIMIT = "LIMIT"
    MARKET = "MARKET"


class Status(enum.Enum):
    OPEN = "OPEN"
    PARTIALLY_FILLED = "PARTIALLY_FILLED"
    FILLED = "FILLED"
    CANCELLED = "CANCELLED"
    REJECTED = "REJECTED"


@dataclass(eq=False)
class Order:
    id: str
    user: str
    symbol: str
    side: Side
    type: OrderType
    price: int          # limit price in paise (0 for market)
    qty: int
    seq: int
    filled: int = 0
    status: Status = Status.OPEN

    @property
    def remaining(self) -> int:
        return self.qty - self.filled

    def fill(self, q: int) -> None:
        self.filled += q
        self.status = Status.FILLED if self.remaining == 0 else Status.PARTIALLY_FILLED


@dataclass(frozen=True)
class Trade:
    id: int
    symbol: str
    buy_order: str
    sell_order: str
    buyer: str
    seller: str
    price: int
    qty: int


def fmt(paise: int) -> str:
    return f"{paise // 100}.{paise % 100:02d}"


class PriceLevels:
    """Sorted prices + a FIFO queue per price. best() is the first price (lowest for asks)."""

    def __init__(self, descending: bool) -> None:
        self._sign = -1 if descending else 1
        self._keys: List[int] = []            # sign * price, ascending
        self._q: Dict[int, Deque[Order]] = {}

    def __bool__(self) -> bool:
        return bool(self._keys)

    def best(self) -> Optional[int]:
        return self._sign * self._keys[0] if self._keys else None

    def queue(self, price: int) -> Deque[Order]:
        return self._q[price]

    def add(self, o: Order) -> None:
        if o.price not in self._q:
            bisect.insort(self._keys, self._sign * o.price)
            self._q[o.price] = deque()
        self._q[o.price].append(o)

    def remove(self, o: Order) -> None:
        q = self._q[o.price]
        q.remove(o)
        if not q:
            self.drop(o.price)

    def drop(self, price: int) -> None:
        del self._q[price]
        self._keys.pop(bisect.bisect_left(self._keys, self._sign * price))

    def depth(self) -> List[str]:
        return [f"{fmt(self._sign * k)}x{sum(o.remaining for o in self._q[self._sign * k])}" for k in self._keys]


class OrderBook:
    def __init__(self, symbol: str, open_price: int) -> None:
        self.symbol = symbol
        self.last_price = open_price
        self.bids = PriceLevels(descending=True)
        self.asks = PriceLevels(descending=False)
        self.resting: Dict[str, Order] = {}
        self.lock = threading.Lock()           # one writer per symbol

    def match(self, o: Order, trade_ids: Callable[[], int]) -> List[Trade]:
        trades: List[Trade] = []
        book = self.asks if o.side is Side.BUY else self.bids
        while o.remaining and book:
            price = book.best()
            if o.type is OrderType.LIMIT and not (o.price >= price if o.side is Side.BUY else o.price <= price):
                break
            level = book.queue(price)
            maker = level[0]                    # earliest order at the best price
            q = min(o.remaining, maker.remaining)
            o.fill(q)
            maker.fill(q)
            self.last_price = price             # trade at the resting order's price
            buy, sell = (o, maker) if o.side is Side.BUY else (maker, o)
            trades.append(Trade(trade_ids(), self.symbol, buy.id, sell.id, buy.user, sell.user, price, q))
            if maker.remaining == 0:
                level.popleft()
                del self.resting[maker.id]
                if not level:
                    book.drop(price)
        if o.remaining:
            if o.type is OrderType.LIMIT:
                (self.bids if o.side is Side.BUY else self.asks).add(o)
                self.resting[o.id] = o
            else:
                o.status = Status.CANCELLED     # market remainder: immediate-or-cancel
        return trades

    def cancel(self, order_id: str) -> Optional[Order]:
        o = self.resting.pop(order_id, None)
        if o is None:
            return None
        (self.bids if o.side is Side.BUY else self.asks).remove(o)
        o.status = Status.CANCELLED
        return o

    def depth(self) -> str:
        return f"{self.symbol} asks: {' '.join(reversed(self.asks.depth()))} | bids: {' '.join(self.bids.depth())}"


class Exchange:
    def __init__(self) -> None:
        self._books: Dict[str, OrderBook] = {}
        self._listeners: List[Callable[[Trade], None]] = []
        self._seq = itertools.count(1)
        self._tid = itertools.count(1)
        self._id_lock = threading.Lock()

    def list_stock(self, symbol: str, open_price: int) -> None:
        if symbol in self._books:
            raise ValueError(f"already listed: {symbol}")
        self._books[symbol] = OrderBook(symbol, open_price)

    def subscribe(self, fn: Callable[[Trade], None]) -> None:
        self._listeners.append(fn)

    def next_seq(self) -> int:
        with self._id_lock:
            return next(self._seq)

    def _next_trade_id(self) -> int:
        with self._id_lock:
            return next(self._tid)

    def book(self, symbol: str) -> OrderBook:
        if symbol not in self._books:
            raise KeyError(f"unknown symbol {symbol}")
        return self._books[symbol]

    def submit(self, o: Order) -> List[Trade]:
        b = self.book(o.symbol)
        with b.lock:
            trades = b.match(o, self._next_trade_id)
            for t in trades:
                for fn in self._listeners:      # Observer: settle, tick, alert - in trade order
                    fn(t)
            return trades

    def cancel(self, symbol: str, order_id: str) -> Optional[Order]:
        b = self.book(symbol)
        with b.lock:
            return b.cancel(order_id)


class OrderRejected(Exception):
    pass


@dataclass
class Account:
    user: str
    cash: int
    blocked_cash: int = 0
    shares: Dict[str, int] = field(default_factory=dict)
    blocked_shares: Dict[str, int] = field(default_factory=dict)
    lock: threading.Lock = field(default_factory=threading.Lock, repr=False)


# pre-trade rules: each raises OrderRejected
def quantity_rule(o: Order, a: Account, last: int, block: int, band: int) -> None:
    if o.qty <= 0:
        raise OrderRejected("quantity must be positive")


def price_band_rule(o: Order, a: Account, last: int, block: int, band: int) -> None:
    if o.type is OrderType.MARKET:
        return
    lo, hi = last * (100 - band) // 100, last * (100 + band) // 100
    if not lo <= o.price <= hi:
        raise OrderRejected(f"price {fmt(o.price)} outside band {fmt(lo)}-{fmt(hi)}")


def funds_rule(o: Order, a: Account, last: int, block: int, band: int) -> None:
    if o.side is Side.BUY and a.cash < block * o.qty:
        raise OrderRejected("insufficient funds")
    if o.side is Side.SELL and a.shares.get(o.symbol, 0) < o.qty:
        raise OrderRejected("insufficient shares")


class Broker:
    """Trading client: wallets, holdings, risk checks, settlement, history."""

    def __init__(self, ex: Exchange, band_percent: int = 20) -> None:
        self.ex, self.band = ex, band_percent
        self.accounts: Dict[str, Account] = {}
        self.orders: Dict[str, Order] = {}
        self.block_per_unit: Dict[str, int] = {}
        self.history: Dict[str, List[Trade]] = {}
        self.rules = [quantity_rule, price_band_rule, funds_rule]
        self._ids = itertools.count(1)
        self._ids_lock = threading.Lock()
        ex.subscribe(self.on_trade)

    def register(self, user: str, cash: int) -> None:
        self.accounts[user] = Account(user, cash)

    def deposit_shares(self, user: str, symbol: str, qty: int) -> None:
        a = self.accounts[user]
        with a.lock:
            a.shares[symbol] = a.shares.get(symbol, 0) + qty

    def place(self, user: str, symbol: str, side: Side, otype: OrderType, price: int, qty: int) -> Order:
        a = self.accounts[user]
        last = self.ex.book(symbol).last_price
        block = price if otype is OrderType.LIMIT else last * (100 + self.band) // 100
        with self._ids_lock:
            oid = f"O{next(self._ids)}"
        o = Order(oid, user, symbol, side, otype, price, qty, self.ex.next_seq())
        self.orders[oid] = o
        with a.lock:
            try:
                for rule in self.rules:
                    rule(o, a, last, block, self.band)
            except OrderRejected:
                o.status = Status.REJECTED
                raise
            if side is Side.BUY:
                a.cash -= block * qty
                a.blocked_cash += block * qty
            else:
                a.shares[symbol] -= qty
                a.blocked_shares[symbol] = a.blocked_shares.get(symbol, 0) + qty
        self.block_per_unit[oid] = block
        self.ex.submit(o)
        if otype is OrderType.MARKET and o.status is Status.CANCELLED:
            self._release(o)
        return o

    def cancel(self, user: str, order_id: str) -> bool:
        o = self.orders.get(order_id)
        if o is None or o.user != user:
            return False
        c = self.ex.cancel(o.symbol, order_id)
        if c is None:
            return False
        self._release(c)
        return True

    def _release(self, o: Order) -> None:
        a = self.accounts[o.user]
        with a.lock:
            if o.side is Side.BUY:
                amt = self.block_per_unit[o.id] * o.remaining
                a.blocked_cash -= amt
                a.cash += amt
            else:
                a.blocked_shares[o.symbol] -= o.remaining
                a.shares[o.symbol] += o.remaining

    def on_trade(self, t: Trade) -> None:
        buyer, seller = self.accounts[t.buyer], self.accounts[t.seller]
        blocked = self.block_per_unit[t.buy_order]
        with buyer.lock:                         # one account lock at a time: no lock-order deadlock
            buyer.blocked_cash -= blocked * t.qty
            buyer.cash += (blocked - t.price) * t.qty
            buyer.shares[t.symbol] = buyer.shares.get(t.symbol, 0) + t.qty
        with seller.lock:
            seller.blocked_shares[t.symbol] -= t.qty
            seller.cash += t.price * t.qty
        self.history.setdefault(t.buyer, []).append(t)
        self.history.setdefault(t.seller, []).append(t)

def rs(rupees: float) -> int:

def main() -> None:

    for u in ("bala", "chen"):
        zerodha.register(u, rs(10_000))
        zerodha.deposit_shares(u, "INFY", 50)

def totals(br, sym="X"):
    cash = sum(a.cash + a.blocked_cash for a in br.accounts.values())
    sh = sum(a.shares.get(sym, 0) + a.blocked_shares.get(sym, 0) for a in br.accounts.values())

// Matching engine core in C++14: price-time priority order book with partial fills,
// market (IOC) orders, cancel by id, and trade listeners (Observer via std::function).
// Single-threaded on purpose: one engine thread per symbol is how real books run.
// Build: g++ -std=c++14 -O2 -Wall main.cpp -o main.exe
#include <cassert>
#include <cstdio>
#include <deque>
#include <functional>
#include <map>
#include <memory>
#include <string>
#include <unordered_map>
#include <vector>

enum class Side { BUY, SELL };
enum class Type { LIMIT, MARKET };

struct Order {
    long id; Side side; Type type; long price; int qty; int filled = 0; bool cancelled = false;
    int remaining() const { return qty - filled; }
};
struct Trade { long buyId, sellId, price; int qty; };

static std::string fmt(long paise) {
    char buf[32]; std::snprintf(buf, sizeof buf, "%ld.%02ld", paise / 100, paise % 100); return buf;
}

class OrderBook {
    using Level = std::deque<std::shared_ptr<Order>>;
    std::map<long, Level, std::greater<long>> bids_;   // highest first
    std::map<long, Level> asks_;                        // lowest first
    std::unordered_map<long, std::shared_ptr<Order>> resting_;
    std::vector<std::function<void(const Trade&)>> listeners_;
    long lastPrice_;

    template <class Book>
    void sweep(Order& in, Book& book, std::vector<Trade>& out) {
        while (in.remaining() > 0 && !book.empty()) {
            auto best = book.begin();
            long px = best->first;
            bool crosses = in.side == Side::BUY ? in.price >= px : in.price <= px;
            if (in.type == Type::LIMIT && !crosses) break;
            Level& level = best->second;
            Order& maker = *level.front();              // earliest at the best price
            int q = std::min(in.remaining(), maker.remaining());
            in.filled += q; maker.filled += q; lastPrice_ = px;
            Trade t = in.side == Side::BUY ? Trade{in.id, maker.id, px, q} : Trade{maker.id, in.id, px, q};
            out.push_back(t);
            for (auto& fn : listeners_) fn(t);
            if (maker.remaining() == 0) { resting_.erase(maker.id); level.pop_front(); }
            if (level.empty()) book.erase(best);
        }
    }
    template <class Book>
    static void removeFrom(Book& book, const std::shared_ptr<Order>& o) {
        auto it = book.find(o->price);
        Level& lv = it->second;
        for (auto i = lv.begin(); i != lv.end(); ++i) if (*i == o) { lv.erase(i); break; }
        if (lv.empty()) book.erase(it);
    }
public:
    explicit OrderBook(long open) : lastPrice_(open) {}
    void onTrade(std::function<void(const Trade&)> fn) { listeners_.push_back(std::move(fn)); }
    long lastPrice() const { return lastPrice_; }

    std::vector<Trade> submit(std::shared_ptr<Order> o) {
        std::vector<Trade> trades;
        if (o->side == Side::BUY) sweep(*o, asks_, trades); else sweep(*o, bids_, trades);
        if (o->remaining() > 0) {
            if (o->type == Type::MARKET) o->cancelled = true;          // immediate-or-cancel
            else {
                if (o->side == Side::BUY) bids_[o->price].push_back(o); else asks_[o->price].push_back(o);
                resting_[o->id] = o;
            }
        }
        return trades;
    }
    bool cancel(long id) {
        auto it = resting_.find(id);
        if (it == resting_.end()) return false;
        auto o = it->second; resting_.erase(it);
        if (o->side == Side::BUY) removeFrom(bids_, o); else removeFrom(asks_, o);
        o->cancelled = true;
        return true;
    }
    long bestBid() const { return bids_.empty() ? -1 : bids_.begin()->first; }
    long bestAsk() const { return asks_.empty() ? -1 : asks_.begin()->first; }
    void print() const {
        std::printf("asks:");
        for (auto it = asks_.rbegin(); it != asks_.rend(); ++it) {
            int s = 0; for (auto& o : it->second) s += o->remaining();
            std::printf(" %sx%d", fmt(it->first).c_str(), s);
        }
        std::printf(" | bids:");
        for (auto& kv : bids_) { int s = 0; for (auto& o : kv.second) s += o->remaining(); std::printf(" %sx%d", fmt(kv.first).c_str(), s); }
        std::printf("\n");
    }
};

int main() {
    long nextId = 0;
    auto mk = [&](Side s, Type t, long px, int q) { return std::make_shared<Order>(Order{++nextId, s, t, px, q}); };
    OrderBook book(150000);
    long volume = 0;
    book.onTrade([&](const Trade& t) {                      // observer 1: tape
        std::printf("trade %d @ %s (buy #%ld x sell #%ld)\n", t.qty, fmt(t.price).c_str(), t.buyId, t.sellId);
    });
    book.onTrade([&](const Trade& t) { volume += t.qty; });  // observer 2: volume counter

    std::printf("== 1. rest three sells ==\n");
    auto s1 = mk(Side::SELL, Type::LIMIT, 150000, 10);
    auto s2 = mk(Side::SELL, Type::LIMIT, 150500, 5);
    auto s3 = mk(Side::SELL, Type::LIMIT, 150500, 5);
    book.submit(s1); book.submit(s2); book.submit(s3);
    book.print();

    std::printf("== 2. buy 12 @ 1510.00 ==\n");
    auto b1 = mk(Side::BUY, Type::LIMIT, 151000, 12);
    auto tr = book.submit(b1);
    assert(tr.size() == 2 && b1->remaining() == 0);
    assert(s1->remaining() == 0 && s2->filled == 2 && s3->filled == 0);   // price, then time
