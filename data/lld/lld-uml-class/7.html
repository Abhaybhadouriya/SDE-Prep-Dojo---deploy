From diagram to code

Every element maps to code mechanically. That is why a good diagram makes the coding phase fast. Here is the stepper's diagram in three languages; each file compiles and runs with a small demo (in lld-code/lld-uml-class/).

import java.util.*;

/** The class diagram from the "Class Diagrams" page, written as code. Run: java ParkingModel.java */
public class ParkingModel {

    // [enum]
    enum VehicleType { BIKE, CAR, TRUCK }                       // <<enumeration>>
    // [/enum]

    // [abstract]
    static abstract class Vehicle {                             // {abstract}, name in italics
        private final String plate;                             // - plate: String {readOnly}
        protected Vehicle(String plate) { this.plate = plate; }
        public String getPlate() { return plate; }              // + getPlate(): String
        public abstract VehicleType getType();                  // + getType(): VehicleType {abstract}
    }
    static class Car extends Vehicle {                          // generalization (hollow triangle)
        Car(String plate) { super(plate); }
        public VehicleType getType() { return VehicleType.CAR; }
    }
    static class Bike extends Vehicle {
        Bike(String plate) { super(plate); }
        public VehicleType getType() { return VehicleType.BIKE; }
    }
    // [/abstract]

    // [interface]
    interface PricingStrategy { long fee(int hours); }          // <<interface>>
    static class HourlyPricing implements PricingStrategy {     // realization (dashed, hollow triangle)
        private final long ratePerHour;                         // - ratePerHour: long
        HourlyPricing(long ratePerHour) { this.ratePerHour = ratePerHour; }
        public long fee(int hours) { return Math.max(1, hours) * ratePerHour; }
    }
    // [/interface]

    // [spot]
    static class ParkingSpot {
        private final String id;                                // - id: String
        private final VehicleType fits;                         // - fits: VehicleType
        private Vehicle parked;                                 // - parked: Vehicle [0..1]  (null = empty)
        ParkingSpot(String id, VehicleType fits) { this.id = id; this.fits = fits; }
        public String getId() { return id; }
        public boolean isFree() { return parked == null; }      // + /free: Boolean (derived)
        public boolean canFit(Vehicle v) { return isFree() && v.getType() == fits; }
        public void park(Vehicle v) {
            if (!canFit(v)) throw new IllegalStateException(id + " cannot take " + v.getPlate());
            parked = v;
        }
        public Vehicle leave() { Vehicle v = parked; parked = null; return v; }
    }
    // [/spot]

    // [level]
    record SpotSpec(String id, VehicleType fits) {}             // plain data in; the Level builds the spot

    static class Level {
        private final int floor;                                // - floor: int
        private final List<ParkingSpot> spots;                  // - spots: ParkingSpot [1..*] {ordered}
        Level(int floor, List<SpotSpec> specs) {                // composition: the level CREATES its spots
            if (specs.isEmpty()) throw new IllegalArgumentException("a level needs at least one spot");
            List<ParkingSpot> made = new ArrayList<>();
            for (SpotSpec s : specs) made.add(new ParkingSpot(s.id(), s.fits()));
            this.floor = floor; this.spots = List.copyOf(made);
        }
        public Optional<ParkingSpot> findSpot(Vehicle v) {
            return spots.stream().filter(s -> s.canFit(v)).findFirst();
        }
    }
    // [/level]

    // [lot]
    record Ticket(int id, String plate, String spotId) {}       // <<record>> value object

    static class ParkingLot {
        private static int nextTicketId = 1;                    // - nextTicketId: int  (underlined = static)
        private final List<Level> levels = new ArrayList<>();  // - levels: Level [1..*], created here
        private final PricingStrategy pricing;                  // - pricing: PricingStrategy (passed in)
        private final Map<Integer, ParkingSpot> active = new HashMap<>();
        ParkingLot(PricingStrategy pricing) { this.pricing = pricing; }
        // composition: the lot creates each Level from a spec, never receives a ready-made one,
        // so no Level can exist outside a lot or be shared by two lots
        public void addLevel(int floor, List<SpotSpec> spots) { levels.add(new Level(floor, spots)); }

        public Ticket park(Vehicle v) {                         // + park(v: Vehicle): Ticket
            for (Level l : levels) {
                Optional<ParkingSpot> s = l.findSpot(v);
                if (s.isPresent()) {
                    s.get().park(v);
                    Ticket t = new Ticket(nextTicketId++, v.getPlate(), s.get().getId());
                    active.put(t.id(), s.get());
                    return t;
                }
            }
            throw new IllegalStateException("no spot for " + v.getType());
        }
        public long unpark(Ticket t, int hours) {               // + unpark(t: Ticket, hours: int): long
            ParkingSpot s = active.remove(t.id());
            if (s == null) throw new IllegalArgumentException("unknown ticket " + t.id());
            s.leave();
            return pricing.fee(hours);
        }
    }
    // [/lot]

    public static void main(String[] args) {
        ParkingLot lot = new ParkingLot(new HourlyPricing(30));
        lot.addLevel(0, List.of(new SpotSpec("G-1", VehicleType.CAR), new SpotSpec("G-2", VehicleType.BIKE)));
        Ticket t1 = lot.park(new Car("KA01AB1234"));
        Ticket t2 = lot.park(new Bike("KA05XY9"));
        System.out.println(t1 + "\n" + t2);
        try { lot.park(new Car("MH12CD5678")); } catch (IllegalStateException e) { System.out.println("expected: " + e.getMessage()); }
        System.out.println("fee for 3h = " + lot.unpark(t1, 3));
        System.out.println("car again: " + lot.park(new Car("MH12CD5678")));
    }
}
"""The class diagram from the "Class Diagrams" page in Python 3.11. Run: python parking_model.py"""
from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum, auto
from typing import Protocol


class VehicleType(Enum):                 # <<enumeration>>
    BIKE = auto()
    CAR = auto()
    TRUCK = auto()


class Vehicle(ABC):                      # {abstract}
    def __init__(self, plate: str) -> None:
        self._plate = plate              # "_" = private by convention

    @property
    def plate(self) -> str:
        return self._plate

    @abstractmethod
    def vehicle_type(self) -> VehicleType: ...


class Car(Vehicle):
    def vehicle_type(self) -> VehicleType:
        return VehicleType.CAR


class Bike(Vehicle):
    def vehicle_type(self) -> VehicleType:
        return VehicleType.BIKE


class PricingStrategy(Protocol):         # <<interface>> (structural typing)
    def fee(self, hours: int) -> int: ...


class HourlyPricing:
    def __init__(self, rate_per_hour: int) -> None:
        self._rate = rate_per_hour

    def fee(self, hours: int) -> int:
        return max(1, hours) * self._rate


class ParkingSpot:
    def __init__(self, spot_id: str, fits: VehicleType) -> None:
        self.spot_id = spot_id
        self._fits = fits
        self._parked: Vehicle | None = None   # multiplicity 0..1

    @property
    def free(self) -> bool:                  # derived attribute /free
        return self._parked is None

    def can_fit(self, v: Vehicle) -> bool:
        return self.free and v.vehicle_type() == self._fits

    def park(self, v: Vehicle) -> None:
        if not self.can_fit(v):
            raise RuntimeError(f"{self.spot_id} cannot take {v.plate}")
        self._parked = v

    def leave(self) -> Vehicle | None:
        v, self._parked = self._parked, None
        return v


@dataclass(frozen=True)
class SpotSpec:                          # plain data in; the Level builds the spot
    spot_id: str
    fits: VehicleType


class Level:
    def __init__(self, floor: int, specs: list[SpotSpec]) -> None:
        if not specs:
            raise ValueError("a level needs at least one spot")   # multiplicity 1..*
        self.floor = floor
        self._spots = [ParkingSpot(s.spot_id, s.fits) for s in specs]   # composition: created here

    def find_spot(self, v: Vehicle) -> ParkingSpot | None:
        return next((s for s in self._spots if s.can_fit(v)), None)


@dataclass(frozen=True)
class Ticket:                            # <<record>>
    ticket_id: int
    plate: str
    spot_id: str


class ParkingLot:
    _next_ticket_id = 1                  # class attribute = static (underlined in UML)

    def __init__(self, pricing: PricingStrategy) -> None:
        self._levels: list[Level] = []    # composition: levels are created by add_level, never passed in
        self._pricing = pricing
        self._active: dict[int, ParkingSpot] = {}

    def add_level(self, floor: int, spots: list[SpotSpec]) -> None:
        self._levels.append(Level(floor, spots))

    def park(self, v: Vehicle) -> Ticket:
        for level in self._levels:
            spot = level.find_spot(v)
            if spot:
                spot.park(v)
                t = Ticket(ParkingLot._next_ticket_id, v.plate, spot.spot_id)
                ParkingLot._next_ticket_id += 1
                self._active[t.ticket_id] = spot
                return t
        raise RuntimeError(f"no spot for {v.vehicle_type().name}")

    def unpark(self, t: Ticket, hours: int) -> int:
        spot = self._active.pop(t.ticket_id)
        spot.leave()
        return self._pricing.fee(hours)


if __name__ == "__main__":
    lot = ParkingLot(HourlyPricing(30))
    lot.add_level(0, [SpotSpec("G-1", VehicleType.CAR), SpotSpec("G-2", VehicleType.BIKE)])
    t1 = lot.park(Car("KA01AB1234"))

    try:
        lot.park(Car("MH12CD5678"))
    except RuntimeError as e:
        print("expected:", e)
    print("fee for 3h =", lot.unpark(t1, 3))
// The class diagram from the "Class Diagrams" page in C++14.
// Build: g++ -std=c++14 -Wall -o parking_model parking_model.cpp && ./parking_model
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

enum class VehicleType { Bike, Car, Truck };                 // <<enumeration>>

class Vehicle {                                              // {abstract}: has a pure virtual
public:
    explicit Vehicle(std::string plate) : plate_(std::move(plate)) {}
    virtual ~Vehicle() = default;
    const std::string& plate() const { return plate_; }      // + plate(): String
    virtual VehicleType type() const = 0;                    // + type(): VehicleType {abstract}
private:
    std::string plate_;                                      // - plate: String
};

class Car : public Vehicle {
public:
    using Vehicle::Vehicle;
    VehicleType type() const override { return VehicleType::Car; }
};

class PricingStrategy {                                      // <<interface>> = only pure virtuals
public:
    virtual ~PricingStrategy() = default;
    virtual long fee(int hours) const = 0;
};

class HourlyPricing : public PricingStrategy {
public:
    explicit HourlyPricing(long rate) : rate_(rate) {}
    long fee(int hours) const override { return (hours < 1 ? 1 : hours) * rate_; }
private:
    long rate_;
};

class ParkingSpot {
public:
    ParkingSpot(std::string id, VehicleType fits) : id_(std::move(id)), fits_(fits) {}
    const std::string& id() const { return id_; }
    bool free() const { return parked_ == nullptr; }          // derived: /free
    bool canFit(const Vehicle& v) const { return free() && v.type() == fits_; }
    void park(const Vehicle* v) {
        if (!canFit(*v)) throw std::runtime_error(id_ + " cannot take " + v->plate());
        parked_ = v;                                         // association: spot does not own the car
    }
    void leave() { parked_ = nullptr; }
private:
    std::string id_;
    VehicleType fits_;
    const Vehicle* parked_ = nullptr;                        // - parked: Vehicle [0..1]
};

struct Ticket { int id; std::string plate; std::string spotId; };

class ParkingLot {
public:
    ParkingLot(std::vector<ParkingSpot> spots, std::unique_ptr<PricingStrategy> p)
        : spots_(std::move(spots)), pricing_(std::move(p)) {
        if (spots_.empty()) throw std::invalid_argument("need at least one spot");   // [1..*]
    }
    Ticket park(const Vehicle& v) {
        for (std::size_t i = 0; i < spots_.size(); ++i) {
            if (spots_[i].canFit(v)) {
                spots_[i].park(&v);
                Ticket t{nextTicketId_++, v.plate(), spots_[i].id()};
                active_[t.id] = i;
                return t;
            }
        }
        throw std::runtime_error("no spot");
    }
    long unpark(const Ticket& t, int hours) {
        auto it = active_.find(t.id);
        if (it == active_.end()) throw std::invalid_argument("unknown ticket");
        spots_[it->second].leave();
        active_.erase(it);
        return pricing_->fee(hours);
    }
private:
    static int nextTicketId_;                                // underlined in UML = static
    std::vector<ParkingSpot> spots_;                         // composition: spots die with the lot
    std::unique_ptr<PricingStrategy> pricing_;
    std::map<int, std::size_t> active_;
};
int ParkingLot::nextTicketId_ = 1;

int main() {
    ParkingLot lot({ParkingSpot("G-1", VehicleType::Car), ParkingSpot("G-2", VehicleType::Bike)},
                   std::unique_ptr<PricingStrategy>(new HourlyPricing(30)));
    Car a("KA01AB1234"), b("MH12CD5678");
    Ticket t = lot.park(a);
    std::cout << "ticket " << t.id << " " << t.plate << " at " << t.spotId << "\n";
    try { lot.park(b); } catch (const std::runtime_error& e) { std::cout << "expected: " << e.what() << "\n"; }
    std::cout << "fee for 3h = " << lot.unpark(t, 3) << "\n";
    std::cout << "second car at " << lot.park(b).spotId << "\n";
    return 0;
}

Mapping notes worth saying out loud: composition becomes a private collection created and owned by the whole (here ParkingLot.addLevel(floor, specs) builds each Level, and each Level builds its spots from plain SpotSpec data; if main built the levels and passed them in, the honest diagram would be aggregation, a hollow diamond) (in C++ it is a value member or unique_ptr); association becomes a reference the object does not own (a raw non-owning pointer in the C++ version); 0..1 becomes null/None; the underlined nextTicketId becomes a static field.

From your world

In C the same diagram is a set of structs: struct parking_lot embeds an array of struct level (composition), each spot holds a struct vehicle * it does not free (association), and struct pricing_ops { long (*fee)(int hours); } is the interface. The abstract Vehicle is a struct with an ops pointer that car and bike fill in. If you can draw your driver's structs, you can draw a class diagram.
