Code: editor undo, game saves and a router commit

Each tab is a complete program that was compiled and run. Java builds the editor above with a private nested Snapshot, commands that store snapshots, and a bounded undo/redo history. Python saves game checkpoints, shows the shallow-copy trap, and measures what a full snapshot costs against a delta. C++14 applies a batch of routing-table changes all-or-nothing by restoring a checkpoint on error. Files live in lld-code/lld-pat-memento/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Supplier<T>, Consumer<T>, Function<A,B>: ready-made one-slot interfaces: "returns a T", "takes a T", "maps A to B". A lambda can fill any of them.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • friend class X; (C++): lets class X read this class's private members, and nobody else.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

public class MementoDemo {

    // ---- Command that stores a memento so it can undo itself ----
    interface Command { void execute(); void undo(); String name(); }

    static final class EditCommand implements Command {
        private final Editor editor;
        private final String name;
        private final java.util.function.Consumer<Editor> action;
        private Editor.Snapshot before;                       // memento taken just before executing
        EditCommand(Editor e, String name, java.util.function.Consumer<Editor> action) {
            this.editor = e; this.name = name; this.action = action;
        }
        @Override public void execute() { before = editor.save(); action.accept(editor); }
        @Override public void undo()    { editor.restore(before); }
        @Override public String name()  { return name; }
    }

    // ---- Caretaker: keeps history, never looks inside a snapshot ----
    static final class History {
        private final Deque<Command> undo = new ArrayDeque<>();
        private final Deque<Command> redo = new ArrayDeque<>();
        private final int limit;
        History(int limit) { this.limit = limit; }

        void run(Command c) {
            c.execute();
            undo.push(c);
            if (undo.size() > limit) undo.removeLast();       // bound memory: drop the oldest
            redo.clear();                                      // a new edit kills the redo branch
        }
        boolean undo() { if (undo.isEmpty()) return false; Command c = undo.pop(); c.undo(); redo.push(c); return true; }
        boolean redo() { if (redo.isEmpty()) return false; Command c = redo.pop(); c.execute(); undo.push(c); return true; }
    }

    public static void main(String[] args) {
        Editor ed = new Editor();
        History h = new History(3);                            // keep at most 3 undo steps

        h.run(new EditCommand(ed, "type Hello", e -> e.type("Hello")));
        h.run(new EditCommand(ed, "type world", e -> e.type(" world")));
        h.run(new EditCommand(ed, "insert ,", e -> { e.moveTo(5); e.type(","); }));
        System.out.println("after edits : " + ed.show());

        h.undo(); System.out.println("undo        : " + ed.show());
        h.undo(); System.out.println("undo        : " + ed.show());
        h.redo(); System.out.println("redo        : " + ed.show());

        h.run(new EditCommand(ed, "backspace 5", e -> e.backspace(5)));   // clears redo stack
        System.out.println("backspace 5 : " + ed.show());
        System.out.println("redo again? : " + h.redo());

        h.run(new EditCommand(ed, "type !", e -> e.type("!")));
        h.run(new EditCommand(ed, "type ?", e -> e.type("?")));
        int steps = 0;
        while (h.undo()) steps++;
        System.out.println("undo all    : " + ed.show() + "  (" + steps + " steps, oldest dropped by limit 3)");

        Editor.Snapshot s = ed.save();
        System.out.println("caretaker sees only: " + s);    // no getters: encapsulation intact
    }
}

// ---- Originator: owns the state and is the ONLY class that can read a memento ----
// Editor is a TOP-LEVEL class (not nested in MementoDemo). Java lets code read private members of
// a nested class anywhere inside the same top-level class, so if Editor were nested in MementoDemo,
// History (also in MementoDemo) could read snapshot.text. Here only Editor's own body can.
final class Editor {
    private final StringBuilder text = new StringBuilder();
    private int cursor = 0;

    void type(String s)  { text.insert(cursor, s); cursor += s.length(); }
    void backspace(int n) { int from = Math.max(0, cursor - n); text.delete(from, cursor); cursor = from; }
    void moveTo(int pos) { cursor = Math.max(0, Math.min(pos, text.length())); }
    String show() { return new StringBuilder(text).insert(cursor, '|').toString(); }

    // Memento: private fields + private constructor. Private members are visible only inside the
    // top-level class that encloses them (Editor), so the caretaker cannot read or create one.
    static final class Snapshot {
        private final String text;               // String is immutable: a safe deep copy
        private final int cursor;
        private Snapshot(String text, int cursor) { this.text = text; this.cursor = cursor; }
        @Override public String toString() { return "Snapshot(opaque)"; }   // no state shown
    }

    Snapshot save() { return new Snapshot(text.toString(), cursor); }
    void restore(Snapshot s) { text.setLength(0); text.append(s.text); cursor = s.cursor; }
}

Follow the last lines: the history limit is 3, so once "backspace 5", "type !" and "type ?" fill it, only those three can be undone and the text stops at "Hello world|", not at empty. That is the memory bound every real editor needs. Because Snapshot stores a String (immutable), the copy is safe; a mutable field would need a deep copy. Immutable snapshots can also be shared between threads without locks; see immutability and safe publication.

"""Memento in Python: game save points, the shallow-copy trap, and the cost of snapshots."""
import copy
import pickle


class Player:
    """Originator."""

    def __init__(self, name: str):
        self.name = name
        self.hp = 100
        self.level = 1
        self.inventory = ["sword"]

    def __repr__(self):
        return f"{self.name}(hp={self.hp}, lvl={self.level}, inv={self.inventory})"

    # --- memento API ---
    def save(self) -> "PlayerMemento":
        return PlayerMemento(copy.deepcopy(self.__dict__))

    def restore(self, m: "PlayerMemento") -> None:
        self.__dict__.update(copy.deepcopy(m._state))   # copy again: the memento stays reusable


class PlayerMemento:
    """Opaque by convention: the leading underscore says 'caretakers do not read _state'."""
    __slots__ = ("_state",)

    def __init__(self, state: dict):
        self._state = state


class SaveSlots:
    """Caretaker: stores and returns mementos by name, never opens them."""

    def __init__(self):
        self._slots = {}

    def put(self, slot: str, m: PlayerMemento):
        self._slots[slot] = m

    def get(self, slot: str) -> PlayerMemento:
        return self._slots[slot]


if __name__ == "__main__":
    p = Player("Asha")
    slots = SaveSlots()
    slots.put("before-boss", p.save())

    p.hp, p.level = 12, 2
    p.inventory.append("cursed ring")
    print("after boss :", p)
    p.restore(slots.get("before-boss"))
    print("restored   :", p)

    # The classic bug: a shallow snapshot shares the mutable list with the live object.
    shallow = dict(p.__dict__)            # copies the dict, NOT the list inside it
    p.inventory.append("potion")
    print("shallow inv:", shallow["inventory"], "<- changed too, snapshot was not a snapshot")

    # Checkpoint cost: a full snapshot grows with the state; a delta stays small.
    p.inventory = [f"item{i}" for i in range(10_000)]
    full = len(pickle.dumps(p.save()._state))
    delta = len(pickle.dumps({"hp": (100, 90)}))       # store only what changed
    print(f"full snapshot {full} bytes vs delta {delta} bytes")
$ python memento.py
after boss : Asha(hp=12, lvl=2, inv=['sword', 'cursed ring'])
restored   : Asha(hp=100, lvl=1, inv=['sword'])
shallow inv: ['sword', 'potion'] <- changed too, snapshot was not a snapshot
full snapshot 108980 bytes vs delta 26 bytes

Two lessons in the output. A shallow copy is not a snapshot: the list inside was shared, so the "saved" inventory changed. And snapshots scale with state size: on this machine (Python 3.11) one full save of a 10,000-item inventory pickled to about 109 KB, while a delta recording one changed field was 26 bytes.

// Memento in C++14: transactional rollback of a routing table, like "commit / rollback" on a router CLI.
// Build: g++ -std=c++14 -Wall memento.cpp -o memento
#include <iostream>
#include <map>
#include <stdexcept>
#include <string>
#include <vector>

class RouteTable {                         // Originator
public:
    class Memento {                        // opaque: only RouteTable can read it (friend)
        friend class RouteTable;
        std::map<std::string, std::string> routes_;
        unsigned long version_;
        Memento(std::map<std::string, std::string> r, unsigned long v) : routes_(std::move(r)), version_(v) {}
    public:
        std::size_t entries() const { return routes_.size(); }   // harmless metadata is fine to expose
    };

    void add(const std::string& prefix, const std::string& nexthop) {
        if (nexthop == "0.0.0.0") throw std::invalid_argument("bad next hop for " + prefix);
        routes_[prefix] = nexthop; ++version_;
    }
    void del(const std::string& prefix) { routes_.erase(prefix); ++version_; }

    Memento save() const { return Memento(routes_, version_); }   // value copy of the map
    void restore(const Memento& m) { routes_ = m.routes_; version_ = m.version_; }

    void print(const char* tag) const {
        std::cout << tag << " v" << version_ << ":";
        for (const auto& kv : routes_) std::cout << ' ' << kv.first << "->" << kv.second;
        std::cout << '\n';
    }
private:
    std::map<std::string, std::string> routes_;
    unsigned long version_ = 0;
};

struct Change { char op; std::string prefix, nexthop; };

// Caretaker: applies a batch atomically using a checkpoint.
bool commit(RouteTable& rt, const std::vector<Change>& batch) {
    RouteTable::Memento checkpoint = rt.save();
    try {
        for (const auto& c : batch) {
            if (c.op == '+') rt.add(c.prefix, c.nexthop); else rt.del(c.prefix);
        }
        return true;
    } catch (const std::exception& e) {
        std::cout << "  error: " << e.what() << " -> rollback to checkpoint ("
                  << checkpoint.entries() << " entries)\n";
        rt.restore(checkpoint);
        return false;
    }
}

int main() {
    RouteTable rt;
    rt.add("10.0.0.0/8", "192.168.1.1");
    rt.add("0.0.0.0/0", "192.168.1.254");
    rt.print("start   ");

    std::cout << "commit batch 1 (good):\n";
    commit(rt, {{'+', "172.16.0.0/12", "192.168.1.2"}, {'-', "10.0.0.0/8", ""}});
    rt.print("now     ");

    std::cout << "commit batch 2 (third change is invalid):\n";
    commit(rt, {{'+', "10.1.0.0/16", "192.168.1.3"}, {'-', "0.0.0.0/0", ""}, {'+', "10.2.0.0/16", "0.0.0.0"}});
    rt.print("now     ");          // half-applied changes are gone: default route is back
    return 0;
}

Batch 2 had already added a route and deleted the default route when its third change failed. Restoring the checkpoint undid both, so the table is never left half-configured. friend class RouteTable is what keeps the memento opaque to commit(), the caretaker.
