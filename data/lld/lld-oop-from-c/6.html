The whole program in C, Java and C++

A complete, runnable version of the device example: two "classes" (loopback and a 4-slot ring device), generic code that calls through the interface, a default stop, and reference counting. Read the three tabs side by side and notice what disappears as you move right. All three files are in lld-code/lld-oop-from-c/. They print the same log up to ring drained and stopped; then C and C++ show the refcount release (last reference dropped or destroyed, then lo still alive), and Java shows nothing more, because the garbage collector frees the objects silently. If a piece of Java syntax is new to you, see Reading Java as a C programmer below.

/* lld-oop-from-c: an "object" in C = struct + pointer to a const ops table.
   Build: gcc -std=c99 -Wall -o netdev netdev.c && ./netdev */
#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>
#include <string.h>

#define container_of(ptr, type, member) \
    ((type *)((char *)(ptr) - offsetof(type, member)))

/* ---- a tiny kref: one counter, one release callback ---- */
struct kref { int refcount; };
static void kref_init(struct kref *k) { k->refcount = 1; }
static void kref_get(struct kref *k)  { k->refcount++; }
static int  kref_put(struct kref *k, void (*release)(struct kref *)) {
    if (--k->refcount == 0) { release(k); return 1; }
    return 0;
}

/* ---- the "interface": a table of function pointers ---- */
struct net_dev;
struct net_dev_ops {
    int  (*open)(struct net_dev *dev);
    int  (*xmit)(struct net_dev *dev, const char *pkt);
    void (*stop)(struct net_dev *dev);        /* may be NULL: "use the default" */
};

/* ---- the "base class": common fields + vtable pointer ---- */
struct net_dev {
    char name[16];
    const struct net_dev_ops *ops;            /* the vptr */
    struct kref ref;
    void (*free_fn)(struct net_dev *dev);     /* how to free the containing object */
};

/* generic code: knows only the base type, calls through the table */
static int dev_send(struct net_dev *dev, const char *pkt) {
    return dev->ops->xmit(dev, pkt);          /* dynamic dispatch */
}
static void dev_stop(struct net_dev *dev) {
    if (dev->ops->stop) dev->ops->stop(dev);
    else printf("[%s] default stop\n", dev->name);
}
static void dev_release(struct kref *k) {
    struct net_dev *dev = container_of(k, struct net_dev, ref);
    printf("[%s] last reference dropped, freeing\n", dev->name);
    dev->free_fn(dev);
}

/* ---- "subclass" 1: loopback, adds a counter ---- */
struct loopback_dev {
    struct net_dev base;                      /* embedded base = inheritance of data */
    int looped;
};
static int lo_open(struct net_dev *dev) { printf("[%s] up\n", dev->name); return 0; }
static int lo_xmit(struct net_dev *dev, const char *pkt) {
    struct loopback_dev *lo = container_of(dev, struct loopback_dev, base); /* "downcast" */
    lo->looped++;
    printf("[%s] looped back '%s' (total %d)\n", dev->name, pkt, lo->looped);
    return 0;
}
static const struct net_dev_ops loopback_ops = { .open = lo_open, .xmit = lo_xmit, .stop = NULL };
static void lo_free(struct net_dev *dev) { free(container_of(dev, struct loopback_dev, base)); }

/* ---- "subclass" 2: a ring device with a small TX ring ---- */
#define RING 4
struct ring_dev {
    struct net_dev base;
    int head, tail;
    const char *slots[RING];
};
static int ring_open(struct net_dev *dev) {
    struct ring_dev *r = container_of(dev, struct ring_dev, base);
    r->head = r->tail = 0;
    printf("[%s] ring of %d slots ready\n", dev->name, RING);
    return 0;
}
static int ring_xmit(struct net_dev *dev, const char *pkt) {
    struct ring_dev *r = container_of(dev, struct ring_dev, base);
    if (r->head - r->tail == RING) { printf("[%s] ring full, drop '%s'\n", dev->name, pkt); return -1; }
    r->slots[r->head++ % RING] = pkt;
    printf("[%s] queued '%s' (%d in ring)\n", dev->name, pkt, r->head - r->tail);
    return 0;
}
static void ring_stop(struct net_dev *dev) { printf("[%s] ring drained and stopped\n", dev->name); }
static const struct net_dev_ops ring_ops = { .open = ring_open, .xmit = ring_xmit, .stop = ring_stop };
static void ring_free(struct net_dev *dev) { free(container_of(dev, struct ring_dev, base)); }

/* ---- "constructors" ---- */
static struct net_dev *loopback_new(const char *name) {
    struct loopback_dev *lo = calloc(1, sizeof *lo);
    strncpy(lo->base.name, name, sizeof lo->base.name - 1);
    lo->base.ops = &loopback_ops; lo->base.free_fn = lo_free;
    kref_init(&lo->base.ref);
    return &lo->base;
}
static struct net_dev *ring_new(const char *name) {
    struct ring_dev *r = calloc(1, sizeof *r);
    strncpy(r->base.name, name, sizeof r->base.name - 1);
    r->base.ops = &ring_ops; r->base.free_fn = ring_free;
    kref_init(&r->base.ref);
    return &r->base;
}

int main(void) {
    struct net_dev *devs[2] = { loopback_new("lo"), ring_new("eth0") };
    for (int i = 0; i < 2; i++) devs[i]->ops->open(devs[i]);
    for (int i = 0; i < 2; i++) dev_send(devs[i], "ping");
    for (int i = 0; i < 4; i++) dev_send(devs[1], "data");   /* 5th packet overall drops */

    kref_get(&devs[0]->ref);                 /* a second user holds lo */
    for (int i = 0; i < 2; i++) dev_stop(devs[i]);
    for (int i = 0; i < 2; i++) kref_put(&devs[i]->ref, dev_release);
    printf("lo still alive, refcount=%d\n", devs[0]->ref.refcount);
    kref_put(&devs[0]->ref, dev_release);    /* second user lets go */
    return 0;
}
// lld-oop-from-c: the same design as netdev.c in Java 21.
// Run: java NetDevDemo.java
import java.util.ArrayDeque;
import java.util.List;

public class NetDevDemo {
    static boolean send(NetDevice dev, String pkt) { return dev.xmit(pkt); }  // generic code

    public static void main(String[] args) {
        List<NetDevice> devs = List.of(new LoopbackDevice("lo"), new RingDevice("eth0"));
        devs.forEach(NetDevice::open);           // Type::method = a function pointer to open(), called per item
        devs.forEach(d -> send(d, "ping"));      // d -> ... = anonymous function (like a fn pointer + captured context)
        for (int i = 0; i < 4; i++) send(devs.get(1), "data");
        devs.forEach(NetDevice::stop);
        // No kref_get/kref_put and no "freeing" lines: when nothing references the objects,
        // the garbage collector frees them silently, so the Java log ends at "ring drained and stopped".
    }
}


interface NetDevice {                       // = struct net_dev_ops (the contract)
    String name();
    void open();
    boolean xmit(String pkt);
    default void stop() {                   // = a NULL slot that means "use the default"
        System.out.println("[" + name() + "] default stop");
    }
}

abstract class BaseDevice implements NetDevice {   // = struct net_dev (shared data)
    private final String name;              // private: only this class touches it
    protected BaseDevice(String name) { this.name = name; }
    public final String name() { return name; }
}

final class LoopbackDevice extends BaseDevice {
    private int looped;                      // = the extra field in loopback_dev
    LoopbackDevice(String name) { super(name); }
    public void open() { System.out.println("[" + name() + "] up"); }
    public boolean xmit(String pkt) {        // no container_of: 'this' is already the full object
        looped++;
        System.out.println("[" + name() + "] looped back '" + pkt + "' (total " + looped + ")");
        return true;
    }
}

final class RingDevice extends BaseDevice {
    private static final int RING = 4;
    private final ArrayDeque<String> ring = new ArrayDeque<>(RING);
    RingDevice(String name) { super(name); }
    public void open() { ring.clear(); System.out.println("[" + name() + "] ring of " + RING + " slots ready"); }
    public boolean xmit(String pkt) {
        if (ring.size() == RING) { System.out.println("[" + name() + "] ring full, drop '" + pkt + "'"); return false; }
        ring.addLast(pkt);
        System.out.println("[" + name() + "] queued '" + pkt + "' (" + ring.size() + " in ring)");
        return true;
    }
    @Override public void stop() { System.out.println("[" + name() + "] ring drained and stopped"); }
}
// lld-oop-from-c: the same design in C++14.
// Build: g++ -std=c++14 -Wall -o netdev_cpp netdev.cpp && ./netdev_cpp
#include <iostream>
#include <memory>
#include <deque>
#include <string>
#include <vector>

class NetDevice {                                   // abstract base = ops table + base struct
public:
    explicit NetDevice(std::string name)            // explicit: no silent conversion from a string
        : name_(std::move(name)) {}                 // std::move: hand over the string buffer, no copy
    virtual ~NetDevice() { std::cout << "[" << name_ << "] destroyed\n"; }  // virtual dtor!
    virtual void open() = 0;                        // pure virtual = mandatory slot
    virtual bool xmit(const std::string& pkt) = 0;
    virtual void stop() { std::cout << "[" << name_ << "] default stop\n"; } // default impl
    const std::string& name() const { return name_; }
private:
    std::string name_;
};

class LoopbackDevice : public NetDevice {
public:
    using NetDevice::NetDevice;                     // reuse the base constructor
    void open() override { std::cout << "[" << name() << "] up\n"; }  // override: error if no base slot matches
    bool xmit(const std::string& pkt) override {
        ++looped_;
        std::cout << "[" << name() << "] looped back '" << pkt << "' (total " << looped_ << ")\n";
        return true;
    }
private:
    int looped_ = 0;
};

class RingDevice : public NetDevice {
public:
    using NetDevice::NetDevice;                     // reuse the base constructor
    void open() override { ring_.clear(); std::cout << "[" << name() << "] ring of " << kRing << " slots ready\n"; }
    bool xmit(const std::string& pkt) override {
        if (ring_.size() == kRing) { std::cout << "[" << name() << "] ring full, drop '" << pkt << "'\n"; return false; }
        ring_.push_back(pkt);
        std::cout << "[" << name() << "] queued '" << pkt << "' (" << ring_.size() << " in ring)\n";
        return true;
    }
    void stop() override { std::cout << "[" << name() << "] ring drained and stopped\n"; }
private:
    static const std::size_t kRing = 4;
    std::deque<std::string> ring_;
};

static bool send(NetDevice& dev, const std::string& pkt) { return dev.xmit(pkt); }

int main() {
    std::vector<std::shared_ptr<NetDevice>> devs = {
        std::make_shared<LoopbackDevice>("lo"),     // make_shared: object + refcount block in one go (kref_init)
        std::make_shared<RingDevice>("eth0") };
    for (auto& d : devs) d->open();
    for (auto& d : devs) send(*d, "ping");
    for (int i = 0; i < 4; ++i) send(*devs[1], "data");
    std::shared_ptr<NetDevice> second_user = devs[0];   // kref_get
    for (auto& d : devs) d->stop();
    devs.clear();                                       // kref_put x2: eth0 dies, lo survives
    std::cout << "lo still alive, use_count=" << second_user.use_count() << "\n";
    second_user.reset();                                // last put: lo destroyed
    return 0;
}
