"""Meeting room and calendar scheduler - Python version of the Java design.

Rooms keep a sorted list of bookings (bisect) and one lock each; people keep their own list and lock.
The scheduler locks participants in user-id order, then the room, so two requests can never deadlock.
"""
from __future__ import annotations

import bisect
import itertools
import threading
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Callable, Iterable, Protocol


class NoRoomAvailableError(Exception): pass
class ConflictError(Exception): pass
class NotFoundError(Exception): pass


@dataclass(frozen=True, order=True)
class TimeSlot:
    """Half-open [start, end): 10:00-11:00 does not clash with 11:00-12:00."""
    start: datetime
    end: datetime

    def __post_init__(self):
        if self.end <= self.start:
            raise ValueError(f"end must be after start: {self.start} .. {self.end}")

    def overlaps(self, other: "TimeSlot") -> bool:
        return self.start < other.end and other.start < self.end

    def shift(self, d: timedelta) -> "TimeSlot":
        return TimeSlot(self.start + d, self.end + d)

    def __str__(self):
        return f"{self.start:%Y-%m-%d %H:%M}-{self.end:%H:%M}"


@dataclass(frozen=True)
class User:
    id: str
    name: str


@dataclass(frozen=True)
class Room:
    id: str
    name: str
    capacity: int
    floor: int


class MeetingStatus(Enum):
    SCHEDULED = "SCHEDULED"
    CANCELLED = "CANCELLED"


class Rsvp(Enum):
    PENDING = "PENDING"
    ACCEPTED = "ACCEPTED"
    DECLINED = "DECLINED"


# ---- recurrence (Strategy as a plain function: first slot -> all occurrences)
Recurrence = Callable[[TimeSlot], list]


def no_repeat(first: TimeSlot) -> list:
    return [first]


def every(step: timedelta, count: int) -> Recurrence:
    if not 1 <= count <= 52:
        raise ValueError("count must be 1..52")
    return lambda first: [first.shift(step * i) for i in range(count)]


def daily(count: int) -> Recurrence: return every(timedelta(days=1), count)
def weekly(count: int) -> Recurrence: return every(timedelta(days=7), count)


@dataclass
class Meeting:
    id: str
    title: str
    organizer: User
    participants: tuple
    room: Room
    slots: tuple
    status: MeetingStatus = MeetingStatus.SCHEDULED
    rsvp: dict = field(default_factory=dict)

    def __post_init__(self):
        for u in self.participants:
            self.rsvp[u.id] = Rsvp.ACCEPTED if u == self.organizer else Rsvp.PENDING

    def __str__(self):
        extra = f" (x{len(self.slots)})" if len(self.slots) > 1 else ""
        return f"{self.id} '{self.title}' in {self.room.name} {self.slots[0]}{extra}"


@dataclass(frozen=True)
class Busy:
    slot: TimeSlot
    meeting_id: str


class RoomCalendar:
    """Bookings of one room, sorted by start. Bookings never overlap, so check only the two neighbours."""

    def __init__(self, room: Room):
        self.room = room
        self._starts: list = []       # sorted start times
        self._items: list = []        # Busy, same order
        self.lock = threading.Lock()

    def _free_locked(self, s: TimeSlot) -> bool:
        i = bisect.bisect_right(self._starts, s.start)       # first booking starting after s.start
        if i > 0 and self._items[i - 1].slot.end > s.start:   # the one starting at or before s
            return False
        return i == len(self._items) or self._items[i].slot.start >= s.end

    def is_free(self, s: TimeSlot) -> bool:
        with self.lock:
            return self._free_locked(s)

    def try_reserve_all(self, slots: Iterable[TimeSlot], meeting_id: str) -> bool:
        slots = list(slots)
        with self.lock:
            if not all(self._free_locked(s) for s in slots):
                return False              # all or nothing
            for s in slots:
                i = bisect.bisect_right(self._starts, s.start)
                self._starts.insert(i, s.start)
                self._items.insert(i, Busy(s, meeting_id))
            return True

    def release(self, meeting_id: str) -> None:
        with self.lock:
            keep = [(st, b) for st, b in zip(self._starts, self._items) if b.meeting_id != meeting_id]
            self._starts = [st for st, _ in keep]
            self._items = [b for _, b in keep]

    def on(self, day: datetime) -> list:
        lo, hi = day, day + timedelta(days=1)
        with self.lock:
            return [b for b in self._items if lo <= b.slot.start < hi]


class UserCalendar:
    """A person's meetings. People may double-book if the policy allows, so this is a plain list."""

    def __init__(self, user: User):
        self.user = user
        self.items: list = []
        self.lock = threading.Lock()

    def clashes_locked(self, slots) -> bool:
        return any(b.slot.overlaps(s) for b in self.items for s in slots)

    def between(self, lo: datetime, hi: datetime) -> list:
        with self.lock:
            return [b for b in self.items if b.slot.start < hi and b.slot.end > lo]


# ---- room selection strategies
class RoomSelectionStrategy(Protocol):
    name: str
    def rank(self, rooms: list, attendees: int) -> list: ...


class BestFitCapacity:
    name = "best-fit capacity"

    def rank(self, rooms, attendees):
        return sorted(rooms, key=lambda r: (r.capacity, r.id))


class NearestFloor:
    def __init__(self, floor: int):
        self.floor = floor
        self.name = f"nearest floor to {floor}"

    def rank(self, rooms, attendees):
        return sorted(rooms, key=lambda r: (abs(r.floor - self.floor), r.capacity, r.id))


class ConflictPolicy(Enum):
    ALLOW = "ALLOW"
    REJECT = "REJECT"


class NotificationService:
    """Observer: records (and optionally prints) the messages it would send."""

    def __init__(self, echo: bool = False):
        self.sent: list = []
        self.echo = echo
        self._lock = threading.Lock()

    def _send(self, text: str):
        with self._lock:
            self.sent.append(text)
        if self.echo:
            print("    [notify]", text)

    def on_scheduled(self, m: Meeting):
        for u in m.participants:
            if u != m.organizer:
                self._send(f"to {u.name}: invite {m.id} '{m.title}' {m.slots[0]} in {m.room.name}")

    def on_cancelled(self, m: Meeting):
        for u in m.participants:
            if u != m.organizer:
                self._send(f"to {u.name}: cancelled {m.id}")

    def on_rsvp(self, m: Meeting, who: User, status: Rsvp):
        self._send(f"to {m.organizer.name}: {who.name} {status.value} {m.id}")


class MeetingScheduler:
    def __init__(self, strategy, policy: ConflictPolicy = ConflictPolicy.REJECT):
        self.strategy = strategy
        self.policy = policy
        self._rooms: dict = {}
        self._users: dict = {}
        self._meetings: dict = {}
        self._listeners: list = []
        self._ids = itertools.count(1)
        self._id_lock = threading.Lock()
        self._registry_lock = threading.Lock()

    def add_room(self, r: Room):
        with self._registry_lock:
            if r.id in self._rooms:
                raise ValueError(f"duplicate room {r.id}")
            self._rooms[r.id] = RoomCalendar(r)

    def add_user(self, u: User):
        with self._registry_lock:
            if u.id in self._users:
                raise ValueError(f"duplicate user {u.id}")
            self._users[u.id] = UserCalendar(u)

    def add_listener(self, listener):
        self._listeners.append(listener)

    def _user(self, uid: str) -> UserCalendar:
        cal = self._users.get(uid)
        if cal is None:
            raise NotFoundError(f"no user {uid}")
        return cal

    def _next_id(self) -> str:
        with self._id_lock:
            return f"M{next(self._ids)}"

    def schedule(self, title: str, organizer: str, slot: TimeSlot, invite: Iterable[str] = (),
                 repeat: Recurrence = no_repeat, room_id: str | None = None) -> Meeting:
        if not title.strip():
            raise ValueError("title required")
        ids = list(dict.fromkeys([organizer, *invite]))      # organizer first, no duplicates
        people = [self._user(i) for i in ids]
        occ = sorted(repeat(slot))
        if room_id is not None:
            rc = self._rooms.get(room_id)
            if rc is None:
                raise NotFoundError(f"no room {room_id}")
            if rc.room.capacity < len(people):
                raise NoRoomAvailableError(f"{rc.room.name} seats {rc.room.capacity}, need {len(people)}")
            candidates = [rc.room]
        else:
            candidates = self.strategy.rank([rc.room for rc in self._rooms.values()
                                             if rc.room.capacity >= len(people)], len(people))
        locked = sorted(people, key=lambda c: c.user.id)      # one global lock order: no deadlock
        mid = self._next_id()
        meeting = None
        for c in locked:
            c.lock.acquire()
        try:
            if self.policy is ConflictPolicy.REJECT:
                for c in locked:
                    if c.clashes_locked(occ):
                        raise ConflictError(f"{c.user.name} is busy at {slot}")
            for r in candidates:
                if self._rooms[r.id].try_reserve_all(occ, mid):
                    for c in locked:
                        c.items.extend(Busy(s, mid) for s in occ)
                        c.items.sort(key=lambda b: b.slot)
                    meeting = Meeting(mid, title, people[0].user, tuple(c.user for c in locked), r, tuple(occ))
                    self._meetings[mid] = meeting
                    break
        finally:
            for c in reversed(locked):
                c.lock.release()
        if meeting is None:
            suffix = f" x{len(occ)}" if len(occ) > 1 else ""
            raise NoRoomAvailableError(f"no room for {len(people)} at {slot}{suffix}")
        for listener in self._listeners:       # notify outside the locks
            listener.on_scheduled(meeting)
        return meeting

    def cancel(self, meeting_id: str, by_user: str) -> None:
        m = self._meetings.get(meeting_id)
        if m is None:
            raise NotFoundError(f"no meeting {meeting_id}")
        if m.organizer.id != by_user:
            raise PermissionError(f"only the organizer can cancel {meeting_id}")
        if self._meetings.pop(meeting_id, None) is None:       # a racing cancel lost
            raise NotFoundError(f"already cancelled {meeting_id}")
        m.status = MeetingStatus.CANCELLED
        self._rooms[m.room.id].release(meeting_id)
        for u in m.participants:
            cal = self._users[u.id]
            with cal.lock:
                cal.items = [b for b in cal.items if b.meeting_id != meeting_id]
        for listener in self._listeners:
            listener.on_cancelled(m)

    def respond(self, meeting_id: str, user_id: str, status: Rsvp) -> None:
        m = self._meetings.get(meeting_id)
        if m is None:
            raise NotFoundError(f"no meeting {meeting_id}")
        if user_id not in m.rsvp:
            raise ValueError(f"{user_id} is not invited to {meeting_id}")
        m.rsvp[user_id] = status
        for listener in self._listeners:
            listener.on_rsvp(m, self._user(user_id).user, status)

    def available_rooms(self, slot: TimeSlot, attendees: int) -> list:
        free = [rc.room for rc in self._rooms.values() if rc.room.capacity >= attendees and rc.is_free(slot)]
        return self.strategy.rank(free, attendees)

    def room_schedule(self, room_id: str, day: datetime) -> list:
        return self._rooms[room_id].on(day)

    def find_free_slots(self, user_ids, lo: datetime, hi: datetime, length: timedelta) -> list:
        busy = sorted(b.slot for uid in user_ids for b in self._user(uid).between(lo, hi))
        free, cursor = [], lo
        for b in busy:
            if b.start > cursor and cursor + length <= b.start:
                free.append(TimeSlot(cursor, b.start))
            cursor = max(cursor, b.end)
        if cursor < hi and cursor + length <= hi:
            free.append(TimeSlot(cursor, hi))
        return free

    def get(self, meeting_id: str):
        return self._meetings.get(meeting_id)
"""Demo driver: the same story as Java Main, with simple asserts."""
from datetime import datetime, timedelta

from meeting_scheduler import (BestFitCapacity, ConflictError, MeetingScheduler, NearestFloor, NoRoomAvailableError,
                               NotFoundError, NotificationService, Room, Rsvp, TimeSlot, User, weekly)

DAY = datetime(2026, 11, 24)          # a Tuesday


def at(h, m=0):
    return DAY.replace(hour=h, minute=m)


def slot(h1, m1, h2, m2):
    return TimeSlot(at(h1, m1), at(h2, m2))


def expect(err, fn, what):
    try:
        fn()
    except err as e:
        print("  rejected as expected:", e)
        return
    raise AssertionError(f"expected {err.__name__}: {what}")


def main():
    s = MeetingScheduler(BestFitCapacity())
    notifier = NotificationService(echo=True)
    s.add_listener(notifier)
    s.add_room(Room("R1", "Focus", 4, 1))
    s.add_room(Room("R2", "Huddle", 6, 2))
    s.add_room(Room("R3", "Board", 20, 3))
    for n in ["alice", "bob", "chen", "dev", "eva", "fay"]:
        s.add_user(User(n, n))

    print("== 1. book with", s.strategy.name, "==")
    review = s.schedule("Design review", "alice", slot(10, 0, 11, 0), invite=["bob", "chen"])
    print(" ", review)
    assert review.room.id == "R1"
    standup = s.schedule("Standup", "dev", slot(10, 30, 11, 0), invite=["eva"])
    print(" ", standup)
    assert standup.room.id == "R2"

    print("== 2. overlaps and the half-open boundary ==")
    expect(NoRoomAvailableError, lambda: s.schedule("Sync", "fay", slot(10, 30, 11, 30), room_id="R1"), "overlap")
    retro = s.schedule("Retro", "eva", slot(11, 0, 12, 0), room_id="R1")
    print(" ", retro, " (starts exactly when the review ends: allowed)")
    expect(ConflictError, lambda: s.schedule("Hiring", "dev", slot(10, 15, 10, 45), invite=["chen"]), "chen busy")
    expect(ValueError, lambda: slot(12, 0, 11, 0), "end before start")

    print("== 3. weekly recurring meeting, all-or-nothing per room ==")
    wk3 = DAY + timedelta(weeks=2)
    s.schedule("Offsite prep", "dev", TimeSlot(wk3.replace(hour=15), wk3.replace(hour=16)), room_id="R1")
    expect(NoRoomAvailableError,
           lambda: s.schedule("1:1", "alice", slot(15, 0, 15, 30), invite=["bob"], repeat=weekly(4), room_id="R1"),
           "week 3 clashes")
    assert len(s.room_schedule("R1", DAY)) == 2
    one = s.schedule("1:1", "alice", slot(15, 0, 15, 30), invite=["bob"], repeat=weekly(4))
    print(" ", one, "->", [str(x) for x in one.slots])
    assert one.room.id == "R2" and len(one.slots) == 4

    print("== 4. RSVP ==")
    s.respond(review.id, "bob", Rsvp.ACCEPTED)
    s.respond(review.id, "chen", Rsvp.DECLINED)
    expect(ValueError, lambda: s.respond(review.id, "eva", Rsvp.ACCEPTED), "not invited")

    print("== 5. find a free time for alice + bob, 09:00-16:00, 30 min ==")
    free = s.find_free_slots(["alice", "bob"], at(9), at(16), timedelta(minutes=30))
    print("  free:", [str(x) for x in free])
    assert free == [slot(9, 0, 10, 0), slot(11, 0, 15, 0), slot(15, 30, 16, 0)]

    print("== 6. cancel ==")
    expect(PermissionError, lambda: s.cancel(review.id, "bob"), "not organizer")
    s.cancel(review.id, "alice")
    expect(NotFoundError, lambda: s.cancel(review.id, "alice"), "double cancel")
    print(" ", s.schedule("Sync", "eva", slot(10, 0, 10, 30), room_id="R1"))

    print("== 7. switch strategy at runtime ==")
    s.strategy = NearestFloor(3)
    coffee = s.schedule("Coffee", "chen", slot(13, 0, 13, 30))
    print(f"  {s.strategy.name}: {coffee}")
    assert coffee.room.id == "R3"
    s.strategy = BestFitCapacity()
    print("  rooms free 13:00-13:30 for 2:", [r.name for r in s.available_rooms(slot(13, 0, 13, 30), 2)])
    print("ALL DEMO ASSERTS PASSED; notifications sent:", len(notifier.sent))


if __name__ == "__main__":
    main()
import threading
import unittest
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timedelta

from meeting_scheduler import (BestFitCapacity, ConflictError, ConflictPolicy, MeetingScheduler, NoRoomAvailableError,
                               NotFoundError, NotificationService, Room, TimeSlot, User, daily, weekly)

D = datetime(2026, 11, 24)


def ts(h1, m1, h2, m2, day=D):
    return TimeSlot(day.replace(hour=h1, minute=m1), day.replace(hour=h2, minute=m2))


class SchedulerTest(unittest.TestCase):
    def make(self, *rooms, users=10, policy=ConflictPolicy.REJECT):
        s = MeetingScheduler(BestFitCapacity(), policy)
        for r in rooms or (Room("R1", "Focus", 4, 1), Room("R2", "Huddle", 6, 2)):
            s.add_room(r)
        for i in range(users):
            s.add_user(User(f"u{i}", f"u{i}"))
        return s

    def test_touching_slots_do_not_overlap(self):
        self.assertFalse(ts(10, 0, 11, 0).overlaps(ts(11, 0, 12, 0)))
        self.assertTrue(ts(10, 0, 11, 0).overlaps(ts(10, 59, 11, 30)))
        self.assertTrue(ts(10, 0, 12, 0).overlaps(ts(10, 30, 11, 0)))     # containment

    def test_overlap_in_same_room_fails_other_room_works(self):
        s = self.make()
        s.schedule("a", "u0", ts(10, 0, 11, 0), room_id="R1")
        with self.assertRaises(NoRoomAvailableError):
            s.schedule("b", "u1", ts(10, 30, 11, 30), room_id="R1")
        self.assertEqual(s.schedule("b", "u1", ts(10, 30, 11, 30), room_id="R2").room.id, "R2")

    def test_best_fit_picks_smallest_room(self):
        s = self.make(Room("big", "Big", 20, 1), Room("small", "Small", 3, 1))
        self.assertEqual(s.schedule("a", "u0", ts(9, 0, 10, 0), invite=["u1"]).room.id, "small")
        self.assertEqual(s.schedule("b", "u2", ts(9, 0, 10, 0), invite=["u3", "u4", "u5"]).room.id, "big")

    def test_participant_conflict_policy(self):
        strict = self.make()
        strict.schedule("a", "u0", ts(9, 0, 10, 0), invite=["u1"])
        with self.assertRaises(ConflictError):
            strict.schedule("b", "u2", ts(9, 30, 10, 30), invite=["u1"])
        loose = self.make(policy=ConflictPolicy.ALLOW)
        loose.schedule("a", "u0", ts(9, 0, 10, 0), invite=["u1"])
        self.assertIsNotNone(loose.schedule("b", "u2", ts(9, 30, 10, 30), invite=["u1"]))

    def test_recurring_is_all_or_nothing(self):
        s = self.make(Room("R1", "Only", 4, 1))
        s.schedule("block", "u9", ts(9, 0, 10, 0, D + timedelta(days=2)))
        with self.assertRaises(NoRoomAvailableError):
            s.schedule("daily", "u0", ts(9, 0, 9, 15), repeat=daily(5))
        self.assertEqual(s.room_schedule("R1", D), [])          # nothing half-booked
        self.assertEqual(len(s.schedule("weekly", "u0", ts(9, 0, 9, 15), repeat=weekly(3)).slots), 3)

    def test_cancel_frees_room_and_notifies(self):
        s = self.make()
        n = NotificationService()
        s.add_listener(n)
        m = s.schedule("a", "u0", ts(9, 0, 10, 0), invite=["u1"], room_id="R1")
        with self.assertRaises(PermissionError):
            s.cancel(m.id, "u1")
        s.cancel(m.id, "u0")
        with self.assertRaises(NotFoundError):
            s.cancel(m.id, "u0")
        self.assertTrue(s.available_rooms(ts(9, 0, 10, 0), 2))
        self.assertIn("to u1: cancelled " + m.id, n.sent)

    def test_free_slots_merge_overlapping_busy(self):
        s = self.make(policy=ConflictPolicy.ALLOW)
        s.schedule("a", "u0", ts(9, 0, 10, 0))
        s.schedule("b", "u1", ts(9, 30, 11, 0))
        s.schedule("c", "u1", ts(11, 10, 11, 20))       # gap 11:00-11:10 is too short for 15 min
        free = s.find_free_slots(["u0", "u1"], D.replace(hour=8), D.replace(hour=12), timedelta(minutes=15))
        self.assertEqual(free, [ts(8, 0, 9, 0), ts(11, 20, 12, 0)])

    def test_concurrent_same_room_one_winner(self):
        s = self.make(users=50)
        start = threading.Event()

        def book(i):
            start.wait()
            try:
                return s.schedule("race", f"u{i}", ts(9, 0, 10, 0), room_id="R1")
            except NoRoomAvailableError:
                return None

        with ThreadPoolExecutor(8) as pool:
            futs = [pool.submit(book, i) for i in range(50)]
            start.set()
            won = [f.result() for f in futs if f.result()]
        self.assertEqual(len(won), 1)

    def test_no_deadlock_with_opposite_lock_orders(self):
        s = self.make(Room("R", "R", 4, 1), Room("Q", "Q", 4, 1))

        def book(k):
            a, b = ("u0", "u1") if k % 2 == 0 else ("u1", "u0")
            st = D + timedelta(minutes=5 * k)
            s.schedule("pair", a, TimeSlot(st, st + timedelta(minutes=5)), invite=[b])

        with ThreadPoolExecutor(8) as pool:
            list(pool.map(book, range(100), timeout=10))


if __name__ == "__main__":
    unittest.main()

// Windows (MinGW g++ 6.3): g++ -std=c++14 -Wall -O2 main.cpp -o meeting
// Linux (real locks + a 50-thread race): g++ -std=c++14 -Wall -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o meeting
// Locks: one per user (taken in sorted-name order), then one per room; a registry lock only for lookups.
#include <algorithm>
#include <cstdio>
#include <functional>
#include <map>
#include <memory>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>

// Locking: a real std::mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). MinGW.org g++ 6.3 (win32 thread model) has no
// std::mutex and -pthread fails to link there, so the default build uses a no-op Mutex with the same
// lock()/unlock(): the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
#include <thread>
typedef std::mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

using namespace std;

static string fmt(long t) {
    char b[32];
    snprintf(b, sizeof b, "day%ld %02ld:%02ld", t / 1440, (t % 1440) / 60, t % 60);
    return b;
}
static long at(int day, int h, int m) { return day * 1440L + h * 60 + m; }

struct TimeSlot {                       // half-open [start, end)
    long start, end;
    TimeSlot(long s, long e) : start(s), end(e) {
        if (e <= s) throw invalid_argument("end must be after start");
    }
    bool overlaps(const TimeSlot& o) const { return start < o.end && o.start < end; }
    string str() const { return fmt(start) + "-" + fmt(end).substr(fmt(end).size() - 5); }
};

struct Room { string id, name; int capacity, floor; };

struct NoRoom : runtime_error { using runtime_error::runtime_error; };
struct Conflict : runtime_error { using runtime_error::runtime_error; };

// Bookings in one room never overlap, so compare a new slot only with its two neighbours.
class RoomCalendar {
    map<long, pair<long, string>> byStart_;   // start -> (end, meetingId), guarded by m_
    mutable Mutex m_;
    bool freeLocked(const TimeSlot& s) const {
        auto next = byStart_.upper_bound(s.start);                 // first start > s.start
        if (next != byStart_.begin()) {
            auto prev = std::prev(next);                           // start <= s.start
            if (prev->second.first > s.start) return false;
        }
        return next == byStart_.end() || next->first >= s.end;
    }
public:
    Room room;
    explicit RoomCalendar(Room r) : room(r) {}
    bool isFree(const TimeSlot& s) const { LockGuard g(m_); return freeLocked(s); }
    bool tryReserveAll(const vector<TimeSlot>& slots, const string& id) {
        LockGuard g(m_);                                           // check and claim under one lock
        for (auto& s : slots) if (!freeLocked(s)) return false;    // all or nothing
        for (auto& s : slots) byStart_[s.start] = make_pair(s.end, id);
        return true;
    }
    void release(const string& id) {
        LockGuard g(m_);
        for (auto it = byStart_.begin(); it != byStart_.end();)
            it = (it->second.second == id) ? byStart_.erase(it) : std::next(it);
    }
    size_t size() const { LockGuard g(m_); return byStart_.size(); }
};

struct RoomSelectionStrategy {               // a vtable = an ops table of function pointers
    virtual ~RoomSelectionStrategy() {}
    virtual vector<Room> rank(vector<Room> rooms) const = 0;
    virtual string name() const = 0;
};
struct BestFitCapacity : RoomSelectionStrategy {
    vector<Room> rank(vector<Room> r) const override {
        sort(r.begin(), r.end(), [](const Room& a, const Room& b) {
            return a.capacity != b.capacity ? a.capacity < b.capacity : a.id < b.id; });
        return r;
    }
    string name() const override { return "best-fit capacity"; }
};
struct NearestFloor : RoomSelectionStrategy {
    int floor;
    explicit NearestFloor(int f) : floor(f) {}
    vector<Room> rank(vector<Room> r) const override {
        int f = floor;
        sort(r.begin(), r.end(), [f](const Room& a, const Room& b) {
            int da = abs(a.floor - f), db = abs(b.floor - f);
            return da != db ? da < db : a.capacity < b.capacity; });
        return r;
    }
    string name() const override { return "nearest floor to " + to_string(floor); }
};

using Recurrence = function<vector<TimeSlot>(const TimeSlot&)>;
static Recurrence weekly(int n) {
    return [n](const TimeSlot& f) {
        vector<TimeSlot> v;
        for (int i = 0; i < n; i++) v.push_back(TimeSlot(f.start + i * 7 * 1440L, f.end + i * 7 * 1440L));
        return v;
    };
}
static Recurrence once() { return [](const TimeSlot& f) { return vector<TimeSlot>{f}; }; }

// Takes several user locks in the order given (callers pass sorted names) and releases them in reverse.
struct OrderedLocks {
    vector<Mutex*> ms;
    explicit OrderedLocks(vector<Mutex*> v) : ms(std::move(v)) { for (Mutex* m : ms) m->lock(); }
    ~OrderedLocks() { for (auto it = ms.rbegin(); it != ms.rend(); ++it) (*it)->unlock(); }
};

class MeetingScheduler {
    struct User { Mutex m; vector<pair<TimeSlot, string>> busy; };   // busy guarded by m
    vector<unique_ptr<RoomCalendar>> rooms_;
    map<string, unique_ptr<User>> users_;
    map<string, Meeting> meetings_;
    vector<MeetingListener*> listeners_;
    const RoomSelectionStrategy* strategy_;
    long nextId_ = 0;
    Mutex reg_;                     // guards the maps above, strategy_ and nextId_; held only briefly
    vector<User*> usersSorted(const vector<string>& names) {          // names must be sorted
        LockGuard g(reg_);
        vector<User*> out;
        for (auto& n : names) {
            auto it = users_.find(n);
            if (it == users_.end()) throw invalid_argument("no user " + n);
            out.push_back(it->second.get());
        }
        return out;
    }
    static vector<Mutex*> mutexes(const vector<User*>& us) { vector<Mutex*> v; for (User* u : us) v.push_back(&u->m); return v; }
public:
    explicit MeetingScheduler(const RoomSelectionStrategy* s) : strategy_(s) {}
    void setStrategy(const RoomSelectionStrategy* s) { LockGuard g(reg_); strategy_ = s; }
    void addRoom(Room r) { LockGuard g(reg_); rooms_.emplace_back(new RoomCalendar(r)); }
    void addUser(const string& u) { LockGuard g(reg_); if (!users_.count(u)) users_[u].reset(new User()); }
    void addListener(MeetingListener* l) { LockGuard g(reg_); listeners_.push_back(l); }
    RoomCalendar& room(const string& id) {
        LockGuard g(reg_);
        for (auto& r : rooms_) if (r->room.id == id) return *r;
        throw invalid_argument("no room " + id);
    }

    Meeting schedule(const string& title, const string& org, TimeSlot slot, vector<string> invite,
                     Recurrence rep, const string& roomId = "") {
        set<string> uniq(invite.begin(), invite.end());
        uniq.insert(org);
        vector<string> people(uniq.begin(), uniq.end());     // sorted = the global lock order
        vector<User*> us = usersSorted(people);
        vector<TimeSlot> occ = rep(slot);
        vector<Room> cands;
        const RoomSelectionStrategy* strategy;
        string id;
        vector<MeetingListener*> ls;
        {
            LockGuard g(reg_);
            for (auto& r : rooms_)
                if (r->room.capacity >= (int)people.size() && (roomId.empty() || r->room.id == roomId)) cands.push_back(r->room);
            strategy = strategy_;
            ls = listeners_;
        }
        cands = strategy->rank(cands);
        Meeting m;
        {
            OrderedLocks held(mutexes(us));              // every attendee, in name order: no deadlock
            for (size_t i = 0; i < people.size(); i++)
                for (auto& b : us[i]->busy)
                    for (auto& s : occ)
                        if (b.first.overlaps(s)) throw Conflict(people[i] + " is busy at " + slot.str());
            { LockGuard g(reg_); id = "M" + to_string(++nextId_); }   // reg_ is a leaf lock
            bool booked = false;
            for (auto& r : cands) {
                if (room(r.id).tryReserveAll(occ, id)) { // room lock is taken inside, after the user locks
                    for (User* u : us) for (auto& s : occ) u->busy.push_back(make_pair(s, id));
                    m = Meeting{id, title, org, people, r, occ};
                    booked = true;
                    break;
                }
            }
            if (!booked) throw NoRoom("no room for " + to_string(people.size()) + " at " + slot.str());
        }
        { LockGuard g(reg_); meetings_.insert(make_pair(id, m)); }
        for (auto* l : ls) l->onScheduled(m);           // call out with no lock held
        return m;
    }


    void cancel(const string& id, const string& by) {
        Meeting m;
        vector<MeetingListener*> ls;
        {
            LockGuard g(reg_);
            auto it = meetings_.find(id);
            if (it == meetings_.end()) throw invalid_argument("no meeting " + id);
            if (it->second.organizer != by) throw logic_error("only the organizer can cancel " + id);
            m = it->second;
            meetings_.erase(it);                     // exactly one cancel wins
            ls = listeners_;
        }
        vector<User*> us = usersSorted(m.people);   // m.people is sorted
        {
            OrderedLocks held(mutexes(us));
            room(m.room.id).release(id);
            for (User* u : us)
                u->busy.erase(remove_if(u->busy.begin(), u->busy.end(),
                                        [&](const pair<TimeSlot, string>& b) { return b.second == id; }), u->busy.end());
        }
        for (auto* l : ls) l->onCancelled(m);
    }

    vector<TimeSlot> findFree(const vector<string>& users, long from, long to, long len) {
        vector<string> sorted(users.begin(), users.end());
        sort(sorted.begin(), sorted.end());
        sorted.erase(unique(sorted.begin(), sorted.end()), sorted.end());
        vector<User*> us = usersSorted(sorted);
        vector<TimeSlot> busy;
        {
            OrderedLocks held(mutexes(us));
            for (User* u : us) for (auto& b : u->busy) if (b.first.start < to && b.first.end > from) busy.push_back(b.first);
        }
        sort(busy.begin(), busy.end(), [](const TimeSlot& a, const TimeSlot& b) { return a.start < b.start; });
        vector<TimeSlot> out;
        long cur = from;
        for (auto& b : busy) {
            if (b.start > cur && cur + len <= b.start) out.push_back(TimeSlot(cur, b.start));
            cur = max(cur, b.end);
        }
        if (cur < to && cur + len <= to) out.push_back(TimeSlot(cur, to));
        return out;
    }
};
