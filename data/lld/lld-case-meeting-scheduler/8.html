/** How a meeting repeats. Kept as data + one method, so new rules are new classes (Strategy). */
interface Recurrence {
    List<TimeSlot> occurrences(TimeSlot first);

    Recurrence NONE = first -> List.of(first);   // C: a lambda = a function pointer plus its captured context

    static Recurrence daily(int count) { return every(Duration.ofDays(1), count); }
    static Recurrence weekly(int count) { return every(Duration.ofDays(7), count); }

    private static Recurrence every(Duration step, int count) {
        if (count < 1 || count > 52) throw new IllegalArgumentException("count must be 1..52");
        return first -> {
            List<TimeSlot> out = new ArrayList<>();
            for (int i = 0; i < count; i++) out.add(first.shift(step.multipliedBy(i)));
            return out;
        };
    }
}

/** One busy interval in a calendar, tagged with the meeting that owns it. */
record Busy(TimeSlot slot, String meetingId) {}

/**
 * The bookings of ONE room, sorted by start time. Bookings in a room never overlap, so a new slot only has
 * to be compared with its two neighbours: the booking that starts at or before it, and the next one after.
 * That is O(log n) instead of scanning every booking. The lock makes "check, then insert" one atomic step.
 */
final class RoomCalendar {
    final Room room;
    private final TreeMap<LocalDateTime, Busy> byStart = new TreeMap<>();
    final ReentrantLock lock = new ReentrantLock();

    RoomCalendar(Room room) { this.room = room; }

    private boolean freeLocked(TimeSlot s) {
        Map.Entry<LocalDateTime, Busy> prev = byStart.floorEntry(s.start());   // starts at or before s
        if (prev != null && prev.getValue().slot().end().isAfter(s.start())) return false;
        Map.Entry<LocalDateTime, Busy> next = byStart.higherEntry(s.start());  // starts after s.start
        return next == null || !next.getKey().isBefore(s.end());
    }

    boolean isFree(TimeSlot s) {
        lock.lock();
        try { return freeLocked(s); } finally { lock.unlock(); }   // C: finally runs on every exit path: the goto out_unlock label
    }

    /** All occurrences or none: a weekly meeting that clashes in week 3 books nothing in this room. */
    boolean tryReserveAll(List<TimeSlot> slots, String meetingId) {
        lock.lock();
        try {
            for (TimeSlot s : slots) if (!freeLocked(s)) return false;
            for (TimeSlot s : slots) byStart.put(s.start(), new Busy(s, meetingId));
            return true;
        } finally { lock.unlock(); }
    }

    void release(List<TimeSlot> slots, String meetingId) {
        lock.lock();
        try {
            for (TimeSlot s : slots) byStart.remove(s.start(), new Busy(s, meetingId));
        } finally { lock.unlock(); }
    }

    List<Busy> on(LocalDate day) {
        lock.lock();
        try {
            return new ArrayList<>(byStart.subMap(day.atStartOfDay(), day.plusDays(1).atStartOfDay()).values());
        } finally { lock.unlock(); }
    }
}

/**
 * The meetings of ONE person. Unlike a room, a person MAY have overlapping meetings (if the policy allows),
 * so we keep a plain sorted list. Guarded by its own lock; the scheduler takes several of these locks in a
 * fixed order (by user id) to avoid deadlock.
 */
final class UserCalendar {
    final User user;
    private final List<Busy> items = new ArrayList<>();
    final ReentrantLock lock = new ReentrantLock();

    UserCalendar(User user) { this.user = user; }

    // callers hold lock
    boolean clashesLocked(List<TimeSlot> slots) {
        for (Busy b : items) for (TimeSlot s : slots) if (b.slot().overlaps(s)) return true;
        return false;
    }
    void addLocked(List<TimeSlot> slots, String meetingId) {
        for (TimeSlot s : slots) items.add(new Busy(s, meetingId));
        items.sort((a, b) -> a.slot().compareTo(b.slot()));
    }
    void removeLocked(String meetingId) { items.removeIf(b -> b.meetingId().equals(meetingId)); }

    List<Busy> between(LocalDateTime from, LocalDateTime to) {
        lock.lock();
        try {
            List<Busy> out = new ArrayList<>();
            for (Busy b : items) if (b.slot().start().isBefore(to) && b.slot().end().isAfter(from)) out.add(b);
            return out;
        } finally { lock.unlock(); }
    }
}

interface RoomSelectionStrategy {

/** Smallest room that fits first: keeps the 20-seat board room free for big meetings. */
final class BestFitCapacity implements RoomSelectionStrategy {

        out.sort(Comparator.comparingInt(Room::capacity).thenComparing(Room::id));   // C: Type::method = a pointer to an existing function
        return out;
    }

/** Nearest floor to the organizer first, then smallest. */
final class NearestFloor implements RoomSelectionStrategy {
    private final int floor;
    NearestFloor(int floor) { this.floor = floor; }
    public List<Room> rank(List<Room> rooms, int attendees) {
        List<Room> out = new ArrayList<>(rooms);
        out.sort(Comparator.comparingInt((Room r) -> Math.abs(r.floor() - floor))
                .thenComparingInt(Room::capacity).thenComparing(Room::id));
        return out;
    }
    public String name() { return "nearest floor to " + floor; }
}

/** The service (facade) that callers use. Owns rooms, people, meetings; delegates rules to strategies. */
final class MeetingScheduler {
    private final Map<String, RoomCalendar> rooms = new ConcurrentHashMap<>();
    private final Map<String, UserCalendar> users = new ConcurrentHashMap<>();
    private final Map<String, Meeting> meetings = new ConcurrentHashMap<>();
    private final List<MeetingListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicLong ids = new AtomicLong();
    private volatile RoomSelectionStrategy strategy;
    private final ParticipantConflictPolicy policy;

    MeetingScheduler(RoomSelectionStrategy strategy, ParticipantConflictPolicy policy) {
        this.strategy = strategy; this.policy = policy;
    }

    void addRoom(Room r) { if (rooms.putIfAbsent(r.id(), new RoomCalendar(r)) != null) throw new IllegalArgumentException("duplicate room " + r.id()); }
    void addUser(User u) { if (users.putIfAbsent(u.id(), new UserCalendar(u)) != null) throw new IllegalArgumentException("duplicate user " + u.id()); }
    void addListener(MeetingListener l) { listeners.add(l); }
    void setStrategy(RoomSelectionStrategy s) { strategy = s; }
    String strategyName() { return strategy.name(); }

    private UserCalendar userCal(String id) {
        UserCalendar c = users.get(id);
        if (c == null) throw new NotFoundException("no user " + id);
        return c;
    }

    Meeting schedule(MeetingRequest req) {
        UserCalendar org = userCal(req.organizerId);
        Set<String> everyone = new LinkedHashSet<>();
        everyone.add(req.organizerId); everyone.addAll(req.participantIds);
        List<UserCalendar> people = new ArrayList<>();
        for (String id : everyone) people.add(userCal(id));
        List<TimeSlot> occ = req.recurrence.occurrences(req.slot);

        List<Room> candidates = new ArrayList<>();
        if (req.roomId != null) {
            RoomCalendar rc = rooms.get(req.roomId);
            if (rc == null) throw new NotFoundException("no room " + req.roomId);
            if (rc.room.capacity() < people.size()) throw new NoRoomAvailableException(rc.room.name() + " seats " + rc.room.capacity() + ", need " + people.size());
            candidates.add(rc.room);
        } else {
            for (RoomCalendar rc : rooms.values()) if (rc.room.capacity() >= people.size()) candidates.add(rc.room);
            candidates = strategy.rank(candidates, people.size());
        }

        // Lock every participant's calendar in ONE global order (user id) so two schedulers never deadlock.
        people.sort(Comparator.comparing(c -> c.user.id()));   // C: builds the cmp() function you would pass to qsort
        String id = "M" + ids.incrementAndGet();
        Meeting m = null;
        for (UserCalendar c : people) c.lock.lock();
        try {
            if (policy == ParticipantConflictPolicy.REJECT)
                for (UserCalendar c : people)
                    if (c.clashesLocked(occ)) throw new ConflictException(c.user.name() + " is busy at " + req.slot);
            for (Room r : candidates) {
                if (rooms.get(r.id()).tryReserveAll(occ, id)) {      // room lock taken INSIDE user locks: fixed order
                    Set<User> ps = new LinkedHashSet<>();
                    for (UserCalendar c : people) { c.addLocked(occ, id); ps.add(c.user); }
                    m = new Meeting(id, req.title, org.user, ps, r, occ);
                    meetings.put(id, m);
                    break;
                }
            }
        } finally {
            for (UserCalendar c : people) c.lock.unlock();
        }
        if (m == null) throw new NoRoomAvailableException("no room for " + people.size() + " at " + req.slot + (occ.size() > 1 ? " x" + occ.size() : ""));
        for (MeetingListener l : listeners) l.onScheduled(m);   // notify OUTSIDE the locks
        return m;
    }

    void cancel(String meetingId, String byUserId) {
        Meeting m = meetings.get(meetingId);
        if (m == null) throw new NotFoundException("no meeting " + meetingId);
        if (!m.organizer().id().equals(byUserId)) throw new IllegalStateException("only the organizer can cancel " + meetingId);
        if (!meetings.remove(meetingId, m)) throw new NotFoundException("already cancelled " + meetingId); // one winner
        m.markCancelled();
        rooms.get(m.room().id()).release(m.slots(), meetingId);
        for (User u : m.participants()) {
            UserCalendar c = users.get(u.id());
            c.lock.lock();
            try { c.removeLocked(meetingId); } finally { c.lock.unlock(); }
        }
        for (MeetingListener l : listeners) l.onCancelled(m);
    }

    void respond(String meetingId, String userId, RsvpStatus s) {
        Meeting m = meetings.get(meetingId);
        if (m == null) throw new NotFoundException("no meeting " + meetingId);
        User u = userCal(userId).user;
        m.respond(u, s);
        for (MeetingListener l : listeners) l.onRsvp(m, u, s);
    }

    List<Room> availableRooms(TimeSlot slot, int attendees) {
        List<Room> out = new ArrayList<>();
        for (RoomCalendar rc : rooms.values()) if (rc.room.capacity() >= attendees && rc.isFree(slot)) out.add(rc.room);
        return strategy.rank(out, attendees);
    }

    List<Busy> roomSchedule(String roomId, LocalDate day) { return rooms.get(roomId).on(day); }

    /** "Find a time": merge everyone's busy intervals inside the window, return gaps >= duration. */
    List<TimeSlot> findFreeSlots(List<String> userIds, LocalDateTime from, LocalDateTime to, Duration length) {
        List<TimeSlot> busy = new ArrayList<>();
        for (String id : userIds) for (Busy b : userCal(id).between(from, to)) busy.add(b.slot());
        busy.sort(Comparator.naturalOrder());
        List<TimeSlot> free = new ArrayList<>();
        LocalDateTime cursor = from;
        for (TimeSlot b : busy) {
            if (b.start().isAfter(cursor) && !cursor.plus(length).isAfter(b.start())) free.add(new TimeSlot(cursor, b.start()));
            if (b.end().isAfter(cursor)) cursor = b.end();
        }
        if (cursor.isBefore(to) && !cursor.plus(length).isAfter(to)) free.add(new TimeSlot(cursor, to));
        return free;
    }

    Meeting get(String id) { return meetings.get(id); }
    int meetingCount() { return meetings.size(); }
}

            return;
        }
        throw new AssertionError("FAILED: expected " + type.getSimpleName() + " for " + what);
    }
    static final LocalDate DAY = LocalDate.of(2026, 11, 24);              // a Tuesday
    static LocalDateTime at(int h, int m) { return DAY.atTime(h, m); }
    static TimeSlot slot(int h1, int m1, int h2, int m2) { return new TimeSlot(at(h1, m1), at(h2, m2)); }

    public static void main(String[] args) throws Exception {

        check(any.size() == 3 && roomsUsed.size() == 3, "40 tries, 3 rooms -> 3 bookings in 3 different rooms");
        System.out.println("  40 tries on any room -> " + any.size() + " booked in " + roomsUsed);

        System.out.println("== 9. no deadlock: 200 meetings with participants locked in opposite orders ==");
        ExecutorService pool = Executors.newFixedThreadPool(8);
        CountDownLatch done = new CountDownLatch(200);
        for (int i = 0; i < 200; i++) {
            final int k = i;
            pool.submit(() -> {

    interface Req { MeetingRequest make(int i); }

    static List<Meeting> race(MeetingScheduler s, int n, Req req) throws Exception {
        ExecutorService pool = Executors.newFixedThreadPool(8);
        CountDownLatch start = new CountDownLatch(1);
        ConcurrentLinkedQueue<Meeting> ok = new ConcurrentLinkedQueue<>();
        List<java.util.concurrent.Future<?>> fs = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            final int k = i;
            fs.add(pool.submit(() -> {
                try { start.await(); ok.add(s.schedule(req.make(k))); }
                catch (NoRoomAvailableException e) { /* lost the race */ }
                catch (InterruptedException e) { Thread.currentThread().interrupt(); }
            }));
        }
        start.countDown();
        for (var f : fs) f.get();
        pool.shutdown();
        return new ArrayList<>(ok);
    }
