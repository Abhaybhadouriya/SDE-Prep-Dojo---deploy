APIs and interfaces

Write these before any method body; they are what the interviewer grades against.

// service API
Meeting schedule(MeetingRequest req);       // throws NoRoomAvailableException, ConflictException, NotFoundException
void    cancel(String meetingId, String byUserId);     // organizer only; IllegalStateException otherwise
void    respond(String meetingId, String userId, RsvpStatus status);
List<Room>     availableRooms(TimeSlot slot, int attendees);
List<Busy>     roomSchedule(String roomId, LocalDate day);
List<TimeSlot> findFreeSlots(List<String> userIds, LocalDateTime from, LocalDateTime to, Duration length);
void setStrategy(RoomSelectionStrategy s);   // runtime switch
void addListener(MeetingListener l);

// building a request
MeetingRequest.builder("1:1", "alice", slot).invite("bob").repeat(Recurrence.weekly(4)).inRoom("R1").build();

// extension points
interface RoomSelectionStrategy { List<Room> rank(List<Room> bigEnough, int attendees); }
interface Recurrence            { List<TimeSlot> occurrences(TimeSlot first); }
interface MeetingListener       { void onScheduled(Meeting m); void onCancelled(Meeting m); void onRsvp(Meeting m, User u, RsvpStatus s); }
