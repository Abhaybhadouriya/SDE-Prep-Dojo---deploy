// Builder demo: an immutable Packet built step by step, validated once in build().
// Run: javac -d out PacketDemo.java && java -cp out PacketDemo
final class Packet {                                   // the PRODUCT: immutable, no setters
    private final String srcIp, dstIp;
    private final int protocol, ttl, dscp;
    private final Integer vlanId;                      // null = untagged
    private final byte[] payload;

    private Packet(Builder b) {                        // only the builder can call this
        srcIp = b.srcIp; dstIp = b.dstIp; protocol = b.protocol;
        ttl = b.ttl; dscp = b.dscp; vlanId = b.vlanId;
        payload = b.payload.clone();                   // defensive copy: builder can't mutate us later
    }

    static Builder to(String dstIp) { return new Builder(dstIp); }   // required field up front
    char payloadAt(int i) { return (char) payload[i]; }

    @Override public String toString() {
        return String.format("Packet[%s -> %s proto=%d ttl=%d dscp=%d vlan=%s bytes=%d]",
            srcIp, dstIp, protocol, ttl, dscp, vlanId == null ? "-" : vlanId, payload.length);
    }

    static final class Builder {                       // the BUILDER: mutable, short-lived
        private final String dstIp;
        private String srcIp = "0.0.0.0";
        private int protocol = 17, ttl = 64, dscp = 0; // sensible defaults (UDP, TTL 64, best effort)
        private Integer vlanId;
        private byte[] payload = new byte[0];

        private Builder(String dstIp) { this.dstIp = dstIp; }
        Builder from(String ip)       { this.srcIp = ip; return this; }
        Builder protocol(int p)       { this.protocol = p; return this; }
        Builder ttl(int t)            { this.ttl = t; return this; }
        Builder dscp(int d)           { this.dscp = d; return this; }
        Builder vlan(int id)          { this.vlanId = id; return this; }
        Builder payload(byte[] data)  { this.payload = data.clone(); return this; }

        // <-- the pattern: validate once, then construct the immutable product
        Packet build() {                               // ALL validation in one place
            if (dstIp == null || dstIp.isEmpty()) throw new IllegalStateException("dst required");
            if (ttl < 1 || ttl > 255)             throw new IllegalStateException("ttl 1..255");
            if (dscp < 0 || dscp > 63)            throw new IllegalStateException("dscp 0..63");
            if (vlanId != null && (vlanId < 1 || vlanId > 4094)) throw new IllegalStateException("vlan 1..4094");
            if (payload.length > 1480)            throw new IllegalStateException("payload > 1480 (MTU 1500 - IP 20)");
            return new Packet(this);
        }
    }
}

final class PacketRecipes {                            // the DIRECTOR: fixed recipes of steps
    static Packet ping(String dst) {
        return Packet.to(dst).protocol(1).payload(new byte[56]).build();
    }
    static Packet voice(String src, String dst, byte[] rtp) {
        return Packet.to(dst).from(src).protocol(17).dscp(46).vlan(20).payload(rtp).build();
    }
}

public class PacketDemo {
    public static void main(String[] args) {
        Packet custom = Packet.to("10.0.0.2").from("10.0.0.1")
                              .protocol(6).ttl(32).vlan(100)
                              .payload("hello".getBytes())
                              .build();
        System.out.println("custom : " + custom);
        System.out.println("ping   : " + PacketRecipes.ping("8.8.8.8"));
        System.out.println("voice  : " + PacketRecipes.voice("10.0.0.5", "10.0.0.9", new byte[160]));

        try {
            Packet.to("10.0.0.2").vlan(5000).build();
        } catch (IllegalStateException e) {
            System.out.println("rejected: " + e.getMessage());
        }

        byte[] buf = "abc".getBytes();
        Packet p = Packet.to("10.0.0.3").payload(buf).build();
        buf[0] = 'X';                                  // caller changes its array afterwards...
        System.out.println("immutable: caller has '" + (char) buf[0] + "', packet still has '" + p.payloadAt(0) + "'");
    }
}
"""Builder demo: an immutable Packet built step by step, validated once in build().
Run: python packet_demo.py
"""
from dataclasses import dataclass
from typing import Optional


@dataclass(frozen=True)                     # the PRODUCT: immutable
class Packet:
    dst_ip: str
    src_ip: str = "0.0.0.0"
    protocol: int = 17                      # UDP
    ttl: int = 64
    dscp: int = 0
    vlan_id: Optional[int] = None
    payload: bytes = b""                    # bytes is immutable, no defensive copy needed


class PacketBuilder:                        # the BUILDER: mutable, short-lived
    def __init__(self, dst_ip: str):        # required field up front
        self._f = {"dst_ip": dst_ip}

    def src(self, ip):        self._f["src_ip"] = ip; return self
    def protocol(self, p):    self._f["protocol"] = p; return self
    def ttl(self, t):         self._f["ttl"] = t; return self
    def dscp(self, d):        self._f["dscp"] = d; return self
    def vlan(self, vid):      self._f["vlan_id"] = vid; return self
    def payload(self, data):  self._f["payload"] = bytes(data); return self

    def build(self) -> Packet:               # ALL validation in one place
        p = Packet(**self._f)
        if not p.dst_ip:
            raise ValueError("dst required")
        if not 1 <= p.ttl <= 255:
            raise ValueError("ttl 1..255")
        if not 0 <= p.dscp <= 63:
            raise ValueError("dscp 0..63")
        if p.vlan_id is not None and not 1 <= p.vlan_id <= 4094:
            raise ValueError("vlan 1..4094")
        if len(p.payload) > 1480:
            raise ValueError("payload > 1480 (MTU 1500 - IP 20)")
        return p


class PacketRecipes:                        # the DIRECTOR: fixed recipes
    @staticmethod
    def ping(dst):
        return PacketBuilder(dst).protocol(1).payload(bytes(56)).build()

    @staticmethod
    def voice(src, dst, rtp):
        return PacketBuilder(dst).src(src).protocol(17).dscp(46).vlan(20).payload(rtp).build()


def show(p: Packet) -> str:
    vlan = "-" if p.vlan_id is None else p.vlan_id
    return (f"Packet[{p.src_ip} -> {p.dst_ip} proto={p.protocol} ttl={p.ttl} "
            f"dscp={p.dscp} vlan={vlan} bytes={len(p.payload)}]")


if __name__ == "__main__":
    custom = (PacketBuilder("10.0.0.2").src("10.0.0.1")
              .protocol(6).ttl(32).vlan(100)
              .payload(b"hello")
              .build())
    print("custom :", show(custom))
    print("ping   :", show(PacketRecipes.ping("8.8.8.8")))
    print("voice  :", show(PacketRecipes.voice("10.0.0.5", "10.0.0.9", bytes(160))))
    try:
        PacketBuilder("10.0.0.2").vlan(5000).build()
    except ValueError as e:
        print("rejected:", e)

    # The Pythonic shortcut: keyword arguments already solve "telescoping constructors".
    quick = Packet("10.0.0.3", ttl=16)
    print("kwargs :", show(quick))
    try:
        quick.ttl = 1                        # frozen dataclass refuses mutation
    except AttributeError as e:
        print("immutable:", type(e).__name__)
// Builder demo (C++14): an immutable Packet built step by step, validated once in build().
// Build: g++ -std=c++14 -Wall -o packet_demo packet_demo.cpp && ./packet_demo
#include <cstdint>
#include <iostream>
#include <stdexcept>
#include <string>
#include <vector>

class Packet {                                        // PRODUCT: const members, no setters
public:
    class Builder;
    const std::string src, dst;
    const int protocol, ttl, dscp, vlan;              // vlan 0 = untagged
    const std::vector<std::uint8_t> payload;

    void print(const char* tag) const {
        std::cout << tag << "Packet[" << src << " -> " << dst << " proto=" << protocol
                  << " ttl=" << ttl << " dscp=" << dscp << " vlan="
                  << (vlan ? std::to_string(vlan) : "-") << " bytes=" << payload.size() << "]\n";
    }
private:
    Packet(std::string s, std::string d, int p, int t, int ds, int v, std::vector<std::uint8_t> pl)
        : src(std::move(s)), dst(std::move(d)), protocol(p), ttl(t), dscp(ds), vlan(v), payload(std::move(pl)) {}
};

class Packet::Builder {                               // BUILDER: mutable, short-lived
    std::string dst_, src_ = "0.0.0.0";
    int protocol_ = 17, ttl_ = 64, dscp_ = 0, vlan_ = 0;
    std::vector<std::uint8_t> payload_;
public:
    explicit Builder(std::string dst) : dst_(std::move(dst)) {}   // required field up front
    Builder& from(std::string ip)    { src_ = std::move(ip); return *this; }
    Builder& protocol(int p)         { protocol_ = p; return *this; }
    Builder& ttl(int t)              { ttl_ = t; return *this; }
    Builder& dscp(int d)             { dscp_ = d; return *this; }
    Builder& vlan(int id)            { vlan_ = id; return *this; }
    Builder& payload(std::vector<std::uint8_t> d) { payload_ = std::move(d); return *this; }

    Packet build() const {                            // ALL validation in one place
        if (dst_.empty())                  throw std::invalid_argument("dst required");
        if (ttl_ < 1 || ttl_ > 255)        throw std::invalid_argument("ttl 1..255");
        if (dscp_ < 0 || dscp_ > 63)       throw std::invalid_argument("dscp 0..63");
        if (vlan_ < 0 || vlan_ > 4094)     throw std::invalid_argument("vlan 1..4094");
        if (payload_.size() > 1480)        throw std::invalid_argument("payload > 1480 (MTU 1500 - IP 20)");
        return Packet(src_, dst_, protocol_, ttl_, dscp_, vlan_, payload_);
    }
};

struct PacketRecipes {                                // DIRECTOR: fixed recipes
    static Packet ping(const std::string& dst) {
        return Packet::Builder(dst).protocol(1).payload(std::vector<std::uint8_t>(56)).build();
    }
    static Packet voice(const std::string& src, const std::string& dst, std::vector<std::uint8_t> rtp) {
        return Packet::Builder(dst).from(src).protocol(17).dscp(46).vlan(20).payload(std::move(rtp)).build();
    }
};

int main() {
    Packet custom = Packet::Builder("10.0.0.2").from("10.0.0.1")
                        .protocol(6).ttl(32).vlan(100)
                        .payload({'h', 'e', 'l', 'l', 'o'})
                        .build();
    custom.print("custom : ");
    PacketRecipes::ping("8.8.8.8").print("ping   : ");
    PacketRecipes::voice("10.0.0.5", "10.0.0.9", std::vector<std::uint8_t>(160)).print("voice  : ");
    try {
        Packet::Builder("10.0.0.2").vlan(5000).build();
    } catch (const std::invalid_argument& e) {
        std::cout << "rejected: " << e.what() << "\n";
    }
    return 0;
}

  • Required field first. Packet.to(dst) makes it impossible to forget the destination. For several required fields, a step builder (each step returns a different interface, so the compiler forces the order) is the strict version.
  • Defensive copies. Arrays and lists are copied into the builder and again into the product. Without that, "immutable" is a lie.
  • Python note. Keyword arguments and @dataclass(frozen=True) already solve telescoping constructors, so many Python programmers never write a builder. A builder still earns its place when construction is multi-step, conditional, or needs validation across fields.
  • C++ note. Setters return Builder&; the product has const members and a private constructor, and Packet::Builder can reach it because a nested class has access to its enclosing class's private members.
