import java.time.LocalDate;
import java.time.YearMonth;
import java.util.*;
import java.util.concurrent.*;

// ---------- model ----------
/** Immutable running total for one agent (in one period). Never store the average itself. */
final class Stats {
    static final Stats EMPTY = new Stats(0, 0);
    final long sum; final long count;
    Stats(long sum, long count) { this.sum = sum; this.count = count; }
    Stats plus(int stars) { return new Stats(sum + stars, count + 1); }
}

/** One row of a ranking: agent + totals. average() is derived, exact comparison uses cross-multiplication. */
final class AgentScore {
    final String agentId; final long sum; final long count;
    AgentScore(String agentId, Stats s) { this.agentId = agentId; this.sum = s.sum; this.count = s.count; }
    /** Average rounded half-up to 2 decimals, as text ("4.67"). Integer maths, no floating drift. */
    String average() {
        long hundredths = (sum * 200 + count) / (2 * count);
        return hundredths / 100 + "." + String.format("%02d", hundredths % 100);
    }
    @Override public String toString() { return agentId + " " + average() + " (" + count + ")"; }
}

class RatingException extends IllegalArgumentException {
    RatingException(String msg) { super(msg); }
}

// ---------- ranking policy (Strategy as a Comparator) ----------
final class Rankings {
    /** Highest average first, compared exactly: a.sum/a.count vs b.sum/b.count  ==  a.sum*b.count vs b.sum*a.count. */
    static final Comparator<AgentScore> BY_AVERAGE =
            (a, b) -> Long.compare(b.sum * a.count, a.sum * b.count);
    /** Default tie-break: more ratings first (more evidence), then agent id A-Z (deterministic output). */
    static final Comparator<AgentScore> DEFAULT = BY_AVERAGE
            .thenComparing(Comparator.comparingLong((AgentScore s) -> s.count).reversed())
            .thenComparing(s -> s.agentId);
    /** Alternative a product manager may want: ties broken alphabetically only. */
    static final Comparator<AgentScore> ALPHABETICAL_TIES = BY_AVERAGE.thenComparing(s -> s.agentId);
    /** Follow-up: Bayesian (damped) average so one lucky 5-star does not top the chart. Same Strategy seam.
     *  The prior is an exact fraction priorSum / priorCount (3.8 stars worth 10 ratings = 38 / 10), not a long
     *  mean that would truncate 3.8 to 3. Score = (sum + priorSum) / (count + priorCount), compared by
     *  cross-multiplication. Overflow: sum <= 5 * count, so a product is about 5 * n^2 and stays inside a long
     *  up to roughly 10^9 ratings per agent; multiplyExact throws instead of wrapping beyond that. */
    static Comparator<AgentScore> bayesian(long priorSum, long priorCount) {
        if (priorCount <= 0 || priorSum < 0) throw new IllegalArgumentException("prior must be a positive weight");
        return (a, b) -> Long.compare(Math.multiplyExact(b.sum + priorSum, a.count + priorCount),
                                      Math.multiplyExact(a.sum + priorSum, b.count + priorCount));
    }
}

// ---------- export formats (Strategy) ----------
interface Exporter { String export(List<AgentScore> rows); }

final class CsvExporter implements Exporter {
    public String export(List<AgentScore> rows) {
        StringBuilder sb = new StringBuilder("agent,average,ratings\n");
        for (AgentScore r : rows) sb.append(csv(r.agentId)).append(',').append(r.average()).append(',').append(r.count).append('\n');
        return sb.toString();
    }
    private static String csv(String v) {                 // quote fields with comma / quote / newline
        return v.matches(".*[\",\\n].*") ? "\"" + v.replace("\"", "\"\"") + "\"" : v;
    }
}

final class JsonExporter implements Exporter {
    public String export(List<AgentScore> rows) {
        StringJoiner j = new StringJoiner(",", "[", "]");
        for (AgentScore r : rows)
            j.add("{\"agent\":\"" + r.agentId.replace("\\", "\\\\").replace("\"", "\\\"") + "\",\"average\":" + r.average() + ",\"ratings\":" + r.count + "}");
        return j.toString();
    }
}

// ---------- service ----------
final class RatingService {
    private final ConcurrentHashMap<String, Stats> allTime = new ConcurrentHashMap<>();
    private final ConcurrentHashMap<YearMonth, ConcurrentHashMap<String, Stats>> byMonth = new ConcurrentHashMap<>();
    private final Comparator<AgentScore> ranking;

    RatingService(Comparator<AgentScore> ranking) { this.ranking = ranking; }
    RatingService() { this(Rankings.DEFAULT); }

    /** Stage 1 + 3: record one customer rating. compute() makes each per-agent update atomic. */
    void rate(String agentId, int stars, LocalDate date) {
        if (agentId == null || agentId.isBlank()) throw new RatingException("agent id required");
        if (stars < 1 || stars > 5) throw new RatingException("stars must be 1..5, got " + stars);
        if (date == null) throw new RatingException("date required");
        allTime.compute(agentId, (k, s) -> (s == null ? Stats.EMPTY : s).plus(stars));
        byMonth.computeIfAbsent(YearMonth.from(date), m -> new ConcurrentHashMap<>())
               .compute(agentId, (k, s) -> (s == null ? Stats.EMPTY : s).plus(stars));
    }

    /** Stage 1 + 2: all agents with their average, highest first, ties broken by the policy. */
    List<AgentScore> averages() { return rank(allTime); }

    /** Stage 3: the ranking for one month (empty list when nobody was rated that month). */
    List<AgentScore> bestOfMonth(YearMonth month) {
        Map<String, Stats> m = byMonth.get(month);
        return m == null ? Collections.emptyList() : rank(m);
    }

    /** Stage 3b: the winners of a month; several agents when they tie exactly on average. */
    List<String> topAgents(YearMonth month) {
        List<AgentScore> r = bestOfMonth(month);
        List<String> out = new ArrayList<>();
        for (AgentScore s : r) if (Rankings.BY_AVERAGE.compare(s, r.get(0)) == 0) out.add(s.agentId);
        return out;
    }

    /** Stage 4: export any ranking in any format. */
    String export(List<AgentScore> rows, Exporter exporter) { return exporter.export(rows); }

    private List<AgentScore> rank(Map<String, Stats> m) {
        List<AgentScore> out = new ArrayList<>();
        for (Map.Entry<String, Stats> e : m.entrySet()) out.add(new AgentScore(e.getKey(), e.getValue()));
        out.sort(ranking);
        return out;
    }
}

// ---------- demo driver: each stage is a "test" like the Atlassian round ----------
public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }
    static LocalDate d(String s) { return LocalDate.parse(s); }

    public static void main(String[] args) throws Exception {
        // Stage 1: accept ratings, show averages highest first
        RatingService s = new RatingService();
        s.rate("alice", 5, d("2025-01-10"));
        s.rate("alice", 4, d("2025-01-20"));
        s.rate("bob", 3, d("2025-01-11"));
        s.rate("carol", 5, d("2025-02-02"));
        s.rate("carol", 4, d("2025-02-03"));
        s.rate("carol", 5, d("2025-02-04"));
        System.out.println("stage 1 averages: " + s.averages());
        check(s.averages().get(0).agentId.equals("carol"), "carol 4.67 first");
        check(s.averages().get(1).average().equals("4.50"), "alice 4.50, not 4 (integer-division bug)");

        // Stage 2: ties. dave 4.50 with 4 ratings ties alice 4.50 with 2 -> more ratings first
        s.rate("dave", 5, d("2025-02-10")); s.rate("dave", 4, d("2025-02-11"));
        s.rate("dave", 5, d("2025-02-12")); s.rate("dave", 4, d("2025-02-13"));
        System.out.println("stage 2 averages: " + s.averages());
        check(s.averages().get(1).agentId.equals("dave") && s.averages().get(2).agentId.equals("alice"), "tie: dave (4 ratings) before alice (2)");
        RatingService alpha = new RatingService(Rankings.ALPHABETICAL_TIES);
        alpha.rate("zed", 4, d("2025-01-01")); alpha.rate("amy", 4, d("2025-01-01"));
        check(alpha.averages().get(0).agentId.equals("amy"), "alphabetical tie policy");

        // Follow-up: one lucky 5-star vs a hundred ratings averaging 4.90; prior 3.8 stars worth 10 ratings
        RatingService damped = new RatingService(Rankings.bayesian(38, 10).thenComparing(x -> x.agentId));
        damped.rate("lucky", 5, d("2025-01-01"));
        for (int k = 0; k < 100; k++) damped.rate("steady", k < 90 ? 5 : 4, d("2025-01-01"));
        System.out.println("bayesian (prior 3.8 x 10): " + damped.averages());
        check(damped.averages().get(0).agentId.equals("steady"), "damped average ranks 100 ratings of 4.90 above one 5");

        // Stage 3: best agents per month
        System.out.println("stage 3 Jan 2025: " + s.bestOfMonth(YearMonth.of(2025, 1)));
        System.out.println("stage 3 Feb 2025: " + s.bestOfMonth(YearMonth.of(2025, 2)));
        check(s.bestOfMonth(YearMonth.of(2025, 1)).get(0).agentId.equals("alice"), "alice best in Jan");
        check(s.topAgents(YearMonth.of(2025, 2)).equals(Arrays.asList("carol")), "carol tops Feb (4.67 > 4.50)");
        check(s.bestOfMonth(YearMonth.of(2025, 3)).isEmpty(), "no ratings in March");

        // Stage 4: export
        String csv = s.export(s.averages(), new CsvExporter());
        String json = s.export(s.bestOfMonth(YearMonth.of(2025, 1)), new JsonExporter());
        System.out.print("stage 4 csv:\n" + csv);
        System.out.println("stage 4 json Jan: " + json);
        check(csv.startsWith("agent,average,ratings\ncarol,4.67,3\n"), "csv header and first row");
        check(json.equals("[{\"agent\":\"alice\",\"average\":4.50,\"ratings\":2},{\"agent\":\"bob\",\"average\":3.00,\"ratings\":1}]"), "json Jan");

        // Validation
        for (int bad : new int[]{0, 6}) {
            try { s.rate("bob", bad, d("2025-01-01")); check(false, "stars " + bad); }
            catch (RatingException e) { System.out.println("rejected: " + e.getMessage()); passed++; }
        }

        // Stage 5: concurrency - 8 threads x 1000 ratings, no lost updates
        RatingService c = new RatingService();
        ExecutorService pool = Executors.newFixedThreadPool(8);
        List<Future<?>> fs = new ArrayList<>();
        for (int t = 0; t < 8; t++) {
            final int stars = t % 5 + 1;
            fs.add(pool.submit(() -> { for (int k = 0; k < 1000; k++) c.rate("agent" + (k % 3), stars, d("2025-05-01")); }));
        }
        for (Future<?> f : fs) f.get();
        pool.shutdown();
        long total = 0;
        for (AgentScore a : c.averages()) total += a.count;
        System.out.println("stage 5 concurrent: " + c.averages());
        check(total == 8000, "8000 ratings counted");
        check(c.bestOfMonth(YearMonth.of(2025, 5)).stream().mapToLong(a -> a.count).sum() == 8000, "monthly too");

        System.out.println("All " + passed + " checks passed");
    }
}
"""Agent Rating System (Atlassian code design) - grows stage by stage, thread safe."""
from __future__ import annotations

import json
import threading
from datetime import date
from functools import cmp_to_key
from typing import Callable, Dict, List, NamedTuple, Optional, Tuple


class RatingError(ValueError):
    pass


class Stats(NamedTuple):  # immutable running total; the average is always derived
    total: int = 0
    count: int = 0

    def plus(self, stars: int) -> "Stats":
        return Stats(self.total + stars, self.count + 1)


class AgentScore(NamedTuple):
    agent: str
    total: int
    count: int

    def average(self) -> str:
        """Rounded half-up to 2 decimals with integer maths (no float drift)."""
        h = (self.total * 200 + self.count) // (2 * self.count)
        return f"{h // 100}.{h % 100:02d}"

    def __repr__(self) -> str:
        return f"{self.agent} {self.average()} ({self.count})"


# ---------- ranking policies (Strategy as a comparator) ----------
def by_average(a: AgentScore, b: AgentScore) -> int:
    """Exact: a.total/a.count vs b.total/b.count  ==  a.total*b.count vs b.total*a.count (highest first)."""
    left, right = b.total * a.count, a.total * b.count
    return (left > right) - (left < right)


def default_ranking(a: AgentScore, b: AgentScore) -> int:
    """Highest average, then more ratings, then agent id A-Z."""
    return by_average(a, b) or (b.count - a.count) or ((a.agent > b.agent) - (a.agent < b.agent))


def alphabetical_ties(a: AgentScore, b: AgentScore) -> int:
    return by_average(a, b) or ((a.agent > b.agent) - (a.agent < b.agent))


# ---------- exporters (Strategy) ----------
def to_csv(rows: List[AgentScore]) -> str:
    def cell(v: str) -> str:
        return '"' + v.replace('"', '""') + '"' if any(c in v for c in ',"\n') else v
    return "agent,average,ratings\n" + "".join(f"{cell(r.agent)},{r.average()},{r.count}\n" for r in rows)


def to_json(rows: List[AgentScore]) -> str:
    # average is emitted as a fixed 2-decimal number, so build it by hand around json.dumps for the name
    return "[" + ",".join(
        '{"agent":' + json.dumps(r.agent) + ',"average":' + r.average() + ',"ratings":' + str(r.count) + "}"
        for r in rows) + "]"


Comparator = Callable[[AgentScore, AgentScore], int]
Month = Tuple[int, int]


class RatingService:
    def __init__(self, ranking: Comparator = default_ranking):
        self._all: Dict[str, Stats] = {}
        self._month: Dict[Month, Dict[str, Stats]] = {}
        self._ranking = ranking
        self._lock = threading.Lock()  # dict read-modify-write is not atomic across threads

    def rate(self, agent: str, stars: int, on: date) -> None:
        if not agent:
            raise RatingError("agent id required")
        if not isinstance(stars, int) or not 1 <= stars <= 5:
            raise RatingError(f"stars must be 1..5, got {stars}")
        if on is None:
            raise RatingError("date required")
        key = (on.year, on.month)
        with self._lock:
            self._all[agent] = self._all.get(agent, Stats()).plus(stars)
            m = self._month.setdefault(key, {})
            m[agent] = m.get(agent, Stats()).plus(stars)

    def averages(self) -> List[AgentScore]:
        with self._lock:
            return self._rank(dict(self._all))

    def best_of_month(self, year: int, month: int) -> List[AgentScore]:
        with self._lock:
            return self._rank(dict(self._month.get((year, month), {})))

    def top_agents(self, year: int, month: int) -> List[str]:
        r = self.best_of_month(year, month)
        return [s.agent for s in r if by_average(s, r[0]) == 0]

    @staticmethod
    def export(rows: List[AgentScore], exporter: Callable[[List[AgentScore]], str]) -> str:
        return exporter(rows)

    def _rank(self, m: Dict[str, Stats]) -> List[AgentScore]:
        rows = [AgentScore(a, s.total, s.count) for a, s in m.items()]
        return sorted(rows, key=cmp_to_key(self._ranking))


def demo() -> None:
    s = RatingService()
    for agent, stars, day in [("alice", 5, "2025-01-10"), ("alice", 4, "2025-01-20"), ("bob", 3, "2025-01-11"),
                              ("carol", 5, "2025-02-02"), ("carol", 4, "2025-02-03"), ("carol", 5, "2025-02-04")]:
        s.rate(agent, stars, date.fromisoformat(day))
    print("stage 1 averages:", s.averages())
    for stars, day in [(5, "2025-02-10"), (4, "2025-02-11"), (5, "2025-02-12"), (4, "2025-02-13")]:
        s.rate("dave", stars, date.fromisoformat(day))
    print("stage 2 averages:", s.averages())
    print("stage 3 Jan 2025:", s.best_of_month(2025, 1))
    print("stage 3 Feb 2025:", s.best_of_month(2025, 2), "top:", s.top_agents(2025, 2))
    print("stage 4 csv:\n" + s.export(s.averages(), to_csv), end="")
    print("stage 4 json Jan:", s.export(s.best_of_month(2025, 1), to_json))
    try:
        s.rate("bob", 6, date(2025, 1, 1))
    except RatingError as e:
        print("rejected:", e)


if __name__ == "__main__":
    demo()

import threading

class Stage1Averages(unittest.TestCase):
    def test_highest_first(self):
        s = RatingService()
        s.rate("a", 3, JAN)
        s.rate("b", 5, JAN)
        self.assertEqual([r.agent for r in s.averages()], ["b", "a"])

    def test_average_is_not_integer_division(self):
        s = RatingService()
        s.rate("a", 5, JAN)
        s.rate("a", 4, JAN)
        self.assertEqual(s.averages()[0].average(), "4.50")

    def test_rounding_half_up(self):
        s = RatingService()
        for stars in (5, 5, 4):  # 14/3 = 4.666..
            s.rate("a", stars, JAN)
        self.assertEqual(s.averages()[0].average(), "4.67")

    def test_validation(self):
        s = RatingService()
        for bad in (0, 6, 3.5):
            with self.assertRaises(RatingError):
                s.rate("a", bad, JAN)
        with self.assertRaises(RatingError):
            s.rate("", 3, JAN)

class Stage5Concurrency(unittest.TestCase):
    def test_no_lost_updates(self):
        s = RatingService()

        def work(stars):
            for k in range(500):
                s.rate(f"agent{k % 3}", stars, JAN)

        ts = [threading.Thread(target=work, args=(t % 5 + 1,)) for t in range(8)]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
        self.assertEqual(sum(r.count for r in s.averages()), 4000)
        self.assertEqual(sum(r.count for r in s.best_of_month(2025, 1)), 4000)

// Agent Rating System - C++14. Build: g++ -std=c++14 -Wall -Wextra main.cpp -o main
// Locking: one mutex (mu_) guards all_ and month_. rate() updates both maps in one critical section; the readers
// copy the Stats they need under the lock and sort the copy after unlocking. A C++14 reader-writer lock would be
// std::shared_timed_mutex (std::shared_mutex is C++17); Java gets the same effect per key with
// ConcurrentHashMap.compute and immutable Stats.
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <algorithm>
#include <cstdlib>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>
#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif

struct Stats { long long sum = 0, count = 0; };

struct AgentScore {
    std::string agent; long long sum, count;
    std::string average() const {                       // half-up, 2 decimals, integer maths
        long long h = (sum * 200 + count) / (2 * count);
        std::string frac = std::to_string(h % 100);
        return std::to_string(h / 100) + "." + (frac.size() < 2 ? "0" : "") + frac;
    }
};

// Ranking policy: "a goes before b". Exact average comparison by cross-multiplication.
using Ranking = std::function<bool(const AgentScore&, const AgentScore&)>;
int byAverage(const AgentScore& a, const AgentScore& b) {   // <0: a first
    long long l = b.sum * a.count, r = a.sum * b.count;
    return l < r ? -1 : (l > r ? 1 : 0);
}
bool defaultRanking(const AgentScore& a, const AgentScore& b) {
    int c = byAverage(a, b);
    if (c != 0) return c < 0;
    if (a.count != b.count) return a.count > b.count;    // more ratings first
    return a.agent < b.agent;                            // then A-Z
}

// Export format: Strategy as an interface (a C ops struct with one function pointer).
struct Exporter {
    virtual ~Exporter() = default;
    virtual std::string exportRows(const std::vector<AgentScore>& rows) const = 0;
};
struct CsvExporter : Exporter {
    std::string exportRows(const std::vector<AgentScore>& rows) const override {
        std::string out = "agent,average,ratings\n";
        for (auto& r : rows) out += r.agent + "," + r.average() + "," + std::to_string(r.count) + "\n";
        return out;
    }
};
struct JsonExporter : Exporter {
    std::string exportRows(const std::vector<AgentScore>& rows) const override {
        std::string out = "[";
        for (size_t i = 0; i < rows.size(); ++i)
            out += std::string(i ? "," : "") + "{\"agent\":\"" + rows[i].agent + "\",\"average\":" +
                   rows[i].average() + ",\"ratings\":" + std::to_string(rows[i].count) + "}";
        return out + "]";
    }
};

class RatingService {
    std::map<std::string, Stats> all_;
    std::map<int, std::map<std::string, Stats>> month_;  // key = year * 100 + month
    Ranking ranking_;
    mutable Mutex mu_;                                   // guards all_ and month_

    std::vector<AgentScore> rank(const std::map<std::string, Stats>& m) const {   // m is a private copy
        std::vector<AgentScore> v;
        for (auto& e : m) v.push_back({e.first, e.second.sum, e.second.count});
        std::sort(v.begin(), v.end(), ranking_);
        return v;
    }

public:
    explicit RatingService(Ranking r = defaultRanking) : ranking_(std::move(r)) {}

    void rate(const std::string& agent, int stars, int year, int month) {
        if (agent.empty()) throw std::invalid_argument("agent id required");
        if (stars < 1 || stars > 5) throw std::invalid_argument("stars must be 1..5, got " + std::to_string(stars));
        std::lock_guard<Mutex> g(mu_);                   // both maps change in one step
        Stats& a = all_[agent]; a.sum += stars; a.count += 1;
        Stats& m = month_[year * 100 + month][agent]; m.sum += stars; m.count += 1;
    }
    std::vector<AgentScore> averages() const {
        std::map<std::string, Stats> copy;
        { std::lock_guard<Mutex> g(mu_); copy = all_; }
        return rank(copy);                               // sort outside the lock
    }
    std::vector<AgentScore> bestOfMonth(int year, int month) const {
        std::map<std::string, Stats> copy;
        {
            std::lock_guard<Mutex> g(mu_);
            auto it = month_.find(year * 100 + month);
            if (it != month_.end()) copy = it->second;
        }
        return rank(copy);
    }
};

static int passed = 0;
void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); }
    ++passed;
}
void print(const char* label, const std::vector<AgentScore>& v) {
    std::cout << label;
    for (auto& s : v) std::cout << " " << s.agent << " " << s.average() << " (" << s.count << ")";
    std::cout << "\n";
}

int main() {
    RatingService s;
    s.rate("alice", 5, 2025, 1); s.rate("alice", 4, 2025, 1); s.rate("bob", 3, 2025, 1);
    s.rate("carol", 5, 2025, 2); s.rate("carol", 4, 2025, 2); s.rate("carol", 5, 2025, 2);
    print("stage 1 averages:", s.averages());
    check(s.averages()[1].average() == "4.50", "alice 4.50");
    for (int stars : {5, 4, 5, 4}) s.rate("dave", stars, 2025, 2);
    print("stage 2 averages:", s.averages());
    check(s.averages()[1].agent == "dave", "tie: more ratings first");
    print("stage 3 Jan 2025:", s.bestOfMonth(2025, 1));
    check(s.bestOfMonth(2025, 1)[0].agent == "alice", "alice best in Jan");
    check(s.bestOfMonth(2025, 3).empty(), "no ratings in March");
    std::unique_ptr<Exporter> csv(new CsvExporter()), json(new JsonExporter());
    std::cout << "stage 4 csv:\n" << csv->exportRows(s.averages());
    std::cout << "stage 4 json Jan: " << json->exportRows(s.bestOfMonth(2025, 1)) << "\n";
    try { s.rate("bob", 6, 2025, 1); check(false, "stars 6"); }
    catch (const std::invalid_argument& e) { std::cout << "rejected: " << e.what() << "\n"; ++passed; }
    std::cout << "All " << passed << " checks passed\n";
    return 0;
}
