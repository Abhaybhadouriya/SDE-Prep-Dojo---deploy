import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.function.*;

public class PoolDemo {

    // The expensive thing we want to reuse (think: a DB connection).
    static final class Connection {
        private static final AtomicInteger NEXT = new AtomicInteger(1);
        final int id = NEXT.getAndIncrement();
        String session = "";              // per-user state that MUST be reset on return
        boolean broken = false;
        Connection() { System.out.println("  [create] conn-" + id + " (slow handshake)"); }
        String query(String sql) { return "conn-" + id + " ran '" + sql + "' as " + session; }
    }

    // A borrowed object. AutoCloseable so try-with-resources always gives it back.
    static final class Lease<T> implements AutoCloseable {
        private final BlockingPool<T> pool;
        private final T obj;
        final long borrowedAt = System.nanoTime();
        final String borrower = Thread.currentThread().getName();
        private boolean returned = false;
        Lease(BlockingPool<T> pool, T obj) { this.pool = pool; this.obj = obj; }
        T get() { if (returned) throw new IllegalStateException("used after release"); return obj; }
        @Override public void close() { if (!returned) { returned = true; pool.release(this); } }
    }

    static final class BlockingPool<T> {
        private final Supplier<T> factory;
        private final Predicate<T> validate;     // still healthy?
        private final Consumer<T> reset;         // wipe the previous user's state
        private final int max;
        private final BlockingQueue<T> idle;
        private final Semaphore permits;         // bounds idle + in-use together
        private final Set<Lease<T>> inUse = ConcurrentHashMap.newKeySet();
        private final AtomicInteger created = new AtomicInteger();

        BlockingPool(int max, Supplier<T> factory, Predicate<T> validate, Consumer<T> reset) {
            this.max = max; this.factory = factory; this.validate = validate; this.reset = reset;
            this.idle = new ArrayBlockingQueue<>(max);
            this.permits = new Semaphore(max, true);   // fair: waiters served in order
        }

        Lease<T> acquire(long timeout, TimeUnit unit) throws InterruptedException, TimeoutException {
            if (!permits.tryAcquire(timeout, unit))
                throw new TimeoutException("pool exhausted: " + max + " in use");
            try {
                T obj = idle.poll();
                while (obj != null && !validate.test(obj)) {        // validate on borrow
                    System.out.println("  [evict] dropping a broken idle object");
                    created.decrementAndGet();
                    obj = idle.poll();
                }
                if (obj == null) { obj = factory.get(); created.incrementAndGet(); }  // lazy growth
                Lease<T> lease = new Lease<>(this, obj);
                inUse.add(lease);
                return lease;
            } catch (RuntimeException e) {
                permits.release();                                // never lose a permit
                throw e;
            }
        }

        private void release(Lease<T> lease) {
            if (!inUse.remove(lease)) throw new IllegalStateException("double release");
            T obj = lease.obj;
            try {
                if (validate.test(obj)) { reset.accept(obj); idle.offer(obj); }   // <-- the pattern: reset, reuse
                else { created.decrementAndGet(); System.out.println("  [evict] broken on return, destroyed"); }
            } catch (RuntimeException e) {                          // reset failed: destroy, do not reuse
                created.decrementAndGet();
                System.out.println("  [evict] reset failed, destroyed: " + e.getMessage());
            } finally {
                permits.release();                                  // release must never lose a permit
            }
        }

        // Leak detection: anything held longer than the threshold is reported.
        List<String> suspectedLeaks(long thresholdMillis) {
            long now = System.nanoTime();
            List<String> out = new ArrayList<>();
            for (Lease<T> l : inUse)
                if (TimeUnit.NANOSECONDS.toMillis(now - l.borrowedAt) > thresholdMillis)
                    out.add("held by " + l.borrower);
            return out;
        }
        String stats() { return "created=" + created.get() + " idle=" + idle.size() + " inUse=" + inUse.size(); }
    }

    public static void main(String[] args) throws Exception {
        BlockingPool<Connection> pool = new BlockingPool<>(2, Connection::new,
                c -> !c.broken, c -> c.session = "");

        System.out.println("1) Borrow, use, return (try-with-resources):");
        try (Lease<Connection> l = pool.acquire(1, TimeUnit.SECONDS)) {
            l.get().session = "asha";
            System.out.println("  " + l.get().query("SELECT 1"));
        }
        try (Lease<Connection> l = pool.acquire(1, TimeUnit.SECONDS)) {
            System.out.println("  reused conn-" + l.get().id + ", session reset to '" + l.get().session + "'");
        }
        System.out.println("  " + pool.stats());

        System.out.println("2) Exhaustion: 2 held, a third caller times out:");
        Lease<Connection> a = pool.acquire(1, TimeUnit.SECONDS);
        Lease<Connection> b = pool.acquire(1, TimeUnit.SECONDS);
        try { pool.acquire(200, TimeUnit.MILLISECONDS); }
        catch (TimeoutException e) { System.out.println("  " + e.getMessage()); }

        System.out.println("3) A waiter is unblocked when someone releases:");
        ExecutorService ex = Executors.newSingleThreadExecutor(r -> new Thread(r, "worker-1"));
        Future<String> f = ex.submit(() -> {
            try (Lease<Connection> l = pool.acquire(2, TimeUnit.SECONDS)) { return "  worker-1 got conn-" + l.get().id; }
        });
        Thread.sleep(100);
        a.close();
        System.out.println(f.get());

        System.out.println("4) Leak detection (b was never closed):");
        Thread.sleep(60);
        System.out.println("  suspected leaks > 50 ms: " + pool.suspectedLeaks(50));

        System.out.println("5) Broken object is evicted, pool heals:");
        b.get().broken = true;
        b.close();
        try (Lease<Connection> l1 = pool.acquire(1, TimeUnit.SECONDS);
             Lease<Connection> l2 = pool.acquire(1, TimeUnit.SECONDS)) {
            System.out.println("  two at once: conn-" + l1.get().id + ", conn-" + l2.get().id);
        }
        System.out.println("  " + pool.stats());
        ex.shutdown();
    }
}

Read the output top to bottom: one handshake served two requests (step 1); the third caller got a clear error instead of hanging forever (step 2); a waiter was woken by a release (step 3); the lease that was never closed was reported with the thread that took it (step 4); a broken connection was destroyed on return and quietly replaced (step 5). Reading like a reviewer: release() must never lose a permit, even when reset throws (a rollback on a dead connection, say). That is why its body sits in try / catch / finally: a failed reset destroys the object and still returns the permit, otherwise the pool shrinks by one for ever.

"""Object Pool in Python: bounded, blocking, context-manager leases, leak report."""
import queue
import threading
import time
from contextlib import contextmanager


class Buffer:
    created = 0

    def __init__(self, size=4096):
        Buffer.created += 1
        self.id = Buffer.created
        self.data = bytearray(size)       # the "expensive" allocation
        self.used = 0
        print(f"  [create] buf-{self.id}")

    def reset(self):
        self.used = 0                     # do not leak the last user's state


class Pool:
    def __init__(self, factory, max_size):
        self._factory = factory
        self._idle = queue.LifoQueue()    # LIFO: most recently used first (cache friendly)
        self._slots = threading.BoundedSemaphore(max_size)
        self._out = {}                    # id(obj) -> (borrower, time)
        self._lock = threading.Lock()

    def acquire(self, timeout=None):
        if not self._slots.acquire(timeout=timeout):
            raise TimeoutError("pool exhausted")
        try:
            obj = self._idle.get_nowait()
        except queue.Empty:
            obj = self._factory()
        with self._lock:
            self._out[id(obj)] = (threading.current_thread().name, time.monotonic())
        return obj

    def release(self, obj):
        with self._lock:
            if self._out.pop(id(obj), None) is None:
                raise RuntimeError("release of an object this pool did not lend")
        try:
            obj.reset()                   # may raise: then drop obj, never reuse it
            self._idle.put(obj)
        finally:
            self._slots.release()         # release must never lose a slot

    @contextmanager
    def lease(self, timeout=None):
        obj = self.acquire(timeout)
        try:
            yield obj
        finally:                          # returned even if the body raises
            self.release(obj)

    def leaks(self, older_than):
        now = time.monotonic()
        with self._lock:
            return [who for who, t in self._out.values() if now - t > older_than]


if __name__ == "__main__":
    pool = Pool(Buffer, max_size=2)
    for i in range(3):                    # three requests, one allocation
        with pool.lease() as buf:
            buf.used = 100 * (i + 1)
            print(f"  request {i} used buf-{buf.id} ({buf.used} bytes)")

    held = pool.acquire()                 # forgot to release: a leak
    with pool.lease() as b2:
        print(f"  second buffer is buf-{b2.id}")
        try:
            pool.acquire(timeout=0.1)
        except TimeoutError as e:
            print(f"  third caller: {e}")
    time.sleep(0.05)
    print(f"  leaks older than 20 ms: {pool.leaks(0.02)}")
    print(f"  total buffers ever created: {Buffer.created}")

with pool.lease() as buf: is Python's version of try-with-resources: the finally inside the context manager returns the object even if the body raises. LifoQueue hands back the most recently used object first, which keeps it warm in CPU caches, so the rarely used ones stay at the bottom, where an idle-timeout evictor (not implemented here) could close them. release() puts the slot back in a finally, so a reset() that raises drops the object but never shrinks the pool.

C++ you need here: std::unique_ptr<T> = an owning pointer that frees its object when it goes out of scope (a kfree() that runs itself on every exit path). A lambda [](args){ body } = an anonymous function: a function pointer plus any variables it captured in the brackets, like a callback with its void *ctx built in. std::function<R(Args)> = a typed slot that can hold any callable (function pointer, lambda, functor). // Object Pool in C++14, kernel flavoured (single-threaded on purpose: the blocking,
// multi-threaded pool is in the Java and Python tabs).
//  1) SlabPool: all objects allocated once, a free list hands them out (like kmem_cache).
//  2) ReservePool: normal allocation first, a pre-filled reserve when allocation fails
//     (the idea behind mempool_t). An RAII handle returns objects automatically.
// Build: g++ -std=c++14 -Wall pool.cpp -o pool
#include <iostream>
#include <memory>
#include <new>
#include <vector>

struct Packet {                        // think: a small skb-like buffer
    char data[256];
    int len = 0;
    void reset() { len = 0; }
};

// ---------- 1) slab-style pool with a free list ----------
template <typename T>
class SlabPool {
    std::vector<T> slab_;              // one big allocation up front
    std::vector<T*> free_;             // free list, used as a stack (LIFO = cache-hot)
    std::size_t outstanding_ = 0;
public:
    struct Releaser {                  // custom deleter: "delete" means "give back"
        SlabPool* pool;
        void operator()(T* p) const { pool->release(p); }
    };
    using Handle = std::unique_ptr<T, Releaser>;

    explicit SlabPool(std::size_t n) : slab_(n) {
        for (auto& o : slab_) free_.push_back(&o);
    }
    Handle acquire() {                 // fail fast when empty (like GFP_ATOMIC)
        if (free_.empty()) return Handle(nullptr, Releaser{this});
        T* p = free_.back();
        free_.pop_back();
        ++outstanding_;
        return Handle(p, Releaser{this});
    }
    std::size_t available() const { return free_.size(); }
    std::size_t outstanding() const { return outstanding_; }
    ~SlabPool() {                      // leak check at teardown, like kmem_cache_destroy
        if (outstanding_) std::cout << "BUG: pool destroyed with " << outstanding_ << " objects out\n";
    }
private:
    void release(T* p) {
        p->reset();                    // wipe state before the next user sees it
        free_.push_back(p);
        --outstanding_;
    }
};

// ---------- 2) mempool-style reserve ----------
static bool g_memory_pressure = false;            // simulate a failing allocator

Packet* try_alloc() {
    if (g_memory_pressure) return nullptr;
    return new (std::nothrow) Packet();
}

class ReservePool {
    std::vector<Packet*> reserve_;                 // pre-allocated, kept for emergencies
    std::size_t min_nr_;
public:
    explicit ReservePool(std::size_t min_nr) : min_nr_(min_nr) {
        for (std::size_t i = 0; i < min_nr; ++i) reserve_.push_back(new Packet());
    }
    ~ReservePool() { for (auto* p : reserve_) delete p; }

    Packet* alloc(const char* who) {
        if (Packet* p = try_alloc()) { std::cout << "  " << who << ": normal allocation\n"; return p; }
        if (!reserve_.empty()) {                   // allocator failed: dip into the reserve
            Packet* p = reserve_.back();
            reserve_.pop_back();
            std::cout << "  " << who << ": from reserve (" << reserve_.size() << " left)\n";
            return p;
        }
        std::cout << "  " << who << ": reserve empty, must wait for a free\n";
        return nullptr;                            // the kernel would sleep here instead
    }
    void free(Packet* p) {
        p->reset();
        if (reserve_.size() < min_nr_) reserve_.push_back(p);   // refill the reserve first
        else delete p;
    }
};

int main() {
    std::cout << "SlabPool with 2 objects:\n";
    {
        SlabPool<Packet> pool(2);
        auto a = pool.acquire();
        auto b = pool.acquire();
        a->len = 64;
        std::cout << "  two out, available=" << pool.available() << "\n";
        auto c = pool.acquire();
        std::cout << "  third acquire " << (c ? "succeeded" : "failed fast (pool empty)") << "\n";
        a.reset();                                 // unique_ptr::reset -> Releaser -> back to pool
        auto d = pool.acquire();
        std::cout << "  after one release: got one, len reset to " << d->len << "\n";
        b.reset();
        d.reset();
        Packet* leaked = pool.acquire().release(); // detach from RAII: a deliberate leak
        (void)leaked;
        std::cout << "  outstanding=" << pool.outstanding() << " (the leaked one)\n";
    }                                              // destructor reports what was never returned

    std::cout << "ReservePool with min_nr=2 (mempool idea):\n";
    ReservePool rp(2);
    Packet* p1 = rp.alloc("io-1");
    g_memory_pressure = true;                      // allocator now fails
    Packet* p2 = rp.alloc("io-2");
    Packet* p3 = rp.alloc("io-3");
    Packet* p4 = rp.alloc("io-4");
    rp.free(p2);                                   // a completed I/O refills the reserve
    Packet* p5 = rp.alloc("io-5");
    rp.free(p1); rp.free(p3); rp.free(p5);
    (void)p4;
    return 0;
}

std::unique_ptr with a custom deleter is the C++ idiom for "release when the scope ends": the deleter does not free memory, it pushes the pointer back on the free list. The destructor check is the same idea as the kernel complaining when you destroy a slab cache that still has objects allocated.

The same idea in C

Object Pool in plain C: the kmem_cache_create / kmem_cache_alloc / kmem_cache_free shape rebuilt in user space as a free list. Six allocations, two real objects. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-object-pool/pool_sketch.c).

/* Object Pool in C: the kmem_cache shape, rebuilt in user space.
   Illustrative, not real kernel code. In a module you would write:
     cache = kmem_cache_create("my_flow", sizeof(struct flow), 0, SLAB_HWCACHE_ALIGN, NULL);
     f = kmem_cache_alloc(cache, GFP_KERNEL);  ...  kmem_cache_free(cache, f);
   Build: gcc -std=c99 -Wall pool_sketch.c -o pool_sketch */
#include <stdio.h>
#include <stdlib.h>

struct flow { struct flow *next_free; int sport, dport; };

struct cache { struct flow *free_list; int created, in_use; };

static struct flow *cache_alloc(struct cache *c)
{
    struct flow *f = c->free_list;
    if (f) c->free_list = f->next_free;              /* reuse: pop the free list */
    else { f = malloc(sizeof *f); c->created++; }    /* grow only when empty */
    c->in_use++;
    return f;
}

static void cache_free(struct cache *c, struct flow *f)
{
    f->sport = f->dport = 0;                         /* reset: no state leaks to the next user */
    f->next_free = c->free_list;                     /* push back, do not free() */
    c->free_list = f;
    c->in_use--;
}

int main(void)
{
    struct cache c = { NULL, 0, 0 };
    for (int round = 0; round < 3; round++) {
        struct flow *a = cache_alloc(&c), *b = cache_alloc(&c);
        a->sport = 1000 + round; b->sport = 2000 + round;
        cache_free(&c, a); cache_free(&c, b);
    }
    printf("6 allocations, %d objects ever created, %d in use\n", c.created, c.in_use);
    return 0;
}
