Seqlocks: StampedLock's kernel ancestor

The optimistic read is the kernel's seqlock. A sequence counter starts even. A writer makes it odd, writes, and makes it even again. A reader records the counter, reads the data, and retries if the counter was odd or has changed:

static DEFINE_SEQLOCK(stats_lock);
static u64 bytes, packets;

void stats_update(u64 b) {                 /* writer */
    write_seqlock(&stats_lock);            /* seq++ (odd) + spinlock */
    bytes += b; packets++;
    write_sequnlock(&stats_lock);          /* seq++ (even) */
}

void stats_read(u64 *b, u64 *p) {          /* reader: never blocks the writer */
    unsigned seq;
    do {
        seq = read_seqbegin(&stats_lock);
        *b = bytes; *p = packets;
    } while (read_seqretry(&stats_lock, seq));
}
void write(int x) {
    std::lock_guard<std::mutex> g(writers_);            // writers exclude each other
    unsigned s = seq_.load(std::memory_order_relaxed);
    seq_.store(s + 1, std::memory_order_relaxed);        // odd
    std::atomic_thread_fence(std::memory_order_release);
    x_.store(x, std::memory_order_relaxed);
    y_.store(2 * x, std::memory_order_relaxed);          // invariant y == 2x
    seq_.store(s + 2, std::memory_order_release);        // even
}
void read(int& x, int& y) const {
    for (;;) {
        unsigned s1 = seq_.load(std::memory_order_acquire);
        if (s1 & 1u) continue;                           // writer active
        x = x_.load(std::memory_order_relaxed);          // payload is atomic: no data race UB
        y = y_.load(std::memory_order_relaxed);
        std::atomic_thread_fence(std::memory_order_acquire);
        if (seq_.load(std::memory_order_relaxed) == s1) return;
    }
}

Seqlocks fit small, plain data that is copied in one go (timestamps, counters, a pair of coordinates). They do not fit data with pointers, because a reader might follow a pointer that a writer is freeing. That case needs RCU.
