import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import java.util.concurrent.locks.*;

/**
 * Read-write locks, StampedLock and immutable snapshots, for the SDE Prep Dojo page
 * "Read-Write Locks, StampedLock & Read-Mostly Data".
 * Run: javac RwDemo.java && java RwDemo

public class RwDemo {

    // ---- 1. a read-mostly cache guarded by ReentrantReadWriteLock -------------------------
    static final class RwCache<K, V> {
        private final Map<K, V> map = new HashMap<>();
        private final ReentrantReadWriteLock rw = new ReentrantReadWriteLock();   // nonfair by default
        final AtomicInteger readersNow = new AtomicInteger(), maxReaders = new AtomicInteger();

        V get(K k) {
            rw.readLock().lock();
            try {
                int n = readersNow.incrementAndGet();
                maxReaders.accumulateAndGet(n, Math::max);
                sleep(2);                                  // pretend the read takes a while
                return map.get(k);
            } finally { readersNow.decrementAndGet(); rw.readLock().unlock(); }
        }
        void put(K k, V v) {
            rw.writeLock().lock();
            try { map.put(k, v); } finally { rw.writeLock().unlock(); }
        }
        /** Downgrade: write, then keep reading under a read lock with no gap. */
        V putAndRead(K k, V v) {
            rw.writeLock().lock();
            try {
                map.put(k, v);
                rw.readLock().lock();                      // allowed while holding the write lock
            } finally { rw.writeLock().unlock(); }         // now only the read lock is held
            try { return map.get(k); } finally { rw.readLock().unlock(); }
        }
        /** Upgrade is NOT supported: with a read lock held, the write lock can never be granted. */
        boolean tryUpgrade() throws InterruptedException {
            rw.readLock().lock();
            try { return rw.writeLock().tryLock(200, TimeUnit.MILLISECONDS); }   // lock() here would hang forever
            finally { rw.readLock().unlock(); }
        }
    }

    // ---- 2. a naive reader-preferring RW lock, to show writer starvation --------------------
    static final class NaiveRwLock {
        private int readers; private boolean writer;
        synchronized void lockRead() throws InterruptedException { while (writer) wait(); readers++; }
        synchronized void unlockRead() { if (--readers == 0) notifyAll(); }
        synchronized void lockWrite() throws InterruptedException { while (writer || readers > 0) wait(); writer = true; }
        synchronized void unlockWrite() { writer = false; notifyAll(); }
    }

    interface RW { void r() throws InterruptedException; void ur(); void w() throws InterruptedException; void uw(); }

    /** 4 readers overlap continuously for ~1 s; a writer arrives at 100 ms. How long does it wait? */
    static long writerWaitMs(RW lock) throws Exception {
        AtomicBoolean stop = new AtomicBoolean();
        List<Thread> readers = new ArrayList<>();
        for (int i = 0; i < 4; i++) {
            final int offset = i;
            Thread t = new Thread(() -> {
                sleep(offset);                             // stagger so read sections always overlap
                while (!stop.get()) {
                    try { lock.r(); } catch (InterruptedException e) { return; }
                    try { sleep(4); } finally { lock.ur(); }
                }
            });
            readers.add(t); t.start();
        }
        sleep(100);
        long t0 = System.nanoTime();
        Thread writer = new Thread(() -> {
            try { lock.w(); lock.uw(); } catch (InterruptedException ignored) { }
        });
        writer.start();
        writer.join(1200);
        long waited = writer.isAlive() ? -1 : (System.nanoTime() - t0) / 1_000_000;
        stop.set(true);
        writer.join();
        if (waited < 0) waited = (System.nanoTime() - t0) / 1_000_000;  // only got in after readers stopped
        for (Thread t : readers) t.join();
        return waited;
    }

    // ---- 3. StampedLock with an optimistic read ---------------------------------------------
    static final class Point {
        private double x, y;
        private final StampedLock sl = new StampedLock();
        final LongAdder optimisticFails = new LongAdder();

        void move(double dx, double dy) {
            long st = sl.writeLock();
            try { x += dx; y += dy; } finally { sl.unlockWrite(st); }
        }
        double distanceFromOrigin() {
            long st = sl.tryOptimisticRead();              // no lock taken, just a version number
            double cx = x, cy = y;                         // copy into locals FIRST
            if (!sl.validate(st)) {                        // a writer ran in between: fall back
                optimisticFails.increment();
                st = sl.readLock();
                try { cx = x; cy = y; } finally { sl.unlockRead(st); }
            }
            return Math.sqrt(cx * cx + cy * cy);
        }
    }

    // ---- 4. copy-on-write snapshot: readers never lock --------------------------------------
    record Config(Map<String, String> flags, long version) {
        Config with(String k, String v) {
            Map<String, String> copy = new HashMap<>(flags);
            copy.put(k, v);
            return new Config(Map.copyOf(copy), version + 1);
        }
    }
    static final AtomicReference<Config> CONFIG = new AtomicReference<>(new Config(Map.of("dark-mode", "off"), 1));
    static String flag(String k) { return CONFIG.get().flags().get(k); }               // one volatile read
    static void setFlag(String k, String v) { CONFIG.updateAndGet(c -> c.with(k, v)); } // CAS loop, may retry

    static void sleep(long ms) { try { Thread.sleep(ms); } catch (InterruptedException e) { Thread.currentThread().interrupt(); } }

    public static void main(String[] args) throws Exception {
        System.out.println("1) ReentrantReadWriteLock: readers share, writers exclude");
        RwCache<String, String> cache = new RwCache<>();
        cache.put("route", "v1");
        ExecutorService pool = Executors.newFixedThreadPool(8);
        List<Future<String>> fs = new ArrayList<>();
        for (int i = 0; i < 40; i++) fs.add(pool.submit(() -> cache.get("route")));
        for (Future<String> f : fs) f.get();
        pool.shutdown();
        System.out.println("  max readers inside at once: " + cache.maxReaders.get());
        System.out.println("  downgrade write -> read returned: " + cache.putAndRead("route", "v2"));
        System.out.println("  upgrade read -> write granted? " + cache.tryUpgrade());

        System.out.println("2) writer waiting while readers keep arriving (one run)");
        NaiveRwLock naive = new NaiveRwLock();
        long naiveWait = writerWaitMs(new RW() {
            public void r() throws InterruptedException { naive.lockRead(); }   public void ur() { naive.unlockRead(); }
            public void w() throws InterruptedException { naive.lockWrite(); }  public void uw() { naive.unlockWrite(); }
        });
        ReentrantReadWriteLock jdk = new ReentrantReadWriteLock();
        long jdkWait = writerWaitMs(new RW() {
            public void r() { jdk.readLock().lock(); }   public void ur() { jdk.readLock().unlock(); }
            public void w() { jdk.writeLock().lock(); }  public void uw() { jdk.writeLock().unlock(); }
        });
        System.out.println("  naive reader-preferring lock: writer waited " + naiveWait + " ms");
        System.out.println("  ReentrantReadWriteLock      : writer waited " + jdkWait + " ms");

        System.out.println("3) StampedLock optimistic reads");
        Point p = new Point();
        AtomicBoolean stop = new AtomicBoolean();
        Thread mover = new Thread(() -> { while (!stop.get()) { p.move(1, 1); sleep(1); } });
        mover.start();
        int reads = 2_000_000;
        for (int i = 0; i < reads; i++) p.distanceFromOrigin();
        stop.set(true); mover.join();
        System.out.println("  " + reads + " reads, " + p.optimisticFails.sum() + " fell back to a read lock");
        StampedLock sl = new StampedLock();
        long w = sl.writeLock();
        System.out.println("  tryOptimisticRead while write-locked returns " + sl.tryOptimisticRead() + " (0 = invalid)");
        sl.unlockWrite(w);

        System.out.println("4) immutable snapshot + AtomicReference (RCU-style)");
        Config before = CONFIG.get();
        setFlag("dark-mode", "on");
        System.out.println("  old reader still sees " + before.flags().get("dark-mode") + " (v" + before.version() + "), new readers see "
                + flag("dark-mode") + " (v" + CONFIG.get().version() + ")");
    }
}

// Build: g++ -std=c++14 -O2 -Wall -Wextra -pthread seqlock.cpp -o seqlock && ./seqlock
// Needs a POSIX-threads g++ (Linux, WSL, or MinGW-w64 "posix"). We built and ran it with g++ 13.3 in
// WSL Ubuntu 24.04. The MinGW.org g++ 6.3 on Windows (win32 thread model) has no std::thread or std::mutex.
// C++14 has std::shared_timed_mutex (std::shared_mutex is C++17).
#include <atomic>
#include <chrono>
#include <iostream>
#include <map>
#include <mutex>
#include <shared_mutex>
#include <string>
#include <thread>
#include <vector>

// ---- 1. read-mostly map with a reader-writer lock --------------------------------
class RouteTable {
public:
    std::string lookup(const std::string& prefix) const {
        std::shared_lock<std::shared_timed_mutex> lk(m_);     // many readers at once
        auto it = routes_.find(prefix);
        return it == routes_.end() ? "drop" : it->second;
    }
    void update(const std::string& prefix, const std::string& nh) {
        std::unique_lock<std::shared_timed_mutex> lk(m_);     // one writer, no readers
        routes_[prefix] = nh;
    }
private:
    mutable std::shared_timed_mutex m_;
    std::map<std::string, std::string> routes_;
};

// ---- 2. seqlock: writers bump a counter to odd, write, bump to even ----------------
// Readers never block a writer; they retry if the counter was odd or changed.
// Payload fields are relaxed atomics so concurrent reads are not a data race in C++.
class SeqPoint {
public:
    void write(int x) {
        std::lock_guard<std::mutex> g(writers_);              // writers still exclude each other
        unsigned s = seq_.load(std::memory_order_relaxed);
        seq_.store(s + 1, std::memory_order_relaxed);         // odd: write in progress
        std::atomic_thread_fence(std::memory_order_release);
        x_.store(x, std::memory_order_relaxed);
        y_.store(2 * x, std::memory_order_relaxed);           // invariant: y == 2x
        seq_.store(s + 2, std::memory_order_release);         // even: done
    }
    // Returns the number of retries; fills a consistent (x, y).
    int read(int& x, int& y) const {
        int retries = 0;
        for (;;) {
            unsigned s1 = seq_.load(std::memory_order_acquire);
            if (s1 & 1u) { ++retries; continue; }             // writer active
            x = x_.load(std::memory_order_relaxed);
            y = y_.load(std::memory_order_relaxed);
            std::atomic_thread_fence(std::memory_order_acquire);
            if (seq_.load(std::memory_order_relaxed) == s1) return retries;
            ++retries;                                        // a write overlapped: try again
        }
    }
private:
    std::atomic<unsigned> seq_{0};
    std::atomic<int> x_{0}, y_{0};
    std::mutex writers_;
};

"""A writer-preferring read-write lock in Python, for the SDE Prep Dojo page
"Read-Write Locks, StampedLock & Read-Mostly Data".  Run: python rwlock.py

The standard library has no RW lock. This one blocks NEW readers while a writer waits,
so a steady stream of readers cannot starve writers.
"""
import threading
import time
from contextlib import contextmanager


class RWLock:
    def __init__(self):
        self._cond = threading.Condition()
        self._readers = 0
        self._writer = False
        self._writers_waiting = 0

    @contextmanager
    def read(self):
        with self._cond:
            while self._writer or self._writers_waiting:   # writer preference
                self._cond.wait()
            self._readers += 1
        try:
            yield
        finally:
            with self._cond:
                self._readers -= 1
                if self._readers == 0:
                    self._cond.notify_all()

    @contextmanager
    def write(self):
        with self._cond:
            self._writers_waiting += 1
            while self._writer or self._readers:
                self._cond.wait()
            self._writers_waiting -= 1
            self._writer = True
        try:
            yield
        finally:
            with self._cond:
                self._writer = False
                self._cond.notify_all()


if __name__ == "__main__":
    lock, data, stop = RWLock(), {"price": 100}, threading.Event()
    max_inside, inside, guard = [0], [0], threading.Lock()

    def reader():
        while not stop.is_set():
            with lock.read():
                with guard:
                    inside[0] += 1
                    max_inside[0] = max(max_inside[0], inside[0])
                time.sleep(0.004)
                with guard:
                    inside[0] -= 1

    rs = [threading.Thread(target=reader) for _ in range(4)]
    for t in rs:
        t.start()
    time.sleep(0.1)
    t0 = time.perf_counter()
    with lock.write():
        data["price"] = 101
    waited = (time.perf_counter() - t0) * 1000
    stop.set()
    for t in rs:
        t.join()
    print(f"max readers inside at once: {max_inside[0]}")
    print(f"writer waited {waited:.1f} ms while 4 readers kept arriving; price={data['price']}")
