import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

enum BerthType { LOWER, MIDDLE, UPPER, SIDE_LOWER, SIDE_UPPER }

enum Quota { GENERAL, TATKAL }

enum Status { CNF, RAC, WL, CANCELLED }

record Station(String code, String name, int km) {}

/** Ordered stops. Segment i is the stretch between stop i and stop i+1. */
final class Route {
    private final List<Station> stops;
    private final Map<String, Integer> index = new HashMap<>();

    Route(List<Station> stops) {
        if (stops.size() < 2 || stops.size() > 64) throw new IllegalArgumentException("2..64 stops");
        this.stops = List.copyOf(stops);
        for (int i = 0; i < stops.size(); i++)
            if (index.put(stops.get(i).code(), i) != null) throw new IllegalArgumentException("repeated stop");
    }

    int indexOf(String code) { Integer i = index.get(code); return i == null ? -1 : i; }
    boolean serves(String from, String to) { int a = indexOf(from), b = indexOf(to); return a >= 0 && b > a; }
    int km(int from, int to) { return stops.get(to).km() - stops.get(from).km(); }
    List<Station> stops() { return stops; }

    /** Bits from..to-1 set: the segments a journey from stop `from` to stop `to` occupies. */
    static long mask(int from, int to) { return ((1L << to) - 1) & ~((1L << from) - 1); }
}

/** One physical berth. racBerth = a side-lower that is sold as two RAC halves. */
record Berth(int no, BerthType type, Quota quota, boolean racBerth) {}

/** One class of a train (3A, SL ...): its berths, waitlist limit and fare rate. */
record TrainClass(String code, boolean ac, List<Berth> berths, int waitlistLimit, long rupeesPerKmX100) {}

record Train(String number, String name, Route route, Map<String, TrainClass> classes) {}

record Passenger(String name, int age, BerthType preference) {}

/** One passenger's seat on a booking. Mutable, guarded by its JourneyInventory's lock. */
final class PassengerTicket {
    final Passenger passenger;
    final Booking booking;   // back-pointer: promotion finds the trip mask in O(1), like skb->sk
    Status status;
    Berth berth;      // CNF: the berth; RAC: the shared side-lower
    int racSlot = -1; // RAC: 0 or 1
    int number;       // RAC n / WL n position, renumbered after every change

    PassengerTicket(Passenger p, Booking b) { this.passenger = p; this.booking = b; }

    String describe() {
        return switch (status) {
            case CNF -> "CNF B1-" + berth.no() + " " + berth.type();
            case RAC -> "RAC " + number + " (B1-" + berth.no() + ")";
            case WL -> "WL " + number;
            case CANCELLED -> "CAN";
        };
    }
}

final class Booking {
    final String pnr, userId, trainNo, classCode, from, to;
    final LocalDate date;
    final int fromIdx, toIdx;
    final Quota quota;
    final long fare;
    final List<PassengerTicket> tickets = new ArrayList<>();

    Booking(String pnr, String userId, String trainNo, LocalDate date, String classCode, String from, String to,
            int fromIdx, int toIdx, Quota quota, long fare, List<Passenger> passengers) {
        this.pnr = pnr; this.userId = userId; this.trainNo = trainNo; this.date = date; this.classCode = classCode;
        this.from = from; this.to = to; this.fromIdx = fromIdx; this.toIdx = toIdx; this.quota = quota; this.fare = fare;
        for (Passenger p : passengers) tickets.add(new PassengerTicket(p, this));
    }

    long mask() { return Route.mask(fromIdx, toIdx); }
    List<PassengerTicket> tickets() { return Collections.unmodifiableList(tickets); }
}

record Availability(int general, int tatkal, int racFree, int waitlisted, int waitlistLimit) {
    /** What the search page shows for the general quota. */
    String display() {
        if (general > 0) return "AVAILABLE " + general;
        if (racFree > 0) return "RAC (" + racFree + " left)";
        if (waitlisted < waitlistLimit) return "WL " + (waitlisted + 1);
        return "REGRET";
    }
}

Strategies.java - berth pickers, fare rules (with the tatkal decorator) and the listener.

import java.util.List;

/** A free berth for the requested segments, and what is already sold on it. */
record Candidate(Berth berth, long occupied) {}

/** Strategy: which free berth a passenger gets. */
interface BerthPicker {
    Candidate pick(List<Candidate> free, BerthType preference);
}

/** Preference first, then best fit: the berth that is already busiest on other segments,
 *  so empty berths stay empty for long-distance passengers (less fragmentation). */
final class PreferenceBestFitPicker implements BerthPicker {
    public Candidate pick(List<Candidate> free, BerthType preference) {
        Candidate best = null;
        for (Candidate c : free) if (better(c, best, preference)) best = c;
        return best;
    }

    private static boolean better(Candidate c, Candidate best, BerthType pref) {
        if (best == null) return true;
        boolean cp = c.berth().type() == pref, bp = best.berth().type() == pref;
        if (cp != bp) return cp;
        int cb = Long.bitCount(c.occupied()), bb = Long.bitCount(best.occupied());
        if (cb != bb) return cb > bb;
        return c.berth().no() < best.berth().no();
    }
}

/** The naive picker, kept to show what best fit buys. */
final class FirstFreePicker implements BerthPicker {
    public Candidate pick(List<Candidate> free, BerthType preference) { return free.isEmpty() ? null : free.get(0); }
}

/** Strategy: fare in rupees. */
interface FareRule {
    long fare(TrainClass cls, int km, Quota quota);
}

/** Toy base fare: km x rate, with a minimum. (Not the real Indian Railways fare table.) */
final class DistanceFare implements FareRule {
    public long fare(TrainClass cls, int km, Quota quota) { return Math.max(100, km * cls.rupeesPerKmX100() / 100); }
}

/** Decorator: tatkal adds a percentage of the base fare, clamped. Toy numbers. */
final class TatkalSurcharge implements FareRule {
    private final FareRule inner;
    private final int percent;
    private final long min, max;

    TatkalSurcharge(FareRule inner, int percent, long min, long max) {
        this.inner = inner; this.percent = percent; this.min = min; this.max = max;
    }

    public long fare(TrainClass cls, int km, Quota quota) {
        long base = inner.fare(cls, km, quota);
        if (quota != Quota.TATKAL) return base;
        return base + Math.min(max, Math.max(min, base * percent / 100));
    }
}

}

record StatusEvent(Booking booking, PassengerTicket ticket, String before, String after) {}

Inventory.java - JourneyInventory: allocation, cancellation, promotion, availability, invariant checker.

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;
import java.util.concurrent.locks.ReentrantLock;

/**
 * Seats of one class of one train on one date. Each berth keeps a bitmask of sold segments,
 * so one berth can be sold NDLS-GWL to one passenger and GWL-NGP to another.
 * Everything is guarded by one fair lock: requests for the same train-date-class are served
 * in arrival order, and different trains or dates never block each other.
 */
final class JourneyInventory {
    final Train train;
    final TrainClass cls;
    final LocalDate date;
    private final BerthPicker picker;
    private final ReentrantLock lock = new ReentrantLock(true);

    private final List<Berth> berths;          // sold whole (CNF)
    private final long[] occupied;             // per berth: sold segments
    private final List<Berth> racBerths;       // side-lowers sold as two RAC halves
    private final long[][] racOccupied;        // per RAC berth, per half: sold segments
    private final List<PassengerTicket> racQueue = new ArrayList<>();
    private final List<PassengerTicket> wlQueue = new ArrayList<>();
    private final Set<Booking> bookings = new LinkedHashSet<>();   // O(1) add/remove, booking order kept

    JourneyInventory(Train train, TrainClass cls, LocalDate date, BerthPicker picker) {
        this.train = train; this.cls = cls; this.date = date; this.picker = picker;
        this.berths = cls.berths().stream().filter(b -> !b.racBerth()).toList();
        this.racBerths = cls.berths().stream().filter(Berth::racBerth).toList();
        this.occupied = new long[berths.size()];
        this.racOccupied = new long[racBerths.size()][2];
    }

    /** Places every passenger (CNF, else RAC, else WL) or none of them. Returns the events. */
    List<StatusEvent> allocate(Booking b) {
        lock.lock();
        try {
            bookings.add(b);
            List<PassengerTicket> placed = new ArrayList<>();
            for (PassengerTicket t : b.tickets) {
                if (tryConfirm(t, b.mask(), b.quota)) { placed.add(t); continue; }
                if (b.quota == Quota.GENERAL && tryRac(t, b.mask())) { racQueue.add(t); placed.add(t); continue; }
                if (b.quota == Quota.GENERAL && wlQueue.size() < cls.waitlistLimit()) {
                    t.status = Status.WL; wlQueue.add(t); placed.add(t); continue;
                }
                for (PassengerTicket p : placed) release(p, b.mask()); // all or nothing
                for (PassengerTicket p : b.tickets) p.status = null;
                bookings.remove(b);
                throw new NoSeatsException(b.quota == Quota.TATKAL ? "REGRET: tatkal quota full" : "REGRET: waitlist full");
            }
            renumber();
            List<StatusEvent> events = new ArrayList<>();
            for (PassengerTicket t : b.tickets) events.add(new StatusEvent(b, t, "NEW", t.describe()));
            return events;
        } finally {
            lock.unlock();
        }
    }

    /** Cancels every passenger of a booking, then promotes RAC and WL passengers into the gaps. */
    List<StatusEvent> cancel(Booking b) {
        lock.lock();
        try {
            List<StatusEvent> events = new ArrayList<>();
            for (PassengerTicket t : b.tickets) {
                if (t.status == Status.CANCELLED) continue;
                String before = t.describe();
                release(t, b.mask());
                t.status = Status.CANCELLED;
                events.add(new StatusEvent(b, t, before, "CAN"));
            }
            promote(events);
            return events;
        } finally {
            lock.unlock();
        }
    }

    Availability availability(int from, int to) {
        long need = Route.mask(from, to);
        lock.lock();
        try {
            int general = 0, tatkal = 0, rac = 0;
            for (int i = 0; i < berths.size(); i++)
                if ((occupied[i] & need) == 0) { if (berths.get(i).quota() == Quota.TATKAL) tatkal++; else general++; }
            for (long[] halves : racOccupied) for (long h : halves) if ((h & need) == 0) rac++;
            return new Availability(general, tatkal, rac, wlQueue.size(), cls.waitlistLimit());
        } finally {
            lock.unlock();
        }
    }

    // ---- under the lock ----

    private boolean tryConfirm(PassengerTicket t, long need, Quota quota) {
        List<Candidate> free = new ArrayList<>();
        for (int i = 0; i < berths.size(); i++)
            if (berths.get(i).quota() == quota && (occupied[i] & need) == 0) free.add(new Candidate(berths.get(i), occupied[i]));
        Candidate c = picker.pick(free, t.passenger.preference());
        if (c == null) return false;
        occupied[berths.indexOf(c.berth())] |= need;
        t.status = Status.CNF; t.berth = c.berth(); t.racSlot = -1;
        return true;
    }

    private boolean tryRac(PassengerTicket t, long need) {
        for (int i = 0; i < racBerths.size(); i++)
            for (int h = 0; h < 2; h++)
                if ((racOccupied[i][h] & need) == 0) {
                    racOccupied[i][h] |= need;
                    t.status = Status.RAC; t.berth = racBerths.get(i); t.racSlot = h;
                    return true;
                }
        return false;
    }

    private void release(PassengerTicket t, long mask) {
        switch (t.status) {
            case CNF -> occupied[berths.indexOf(t.berth)] &= ~mask;
            case RAC -> { racOccupied[racBerths.indexOf(t.berth)][t.racSlot] &= ~mask; racQueue.remove(t); }
            case WL -> wlQueue.remove(t);
            default -> { }
        }
    }

    /** RAC -> CNF first, then WL -> CNF or RAC, each queue in booking order. A passenger whose
     *  segments do not fit is skipped, so a later one with a shorter trip may move ahead. */
    private void promote(List<StatusEvent> events) {
        for (PassengerTicket t : new ArrayList<>(racQueue)) {
            Booking b = bookingOf(t);
            String before = t.describe();
            int rb = racBerths.indexOf(t.berth), slot = t.racSlot;
            if (tryConfirm(t, b.mask(), Quota.GENERAL)) {
                racOccupied[rb][slot] &= ~b.mask();
                racQueue.remove(t);
                events.add(new StatusEvent(b, t, before, null));
            }
        }
        for (PassengerTicket t : new ArrayList<>(wlQueue)) {
            Booking b = bookingOf(t);
            String before = t.describe();
            if (tryConfirm(t, b.mask(), Quota.GENERAL) || (tryRac(t, b.mask()) && racQueue.add(t))) {
                wlQueue.remove(t);
                events.add(new StatusEvent(b, t, before, null));
            }
        }
        renumber();
        events.replaceAll(e -> e.after() == null ? new StatusEvent(e.booking(), e.ticket(), e.before(), e.ticket().describe()) : e);
    }

    private void renumber() {
        for (int i = 0; i < racQueue.size(); i++) racQueue.get(i).number = i + 1;
        for (int i = 0; i < wlQueue.size(); i++) wlQueue.get(i).number = i + 1;
    }

    private static Booking bookingOf(PassengerTicket t) { return t.booking; }   // O(1), no scan under the lock

    private boolean fitsGeneralBerth(long m) {
        for (int i = 0; i < berths.size(); i++) if (berths.get(i).quota() == Quota.GENERAL && (occupied[i] & m) == 0) return true;
        return false;
    }

    private boolean fitsRacHalf(long m) {
        for (long[] halves : racOccupied) for (long h : halves) if ((h & m) == 0) return true;
        return false;
    }

    /** Test hook: rebuilds the masks from the tickets, checks nothing is sold twice and no waiting passenger fits a free place. */
    void checkInvariants() {
        lock.lock();
        try {
            long[] cnf = new long[berths.size()];
            long[][] rac = new long[racBerths.size()][2];
            int racCount = 0, wlCount = 0;
            for (Booking b : bookings)
                for (PassengerTicket t : b.tickets) {
                    if (t.status == Status.CNF) {
                        int i = berths.indexOf(t.berth);
                        if ((cnf[i] & b.mask()) != 0) throw new AssertionError("berth " + t.berth.no() + " sold twice");
                        cnf[i] |= b.mask();
                    } else if (t.status == Status.RAC) {
                        int i = racBerths.indexOf(t.berth);
                        if ((rac[i][t.racSlot] & b.mask()) != 0) throw new AssertionError("RAC half sold twice");
                        rac[i][t.racSlot] |= b.mask();
                        racCount++;
                    } else if (t.status == Status.WL) wlCount++;
                }
            for (int i = 0; i < cnf.length; i++) if (cnf[i] != occupied[i]) throw new AssertionError("mask drift on berth " + i);
            for (int i = 0; i < rac.length; i++)
                for (int h = 0; h < 2; h++) if (rac[i][h] != racOccupied[i][h]) throw new AssertionError("RAC mask drift");
            if (racCount != racQueue.size() || wlCount != wlQueue.size()) throw new AssertionError("queue drift");
            if (wlCount > cls.waitlistLimit()) throw new AssertionError("waitlist over limit");
            for (PassengerTicket t : racQueue) if (fitsGeneralBerth(bookingOf(t).mask())) throw new AssertionError("missed RAC promotion");
            for (PassengerTicket t : wlQueue) {
                long m = bookingOf(t).mask();
                if (fitsGeneralBerth(m) || fitsRacHalf(m)) throw new AssertionError("missed WL promotion");
            }
        } finally {
            lock.unlock();
        }
    }
}

ReservationService.java - catalogue with the station index, and the service.

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicLong;

/** Read-mostly catalogue with a reverse index station -> trains. */
final class TrainCatalog {
    private final Map<String, Train> trains = new HashMap<>();
    private final Map<String, Set<String>> byStation = new HashMap<>();

    synchronized void add(Train t) {
        trains.put(t.number(), t);
        for (Station s : t.route().stops()) byStation.computeIfAbsent(s.code(), k -> new HashSet<>()).add(t.number());
    }

    synchronized Train get(String number) {
        Train t = trains.get(number);
        if (t == null) throw new IllegalArgumentException("no train " + number);
        return t;
    }

    /** Intersect the two stations' train sets, then keep trains that visit `from` before `to`. */
    synchronized List<Train> between(String from, String to) {
        Set<String> a = byStation.getOrDefault(from, Set.of()), b = byStation.getOrDefault(to, Set.of());
        List<Train> out = new ArrayList<>();
        for (String n : a) if (b.contains(n) && trains.get(n).route().serves(from, to)) out.add(trains.get(n));
        out.sort((x, y) -> x.number().compareTo(y.number()));
        return out;
    }
}

final class ReservationService {
    static final int MAX_GENERAL = 6, MAX_TATKAL = 4;

    private final TrainCatalog catalog;
    private final Clock clock;
    private final FareRule fares;
    private final BerthPicker picker;
    private final Map<String, JourneyInventory> inventories = new ConcurrentHashMap<>();
    private final Map<String, Booking> bookings = new ConcurrentHashMap<>();
    private final List<BookingListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicLong pnrSeq = new AtomicLong(4_000_000_000L);

    ReservationService(TrainCatalog catalog, Clock clock, FareRule fares, BerthPicker picker) {
        this.catalog = catalog; this.clock = clock; this.fares = fares; this.picker = picker;
    }

    void addListener(BookingListener l) { listeners.add(l); }

    List<Train> search(String from, String to) { return catalog.between(from, to); }

    Availability availability(String trainNo, LocalDate date, String cls, String from, String to) {
        Train t = catalog.get(trainNo);
        requireServes(t, from, to);
        return inventory(t, cls, date).availability(t.route().indexOf(from), t.route().indexOf(to));
    }

    Booking book(String userId, String trainNo, LocalDate date, String cls, String from, String to,
                 Quota quota, List<Passenger> passengers) {
        Train t = catalog.get(trainNo);
        requireServes(t, from, to);
        TrainClass tc = trainClass(t, cls);
        int max = quota == Quota.TATKAL ? MAX_TATKAL : MAX_GENERAL;
        if (passengers.isEmpty() || passengers.size() > max) throw new IllegalArgumentException("1.." + max + " passengers");
        LocalDateTime now = clock.now();
        if (!now.isBefore(date.atStartOfDay())) throw new BookingClosedException("train has left");
        if (quota == Quota.TATKAL && now.isBefore(tatkalOpens(tc, date)))
            throw new BookingClosedException("tatkal opens at " + tatkalOpens(tc, date));
        int a = t.route().indexOf(from), b = t.route().indexOf(to);
        long fare = fares.fare(tc, t.route().km(a, b), quota) * passengers.size();
        Booking booking = new Booking(String.valueOf(pnrSeq.incrementAndGet()), userId, trainNo, date, cls,
                from, to, a, b, quota, fare, passengers);
        List<StatusEvent> events = inventory(t, cls, date).allocate(booking); // throws NoSeatsException
        bookings.put(booking.pnr, booking);
        publish(events);
        return booking;
    }

    Booking cancel(String pnr) {
        Booking b = bookings.get(pnr);
        if (b == null) throw new IllegalArgumentException("no PNR " + pnr);
        Train t = catalog.get(b.trainNo);
        publish(inventory(t, b.classCode, b.date).cancel(b));
        return b;
    }

    Booking pnr(String pnr) { return bookings.get(pnr); }

    JourneyInventory inventoryFor(String trainNo, String cls, LocalDate date) {
        return inventory(catalog.get(trainNo), cls, date);
    }

    /** AC classes open at 10:00, non-AC at 11:00, one day before the journey. */
    static LocalDateTime tatkalOpens(TrainClass tc, LocalDate date) {
        return date.minusDays(1).atTime(tc.ac() ? LocalTime.of(10, 0) : LocalTime.of(11, 0));
    }

    private JourneyInventory inventory(Train t, String cls, LocalDate date) {
        TrainClass tc = trainClass(t, cls);
        return inventories.computeIfAbsent(t.number() + "|" + date + "|" + cls, k -> new JourneyInventory(t, tc, date, picker));
    }

    private static TrainClass trainClass(Train t, String cls) {
        TrainClass tc = t.classes().get(cls);
        if (tc == null) throw new IllegalArgumentException("no class " + cls + " on " + t.number());
        return tc;
    }

    private static void requireServes(Train t, String from, String to) {
        if (!t.route().serves(from, to)) throw new IllegalArgumentException(t.number() + " does not run " + from + "->" + to);
    }

    /** Called after the inventory lock is released: a slow SMS gateway never blocks booking. */
    private void publish(List<StatusEvent> events) {
        for (StatusEvent e : events)
            for (BookingListener l : listeners) l.onStatusChange(e.booking(), e.ticket(), e.before(), e.after());
    }
}

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Random;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.Future;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static void expectThrows(Class<? extends RuntimeException> type, Runnable r, String what) {
        try {
            r.run();
        } catch (RuntimeException e) {
            check(type.isInstance(e), what + " (got " + e.getClass().getSimpleName() + ")");
            System.out.println("  rejected as expected: " + e.getMessage());
            return;
        }
        throw new AssertionError("FAILED: expected " + type.getSimpleName() + " for " + what);
    }

    static Station st(String code, int km) { return new Station(code, code, km); }

    /** 3A coach B1: berths 1-4 and 8 general, 5-6 tatkal quota, 7 side-lower sold as two RAC halves. */
    static TrainClass coach() {
        List<Berth> b = List.of(
                new Berth(1, BerthType.LOWER, Quota.GENERAL, false), new Berth(2, BerthType.MIDDLE, Quota.GENERAL, false),
                new Berth(3, BerthType.UPPER, Quota.GENERAL, false), new Berth(4, BerthType.LOWER, Quota.GENERAL, false),
                new Berth(5, BerthType.MIDDLE, Quota.TATKAL, false), new Berth(6, BerthType.UPPER, Quota.TATKAL, false),
                new Berth(7, BerthType.SIDE_LOWER, Quota.GENERAL, true), new Berth(8, BerthType.SIDE_UPPER, Quota.GENERAL, false));
        return new TrainClass("3A", true, b, 3, 250);
    }

    static Passenger p(String name) { return new Passenger(name, 30, null); }

    static String show(Booking b) {
        StringBuilder s = new StringBuilder("  PNR " + b.pnr + " " + b.from + "->" + b.to + " Rs " + b.fare + ":");
        for (PassengerTicket t : b.tickets()) s.append(" ").append(t.passenger.name()).append("=").append(t.describe());
        return s.toString();
    }

        Booking mohan = irctc.book("u9", T, day, "3A", "GWL", "BPL", Quota.GENERAL, List.of(p("Mohan")));

        clock.set(LocalDateTime.of(2026, 12, 9, 10, 0));
        int users = 100;
        AtomicInteger won = new AtomicInteger(), regret = new AtomicInteger();
        ConcurrentLinkedQueue<Booking> winners = new ConcurrentLinkedQueue<>();
        runConcurrently(users, i -> {
            try {
                winners.add(irctc.book("t" + i, T, day, "3A", "NDLS", "NGP", Quota.TATKAL, List.of(p("T" + i))));
                won.incrementAndGet();
            } catch (NoSeatsException e) {
                regret.incrementAndGet();
            }
        });
        System.out.println("  " + users + " users at 10:00:00 -> " + won.get() + " confirmed, " + regret.get() + " regret");
        winners.forEach(b -> System.out.println(show(b)));
        check(won.get() == 2 && regret.get() == 98, "exactly the 2 tatkal berths are sold");
        irctc.inventoryFor(T, "3A", day).checkInvariants();

        System.out.println("== 8. random race: 300 bookings then 150 cancellations at once ==");
        LocalDate day2 = day.plusDays(1);
        clock.set(LocalDateTime.of(2026, 12, 1, 9, 0));
        ConcurrentLinkedQueue<Booking> made = new ConcurrentLinkedQueue<>();
        AtomicInteger full = new AtomicInteger();
        String[] stops = {"NDLS", "AGC", "GWL", "BPL", "NGP"};
        runConcurrently(300, i -> {
            Random r = new Random(i);
            int a = r.nextInt(4), b = a + 1 + r.nextInt(4 - a);
            List<Passenger> ps = r.nextBoolean() ? List.of(p("R" + i)) : List.of(p("R" + i), p("S" + i));
            try {
                made.add(irctc.book("r" + i, T, day2, "3A", stops[a], stops[b], Quota.GENERAL, ps));
            } catch (NoSeatsException e) {
                full.incrementAndGet();
            }
        });
        JourneyInventory inv2 = irctc.inventoryFor(T, "3A", day2);
        inv2.checkInvariants();
        List<Booking> list = new ArrayList<>(made);
        System.out.println("  booked " + list.size() + ", regret " + full.get() + "; invariants hold");
        check(list.size() + full.get() == 300, "every request answered");
        runConcurrently(list.size() / 2, i -> irctc.cancel(list.get(i * 2).pnr));
        inv2.checkInvariants();
        long cnf = list.stream().flatMap(b -> b.tickets().stream()).filter(t -> t.status == Status.CNF).count();
        System.out.println("  cancelled every second booking: " + cnf + " CNF passengers left; no double sale, no missed promotion");
        check(list.stream().allMatch(b -> b.tickets().stream().allMatch(t -> t.status != null)), "every passenger has a status");

        System.out.println("== 9. fares and validation ==");
        clock.set(LocalDateTime.of(2026, 12, 9, 12, 0));
        Booking tq = irctc.book("x1", "12001", day, "3A", "NDLS", "BPL", Quota.TATKAL, List.of(p("Neha")));

                Quota.GENERAL, List.of(p("X"))), "train does not serve BPL");

                List.of(p("a"), p("b"), p("c"), p("d"), p("e"))), "5 passengers on tatkal");

    interface Task { void run(int i) throws Exception; }

    static void runConcurrently(int n, Task task) throws Exception {
        ExecutorService pool = Executors.newFixedThreadPool(32);
        CountDownLatch start = new CountDownLatch(1);
        List<Future<Object>> futures = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            final int id = i;
            futures.add(pool.submit(() -> {
                start.await();
                task.run(id);
                return null;
            }));
        }
        start.countDown();
        for (Future<Object> f : futures) f.get(30, TimeUnit.SECONDS); // rethrows any unexpected failure
        pool.shutdown();
    }
