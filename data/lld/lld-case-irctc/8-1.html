irctc.py - the full model, inventory, strategies (as plain functions) and service, with a threading.Lock per journey.

"""Train ticket booking (IRCTC-style): segment bitmasks per berth, RAC halves, waitlist promotion."""
from __future__ import annotations

import itertools
import threading
from dataclasses import dataclass, field
from datetime import date, datetime, time, timedelta
from enum import Enum
from typing import Callable, Optional


class BerthType(Enum):
    LOWER = "LOWER"; MIDDLE = "MIDDLE"; UPPER = "UPPER"; SIDE_LOWER = "SIDE_LOWER"; SIDE_UPPER = "SIDE_UPPER"


class Quota(Enum):
    GENERAL = "GENERAL"; TATKAL = "TATKAL"


class Status(Enum):
    CNF = "CNF"; RAC = "RAC"; WL = "WL"; CANCELLED = "CAN"


class NoSeats(Exception):
    pass


class BookingClosed(Exception):
    pass


@dataclass(frozen=True)
class Station:
    code: str
    km: int


class Route:
    def __init__(self, stops: list[Station]):
        if not 2 <= len(stops) <= 64:
            raise ValueError("2..64 stops")
        self.stops = list(stops)
        self.index = {s.code: i for i, s in enumerate(stops)}
        if len(self.index) != len(stops):
            raise ValueError("repeated stop")

    def serves(self, a: str, b: str) -> bool:
        return a in self.index and b in self.index and self.index[a] < self.index[b]

    def km(self, i: int, j: int) -> int:
        return self.stops[j].km - self.stops[i].km

    @staticmethod
    def mask(i: int, j: int) -> int:
        """Bits i..j-1: the segments a trip from stop i to stop j occupies."""
        return ((1 << j) - 1) & ~((1 << i) - 1)


@dataclass(frozen=True)
class Berth:
    no: int
    type: BerthType
    quota: Quota = Quota.GENERAL
    rac: bool = False


@dataclass(frozen=True)
class TrainClass:
    code: str
    ac: bool
    berths: tuple[Berth, ...]
    waitlist_limit: int
    paise_per_km: int


@dataclass(frozen=True)
class Train:
    number: str
    route: Route
    classes: dict


@dataclass(frozen=True)
class Passenger:
    name: str
    preference: Optional[BerthType] = None


@dataclass(eq=False)
class Ticket:
    passenger: Passenger
    booking: "Booking" = None
    status: Optional[Status] = None
    berth: Optional[Berth] = None
    half: int = -1
    number: int = 0

    def describe(self) -> str:
        if self.status is Status.CNF:
            return f"CNF B1-{self.berth.no} {self.berth.type.value}"
        if self.status is Status.RAC:
            return f"RAC {self.number} (B1-{self.berth.no})"
        if self.status is Status.WL:
            return f"WL {self.number}"
        return "CAN"


@dataclass(eq=False)
class Booking:
    pnr: str
    train_no: str
    day: date
    cls: str
    src: str
    dst: str
    i: int
    j: int
    quota: Quota
    fare: int
    tickets: list = field(default_factory=list)

    @property
    def mask(self) -> int:
        return Route.mask(self.i, self.j)


# ---- strategies ----

def best_fit_picker(free: list[tuple[Berth, int]], pref: Optional[BerthType]):
    """Preference first, then the berth already busiest on other segments, then lowest number."""
    if not free:
        return None
    return min(free, key=lambda c: (c[0].type is not pref, -bin(c[1]).count("1"), c[0].no))


def distance_fare(cls: TrainClass, km: int, quota: Quota) -> int:
    return max(100, km * cls.paise_per_km // 100)


def tatkal_surcharge(inner, percent=30, lo=400, hi=500):
    def fare(cls, km, quota):
        base = inner(cls, km, quota)
        return base if quota is not Quota.TATKAL else base + min(hi, max(lo, base * percent // 100))
    return fare


# ---- the contended core ----

class JourneyInventory:
    """One class of one train on one date. One lock; nothing slow happens while it is held."""

    def __init__(self, cls: TrainClass, picker=best_fit_picker):
        self.cls, self.picker = cls, picker
        self.lock = threading.Lock()
        self.berths = [b for b in cls.berths if not b.rac]
        self.occ = {b.no: 0 for b in self.berths}
        self.rac_berths = [b for b in cls.berths if b.rac]
        self.rac_occ = {b.no: [0, 0] for b in self.rac_berths}
        self.rac_q: list[Ticket] = []
        self.wl_q: list[Ticket] = []
        self.bookings: list[Booking] = []

    def allocate(self, b: Booking) -> list:
        with self.lock:
            placed = []
            for t in b.tickets:
                if self._confirm(t, b.mask, b.quota) or (b.quota is Quota.GENERAL and self._rac(t, b.mask)):
                    placed.append(t)
                    continue
                if b.quota is Quota.GENERAL and len(self.wl_q) < self.cls.waitlist_limit:
                    t.status = Status.WL
                    self.wl_q.append(t)
                    placed.append(t)
                    continue
                for p in placed:  # all or nothing
                    self._release(p, b.mask)
                    p.status = None
                raise NoSeats("REGRET: tatkal quota full" if b.quota is Quota.TATKAL else "REGRET: waitlist full")
            self.bookings.append(b)
            self._renumber()
            return [(b, t, "NEW", t.describe()) for t in b.tickets]

    def cancel(self, b: Booking) -> list:
        with self.lock:
            events = []
            for t in b.tickets:
                if t.status in (None, Status.CANCELLED):
                    continue
                before = t.describe()
                self._release(t, b.mask)
                t.status = Status.CANCELLED
                events.append((b, t, before, "CAN"))
            moved = self._promote()
            self._renumber()
            return events + [(t.booking, t, before, t.describe()) for t, before in moved]

    def availability(self, i: int, j: int) -> dict:
        need = Route.mask(i, j)
        with self.lock:
            gen = sum(1 for b in self.berths if b.quota is Quota.GENERAL and not self.occ[b.no] & need)
            tq = sum(1 for b in self.berths if b.quota is Quota.TATKAL and not self.occ[b.no] & need)
            rac = sum(1 for h in self.rac_occ.values() for x in h if not x & need)
            return {"general": gen, "tatkal": tq, "rac": rac, "wl": len(self.wl_q), "wl_limit": self.cls.waitlist_limit}

    # -- under the lock --
    def _confirm(self, t: Ticket, need: int, quota: Quota) -> bool:
        free = [(b, self.occ[b.no]) for b in self.berths if b.quota is quota and not self.occ[b.no] & need]
        pick = self.picker(free, t.passenger.preference)
        if pick is None:
            return False
        self.occ[pick[0].no] |= need
        t.status, t.berth, t.half = Status.CNF, pick[0], -1
        return True

    def _rac(self, t: Ticket, need: int) -> bool:
        for b in self.rac_berths:
            for h in (0, 1):
                if not self.rac_occ[b.no][h] & need:
                    self.rac_occ[b.no][h] |= need
                    t.status, t.berth, t.half = Status.RAC, b, h
                    self.rac_q.append(t)
                    return True
        return False

    def _release(self, t: Ticket, need: int) -> None:
        if t.status is Status.CNF:
            self.occ[t.berth.no] &= ~need
        elif t.status is Status.RAC:
            self.rac_occ[t.berth.no][t.half] &= ~need
            self.rac_q.remove(t)
        elif t.status is Status.WL:
            self.wl_q.remove(t)

    def _promote(self) -> list:
        moved = []
        for t in list(self.rac_q):
            before, berth, half = t.describe(), t.berth, t.half
            if self._confirm(t, t.booking.mask, Quota.GENERAL):
                self.rac_occ[berth.no][half] &= ~t.booking.mask
                self.rac_q.remove(t)
                moved.append((t, before))
        for t in list(self.wl_q):
            before = t.describe()
            if self._confirm(t, t.booking.mask, Quota.GENERAL) or self._rac(t, t.booking.mask):
                self.wl_q.remove(t)
                moved.append((t, before))
        return moved

    def _renumber(self) -> None:
        for n, t in enumerate(self.rac_q, 1):
            t.number = n
        for n, t in enumerate(self.wl_q, 1):
            t.number = n

    def check_invariants(self) -> None:
        with self.lock:
            cnf = {b.no: 0 for b in self.berths}
            rac = {b.no: [0, 0] for b in self.rac_berths}
            for bk in self.bookings:
                for t in bk.tickets:
                    if t.status is Status.CNF:
                        assert not cnf[t.berth.no] & bk.mask, "berth sold twice"
                        cnf[t.berth.no] |= bk.mask
                    elif t.status is Status.RAC:
                        assert not rac[t.berth.no][t.half] & bk.mask, "RAC half sold twice"
                        rac[t.berth.no][t.half] |= bk.mask
            assert cnf == self.occ and rac == self.rac_occ, "mask drift"
            assert len(self.wl_q) <= self.cls.waitlist_limit
            for t in self.rac_q + self.wl_q:
                m = t.booking.mask
                assert not any(b.quota is Quota.GENERAL and not self.occ[b.no] & m for b in self.berths), "missed promotion"


class ReservationService:
    def __init__(self, clock: Callable[[], datetime], fare=tatkal_surcharge(distance_fare), picker=best_fit_picker):
        self.clock, self.fare, self.picker = clock, fare, picker
        self.trains: dict[str, Train] = {}
        self.by_station: dict[str, set] = {}
        self.inventories: dict[str, JourneyInventory] = {}
        self.bookings: dict[str, Booking] = {}
        self.listeners: list = []
        self._lock = threading.Lock()
        self._pnr = itertools.count(4_000_000_001)

    def add_train(self, t: Train) -> None:
        self.trains[t.number] = t
        for s in t.route.stops:
            self.by_station.setdefault(s.code, set()).add(t.number)

    def search(self, a: str, b: str) -> list[str]:
        both = self.by_station.get(a, set()) & self.by_station.get(b, set())
        return sorted(n for n in both if self.trains[n].route.serves(a, b))

    def inventory(self, train_no: str, cls: str, day: date) -> JourneyInventory:
        key = f"{train_no}|{day}|{cls}"
        with self._lock:
            if key not in self.inventories:
                self.inventories[key] = JourneyInventory(self.trains[train_no].classes[cls], self.picker)
            return self.inventories[key]

    def availability(self, train_no, day, cls, a, b) -> dict:
        r = self._train(train_no, a, b).route
        return self.inventory(train_no, cls, day).availability(r.index[a], r.index[b])

    @staticmethod
    def tatkal_opens(tc: TrainClass, day: date) -> datetime:
        return datetime.combine(day - timedelta(days=1), time(10 if tc.ac else 11, 0))

    def book(self, user, train_no, day, cls, a, b, quota, passengers) -> Booking:
        t = self._train(train_no, a, b)
        tc = t.classes[cls]
        limit = 4 if quota is Quota.TATKAL else 6
        if not 1 <= len(passengers) <= limit:
            raise ValueError(f"1..{limit} passengers")
        now = self.clock()
        if now >= datetime.combine(day, time(0, 0)):
            raise BookingClosed("train has left")
        if quota is Quota.TATKAL and now < self.tatkal_opens(tc, day):
            raise BookingClosed(f"tatkal opens at {self.tatkal_opens(tc, day)}")
        i, j = t.route.index[a], t.route.index[b]
        with self._lock:
            pnr = str(next(self._pnr))
        bk = Booking(pnr, train_no, day, cls, a, b, i, j, quota, self.fare(tc, t.route.km(i, j), quota) * len(passengers))
        bk.tickets = [Ticket(p, bk) for p in passengers]
        events = self.inventory(train_no, cls, day).allocate(bk)
        self.bookings[pnr] = bk
        self._publish(events)
        return bk

    def cancel(self, pnr: str) -> Booking:
        bk = self.bookings.get(pnr)
        if bk is None:
            raise ValueError(f"no PNR {pnr}")
        self._publish(self.inventory(bk.train_no, bk.cls, bk.day).cancel(bk))
        return bk

    def _train(self, n, a, b) -> Train:
        t = self.trains.get(n)
        if t is None or not t.route.serves(a, b):
            raise ValueError(f"{n} does not run {a}->{b}")
        return t

    def _publish(self, events) -> None:  # outside the inventory lock
        for e in events:
            for listener in self.listeners:
                listener(*e)


def coach() -> TrainClass:
    G, T = Quota.GENERAL, Quota.TATKAL
    L, M, U, SL, SU = BerthType.LOWER, BerthType.MIDDLE, BerthType.UPPER, BerthType.SIDE_LOWER, BerthType.SIDE_UPPER
    return TrainClass("3A", True, (Berth(1, L), Berth(2, M), Berth(3, U), Berth(4, L), Berth(5, M, T), Berth(6, U, T),
                                   Berth(7, SL, G, True), Berth(8, SU)), 3, 250)


def demo_service(clock) -> ReservationService:
    s = ReservationService(clock)
    st = [Station("NDLS", 0), Station("AGC", 195), Station("GWL", 313), Station("BPL", 701), Station("NGP", 1091)]
    s.add_train(Train("12627", Route(st), {"3A": coach()}))
    s.add_train(Train("12001", Route(st[:4]), {"3A": coach()}))
    s.add_train(Train("12951", Route([st[0], Station("KOTA", 465), Station("BCT", 1386)]), {"3A": coach()}))
    return s

def check(cond, msg="check failed"):
    """Like assert, but never stripped by python -O: the call inside it always runs."""
    if not cond:
        raise AssertionError(msg)

test_irctc.py - unit tests: masks, segment reuse, RAC/WL/regret, all-or-nothing, promotion order, short trip in a gap, tatkal fare, wrong direction.

import unittest
from datetime import date, datetime

from irctc import NoSeats, Passenger, Quota, Route, Status, demo_service

DAY = date(2026, 12, 10)


class IrctcTest(unittest.TestCase):
    def setUp(self):
        self.s = demo_service(lambda: datetime(2026, 12, 1, 9, 0))

    def book(self, a, b, *names):
        return self.s.book("u", "12627", DAY, "3A", a, b, Quota.GENERAL, [Passenger(n) for n in names])

    def test_mask(self):
        self.assertEqual(Route.mask(0, 2), 0b11)
        self.assertEqual(Route.mask(2, 4), 0b1100)
        self.assertEqual(Route.mask(0, 2) & Route.mask(2, 4), 0)

    def test_segment_reuse_and_best_fit(self):
        a = self.book("NDLS", "GWL", "A")
        b = self.book("GWL", "NGP", "B")
        self.assertEqual(a.tickets[0].berth, b.tickets[0].berth)

    def test_rac_then_wl_then_regret(self):
        self.book("NDLS", "NGP", "a", "b", "c", "d", "e")
        r = self.book("NDLS", "NGP", "r1", "r2")
        self.assertEqual([t.status for t in r.tickets], [Status.RAC, Status.RAC])
        w = self.book("NDLS", "NGP", "w1", "w2", "w3")
        self.assertEqual([t.describe() for t in w.tickets], ["WL 1", "WL 2", "WL 3"])
        with self.assertRaises(NoSeats):
            self.book("NDLS", "NGP", "x")

    def test_all_or_nothing(self):
        self.book("NDLS", "NGP", "a", "b", "c", "d", "e")
        self.book("NDLS", "NGP", "r1", "r2", "w1", "w2")
        with self.assertRaises(NoSeats):
            self.book("NDLS", "NGP", "w3", "w4")
        self.assertEqual(self.s.availability("12627", DAY, "3A", "NDLS", "NGP")["wl"], 2)

    def test_promotion_order(self):
        first = self.book("NDLS", "NGP", "a", "b", "c", "d", "e")
        r = self.book("NDLS", "NGP", "r1", "r2")
        w = self.book("NDLS", "NGP", "w1")
        self.s.cancel(first.pnr)
        self.assertTrue(all(t.status is Status.CNF for t in r.tickets + w.tickets))
        self.s.inventory("12627", "3A", DAY).check_invariants()

    def test_short_trip_takes_gap(self):
        self.book("NDLS", "NGP", "a", "b", "c", "d")
        x = self.book("GWL", "NGP", "x")
        self.book("NDLS", "GWL", "y")  # fills berth 8 end to end with x
        r = self.book("NDLS", "NGP", "r")
        self.assertEqual(r.tickets[0].status, Status.RAC)
        self.s.cancel(x.pnr)
        self.assertEqual(r.tickets[0].status, Status.RAC)  # GWL-NGP gap does not fit NDLS-NGP
        g = self.book("BPL", "NGP", "g")
        self.assertEqual(g.tickets[0].status, Status.CNF)

    def test_tatkal_fare_cap(self):
        s = demo_service(lambda: datetime(2026, 12, 9, 12, 0))
        b = s.book("u", "12001", DAY, "3A", "NDLS", "BPL", Quota.TATKAL, [Passenger("n")])
        self.assertEqual(b.fare, 1752 + 500)

    def test_wrong_direction(self):
        self.assertEqual(self.s.search("BPL", "NDLS"), [])
        with self.assertRaises(ValueError):
            self.book("NGP", "NDLS", "x")


if __name__ == "__main__":
    unittest.main()

main.cpp - the allocator core: per-berth uint64_t masks, RAC halves, queues, promotion, and the picker as a virtual interface (a hand-written vtable in C). Single-threaded; the comment at the top says where the mutex goes.

// IRCTC core in C++14: per-berth segment bitmasks, RAC halves, waitlist promotion.
// Locking: every public JourneyInventory method body runs under one mutex per (train, date, class) - one
// inventory object - exactly like the Java ReentrantLock: check-and-claim of a berth mask is one step.
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run-cpp.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <algorithm>
#include <cstdint>
#include <iostream>
#include <memory>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>

#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif

enum class Status { CNF, RAC, WL, CAN };

// CHECK, not assert: the calls inside have side effects and must survive -DNDEBUG.
#define CHECK(x) do { if (!(x)) { std::cerr << "FAILED: " #x "\n"; return 1; } } while (0)

static uint64_t segMask(int from, int to) { return ((1ULL << to) - 1) & ~((1ULL << from) - 1); }

static int popcount64(uint64_t x) { int n = 0; while (x) { x &= x - 1; ++n; } return n; }

struct Berth { int no; std::string type; bool tatkal; bool rac; };

struct Ticket {
    std::string name;
    uint64_t mask;
    Status status = Status::CAN;
    int berth = -1;   // index into berths (CNF) or racBerths (RAC)
    int half = -1;
    int number = 0;
    std::string describe(const std::vector<Berth>& b, const std::vector<Berth>& rb) const {
        switch (status) {
            case Status::CNF: return "CNF B1-" + std::to_string(b[berth].no) + " " + b[berth].type;
            case Status::RAC: return "RAC " + std::to_string(number) + " (B1-" + std::to_string(rb[berth].no) + ")";
            case Status::WL: return "WL " + std::to_string(number);
            default: return "CAN";
        }
    }
};

// Strategy as an interface: the C++ vtable you would write by hand as a struct of function pointers.
struct BerthPicker {
    virtual ~BerthPicker() {}
    virtual int pick(const std::vector<int>& freeIdx, const std::vector<uint64_t>& occ) const = 0;
};
struct BestFit : BerthPicker {
    int pick(const std::vector<int>& freeIdx, const std::vector<uint64_t>& occ) const override {
        int best = -1;
        for (int i : freeIdx)
            if (best < 0 || popcount64(occ[i]) > popcount64(occ[best])) best = i;
        return best;
    }
};

class JourneyInventory {
public:
    JourneyInventory(std::vector<Berth> all, int wlLimit, std::unique_ptr<BerthPicker> p)
        : wlLimit_(wlLimit), picker_(std::move(p)) {
        for (auto& b : all) (b.rac ? racBerths_ : berths_).push_back(b);
        occ_.assign(berths_.size(), 0);
        racOcc_.assign(racBerths_.size() * 2, 0);
    }

    // All passengers of one booking are placed, or none (rollback).
    bool allocate(std::vector<std::shared_ptr<Ticket>>& tickets, bool tatkal) {
        std::lock_guard<Mutex> g(mu_);
        std::vector<std::shared_ptr<Ticket>> placed;
        for (auto& t : tickets) {
            if (confirm(*t, tatkal) || (!tatkal && rac(t))) { placed.push_back(t); continue; }
            if (!tatkal && (int)wl_.size() < wlLimit_) { t->status = Status::WL; wl_.push_back(t); placed.push_back(t); continue; }
            for (auto& p : placed) { release(p); p->status = Status::CAN; }
            return false;
        }
        renumber();
        return true;
    }

    void cancel(std::vector<std::shared_ptr<Ticket>>& tickets) {
        std::lock_guard<Mutex> g(mu_);
        for (auto& t : tickets) { release(t); t->status = Status::CAN; }
        for (auto t : std::vector<std::shared_ptr<Ticket>>(racQ_)) {   // RAC -> CNF
            int rb = t->berth, h = t->half;
            if (confirm(*t, false)) { racOcc_[rb * 2 + h] &= ~t->mask; erase(racQ_, t); }
        }
        for (auto t : std::vector<std::shared_ptr<Ticket>>(wl_))       // WL -> CNF or RAC
            if (confirm(*t, false) || rac(t)) erase(wl_, t);
        renumber();
    }

    int freeGeneral(uint64_t need) const {
        std::lock_guard<Mutex> g(mu_);
        int n = 0;
        for (size_t i = 0; i < berths_.size(); ++i) if (!berths_[i].tatkal && !(occ_[i] & need)) ++n;
        return n;
    }
    std::string show(const Ticket& t) const { return t.describe(berths_, racBerths_); }

private:
    bool confirm(Ticket& t, bool tatkal) {
        std::vector<int> freeIdx;
        for (size_t i = 0; i < berths_.size(); ++i)
            if (berths_[i].tatkal == tatkal && !(occ_[i] & t.mask)) freeIdx.push_back((int)i);
        int i = picker_->pick(freeIdx, occ_);
        if (i < 0) return false;
        occ_[i] |= t.mask; t.status = Status::CNF; t.berth = i; t.half = -1;
        return true;
    }
    bool rac(const std::shared_ptr<Ticket>& t) {
        for (size_t i = 0; i < racOcc_.size(); ++i)
            if (!(racOcc_[i] & t->mask)) {
                racOcc_[i] |= t->mask; t->status = Status::RAC; t->berth = (int)i / 2; t->half = (int)i % 2;
                racQ_.push_back(t);
                return true;
            }
        return false;
    }
    void release(const std::shared_ptr<Ticket>& t) {
        if (t->status == Status::CNF) occ_[t->berth] &= ~t->mask;
        else if (t->status == Status::RAC) { racOcc_[t->berth * 2 + t->half] &= ~t->mask; erase(racQ_, t); }
        else if (t->status == Status::WL) erase(wl_, t);
    }
    static void erase(std::vector<std::shared_ptr<Ticket>>& v, const std::shared_ptr<Ticket>& t) {
        v.erase(std::remove(v.begin(), v.end(), t), v.end());
    }
    void renumber() {
        for (size_t i = 0; i < racQ_.size(); ++i) racQ_[i]->number = (int)i + 1;
        for (size_t i = 0; i < wl_.size(); ++i) wl_[i]->number = (int)i + 1;
    }

    std::vector<Berth> berths_, racBerths_;
    std::vector<uint64_t> occ_, racOcc_;
    std::vector<std::shared_ptr<Ticket>> racQ_, wl_;
    int wlLimit_;
    std::unique_ptr<BerthPicker> picker_;
    mutable Mutex mu_;                     // one lock per journey inventory
};

using Booking = std::vector<std::shared_ptr<Ticket>>;

static Booking make(std::initializer_list<const char*> names, int from, int to) {
    Booking b;
    for (auto n : names) { auto t = std::make_shared<Ticket>(); t->name = n; t->mask = segMask(from, to); b.push_back(t); }
    return b;
}

int main() {
    // stops: 0 NDLS, 1 AGC, 2 GWL, 3 BPL, 4 NGP
    std::vector<Berth> coach = {{1, "LOWER", false, false}, {2, "MIDDLE", false, false}, {3, "UPPER", false, false},
                                {4, "LOWER", false, false}, {5, "MIDDLE", true, false},  {6, "UPPER", true, false},
                                {7, "SIDE_LOWER", false, true}, {8, "SIDE_UPPER", false, false}};
    JourneyInventory inv(coach, 3, std::unique_ptr<BerthPicker>(new BestFit()));
    auto print = [&](const char* tag, const Booking& b) {
        std::cout << "  " << tag << ":";
        for (auto& t : b) std::cout << " " << t->name << "=" << inv.show(*t);
        std::cout << "\n";
    };

    Booking alice = make({"Alice"}, 0, 2), bob = make({"Bob"}, 2, 4);
    CHECK(inv.allocate(alice, false) && inv.allocate(bob, false));
    print("NDLS->GWL", alice); print("GWL->NGP", bob);
    CHECK(alice[0]->berth == bob[0]->berth);

    Booking family = make({"Chitra", "Chirag", "Chetan", "Chhavi"}, 0, 4);
    Booking rac = make({"Dev", "Esha"}, 0, 4), wl = make({"Farah", "Gopal", "Hari"}, 0, 4);
    CHECK(inv.allocate(family, false) && inv.allocate(rac, false) && inv.allocate(wl, false));
    print("family", family); print("rac", rac); print("wl", wl);
    Booking ira = make({"Ira"}, 0, 4);
    CHECK(!inv.allocate(ira, false));
    std::cout << "  Ira: REGRET (waitlist full)\n";

    inv.cancel(family);
    std::cout << "  after family cancels:\n";
    print("rac", rac); print("wl", wl);
    CHECK(rac[0]->status == Status::CNF && wl[1]->status == Status::CNF && wl[2]->status == Status::RAC);

    Booking t1 = make({"T1"}, 0, 4), t2 = make({"T2"}, 0, 4), t3 = make({"T3"}, 0, 4);
    CHECK(inv.allocate(t1, true) && inv.allocate(t2, true) && !inv.allocate(t3, true));
    print("tatkal", t1); print("tatkal", t2);
    std::cout << "  T3: REGRET (tatkal quota full)\n";
    CHECK(inv.freeGeneral(segMask(2, 3)) == 0);
    std::cout << "C++ demo OK\n";
    return 0;
}
