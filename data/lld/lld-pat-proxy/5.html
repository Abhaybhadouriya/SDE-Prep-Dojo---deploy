Code: five proxies and a dynamic one

Each tab is a complete program that was compiled and run; the outputs are real. Java builds the chain from the stepper with an injectable clock (so the demo is deterministic) and adds a JDK dynamic proxy. Python does the same and shows a __getattr__ forwarding proxy plus weakref.proxy. C++14 has virtual, caching and protection proxies over an image store, plus an operator-> smart reference. Files live in lld-code/lld-pat-proxy/.

import java.lang.reflect.*;
import java.util.*;
import java.util.concurrent.*;
import java.util.function.LongSupplier;

public class ProxyDemo {

    // ---- Subject: the interface both the real object and every proxy implement ----
    record Quote(String symbol, long pricePaise, String source) { }

    interface QuoteService {
        Quote quote(String symbol);
    }

    // ---- Real subject: expensive to create, slow to call ----
    static final class ExchangeQuoteService implements QuoteService {
        ExchangeQuoteService() { System.out.println("  [real] connecting to exchange feed (expensive)"); }
        @Override public Quote quote(String symbol) {
            System.out.println("  [real] fetching " + symbol);
            long price = 100_00L + Math.abs(symbol.hashCode() % 900_00);   // deterministic fake price
            return new Quote(symbol, price, "exchange");
        }
    }

    // ---- Virtual proxy: delay creating the real object until the first call ----
    static final class LazyQuoteProxy implements QuoteService {
        private volatile QuoteService real;                      // created on first use
        @Override public Quote quote(String symbol) {
            QuoteService r = real;
            if (r == null) {
                synchronized (this) {
                    if (real == null) real = new ExchangeQuoteService();
                    r = real;
                }
            }
            return r.quote(symbol);
        }
    }

    // ---- Caching proxy: answer repeated questions without calling the real object ----
    // One in-flight load per symbol (no stampede) and NO lock held during the slow call:
    // a slow AAPL fetch never blocks cached INFY reads. TTL, but no size bound (see Q&A).
    static final class CachingQuoteProxy implements QuoteService {
        private record Entry(Quote quote, long expiresAtMs) { }
        private final QuoteService target;
        private final long ttlMs;
        private final LongSupplier clock;
        private final ConcurrentHashMap<String, CompletableFuture<Entry>> cache = new ConcurrentHashMap<>();
        CachingQuoteProxy(QuoteService target, long ttlMs, LongSupplier clock) {
            this.target = target; this.ttlMs = ttlMs; this.clock = clock;
        }
        private static boolean usable(CompletableFuture<Entry> f, long now) {
            return !f.isDone()                                     // a load is in flight: wait for it
                || (!f.isCompletedExceptionally() && now < f.join().expiresAtMs());
        }
        @Override public Quote quote(String symbol) {
            long now = clock.getAsLong();
            CompletableFuture<Entry> mine = new CompletableFuture<>();
            CompletableFuture<Entry> f = cache.compute(symbol,          // short, per-key atomic step
                    (k, old) -> old != null && usable(old, now) ? old : mine);
            if (f != mine)                                          // hit (or join someone's load)
                return new Quote(symbol, f.join().quote().pricePaise(), "cache");
            try {                                                   // we own the load; no lock held
                Quote fresh = target.quote(symbol);                 // <-- the pattern: forward on a miss
                mine.complete(new Entry(fresh, now + ttlMs));
                return fresh;
            } catch (RuntimeException ex) {
                cache.remove(symbol, mine);                         // do not cache failures
                mine.completeExceptionally(ex);
                throw ex;
            }
        }
    }

    // ---- Rate-limit proxy: at most N calls per fixed one-second window ----
    static final class RateLimitProxy implements QuoteService {
        private final QuoteService target;
        private final int maxPerSecond;
        private final LongSupplier clock;
        private long windowStartMs = -1;
        private int used;
        RateLimitProxy(QuoteService target, int maxPerSecond, LongSupplier clock) {
            this.target = target; this.maxPerSecond = maxPerSecond; this.clock = clock;
        }
        private synchronized void takePermit() {                   // short critical section
            long now = clock.getAsLong();
            if (windowStartMs < 0 || now - windowStartMs >= 1000) { windowStartMs = now; used = 0; }
            if (used >= maxPerSecond) throw new IllegalStateException("429 rate limit exceeded");
            used++;
        }
        @Override public Quote quote(String symbol) {
            takePermit();
            return target.quote(symbol);                            // slow call runs outside the lock
        }
    }

    // ---- Protection proxy: only callers with the right role get through ----
    record User(String name, Set<String> roles) { }
    static final class AuthQuoteProxy implements QuoteService {
        private final QuoteService target;
        private final User user;
        AuthQuoteProxy(QuoteService target, User user) { this.target = target; this.user = user; }
        @Override public Quote quote(String symbol) {
            if (!user.roles().contains("TRADER")) throw new SecurityException(user.name() + " may not read quotes");
            return target.quote(symbol);                            // <-- the pattern: check, then forward
        }
    }

    // ---- Dynamic proxy: one InvocationHandler that logs calls on ANY interface ----
    @SuppressWarnings("unchecked")
    static <T> T logged(T target, Class<T> iface) {
        InvocationHandler h = (proxy, method, args) -> {
            System.out.println("  [log] -> " + iface.getSimpleName() + "." + method.getName() + Arrays.toString(args));
            try {
                Object result = method.invoke(target, args);
                System.out.println("  [log] <- " + method.getName() + " ok");
                return result;
            } catch (InvocationTargetException ex) {
                throw ex.getCause();                              // rethrow the real exception
            }
        };
        return (T) Proxy.newProxyInstance(iface.getClassLoader(), new Class<?>[]{iface}, h);
    }

    public static void main(String[] args) {
        long[] now = {0};                                         // fake clock we control
        LongSupplier clock = () -> now[0];

        // Client code is written against QuoteService and cannot tell which object it holds.
        QuoteService lazy = new LazyQuoteProxy();
        QuoteService cached = new CachingQuoteProxy(lazy, 500, clock);
        QuoteService limited = new RateLimitProxy(cached, 3, clock);
        User asha = new User("asha", Set.of("TRADER"));
        User ravi = new User("ravi", Set.of("VIEWER"));

        System.out.println("1) proxies built; real service not created yet");
        QuoteService forAsha = new AuthQuoteProxy(limited, asha);
        System.out.println("2) " + forAsha.quote("INFY"));          // creates real, fetches, caches
        System.out.println("3) " + forAsha.quote("INFY"));          // cache hit, real not called
        now[0] = 600;                                             // 600 ms later: cache TTL (500 ms) expired, same 1 s window
        System.out.println("4) " + forAsha.quote("INFY"));          // refetch
        try {
            forAsha.quote("TCS");                                 // 4th call in the same 1 s window
        } catch (IllegalStateException ex) {
            System.out.println("5) blocked: " + ex.getMessage());
        }
        now[0] = 1_200;                                           // new window
        System.out.println("6) " + forAsha.quote("TCS"));

        try {
            new AuthQuoteProxy(limited, ravi).quote("INFY");
        } catch (SecurityException ex) {
            System.out.println("7) denied: " + ex.getMessage());
        }

        QuoteService dyn = logged(cached, QuoteService.class);
        System.out.println("8) " + dyn.quote("TCS"));
        System.out.println("   dynamic proxy class: " + dyn.getClass().getSimpleName().replaceAll("\\d+", "N")
                + ", isProxyClass=" + Proxy.isProxyClass(dyn.getClass()));

        // The JDK's own protection proxy:
        List<String> ro = Collections.unmodifiableList(new ArrayList<>(List.of("a", "b")));
        try { ro.add("c"); } catch (UnsupportedOperationException ex) {
            System.out.println("9) unmodifiableList refused add()");
        }
    }
}

Read the output against the code: call 3 is a cache hit, call 4 refetches because 600 ms passed (TTL 500 ms), and call 5 is refused because the limiter sits in front of the cache, so the three earlier calls (including the hit) used up the window. Proxy.newProxyInstance generated a class named like $Proxy0 at run time that implements QuoteService and sends every call to our InvocationHandler. It only works for interfaces.

"""Proxy: same interface as the real object, but controls access to it (lazy, cache, rate limit, auth)."""
from dataclasses import dataclass
from typing import Callable, Protocol
import weakref
import zlib


@dataclass(frozen=True)
class Quote:
    symbol: str
    price_paise: int
    source: str


class QuoteService(Protocol):                      # Subject
    def quote(self, symbol: str) -> Quote: ...


class ExchangeQuoteService:                        # Real subject: expensive to build, slow to call
    def __init__(self):
        print("  [real] connecting to exchange feed (expensive)")

    def quote(self, symbol):
        print(f"  [real] fetching {symbol}")
        return Quote(symbol, 10_000 + zlib.crc32(symbol.encode()) % 90_000, "exchange")


class LazyQuoteProxy:                              # Virtual proxy
    def __init__(self, factory: Callable[[], QuoteService]):
        self._factory, self._real = factory, None

    def quote(self, symbol):
        if self._real is None:
            self._real = self._factory()           # built on first use only
        return self._real.quote(symbol)


class CachingQuoteProxy:                           # Caching proxy with a TTL
    def __init__(self, target: QuoteService, ttl: float, clock: Callable[[], float]):
        self._target, self._ttl, self._clock = target, ttl, clock
        self._cache: dict[str, tuple[Quote, float]] = {}

    def quote(self, symbol):
        now = self._clock()
        hit = self._cache.get(symbol)
        if hit and now < hit[1]:
            return Quote(symbol, hit[0].price_paise, "cache")
        fresh = self._target.quote(symbol)
        self._cache[symbol] = (fresh, now + self._ttl)
        return fresh


class RateLimitProxy:                              # at most N calls per fixed 1-second window
    def __init__(self, target: QuoteService, max_per_sec: int, clock: Callable[[], float]):
        self._target, self._max, self._clock = target, max_per_sec, clock
        self._window, self._used = None, 0

    def quote(self, symbol):
        now = self._clock()
        if self._window is None or now - self._window >= 1.0:
            self._window, self._used = now, 0
        if self._used >= self._max:
            raise RuntimeError("429 rate limit exceeded")
        self._used += 1
        return self._target.quote(symbol)


class AuthQuoteProxy:                              # Protection proxy
    def __init__(self, target: QuoteService, roles: set[str]):
        self._target, self._roles = target, roles

    def quote(self, symbol):
        if "TRADER" not in self._roles:
            raise PermissionError("role TRADER required")
        return self._target.quote(symbol)


class LoggingProxy:                                # "Dynamic" proxy: forwards ANY attribute
    def __init__(self, target):
        self._target = target

    def __getattr__(self, name):                   # only called for names not found on the proxy
        attr = getattr(self._target, name)
        if not callable(attr):
            return attr

        def wrapper(*args, **kwargs):
            print(f"  [log] -> {type(self._target).__name__}.{name}{args}")
            result = attr(*args, **kwargs)
            print(f"  [log] <- {name} ok")
            return result
        return wrapper


if __name__ == "__main__":
    now = [0.0]

__getattr__ runs only for attributes the proxy itself does not have, so one small class can forward any method of any target: Python's version of a dynamic proxy. Magic methods like __len__ are looked up on the type, not the instance, so a full transparent proxy must define them explicitly. weakref.proxy is a standard-library smart reference: it forwards calls but does not keep the target alive.

C++ you need here: std::unique_ptr<T> = an owning pointer that frees its object when it goes out of scope (a kfree() that runs itself on every exit path). A lambda [](args){ body } = an anonymous function: a function pointer plus any variables it captured in the brackets, like a callback with its void *ctx built in. std::function<R(Args)> = a typed slot that can hold any callable (function pointer, lambda, functor). Ownership and RAII in more depth: kref and shared ownership.

// Proxy in C++14: virtual (lazy), protection, and caching proxies behind one interface,
// plus a "smart reference" proxy that overloads operator-> like a tiny refcounted pointer.
// Build: g++ -std=c++14 -Wall proxy.cpp -o proxy
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>

// Subject
class ImageStore {
public:
    virtual ~ImageStore() = default;
    virtual std::string load(const std::string& id) = 0;
};

// Real subject: expensive constructor (think: mmap a 2 GB archive, open a socket)
class DiskImageStore : public ImageStore {
public:
    DiskImageStore() { std::cout << "  [real] opening image archive (expensive)\n"; }
    std::string load(const std::string& id) override {
        std::cout << "  [real] reading " << id << " from disk\n";
        return "<pixels of " + id + ">";
    }
};

// Virtual proxy: builds the real store on first use.
class LazyImageStore : public ImageStore {
public:
    std::string load(const std::string& id) override {
        if (!real_) real_ = std::make_unique<DiskImageStore>();
        return real_->load(id);
    }
private:
    std::unique_ptr<ImageStore> real_;
};

// Caching proxy: remembers results.
class CachingImageStore : public ImageStore {
public:
    explicit CachingImageStore(ImageStore& target) : target_(target) {}
    std::string load(const std::string& id) override {
        auto it = cache_.find(id);
        if (it != cache_.end()) { std::cout << "  [cache] hit " << id << "\n"; return it->second; }
        return cache_[id] = target_.load(id);
    }
private:
    ImageStore& target_;
    std::map<std::string, std::string> cache_;
};

// Protection proxy: checks permission before delegating.
class PrivateAlbumGuard : public ImageStore {
public:
    PrivateAlbumGuard(ImageStore& target, std::string user) : target_(target), user_(std::move(user)) {}
    std::string load(const std::string& id) override {
        if (id.compare(0, 8, "private/") == 0 && user_ != "owner")
            throw std::runtime_error(user_ + " may not open " + id);
        return target_.load(id);
    }
private:
    ImageStore& target_;
    std::string user_;
};

// Smart reference proxy: operator-> forwards to the object and counts accesses.
template <typename T>
class Counted {
public:
    explicit Counted(T& obj) : obj_(obj) {}
    T* operator->() { ++hits_; return &obj_; }
    int hits() const { return hits_; }
private:
    T& obj_;
    int hits_ = 0;
};

static void show(ImageStore& store, const std::string& id) {   // client: knows only ImageStore
    try { std::string px = store.load(id); std::cout << id << " -> " << px << "\n"; }
    catch (const std::exception& e) { std::cout << id << " -> denied: " << e.what() << "\n"; }
}

int main() {
    LazyImageStore lazy;
    CachingImageStore cached(lazy);
    PrivateAlbumGuard asGuest(cached, "guest");
    PrivateAlbumGuard asOwner(cached, "owner");
    std::cout << "proxies built; archive not opened yet\n";

    show(asGuest, "beach.jpg");          // opens archive, reads, caches
    show(asGuest, "beach.jpg");          // cache hit
    show(asGuest, "private/id.jpg");     // blocked before any disk access
    show(asOwner, "private/id.jpg");     // allowed

    Counted<ImageStore> counted(cached);
    counted->load("beach.jpg");
    counted->load("beach.jpg");
    std::cout << "smart reference saw " << counted.hits() << " accesses\n";
    return 0;
}

The guest's request for a private image was refused before any disk access: the protection proxy sits in front of the cache. Overloading operator-> is the C++ way to write a proxy that looks like a pointer; std::shared_ptr and std::unique_ptr are the standard smart-reference proxies. A small detail we hit: the client stores the loaded string in a variable before printing, because in C++14 the order of evaluation inside one cout << a << f() chain is not guaranteed.

The same idea in C

Protection proxy in plain C: a wrapper ops table with the same signature that checks permission and then forwards, the way an LSM hook runs before the real file operation. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-proxy/proxy_sketch.c).

/* Protection proxy in C: a wrapper ops table that checks permission, then forwards.
   Illustrative, not real kernel code (LSM hooks check before the real file op runs).
   Build: gcc -std=c99 -Wall proxy_sketch.c -o proxy_sketch */
#include <stdio.h>
#include <string.h>

struct store;
struct store_ops { int (*get)(struct store *s, const char *key, char *out, size_t n); };
struct store { const struct store_ops *ops; struct store *real; int uid; };

static int real_get(struct store *s, const char *key, char *out, size_t n)
{
    (void)s;
    snprintf(out, n, "value-of-%s", key);
    return 0;
}

static int guarded_get(struct store *s, const char *key, char *out, size_t n)
{
    if (s->uid != 0 && strncmp(key, "secret/", 7) == 0)
        return -13;                                   /* -EACCES: the real store is never called */
    return s->real->ops->get(s->real, key, out, n);   /* same interface: forward */
}

static const struct store_ops real_ops = { real_get }, guard_ops = { guarded_get };

int main(void)
{
    struct store real = { &real_ops, NULL, 0 };
    struct store user = { &guard_ops, &real, 1000 };  /* the proxy a normal user gets */
    const char *keys[] = { "motd", "secret/key" };
    char buf[64];
    for (int i = 0; i < 2; i++) {
        int rc = user.ops->get(&user, keys[i], buf, sizeof buf);
        printf("%-10s -> %s\n", keys[i], rc == 0 ? buf : "denied (-EACCES)");
    }
    return 0;
}
