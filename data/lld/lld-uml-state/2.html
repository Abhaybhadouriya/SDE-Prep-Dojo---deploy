Example 1: an e-commerce order lifecycle

Requirements: an order is created, paid, packed, shipped and delivered. It can be cancelled any time before it ships; cancelling a paid order refunds it. A delivered order can be returned within 7 days.

Notice what drawing it forced: is cancel allowed from Packed? (Yes, the parcel has not left.) What exactly is the return window measured from? (Delivery, so the Delivered transition must record a timestamp: an action.) These are the questions an interviewer wants you to ask out loud.

The runnable code (lld-code/lld-uml-state/StateMachines.java) keeps the diagram as data: a table from (state, event) to next state, plus the guard and actions in one place. Every arrow on the diagram is in it: both refunding cancels (from Paid and from Packed), the refund on return, and the after(7 days) time event, which a scheduler delivers by calling tick(); the closed end state is CLOSED in code.

enum OrderStatus { CREATED, PAID, PACKED, SHIPPED, DELIVERED, CANCELLED, RETURNED, CLOSED }
enum OrderEvent { PAY, PACK, SHIP, DELIVER, CANCEL, RETURN }

static final class Order {
    private static final Map<OrderStatus, Map<OrderEvent, OrderStatus>> T = new EnumMap<>(OrderStatus.class);
    static {
        for (OrderStatus s : OrderStatus.values()) T.put(s, new EnumMap<>(OrderEvent.class));
        T.get(OrderStatus.CREATED).put(OrderEvent.PAY, OrderStatus.PAID);
        T.get(OrderStatus.PAID).put(OrderEvent.PACK, OrderStatus.PACKED);
        T.get(OrderStatus.PACKED).put(OrderEvent.SHIP, OrderStatus.SHIPPED);
        T.get(OrderStatus.SHIPPED).put(OrderEvent.DELIVER, OrderStatus.DELIVERED);
        for (OrderStatus s : EnumSet.of(OrderStatus.CREATED, OrderStatus.PAID, OrderStatus.PACKED))
            T.get(s).put(OrderEvent.CANCEL, OrderStatus.CANCELLED);
        T.get(OrderStatus.DELIVERED).put(OrderEvent.RETURN, OrderStatus.RETURNED);
    }
    static final Set<OrderStatus> FINAL = EnumSet.of(OrderStatus.CANCELLED, OrderStatus.RETURNED, OrderStatus.CLOSED);

    private OrderStatus status = OrderStatus.CREATED;
    private int deliveredOnDay = -1;

    OrderStatus status() { return status; }

    void fire(OrderEvent e, int today) {
        OrderStatus next = T.get(status).get(e);
        if (next == null) throw new IllegalStateException(e + " not allowed in " + status);
        if (e == OrderEvent.RETURN && today - deliveredOnDay > 7)           // guard [within 7 days]
            throw new IllegalStateException("RETURN guard failed: delivered " + (today - deliveredOnDay) + " days ago");
        if (next == OrderStatus.DELIVERED) deliveredOnDay = today;           // transition action
        boolean refund = next == OrderStatus.RETURNED                          // return / refund
            || (next == OrderStatus.CANCELLED                                  // cancel / refund, but only
                && (status == OrderStatus.PAID || status == OrderStatus.PACKED)); // from Paid or Packed
        if (refund) System.out.println("    action: refund payment");
        System.out.println("    " + status + " --" + e + "--> " + next);
        status = next;
    }

    // time event after(7 days): a scheduler calls tick() daily; it is just another event for the machine
    void tick(int today) {
        if (status == OrderStatus.DELIVERED && today - deliveredOnDay > 7) {
            System.out.println("    " + status + " --after(7 days)--> " + OrderStatus.CLOSED);
            status = OrderStatus.CLOSED;
        }
    }
}
