"""Stack Overflow / Flipkart Overflow style Q&A platform - in-memory, thread-safe."""
from __future__ import annotations

import itertools
import re
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Dict, Iterable, List, Optional, Set

Listener = Callable[[str, str], None]          # (user name, message)
QuestionFilter = Callable[["Question"], bool]


class QaError(Exception):
    pass


class VoteType(Enum):
    UP = 1
    DOWN = -1


class Action(Enum):
    ASK = "ask"
    ANSWER = "answer"
    COMMENT = "comment"
    UPVOTE = "upvote"
    DOWNVOTE = "downvote"
    ACCEPT = "accept"


class User:
    def __init__(self, uid: str, name: str) -> None:
        self.id, self.name = uid, name
        self.topics: Set[str] = set()
        self._rep = 1
        self._lock = threading.Lock()

    @property
    def reputation(self) -> int:
        return self._rep

    def add_reputation(self, delta: int) -> int:
        with self._lock:                       # floor at 1, like Stack Overflow
            self._rep = max(1, self._rep + delta)
            return self._rep


@dataclass
class Comment:
    id: str
    author: User
    text: str
    at: int


@dataclass(eq=False)
class Post:
    id: str
    author: User
    body: str
    created_at: int
    votes: Dict[str, VoteType] = field(default_factory=dict)   # voter id -> vote, guarded by lock
    comments: List[Comment] = field(default_factory=list)
    lock: threading.Lock = field(default_factory=threading.Lock)

    def score(self) -> int:
        with self.lock:
            return sum(v.value for v in self.votes.values())

    def question(self) -> "Question":
        raise NotImplementedError


@dataclass(eq=False)
class Question(Post):
    title: str = ""
    tags: frozenset = frozenset()
    answers: List["Answer"] = field(default_factory=list)
    followers: Set[str] = field(default_factory=set)
    accepted: Optional["Answer"] = None

    def question(self) -> "Question":
        return self

    def answered(self) -> bool:
        return bool(self.answers)


@dataclass(eq=False)
class Answer(Post):
    parent: Optional[Question] = None

    def question(self) -> Question:
        return self.parent


# ---------- strategies and policies ----------
class ReputationPolicy(ABC):
    @abstractmethod
    def for_author(self, is_question: bool, v: VoteType) -> int: ...

    @abstractmethod
    def for_voter(self, is_question: bool, v: VoteType) -> int: ...

    accepted = 15
    acceptor = 2


class StackOverflowReputation(ReputationPolicy):
    def for_author(self, is_question, v):
        return 10 if v is VoteType.UP else -2

    def for_voter(self, is_question, v):
        return -1 if (not is_question and v is VoteType.DOWN) else 0


class PermissionRule(ABC):
    @abstractmethod
    def check(self, user: User, action: Action, tags: Iterable[str]) -> None: ...


class ReputationPrivileges(PermissionRule):
    MIN = {Action.UPVOTE: 15, Action.DOWNVOTE: 125}

    def check(self, user, action, tags):
        need = self.MIN.get(action)
        if need is not None and user.reputation < need:
            raise QaError(f"{user.name} needs {need} reputation to {action.value}")


class TopicSubscriptionRule(PermissionRule):
    def check(self, user, action, tags):
        if action is Action.ACCEPT:
            return
        tags = set(tags)
        if not tags & user.topics:
            raise QaError(f"{user.name} is not subscribed to any of {sorted(tags)}")


def any_topic(topics: Iterable[str]) -> QuestionFilter:
    want = set(topics)
    return lambda q: bool(q.tags & want)


def answered(yes: bool) -> QuestionFilter:
    return lambda q: q.answered() == yes


def everything(q: Question) -> bool:
    return True


def answer_order(q: Question) -> List[Answer]:
    """Accepted first, then highest score, then oldest."""
    acc = q.accepted
    return sorted(q.answers, key=lambda a: (a is not acc, -a.score(), a.created_at))


def words(text: str) -> List[str]:
    return [w for w in re.split(r"[^a-z0-9+#]+", text.lower()) if len(w) > 1]


# ---------- service (facade) ----------
class QaService:
    def __init__(self, reputation: ReputationPolicy, rules: List[PermissionRule]) -> None:
        self._rep = reputation
        self._rules = rules
        self._users: Dict[str, User] = {}
        self._questions: Dict[str, Question] = {}
        self._posts: Dict[str, Post] = {}
        self._index: Dict[str, Set[str]] = {}
        self._listeners: List[Listener] = []
        self._ids = threading.Lock()             # guards counters, maps and the word index
        self._clock = itertools.count(1)
        self._counters = {k: itertools.count(1) for k in "UQAC"}

    def _next(self, kind: str) -> str:
        with self._ids:
            return f"{kind}{next(self._counters[kind])}"

    def _tick(self) -> int:
        with self._ids:
            return next(self._clock)

    def add_listener(self, listener: Listener) -> None:
        self._listeners.append(listener)

    def register(self, name: str) -> str:
        uid = self._next("U")
        with self._ids:
            self._users[uid] = User(uid, name)
        return uid

    def subscribe(self, uid: str, *topics: str) -> None:
        self._user(uid).topics.update(topics)

    def unsubscribe(self, uid: str, topic: str) -> None:
        self._user(uid).topics.discard(topic)

    def bonus(self, uid: str, points: int) -> None:
        self._user(uid).add_reputation(points)

    def ask(self, uid: str, title: str, body: str, *tags: str) -> str:
        u = self._user(uid)
        if not title.strip():
            raise QaError("title is empty")
        if not tags:
            raise QaError("a question needs at least one tag")
        self._allow(u, Action.ASK, tags)       # validate before taking an id
        q = Question(self._next("Q"), u, body, self._tick(), title=title.strip(), tags=frozenset(tags))
        q.followers.add(u.id)
        with self._ids:
            self._questions[q.id] = q
            self._posts[q.id] = q
            for w in words(q.title + " " + q.body):
                self._index.setdefault(w, set()).add(q.id)
        return q.id

    def answer(self, uid: str, qid: str, body: str) -> str:
        u, q = self._user(uid), self._question(qid)
        if not body.strip():
            raise QaError("answer is empty")
        self._allow(u, Action.ANSWER, q.tags)
        a = Answer(self._next("A"), u, body.strip(), self._tick(), parent=q)
        with self._ids:
            self._posts[a.id] = a
        with q.lock:
            q.answers.append(a)
            q.followers.add(u.id)
            targets = [f for f in q.followers if f != u.id]
        self._fire([(f, f"{u.name} answered {q.id}: {q.title}") for f in sorted(targets)])
        return a.id

    def comment(self, uid: str, post_id: str, text: str) -> str:
        u, p = self._user(uid), self._post(post_id)
        self._allow(u, Action.COMMENT, p.question().tags)
        c = Comment(self._next("C"), u, text, self._tick())
        with p.lock:
            p.comments.append(c)
        if p.author.id != u.id:
            self._fire([(p.author.id, f"{u.name} commented on {p.id}")])
        return c.id

    def vote(self, uid: str, post_id: str, v: VoteType) -> None:
        u, p = self._user(uid), self._post(post_id)
        if p.author.id == u.id:
            raise QaError(f"{u.name} cannot vote on own post")
        self._allow(u, Action.UPVOTE if v is VoteType.UP else Action.DOWNVOTE, p.question().tags)
        with p.lock:
            old = p.votes.get(u.id)
            if old is v:
                raise QaError(f"{u.name} already voted {v.name} on {p.id}")
            p.votes[u.id] = v
            if old is not None:
                self._apply(u, p, old, -1)
            self._apply(u, p, v, +1)

    def retract_vote(self, uid: str, post_id: str) -> None:
        u, p = self._user(uid), self._post(post_id)
        with p.lock:
            old = p.votes.pop(u.id, None)
            if old is None:
                raise QaError(f"{u.name} has no vote on {p.id}")
            self._apply(u, p, old, -1)

    def accept(self, uid: str, answer_id: str) -> None:
        u, p = self._user(uid), self._post(answer_id)
        if not isinstance(p, Answer):
            raise QaError(f"{answer_id} is not an answer")
        q = p.parent
        if q.author.id != u.id:
            raise QaError(f"only {q.author.name} can accept answers on {q.id}")
        self._allow(u, Action.ACCEPT, q.tags)
        events = []
        with q.lock:
            old = q.accepted
            if old is p:
                return
            old_paid = old is not None and old.author.id != u.id    # own answers earn nothing
            new_pays = p.author.id != u.id
            if old_paid:
                old.author.add_reputation(-self._rep.accepted)
            u.add_reputation(self._rep.acceptor * (int(new_pays) - int(old_paid)))
            q.accepted = p
            if new_pays:
                p.author.add_reputation(self._rep.accepted)
                events.append((p.author.id, f"your answer {p.id} was accepted on {q.id}"))
        self._fire(events)

    def feed(self, uid: str, flt: QuestionFilter = everything) -> List[Question]:
        u = self._user(uid)
        mine = any_topic(u.topics)
        with self._ids:
            qs = list(self._questions.values())
        return sorted((q for q in qs if mine(q) and flt(q)), key=lambda q: -q.created_at)

    def search(self, text: str, flt: QuestionFilter = everything) -> List[Question]:
        with self._ids:
            ids: Optional[Set[str]] = None
            for w in words(text):
                hit = self._index.get(w, set())
                ids = set(hit) if ids is None else ids & hit
            pool = list(self._questions.values()) if ids is None else [self._questions[i] for i in ids]
        return sorted((q for q in pool if flt(q)), key=lambda q: (-q.score(), -q.created_at))

    def show(self, qid: str) -> str:
        q = self._question(qid)
        lines = [f"{q.id} [{q.score()}] {q.title} {sorted(q.tags)} by {q.author.name}"]
        for a in answer_order(q):
            mark = "* " if a is q.accepted else "  "
            lines.append(f"   {mark}{a.id} [{a.score()}] {a.body} - {a.author.name}")
            lines.extend(f"        {c.author.name}: {c.text}" for c in a.comments)
        return "\n".join(lines)

    def reputation_of(self, uid: str) -> int:
        return self._user(uid).reputation

    def score_of(self, post_id: str) -> int:
        return self._post(post_id).score()

    # ----- helpers -----
    def _apply(self, voter: User, p: Post, v: VoteType, sign: int) -> None:
        is_q = isinstance(p, Question)
        p.author.add_reputation(sign * self._rep.for_author(is_q, v))
        voter.add_reputation(sign * self._rep.for_voter(is_q, v))

    def _allow(self, u: User, action: Action, tags: Iterable[str]) -> None:
        for r in self._rules:
            r.check(u, action, tags)

    def _fire(self, events) -> None:           # no lock held here
        for uid, msg in events:
            for listener in self._listeners:
                listener(self._users[uid].name, msg)

    def _user(self, uid: str) -> User:
        try:
            return self._users[uid]
        except KeyError:
            raise QaError(f"unknown user {uid}") from None

    def _question(self, qid: str) -> Question:
        try:
            return self._questions[qid]
        except KeyError:
            raise QaError(f"unknown question {qid}") from None

    def _post(self, pid: str) -> Post:
        try:
            return self._posts[pid]
        except KeyError:
            raise QaError(f"unknown post {pid}") from None

"""Demo driver: the Flipkart Overflow test sequence, Stack Overflow reputation, and a vote race."""
import threading

from qa import (QaError, QaService, ReputationPrivileges, StackOverflowReputation,
                TopicSubscriptionRule, VoteType, answered, any_topic)


def check(cond, msg="check failed"):
    """Like assert, but never stripped by python -O: the call inside it always runs."""
    if not cond:
        raise AssertionError(msg)


def ids(qs):
    return [q.id for q in qs]


def rejected(fn, label):
    try:
        fn()
    except QaError as e:
        print(f"{label} -> rejected: {e}")
        return True
    return False


def demo() -> None:
    inbox = []

    def console(user, msg):
        print(f"  [notify {user}] {msg}")
        inbox.append(f"{user}: {msg}")

    print("== Flipkart Overflow: topics, feed, filters ==")
    flip = QaService(StackOverflowReputation(), [TopicSubscriptionRule()])
    flip.add_listener(console)
    asha, bala, chen = flip.register("asha"), flip.register("bala"), flip.register("chen")
    flip.subscribe(asha, "java", "hadoop", "jdk")
    q1 = flip.ask(asha, "What are new open source JDKs?", "Looking beyond Oracle.", "java", "jdk")
    q2 = flip.ask(asha, "Does Hadoop work on JDK 11?", "Cluster upgrade.", "hadoop", "jdk")
    print("feed asha         ", ids(flip.feed(asha)))
    print("feed asha [java]  ", ids(flip.feed(asha, any_topic(["java"]))))
    print("feed asha answered", ids(flip.feed(asha, answered(True))))
    check(ids(flip.feed(asha, any_topic(["java"]))) == [q1])
    flip.subscribe(bala, "apache", "hadoop")
    check(ids(flip.feed(bala)) == [q2])
    check(rejected(lambda: flip.ask(bala, "Best JVM flags?", "", "java"), "bala asks in [java]"))
    q3 = flip.ask(bala, "Does Apache Spark stream from HDFS?", "", "apache", "hadoop")
    a1 = flip.answer(bala, q2, "Yes, Hadoop 3 and above supports it.")
    print("feed bala         ", ids(flip.feed(bala)))
    check(ids(flip.feed(asha, answered(True))) == [q2])
    flip.subscribe(chen, "apache", "hadoop", "java")
    check(len(flip.feed(chen)) == 3)
    flip.vote(chen, q2, VoteType.UP)
    flip.vote(chen, a1, VoteType.UP)
    check(rejected(lambda: flip.vote(chen, a1, VoteType.UP), "chen upvotes A1 again"))
    a2 = flip.answer(chen, q2, "Only from Hadoop 3.3 for runtime.")
    flip.vote(bala, a2, VoteType.UP)
    flip.vote(asha, a2, VoteType.UP)
    flip.accept(asha, a1)
    print(flip.show(q2))
    check(flip.show(q2).splitlines()[1].strip().startswith("* " + a1))
    check("asha: bala answered Q2: Does Hadoop work on JDK 11?" in inbox)
    _ = q3

    print("\n== Stack Overflow: reputation and privileges ==")
    so = QaService(StackOverflowReputation(), [ReputationPrivileges()])
    ann, ben, cat, dev = (so.register(n) for n in ("ann", "ben", "cat", "dev"))
    sq = so.ask(ann, "Why does my C struct have padding?", "sizeof is bigger than the sum", "c", "memory")
    sa = so.answer(ben, sq, "Alignment: each field starts at a multiple of its size.")
    check(rejected(lambda: so.vote(cat, sa, VoteType.UP), "cat (rep 1) upvotes"))
    so.bonus(cat, 100)
    so.bonus(dev, 200)
    so.vote(cat, sa, VoteType.UP)
    so.vote(dev, sa, VoteType.UP)
    so.vote(dev, sa, VoteType.DOWN)
    print("dev flips to DOWN: ben", so.reputation_of(ben), "dev", so.reputation_of(dev))
    check((so.reputation_of(ben), so.reputation_of(dev)) == (9, 200))
    so.accept(ann, sa)
    print("ann accepts: ben", so.reputation_of(ben), "ann", so.reputation_of(ann))
    check((so.reputation_of(ben), so.reputation_of(ann)) == (24, 3))
    print("search 'struct padding':", ids(so.search("struct padding")))

    print("\n== Concurrency: 100 voters at once ==")
    rush = QaService(StackOverflowReputation(), [])
    asker, author = rush.register("asker"), rush.register("author")
    rq = rush.ask(asker, "Hot question", "", "c")
    ra = rush.answer(author, rq, "Hot answer")
    voters = [rush.register(f"v{i}") for i in range(100)]
    go = threading.Event()

    def worker(v):
        go.wait()
        rush.vote(v, ra, VoteType.UP)

    threads = [threading.Thread(target=worker, args=(v,)) for v in voters]
    for t in threads:
        t.start()
    go.set()
    for t in threads:
        t.join()
    print("answer score", rush.score_of(ra), "author rep", rush.reputation_of(author))
    check(rush.score_of(ra) == 100 and rush.reputation_of(author) == 1001)
    print("demo OK")


if __name__ == "__main__":
    demo()

import threading
import unittest

from qa import (QaError, QaService, ReputationPrivileges, StackOverflowReputation,
                TopicSubscriptionRule, VoteType, answered, any_topic)


class FlipkartOverflowTest(unittest.TestCase):
    def setUp(self):
        self.inbox = []
        self.svc = QaService(StackOverflowReputation(), [TopicSubscriptionRule()])
        self.svc.add_listener(lambda u, m: self.inbox.append(f"{u}: {m}"))
        self.a, self.b = self.svc.register("asha"), self.svc.register("bala")
        self.svc.subscribe(self.a, "java", "jdk")
        self.svc.subscribe(self.b, "jdk")
        self.q = self.svc.ask(self.a, "JDK 21 features?", "virtual threads", "java", "jdk")

    def test_feed_filters(self):
        self.assertEqual([q.id for q in self.svc.feed(self.b)], [self.q])
        self.assertEqual(self.svc.feed(self.b, any_topic(["spring"])), [])
        self.assertEqual(self.svc.feed(self.a, answered(True)), [])
        self.svc.answer(self.b, self.q, "Virtual threads and records patterns")
        self.assertEqual(len(self.svc.feed(self.a, answered(True))), 1)

    def test_subscription_rules(self):
        with self.assertRaises(QaError):
            self.svc.ask(self.b, "Spring?", "", "spring")
        self.svc.unsubscribe(self.b, "jdk")
        with self.assertRaises(QaError):
            self.svc.answer(self.b, self.q, "no longer allowed")

    def test_no_duplicate_or_self_vote(self):
        a1 = self.svc.answer(self.b, self.q, "Loom")
        self.svc.vote(self.a, a1, VoteType.UP)
        with self.assertRaises(QaError):
            self.svc.vote(self.a, a1, VoteType.UP)
        with self.assertRaises(QaError):
            self.svc.vote(self.b, a1, VoteType.UP)
        self.assertEqual(self.svc.score_of(a1), 1)

    def test_answer_order_and_accept(self):
        a1 = self.svc.answer(self.b, self.q, "first")
        c = self.svc.register("chen")
        self.svc.subscribe(c, "java")
        a2 = self.svc.answer(c, self.q, "second")
        self.svc.vote(self.a, a2, VoteType.UP)
        self.assertIn(a2, self.svc.show(self.q).splitlines()[1])
        with self.assertRaises(QaError):
            self.svc.accept(c, a1)                       # only the asker
        self.svc.accept(self.a, a1)
        self.assertIn("* " + a1, self.svc.show(self.q).splitlines()[1])
        self.assertIn("bala: your answer A1 was accepted on Q1", self.inbox)


class ReputationTest(unittest.TestCase):
    def setUp(self):
        self.svc = QaService(StackOverflowReputation(), [ReputationPrivileges()])
        self.ann, self.ben, self.cat = (self.svc.register(n) for n in ("ann", "ben", "cat"))
        self.q = self.svc.ask(self.ann, "Padding in structs", "", "c")
        self.a = self.svc.answer(self.ben, self.q, "alignment")

    def test_privileges(self):
        with self.assertRaises(QaError):
            self.svc.vote(self.cat, self.a, VoteType.UP)
        self.svc.bonus(self.cat, 14)                       # now 15
        self.svc.vote(self.cat, self.a, VoteType.UP)
        with self.assertRaises(QaError):
            self.svc.vote(self.cat, self.a, VoteType.DOWN)    # needs 125

    def test_flip_retract_and_floor(self):
        self.svc.bonus(self.cat, 200)
        self.svc.vote(self.cat, self.a, VoteType.UP)
        self.assertEqual(self.svc.reputation_of(self.ben), 11)
        self.svc.vote(self.cat, self.a, VoteType.DOWN)
        self.assertEqual(self.svc.reputation_of(self.ben), 1)        # 11 - 10 - 2, floored at 1
        self.assertEqual(self.svc.reputation_of(self.cat), 200)
        self.svc.retract_vote(self.cat, self.a)
        self.assertEqual(self.svc.reputation_of(self.cat), 201)

    def test_moving_accept(self):
        self.svc.bonus(self.cat, 10)
        a2 = self.svc.answer(self.cat, self.q, "offsetof")
        self.svc.accept(self.ann, self.a)
        self.svc.accept(self.ann, a2)
        self.assertEqual(self.svc.reputation_of(self.ben), 1)
        self.assertEqual(self.svc.reputation_of(self.cat), 26)
        self.assertEqual(self.svc.reputation_of(self.ann), 3)        # acceptor bonus counted once

    def test_search(self):
        self.assertEqual([q.id for q in self.svc.search("structs padding")], [self.q])
        self.assertEqual(self.svc.search("golang"), [])


class ConcurrencyTest(unittest.TestCase):
    def test_same_user_many_clicks(self):
        svc = QaService(StackOverflowReputation(), [])
        a, b = svc.register("a"), svc.register("b")
        q = svc.ask(a, "hot", "", "c")
        dup = []
        go = threading.Event()

        def click():
            go.wait()
            try:
                svc.vote(b, q, VoteType.UP)
            except QaError:
                dup.append(1)

        ts = [threading.Thread(target=click) for _ in range(20)]
        for t in ts:
            t.start()
        go.set()
        for t in ts:
            t.join()
        self.assertEqual(svc.score_of(q), 1)
        self.assertEqual(len(dup), 19)
        self.assertEqual(svc.reputation_of(a), 11)


if __name__ == "__main__":
    unittest.main()

// Stack Overflow / Flipkart Overflow Q&A platform - C++14, single file.
// Locking, as in the Java version: each Post has its own mutex (votes, comments, accepted answer), so votes on
// different posts never wait for each other; User::reputation is a std::atomic<int> updated with a
// compare-exchange loop (like atomic_cmpxchg() in the kernel); one service mutex mu_ guards the maps, the word
// index and the id counters. Lock order: mu_ before a post lock, never the reverse. Listeners run after unlock.
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run-cpp.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <algorithm>
#include <atomic>
#include <cctype>
#include <cstdlib>
#include <functional>
#include <iterator>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <set>
#include <sstream>

#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif

struct QaError : std::runtime_error { explicit QaError(const std::string& m) : std::runtime_error(m) {} };

enum class VoteType { UP, DOWN };
enum class Action { ASK, ANSWER, COMMENT, UPVOTE, DOWNVOTE, ACCEPT };

struct User {
    std::string id, name;
    std::set<std::string> topics;
    std::atomic<int> reputation{1};
    User(std::string i, std::string n) : id(std::move(i)), name(std::move(n)) {}
    void addReputation(int d) {                 // lock-free, floor at 1: retry until no other thread got in between
        int cur = reputation.load();
        while (!reputation.compare_exchange_weak(cur, std::max(1, cur + d))) {}
    }
};

struct Question;
struct Post {                                   // base of Question and Answer
    std::string id; User* author; std::string body; long createdAt;
    std::map<std::string, VoteType> votes;      // voter id -> vote
    std::vector<std::pair<User*, std::string>> comments;
    mutable Mutex mu;                           // guards votes, comments (and answers/accepted on a Question)
    Post(std::string i, User* a, std::string b, long t) : id(std::move(i)), author(a), body(std::move(b)), createdAt(t) {}
    virtual ~Post() {}
    virtual Question* question() = 0;
    virtual bool isQuestion() const = 0;
    int score() const {
        std::lock_guard<Mutex> g(mu);
        int s = 0;
        for (const auto& kv : votes) s += kv.second == VoteType::UP ? 1 : -1;
        return s;
    }
};

struct Answer;
struct Question : Post {
    std::string title; std::set<std::string> tags;
    std::vector<Answer*> answers; std::set<std::string> followers; Answer* accepted = nullptr;
    Question(std::string i, User* a, std::string ti, std::string b, std::set<std::string> tg, long t)
        : Post(std::move(i), a, std::move(b), t), title(std::move(ti)), tags(std::move(tg)) {}
    Question* question() override { return this; }
    bool isQuestion() const override { return true; }
};

struct Answer : Post {
    Question* parent;
    Answer(std::string i, User* a, Question* q, std::string b, long t) : Post(std::move(i), a, std::move(b), t), parent(q) {}
    Question* question() override { return parent; }
    bool isQuestion() const override { return false; }
};

// ---------- strategies and policies ----------
struct ReputationPolicy {
    virtual ~ReputationPolicy() {}
    virtual int forAuthor(bool isQ, VoteType v) const = 0;
    virtual int forVoter(bool isQ, VoteType v) const = 0;
    virtual int forAccepted() const { return 15; }
    virtual int forAcceptor() const { return 2; }
};
struct StackOverflowReputation : ReputationPolicy {
    int forAuthor(bool, VoteType v) const override { return v == VoteType::UP ? 10 : -2; }
    int forVoter(bool isQ, VoteType v) const override { return (!isQ && v == VoteType::DOWN) ? -1 : 0; }
};

struct PermissionRule {                         // like a netfilter hook: each may drop the request
    virtual ~PermissionRule() {}
    virtual void check(const User& u, Action a, const std::set<std::string>& tags) const = 0;
};
struct ReputationPrivileges : PermissionRule {
    void check(const User& u, Action a, const std::set<std::string>&) const override {
        if (a == Action::UPVOTE && u.reputation < 15) throw QaError(u.name + " needs 15 reputation to upvote");
        if (a == Action::DOWNVOTE && u.reputation < 125) throw QaError(u.name + " needs 125 reputation to downvote");
    }
};
struct TopicSubscriptionRule : PermissionRule {
    void check(const User& u, Action a, const std::set<std::string>& tags) const override {
        if (a == Action::ACCEPT) return;
        for (const auto& t : tags) if (u.topics.count(t)) return;
        throw QaError(u.name + " is not subscribed to a topic of this question");
    }
};

using Filter = std::function<bool(const Question&)>;
static Filter anyTopic(std::set<std::string> want) {
    return [want](const Question& q) { for (const auto& t : q.tags) if (want.count(t)) return true; return false; };
}
static Filter answered(bool yes) { return [yes](const Question& q) { return q.answers.empty() != yes; }; }
static Filter everything() { return [](const Question&) { return true; }; }

static std::vector<std::string> words(const std::string& text) {
    std::vector<std::string> r; std::string w;
    for (char c : text + " ") {
        unsigned char u = static_cast<unsigned char>(c);
        if (std::isalnum(u) || c == '+' || c == '#') w += static_cast<char>(std::tolower(u));
        else { if (w.size() > 1) r.push_back(w); w.clear(); }
    }
    return r;
}

// ---------- service (facade) ----------
class QaService {
public:
    using Listener = std::function<void(const std::string&, const std::string&)>;
    QaService(std::unique_ptr<ReputationPolicy> rep, std::vector<std::unique_ptr<PermissionRule>> rules)
        : rep_(std::move(rep)), rules_(std::move(rules)) {}

    void addListener(Listener l) { listeners_.push_back(std::move(l)); }

    std::string registerUser(const std::string& name) {
        std::lock_guard<Mutex> g(mu_);
        std::string id = "U" + std::to_string(nu_++);
        users_.emplace(std::piecewise_construct, std::forward_as_tuple(id), std::forward_as_tuple(id, name));
        return id;
    }
    void subscribe(const std::string& uid, std::initializer_list<std::string> ts) {
        std::lock_guard<Mutex> g(mu_);
        for (auto& t : ts) user(uid).topics.insert(t);
    }
    void bonus(const std::string& uid, int pts) { userRef(uid).addReputation(pts); }

    std::string ask(const std::string& uid, const std::string& title, const std::string& body, std::set<std::string> tags) {
        std::lock_guard<Mutex> g(mu_);
        User& u = user(uid);
        if (title.empty()) throw QaError("title is empty");
        if (tags.empty()) throw QaError("a question needs at least one tag");
        allow(u, Action::ASK, tags);
        std::string id = "Q" + std::to_string(nq_++);
        std::unique_ptr<Question> q(new Question(id, &u, title, body, tags, ++clock_));
        q->followers.insert(u.id);
        for (const auto& w : words(title + " " + body)) index_[w].insert(id);
        questions_[id] = q.get();
        posts_[id] = std::move(q);
        return id;
    }

    std::string answer(const std::string& uid, const std::string& qid, const std::string& body) {
        Notes notes;
        std::string id;
        {
            std::lock_guard<Mutex> g(mu_);
            User& u = user(uid);
            Question& q = question(qid);
            if (body.empty()) throw QaError("answer is empty");
            allow(u, Action::ANSWER, q.tags);
            id = "A" + std::to_string(na_++);
            std::unique_ptr<Answer> a(new Answer(id, &u, &q, body, ++clock_));
            {
                std::lock_guard<Mutex> pg(q.mu);                 // mu_ then the post lock: the one lock order
                q.answers.push_back(a.get());
                q.followers.insert(u.id);
                for (const auto& f : q.followers)
                    if (f != u.id) notes.push_back({users_.at(f).name, u.name + " answered " + q.id + ": " + q.title});
            }
            posts_[id] = std::move(a);
        }
        fire(notes);                                             // after every unlock
        return id;
    }

    void comment(const std::string& uid, const std::string& pid, const std::string& text) {
        User* up; Post* pp;
        {
            std::lock_guard<Mutex> g(mu_);
            up = &user(uid); pp = &post(pid);
            allow(*up, Action::COMMENT, pp->question()->tags);
        }
        { std::lock_guard<Mutex> g(pp->mu); pp->comments.push_back({up, text}); }
        if (pp->author != up) fire({{pp->author->name, up->name + " commented on " + pp->id}});
    }

    void vote(const std::string& uid, const std::string& pid, VoteType v) {
        User* up; Post* pp;
        {
            std::lock_guard<Mutex> g(mu_);                       // map lookup + permission check only
            up = &user(uid); pp = &post(pid);
            if (pp->author == up) throw QaError(up->name + " cannot vote on own post");
            allow(*up, v == VoteType::UP ? Action::UPVOTE : Action::DOWNVOTE, pp->question()->tags);
        }
        User& u = *up; Post& p = *pp;
        std::lock_guard<Mutex> g(p.mu);                          // read old vote + write new vote: one step
        auto it = p.votes.find(u.id);
        if (it != p.votes.end() && it->second == v) throw QaError(u.name + " already voted on " + p.id);
        if (it != p.votes.end()) apply(u, p, it->second, -1);
        p.votes[u.id] = v;
        apply(u, p, v, +1);
    }

    void retractVote(const std::string& uid, const std::string& pid) {
        User* up; Post* pp;
        { std::lock_guard<Mutex> g(mu_); up = &user(uid); pp = &post(pid); }
        User& u = *up; Post& p = *pp;
        std::lock_guard<Mutex> g(p.mu);
        auto it = p.votes.find(u.id);
        if (it == p.votes.end()) throw QaError(u.name + " has no vote on " + p.id);
        VoteType old = it->second;
        p.votes.erase(it);
        apply(u, p, old, -1);
    }

    void accept(const std::string& uid, const std::string& aid) {
        User* up; Post* pp;
        { std::lock_guard<Mutex> g(mu_); up = &user(uid); pp = &post(aid); }
        User& u = *up; Post& p = *pp;
        if (p.isQuestion()) throw QaError(aid + " is not an answer");
        Answer& a = static_cast<Answer&>(p);
        Question& q = *a.parent;
        if (q.author != &u) throw QaError("only " + q.author->name + " can accept answers on " + q.id);
        bool newPays;
        {
            std::lock_guard<Mutex> g(q.mu);                      // the question's lock guards q.accepted
            Answer* old = q.accepted;
            if (old == &a) return;
            bool oldPaid = old && old->author != &u;
            newPays = a.author != &u;
            if (oldPaid) old->author->addReputation(-rep_->forAccepted());
            u.addReputation(rep_->forAcceptor() * ((newPays ? 1 : 0) - (oldPaid ? 1 : 0)));
            q.accepted = &a;
            if (newPays) a.author->addReputation(rep_->forAccepted());
        }
        if (newPays) fire({{a.author->name, "your answer " + a.id + " was accepted on " + q.id}});
    }

    std::vector<const Question*> feed(const std::string& uid, const Filter& f) {
        std::lock_guard<Mutex> g(mu_);
        Filter mine = anyTopic(user(uid).topics);
        std::vector<const Question*> r;
        for (const auto& kv : questions_) if (mine(*kv.second) && f(*kv.second)) r.push_back(kv.second);
        std::sort(r.begin(), r.end(), [](const Question* a, const Question* b) { return a->createdAt > b->createdAt; });
        return r;
    }

    std::vector<const Question*> search(const std::string& text) {
        std::lock_guard<Mutex> g(mu_);
        std::set<std::string> ids; bool first = true;
        for (const auto& w : words(text)) {
            const std::set<std::string>& hit = index_[w];
            if (first) { ids = hit; first = false; continue; }
            std::set<std::string> both;
            std::set_intersection(ids.begin(), ids.end(), hit.begin(), hit.end(), std::inserter(both, both.begin()));
            ids.swap(both);
        }
        std::vector<const Question*> r;
        for (const auto& id : ids) r.push_back(questions_[id]);
        std::sort(r.begin(), r.end(), [](const Question* a, const Question* b) {
            return a->score() != b->score() ? a->score() > b->score() : a->createdAt > b->createdAt; });
        return r;
    }

    std::string show(const std::string& qid) {
        std::lock_guard<Mutex> g(mu_);
        Question& q = question(qid);
        std::vector<Answer*> as;
        Answer* acc;
        { std::lock_guard<Mutex> pg(q.mu); as = q.answers; acc = q.accepted; }
        std::stable_sort(as.begin(), as.end(), [acc](const Answer* a, const Answer* b) {
            if ((a == acc) != (b == acc)) return a == acc;
            if (a->score() != b->score()) return a->score() > b->score();
            return a->createdAt < b->createdAt;
        });
        std::ostringstream os;
        os << q.id << " [" << q.score() << "] " << q.title << " by " << q.author->name;
        for (const Answer* a : as) {
            os << "\n   " << (a == acc ? "* " : "  ") << a->id << " [" << a->score() << "] " << a->body << " - " << a->author->name;
            std::lock_guard<Mutex> ag(a->mu);
            for (const auto& c : a->comments) os << "\n        " << c.first->name << ": " << c.second;
        }
        return os.str();
    }

    int reputationOf(const std::string& uid) { return userRef(uid).reputation.load(); }
    int scoreOf(const std::string& pid) { Post* p; { std::lock_guard<Mutex> g(mu_); p = &post(pid); } return p->score(); }

private:
    void apply(User& voter, Post& p, VoteType v, int sign) {
        p.author->addReputation(sign * rep_->forAuthor(p.isQuestion(), v));
        voter.addReputation(sign * rep_->forVoter(p.isQuestion(), v));
    }
    void allow(const User& u, Action a, const std::set<std::string>& tags) { for (const auto& r : rules_) r->check(u, a, tags); }
    using Notes = std::vector<std::pair<std::string, std::string>>;   // (user name, message), sent after unlock
    void fire(const Notes& notes) { for (const auto& n : notes) for (auto& l : listeners_) l(n.first, n.second); }
    User& userRef(const std::string& id) { std::lock_guard<Mutex> g(mu_); return user(id); }
    User& user(const std::string& id) {
        auto it = users_.find(id);
        if (it == users_.end()) throw QaError("unknown user " + id);
        return it->second;
    }
    Question& question(const std::string& id) {
        auto it = questions_.find(id);
        if (it == questions_.end()) throw QaError("unknown question " + id);
        return *it->second;
    }
    Post& post(const std::string& id) {
        auto it = posts_.find(id);
        if (it == posts_.end()) throw QaError("unknown post " + id);
        return *it->second;
    }

    std::unique_ptr<ReputationPolicy> rep_;
    std::vector<std::unique_ptr<PermissionRule>> rules_;
    std::map<std::string, User> users_;
    std::map<std::string, std::unique_ptr<Post>> posts_;   // owns every question and answer
    std::map<std::string, Question*> questions_;
    std::map<std::string, std::set<std::string>> index_;   // word -> question ids
    std::vector<Listener> listeners_;
    long clock_ = 0;
    int nu_ = 1, nq_ = 1, na_ = 1;
    Mutex mu_;                                             // guards the maps, index_, clock_ and the id counters
};

static std::string ids(const std::vector<const Question*>& qs) {
    std::string s = "[";
    for (size_t i = 0; i < qs.size(); ++i) s += (i ? ", " : "") + qs[i]->id;
    return s + "]";

template <class F> static bool rejected(F f, const std::string& label) {
    try { f(); } catch (const QaError& e) { std::cout << label << " -> rejected: " << e.what() << "\n"; return true; }
    return false;
}
