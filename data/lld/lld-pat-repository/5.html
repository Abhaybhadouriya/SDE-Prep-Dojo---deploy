Code: a lending library in three layers

Each tab is a complete program that was compiled and run. Java shows a generic InMemoryRepository<T, ID> base so each concrete repository is a few lines. Python uses abstract base classes for two repositories and plain duck typing for the third. C++14 uses pure virtual interfaces (an ops table) and returns copies so callers never hold pointers into the repository's storage. Files live in lld-code/lld-pat-repository/.

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;
import java.util.stream.Collectors;

// Repository + Service layer in one file (a real project splits it into packages:
// model/ repository/ service/ exception/ Main.java).
public class LibraryDemo {

    // ---------- model (entities): data + their own invariants, no storage code ----------
    enum BookStatus { AVAILABLE, LENT }

    static final class Book {
        private final String id; private final String title; private BookStatus status = BookStatus.AVAILABLE;
        Book(String id, String title) { this.id = id; this.title = title; }
        String id() { return id; }
        String title() { return title; }
        BookStatus status() { return status; }
        void markLent() {
            if (status == BookStatus.LENT) throw new IllegalStateException(id + " already lent");
            status = BookStatus.LENT;
        }
        void markReturned() { status = BookStatus.AVAILABLE; }
    }

    record Member(String id, String name) { }

    static final class Loan {
        private final String id; private final String memberId; private final String bookId;
        private boolean open = true;
        Loan(String id, String memberId, String bookId) { this.id = id; this.memberId = memberId; this.bookId = bookId; }
        String id() { return id; }
        String memberId() { return memberId; }
        String bookId() { return bookId; }
        boolean isOpen() { return open; }
        void close() { open = false; }
        @Override public String toString() { return "Loan[" + id + " " + memberId + "->" + bookId + (open ? " open]" : " closed]"); }
    }

    // ---------- exceptions: named business errors ----------
    static class NotFoundException extends RuntimeException { NotFoundException(String m) { super(m); } }
    static class LimitExceededException extends RuntimeException { LimitExceededException(String m) { super(m); } }

    // ---------- repository interfaces: collection-like, in domain words ----------
    interface Repository<T, ID> {
        Optional<T> findById(ID id);
        List<T> findAll();
        T save(T entity);                   // insert or update
        void deleteById(ID id);
    }
    interface BookRepository extends Repository<Book, String> {
        List<Book> findByStatus(BookStatus status);
    }
    interface MemberRepository extends Repository<Member, String> { }
    interface LoanRepository extends Repository<Loan, String> {
        List<Loan> findOpenByMember(String memberId);
        String nextId();
    }

    // ---------- in-memory implementations (what a machine-coding round expects) ----------
    static class InMemoryRepository<T, ID> implements Repository<T, ID> {
        protected final Map<ID, T> store = new ConcurrentHashMap<>();
        private final java.util.function.Function<T, ID> idOf;
        InMemoryRepository(java.util.function.Function<T, ID> idOf) { this.idOf = idOf; }
        public Optional<T> findById(ID id) { return Optional.ofNullable(store.get(id)); }
        public List<T> findAll() { return new ArrayList<>(store.values()); }    // a copy, never the map
        public T save(T e) { store.put(idOf.apply(e), e); return e; }
        public void deleteById(ID id) { store.remove(id); }
    }
    static final class InMemoryBookRepository extends InMemoryRepository<Book, String> implements BookRepository {
        InMemoryBookRepository() { super(Book::id); }
        public List<Book> findByStatus(BookStatus s) {
            return store.values().stream().filter(b -> b.status() == s)
                        .sorted(Comparator.comparing(Book::id)).collect(Collectors.toList());
        }
    }
    static final class InMemoryMemberRepository extends InMemoryRepository<Member, String> implements MemberRepository {
        InMemoryMemberRepository() { super(Member::id); }
    }
    static final class InMemoryLoanRepository extends InMemoryRepository<Loan, String> implements LoanRepository {
        private final AtomicLong seq = new AtomicLong();
        InMemoryLoanRepository() { super(Loan::id); }
        public List<Loan> findOpenByMember(String memberId) {
            return store.values().stream().filter(l -> l.isOpen() && l.memberId().equals(memberId))
                        .collect(Collectors.toList());
        }
        public String nextId() { return "L" + seq.incrementAndGet(); }
    }

    // ---------- service: the use cases and business rules; depends only on interfaces ----------
    static final class LendingService {
        static final int MAX_OPEN_LOANS = 2;
        private final BookRepository books; private final MemberRepository members; private final LoanRepository loans;
        LendingService(BookRepository books, MemberRepository members, LoanRepository loans) {
            this.books = books; this.members = members; this.loans = loans;    // constructor injection
        }
        synchronized Loan borrow(String memberId, String bookId) {          // one lock: simple and correct
            // <-- the pattern: the service sees only repository interfaces
            Member m = members.findById(memberId).orElseThrow(() -> new NotFoundException("member " + memberId));
            Book b = books.findById(bookId).orElseThrow(() -> new NotFoundException("book " + bookId));
            if (loans.findOpenByMember(m.id()).size() >= MAX_OPEN_LOANS)
                throw new LimitExceededException(m.name() + " already has " + MAX_OPEN_LOANS + " books");
            b.markLent();                                                  // entity guards its own state
            books.save(b);
            return loans.save(new Loan(loans.nextId(), m.id(), b.id()));
        }
        synchronized void giveBack(String loanId) {
            Loan l = loans.findById(loanId).orElseThrow(() -> new NotFoundException("loan " + loanId));
            l.close(); loans.save(l);
            Book b = books.findById(l.bookId()).orElseThrow();
            b.markReturned(); books.save(b);
        }
        List<String> availableTitles() {
            return books.findByStatus(BookStatus.AVAILABLE).stream().map(Book::title).collect(Collectors.toList());
        }
    }

    // ---------- driver: the composition root wires concrete classes once ----------
    public static void main(String[] args) {
        BookRepository books = new InMemoryBookRepository();
        MemberRepository members = new InMemoryMemberRepository();
        LoanRepository loans = new InMemoryLoanRepository();
        LendingService svc = new LendingService(books, members, loans);

        books.save(new Book("B1", "Linux Device Drivers"));
        books.save(new Book("B2", "TCP/IP Illustrated"));
        books.save(new Book("B3", "Clean Code"));
        members.save(new Member("M1", "Asha"));

        Loan l1 = svc.borrow("M1", "B1");
        Loan l2 = svc.borrow("M1", "B2");
        System.out.println("borrowed " + l1 + " and " + l2);
        System.out.println("available: " + svc.availableTitles());
        tryRun(() -> svc.borrow("M1", "B3"));     // limit
        tryRun(() -> svc.borrow("M9", "B3"));     // unknown member
        svc.giveBack(l1.id());
        System.out.println("after return: " + svc.availableTitles());
        tryRun(() -> System.out.println("borrowed " + svc.borrow("M1", "B3")));
    }
    static void tryRun(Runnable r) {
        try { r.run(); } catch (RuntimeException e) { System.out.println("error: " + e.getClass().getSimpleName() + ": " + e.getMessage()); }
    }
}

findAll() returns a copy of the values. Returning the live map would let callers bypass the repository and mutate storage directly, the same bug as handing out a pointer to a kernel-internal list without its lock.

"""Repository + service layer: entities, repository interfaces, in-memory repos, a service, a driver."""
from __future__ import annotations

import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass
from itertools import count


# ---------- model ----------
@dataclass
class Book:
    id: str
    title: str
    lent: bool = False


@dataclass(frozen=True)
class Member:
    id: str
    name: str


@dataclass
class Loan:
    id: str
    member_id: str
    book_id: str
    open: bool = True


class NotFound(Exception):
    pass


class LimitExceeded(Exception):
    pass


# ---------- repository interfaces ----------
class BookRepository(ABC):
    @abstractmethod
    def find_by_id(self, book_id: str) -> Book | None: ...
    @abstractmethod
    def save(self, book: Book) -> Book: ...
    @abstractmethod
    def find_available(self) -> list[Book]: ...


class LoanRepository(ABC):
    @abstractmethod
    def find_by_id(self, loan_id: str) -> Loan | None: ...
    @abstractmethod
    def save(self, loan: Loan) -> Loan: ...
    @abstractmethod
    def open_loans_of(self, member_id: str) -> list[Loan]: ...
    @abstractmethod
    def next_id(self) -> str: ...


# ---------- in-memory implementations ----------
class InMemoryBookRepository(BookRepository):
    def __init__(self):
        self._rows: dict[str, Book] = {}

    def find_by_id(self, book_id):
        return self._rows.get(book_id)

    def save(self, book):
        self._rows[book.id] = book
        return book

    def find_available(self):
        return sorted((b for b in self._rows.values() if not b.lent), key=lambda b: b.id)


class InMemoryLoanRepository(LoanRepository):
    def __init__(self):
        self._rows: dict[str, Loan] = {}
        self._seq = count(1)

    def find_by_id(self, loan_id):
        return self._rows.get(loan_id)

    def save(self, loan):
        self._rows[loan.id] = loan
        return loan

    def open_loans_of(self, member_id):
        return [l for l in self._rows.values() if l.open and l.member_id == member_id]

    def next_id(self):
        return f"L{next(self._seq)}"


class InMemoryMemberRepository:
    """Duck-typed: Python does not need an ABC to be swappable."""
    def __init__(self):
        self._rows: dict[str, Member] = {}

    def find_by_id(self, member_id):
        return self._rows.get(member_id)

    def save(self, m):
        self._rows[m.id] = m
        return m


# ---------- service ----------
class LendingService:
    MAX_OPEN = 2

    def __init__(self, books: BookRepository, members, loans: LoanRepository):
        self._books, self._members, self._loans = books, members, loans
        self._lock = threading.Lock()

    def borrow(self, member_id: str, book_id: str) -> Loan:
        with self._lock:
            member = self._members.find_by_id(member_id) or self._missing("member", member_id)
            book = self._books.find_by_id(book_id) or self._missing("book", book_id)
            if len(self._loans.open_loans_of(member.id)) >= self.MAX_OPEN:
                raise LimitExceeded(f"{member.name} already has {self.MAX_OPEN} books")
            if book.lent:
                raise LimitExceeded(f"{book.id} already lent")
            book.lent = True
            self._books.save(book)
            return self._loans.save(Loan(self._loans.next_id(), member.id, book.id))

    def give_back(self, loan_id: str) -> None:
        with self._lock:
            loan = self._loans.find_by_id(loan_id) or self._missing("loan", loan_id)
            loan.open = False
            self._loans.save(loan)
            book = self._books.find_by_id(loan.book_id)
            book.lent = False
            self._books.save(book)

    def available_titles(self) -> list[str]:
        return [b.title for b in self._books.find_available()]

    @staticmethod
    def _missing(kind, key):
        raise NotFound(f"{kind} {key}")


def attempt(fn):
    try:
        return fn()
    except (NotFound, LimitExceeded) as e:
        print(f"error: {type(e).__name__}: {e}")


if __name__ == "__main__":
    books, members, loans = InMemoryBookRepository(), InMemoryMemberRepository(), InMemoryLoanRepository()
    svc = LendingService(books, members, loans)
    for b in (Book("B1", "Linux Device Drivers"), Book("B2", "TCP/IP Illustrated"), Book("B3", "Clean Code")):
        books.save(b)
    members.save(Member("M1", "Asha"))

    l1 = svc.borrow("M1", "B1")
    l2 = svc.borrow("M1", "B2")
    print("borrowed", l1.id, l2.id)
    print("available:", svc.available_titles())
    attempt(lambda: svc.borrow("M1", "B3"))
    attempt(lambda: svc.borrow("M9", "B3"))
    svc.give_back(l1.id)
    print("after return:", svc.available_titles())
    print("borrowed", attempt(lambda: svc.borrow("M1", "B3")))

In Python the ABC is optional. It documents the contract and makes a missing method fail at construction time instead of at the first call. InMemoryMemberRepository shows that duck typing alone is enough to be swappable.

// Repository + service layer in C++14. Build: g++ -std=c++14 -Wall library.cpp -o library
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

// ---------- model ----------
struct Book { std::string id, title; bool lent = false; };
struct Loan { std::string id, memberId, bookId; bool open = true; };

struct NotFound : std::runtime_error { using std::runtime_error::runtime_error; };
struct LimitExceeded : std::runtime_error { using std::runtime_error::runtime_error; };

// ---------- repository interfaces (pure virtual = an ops table) ----------
class BookRepository {
public:
    virtual ~BookRepository() = default;
    virtual bool findById(const std::string& id, Book& out) const = 0;   // copy out: callers never hold our storage
    virtual void save(const Book& b) = 0;
    virtual std::vector<Book> findAvailable() const = 0;
};
class LoanRepository {
public:
    virtual ~LoanRepository() = default;
    virtual bool findById(const std::string& id, Loan& out) const = 0;
    virtual void save(const Loan& l) = 0;
    virtual int countOpen(const std::string& memberId) const = 0;
    virtual std::string nextId() = 0;
};

// ---------- in-memory implementations ----------
class InMemoryBookRepository : public BookRepository {
    std::map<std::string, Book> rows_;
public:
    bool findById(const std::string& id, Book& out) const override {
        auto it = rows_.find(id);
        if (it == rows_.end()) return false;
        out = it->second; return true;
    }
    void save(const Book& b) override { rows_[b.id] = b; }
    std::vector<Book> findAvailable() const override {
        std::vector<Book> v;
        for (const auto& kv : rows_) if (!kv.second.lent) v.push_back(kv.second);
        return v;
    }
};
class InMemoryLoanRepository : public LoanRepository {
    std::map<std::string, Loan> rows_;
    int seq_ = 0;
public:
    bool findById(const std::string& id, Loan& out) const override {
        auto it = rows_.find(id);
        if (it == rows_.end()) return false;
        out = it->second; return true;
    }
    void save(const Loan& l) override { rows_[l.id] = l; }
    int countOpen(const std::string& m) const override {
        int n = 0;
        for (const auto& kv : rows_) if (kv.second.open && kv.second.memberId == m) ++n;
        return n;
    }
    std::string nextId() override { return "L" + std::to_string(++seq_); }
};

// ---------- service: rules live here, storage lives behind the interfaces ----------
class LendingService {
    BookRepository& books_;
    LoanRepository& loans_;
    // Single-threaded demo. In a threaded build add a std::mutex and lock it in borrow/giveBack.
public:
    static const int kMaxOpen = 2;
    LendingService(BookRepository& b, LoanRepository& l) : books_(b), loans_(l) {}

    Loan borrow(const std::string& member, const std::string& bookId) {
        Book b;
        if (!books_.findById(bookId, b)) throw NotFound("book " + bookId);
        if (b.lent) throw LimitExceeded(bookId + " already lent");
        if (loans_.countOpen(member) >= kMaxOpen) throw LimitExceeded(member + " has 2 books");
        b.lent = true;
        books_.save(b);                                   // write back: the repo owns the state
        Loan l{loans_.nextId(), member, bookId, true};
        loans_.save(l);
        return l;
    }
    void giveBack(const std::string& loanId) {
        Loan l; Book b;
        if (!loans_.findById(loanId, l)) throw NotFound("loan " + loanId);
        l.open = false; loans_.save(l);
        books_.findById(l.bookId, b);
        b.lent = false; books_.save(b);
    }
    void printAvailable(const char* label) {
        std::cout << label;
        for (const auto& b : books_.findAvailable()) std::cout << " [" << b.title << "]";
        std::cout << "\n";
    }
};

int main() {
    InMemoryBookRepository books;
    InMemoryLoanRepository loans;
    LendingService svc(books, loans);
    books.save({"B1", "Linux Device Drivers"});
    books.save({"B2", "TCP/IP Illustrated"});
    books.save({"B3", "Clean Code"});

    Loan l1 = svc.borrow("M1", "B1");
    Loan l2 = svc.borrow("M1", "B2");
    std::cout << "borrowed " << l1.id << " and " << l2.id << "\n";
    svc.printAvailable("available:");
    try { svc.borrow("M1", "B3"); } catch (const std::exception& e) { std::cout << "error: " << e.what() << "\n"; }
    try { svc.borrow("M1", "B9"); } catch (const std::exception& e) { std::cout << "error: " << e.what() << "\n"; }
    svc.giveBack(l1.id);
    svc.printAvailable("after return:");
    std::cout << "borrowed " << svc.borrow("M1", "B3").id << "\n";
    return 0;
}

The service holds references to the abstract classes, the C++ version of a struct ops *. findById copies the entity out; the service changes its copy and calls save(), so the repository stays the single owner of state, just like a database row.

Swapping the store: the review-call answer

"How would you move this to MySQL?" The prepared answer: write JdbcBookRepository implements BookRepository (or a Spring Data interface), change one line in the composition root, and leave the service and entities untouched. The service tests keep running against the in-memory version. Two honest caveats to add: the lock on the service becomes a database transaction (and probably row locks or optimistic versioning on Book), and queries like findOpenByMember become indexed SQL instead of a scan.
