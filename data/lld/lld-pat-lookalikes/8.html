Code: look-alikes side by side

Each program puts look-alike patterns over one interface so you can see that the code shape is the same and only the intent differs. Java and Python: Decorator, Proxy and Adapter around a Storage, then Strategy (fare chosen from outside) vs State (trip moves itself, and refuses to cancel mid-trip). C++14: the same three wrappers, then a netfilter-style Chain of Responsibility where one hook stops the packet. All compiled and run; files in lld-code/lld-pat-lookalikes/.

Map before you read: which class plays which role in the Java tab (LookAlikes.java). The Python and C++ tabs use the same roles under their own names. Key lines carry a // <-- the pattern comment.

import java.util.*;

// Same shapes, different intent. Part 1: three wrappers around one interface.
// Part 2: Strategy vs State, both "context delegates to an interface".
public class LookAlikes {

    // ---------- Part 1: the wrapper family ----------
    interface Storage { String read(String key); }

    static final class DiskStorage implements Storage {
        int reads = 0;
        public String read(String key) { reads++; return "data(" + key + ")"; }
    }

    // DECORATOR: same interface, ADDS behaviour, stackable, client builds the stack.
    static final class Logged implements Storage {
        private final Storage inner;
        Logged(Storage inner) { this.inner = inner; }
        public String read(String key) {
            String v = inner.read(key);
            System.out.println("    [log] read " + key);
            return v;
        }
    }

    // PROXY: same interface, CONTROLS ACCESS to the real object (here: auth + cache).
    static final class GuardedCache implements Storage {
        private final Storage real;
        private final Set<String> allowed;
        private final Map<String, String> cache = new HashMap<>();
        GuardedCache(Storage real, Set<String> allowed) { this.real = real; this.allowed = allowed; }
        public String read(String key) {
            if (!allowed.contains(key)) return "DENIED";
            return cache.computeIfAbsent(key, real::read);   // real object touched at most once per key
        }
    }

    // ADAPTER: DIFFERENT interface underneath; converts it to the one clients expect.
    static final class LegacyBlobStore {                     // third-party, cannot change
        byte[] fetchBytes(int bucket, String path) { return ("blob:" + bucket + "/" + path).getBytes(); }
    }
    static final class BlobAdapter implements Storage {
        private final LegacyBlobStore legacy;
        BlobAdapter(LegacyBlobStore legacy) { this.legacy = legacy; }
        public String read(String key) { return new String(legacy.fetchBytes(7, key)); }
    }

    // ---------- Part 2: Strategy vs State ----------
    interface Fare { int price(int km); }                     // STRATEGY: client picks; strategies never switch themselves
    static final Fare NORMAL = km -> 10 * km;
    static final Fare SURGE = km -> 18 * km;

    static final class Ride {
        private Fare fare;
        Ride(Fare fare) { this.fare = fare; }
        void setFare(Fare f) { fare = f; }                    // chosen from OUTSIDE
        // <-- the pattern: Strategy: chosen from outside
        int quote(int km) { return fare.price(km); }
    }

    interface TripState { TripState next(Trip t, String event); }  // STATE: states pick the next state
    static final class Trip {
        TripState state = Requested.I;
        void on(String event) {
            TripState before = state;
            // <-- the pattern: State: the object changes its own behaviour
            state = state.next(this, event);                  // chosen from INSIDE
            System.out.println("    " + before + " --" + event + "--> " + state);
        }
    }
    enum Requested implements TripState { I;
        public TripState next(Trip t, String e) { return e.equals("accept") ? Accepted.I : e.equals("cancel") ? Cancelled.I : this; }
        public String toString() { return "Requested"; }
    }
    enum Accepted implements TripState { I;
        public TripState next(Trip t, String e) { return e.equals("start") ? OnTrip.I : e.equals("cancel") ? Cancelled.I : this; }
        public String toString() { return "Accepted"; }
    }
    enum OnTrip implements TripState { I;
        public TripState next(Trip t, String e) { return e.equals("end") ? Done.I : this; }   // cannot cancel mid-trip
        public String toString() { return "OnTrip"; }
    }
    enum Done implements TripState { I; public TripState next(Trip t, String e) { return this; } public String toString() { return "Done"; } }
    enum Cancelled implements TripState { I; public TripState next(Trip t, String e) { return this; } public String toString() { return "Cancelled"; } }

    public static void main(String[] args) {
        System.out.println("Part 1: three wrappers, one interface");
        DiskStorage disk = new DiskStorage();
        Storage decorated = new Logged(new Logged(disk));     // decorators stack
        System.out.println("  decorator  -> " + decorated.read("a"));
        Storage proxied = new GuardedCache(disk, Set.of("a"));
        proxied.read("a"); proxied.read("a");
        System.out.println("  proxy      -> " + proxied.read("a") + ", " + proxied.read("secret")
                + " (disk reads so far: " + disk.reads + ")");
        Storage adapted = new BlobAdapter(new LegacyBlobStore());
        System.out.println("  adapter    -> " + adapted.read("a"));

        System.out.println("Part 2: Strategy (client switches) vs State (object switches itself)");
        Ride r = new Ride(NORMAL);
        System.out.println("  strategy normal: " + r.quote(5));
        r.setFare(SURGE);
        System.out.println("  strategy surge : " + r.quote(5));
        Trip t = new Trip();
        for (String e : List.of("accept", "start", "cancel", "end")) t.on(e);
    }
}

Note "disk reads so far: 2": the decorator stack hit the disk once, and the proxy hit it once for three reads of a and zero times for the denied key. The trip ignored cancel while OnTrip because the state, not the caller, owns the rules.

"""Same shapes, different intent: three wrappers, then Strategy vs State."""


class Disk:                                    # the real object
    def __init__(self):
        self.reads = 0

    def read(self, key):
        self.reads += 1
        return f"data({key})"


class Logged:                                  # DECORATOR: adds behaviour, stackable
    def __init__(self, inner):
        self.inner = inner

    def read(self, key):
        value = self.inner.read(key)
        print(f"    [log] read {key}")
        return value


class GuardedCache:                            # PROXY: controls access to the real object
    def __init__(self, real, allowed):
        self.real, self.allowed, self.cache = real, set(allowed), {}

    def read(self, key):
        if key not in self.allowed:
            return "DENIED"
        if key not in self.cache:
            self.cache[key] = self.real.read(key)
        return self.cache[key]


class LegacyBlobStore:                         # cannot change this API
    def fetch_bytes(self, bucket, path):
        return f"blob:{bucket}/{path}".encode()


class BlobAdapter:                             # ADAPTER: converts a different interface
    def __init__(self, legacy):
        self.legacy = legacy

    def read(self, key):
        return self.legacy.fetch_bytes(7, key).decode()


# Strategy: a plain function the CLIENT chooses
def normal(km): return 10 * km
def surge(km): return 18 * km


# State: each state returns the NEXT state itself
TRANSITIONS = {
    "Requested": {"accept": "Accepted", "cancel": "Cancelled"},
    "Accepted": {"start": "OnTrip", "cancel": "Cancelled"},
    "OnTrip": {"end": "Done"},                 # no cancel while on the trip
}


class Trip:
    def __init__(self):
        self.state = "Requested"

    def on(self, event):
        before = self.state
        self.state = TRANSITIONS.get(self.state, {}).get(event, self.state)
        print(f"    {before} --{event}--> {self.state}")


if __name__ == "__main__":
    print("Part 1: three wrappers, one interface")
    disk = Disk()
    print("  decorator  ->", Logged(Logged(disk)).read("a"))
    proxy = GuardedCache(disk, {"a"})
    proxy.read("a"), proxy.read("a")
    print("  proxy      ->", proxy.read("a"), proxy.read("secret"), f"(disk reads so far: {disk.reads})")
    print("  adapter    ->", BlobAdapter(LegacyBlobStore()).read("a"))

    print("Part 2: Strategy (client switches) vs State (object switches itself)")
    fare = normal
    print("  strategy normal:", fare(5))
    fare = surge
    print("  strategy surge :", fare(5))
    trip = Trip()
    for e in ("accept", "start", "cancel", "end"):
        trip.on(e)

In Python, Strategy often collapses into "pass a function" and State into a transition table. The intent still differs: the caller rebinds fare; only Trip.on may change state.

// Same shapes, different intent, in C++14: Decorator vs Proxy vs Adapter,
// then Chain of Responsibility vs Decorator (who may stop the call?).
// Build: g++ -std=c++14 -Wall lookalikes.cpp -o lookalikes
#include <iostream>
#include <map>
#include <memory>
#include <set>
#include <string>
#include <vector>

struct Storage {
    virtual ~Storage() = default;
    virtual std::string read(const std::string& key) = 0;
};

struct Disk : Storage {
    int reads = 0;
    std::string read(const std::string& k) override { ++reads; return "data(" + k + ")"; }
};

struct Logged : Storage {                       // DECORATOR: always forwards, adds behaviour
    std::shared_ptr<Storage> inner;
    explicit Logged(std::shared_ptr<Storage> s) : inner(std::move(s)) {}
    std::string read(const std::string& k) override {
        auto v = inner->read(k);
        std::cout << "    [log] read " << k << "\n";
        return v;
    }
};

struct GuardedCache : Storage {                 // PROXY: decides whether the real object is touched
    std::shared_ptr<Storage> real;
    std::set<std::string> allowed;
    std::map<std::string, std::string> cache;
    GuardedCache(std::shared_ptr<Storage> r, std::set<std::string> a) : real(std::move(r)), allowed(std::move(a)) {}
    std::string read(const std::string& k) override {
        if (!allowed.count(k)) return "DENIED";
        auto it = cache.find(k);
        if (it != cache.end()) return it->second;
        return cache[k] = real->read(k);
    }
};

struct LegacyBlobStore {                        // a C-style API we cannot change
    std::string fetch(int bucket, const char* path) { return "blob:" + std::to_string(bucket) + "/" + path; }
};

struct BlobAdapter : Storage {                  // ADAPTER: converts a different interface
    LegacyBlobStore legacy;
    std::string read(const std::string& k) override { return legacy.fetch(7, k.c_str()); }
};

// Chain of Responsibility: like netfilter hooks, any handler may stop the packet.
enum Verdict { ACCEPT, DROP };
struct Hook {
    std::string name;
    Verdict (*fn)(int port);                    // a plain function pointer, kernel style
};
Verdict rate_ok(int) { return ACCEPT; }
Verdict block_telnet(int port) { return port == 23 ? DROP : ACCEPT; }
Verdict log_hook(int port) { std::cout << "    [log] port " << port << "\n"; return ACCEPT; }

Verdict run_chain(const std::vector<Hook>& chain, int port) {
    for (const auto& h : chain)
        if (h.fn(port) == DROP) { std::cout << "    dropped by " << h.name << "\n"; return DROP; }
    return ACCEPT;
}

int main() {
    std::cout << "Decorator vs Proxy vs Adapter\n";
    auto disk = std::make_shared<Disk>();
    Logged twice(std::make_shared<Logged>(disk));
    std::cout << "  decorator  -> " << twice.read("a") << "\n";
    GuardedCache proxy(disk, {"a"});
    proxy.read("a"); proxy.read("a");
    std::cout << "  proxy      -> " << proxy.read("a") << ", " << proxy.read("secret")
              << " (disk reads so far: " << disk->reads << ")\n";
    BlobAdapter adapter;
    std::cout << "  adapter    -> " << adapter.read("a") << "\n";

    std::cout << "Chain of Responsibility: a link may stop the call\n";
    std::vector<Hook> chain = {{"rate_ok", rate_ok}, {"block_telnet", block_telnet}, {"log_hook", log_hook}};
    for (int port : {443, 23})
        std::cout << "  port " << port << ": " << (run_chain(chain, port) == ACCEPT ? "ACCEPT" : "DROP") << "\n";
    return 0;
}

The hook array with function pointers is the shape of a netfilter hook list. The port-23 packet never reaches log_hook: a link stopped the chain, which a Decorator never does.
