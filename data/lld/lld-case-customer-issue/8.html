import java.util.*;
import java.util.concurrent.*;
import java.util.function.Predicate;

// ---------- model ----------
enum IssueType { PAYMENT, MUTUAL_FUND, GOLD, INSURANCE }

// Issue lifecycle as a small state machine: each status lists where it may go next.
enum IssueStatus {
    OPEN, WAITING, IN_PROGRESS, RESOLVED;
    boolean canMoveTo(IssueStatus next) {
        switch (this) {
            case OPEN:        return next == WAITING || next == IN_PROGRESS;
            case WAITING:     return next == IN_PROGRESS;
            case IN_PROGRESS: return next == RESOLVED;
            default:          return false;                 // RESOLVED is terminal
        }
    }
}

class IssueException extends RuntimeException {
    IssueException(String msg) { super(msg); }
}

final class Issue {
    final String id, transactionId, subject, description, customerEmail;
    final IssueType type;
    private IssueStatus status = IssueStatus.OPEN;
    private String agentId;                                  // null until assigned
    private String resolution = "";

    Issue(String id, String txnId, IssueType type, String subject, String description, String email) {
        this.id = id; this.transactionId = txnId; this.type = type;
        this.subject = subject; this.description = description; this.customerEmail = email;
    }
    IssueStatus status() { return status; }
    String agentId() { return agentId; }
    String resolution() { return resolution; }

    void moveTo(IssueStatus next) {                          // every status change goes through here
        if (!status.canMoveTo(next))
            throw new IssueException(id + ": cannot move " + status + " -> " + next);
        status = next;
    }
    void setAgent(String agentId) { this.agentId = agentId; }
    void setResolution(String r) { this.resolution = r == null ? "" : r; }
    @Override public String toString() { return id + "[" + type + "," + status + (agentId == null ? "" : "," + agentId) + "]"; }
}

final class Agent {
    final String id, name;                                   // id = agent email
    final Set<IssueType> expertise;
    private String currentIssue;                             // one issue at a time
    private final Deque<String> waitlist = new ArrayDeque<>();
    private final List<String> history = new ArrayList<>();  // resolved issue ids, in order

    Agent(String id, String name, Collection<IssueType> expertise) {
        this.id = id; this.name = name; this.expertise = EnumSet.copyOf(expertise);
    }
    boolean isFree() { return currentIssue == null; }
    boolean canHandle(IssueType t) { return expertise.contains(t); }
    int waitlistSize() { return waitlist.size(); }
    String currentIssue() { return currentIssue; }
    List<String> history() { return Collections.unmodifiableList(history); }

    void start(String issueId) { currentIssue = issueId; }
    void enqueue(String issueId) { waitlist.addLast(issueId); }
    String finishCurrent() {                                 // returns next waiting issue id, or null
        history.add(currentIssue);
        currentIssue = waitlist.pollFirst();
        return currentIssue;
    }
}

// ---------- assignment strategies ----------
interface AssignmentStrategy {
    /** Pick an agent for the issue from agents that have the expertise; empty if none qualify. */
    Optional<Agent> pick(Issue issue, List<Agent> experts);
}

/** Default: any free expert (first onboarded wins); if all are busy, the expert with the shortest waitlist. */
final class FreeFirstStrategy implements AssignmentStrategy {
    public Optional<Agent> pick(Issue issue, List<Agent> experts) {
        for (Agent a : experts) if (a.isFree()) return Optional.of(a);
        return experts.stream().min(Comparator.comparingInt(Agent::waitlistSize));
    }
}

/** Alternative: rotate through experts per issue type, free or not (spreads load, ignores busyness). */
final class RoundRobinStrategy implements AssignmentStrategy {
    private final Map<IssueType, Integer> next = new EnumMap<>(IssueType.class);
    public Optional<Agent> pick(Issue issue, List<Agent> experts) {
        if (experts.isEmpty()) return Optional.empty();
        int i = next.getOrDefault(issue.type, 0);
        next.put(issue.type, i + 1);
        return Optional.of(experts.get(i % experts.size()));
    }
}

// ---------- filters (small Specification) ----------
final class IssueFilters {
    static Predicate<Issue> all() { return i -> true; }
    static Predicate<Issue> byEmail(String e) { return i -> i.customerEmail.equalsIgnoreCase(e); }
    static Predicate<Issue> byType(IssueType t) { return i -> i.type == t; }
    static Predicate<Issue> byStatus(IssueStatus s) { return i -> i.status() == s; }
    static Predicate<Issue> byAgent(String a) { return i -> a.equals(i.agentId()); }
}

// ---------- service (facade) ----------
final class IssueService {
    private final Map<String, Issue> issues = new LinkedHashMap<>();
    private final Map<String, Agent> agents = new LinkedHashMap<>();
    private final AssignmentStrategy strategy;
    private int seq = 0;

    IssueService(AssignmentStrategy strategy) { this.strategy = strategy; }

    synchronized Issue createIssue(String txnId, IssueType type, String subject, String desc, String email) {
        if (txnId == null || txnId.isBlank() || email == null || !email.contains("@"))
            throw new IssueException("transactionId and a valid email are required");
        Issue issue = new Issue("I" + (++seq), txnId, type, subject, desc, email);
        issues.put(issue.id, issue);
        return issue;
    }

    synchronized Agent addAgent(String email, String name, List<IssueType> expertise) {
        if (agents.containsKey(email)) throw new IssueException("agent exists: " + email);
        if (expertise == null || expertise.isEmpty()) throw new IssueException("agent needs at least one issue type");
        Agent a = new Agent(email, name, expertise);
        agents.put(email, a);
        return a;
    }

    /** Returns the chosen agent id, or empty if no agent has the expertise (issue stays OPEN). */
    synchronized Optional<String> assignIssue(String issueId) {
        Issue issue = get(issueId);
        if (issue.status() != IssueStatus.OPEN) throw new IssueException(issueId + " already " + issue.status());
        List<Agent> experts = new ArrayList<>();
        for (Agent a : agents.values()) if (a.canHandle(issue.type)) experts.add(a);
        Optional<Agent> chosen = strategy.pick(issue, experts);
        if (!chosen.isPresent()) return Optional.empty();
        Agent a = chosen.get();
        issue.setAgent(a.id);
        if (a.isFree()) { issue.moveTo(IssueStatus.IN_PROGRESS); a.start(issue.id); }
        else            { issue.moveTo(IssueStatus.WAITING);     a.enqueue(issue.id); }
        return Optional.of(a.id);
    }

    synchronized List<Issue> getIssues(Predicate<Issue> filter) {
        List<Issue> out = new ArrayList<>();
        for (Issue i : issues.values()) if (filter.test(i)) out.add(i);
        return out;
    }

    /** RESOLVED delegates to resolveIssue; same status = update the notes. Other moves belong to the system
     *  (assignment and the waitlist), so a caller cannot push WAITING -> IN_PROGRESS by hand. */
    synchronized void updateIssue(String issueId, IssueStatus status, String resolution) {
        if (status == IssueStatus.RESOLVED) { resolveIssue(issueId, resolution); return; }
        Issue issue = get(issueId);
        if (status != issue.status())
            throw new IssueException(issueId + ": only the system moves " + issue.status() + " -> " + status);
        issue.setResolution(resolution);
    }

    synchronized void resolveIssue(String issueId, String resolution) {
        Issue issue = get(issueId);
        if (issue.status() == IssueStatus.RESOLVED) throw new IssueException(issueId + " is already RESOLVED");
        Agent a = issue.agentId() == null ? null : agents.get(issue.agentId());
        if (a == null || !issueId.equals(a.currentIssue()))
            throw new IssueException(issueId + " is not the issue its agent is working on");
        issue.moveTo(IssueStatus.RESOLVED);
        issue.setResolution(resolution);
        String next = a.finishCurrent();                     // agent pulls the next waiting issue
        if (next != null) issues.get(next).moveTo(IssueStatus.IN_PROGRESS);
    }

    synchronized Map<String, List<String>> viewAgentsWorkHistory() {
        Map<String, List<String>> out = new LinkedHashMap<>();
        for (Agent a : agents.values()) out.put(a.id, new ArrayList<>(a.history()));
        return out;
    }

    synchronized Agent agent(String id) { return agents.get(id); }

    private Issue get(String id) {
        Issue i = issues.get(id);
        if (i == null) throw new IssueException("no such issue: " + id);
        return i;
    }
}

public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }

    public static void main(String[] args) throws Exception {
        IssueService s = new IssueService(new FreeFirstStrategy());
        Issue i1 = s.createIssue("T1", IssueType.PAYMENT, "Payment failed", "Money debited", "asha@test.com");
        Issue i2 = s.createIssue("T2", IssueType.GOLD, "Gold not credited", "Bought 1g", "ben@test.com");
        Issue i3 = s.createIssue("T3", IssueType.PAYMENT, "Pending txn", "Stuck 2 days", "ben@test.com");
        Issue i4 = s.createIssue("T4", IssueType.INSURANCE, "Policy missing", "No PDF", "chitra@test.com");
        Issue i5 = s.createIssue("T5", IssueType.GOLD, "Gold sell failed", "Error 500", "asha@test.com");
        s.addAgent("a1@pp.com", "Agent1", Arrays.asList(IssueType.PAYMENT, IssueType.GOLD));
        s.addAgent("a2@pp.com", "Agent2", Arrays.asList(IssueType.PAYMENT));
        System.out.println("created " + s.getIssues(IssueFilters.all()));

        for (Issue i : Arrays.asList(i1, i2, i3, i4, i5)) {
            Optional<String> a = s.assignIssue(i.id);
            System.out.println("assign " + i.id + " -> " + (a.isPresent() ? a.get() : "no agent with expertise, stays OPEN"));
        }
        check(i1.status() == IssueStatus.IN_PROGRESS && "a1@pp.com".equals(i1.agentId()), "I1 to free a1");
        check(i2.status() == IssueStatus.WAITING && "a1@pp.com".equals(i2.agentId()), "I2 waits on a1 (only gold expert)");
        check(i3.status() == IssueStatus.IN_PROGRESS && "a2@pp.com".equals(i3.agentId()), "I3 to free a2");
        check(i4.status() == IssueStatus.OPEN, "I4 has no expert");
        check(s.agent("a1@pp.com").waitlistSize() == 2, "a1 waitlist = I2, I5");

        System.out.println("issues of ben@test.com: " + s.getIssues(IssueFilters.byEmail("ben@test.com")));
        System.out.println("waiting gold issues:    " + s.getIssues(IssueFilters.byType(IssueType.GOLD).and(IssueFilters.byStatus(IssueStatus.WAITING))));
        check(s.getIssues(IssueFilters.byEmail("BEN@test.com")).size() == 2, "filter by email ignores case");

        s.resolveIssue("I1", "Refund initiated");
        System.out.println("resolve I1 -> a1 now on " + s.agent("a1@pp.com").currentIssue());
        check(i2.status() == IssueStatus.IN_PROGRESS, "a1 picks I2 from its waitlist");
        s.updateIssue("I3", IssueStatus.RESOLVED, "Bank reversed the txn");
        s.resolveIssue("I2", "Gold credited");
        s.resolveIssue("I5", "Sell retried");
        check(s.agent("a1@pp.com").isFree(), "a1 free after waitlist drains");

        try { s.resolveIssue("I1", "again"); check(false, "double resolve must fail"); }
        catch (IssueException e) { System.out.println("rejected: " + e.getMessage()); passed++; }
        try { s.updateIssue("I4", IssueStatus.RESOLVED, "x"); check(false, "resolve unassigned must fail"); }
        catch (IssueException e) { System.out.println("rejected: " + e.getMessage()); passed++; }

        s.updateIssue("I4", IssueStatus.OPEN, "Asked customer for policy number");
        try { s.updateIssue("I4", IssueStatus.IN_PROGRESS, "x"); check(false, "manual move must fail"); }
        catch (IssueException e) { System.out.println("rejected: " + e.getMessage()); passed++; }

        s.addAgent("a3@pp.com", "Agent3", Arrays.asList(IssueType.INSURANCE));
        System.out.println("assign I4 again -> " + s.assignIssue("I4").get());
        System.out.println("work history " + s.viewAgentsWorkHistory());
        check(s.viewAgentsWorkHistory().get("a1@pp.com").equals(Arrays.asList("I1", "I2", "I5")), "a1 history in order");

        // strategy swap: round robin spreads payment issues over a1 and a2 even when a1 is free
        IssueService rr = new IssueService(new RoundRobinStrategy());
        rr.addAgent("a1@pp.com", "Agent1", Arrays.asList(IssueType.PAYMENT));
        rr.addAgent("a2@pp.com", "Agent2", Arrays.asList(IssueType.PAYMENT));
        List<String> got = new ArrayList<>();
        for (int k = 0; k < 4; k++) got.add(rr.assignIssue(rr.createIssue("R" + k, IssueType.PAYMENT, "s", "d", "x@y.com").id).get());
        System.out.println("round robin " + got);
        check(got.equals(Arrays.asList("a1@pp.com", "a2@pp.com", "a1@pp.com", "a2@pp.com")), "round robin order");

        // concurrency: 8 threads create+assign 400 issues; each agent works on at most one
        IssueService c = new IssueService(new FreeFirstStrategy());
        for (int k = 0; k < 5; k++) c.addAgent("c" + k + "@pp.com", "C" + k, Arrays.asList(IssueType.PAYMENT));
        ExecutorService pool = Executors.newFixedThreadPool(8);
        List<Future<?>> fs = new ArrayList<>();
        for (int t = 0; t < 8; t++) fs.add(pool.submit(() -> {
            for (int k = 0; k < 50; k++) c.assignIssue(c.createIssue("C", IssueType.PAYMENT, "s", "d", "c@y.com").id);
        }));
        for (Future<?> f : fs) f.get();
        pool.shutdown();
        long inProgress = c.getIssues(IssueFilters.byStatus(IssueStatus.IN_PROGRESS)).size();
        long waiting = c.getIssues(IssueFilters.byStatus(IssueStatus.WAITING)).size();
        System.out.println("concurrent: in progress=" + inProgress + " waiting=" + waiting);
        check(inProgress == 5 && waiting == 395, "one issue per agent, rest waiting");

        System.out.println("All " + passed + " checks passed");
    }
}
"""Customer Issue Resolution (PhonePe machine coding) - in-memory, thread safe, runnable."""
from __future__ import annotations

import threading
from collections import deque
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Deque, Dict, List, Optional


class IssueType(Enum):
    PAYMENT = "PAYMENT"
    MUTUAL_FUND = "MUTUAL_FUND"
    GOLD = "GOLD"
    INSURANCE = "INSURANCE"


class IssueStatus(Enum):
    OPEN = "OPEN"
    WAITING = "WAITING"
    IN_PROGRESS = "IN_PROGRESS"
    RESOLVED = "RESOLVED"


# Issue lifecycle as a transition table (RESOLVED is terminal).
ALLOWED = {
    IssueStatus.OPEN: {IssueStatus.WAITING, IssueStatus.IN_PROGRESS},
    IssueStatus.WAITING: {IssueStatus.IN_PROGRESS},
    IssueStatus.IN_PROGRESS: {IssueStatus.RESOLVED},
    IssueStatus.RESOLVED: set(),
}


class IssueError(Exception):
    pass


@dataclass
class Issue:
    id: str
    transaction_id: str
    type: IssueType
    subject: str
    description: str
    email: str
    status: IssueStatus = IssueStatus.OPEN
    agent_id: Optional[str] = None
    resolution: str = ""

    def move_to(self, nxt: IssueStatus) -> None:
        if nxt not in ALLOWED[self.status]:
            raise IssueError(f"{self.id}: cannot move {self.status.value} -> {nxt.value}")
        self.status = nxt

    def __repr__(self) -> str:
        who = f",{self.agent_id}" if self.agent_id else ""
        return f"{self.id}[{self.type.value},{self.status.value}{who}]"


@dataclass
class Agent:
    id: str  # agent email
    name: str
    expertise: frozenset
    current: Optional[str] = None
    waitlist: Deque[str] = field(default_factory=deque)
    history: List[str] = field(default_factory=list)

    def is_free(self) -> bool:
        return self.current is None

    def finish_current(self) -> Optional[str]:
        self.history.append(self.current)
        self.current = self.waitlist.popleft() if self.waitlist else None
        return self.current


# ---------- assignment strategies ----------
class AssignmentStrategy:
    def pick(self, issue: Issue, experts: List[Agent]) -> Optional[Agent]:
        raise NotImplementedError


class FreeFirstStrategy(AssignmentStrategy):
    """Any free expert (first onboarded wins); else the expert with the shortest waitlist."""

    def pick(self, issue, experts):
        for a in experts:
            if a.is_free():
                return a
        return min(experts, key=lambda a: len(a.waitlist), default=None)


class RoundRobinStrategy(AssignmentStrategy):
    """Rotate through experts per issue type, free or not."""

    def __init__(self):
        self._next: Dict[IssueType, int] = {}

    def pick(self, issue, experts):
        if not experts:
            return None
        i = self._next.get(issue.type, 0)
        self._next[issue.type] = i + 1
        return experts[i % len(experts)]


# ---------- filters (small Specification: plain predicates that compose) ----------
Filter = Callable[[Issue], bool]


def by_email(e: str) -> Filter:
    return lambda i: i.email.lower() == e.lower()


def by_type(t: IssueType) -> Filter:
    return lambda i: i.type == t


def by_status(s: IssueStatus) -> Filter:
    return lambda i: i.status == s


def all_of(*fs: Filter) -> Filter:
    return lambda i: all(f(i) for f in fs)


# ---------- service (facade) ----------
class IssueService:
    def __init__(self, strategy: AssignmentStrategy):
        self._issues: Dict[str, Issue] = {}
        self._agents: Dict[str, Agent] = {}
        self._strategy = strategy
        self._seq = 0
        self._lock = threading.RLock()

    def create_issue(self, txn_id, issue_type, subject, description, email) -> Issue:
        if not txn_id or not email or "@" not in email:
            raise IssueError("transaction id and a valid email are required")
        with self._lock:
            self._seq += 1
            issue = Issue(f"I{self._seq}", txn_id, issue_type, subject, description, email)
            self._issues[issue.id] = issue
            return issue

    def add_agent(self, email, name, expertise) -> Agent:
        if not expertise:
            raise IssueError("agent needs at least one issue type")
        with self._lock:
            if email in self._agents:
                raise IssueError(f"agent exists: {email}")
            a = Agent(email, name, frozenset(expertise))
            self._agents[email] = a
            return a

    def assign_issue(self, issue_id) -> Optional[str]:
        with self._lock:
            issue = self._get(issue_id)
            if issue.status != IssueStatus.OPEN:
                raise IssueError(f"{issue_id} already {issue.status.value}")
            experts = [a for a in self._agents.values() if issue.type in a.expertise]
            a = self._strategy.pick(issue, experts)
            if a is None:
                return None  # no expert: stays OPEN
            issue.agent_id = a.id
            if a.is_free():
                issue.move_to(IssueStatus.IN_PROGRESS)
                a.current = issue.id
            else:
                issue.move_to(IssueStatus.WAITING)
                a.waitlist.append(issue.id)
            return a.id

    def get_issues(self, flt: Filter = lambda i: True) -> List[Issue]:
        with self._lock:
            return [i for i in self._issues.values() if flt(i)]

    def update_issue(self, issue_id, status: IssueStatus, resolution: str) -> None:
        with self._lock:
            if status == IssueStatus.RESOLVED:
                self.resolve_issue(issue_id, resolution)
                return
            issue = self._get(issue_id)
            if status != issue.status:
                raise IssueError(f"{issue_id}: only the system moves {issue.status.value} -> {status.value}")
            issue.resolution = resolution

    def resolve_issue(self, issue_id, resolution) -> None:
        with self._lock:
            issue = self._get(issue_id)
            if issue.status == IssueStatus.RESOLVED:
                raise IssueError(f"{issue_id} is already RESOLVED")
            a = self._agents.get(issue.agent_id) if issue.agent_id else None
            if a is None or a.current != issue_id:
                raise IssueError(f"{issue_id} is not the issue its agent is working on")
            issue.move_to(IssueStatus.RESOLVED)
            issue.resolution = resolution
            nxt = a.finish_current()
            if nxt is not None:
                self._issues[nxt].move_to(IssueStatus.IN_PROGRESS)

    def view_agents_work_history(self) -> Dict[str, List[str]]:
        with self._lock:
            return {a.id: list(a.history) for a in self._agents.values()}

    def agent(self, agent_id) -> Agent:
        return self._agents[agent_id]

    def _get(self, issue_id) -> Issue:
        if issue_id not in self._issues:
            raise IssueError(f"no such issue: {issue_id}")
        return self._issues[issue_id]


def demo() -> None:
    s = IssueService(FreeFirstStrategy())
    s.create_issue("T1", IssueType.PAYMENT, "Payment failed", "Money debited", "asha@test.com")
    s.create_issue("T2", IssueType.GOLD, "Gold not credited", "Bought 1g", "ben@test.com")
    s.create_issue("T3", IssueType.PAYMENT, "Pending txn", "Stuck 2 days", "ben@test.com")
    s.create_issue("T4", IssueType.INSURANCE, "Policy missing", "No PDF", "chitra@test.com")
    s.create_issue("T5", IssueType.GOLD, "Gold sell failed", "Error 500", "asha@test.com")
    s.add_agent("a1@pp.com", "Agent1", [IssueType.PAYMENT, IssueType.GOLD])
    s.add_agent("a2@pp.com", "Agent2", [IssueType.PAYMENT])
    for iid in ["I1", "I2", "I3", "I4", "I5"]:
        print(f"assign {iid} ->", s.assign_issue(iid) or "no agent with expertise, stays OPEN")
    print("issues of ben@test.com:", s.get_issues(by_email("ben@test.com")))
    print("waiting gold issues:   ", s.get_issues(all_of(by_type(IssueType.GOLD), by_status(IssueStatus.WAITING))))
    s.resolve_issue("I1", "Refund initiated")
    print("resolve I1 -> a1 now on", s.agent("a1@pp.com").current)
    s.update_issue("I3", IssueStatus.RESOLVED, "Bank reversed the txn")
    s.resolve_issue("I2", "Gold credited")
    s.resolve_issue("I5", "Sell retried")
    try:
        s.resolve_issue("I1", "again")
    except IssueError as e:
        print("rejected:", e)
    s.add_agent("a3@pp.com", "Agent3", [IssueType.INSURANCE])
    print("assign I4 again ->", s.assign_issue("I4"))
    print("work history", s.view_agents_work_history())


if __name__ == "__main__":
    demo()

    def test_no_expert_stays_open(self):

    def test_resolve_pulls_next_from_waitlist(self):

    def test_illegal_transitions(self):
        s = fresh()
        i1, i2 = new(s, G), new(s, G)
        s.assign_issue(i1)
        s.assign_issue(i2)

            s.resolve_issue(i2, "skip the queue")                  # waiting, not current
        with self.assertRaises(IssueError):

        with self.assertRaises(IssueError):
            s.assign_issue(i1)                                     # already assigned
        s.resolve_issue(i1, "ok")
        with self.assertRaises(IssueError):

    def test_filters_compose(self):

    def test_round_robin(self):

    def test_concurrent_assign(self):
        s = IssueService(FreeFirstStrategy())
        for k in range(5):
            s.add_agent(f"c{k}@pp.com", f"C{k}", [P])

        def work():
            for _ in range(50):
                s.assign_issue(new(s, P))

        ts = [threading.Thread(target=work) for _ in range(8)]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
        self.assertEqual(len(s.get_issues(by_status(IssueStatus.IN_PROGRESS))), 5)
        self.assertEqual(len(s.get_issues(by_status(IssueStatus.WAITING))), 395)

// Customer Issue Resolution - C++14, single file. Build: g++ -std=c++14 -Wall -Wextra main.cpp -o main
// Locking, as in the Java synchronized methods: every public IssueService method takes one mutex (mu_) at the
// top with std::lock_guard. resolveIssue() marks the issue resolved AND pulls the next waitlisted issue in the
// same critical section, so no other thread can grab the agent in between. The strategy runs under the lock (pure).
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <deque>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <cstdlib>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>
#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif

enum class IssueType { PAYMENT, MUTUAL_FUND, GOLD, INSURANCE };
enum class Status { OPEN, WAITING, IN_PROGRESS, RESOLVED };

const char* name(Status s) {
    switch (s) {
        case Status::OPEN: return "OPEN";
        case Status::WAITING: return "WAITING";
        case Status::IN_PROGRESS: return "IN_PROGRESS";
        default: return "RESOLVED";
    }
}

bool canMove(Status from, Status to) {
    return (from == Status::OPEN && (to == Status::WAITING || to == Status::IN_PROGRESS)) ||
           (from == Status::WAITING && to == Status::IN_PROGRESS) ||
           (from == Status::IN_PROGRESS && to == Status::RESOLVED);
}

struct IssueError : std::runtime_error { using std::runtime_error::runtime_error; };

struct Issue {
    std::string id, txn, subject, email, agent, resolution;
    IssueType type;
    Status status = Status::OPEN;
    void moveTo(Status s) {
        if (!canMove(status, s)) throw IssueError(id + ": cannot move " + name(status) + " -> " + name(s));
        status = s;
    }
};

struct Agent {
    std::string id, name;
    std::set<IssueType> expertise;
    std::string current;                 // empty = free
    std::deque<std::string> waitlist;
    std::vector<std::string> history;
    bool isFree() const { return current.empty(); }
};

// Strategy: the C-style equivalent is a struct of function pointers; here a pure virtual interface.
struct AssignmentStrategy {
    virtual ~AssignmentStrategy() = default;
    virtual Agent* pick(const Issue& issue, const std::vector<Agent*>& experts) = 0;
};

struct FreeFirstStrategy : AssignmentStrategy {
    Agent* pick(const Issue&, const std::vector<Agent*>& experts) override {
        for (Agent* a : experts) if (a->isFree()) return a;
        Agent* best = nullptr;
        for (Agent* a : experts) if (!best || a->waitlist.size() < best->waitlist.size()) best = a;
        return best;
    }
};

using Filter = std::function<bool(const Issue&)>;

class IssueService {
    std::map<std::string, Issue> issues_;           // ordered by id string; fine for a demo
    std::vector<std::string> order_;                // creation order
    std::vector<std::unique_ptr<Agent>> agents_;    // onboarding order; stable addresses
    std::unique_ptr<AssignmentStrategy> strategy_;
    int seq_ = 0;
    Mutex mu_;                                      // guards everything above, including each Agent's queue

    Issue& get(const std::string& id) {
        auto it = issues_.find(id);
        if (it == issues_.end()) throw IssueError("no such issue: " + id);
        return it->second;
    }
    Agent* agentById(const std::string& id) {
        for (auto& a : agents_) if (a->id == id) return a.get();
        return nullptr;
    }

public:
    explicit IssueService(std::unique_ptr<AssignmentStrategy> s) : strategy_(std::move(s)) {}

    std::string createIssue(const std::string& txn, IssueType t, const std::string& subject, const std::string& email) {
        std::lock_guard<Mutex> g(mu_);
        if (txn.empty() || email.find('@') == std::string::npos) throw IssueError("txn and valid email required");
        Issue i;
        i.id = "I" + std::to_string(++seq_); i.txn = txn; i.type = t; i.subject = subject; i.email = email;
        order_.push_back(i.id);
        issues_[i.id] = i;
        return order_.back();
    }

    void addAgent(const std::string& email, const std::string& nm, std::set<IssueType> exp) {
        std::lock_guard<Mutex> g(mu_);
        if (agentById(email)) throw IssueError("agent exists: " + email);
        std::unique_ptr<Agent> a(new Agent());
        a->id = email; a->name = nm; a->expertise = std::move(exp);
        agents_.push_back(std::move(a));
    }

    std::string assignIssue(const std::string& id) {        // "" = no expert, stays OPEN
        std::lock_guard<Mutex> g(mu_);
        Issue& i = get(id);
        if (i.status != Status::OPEN) throw IssueError(id + " already " + name(i.status));
        std::vector<Agent*> experts;
        for (auto& a : agents_) if (a->expertise.count(i.type)) experts.push_back(a.get());
        Agent* a = strategy_->pick(i, experts);
        if (!a) return "";
        i.agent = a->id;
        if (a->isFree()) { i.moveTo(Status::IN_PROGRESS); a->current = id; }
        else { i.moveTo(Status::WAITING); a->waitlist.push_back(id); }
        return a->id;
    }

    std::vector<std::string> getIssues(const Filter& f) {
        std::lock_guard<Mutex> g(mu_);
        std::vector<std::string> out;
        for (auto& id : order_) {
            const Issue& i = issues_[id];
            if (f(i)) out.push_back(i.id + "[" + name(i.status) + (i.agent.empty() ? "" : "," + i.agent) + "]");
        }
        return out;
    }

    void resolveIssue(const std::string& id, const std::string& resolution) {
        std::lock_guard<Mutex> g(mu_);
        Issue& i = get(id);
        if (i.status == Status::RESOLVED) throw IssueError(id + " is already RESOLVED");
        Agent* a = i.agent.empty() ? nullptr : agentById(i.agent);
        if (!a || a->current != id) throw IssueError(id + " is not the issue its agent is working on");
        i.moveTo(Status::RESOLVED);
        i.resolution = resolution;
        a->history.push_back(id);
        a->current.clear();
        if (!a->waitlist.empty()) {
            a->current = a->waitlist.front();
            a->waitlist.pop_front();
            get(a->current).moveTo(Status::IN_PROGRESS);
        }
    }

    std::map<std::string, std::vector<std::string>> workHistory() {
        std::lock_guard<Mutex> g(mu_);
        std::map<std::string, std::vector<std::string>> out;
        for (auto& a : agents_) out[a->id] = a->history;
        return out;
    }
    std::string currentOf(const std::string& agentId) {
        std::lock_guard<Mutex> g(mu_);
        return agentById(agentId)->current;
    }
};

static int passed = 0;
void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); }
    ++passed;
}

    std::cout << label;
    for (auto& s : v) std::cout << " " << s;
    std::cout << "\n";
}
