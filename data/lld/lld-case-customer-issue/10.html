// Follow-up: least-loaded strategy, written live in two minutes, no other file changes
final class LeastLoadedStrategy implements AssignmentStrategy {
    public Optional<Agent> pick(Issue issue, List<Agent> experts) {
        return experts.stream().min(Comparator
                .comparingInt((Agent a) -> (a.isFree() ? 0 : 1) + a.waitlistSize())
                .thenComparing(a -> a.id));
    }
}
