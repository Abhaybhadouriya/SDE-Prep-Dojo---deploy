Code: coffee pricing, service wrappers, a block-device stack

Each tab is a complete program that was compiled and run. Java does the classic coffee pricing and then stacks real JDK stream decorators (buffering, character encoding, gzip). Python wraps a flaky price service with retry, caching and timing at run time, and contrasts this with Python's @decorator syntax. C++14 stacks a stats layer and a crypt layer over a RAM disk, like device-mapper. Files live in lld-code/lld-pat-decorator/.

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.zip.*;

public class DecoratorDemo {

    // ---- Component: what the till and the receipt printer use ----
    interface Beverage {
        String description();
        int pricePaise();
    }

    // ---- Concrete components: the base drinks ----
    static final class Espresso implements Beverage {
        public String description() { return "Espresso"; }
        public int pricePaise() { return 12_000; }
    }
    static final class Chai implements Beverage {
        public String description() { return "Chai"; }
        public int pricePaise() { return 4_000; }
    }

    // ---- Base decorator: IS a Beverage and HAS a Beverage; forwards by default ----
    static abstract class AddOn implements Beverage {
        protected final Beverage inner;
        AddOn(Beverage inner) { this.inner = inner; }
        public String description() { return inner.description(); }
        public int pricePaise() { return inner.pricePaise(); }
    }

    // ---- Concrete decorators: each adds one responsibility around the call ----
    static final class Milk extends AddOn {
        Milk(Beverage b) { super(b); }
        @Override public String description() { return inner.description() + " + milk"; }
        // <-- the pattern: forward to the wrapped object, then add
        @Override public int pricePaise() { return inner.pricePaise() + 2_000; }
    }
    static final class ExtraShot extends AddOn {
        ExtraShot(Beverage b) { super(b); }
        @Override public String description() { return inner.description() + " + shot"; }
        @Override public int pricePaise() { return inner.pricePaise() + 3_000; }
    }
    static final class LargeSize extends AddOn {            // changes the result, not just adds
        LargeSize(Beverage b) { super(b); }
        @Override public String description() { return inner.description() + " [large x1.5]"; }
        @Override public int pricePaise() { return inner.pricePaise() * 3 / 2; }
    }

    static void ring(Beverage b) {                          // client: sees only Beverage
        System.out.printf("%-38s Rs %6.2f%n", b.description(), b.pricePaise() / 100.0);
    }

    public static void main(String[] args) throws IOException {
        ring(new Espresso());
        ring(new Milk(new Espresso()));
        ring(new Milk(new Milk(new ExtraShot(new Espresso()))));   // stack any number
        ring(new LargeSize(new Milk(new Chai())));                 // size applied after milk
        ring(new Milk(new LargeSize(new Chai())));                 // order matters

        // The JDK I/O library is built from decorators: each wrapper IS an OutputStream.
        ByteArrayOutputStream sink = new ByteArrayOutputStream();
        try (Writer w = new BufferedWriter(new OutputStreamWriter(
                new GZIPOutputStream(sink), StandardCharsets.UTF_8))) {
            for (int i = 0; i < 200; i++) w.write("hello decorator ");
        }
        System.out.println("gzipped 3200 chars into " + sink.size() + " bytes");

        try (BufferedReader r = new BufferedReader(new InputStreamReader(
                new GZIPInputStream(new ByteArrayInputStream(sink.toByteArray())),
                StandardCharsets.UTF_8))) {
            System.out.println("read back: " + r.readLine().substring(0, 32) + "...");
        }
    }
}

"""Decorator: wrap a slow price service with caching, retry and timing, at run time."""
from __future__ import annotations
import functools
import time
from typing import Protocol


class PriceService(Protocol):                      # Component
    def price(self, sku: str) -> int: ...


class RemotePriceService:                          # Concrete component (flaky and slow)
    def __init__(self) -> None:
        self.calls = 0

    def price(self, sku: str) -> int:
        self.calls += 1
        if self.calls == 2:                        # fail once, to exercise Retry
            raise ConnectionError("timeout")
        return {"apple": 120, "milk": 56}[sku]


class CachingPrice:                                # Concrete decorator 1
    def __init__(self, inner: PriceService) -> None:
        self._inner, self._cache = inner, {}

    def price(self, sku: str) -> int:
        if sku not in self._cache:
            self._cache[sku] = self._inner.price(sku)
        return self._cache[sku]


class RetryingPrice:                               # Concrete decorator 2
    def __init__(self, inner: PriceService, attempts: int = 3) -> None:
        self._inner, self._attempts = inner, attempts

    def price(self, sku: str) -> int:
        for n in range(1, self._attempts + 1):
            try:
                return self._inner.price(sku)
            except ConnectionError as e:
                print(f"  retry {n}: {e}")
        raise ConnectionError(f"gave up on {sku}")


class TimedPrice:                                  # Concrete decorator 3
    def __init__(self, inner: PriceService, log: list[str]) -> None:
        self._inner, self._log = inner, log

    def price(self, sku: str) -> int:
        t0 = time.perf_counter()
        try:
            return self._inner.price(sku)
        finally:
            self._log.append(f"price({sku}) {1000 * (time.perf_counter() - t0):.3f} ms")


def checkout(svc: PriceService, skus: list[str]) -> int:     # client sees one interface
    return sum(svc.price(s) for s in skus)


# Python's @decorator syntax: a function that wraps a function. Same idea, smaller scale.
def logged(fn):
    @functools.wraps(fn)
    def wrapper(*args):
        result = fn(*args)
        print(f"  {fn.__name__}({', '.join(map(str, args))}) -> {result}")
        return result
    return wrapper


@logged
def gst(amount: int) -> int:
    return amount * 118 // 100


if __name__ == "__main__":
    remote = RemotePriceService()
    log: list[str] = []
    svc = TimedPrice(CachingPrice(RetryingPrice(remote)), log)   # built at run time
    print("total:", checkout(svc, ["apple", "milk", "apple", "milk"]))
    print("remote calls:", remote.calls, "| timings logged:", len(log))
    gst(1000)
    print("wraps keeps the name:", gst.__name__)
$ python decorator.py
  retry 1: timeout
total: 352
remote calls: 3 | timings logged: 4
  gst(1000) -> 1180
wraps keeps the name: gst

The stack is Timed(Caching(Retrying(remote))). The second remote call fails once and is retried; repeated SKUs are served from the cache, so the remote saw only three calls for four lookups. The order is a design choice: with caching outside retry, a cache hit skips the retry logic entirely. Python's @logged is the same idea applied to a function rather than an object; functools.wraps copies the name and docstring so the wrapper does not hide the original.

C++ you need here: std::unique_ptr<T> = an owning pointer that frees its object when it goes out of scope (a kfree() that runs itself on every exit path). A lambda [](args){ body } = an anonymous function: a function pointer plus any variables it captured in the brackets, like a callback with its void *ctx built in. std::function<R(Args)> = a typed slot that can hold any callable (function pointer, lambda, functor). Ownership and RAII in more depth: kref and shared ownership.

// Decorator in C++14: stack layers over a block device, the way device-mapper stacks targets.
// Build: g++ -std=c++14 -Wall decorator.cpp -o decorator
#include <cstdint>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

// Component
class BlockDevice {
public:
    virtual ~BlockDevice() = default;
    virtual void write(std::size_t block, std::string data) = 0;
    virtual std::string read(std::size_t block) = 0;
    virtual std::string name() const = 0;
};

// Concrete component: a RAM disk
class RamDisk : public BlockDevice {
public:
    explicit RamDisk(std::size_t blocks) : blocks_(blocks) {}
    void write(std::size_t b, std::string d) override { blocks_.at(b) = std::move(d); }
    std::string read(std::size_t b) override { return blocks_.at(b); }
    std::string name() const override { return "ramdisk"; }
private:
    std::vector<std::string> blocks_;
};

// Base decorator: is a BlockDevice, owns the BlockDevice below it
class Layer : public BlockDevice {
public:
    explicit Layer(std::unique_ptr<BlockDevice> lower) : lower_(std::move(lower)) {}
    void write(std::size_t b, std::string d) override { lower_->write(b, std::move(d)); }
    std::string read(std::size_t b) override { return lower_->read(b); }
    std::string name() const override { return lower_->name(); }
protected:
    std::unique_ptr<BlockDevice> lower_;
};

// Concrete decorator 1: toy encryption (XOR) - think dm-crypt
class CryptLayer : public Layer {
public:
    CryptLayer(std::unique_ptr<BlockDevice> lower, char key) : Layer(std::move(lower)), key_(key) {}
    void write(std::size_t b, std::string d) override { lower_->write(b, xorred(d)); }
    std::string read(std::size_t b) override { return xorred(lower_->read(b)); }
    std::string name() const override { return "crypt(" + lower_->name() + ")"; }
private:
    std::string xorred(std::string s) const { for (auto& c : s) c ^= key_; return s; }
    char key_;
};

// Concrete decorator 2: I/O statistics - think iostat counters on a dm device
class StatsLayer : public Layer {
public:
    using Layer::Layer;
    void write(std::size_t b, std::string d) override { ++writes_; Layer::write(b, std::move(d)); }
    std::string read(std::size_t b) override { ++reads_; return Layer::read(b); }
    std::string name() const override { return "stats(" + lower_->name() + ")"; }
    void report() const { std::cout << "  stats: reads=" << reads_ << " writes=" << writes_ << "\n"; }
private:
    int reads_ = 0, writes_ = 0;
};

int main() {
    // Keep a raw pointer to the bottom so we can peek at what is really stored.
    auto disk = std::make_unique<RamDisk>(4);
    RamDisk* raw = disk.get();

    auto stats = std::make_unique<StatsLayer>(
        std::make_unique<CryptLayer>(std::move(disk), 0x20));
    StatsLayer* statsView = stats.get();
    std::unique_ptr<BlockDevice> dev = std::move(stats);      // client holds the Component

    std::cout << "stack: " << dev->name() << "\n";
    dev->write(0, "kernel");
    dev->write(1, "Hello");
    std::cout << "read 0 through stack: " << dev->read(0) << "\n";
    std::cout << "read 1 through stack: " << dev->read(1) << "\n";
    std::cout << "raw block 1 on disk : " << raw->read(1) << "\n";   // case flipped by XOR 0x20
    statsView->report();
    return 0;
}

read 1 through stack: Hello

  stats: reads=2 writes=2

Each layer owns the layer below through std::unique_ptr, so destroying the top destroys the stack. The client reads "Hello" back through the stack while the bytes at rest are transformed, exactly the contract dm-crypt gives a file system. The XOR is a toy, not encryption.

The same idea in C

Decorator in plain C: a struct blk_ops whose read wraps an inner device's read, stacked like device-mapper targets. The outer layer is the only one the caller holds; each layer forwards to inner and adds its bit. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-decorator/decorator_sketch.c).

/* Decorator in C: a struct blk_ops that wraps an inner blk_ops and adds behaviour.
   Illustrative, not real kernel code (device-mapper targets stack on block devices this way).
   Build: gcc -std=c99 -Wall decorator_sketch.c -o decorator_sketch */
#include <stdio.h>

struct blk;
struct blk_ops { int (*read)(struct blk *b, long sector); };
struct blk { const struct blk_ops *ops; struct blk *inner; int hits; };

static int disk_read(struct blk *b, long sector) { (void)b; return (int)(sector * 7 % 256); }

static int count_read(struct blk *b, long sector)       /* decorator 1: count calls */
{
    b->hits++;
    return b->inner->ops->read(b->inner, sector);      /* forward to the wrapped device */
}
static int crypt_read(struct blk *b, long sector)       /* decorator 2: "decrypt" the result */
{
    return b->inner->ops->read(b->inner, sector) ^ 0x5a;
}

static const struct blk_ops disk_ops = { disk_read }, count_ops = { count_read }, crypt_ops = { crypt_read };

int main(void)
{
    struct blk disk  = { &disk_ops,  NULL,   0 };
    struct blk count = { &count_ops, &disk,  0 };     /* count(disk) */
    struct blk crypt = { &crypt_ops, &count, 0 };     /* crypt(count(disk)): the onion */
    struct blk *top = &crypt;                          /* callers only see the outer layer */
    for (long s = 1; s <= 3; s++) printf("sector %ld -> %d\n", s, top->ops->read(top, s));
    printf("reads counted by the middle layer: %d\n", count.hits);
    return 0;
}
