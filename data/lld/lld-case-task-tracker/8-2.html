// Task Tracker (Jira / Trello / To-do) in C++14: the same design as the Java version.
// Windows (MinGW g++ 6.3): g++ -std=c++14 -Wall -Wextra main.cpp -o tracker && ./tracker
// Linux (real board locks): g++ -std=c++14 -Wall -Wextra -pthread -DHAVE_STD_MUTEX main.cpp -o tracker
// MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and -pthread fails to link there, so BoardMutex
// falls back to a no-op with the same lock()/unlock(): the same locking code builds both ways.
#include <algorithm>
#include <cstdio>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <set>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>
#if defined(HAVE_STD_MUTEX) || defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
#include <mutex>
using BoardMutex = std::mutex;
#else
struct BoardMutex { void lock() {} void unlock() {} };   // MinGW g++ 6.3 "win32" threads: no std::mutex
#endif
#include <cassert>

// RAII lock like std::lock_guard; works with either mutex type above.
struct LockGuard {
    BoardMutex& m;
    explicit LockGuard(BoardMutex& mu) : m(mu) { m.lock(); }
    ~LockGuard() { m.unlock(); }
    LockGuard(const LockGuard&) = delete;
    LockGuard& operator=(const LockGuard&) = delete;
};

using std::string;
using std::vector;

enum class IssueType { STORY, TASK, BUG, SUBTASK };
enum class Priority { LOW, MEDIUM, HIGH, CRITICAL };
enum class Field { TITLE, DESCRIPTION, PRIORITY };
enum class SprintState { PLANNED, ACTIVE, CLOSED };
const char* name(IssueType t) { static const char* n[] = {"STORY", "TASK", "BUG", "SUBTASK"}; return n[(int)t]; }
const char* name(Priority p) { static const char* n[] = {"LOW", "MEDIUM", "HIGH", "CRITICAL"}; return n[(int)p]; }
const char* name(Field f) { static const char* n[] = {"TITLE", "DESCRIPTION", "PRIORITY"}; return n[(int)f]; }

struct TrackerError : std::runtime_error { using std::runtime_error::runtime_error; };
struct TransitionDenied : TrackerError { using TrackerError::TrackerError; };
struct StaleUpdate : TrackerError { using TrackerError::TrackerError; };

// A workflow is data: states plus allowed moves.
class Workflow {
public:
    string name, initial;
    Workflow(string n, string init) : name(std::move(n)), initial(std::move(init)) { add(initial); }
    Workflow& allow(const string& from, std::initializer_list<string> to) {
        add(from);
        for (const string& t : to) { next_[from].insert(t); add(t); }
        return *this;
    }
    Workflow& done(const string& s) { done_.insert(s); return *this; }
    bool allows(const string& from, const string& to) const {
        auto it = next_.find(from);
        return it != next_.end() && it->second.count(to);
    }
    bool isDone(const string& s) const { return done_.count(s) > 0; }
    const vector<string>& states() const { return order_; }
    static Workflow todo() { Workflow w("To-do", "TODO"); w.allow("TODO", {"DONE"}).allow("DONE", {"TODO"}).done("DONE"); return w; }
    static Workflow scrum() {
        Workflow w("Scrum", "TODO");
        w.allow("TODO", {"IN_PROGRESS"}).allow("IN_PROGRESS", {"IN_REVIEW", "TODO"})
         .allow("IN_REVIEW", {"DONE", "IN_PROGRESS"}).allow("DONE", {"TODO"}).done("DONE");
        return w;
    }
private:
    void add(const string& s) { if (!next_.count(s)) { next_[s]; order_.push_back(s); } }
    std::map<string, std::set<string>> next_;
    std::set<string> done_;
    vector<string> order_;
};

struct Issue {
    string key, title, description, status, reporter, assignee, parent, sprint;   // "" means none
    IssueType type; Priority priority;
    std::set<string> labels;
    vector<string> subtasks;
    int version = 1;
    long createdAt = 0, updatedAt = 0;
};

struct IssueView {           // an immutable copy for readers
    string key, title, description, status, assignee, parent, sprint;
    IssueType type; Priority priority; bool done; int version;
    vector<string> labels, subtasks;
    int number() const { return std::stoi(key.substr(key.find('-') + 1)); }
    string line() const { return key + " [" + name(priority) + "] " + title + " (" + (assignee.empty() ? "unassigned" : assignee) + ")"; }
};

struct Event {
    long seq, at; string actor, issue, action, from, to;
    string str() const {
        char buf[200];
        string detail = from == "-" ? to : from + " -> " + to;
        std::snprintf(buf, sizeof buf, "#%-2ld t=%02ld %-6s %-6s %-14s %s", seq, at, actor.c_str(), issue.c_str(), action.c_str(), detail.c_str());
        return buf;
    }
};

struct Sprint { string id, name; SprintState state = SprintState::PLANNED; vector<string> issues; };

struct Board;
using Guard = std::function<string(const Issue&, const string& to, const Board&)>;   // "" = allowed

struct Board {
    string key, name; Workflow workflow;
    vector<string> members, order;            // order = insertion order of issue keys
    vector<Guard> guards;
    std::map<string, Issue> issues;
    std::map<string, Sprint> sprints;
    vector<string> sprintOrder;
    vector<Event> log;
    int nextNumber = 1;
    BoardMutex mu;
    Board(string k, string n, Workflow w) : key(std::move(k)), name(std::move(n)), workflow(std::move(w)) {}
    bool member(const string& u) const { return std::find(members.begin(), members.end(), u) != members.end(); }
};

Guard needsAssignee(string state) {
    return [state](const Issue& i, const string& to, const Board&) {
        return to == state && i.assignee.empty() ? i.key + " needs an assignee before " + state : string();
    };
}
Guard subtasksDone() {
    return [](const Issue& i, const string& to, const Board& b) {
        if (!b.workflow.isDone(to)) return string();
        for (const string& k : i.subtasks)
            if (!b.workflow.isDone(b.issues.at(k).status)) return i.key + " has an open subtask " + k;
        return string();
    };
}

// Filters (Specification): combine with &, |, !
struct Spec {
    std::function<bool(const IssueView&)> fn;
    bool operator()(const IssueView& v) const { return fn(v); }
};
Spec operator&(Spec a, Spec b) { return {[a, b](const IssueView& v) { return a(v) && b(v); }}; }
Spec operator|(Spec a, Spec b) { return {[a, b](const IssueView& v) { return a(v) || b(v); }}; }
Spec operator!(Spec a) { return {[a](const IssueView& v) { return !a(v); }}; }
namespace Filters {
Spec open() { return {[](const IssueView& v) { return !v.done; }}; }
Spec status(string s) { return {[s](const IssueView& v) { return v.status == s; }}; }
Spec unassigned() { return {[](const IssueView& v) { return v.assignee.empty(); }}; }
Spec priorityAtLeast(Priority p) { return {[p](const IssueView& v) { return v.priority >= p; }}; }
Spec label(string l) { return {[l](const IssueView& v) { return std::count(v.labels.begin(), v.labels.end(), l) > 0; }}; }
Spec type(IssueType t) { return {[t](const IssueView& v) { return v.type == t; }}; }
Spec inSprint(string s) { return {[s](const IssueView& v) { return v.sprint == s; }}; }
Spec backlog() { return {[](const IssueView& v) { return v.sprint.empty(); }}; }
string lower(string s) { std::transform(s.begin(), s.end(), s.begin(), ::tolower); return s; }
Spec text(string q) {
    q = lower(q);
    return {[q](const IssueView& v) { return lower(v.title).find(q) != string::npos || lower(v.description).find(q) != string::npos; }};
}
bool byPriority(const IssueView& a, const IssueView& b) {
    return a.priority != b.priority ? a.priority > b.priority : a.number() < b.number();
}
}  // namespace Filters

using Listener = std::function<void(const Event&)>;

class TaskTracker {
public:
    explicit TaskTracker(std::function<long()> clock) : clock_(std::move(clock)) {}
    void addListener(Listener l) { listeners_.push_back(std::move(l)); }
    void addUser(const string& id) { if (!users_.insert(id).second) throw TrackerError("user " + id + " exists"); }
    void createBoard(const string& key, const string& nm, Workflow wf, const string& owner) {
        user(owner);
        if (boards_.count(key)) throw TrackerError("board " + key + " exists");
        boards_[key].reset(new Board(key, nm, std::move(wf)));
        boards_[key]->members.push_back(owner);
    }
    void addMember(const string& bk, const string& actor, const string& uid) {
        user(uid);
        write<int>(board(bk), actor, [&](Board& b, vector<Event>&) { if (!b.member(uid)) b.members.push_back(uid); return 0; });
    }
    void addGuard(const string& bk, Guard g) { Board& b = board(bk); LockGuard l(b.mu); b.guards.push_back(std::move(g)); }

    string createIssue(const string& bk, const string& actor, IssueType t, const string& title, Priority p) {
        if (t == IssueType::SUBTASK) throw TrackerError("use addSubtask for subtasks");
        return write<string>(board(bk), actor, [&](Board& b, vector<Event>& out) { return newIssue(b, out, actor, t, title, p, nullptr); });
    }
    string addSubtask(const string& parentKey, const string& actor, const string& title) {
        return write<string>(boardOf(parentKey), actor, [&](Board& b, vector<Event>& out) {
            Issue& parent = issue(b, parentKey);
            if (parent.type == IssueType::SUBTASK) throw TrackerError("a subtask cannot have subtasks");
            string key = newIssue(b, out, actor, IssueType::SUBTASK, title, parent.priority, &parent);
            Issue& p = issue(b, parentKey);        // re-fetch: map nodes are stable, but be explicit
            p.subtasks.push_back(key);
            touch(p);
            log(b, out, actor, parentKey, "SUBTASK_ADDED", "", key);
            return key;
        });
    }
    void assign(const string& key, const string& actor, const string& assignee) {
        write<int>(boardOf(key), actor, [&](Board& b, vector<Event>& out) {
            if (!assignee.empty() && !b.member(assignee)) throw TrackerError(assignee + " is not a member of " + b.key);
            Issue& i = issue(b, key);
            if (i.assignee == assignee) return 0;
            log(b, out, actor, key, "ASSIGNED", i.assignee, assignee);
            i.assignee = assignee;
            touch(i);
            return 0;
        });
    }
    void edit(const string& key, const string& actor, Field f, const string& value, int expected = -1) {
        write<int>(boardOf(key), actor, [&](Board& b, vector<Event>& out) {
            Issue& i = issue(b, key);
            if (expected >= 0 && expected != i.version)
                throw StaleUpdate(key + " changed since you loaded it (you have v" + std::to_string(expected) +
                                  ", current is v" + std::to_string(i.version) + ")");
            string old;
            if (f == Field::TITLE) { old = i.title; i.title = value; }
            else if (f == Field::DESCRIPTION) { old = i.description; i.description = value; }
            else { old = name(i.priority); i.priority = parsePriority(value); }
            touch(i);
            log(b, out, actor, key, string("EDIT_") + name(f), old, value);
            return 0;
        });
    }
    void addLabel(const string& key, const string& actor, const string& label) {
        write<int>(boardOf(key), actor, [&](Board& b, vector<Event>& out) {
            Issue& i = issue(b, key);
            if (i.labels.insert(label).second) { touch(i); log(b, out, actor, key, "LABEL_ADDED", "", label); }
            return 0;
        });
    }
    void move(const string& key, const string& actor, const string& to) {
        write<int>(boardOf(key), actor, [&](Board& b, vector<Event>& out) {
            Issue& i = issue(b, key);
            if (i.status == to) return 0;
            if (!b.workflow.allows(i.status, to))
                throw TransitionDenied(key + ": " + i.status + " -> " + to + " is not in the " + b.workflow.name + " workflow");
            for (const Guard& g : b.guards) { string why = g(i, to, b); if (!why.empty()) throw TransitionDenied(why); }
            log(b, out, actor, key, "MOVED", i.status, to);
            i.status = to;
            touch(i);
            return 0;
        });
    }
    vector<string> remove(const string& key, const string& actor) {
        return write<vector<string>>(boardOf(key), actor, [&](Board& b, vector<Event>& out) {
            Issue& i = issue(b, key);
            vector<string> gone = i.subtasks;
            gone.push_back(key);
            if (!i.parent.empty()) { Issue& p = b.issues.at(i.parent); erase(p.subtasks, key); touch(p); }
            if (!i.sprint.empty()) erase(b.sprints.at(i.sprint).issues, key);
            for (const string& k : gone) { b.issues.erase(k); erase(b.order, k); log(b, out, actor, k, "REMOVED", "", "deleted"); }
            return gone;
        });
    }
    string createSprint(const string& bk, const string& actor, const string& nm) {
        return write<string>(board(bk), actor, [&](Board& b, vector<Event>&) {
            string id = "S" + std::to_string(++sprintIds_);
            b.sprints[id] = Sprint{id, nm, SprintState::PLANNED, {}};
            b.sprintOrder.push_back(id);
            sprintBoard_[id] = b.key;
            return id;
        });
    }
    void addToSprint(const string& sid, const string& actor, const string& key) {
        write<int>(boardOfSprint(sid), actor, [&](Board& b, vector<Event>& out) {
            Sprint& s = b.sprints.at(sid);
            if (s.state == SprintState::CLOSED) throw TrackerError(s.name + " is closed");
            Issue& i = issue(b, key);
            if (i.type == IssueType::SUBTASK) throw TrackerError("subtasks follow their parent " + i.parent);
            if (i.sprint == sid) return 0;
            if (!i.sprint.empty()) erase(b.sprints.at(i.sprint).issues, key);
            s.issues.push_back(key);
            setSprint(b, out, actor, i, sid);
            return 0;
        });
    }
    void startSprint(const string& sid, const string& actor) {
        write<int>(boardOfSprint(sid), actor, [&](Board& b, vector<Event>&) {
            Sprint& s = b.sprints.at(sid);
            if (s.state != SprintState::PLANNED) throw TrackerError(s.name + " is not planned");
            for (auto& kv : b.sprints)
                if (kv.second.state == SprintState::ACTIVE) throw TrackerError(kv.second.name + " is already active on " + b.key);
            s.state = SprintState::ACTIVE;
            return 0;
        });
    }
    vector<string> completeSprint(const string& sid, const string& actor) {
        return write<vector<string>>(boardOfSprint(sid), actor, [&](Board& b, vector<Event>& out) {
            Sprint& s = b.sprints.at(sid);
            if (s.state != SprintState::ACTIVE) throw TrackerError(s.name + " is not active");
            vector<string> carried;
            for (const string& k : s.issues)
                if (!b.workflow.isDone(b.issues.at(k).status)) { carried.push_back(k); setSprint(b, out, actor, b.issues.at(k), ""); }
            for (const string& k : carried) erase(s.issues, k);
            s.state = SprintState::CLOSED;
            return carried;
        });
    }

    IssueView get(const string& key) {
        Board& b = boardOf(key);
        LockGuard l(b.mu);
        return view(b, issue(b, key));
    }
    vector<IssueView> search(const string& bk, const Spec& spec) {
        vector<IssueView> all = snapshot(board(bk)), out;
        for (const IssueView& v : all) if (spec(v)) out.push_back(v);
        std::stable_sort(out.begin(), out.end(), Filters::byPriority);
        return out;
    }
    vector<Event> history(const string& key) {
        Board& b = boardOf(key);
        LockGuard l(b.mu);
        vector<Event> out;
        for (const Event& e : b.log) if (e.issue == key) out.push_back(e);
        return out;
    }
    vector<Event> activityOf(const string& uid) {
        vector<Event> out;
        for (auto& kv : boards_) {
            LockGuard l(kv.second->mu);
            for (const Event& e : kv.second->log) if (e.actor == uid) out.push_back(e);
        }
        std::sort(out.begin(), out.end(), [](const Event& a, const Event& b) { return a.seq < b.seq; });
        return out;
    }
    SprintState sprintState(const string& sid) { Board& b = boardOfSprint(sid); LockGuard l(b.mu); return b.sprints.at(sid).state; }
    string render(const string& bk) {
        Board& b = board(bk);
        vector<IssueView> all = snapshot(b);
        string active = "none";
        {
            LockGuard l(b.mu);
            for (const string& id : b.sprintOrder) if (b.sprints.at(id).state == SprintState::ACTIVE) { active = b.sprints.at(id).name; break; }
        }
        std::ostringstream os;
        os << b.key << " " << b.name << " [" << b.workflow.name << "]  active sprint: " << active << "\n";
        char buf[200];
        for (const string& st : b.workflow.states()) {
            vector<IssueView> col = search(bk, Filters::status(st));
            if (col.empty()) { std::snprintf(buf, sizeof buf, "  %-11s | -\n", st.c_str()); os << buf; }
            for (size_t k = 0; k < col.size(); k++) {
                std::snprintf(buf, sizeof buf, "  %-11s | %s\n", k == 0 ? st.c_str() : "", col[k].line().c_str());
                os << buf;
            }
        }
        return os.str();
    }

private:
    std::function<long()> clock_;
    std::set<string> users_;
    std::map<string, std::unique_ptr<Board>> boards_;
    std::map<string, string> sprintBoard_;
    vector<Listener> listeners_;
    long seq_ = 0, sprintIds_ = 0;

    // Every write: membership check and mutation under the board lock; listeners after it is released.
    template <class T, class F> T write(Board& b, const string& actor, F body) {
        vector<Event> out;
        T result;
        {
            LockGuard l(b.mu);
            if (!b.member(actor)) throw TrackerError(actor + " is not a member of " + b.key);
            result = body(b, out);
        }
        for (const Event& e : out) for (const Listener& fn : listeners_) fn(e);
        return result;
    }
    string newIssue(Board& b, vector<Event>& out, const string& actor, IssueType t, const string& title, Priority p, const Issue* parent) {
        if (title.find_first_not_of(' ') == string::npos) throw TrackerError("title is required");
        string key = b.key + "-" + std::to_string(b.nextNumber++);
        Issue i;
        i.key = key; i.type = t; i.title = title; i.priority = p; i.status = b.workflow.initial; i.reporter = actor;
        i.createdAt = i.updatedAt = clock_();
        if (parent) { i.parent = parent->key; i.sprint = parent->sprint; }
        b.issues[key] = i;
        b.order.push_back(key);
        log(b, out, actor, key, "CREATED", "", string(name(t)) + " '" + title + "'");
        return key;
    }
    void setSprint(Board& b, vector<Event>& out, const string& actor, Issue& i, const string& sid) {
        string from = i.sprint.empty() ? "backlog" : b.sprints.at(i.sprint).name;
        string to = sid.empty() ? "backlog" : b.sprints.at(sid).name;
        i.sprint = sid;
        touch(i);
        log(b, out, actor, i.key, "SPRINT", from, to);
        for (const string& k : i.subtasks) { b.issues.at(k).sprint = sid; touch(b.issues.at(k)); }
    }
    void log(Board& b, vector<Event>& out, const string& actor, const string& key, const string& action, const string& from, const string& to) {
        Event e{++seq_, clock_(), actor, key, action, from.empty() ? "-" : from, to.empty() ? "-" : to};
        b.log.push_back(e);
        out.push_back(e);
    }
    void touch(Issue& i) { i.version++; i.updatedAt = clock_(); }
    static void erase(vector<string>& v, const string& k) { v.erase(std::remove(v.begin(), v.end(), k), v.end()); }
    static Priority parsePriority(const string& v) {
        for (int p = 0; p < 4; p++) if (v == name((Priority)p)) return (Priority)p;
        throw TrackerError("unknown priority " + v);
    }
    static IssueView view(const Board& b, const Issue& i) {
        IssueView v;
        v.key = i.key; v.title = i.title; v.description = i.description; v.status = i.status; v.assignee = i.assignee;
        v.parent = i.parent; v.sprint = i.sprint; v.type = i.type; v.priority = i.priority;
        v.done = b.workflow.isDone(i.status); v.version = i.version;
        v.labels.assign(i.labels.begin(), i.labels.end()); v.subtasks = i.subtasks;
        return v;
    }
    vector<IssueView> snapshot(Board& b) {
        LockGuard l(b.mu);
        vector<IssueView> all;
        for (const string& k : b.order) all.push_back(view(b, b.issues.at(k)));
        return all;
    }
    void user(const string& id) { if (!users_.count(id)) throw TrackerError("no user " + id); }
    Board& board(const string& k) { auto it = boards_.find(k); if (it == boards_.end()) throw TrackerError("no board " + k); return *it->second; }
    Board& boardOf(const string& key) {
        size_t d = key.find('-');
        if (d == string::npos) throw TrackerError("bad issue key " + key);
        return board(key.substr(0, d));
    }
    Board& boardOfSprint(const string& sid) {
        auto it = sprintBoard_.find(sid);
        if (it == sprintBoard_.end()) throw TrackerError("no sprint " + sid);
        return board(it->second);
    }
    static Issue& issue(Board& b, const string& key) {
        auto it = b.issues.find(key);
        if (it == b.issues.end()) throw TrackerError("no issue " + key);
        return it->second;
    }
};

static int checks = 0;
#define CHECK(cond) do { assert(cond); checks++; } while (0)
template <class E, class F> void expectFail(F f) {
    try { f(); } catch (const E&) { checks++; return; }
    assert(!"expected an exception");
}
template <class F> void tryIt(const string& label, F f) {
    try { f(); std::cout << "  ok      " << label << "\n"; }
    catch (const TrackerError& e) { std::cout << "  refused " << label << ": " << e.what() << "\n"; }
}
void show(const string& title, const vector<IssueView>& rows) {
    std::cout << "  " << title << ":\n";
    if (rows.empty()) std::cout << "    (none)\n";
    for (const IssueView& v : rows) std::cout << "    " << v.line() << " " << v.status << "\n";
}
string join(const vector<string>& v) {
    string s = "[";
    for (size_t i = 0; i < v.size(); i++) s += (i ? ", " : "") + v[i];
    return s + "]";
}
std::function<long()> counterClock() { auto t = std::make_shared<long>(0); return [t] { return ++*t; }; }

void demo() {
    TaskTracker tt(counterClock());
    tt.addListener([](const Event& e) {
        if (e.action == "ASSIGNED" && e.to != "-") std::cout << "  [notify] " << e.to << ": you were assigned " << e.issue << "\n";
    });
    for (const char* u : {"alice", "bob", "carol", "dave"}) tt.addUser(u);
    tt.createBoard("PAY", "Payments", Workflow::scrum(), "alice");
    tt.addMember("PAY", "alice", "bob");
    tt.addMember("PAY", "alice", "carol");
    tt.addGuard("PAY", needsAssignee("IN_PROGRESS"));
    tt.addGuard("PAY", subtasksDone());

    std::cout << "== 1. Create issues and subtasks, assign\n";
    string story = tt.createIssue("PAY", "alice", IssueType::STORY, "UPI refund flow", Priority::HIGH);
    string bug = tt.createIssue("PAY", "alice", IssueType::BUG, "Double debit on retry", Priority::CRITICAL);
    string task = tt.createIssue("PAY", "bob", IssueType::TASK, "Write on-call runbook", Priority::LOW);
    string api = tt.addSubtask(story, "alice", "Refund API");
    string db = tt.addSubtask(story, "alice", "Refund table migration");
    tt.assign(bug, "alice", "bob");
    tt.assign(story, "alice", "carol");
    tt.assign(api, "carol", "carol");
    tt.addLabel(story, "alice", "upi");
    tt.addLabel(bug, "alice", "upi");
    tryIt("assign " + task + " to dave", [&] { tt.assign(task, "alice", "dave"); });
    tryIt("dave creates an issue", [&] { tt.createIssue("PAY", "dave", IssueType::TASK, "x", Priority::LOW); });

    std::cout << "\n== 2. Sprint planning\n";
    string s1 = tt.createSprint("PAY", "alice", "Sprint 1");
    tt.addToSprint(s1, "alice", story);
    tt.addToSprint(s1, "alice", bug);
    tt.startSprint(s1, "alice");
    tryIt("add subtask " + db + " to the sprint", [&] { tt.addToSprint(s1, "alice", db); });
    string s2 = tt.createSprint("PAY", "alice", "Sprint 2");
    tryIt("start Sprint 2 while Sprint 1 is active", [&] { tt.startSprint(s2, "alice"); });

    std::cout << "\n== 3. Workflow moves\n";
    tryIt(task + " TODO -> IN_PROGRESS", [&] { tt.move(task, "bob", "IN_PROGRESS"); });
    tryIt(task + " TODO -> DONE", [&] { tt.move(task, "bob", "DONE"); });
    std::vector<std::pair<string, string>> walkers = {{bug, "bob"}, {story, "carol"}, {api, "carol"}};
    for (auto& kw : walkers)
        for (const char* st : {"IN_PROGRESS", "IN_REVIEW", "DONE"})
            tryIt(kw.first + " -> " + st, [&] { tt.move(kw.first, kw.second, st); });

    std::cout << "\n== 4. Two people edit the same issue\n";
    int v = tt.get(task).version;
    tryIt("alice renames " + task + " (she loaded v" + std::to_string(v) + ")",
          [&] { tt.edit(task, "alice", Field::TITLE, "Write payments on-call runbook", v); });
    tryIt("bob sets priority HIGH (he also loaded v" + std::to_string(v) + ")",
          [&] { tt.edit(task, "bob", Field::PRIORITY, "HIGH", v); });

    std::cout << "\n== 5. Filters\n";
    using namespace Filters;
    show("open and HIGH or above", tt.search("PAY", open() & priorityAtLeast(Priority::HIGH)));
    show("label upi and not done", tt.search("PAY", label("upi") & open()));
    show("unassigned or LOW", tt.search("PAY", unassigned() | !priorityAtLeast(Priority::MEDIUM)));
    show("text 'refund' in the active sprint", tt.search("PAY", text("refund") & inSprint(s1)));

    std::cout << "\n== 6. Board view\n" << tt.render("PAY");

    std::cout << "\n== 7. Close the sprint, remove a task\n";
    std::cout << "  carried over to the backlog: " << join(tt.completeSprint(s1, "alice")) << "\n";
    vector<string> backlogKeys;
    for (const IssueView& iv : tt.search("PAY", backlog() & !type(IssueType::SUBTASK))) backlogKeys.push_back(iv.key);
    std::cout << "  backlog now: " << join(backlogKeys) << "\n";
    std::cout << "  removed: " << join(tt.remove(task, "bob")) << "\n";

    std::cout << "\n== 8. Audit history of " << bug << "\n";
    for (const Event& e : tt.history(bug)) std::cout << "  " << e.str() << "\n";
    std::cout << "\n== 9. Everything carol did\n";
    for (const Event& e : tt.activityOf("carol")) std::cout << "  " << e.str() << "\n";
}

void runTests() {
    using namespace Filters;
    std::unique_ptr<TaskTracker> tt(fresh(Workflow::scrum()));
    tt->addGuard("T", needsAssignee("IN_PROGRESS"));
    tt->addGuard("T", subtasksDone());
    string k = tt->createIssue("T", "a", IssueType::TASK, "t1", Priority::MEDIUM);
    expectFail<TransitionDenied>([&] { tt->move(k, "a", "DONE"); });
    expectFail<TransitionDenied>([&] { tt->move(k, "a", "IN_PROGRESS"); });
    tt->assign(k, "a", "b");
    tt->move(k, "b", "IN_PROGRESS");
    CHECK(tt->get(k).status == "IN_PROGRESS");

    string p = tt->createIssue("T", "a", IssueType::STORY, "parent", Priority::HIGH);
    string c = tt->addSubtask(p, "a", "child");
    expectFail<TrackerError>([&] { tt->addSubtask(c, "a", "grandchild"); });
    tt->assign(p, "a", "a");
    tt->move(p, "a", "IN_PROGRESS");
    tt->move(p, "a", "IN_REVIEW");
    expectFail<TransitionDenied>([&] { tt->move(p, "a", "DONE"); });
    tt->assign(c, "a", "a");
    for (const char* st : {"IN_PROGRESS", "IN_REVIEW", "DONE"}) tt->move(c, "a", st);
    tt->move(p, "a", "DONE");
    CHECK(tt->get(p).done);

    expectFail<TrackerError>([&] { tt->createIssue("T", "x", IssueType::TASK, "t", Priority::LOW); });
    expectFail<TrackerError>([&] { tt->assign(k, "a", "x"); });
    expectFail<TrackerError>([&] { tt->move("T-99", "a", "DONE"); });
    expectFail<TrackerError>([&] { tt->edit(k, "a", Field::PRIORITY, "URGENT"); });

    int v = tt->get(k).version;
    tt->edit(k, "a", Field::TITLE, "renamed", v);
    CHECK(tt->get(k).version == v + 1);
    expectFail<StaleUpdate>([&] { tt->edit(k, "b", Field::TITLE, "lost update", v); });
    CHECK(tt->get(k).title == "renamed");

    std::unique_ptr<TaskTracker> sp(fresh(Workflow::scrum()));
    string a1 = sp->createIssue("T", "a", IssueType::STORY, "s", Priority::HIGH);
    string a2 = sp->addSubtask(a1, "a", "sub");
    string a3 = sp->createIssue("T", "a", IssueType::TASK, "quick", Priority::LOW);
    string s1 = sp->createSprint("T", "a", "S1"), s2 = sp->createSprint("T", "a", "S2");
    sp->addToSprint(s1, "a", a1);
    sp->addToSprint(s1, "a", a3);
    CHECK(sp->get(a2).sprint == s1);
    sp->startSprint(s1, "a");
    expectFail<TrackerError>([&] { sp->startSprint(s2, "a"); });
    for (const char* st : {"IN_PROGRESS", "IN_REVIEW", "DONE"}) sp->move(a3, "a", st);
    CHECK(sp->completeSprint(s1, "a") == vector<string>{a1});
    CHECK(sp->get(a1).sprint.empty() && sp->get(a2).sprint.empty() && sp->get(a3).sprint == s1);
    CHECK(sp->sprintState(s1) == SprintState::CLOSED);
    expectFail<TrackerError>([&] { sp->addToSprint(s1, "a", a1); });

    std::unique_ptr<TaskTracker> rm(fresh(Workflow::scrum()));
    string r1 = rm->createIssue("T", "a", IssueType::STORY, "r", Priority::LOW);
    string r2 = rm->addSubtask(r1, "a", "r-sub");
    string r3 = rm->addSubtask(r1, "a", "r-sub2");
    rm->remove(r3, "a");
    CHECK(rm->get(r1).subtasks == vector<string>{r2});
    CHECK((rm->remove(r1, "b") == vector<string>{r2, r1}));
    expectFail<TrackerError>([&] { rm->get(r1); });
    vector<Event> h = rm->history(r1);
    CHECK(h.front().action == "CREATED" && h.back().action == "REMOVED");

    std::unique_ptr<TaskTracker> f(fresh(Workflow::todo()));
    string f1 = f->createIssue("T", "a", IssueType::TASK, "Buy milk", Priority::LOW);
    string f2 = f->createIssue("T", "a", IssueType::TASK, "Pay rent", Priority::CRITICAL);
    string f3 = f->createIssue("T", "a", IssueType::TASK, "Pay electricity", Priority::CRITICAL);
    f->addLabel(f1, "a", "home");
    f->move(f2, "a", "DONE");
    vector<IssueView> r = f->search("T", text("pay") & open());
    CHECK(r.size() == 1 && r[0].key == f3);
    r = f->search("T", label("home") | priorityAtLeast(Priority::HIGH));
    CHECK(r.size() == 3 && r[0].key == f2 && r[1].key == f3 && r[2].key == f1);
    CHECK(f->search("T", !open()).size() == 1);
    f->move(f2, "a", "TODO");
    CHECK(!f->get(f2).done);
    vector<Event> hist = f->history(f2);
    CHECK(hist.size() == 3 && hist[1].to == "DONE" && hist[2].from == "DONE");

    std::unique_ptr<TaskTracker> ls(fresh(Workflow::scrum()));
    vector<string> seen;
    ls->addListener([&seen](const Event& e) { seen.push_back(e.action); });
    string l1 = ls->createIssue("T", "a", IssueType::TASK, "l", Priority::LOW);
    expectFail<TransitionDenied>([&] { ls->move(l1, "a", "DONE"); });
    CHECK(seen == vector<string>{"CREATED"});
}

Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was All 30 checks passed..

What this C++ version leaves out: nothing in the demo: it prints exactly the Java demo and then "All 30 checks passed." against Java's 38 (8 of the Java unit tests in runTests are not ported). On MinGW the board lock is a no-op; the Linux -DHAVE_STD_MUTEX build uses a real std::mutex per board (compiled and run, same output).

g++ -std=c++14 -Wall -Wextra main.cpp -o tracker && ./tracker prints the same demo, then All 30 checks passed.

From your world

The C++ version guards each board with a BoardMutex. With a normal toolchain that is std::mutex; the MinGW g++ 6.3 used to test this page has the "win32" thread model, which ships no std::mutex, so the file falls back to a no-op lock and the 8-thread test lives in the Java and Python versions. It is the same trick as a kernel spinlock_t that compiles to nothing on a uniprocessor build: the call sites stay, the implementation is chosen by configuration.
