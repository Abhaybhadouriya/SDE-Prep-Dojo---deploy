You will seeRead it as
x -> x.price (lambda)An anonymous function that can capture local variables: a callback and its void *ctx in one value. (a, b) -> a.x - b.x is the cmp you would hand to qsort.
Type::method (method reference)A pointer to a function that already exists, written without a wrapper: Post::score is &post_score.
.stream().filter(..).map(..).collect(..)A for-loop written as a pipeline: filter is if (!pred(x)) continue;, map transforms each element, collect/toList appends the survivors to a new array.
record P(int x, int y) {}An immutable struct: the compiler writes the constructor, the getters x()/y(), equals, hashCode and toString.
map.computeIfAbsent(k, f)Lookup-or-create in one call: find k; if it is missing, call f, insert the result and return it.
Comparator.comparing(f).reversed().thenComparing(g)Builds the cmp() function field by field: compare by f, negate the result, break ties with g.
case X -> ... (switch arrow)A case with no fall-through and an implied break; the whole switch can also return a value.
synchronizedTake the object's built-in mutex on entry and release it on every exit. It is reentrant: the owning thread may enter again.
throw new XException(..)Unwinds to the nearest catch: an error return that a caller cannot silently ignore.
List.of(..), Map.of(..)Immutable literal collections, like a static const initialiser list; writes to them throw.

final class Event {
    final long seq, at;
    final String actor, issueKey, action, from, to;
    Event(long seq, long at, String actor, String issueKey, String action, String from, String to) {
        this.seq = seq; this.at = at; this.actor = actor; this.issueKey = issueKey; this.action = action;
        this.from = from == null ? "-" : from; this.to = to == null ? "-" : to;

    @Override public String toString() {
        String detail = from.equals("-") ? to : from + " -> " + to;

import java.util.*;

/** A workflow is data: states plus allowed moves (a transition table), configurable per board. */
final class Workflow {
    final String name, initial;
    private final Map<String, Set<String>> next = new LinkedHashMap<>();
    private final Set<String> done = new HashSet<>();

    Workflow(String name, String initial) {
        this.name = name; this.initial = initial;
        next.put(initial, new LinkedHashSet<>());
    }

    Workflow allow(String from, String... to) {
        next.computeIfAbsent(from, k -> new LinkedHashSet<>()).addAll(List.of(to));   // C: look up k; if missing, allocate, insert and return it
        for (String t : to) next.computeIfAbsent(t, k -> new LinkedHashSet<>());
        return this;
    }

    Workflow done(String... states) { done.addAll(List.of(states)); return this; }

    boolean allows(String from, String to) { return next.getOrDefault(from, Set.of()).contains(to); }
    boolean isDone(String state) { return done.contains(state); }
    Set<String> states() { return Collections.unmodifiableSet(next.keySet()); }

    /** A personal to-do list: open or done, and you can reopen. */
    static Workflow todo() {
        return new Workflow("To-do", "TODO").allow("TODO", "DONE").allow("DONE", "TODO").done("DONE");
    }

    /** A typical Jira Scrum board. */
    static Workflow scrum() {
        return new Workflow("Scrum", "TODO")
                .allow("TODO", "IN_PROGRESS")
                .allow("IN_PROGRESS", "IN_REVIEW", "TODO")
                .allow("IN_REVIEW", "DONE", "IN_PROGRESS")
                .allow("DONE", "TODO")
                .done("DONE");
    }
}

/** A rule checked before a move (Strategy). Returns null if the move is fine, else the reason. */
interface TransitionGuard {
    String check(Issue issue, String to, Board board);

    static TransitionGuard needsAssignee(String state) {
        return (i, to, b) -> to.equals(state) && i.assignee == null ? i.key + " needs an assignee before " + state : null;
    }

    static TransitionGuard subtasksDone() {
        return (i, to, b) -> {
            if (!b.workflow.isDone(to)) return null;
            for (String k : i.subtasks)
                if (!b.workflow.isDone(b.issues.get(k).status)) return i.key + " has an open subtask " + k;
            return null;
        };
    }
}

import java.util.Comparator;
import java.util.function.Predicate;

/** Small composable filters (the Specification idea): combine with and(), or(), negate(). */
final class Filters {
    private Filters() {}

    static Predicate<IssueView> open() { return i -> !i.done(); }   // C: a lambda = a function pointer plus its captured context
    static Predicate<IssueView> status(String s) { return i -> i.status().equals(s); }
    static Predicate<IssueView> assignee(String user) { return i -> user.equals(i.assignee()); }
    static Predicate<IssueView> unassigned() { return i -> i.assignee() == null; }
    static Predicate<IssueView> priorityAtLeast(Priority p) { return i -> i.priority().compareTo(p) >= 0; }
    static Predicate<IssueView> label(String l) { return i -> i.labels().contains(l); }
    static Predicate<IssueView> type(IssueType t) { return i -> i.type() == t; }
    static Predicate<IssueView> inSprint(String id) { return i -> id.equals(i.sprint()); }
    static Predicate<IssueView> backlog() { return i -> i.sprint() == null; }

    static Predicate<IssueView> text(String q) {
        String lq = q.toLowerCase();
        return i -> i.title().toLowerCase().contains(lq) || i.description().toLowerCase().contains(lq);
    }

    /** Highest priority first, then the oldest issue (lowest number). */
    static final Comparator<IssueView> BY_PRIORITY =
            Comparator.comparing(IssueView::priority).reversed().thenComparingInt(IssueView::number);   // C: Type::method = a pointer to an existing function
}
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicLong;
import java.util.function.LongSupplier;
import java.util.function.Predicate;

/** Observer: notifications, webhooks, search indexing. The audit log is NOT a listener: it is written in the lock. */
interface TrackerListener {
    void onEvent(Event e);
}

/** The service facade. One lock per board; listeners are called after the lock is released. */
final class TaskTracker {
    private final Map<String, User> users = new ConcurrentHashMap<>();
    private final Map<String, Board> boards = new ConcurrentHashMap<>();
    private final Map<String, String> sprintBoard = new ConcurrentHashMap<>();
    private final List<TrackerListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicLong seq = new AtomicLong();
    private final AtomicLong sprintIds = new AtomicLong();
    private final LongSupplier clock;

    TaskTracker(LongSupplier clock) { this.clock = clock; }

    void addListener(TrackerListener l) { listeners.add(l); }

    User addUser(String id, String name) {
        User u = new User(id, name);
        if (users.putIfAbsent(id, u) != null) throw new TrackerException("user " + id + " exists");
        return u;
    }

    void createBoard(String key, String name, Workflow wf, String owner) {
        user(owner);
        Board b = new Board(key, name, wf);
        b.members.add(owner);
        if (boards.putIfAbsent(key, b) != null) throw new TrackerException("board " + key + " exists");
    }

    void addMember(String boardKey, String actor, String userId) {
        user(userId);
        write(board(boardKey), actor, (b, out) -> b.members.add(userId));
    }

    void addGuard(String boardKey, TransitionGuard g) {
        Board b = board(boardKey);
        synchronized (b) { b.guards.add(g); }
    }

    // ---------------- issues ----------------

    String createIssue(String boardKey, String actor, IssueType type, String title, Priority p) {
        if (type == IssueType.SUBTASK) throw new TrackerException("use addSubtask for subtasks");
        return write(board(boardKey), actor, (b, out) -> newIssue(b, out, actor, type, title, p, null));
    }

    String addSubtask(String parentKey, String actor, String title) {
        return write(boardOf(parentKey), actor, (b, out) -> {
            Issue parent = issue(b, parentKey);
            if (parent.type == IssueType.SUBTASK) throw new TrackerException("a subtask cannot have subtasks");
            String key = newIssue(b, out, actor, IssueType.SUBTASK, title, parent.priority, parent);
            parent.subtasks.add(key);
            touch(parent);
            log(b, out, actor, parentKey, "SUBTASK_ADDED", null, key);
            return key;
        });
    }

    void assign(String key, String actor, String assignee) {
        write(boardOf(key), actor, (b, out) -> {
            if (assignee != null && !b.members.contains(assignee))
                throw new TrackerException(assignee + " is not a member of " + b.key);
            Issue i = issue(b, key);
            if (Objects.equals(i.assignee, assignee)) return null;
            log(b, out, actor, key, "ASSIGNED", i.assignee, assignee);
            i.assignee = assignee;
            touch(i);
            return null;
        });
    }

    void edit(String key, String actor, Field f, String value) { edit(key, actor, f, value, -1); }

    /** expectedVersion = the version the caller's screen showed; -1 skips the check. */
    void edit(String key, String actor, Field f, String value, int expectedVersion) {
        write(boardOf(key), actor, (b, out) -> {
            Issue i = issue(b, key);
            checkVersion(i, expectedVersion);
            String old;
            switch (f) {
                case TITLE -> { old = i.title; i.title = value; }   // C: case X -> ... is a switch arm with no fall-through
                case DESCRIPTION -> { old = i.description; i.description = value; }
                case PRIORITY -> { old = i.priority.name(); i.priority = parsePriority(value); }
                default -> throw new IllegalStateException();
            }
            touch(i);
            log(b, out, actor, key, "EDIT_" + f, old, value);
            return null;
        });
    }

    void addLabel(String key, String actor, String label) {
        write(boardOf(key), actor, (b, out) -> {
            Issue i = issue(b, key);
            if (i.labels.add(label)) { touch(i); log(b, out, actor, key, "LABEL_ADDED", null, label); }
            return null;
        });
    }

    /** Move along the board's workflow. Checks the transition table first, then every guard. */
    void move(String key, String actor, String to) {
        write(boardOf(key), actor, (b, out) -> {
            Issue i = issue(b, key);
            if (i.status.equals(to)) return null;
            if (!b.workflow.allows(i.status, to))
                throw new TransitionDenied(key + ": " + i.status + " -> " + to + " is not in the " + b.workflow.name + " workflow");
            for (TransitionGuard g : b.guards) {
                String why = g.check(i, to, b);
                if (why != null) throw new TransitionDenied(why);
            }
            log(b, out, actor, key, "MOVED", i.status, to);
            i.status = to;
            touch(i);
            return null;
        });
    }

    /** Deletes the issue and its subtasks. The audit log keeps their history. */
    List<String> remove(String key, String actor) {
        return write(boardOf(key), actor, (b, out) -> {
            Issue i = issue(b, key);
            List<String> gone = new ArrayList<>(i.subtasks);
            gone.add(key);
            if (i.parentKey != null) {
                Issue p = b.issues.get(i.parentKey);
                p.subtasks.remove(key);
                touch(p);
            }
            if (i.sprintId != null) b.sprints.get(i.sprintId).issues.remove(key);
            for (String k : gone) { b.issues.remove(k); log(b, out, actor, k, "REMOVED", null, "deleted"); }
            return gone;
        });
    }

    // ---------------- sprints ----------------

    String createSprint(String boardKey, String actor, String name) {
        return write(board(boardKey), actor, (b, out) -> {
            String id = "S" + sprintIds.incrementAndGet();
            b.sprints.put(id, new Sprint(id, b.key, name));
            sprintBoard.put(id, b.key);
            return id;
        });
    }

    void addToSprint(String sprintId, String actor, String key) {
        write(boardOfSprint(sprintId), actor, (b, out) -> {
            Sprint s = b.sprints.get(sprintId);
            if (s.state == SprintState.CLOSED) throw new TrackerException(s.name + " is closed");
            Issue i = issue(b, key);
            if (i.type == IssueType.SUBTASK) throw new TrackerException("subtasks follow their parent " + i.parentKey);
            if (sprintId.equals(i.sprintId)) return null;
            if (i.sprintId != null) b.sprints.get(i.sprintId).issues.remove(key);
            s.issues.add(key);
            setSprint(b, out, actor, i, sprintId);
            return null;
        });
    }

    void startSprint(String sprintId, String actor) {
        write(boardOfSprint(sprintId), actor, (b, out) -> {
            Sprint s = b.sprints.get(sprintId);
            if (s.state != SprintState.PLANNED) throw new TrackerException(s.name + " is " + s.state);
            for (Sprint other : b.sprints.values())
                if (other.state == SprintState.ACTIVE) throw new TrackerException(other.name + " is already active on " + b.key);
            s.state = SprintState.ACTIVE;
            return null;
        });
    }

    /** Closes the sprint. Unfinished issues (and their subtasks) go back to the backlog. Returns their keys. */
    List<String> completeSprint(String sprintId, String actor) {
        return write(boardOfSprint(sprintId), actor, (b, out) -> {
            Sprint s = b.sprints.get(sprintId);
            if (s.state != SprintState.ACTIVE) throw new TrackerException(s.name + " is not active");
            List<String> carried = new ArrayList<>();
            for (String k : s.issues) {
                Issue i = b.issues.get(k);
                if (!b.workflow.isDone(i.status)) { carried.add(k); setSprint(b, out, actor, i, null); }
            }
            s.issues.removeAll(carried);
            s.state = SprintState.CLOSED;
            return carried;
        });
    }

    // ---------------- reads (return copies) ----------------

    IssueView get(String key) {
        Board b = boardOf(key);
        synchronized (b) { return issue(b, key).view(b.workflow.isDone(issue(b, key).status)); }
    }

    /** Copy a snapshot under the lock, filter and sort outside it. */
    List<IssueView> search(String boardKey, Predicate<IssueView> filter) {
        List<IssueView> all = snapshot(board(boardKey));
        return all.stream().filter(filter).sorted(Filters.BY_PRIORITY).toList();   // C: stream() = a for-loop written as filter/map/collect
    }

    List<Event> history(String key) {
        Board b = boardOf(key);
        synchronized (b) { return b.log.stream().filter(e -> e.issueKey.equals(key)).toList(); }
    }

    List<Event> activityOf(String userId) {
        List<Event> all = new ArrayList<>();
        for (Board b : boards.values()) synchronized (b) { for (Event e : b.log) if (e.actor.equals(userId)) all.add(e); }
        all.sort(Comparator.comparingLong(e -> e.seq));
        return all;
    }

    SprintState sprintState(String sprintId) {
        Board b = boardOfSprint(sprintId);
        synchronized (b) { return b.sprints.get(sprintId).state; }
    }

    int logSize(String boardKey) {
        Board b = board(boardKey);
        synchronized (b) { return b.log.size(); }
    }

    /** A Trello-style view: one column per workflow state. */
    String render(String boardKey) {
        Board b = board(boardKey);
        List<IssueView> all = snapshot(b);
        String active;
        synchronized (b) {
            active = b.sprints.values().stream().filter(s -> s.state == SprintState.ACTIVE).map(s -> s.name).findFirst().orElse("none");
        }
        StringBuilder sb = new StringBuilder(b.key + " " + b.name + " [" + b.workflow.name + "]  active sprint: " + active + "\n");
        for (String st : b.workflow.states()) {
            List<IssueView> col = all.stream().filter(Filters.status(st)).sorted(Filters.BY_PRIORITY).toList();
            if (col.isEmpty()) sb.append(String.format("  %-11s | -%n", st));
            for (int k = 0; k < col.size(); k++)
                sb.append(String.format("  %-11s | %s%n", k == 0 ? st : "", col.get(k).line()));
        }
        return sb.toString();
    }

    // ---------------- internals ----------------

    private interface Body<T> { T run(Board b, List<Event> out); }

    /** Every write: check membership, mutate under the board lock, then notify listeners outside it. */
    private <T> T write(Board b, String actor, Body<T> body) {
        List<Event> out = new ArrayList<>();
        T result;
        synchronized (b) {
            if (!b.members.contains(actor)) throw new TrackerException(actor + " is not a member of " + b.key);
            result = body.run(b, out);
        }
        for (Event e : out) for (TrackerListener l : listeners) l.onEvent(e);
        return result;
    }

    private String newIssue(Board b, List<Event> out, String actor, IssueType type, String title, Priority p, Issue parent) {
        if (title == null || title.isBlank()) throw new TrackerException("title is required");
        String key = b.key + "-" + b.nextNumber++;
        Issue i = new Issue(key, type, title, p, b.workflow.initial, actor, clock.getAsLong());
        if (parent != null) { i.parentKey = parent.key; i.sprintId = parent.sprintId; }
        b.issues.put(key, i);
        log(b, out, actor, key, "CREATED", null, type + " '" + title + "'");
        return key;
    }

    private void setSprint(Board b, List<Event> out, String actor, Issue i, String sprintId) {
        String from = i.sprintId == null ? "backlog" : b.sprints.get(i.sprintId).name;
        String to = sprintId == null ? "backlog" : b.sprints.get(sprintId).name;
        i.sprintId = sprintId;
        touch(i);
        log(b, out, actor, i.key, "SPRINT", from, to);
        for (String k : i.subtasks) { b.issues.get(k).sprintId = sprintId; touch(b.issues.get(k)); }
    }

    private void log(Board b, List<Event> out, String actor, String key, String action, String from, String to) {
        Event e = new Event(seq.incrementAndGet(), clock.getAsLong(), actor, key, action, from, to);
        b.log.add(e);
        out.add(e);
    }

    private void touch(Issue i) { i.version++; i.updatedAt = clock.getAsLong(); }

    private static void checkVersion(Issue i, int expected) {
        if (expected >= 0 && expected != i.version)
            throw new StaleUpdate(i.key + " changed since you loaded it (you have v" + expected + ", current is v" + i.version + ")");
    }

    private static Priority parsePriority(String v) {
        try { return Priority.valueOf(v); }
        catch (IllegalArgumentException e) { throw new TrackerException("unknown priority " + v); }
    }

    private List<IssueView> snapshot(Board b) {
        synchronized (b) {
            List<IssueView> all = new ArrayList<>();
            for (Issue i : b.issues.values()) all.add(i.view(b.workflow.isDone(i.status)));
            return all;
        }
    }

    private User user(String id) {
        User u = users.get(id);
        if (u == null) throw new TrackerException("no user " + id);
        return u;
    }

    private Board board(String key) {
        Board b = boards.get(key);
        if (b == null) throw new TrackerException("no board " + key);
        return b;
    }

    private Board boardOf(String issueKey) {
        int dash = issueKey.indexOf('-');
        if (dash < 0) throw new TrackerException("bad issue key " + issueKey);
        return board(issueKey.substring(0, dash));
    }

    private Board boardOfSprint(String sprintId) {
        String bk = sprintBoard.get(sprintId);
        if (bk == null) throw new TrackerException("no sprint " + sprintId);
        return board(bk);
    }

    private static Issue issue(Board b, String key) {
        Issue i = b.issues.get(key);
        if (i == null) throw new TrackerException("no issue " + key);
        return i;
    }
}

/*
 * Task Tracker (Jira / Trello / To-do) - machine-coding demo.
 * Scope agreed: boards with members and a configurable workflow, issues (story, task, bug) with one level
 * of subtasks, assign, edit, labels, workflow moves with guards, sprints (one active per board, unfinished
 * work returns to the backlog), composable filters, an append-only activity log per board.
 * Out of scope: comments, attachments, permissions beyond membership, persistence, UI.
 */
public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static void expectFail(Class<? extends RuntimeException> type, String what, Runnable r) {
        try { r.run(); }
        catch (RuntimeException e) {
            if (type.isInstance(e)) { checks++; return; }
            throw e;
        }
        throw new AssertionError("expected " + type.getSimpleName() + ": " + what);
    }

    static void tryIt(String label, Runnable r) {
        try { r.run(); System.out.println("  ok      " + label); }
        catch (TrackerException e) { System.out.println("  refused " + label + ": " + e.getMessage()); }
    }

    static void show(String title, List<IssueView> rows) {
        System.out.println("  " + title + ":");
        if (rows.isEmpty()) System.out.println("    (none)");
        for (IssueView v : rows) System.out.println("    " + v.line() + " " + v.status());
    }

    static TaskTracker newTracker() {
        AtomicLong t = new AtomicLong();
        return new TaskTracker(t::incrementAndGet);   // a fake clock keeps the output deterministic
    }

    public static void main(String[] args) throws Exception {
        demo();
        runTests();
        System.out.println("\nAll " + checks + " checks passed.");
    }

    static void demo() {
        TaskTracker tt = newTracker();
        tt.addListener(e -> {
            if (e.action.equals("ASSIGNED") && !e.to.equals("-"))
                System.out.println("  [notify] " + e.to + ": you were assigned " + e.issueKey);
        });
        for (String u : List.of("alice", "bob", "carol", "dave")) tt.addUser(u, u);
        tt.createBoard("PAY", "Payments", Workflow.scrum(), "alice");
        tt.addMember("PAY", "alice", "bob");
        tt.addMember("PAY", "alice", "carol");
        tt.addGuard("PAY", TransitionGuard.needsAssignee("IN_PROGRESS"));
        tt.addGuard("PAY", TransitionGuard.subtasksDone());

        System.out.println("== 1. Create issues and subtasks, assign");
        String story = tt.createIssue("PAY", "alice", IssueType.STORY, "UPI refund flow", Priority.HIGH);
        String bug = tt.createIssue("PAY", "alice", IssueType.BUG, "Double debit on retry", Priority.CRITICAL);
        String task = tt.createIssue("PAY", "bob", IssueType.TASK, "Write on-call runbook", Priority.LOW);
        String api = tt.addSubtask(story, "alice", "Refund API");
        String db = tt.addSubtask(story, "alice", "Refund table migration");
        tt.assign(bug, "alice", "bob");
        tt.assign(story, "alice", "carol");
        tt.assign(api, "carol", "carol");
        tt.addLabel(story, "alice", "upi");
        tt.addLabel(bug, "alice", "upi");
        tryIt("assign " + task + " to dave", () -> tt.assign(task, "alice", "dave"));
        tryIt("dave creates an issue", () -> tt.createIssue("PAY", "dave", IssueType.TASK, "x", Priority.LOW));

        System.out.println("\n== 2. Sprint planning");
        String s1 = tt.createSprint("PAY", "alice", "Sprint 1");
        tt.addToSprint(s1, "alice", story);
        tt.addToSprint(s1, "alice", bug);
        tt.startSprint(s1, "alice");
        tryIt("add subtask " + db + " to the sprint", () -> tt.addToSprint(s1, "alice", db));
        String s2 = tt.createSprint("PAY", "alice", "Sprint 2");
        tryIt("start Sprint 2 while Sprint 1 is active", () -> tt.startSprint(s2, "alice"));

        System.out.println("\n== 3. Workflow moves");
        tryIt(task + " TODO -> IN_PROGRESS", () -> tt.move(task, "bob", "IN_PROGRESS"));
        tryIt(task + " TODO -> DONE", () -> tt.move(task, "bob", "DONE"));
        for (String st : List.of("IN_PROGRESS", "IN_REVIEW", "DONE")) tryIt(bug + " -> " + st, () -> tt.move(bug, "bob", st));
        for (String st : List.of("IN_PROGRESS", "IN_REVIEW", "DONE")) tryIt(story + " -> " + st, () -> tt.move(story, "carol", st));
        for (String st : List.of("IN_PROGRESS", "IN_REVIEW", "DONE")) tryIt(api + " -> " + st, () -> tt.move(api, "carol", st));

        System.out.println("\n== 4. Two people edit the same issue");
        int v = tt.get(task).version();
        tryIt("alice renames " + task + " (she loaded v" + v + ")",
                () -> tt.edit(task, "alice", Field.TITLE, "Write payments on-call runbook", v));
        tryIt("bob sets priority HIGH (he also loaded v" + v + ")",
                () -> tt.edit(task, "bob", Field.PRIORITY, "HIGH", v));

        System.out.println("\n== 5. Filters");
        show("open and HIGH or above", tt.search("PAY", Filters.open().and(Filters.priorityAtLeast(Priority.HIGH))));
        show("label upi and not done", tt.search("PAY", Filters.label("upi").and(Filters.open())));
        show("unassigned or LOW", tt.search("PAY", Filters.unassigned().or(Filters.priorityAtLeast(Priority.MEDIUM).negate())));
        show("text 'refund' in the active sprint", tt.search("PAY", Filters.text("refund").and(Filters.inSprint(s1))));

        System.out.println("\n== 6. Board view");
        System.out.print(tt.render("PAY"));

        System.out.println("\n== 7. Close the sprint, remove a task");
        System.out.println("  carried over to the backlog: " + tt.completeSprint(s1, "alice"));
        System.out.println("  backlog now: " + tt.search("PAY", Filters.backlog().and(Filters.type(IssueType.SUBTASK).negate()))
                .stream().map(IssueView::key).toList());
        System.out.println("  removed: " + tt.remove(task, "bob"));

        System.out.println("\n== 8. Audit history of " + bug);
        tt.history(bug).forEach(e -> System.out.println("  " + e));

        System.out.println("\n== 9. Everything carol did");
        tt.activityOf("carol").forEach(e -> System.out.println("  " + e));
    }

    // ---------------------------------------------------------------- tests

    static TaskTracker board(String wfName) {
        TaskTracker tt = newTracker();
        for (String u : List.of("a", "b", "x")) tt.addUser(u, u);
        tt.createBoard("T", "Test", wfName.equals("todo") ? Workflow.todo() : Workflow.scrum(), "a");
        tt.addMember("T", "a", "b");
        return tt;
    }

    static void runTests() throws Exception {
        // 1. the transition table
        Workflow w = Workflow.scrum();
        check(w.allows("TODO", "IN_PROGRESS") && !w.allows("TODO", "DONE"), "scrum table");
        check(w.isDone("DONE") && !w.isDone("IN_REVIEW"), "done states");

        // 2. illegal move and guard: assignee needed before IN_PROGRESS
        TaskTracker tt = board("scrum");
        tt.addGuard("T", TransitionGuard.needsAssignee("IN_PROGRESS"));
        tt.addGuard("T", TransitionGuard.subtasksDone());
        String k = tt.createIssue("T", "a", IssueType.TASK, "t1", Priority.MEDIUM);
        expectFail(TransitionDenied.class, "TODO -> DONE", () -> tt.move(k, "a", "DONE"));
        expectFail(TransitionDenied.class, "no assignee", () -> tt.move(k, "a", "IN_PROGRESS"));
        tt.assign(k, "a", "b");
        tt.move(k, "b", "IN_PROGRESS");
        check(tt.get(k).status().equals("IN_PROGRESS"), "moved after assign");

        // 3. parent cannot be DONE while a subtask is open; subtask of subtask is refused
        String p = tt.createIssue("T", "a", IssueType.STORY, "parent", Priority.HIGH);
        String c = tt.addSubtask(p, "a", "child");
        expectFail(TrackerException.class, "nested subtask", () -> tt.addSubtask(c, "a", "grandchild"));
        tt.assign(p, "a", "a");
        tt.move(p, "a", "IN_PROGRESS");
        tt.move(p, "a", "IN_REVIEW");
        expectFail(TransitionDenied.class, "open subtask", () -> tt.move(p, "a", "DONE"));
        tt.assign(c, "a", "a");
        for (String st : List.of("IN_PROGRESS", "IN_REVIEW", "DONE")) tt.move(c, "a", st);
        tt.move(p, "a", "DONE");
        check(tt.get(p).done(), "parent done after child");

        // 4. membership and bad input
        expectFail(TrackerException.class, "non-member actor", () -> tt.createIssue("T", "x", IssueType.TASK, "t", Priority.LOW));
        expectFail(TrackerException.class, "non-member assignee", () -> tt.assign(k, "a", "x"));
        expectFail(TrackerException.class, "unknown issue", () -> tt.move("T-99", "a", "DONE"));
        expectFail(TrackerException.class, "blank title", () -> tt.createIssue("T", "a", IssueType.TASK, " ", Priority.LOW));
        expectFail(TrackerException.class, "bad priority", () -> tt.edit(k, "a", Field.PRIORITY, "URGENT"));

        // 5. optimistic version check
        int v = tt.get(k).version();
        tt.edit(k, "a", Field.TITLE, "renamed", v);
        check(tt.get(k).version() == v + 1, "version bumps");
        expectFail(StaleUpdate.class, "stale edit", () -> tt.edit(k, "b", Field.TITLE, "lost update", v));
        check(tt.get(k).title().equals("renamed"), "stale edit did not land");

        // 6. sprints: one active, closed sprint is frozen, unfinished work and subtasks return to the backlog
        TaskTracker sp = board("scrum");
        String a1 = sp.createIssue("T", "a", IssueType.STORY, "s", Priority.HIGH);
        String a2 = sp.addSubtask(a1, "a", "sub");
        String a3 = sp.createIssue("T", "a", IssueType.TASK, "quick", Priority.LOW);
        String s1 = sp.createSprint("T", "a", "S1"), s2 = sp.createSprint("T", "a", "S2");
        sp.addToSprint(s1, "a", a1);
        sp.addToSprint(s1, "a", a3);
        check(s1.equals(sp.get(a2).sprint()), "subtask follows parent into sprint");
        sp.startSprint(s1, "a");
        expectFail(TrackerException.class, "second active sprint", () -> sp.startSprint(s2, "a"));
        for (String st : List.of("IN_PROGRESS", "IN_REVIEW", "DONE")) sp.move(a3, "a", st);
        check(sp.completeSprint(s1, "a").equals(List.of(a1)), "only unfinished carried over");
        check(sp.get(a1).sprint() == null && sp.get(a2).sprint() == null && s1.equals(sp.get(a3).sprint()), "backlog after close");
        check(sp.sprintState(s1) == SprintState.CLOSED, "closed");
        expectFail(TrackerException.class, "closed sprint", () -> sp.addToSprint(s1, "a", a1));

        // 7. remove cascades to subtasks; the history survives
        TaskTracker rm = board("scrum");
        String r1 = rm.createIssue("T", "a", IssueType.STORY, "r", Priority.LOW);
        String r2 = rm.addSubtask(r1, "a", "r-sub");
        String r3 = rm.addSubtask(r1, "a", "r-sub2");
        rm.remove(r3, "a");
        check(rm.get(r1).subtasks().equals(List.of(r2)), "parent forgets removed subtask");
        check(rm.remove(r1, "b").equals(List.of(r2, r1)), "cascade");
        expectFail(TrackerException.class, "gone", () -> rm.get(r1));
        List<Event> h = rm.history(r1);
        check(h.get(0).action.equals("CREATED") && h.get(h.size() - 1).action.equals("REMOVED"), "history kept");

        // 8. filters compose; sort = priority desc, then number
        TaskTracker f = board("todo");
        String f1 = f.createIssue("T", "a", IssueType.TASK, "Buy milk", Priority.LOW);
        String f2 = f.createIssue("T", "a", IssueType.TASK, "Pay rent", Priority.CRITICAL);
        String f3 = f.createIssue("T", "a", IssueType.TASK, "Pay electricity", Priority.CRITICAL);
        f.addLabel(f1, "a", "home");
        f.move(f2, "a", "DONE");
        Predicate<IssueView> payOpen = Filters.text("pay").and(Filters.open());
        check(f.search("T", payOpen).stream().map(IssueView::key).toList().equals(List.of(f3)), "and");
        check(f.search("T", Filters.label("home").or(Filters.priorityAtLeast(Priority.HIGH)))
                .stream().map(IssueView::key).toList().equals(List.of(f2, f3, f1)), "or + sort");
        check(f.search("T", Filters.open().negate()).size() == 1, "negate");
        f.move(f2, "a", "TODO");   // to-do lists can reopen
        check(!f.get(f2).done(), "reopen");

        // 9. the audit log: every change, in order
        List<Event> hist = f.history(f2);
        check(hist.size() == 3 && hist.get(1).to.equals("DONE") && hist.get(2).from.equals("DONE"), "history of f2");
        for (int i = 1; i < hist.size(); i++) check(hist.get(i).seq > hist.get(i - 1).seq, "seq increases");

        // 10. listeners run after the write; a refused write notifies nobody
        TaskTracker ls = board("scrum");
        List<String> seen = new CopyOnWriteArrayList<>();
        ls.addListener(e -> seen.add(e.action));
        String l1 = ls.createIssue("T", "a", IssueType.TASK, "l", Priority.LOW);
        expectFail(TransitionDenied.class, "denied", () -> ls.move(l1, "a", "DONE"));
        check(seen.equals(List.of("CREATED")), "listener saw only the successful write");

        // 11. concurrency: 8 threads create 250 issues each on one board, and move them
        TaskTracker cc = board("todo");
        ExecutorService pool = Executors.newFixedThreadPool(8);
        List<Future<List<String>>> futures = new ArrayList<>();
        for (int t = 0; t < 8; t++) {
            String actor = t % 2 == 0 ? "a" : "b";
            futures.add(pool.submit(() -> {
                List<String> mine = new ArrayList<>();
                for (int n = 0; n < 250; n++) {
                    String key = cc.createIssue("T", actor, IssueType.TASK, "job " + n, Priority.MEDIUM);
                    cc.move(key, actor, "DONE");
                    mine.add(key);
                }
                return mine;
            }));
        }
        Set<String> keys = new HashSet<>();
        for (Future<List<String>> fu : futures) keys.addAll(fu.get());
        pool.shutdown();
        check(keys.size() == 2000, "2000 unique keys");
        check(cc.search("T", Filters.open()).isEmpty(), "all done");
        check(cc.logSize("T") == 4000, "2 events per issue");
    }
}
