Code: notification channels

A complete, runnable program in each language. Notifier owns the workflow; three creators pick their channel; a fourth (WhatsAppNotifier) is added at the end without touching Notifier. The format() hook shows that creators can also customise other steps.

// Factory Method: the base class owns the workflow; each subclass decides which product to create.
// Run: java FactoryMethodDemo.java   (Java 21)
import java.util.List;

public class FactoryMethodDemo {

    // ----- Product -----
    interface Channel {
        String name();
        boolean send(String to, String body);   // true = accepted for delivery
    }

    // ----- Concrete products -----
    static final class EmailChannel implements Channel {
        public String name() { return "email"; }
        public boolean send(String to, String body) {
            System.out.println("  [email -> " + to + "@example.com] " + body.replace("\n", " | "));
            return true;
        }
    }
    static final class SmsChannel implements Channel {
        static final int MAX = 40;               // tiny limit so the demo shows truncation
        public String name() { return "sms"; }
        public boolean send(String to, String body) {
            String b = body.length() > MAX ? body.substring(0, MAX - 3) + "..." : body;
            System.out.println("  [sms   -> +91-" + to + "] " + b);
            return true;
        }
    }
    static final class PushChannel implements Channel {
        public String name() { return "push"; }
        public boolean send(String to, String body) {
            if (to.isBlank()) return false;      // no device token
            System.out.println("  [push  -> device:" + to + "] " + body);
            return true;
        }
    }

    // ----- Creator: declares the factory method and USES it inside its own logic -----
    abstract static class Notifier {
        protected abstract Channel createChannel();          // the factory method

        protected String format(String msg) { return msg; }  // optional hook with a default

        public final String notifyUser(String user, String msg) {
            // <-- the pattern: the base class calls the hook; the subclass picks the class
            Channel ch = createChannel();                    // subclass picks the concrete class
            boolean ok = ch.send(user, format(msg));
            return (ok ? "sent via " : "FAILED via ") + ch.name();
        }
    }

    // ----- Concrete creators: one small override each -----
    static final class EmailNotifier extends Notifier {
        protected Channel createChannel() { return new EmailChannel(); }
        @Override protected String format(String msg) { return "Hi,\n" + msg + "\n- Team Dojo"; }
    }
    static final class SmsNotifier extends Notifier {
        protected Channel createChannel() { return new SmsChannel(); }
    }
    static final class PushNotifier extends Notifier {
        protected Channel createChannel() { return new PushChannel(); }
    }

    // ----- Added later: a new channel = one product + one creator. Notifier is not edited. -----
    static final class WhatsAppChannel implements Channel {
        public String name() { return "whatsapp"; }
        public boolean send(String to, String body) {
            System.out.println("  [wa    -> " + to + "] " + body);
            return true;
        }
    }
    static final class WhatsAppNotifier extends Notifier {
        protected Channel createChannel() { return new WhatsAppChannel(); }
    }

    public static void main(String[] args) {
        List<Notifier> notifiers = List.of(
                new EmailNotifier(), new SmsNotifier(), new PushNotifier(), new WhatsAppNotifier());
        String msg = "Your order #881 is out for delivery and will arrive by 7 pm";
        for (Notifier n : notifiers) {
            String result = n.notifyUser("u42", msg);
            System.out.println(n.getClass().getSimpleName() + ": " + result);
        }
        // Failure path: same workflow, the product decides.
        System.out.println("PushNotifier (no token): " + new PushNotifier().notifyUser(" ", msg));
    }
}
"""Factory Method: the base class owns the workflow; each subclass decides which product to create.
Run: python factory_method.py   (Python 3.11)"""
from __future__ import annotations

from abc import ABC, abstractmethod


# ----- Product -----
class Channel(ABC):
    name = "?"

    @abstractmethod
    def send(self, to: str, body: str) -> bool: ...


# ----- Concrete products -----
class EmailChannel(Channel):
    name = "email"

    def send(self, to: str, body: str) -> bool:
        print(f"  [email -> {to}@example.com] " + body.replace("\n", " | "))
        return True


class SmsChannel(Channel):
    name = "sms"
    MAX = 40  # tiny limit so the demo shows truncation

    def send(self, to: str, body: str) -> bool:
        b = body if len(body) <= self.MAX else body[: self.MAX - 3] + "..."
        print(f"  [sms   -> +91-{to}] {b}")
        return True


class PushChannel(Channel):
    name = "push"

    def send(self, to: str, body: str) -> bool:
        if not to.strip():
            return False  # no device token
        print(f"  [push  -> device:{to}] {body}")
        return True


# ----- Creator -----
class Notifier(ABC):
    @abstractmethod
    def create_channel(self) -> Channel:
        """The factory method."""

    def format(self, msg: str) -> str:  # optional hook
        return msg

    def notify_user(self, user: str, msg: str) -> str:
        ch = self.create_channel()      # subclass picks the concrete class
        ok = ch.send(user, self.format(msg))
        return ("sent via " if ok else "FAILED via ") + ch.name


# ----- Concrete creators -----
class EmailNotifier(Notifier):
    def create_channel(self) -> Channel:
        return EmailChannel()

    def format(self, msg: str) -> str:
        return f"Hi,\n{msg}\n- Team Dojo"


class SmsNotifier(Notifier):
    def create_channel(self) -> Channel:
        return SmsChannel()


class PushNotifier(Notifier):
    def create_channel(self) -> Channel:
        return PushChannel()


# ----- Added later: one product + one creator. Notifier is not edited. -----
class WhatsAppChannel(Channel):
    name = "whatsapp"

    def send(self, to: str, body: str) -> bool:
        print(f"  [wa    -> {to}] {body}")
        return True


class WhatsAppNotifier(Notifier):
    def create_channel(self) -> Channel:
        return WhatsAppChannel()


if __name__ == "__main__":
    msg = "Your order #881 is out for delivery and will arrive by 7 pm"
    for n in (EmailNotifier(), SmsNotifier(), PushNotifier(), WhatsAppNotifier()):
        result = n.notify_user("u42", msg)
        print(f"{type(n).__name__}: {result}")
    print("PushNotifier (no token): " + PushNotifier().notify_user(" ", msg))

#include <iostream>
#include <memory>
#include <string>
#include <vector>

// ----- Product -----
struct Channel {
    virtual ~Channel() = default;
    virtual const char* name() const = 0;
    virtual bool send(const std::string& to, const std::string& body) = 0;
};

// ----- Concrete products -----
struct EmailChannel : Channel {
    const char* name() const override { return "email"; }
    bool send(const std::string& to, const std::string& body) override {
        std::string b = body;
        for (std::string::size_type i = 0; (i = b.find('\n', i)) != std::string::npos;) b.replace(i, 1, " | ");
        std::cout << "  [email -> " << to << "@example.com] " << b << "\n";
        return true;
    }
};
struct SmsChannel : Channel {
    static constexpr std::size_t MAX = 40;  // tiny limit so the demo shows truncation
    const char* name() const override { return "sms"; }
    bool send(const std::string& to, const std::string& body) override {
        std::string b = body.size() > MAX ? body.substr(0, MAX - 3) + "..." : body;
        std::cout << "  [sms   -> +91-" << to << "] " << b << "\n";
        return true;
    }
};
constexpr std::size_t SmsChannel::MAX;
struct PushChannel : Channel {
    const char* name() const override { return "push"; }
    bool send(const std::string& to, const std::string& body) override {
        if (to.find_first_not_of(' ') == std::string::npos) return false;  // no device token
        std::cout << "  [push  -> device:" << to << "] " << body << "\n";
        return true;
    }
};

// ----- Creator -----
class Notifier {
public:
    virtual ~Notifier() = default;
    virtual const char* label() const = 0;
    std::string notifyUser(const std::string& user, const std::string& msg) {
        std::unique_ptr<Channel> ch = createChannel();   // subclass picks the concrete class
        bool ok = ch->send(user, format(msg));
        return std::string(ok ? "sent via " : "FAILED via ") + ch->name();
    }
protected:
    virtual std::unique_ptr<Channel> createChannel() const = 0;     // the factory method
    virtual std::string format(const std::string& msg) const { return msg; }  // optional hook
};

// ----- Concrete creators -----
class EmailNotifier : public Notifier {
public:
    const char* label() const override { return "EmailNotifier"; }
protected:
    std::unique_ptr<Channel> createChannel() const override { return std::make_unique<EmailChannel>(); }
    std::string format(const std::string& msg) const override { return "Hi,\n" + msg + "\n- Team Dojo"; }
};
class SmsNotifier : public Notifier {
public:
    const char* label() const override { return "SmsNotifier"; }
protected:
    std::unique_ptr<Channel> createChannel() const override { return std::make_unique<SmsChannel>(); }
};
class PushNotifier : public Notifier {
public:
    const char* label() const override { return "PushNotifier"; }
protected:
    std::unique_ptr<Channel> createChannel() const override { return std::make_unique<PushChannel>(); }
};

// ----- Added later: one product + one creator. Notifier is not edited. -----
struct WhatsAppChannel : Channel {
    const char* name() const override { return "whatsapp"; }
    bool send(const std::string& to, const std::string& body) override {
        std::cout << "  [wa    -> " << to << "] " << body << "\n";
        return true;
    }
};
class WhatsAppNotifier : public Notifier {
public:
    const char* label() const override { return "WhatsAppNotifier"; }
protected:
    std::unique_ptr<Channel> createChannel() const override { return std::make_unique<WhatsAppChannel>(); }
};

int main() {
    std::vector<std::unique_ptr<Notifier>> notifiers;
    notifiers.push_back(std::make_unique<EmailNotifier>());
    notifiers.push_back(std::make_unique<SmsNotifier>());
    notifiers.push_back(std::make_unique<PushNotifier>());
    notifiers.push_back(std::make_unique<WhatsAppNotifier>());
    const std::string msg = "Your order #881 is out for delivery and will arrive by 7 pm";
    for (auto& n : notifiers) {
        std::string result = n->notifyUser("u42", msg);
        std::cout << n->label() << ": " << result << "\n";
    }
    PushNotifier p;
    std::cout << "PushNotifier (no token): " << p.notifyUser(" ", msg) << "\n";
    return 0;
}

Reading the code like a reviewer

  • notifyUser is final in Java. Subclasses may change the product and the hook, not the workflow. (Python has no final at run time; C++ makes it non-virtual.)
  • createChannel is protected: it is an extension point for subclasses, not a public API for clients.
  • The product is created per call here, which is simple and safe when channels are cheap. If a channel holds a connection pool, create it once in the constructor (still via the factory method) or cache it.
  • C++ note: never call a virtual factory method from the base constructor. During base construction the object is still a base, so the call does not reach the subclass override (pure virtual call = crash). Java does dispatch to the override, but the subclass fields are not initialised yet, which is just as dangerous. Call it from a normal method, as here.

From your world

The same idea in C, in the shape the kernel uses (our own simplified sketch, not kernel source):

struct channel_ops { int (*send)(void *self, const char *to, const char *body); };
struct channel     { const struct channel_ops *ops; };

struct notifier_family {                        /* the "creator" */
    const char *name;
    struct channel *(*create)(void);            /* the factory method */
};

int notify_user(const struct notifier_family *f, const char *to, const char *msg)
{
    struct channel *ch = f->create();           /* family decides the concrete object */
    if (!ch)
        return -ENOMEM;
    int ret = ch->ops->send(ch, to, msg);       /* workflow only uses the interface */
    kfree(ch);
    return ret;
}

When the create op is fixed by the type of the struct it lives in (each "subclass" fills it in once, the way every file system sets super_operations->alloc_inode), this is the Factory Method analogue: the caller never chooses the product, the type does. Contrast struct net_proto_family registered through sock_register() and looked up by the family number passed to socket(2): selecting a creator by a key at call time is a parameterised factory or registry, see Simple Factory. A vtable and a registered ops struct are both tables of function pointers; the difference the pattern cares about is who picks the entry: the object's type (Factory Method) or a key at call time (registry).
