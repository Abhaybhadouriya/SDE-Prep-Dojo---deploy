import java.util.*;

// ---------- model ----------
enum GameState { IN_PROGRESS, WON, DRAW }

final class Player {
    final int index;            // 0..k-1, the turn order
    final String name;
    final char symbol;
    Player(int index, String name, char symbol) { this.index = index; this.name = name; this.symbol = symbol; }
}

final class Move {
    final int row, col;
    final Player player;
    Move(int row, int col, Player player) { this.row = row; this.col = col; this.player = player; }
}

class InvalidMoveException extends RuntimeException {
    private static final long serialVersionUID = 1L;
    InvalidMoveException(String msg) { super(msg); }
}

// ---------- board: owns cells only, knows nothing about rules ----------
final class Board {
    static final char EMPTY = '.';
    final int n;
    private final char[][] cells;
    private int filled = 0;

    Board(int n) {
        if (n < 3) throw new IllegalArgumentException("board must be at least 3x3");
        this.n = n;
        cells = new char[n][n];
        for (char[] r : cells) Arrays.fill(r, EMPTY);
    }
    boolean inside(int r, int c) { return r >= 0 && r < n && c >= 0 && c < n; }
    char at(int r, int c) { return cells[r][c]; }
    boolean isFull() { return filled == n * n; }

    void place(int r, int c, char s) {
        if (!inside(r, c)) throw new InvalidMoveException("cell (" + r + "," + c + ") is off the board");
        if (cells[r][c] != EMPTY) throw new InvalidMoveException("cell (" + r + "," + c + ") is taken");
        cells[r][c] = s;
        filled++;
    }
    void clear(int r, int c) { cells[r][c] = EMPTY; filled--; }

    String render() {
        StringBuilder sb = new StringBuilder();
        for (char[] r : cells) sb.append("  ").append(new String(r)).append('\n');
        return sb.toString();
    }
}

// ---------- strategy: how do we decide that the last move won? ----------
interface WinStrategy {
    boolean isWinningMove(Board b, Move m);   // called after the move is placed
    void undo(Board b, Move m);               // called before the move is removed
}

/** Full row / column / diagonal of the same symbol. O(1) per move: per-player line counters. */
final class CounterWinStrategy implements WinStrategy {
    private final int n;
    private final int[][] rows, cols;       // [player][line]
    private final int[] diag, anti;         // [player]
    CounterWinStrategy(int n, int players) {
        this.n = n;
        rows = new int[players][n]; cols = new int[players][n];
        diag = new int[players]; anti = new int[players];
    }
    public boolean isWinningMove(Board b, Move m) { return bump(m, +1); }
    public void undo(Board b, Move m) { bump(m, -1); }
    private boolean bump(Move m, int d) {
        int p = m.player.index, r = m.row, c = m.col;
        rows[p][r] += d; cols[p][c] += d;
        if (r == c) diag[p] += d;
        if (r + c == n - 1) anti[p] += d;
        return rows[p][r] == n || cols[p][c] == n || diag[p] == n || anti[p] == n;
    }
}

/** k in a row anywhere (Gomoku, Connect-style). O(k) per move: walk 4 directions from the move. */
final class KInARowWinStrategy implements WinStrategy {
    private static final int[][] DIRS = {{0, 1}, {1, 0}, {1, 1}, {1, -1}};
    private final int k;
    KInARowWinStrategy(int k) { this.k = k; }
    public boolean isWinningMove(Board b, Move m) {
        for (int[] d : DIRS) {
            int run = 1 + count(b, m, d[0], d[1]) + count(b, m, -d[0], -d[1]);
            if (run >= k) return true;
        }
        return false;
    }
    private int count(Board b, Move m, int dr, int dc) {
        int r = m.row + dr, c = m.col + dc, cnt = 0;
        while (b.inside(r, c) && b.at(r, c) == m.player.symbol && cnt < k) { cnt++; r += dr; c += dc; }
        return cnt;
    }
    public void undo(Board b, Move m) { /* stateless: reads the board */ }
}

// ---------- game: turn order, state machine, history ----------
final class Game {
    private final Board board;
    private final List<Player> players;
    private final WinStrategy rule;
    private final Deque<Move> history = new ArrayDeque<>();
    private GameState state = GameState.IN_PROGRESS;
    private Player winner = null;
    private int turn = 0;

    Game(int n, List<Player> players, WinStrategy rule) {
        if (players.size() < 2) throw new IllegalArgumentException("need at least 2 players");
        Set<Character> seen = new HashSet<>();
        for (Player p : players)
            if (!seen.add(p.symbol) || p.symbol == Board.EMPTY) throw new IllegalArgumentException("symbols must be unique");
        this.board = new Board(n);
        this.players = List.copyOf(players);
        this.rule = rule;
    }

    /** Classic game: n x n, win = full line, O(1) check. */
    static Game classic(int n, List<Player> players) {
        return new Game(n, players, new CounterWinStrategy(n, players.size()));
    }

    synchronized GameState makeMove(String playerName, int r, int c) {
        if (state != GameState.IN_PROGRESS) throw new InvalidMoveException("game is over: " + state);
        Player p = players.get(turn);
        if (!p.name.equals(playerName)) throw new InvalidMoveException("not " + playerName + "'s turn, it is " + p.name + "'s");
        board.place(r, c, p.symbol);                 // validates bounds + empty, throws before any change
        Move m = new Move(r, c, p);
        history.push(m);
        if (rule.isWinningMove(board, m)) { state = GameState.WON; winner = p; }
        else if (board.isFull()) state = GameState.DRAW;
        else turn = (turn + 1) % players.size();
        return state;
    }

    synchronized void undo() {
        if (history.isEmpty()) throw new InvalidMoveException("nothing to undo");
        Move m = history.pop();
        rule.undo(board, m);
        board.clear(m.row, m.col);
        state = GameState.IN_PROGRESS;
        winner = null;
        turn = m.player.index;                       // that player moves again
    }

    synchronized GameState state() { return state; }
    synchronized Optional<Player> winner() { return Optional.ofNullable(winner); }   // C: Optional = a pointer that may be NULL, made explicit
    synchronized Player current() { return players.get(turn); }
    synchronized int movesPlayed() { return history.size(); }
    synchronized String render() { return board.render(); }
}

public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        passed++;
    }
    static void play(Game g, int[][] moves, String... names) {
        for (int i = 0; i < moves.length; i++) g.makeMove(names[i % names.length], moves[i][0], moves[i][1]);
    }
    static String rejected(Runnable r) {
        try { r.run(); return "accepted"; } catch (InvalidMoveException e) { return e.getMessage(); }
    }

    public static void main(String[] args) {
        List<Player> xo = List.of(new Player(0, "Asha", 'X'), new Player(1, "Ben", 'O'));   // C: immutable literal collection

        // 1. X wins on the main diagonal
        Game g = Game.classic(3, xo);
        play(g, new int[][]{{0, 0}, {0, 1}, {1, 1}, {0, 2}, {2, 2}}, "Asha", "Ben");
        System.out.println("Game 1 (3x3):\n" + g.render() + "  state=" + g.state() + " winner=" + g.winner().get().name);
        check(g.state() == GameState.WON && g.winner().get().symbol == 'X', "diagonal win");

        // 2. invalid moves leave the game unchanged
        Game g2 = Game.classic(3, xo);
        g2.makeMove("Asha", 1, 1);
        String e1 = rejected(() -> g2.makeMove("Ben", 1, 1));   // C: a lambda = a function pointer plus its captured context
        String e2 = rejected(() -> g2.makeMove("Asha", 0, 0));
        String e3 = rejected(() -> g2.makeMove("Ben", 3, 0));
        System.out.println("rejected: " + e1 + " | " + e2 + " | " + e3);
        check(e1.contains("taken") && e2.contains("not Asha's turn") && e3.contains("off the board"), "invalid moves");
        check(g2.movesPlayed() == 1 && g2.current().name.equals("Ben"), "state untouched after rejects");

        // 3. draw
        Game g3 = Game.classic(3, xo);
        play(g3, new int[][]{{0, 0}, {0, 1}, {0, 2}, {1, 1}, {1, 0}, {1, 2}, {2, 1}, {2, 0}, {2, 2}}, "Asha", "Ben");
        System.out.println("Game 3 (draw):\n" + g3.render() + "  state=" + g3.state());
        check(g3.state() == GameState.DRAW && g3.winner().isEmpty(), "draw");
        check(rejected(() -> g3.makeMove("Asha", 0, 0)).contains("game is over"), "no move after end");

        // 4. undo a winning move, then play elsewhere
        g.undo();
        check(g.state() == GameState.IN_PROGRESS && g.current().name.equals("Asha"), "undo reopens game");
        g.makeMove("Asha", 2, 2);   // replay: only wins again if the diagonal counter went 3 -> 2 -> 3
        check(g.state() == GameState.WON, "counters were rolled back by undo");

        // 5. 4x4 with three players, column win for player C
        List<Player> three = List.of(new Player(0, "A", 'X'), new Player(1, "B", 'O'), new Player(2, "C", 'Z'));
        Game g4 = Game.classic(4, three);
        play(g4, new int[][]{{0, 0}, {0, 1}, {0, 3}, {1, 0}, {1, 1}, {1, 3}, {2, 1}, {2, 0}, {2, 3}, {3, 2}, {3, 1}, {3, 3}},
             "A", "B", "C");
        System.out.println("Game 4 (4x4, 3 players):\n" + g4.render() + "  state=" + g4.state() + " winner=" + g4.winner().get().name);
        check(g4.winner().get().name.equals("C"), "3-player column win");

        // 6. Gomoku-style: 6x6, 4 in a row wins (anti-diagonal), O(k) strategy
        Game g5 = new Game(6, xo, new KInARowWinStrategy(4));
        play(g5, new int[][]{{0, 5}, {0, 0}, {1, 4}, {0, 1}, {2, 3}, {5, 5}, {3, 2}}, "Asha", "Ben");
        System.out.println("Game 5 (6x6, 4 in a row): state=" + g5.state() + " winner=" + g5.winner().get().name);
        check(g5.state() == GameState.WON && g5.winner().get().name.equals("Asha"), "k-in-a-row anti-diagonal");

        System.out.println("All " + passed + " checks passed");
    }
}

"""Tic-Tac-Toe LLD: N x N board, k players, pluggable win rule, O(1) win check, undo."""
import threading
from dataclasses import dataclass
from enum import Enum
from typing import List, Optional


class GameState(Enum):
    IN_PROGRESS = "IN_PROGRESS"
    WON = "WON"
    DRAW = "DRAW"


@dataclass(frozen=True)
class Player:
    index: int      # turn order 0..k-1
    name: str
    symbol: str


@dataclass(frozen=True)
class Move:
    row: int
    col: int
    player: Player


class InvalidMoveError(Exception):
    pass


class Board:
    EMPTY = "."

    def __init__(self, n: int):
        if n < 3:
            raise ValueError("board must be at least 3x3")
        self.n = n
        self._cells = [[Board.EMPTY] * n for _ in range(n)]
        self._filled = 0

    def inside(self, r: int, c: int) -> bool:
        return 0 <= r < self.n and 0 <= c < self.n

    def at(self, r: int, c: int) -> str:
        return self._cells[r][c]

    def is_full(self) -> bool:
        return self._filled == self.n * self.n

    def place(self, r: int, c: int, s: str) -> None:
        if not self.inside(r, c):
            raise InvalidMoveError(f"cell ({r},{c}) is off the board")
        if self._cells[r][c] != Board.EMPTY:
            raise InvalidMoveError(f"cell ({r},{c}) is taken")
        self._cells[r][c] = s
        self._filled += 1

    def clear(self, r: int, c: int) -> None:
        self._cells[r][c] = Board.EMPTY
        self._filled -= 1

    def render(self) -> str:
        return "\n".join("  " + "".join(row) for row in self._cells)


class WinStrategy:
    def is_winning_move(self, board: Board, m: Move) -> bool:
        raise NotImplementedError

    def undo(self, board: Board, m: Move) -> None:
        raise NotImplementedError


class CounterWinStrategy(WinStrategy):
    """Full line wins. O(1) per move with per-player counters."""

    def __init__(self, n: int, players: int):
        self.n = n
        self.rows = [[0] * n for _ in range(players)]
        self.cols = [[0] * n for _ in range(players)]
        self.diag = [0] * players
        self.anti = [0] * players

    def _bump(self, m: Move, d: int) -> bool:
        p, r, c, n = m.player.index, m.row, m.col, self.n
        self.rows[p][r] += d
        self.cols[p][c] += d
        if r == c:
            self.diag[p] += d
        if r + c == n - 1:
            self.anti[p] += d
        return n in (self.rows[p][r], self.cols[p][c], self.diag[p], self.anti[p])

    def is_winning_move(self, board: Board, m: Move) -> bool:
        return self._bump(m, +1)

    def undo(self, board: Board, m: Move) -> None:
        self._bump(m, -1)


class KInARowWinStrategy(WinStrategy):
    """k in a row anywhere. O(k) per move: walk 4 directions from the last move."""
    DIRS = ((0, 1), (1, 0), (1, 1), (1, -1))

    def __init__(self, k: int):
        self.k = k

    def _count(self, b: Board, m: Move, dr: int, dc: int) -> int:
        r, c, cnt = m.row + dr, m.col + dc, 0
        while b.inside(r, c) and b.at(r, c) == m.player.symbol and cnt < self.k:
            cnt += 1
            r += dr
            c += dc
        return cnt

    def is_winning_move(self, board: Board, m: Move) -> bool:
        return any(1 + self._count(board, m, dr, dc) + self._count(board, m, -dr, -dc) >= self.k
                   for dr, dc in self.DIRS)

    def undo(self, board: Board, m: Move) -> None:
        pass  # stateless


class Game:
    def __init__(self, n: int, players: List[Player], rule: WinStrategy):
        if len(players) < 2:
            raise ValueError("need at least 2 players")
        symbols = [p.symbol for p in players]
        if len(set(symbols)) != len(symbols) or Board.EMPTY in symbols:
            raise ValueError("symbols must be unique")
        self._board = Board(n)
        self._players = list(players)
        self._rule = rule
        self._history: List[Move] = []
        self._state = GameState.IN_PROGRESS
        self._winner: Optional[Player] = None
        self._turn = 0
        self._lock = threading.Lock()

    @staticmethod
    def classic(n: int, players: List[Player]) -> "Game":
        return Game(n, players, CounterWinStrategy(n, len(players)))

    def make_move(self, player_name: str, r: int, c: int) -> GameState:
        with self._lock:
            if self._state != GameState.IN_PROGRESS:
                raise InvalidMoveError(f"game is over: {self._state.value}")
            p = self._players[self._turn]
            if p.name != player_name:
                raise InvalidMoveError(f"not {player_name}'s turn, it is {p.name}'s")
            self._board.place(r, c, p.symbol)
            m = Move(r, c, p)
            self._history.append(m)
            if self._rule.is_winning_move(self._board, m):
                self._state, self._winner = GameState.WON, p
            elif self._board.is_full():
                self._state = GameState.DRAW
            else:
                self._turn = (self._turn + 1) % len(self._players)
            return self._state

    def undo(self) -> None:
        with self._lock:
            if not self._history:
                raise InvalidMoveError("nothing to undo")
            m = self._history.pop()
            self._rule.undo(self._board, m)
            self._board.clear(m.row, m.col)
            self._state, self._winner = GameState.IN_PROGRESS, None
            self._turn = m.player.index

    @property
    def state(self) -> GameState:
        return self._state

    @property
    def winner(self) -> Optional[Player]:
        return self._winner

    @property
    def current(self) -> Player:
        return self._players[self._turn]

    @property
    def moves_played(self) -> int:
        return len(self._history)

    def render(self) -> str:
        return self._board.render()


def play(g: Game, moves, *names) -> None:
    for i, (r, c) in enumerate(moves):
        g.make_move(names[i % len(names)], r, c)


def main() -> None:
    xo = [Player(0, "Asha", "X"), Player(1, "Ben", "O")]
    g = Game.classic(3, xo)
    play(g, [(0, 0), (0, 1), (1, 1), (0, 2), (2, 2)], "Asha", "Ben")

    errors = []

    play(g3, [(0, 0), (0, 1), (0, 2), (1, 1), (1, 0), (1, 2), (2, 1), (2, 0), (2, 2)], "Asha", "Ben")
    print(f"Game 3 (draw):\n{g3.render()}\n  state={g3.state.value}")

    three = [Player(0, "A", "X"), Player(1, "B", "O"), Player(2, "C", "Z")]
    g4 = Game.classic(4, three)
    play(g4, [(0, 0), (0, 1), (0, 3), (1, 0), (1, 1), (1, 3), (2, 1), (2, 0), (2, 3), (3, 2), (3, 1), (3, 3)],
         "A", "B", "C")
    print(f"Game 4 (4x4, 3 players):\n{g4.render()}\n  state={g4.state.value} winner={g4.winner.name}")

    g5 = Game(6, xo, KInARowWinStrategy(4))
    play(g5, [(0, 5), (0, 0), (1, 4), (0, 1), (2, 3), (5, 5), (3, 2)], "Asha", "Ben")
    print(f"Game 5 (6x6, 4 in a row): state={g5.state.value} winner={g5.winner.name}")

    main()

    def test_concurrent_same_cell_only_one_wins(self):

Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was All 7 checks passed.

// Tic-Tac-Toe LLD in C++14: N x N, k players, pluggable win rule (Strategy), O(1) win check, undo.
#include <cstdlib>
#include <iostream>
#include <memory>
#include <set>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>
#include <mutex>
// Locking: one facade lock, a real std::recursive_mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). Recursive because public methods call each other,
// just as Java's synchronized is re-entrant. MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and
// -pthread fails to link there, so the default build uses a no-op Mutex with the same lock()/unlock():
// the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::recursive_mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

enum class GameState { IN_PROGRESS, WON, DRAW };
static const char* name(GameState s) {
    return s == GameState::WON ? "WON" : s == GameState::DRAW ? "DRAW" : "IN_PROGRESS";
}

struct Player { int index; std::string name; char symbol; };
struct Move { int row, col; const Player* player; };

struct InvalidMove : std::runtime_error {
    explicit InvalidMove(const std::string& m) : std::runtime_error(m) {}
};

class Board {
public:
    static const char EMPTY = '.';
    explicit Board(int n) : n_(n), cells_(n, std::string(n, EMPTY)) {
        if (n < 3) throw std::invalid_argument("board must be at least 3x3");
    }
    int n() const { return n_; }
    bool inside(int r, int c) const { return r >= 0 && r < n_ && c >= 0 && c < n_; }
    char at(int r, int c) const { return cells_[r][c]; }
    bool isFull() const { return filled_ == n_ * n_; }
    void place(int r, int c, char s) {
        std::string cell = "cell (" + std::to_string(r) + "," + std::to_string(c) + ")";
        if (!inside(r, c)) throw InvalidMove(cell + " is off the board");
        if (cells_[r][c] != EMPTY) throw InvalidMove(cell + " is taken");
        cells_[r][c] = s;
        ++filled_;
    }
    void clear(int r, int c) { cells_[r][c] = EMPTY; --filled_; }
    std::string render() const {
        std::string out;
        for (const auto& row : cells_) out += "  " + row + "\n";
        return out;
    }
private:
    int n_, filled_ = 0;
    std::vector<std::string> cells_;
};

// Strategy interface = a C ops table with one implementation per rule.
struct WinStrategy {
    virtual ~WinStrategy() = default;
    virtual bool isWinningMove(const Board& b, const Move& m) = 0;
    virtual void undo(const Board& b, const Move& m) = 0;
};

class CounterWinStrategy : public WinStrategy {
public:
    CounterWinStrategy(int n, int players)
        : n_(n), rows_(players, std::vector<int>(n)), cols_(players, std::vector<int>(n)),
          diag_(players), anti_(players) {}
    bool isWinningMove(const Board&, const Move& m) override { return bump(m, +1); }
    void undo(const Board&, const Move& m) override { bump(m, -1); }
private:
    bool bump(const Move& m, int d) {
        int p = m.player->index, r = m.row, c = m.col;
        rows_[p][r] += d; cols_[p][c] += d;
        if (r == c) diag_[p] += d;
        if (r + c == n_ - 1) anti_[p] += d;
        return rows_[p][r] == n_ || cols_[p][c] == n_ || diag_[p] == n_ || anti_[p] == n_;
    }
    int n_;
    std::vector<std::vector<int>> rows_, cols_;
    std::vector<int> diag_, anti_;
};

class KInARowWinStrategy : public WinStrategy {
public:
    explicit KInARowWinStrategy(int k) : k_(k) {}
    bool isWinningMove(const Board& b, const Move& m) override {
        static const int dirs[4][2] = {{0, 1}, {1, 0}, {1, 1}, {1, -1}};
        for (const auto& d : dirs)
            if (1 + count(b, m, d[0], d[1]) + count(b, m, -d[0], -d[1]) >= k_) return true;
        return false;
    }
    void undo(const Board&, const Move&) override {}
private:
    int count(const Board& b, const Move& m, int dr, int dc) const {
        int r = m.row + dr, c = m.col + dc, cnt = 0;
        while (b.inside(r, c) && b.at(r, c) == m.player->symbol && cnt < k_) { ++cnt; r += dr; c += dc; }
        return cnt;
    }
    int k_;
};

class Game {
    std::unique_ptr<Mutex> mu_{new Mutex()};   // the facade lock; behind a pointer so Game stays movable (Game::classic returns one)
public:
    Game(int n, std::vector<Player> players, std::unique_ptr<WinStrategy> rule)
        : board_(n), players_(std::move(players)), rule_(std::move(rule)) {
        if (players_.size() < 2) throw std::invalid_argument("need at least 2 players");
        std::set<char> seen;
        for (const auto& p : players_)
            if (!seen.insert(p.symbol).second || p.symbol == Board::EMPTY)
                throw std::invalid_argument("symbols must be unique");
    }
    static Game classic(int n, const std::vector<Player>& players) {
        return Game(n, players, std::unique_ptr<WinStrategy>(new CounterWinStrategy(n, (int)players.size())));
    }
    GameState makeMove(const std::string& who, int r, int c) { LockGuard g(*mu_);
        if (state_ != GameState::IN_PROGRESS) throw InvalidMove(std::string("game is over: ") + name(state_));
        const Player& p = players_[turn_];
        if (p.name != who) throw InvalidMove("not " + who + "'s turn, it is " + p.name + "'s");
        board_.place(r, c, p.symbol);
        Move m{r, c, &p};
        history_.push_back(m);
        if (rule_->isWinningMove(board_, m)) { state_ = GameState::WON; winner_ = &p; }
        else if (board_.isFull()) state_ = GameState::DRAW;
        else turn_ = (turn_ + 1) % (int)players_.size();
        return state_;
    }
    void undo() { LockGuard g(*mu_);
        if (history_.empty()) throw InvalidMove("nothing to undo");
        Move m = history_.back();
        history_.pop_back();
        rule_->undo(board_, m);
        board_.clear(m.row, m.col);
        state_ = GameState::IN_PROGRESS;
        winner_ = nullptr;
        turn_ = m.player->index;
    }
    GameState state() const { LockGuard g(*mu_); return state_; }
    const Player* winner() const { LockGuard g(*mu_); return winner_; }
    const Player& current() const { LockGuard g(*mu_); return players_[turn_]; }
    int movesPlayed() const { LockGuard g(*mu_); return (int)history_.size(); }
    std::string render() const { LockGuard g(*mu_); return board_.render(); }
private:
    Board board_;
    std::vector<Player> players_;
    std::unique_ptr<WinStrategy> rule_;
    std::vector<Move> history_;
    GameState state_ = GameState::IN_PROGRESS;
    const Player* winner_ = nullptr;
    int turn_ = 0;
};

}
