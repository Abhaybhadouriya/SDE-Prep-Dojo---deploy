/** Anything people can vote on (posts and comments). One vote per user; voting again replaces it. */
abstract class Votable {
    private final Map<String, Integer> votes = new ConcurrentHashMap<>();
    private final AtomicInteger up = new AtomicInteger();
    private final AtomicInteger down = new AtomicInteger();

    /** direction: +1 upvote, -1 downvote, 0 clear. Idempotent: the same vote twice changes nothing. */
    void vote(String userId, int direction) {
        Integer prev = direction == 0 ? votes.remove(userId) : votes.put(userId, direction);
        if (prev != null) count(prev, -1);
        if (direction != 0) count(direction, +1);
    }

    private void count(int direction, int delta) {
        (direction > 0 ? up : down).addAndGet(delta);
    }

    int upvotes() { return up.get(); }

    int downvotes() { return down.get(); }

    int score() { return up.get() - down.get(); }
}

/** A comment can hold replies: a small Composite, so the comment count is a recursive sum. */
final class Comment extends Votable {
    final String id;
    final String authorId;
    final String text;
    final long createdAt;
    final List<Comment> replies = new CopyOnWriteArrayList<>();

    Comment(String id, String authorId, String text, long createdAt) {
        this.id = id;
        this.authorId = authorId;
        this.text = text;
        this.createdAt = createdAt;
    }

    int threadSize() {
        int n = 1;
        for (Comment r : replies) n += r.threadSize();
        return n;
    }
}

final class Post extends Votable {
    final String id;
    final long seq;                 // global order; breaks timestamp ties
    final String authorId;
    final String text;
    final long createdAt;
    final List<Comment> comments = new CopyOnWriteArrayList<>();
    volatile boolean deleted;

    Post(String id, long seq, String authorId, String text, long createdAt) {
        this.id = id;
        this.seq = seq;
        this.authorId = authorId;
        this.text = text;
        this.createdAt = createdAt;
    }

    int commentCount() {
        int n = 0;
        for (Comment c : comments) n += c.threadSize();
        return n;
    }

    /** Newest first: time, then sequence. */
    static int newestFirst(Post a, Post b) {
        if (a.createdAt != b.createdAt) return Long.compare(b.createdAt, a.createdAt);
        return Long.compare(b.seq, a.seq);
    }
}

/** Who follows whom. Both directions are kept: following for pull, followers for push. */
final class FollowGraph {
    private final Map<String, Set<String>> following = new ConcurrentHashMap<>();
    private final Map<String, Set<String>> followers = new ConcurrentHashMap<>();

    void follow(String a, String b) {
        following.computeIfAbsent(a, k -> ConcurrentHashMap.newKeySet()).add(b);
        followers.computeIfAbsent(b, k -> ConcurrentHashMap.newKeySet()).add(a);
    }

    void unfollow(String a, String b) {
        following.getOrDefault(a, Set.of()).remove(b);
        followers.getOrDefault(b, Set.of()).remove(a);
    }

    Set<String> following(String a) { return Collections.unmodifiableSet(following.getOrDefault(a, Set.of())); }

    Set<String> followers(String b) { return Collections.unmodifiableSet(followers.getOrDefault(b, Set.of())); }

    boolean follows(String a, String b) { return following.getOrDefault(a, Set.of()).contains(b); }
}

/** Posts by id plus each author's timeline (append-only, oldest first). */
final class PostStore {
    private final Map<String, Post> byId = new ConcurrentHashMap<>();
    private final Map<String, List<Post>> timelines = new ConcurrentHashMap<>();
    private final List<Post> all = new ArrayList<>();

    void add(Post p) {
        byId.put(p.id, p);
        List<Post> t = timelines.computeIfAbsent(p.authorId, k -> new ArrayList<>());
        synchronized (t) { t.add(p); }
        synchronized (all) { all.add(p); }
    }

    Post get(String id) {
        Post p = byId.get(id);
        if (p == null || p.deleted) throw new FeedException("no such post: " + id);
        return p;
    }

    /** A copy, so readers never see a list being appended to. */
    List<Post> timeline(String authorId) {
        List<Post> t = timelines.get(authorId);
        if (t == null) return List.of();
        synchronized (t) { return new ArrayList<>(t); }
    }

    /** Most recent posts by anyone, newest first. */
    List<Post> recent(int limit) {
        List<Post> out = new ArrayList<>();
        synchronized (all) {
            for (int i = all.size() - 1; i >= 0 && out.size() < limit; i--)
                if (!all.get(i).deleted) out.add(all.get(i));
        }
        return out;
    }
}

/** Observer: told about every new post and follow. */
interface FeedListener {
    void onPost(Post p);

    default void onFollow(String follower, String followee) {}   // C: default method = an ops-table slot with a generic body
}

/** Strategy 1 - which posts are candidates for a user's feed (newest first, at most limit). */
interface FeedGenerator {
    List<Post> candidates(String userId, int limit);
}

/** Fan-out on read: merge the timelines of everyone I follow (and mine) with a heap. O(limit log k). */
final class PullFeedGenerator implements FeedGenerator {
    private final PostStore posts;
    private final FollowGraph graph;

    PullFeedGenerator(PostStore posts, FollowGraph graph) {
        this.posts = posts;
        this.graph = graph;
    }

    private record Cursor(List<Post> timeline, int index) {
        Post post() { return timeline.get(index); }
    }

    @Override public List<Post> candidates(String userId, int limit) {
        PriorityQueue<Cursor> heap = new PriorityQueue<>((a, b) -> Post.newestFirst(a.post(), b.post()));   // C: a lambda = a function pointer plus its captured context
        Set<String> sources = new HashSet<>(graph.following(userId));
        sources.add(userId);
        for (String author : sources) {
            List<Post> t = posts.timeline(author);
            if (!t.isEmpty()) heap.add(new Cursor(t, t.size() - 1));   // newest end of each timeline
        }
        List<Post> out = new ArrayList<>();
        while (!heap.isEmpty() && out.size() < limit) {
            Cursor c = heap.poll();
            if (!c.post().deleted) out.add(c.post());
            if (c.index() > 0) heap.add(new Cursor(c.timeline(), c.index() - 1));
        }
        return out;
    }
}

/** Fan-out on write: every new post is pushed into each follower's inbox. Reads are just a copy. */
final class PushFeedGenerator implements FeedGenerator, FeedListener {
    private final PostStore posts;
    private final FollowGraph graph;
    private final int inboxCap;
    private final Map<String, Deque<Post>> inbox = new ConcurrentHashMap<>();

    PushFeedGenerator(PostStore posts, FollowGraph graph, int inboxCap) {
        this.posts = posts;
        this.graph = graph;
        this.inboxCap = inboxCap;
    }

    private Deque<Post> box(String user) { return inbox.computeIfAbsent(user, k -> new ArrayDeque<>()); }   // C: look up k; if missing, allocate, insert and return it

    private void deliver(String user, Post p) {
        Deque<Post> b = box(user);
        synchronized (b) {
            b.addFirst(p);
            if (b.size() > inboxCap) b.removeLast();      // keep only the newest N, like a ring buffer
        }
    }

    @Override public void onPost(Post p) {
        deliver(p.authorId, p);
        for (String f : graph.followers(p.authorId)) deliver(f, p);
    }

    /** Backfill: a new follow brings the followee's recent posts into my inbox, in order. */
    @Override public void onFollow(String follower, String followee) {
        Deque<Post> b = box(follower);
        synchronized (b) {
            List<Post> merged = new ArrayList<>(b);
            List<Post> t = posts.timeline(followee);
            for (int i = Math.max(0, t.size() - inboxCap); i < t.size(); i++)
                if (!merged.contains(t.get(i))) merged.add(t.get(i));
            merged.sort(Post::newestFirst);   // C: Type::method = a pointer to an existing function
            b.clear();
            for (int i = 0; i < merged.size() && i < inboxCap; i++) b.addLast(merged.get(i));
        }
    }

    @Override public List<Post> candidates(String userId, int limit) {
        List<Post> out = new ArrayList<>();
        Deque<Post> b = box(userId);
        synchronized (b) {
            for (Post p : b) {
                if (out.size() >= limit) break;
                // lazy cleanup: skip deleted posts and authors I no longer follow
                if (!p.deleted && (p.authorId.equals(userId) || graph.follows(userId, p.authorId))) out.add(p);
            }
        }
        return out;
    }
}

/** Everyone's recent posts (the Flipkart console variant: followed users are only ranked first). */
final class GlobalFeedGenerator implements FeedGenerator {
    private final PostStore posts;

    GlobalFeedGenerator(PostStore posts) { this.posts = posts; }

    @Override public List<Post> candidates(String userId, int limit) { return posts.recent(limit); }
}

/** Strategy 2 - how a user's candidates are ordered. Each user may pick one. */
interface FeedRanker {
    Comparator<Post> order(String viewerId, FollowGraph graph);
}

enum Rankers implements FeedRanker {
    RECENT {
        public Comparator<Post> order(String v, FollowGraph g) { return Post::newestFirst; }
    },
    TOP {
        public Comparator<Post> order(String v, FollowGraph g) {
            return Comparator.comparingInt(Post::score).reversed().thenComparing(Post::newestFirst);   // C: builds the cmp() function you would pass to qsort
        }
    },
    /** Flipkart rule: followed authors first, then score, then comment count, then newest. */
    ENGAGEMENT {
        public Comparator<Post> order(String v, FollowGraph g) {
            Comparator<Post> followedFirst = Comparator.comparing(p -> !g.follows(v, p.authorId));
            return followedFirst
                    .thenComparing(Comparator.comparingInt(Post::score).reversed())
                    .thenComparing(Comparator.comparingInt(Post::commentCount).reversed())
                    .thenComparing(Post::newestFirst);
        }
    }
}

/** Facade: the one entry point. Owns the stores, publishes events, applies generator + ranker. */
final class SocialNetwork {
    static final int CANDIDATE_LIMIT = 500;

    private final Clock clock;
    private final Map<String, User> users = new ConcurrentHashMap<>();
    private final Map<String, Comment> comments = new ConcurrentHashMap<>();
    private final Map<String, FeedRanker> rankerOf = new ConcurrentHashMap<>();
    private final List<FeedListener> listeners = new CopyOnWriteArrayList<>();
    private final AtomicLong postSeq = new AtomicLong();
    private final AtomicLong commentSeq = new AtomicLong();
    final PostStore posts = new PostStore();
    final FollowGraph graph = new FollowGraph();
    private volatile FeedGenerator generator;
    private final FeedRanker defaultRanker;

    SocialNetwork(Clock clock, FeedRanker defaultRanker) {
        this.clock = clock;
        this.defaultRanker = defaultRanker;
        this.generator = new PullFeedGenerator(posts, graph);
    }

    void useGenerator(FeedGenerator g) {
        this.generator = g;
        if (g instanceof FeedListener l) listeners.add(l);
    }

    // ---------- users and follows ----------
    User signup(String id) {
        if (id == null || id.isBlank()) throw new FeedException("user id required");
        User u = new User(id);
        if (users.putIfAbsent(id, u) != null) throw new FeedException("user exists: " + id);
        return u;
    }

    private void mustExist(String id) {
        if (!users.containsKey(id)) throw new FeedException("no such user: " + id);
    }

    void follow(String a, String b) {
        mustExist(a);
        mustExist(b);
        if (a.equals(b)) throw new FeedException("cannot follow yourself");
        if (graph.follows(a, b)) return;                       // idempotent
        graph.follow(a, b);
        for (FeedListener l : listeners) l.onFollow(a, b);
    }

    void unfollow(String a, String b) {
        mustExist(a);
        graph.unfollow(a, b);
    }

    void setRanking(String userId, FeedRanker r) {
        mustExist(userId);
        rankerOf.put(userId, r);
    }

    // ---------- posts, comments, votes ----------
    Post post(String authorId, String text) {
        mustExist(authorId);
        if (text == null || text.isBlank()) throw new FeedException("empty post");
        long seq = postSeq.incrementAndGet();
        Post p = new Post(String.format("%03d", seq), seq, authorId, text, clock.now());
        posts.add(p);
        for (FeedListener l : listeners) l.onPost(p);         // Observer: push fan-out, notifications ...
        return p;
    }

    void deletePost(String userId, String postId) {
        Post p = posts.get(postId);
        if (!p.authorId.equals(userId)) throw new FeedException("only the author can delete " + postId);
        p.deleted = true;                                      // tombstone; feeds skip it
    }

    Comment comment(String userId, String postId, String text) {
        mustExist(userId);
        Comment c = newComment(userId, text);
        posts.get(postId).comments.add(c);
        return c;
    }

    Comment reply(String userId, String commentId, String text) {
        mustExist(userId);
        Comment parent = comments.get(commentId);
        if (parent == null) throw new FeedException("no such comment: " + commentId);
        Comment c = newComment(userId, text);
        parent.replies.add(c);
        return c;
    }

    private Comment newComment(String userId, String text) {
        if (text == null || text.isBlank()) throw new FeedException("empty comment");
        Comment c = new Comment("c" + commentSeq.incrementAndGet(), userId, text, clock.now());
        comments.put(c.id, c);
        return c;
    }

    void votePost(String userId, String postId, int direction) {
        mustExist(userId);
        posts.get(postId).vote(userId, direction);
    }

    void voteComment(String userId, String commentId, int direction) {
        mustExist(userId);
        Comment c = comments.get(commentId);
        if (c == null) throw new FeedException("no such comment: " + commentId);
        c.vote(userId, direction);
    }

    // ---------- the feed ----------
    /** cursor = null for page 1, else the nextCursor of the previous page (an offset here). */
    FeedPage feed(String userId, int pageSize, String cursor) {
        mustExist(userId);
        if (pageSize <= 0) throw new FeedException("page size must be positive");
        int from;
        try {
            from = cursor == null ? 0 : Integer.parseInt(cursor);
        } catch (NumberFormatException e) {
            throw new FeedException("bad cursor: " + cursor);
        }
        List<Post> ranked = new ArrayList<>(generator.candidates(userId, CANDIDATE_LIMIT));
        ranked.sort(rankerOf.getOrDefault(userId, defaultRanker).order(userId, graph));
        int to = Math.min(ranked.size(), from + pageSize);
        List<Post> page = from >= ranked.size() ? List.of() : List.copyOf(ranked.subList(from, to));
        return new FeedPage(page, to < ranked.size() ? String.valueOf(to) : null);
    }

    long now() { return clock.now(); }

    /** "2m ago" style, the Flipkart bonus. */
    static String ago(long now, long then) {
        long s = Math.max(0, (now - then) / 1000);
        if (s < 60) return "just now";
        if (s < 3600) return (s / 60) + "m ago";
        if (s < 86400) return (s / 3600) + " hr ago";
        return (s / 86400) + "d ago";
    }
}

final class Console {
    private final SocialNetwork net;
    private String current;
    private final Map<String, BiFunction<Console, String[], String>> commands = Map.of(   // C: immutable literal collection
            "signup", (c, a) -> { c.net.signup(a[1]); return "signed up " + a[1]; },
            "login", (c, a) -> { c.login(a[1]); return c.show(); },
            "post", (c, a) -> "posted " + c.net.post(c.me(), a[1]).id,
            "follow", (c, a) -> { c.net.follow(c.me(), a[1]); return "following " + a[1]; },
            "unfollow", (c, a) -> { c.net.unfollow(c.me(), a[1]); return "unfollowed " + a[1]; },
            "reply", (c, a) -> "comment " + c.net.comment(c.me(), a[1], a[2]).id,
            "upvote", (c, a) -> { c.net.votePost(c.me(), a[1], +1); return "upvoted " + a[1]; },
            "downvote", (c, a) -> { c.net.votePost(c.me(), a[1], -1); return "downvoted " + a[1]; },
            "shownewsfeed", (c, a) -> c.show(),
            "ranking", (c, a) -> { c.net.setRanking(c.me(), Rankers.valueOf(a[1].toUpperCase())); return "ranking " + a[1]; });

    Console(SocialNetwork net) { this.net = net; }

    private void login(String user) {
        net.feed(user, 1, null);                         // validates the user exists
        current = user;
    }

    private String me() {
        if (current == null) throw new FeedException("login first");
        return current;
    }

    private String show() {
        StringBuilder sb = new StringBuilder();
        List<Post> items = net.feed(me(), 10, null).items();
        if (items.isEmpty()) return "(feed is empty)";
        for (Post p : items) {
            sb.append(String.format("id: %s (%d up, %d down, %d comments) %s: %s [%s]%n", p.id, p.upvotes(),
                    p.downvotes(), p.commentCount(), p.authorId, p.text, SocialNetwork.ago(net.now(), p.createdAt)));
        }
        return sb.toString().stripTrailing();
    }

    String run(String line) {
        String[] args = line.trim().split("~");
        var handler = commands.get(args[0].toLowerCase());   // C: var = type inferred by the compiler (__auto_type)
        if (handler == null) return "unknown command: " + args[0];
        try {
            return handler.apply(this, args);
        } catch (FeedException e) {
            return "error: " + e.getMessage();
        } catch (ArrayIndexOutOfBoundsException | IllegalArgumentException e) {
            return "error: bad arguments for " + args[0];
        }
    }
}

"""Social network and news feed: follows, posts, votes, comments, pull vs push Strategy, ranking, pagination."""
from __future__ import annotations

import heapq
import itertools
import threading
from collections import deque
from dataclasses import dataclass, field
from typing import Callable, Deque, Dict, List, Optional, Protocol, Set, Tuple


class FeedError(Exception):
    pass


class Votable:
    """One vote per user; voting again replaces the old vote. direction: +1, -1 or 0 (clear)."""

    def __init__(self) -> None:
        self._votes: Dict[str, int] = {}
        self._lock = threading.Lock()
        self.up = 0
        self.down = 0

    def vote(self, user_id: str, direction: int) -> None:
        with self._lock:
            prev = self._votes.pop(user_id, None)
            if prev is not None:
                self._count(prev, -1)
            if direction != 0:
                self._votes[user_id] = direction
                self._count(direction, +1)

    def _count(self, direction: int, delta: int) -> None:
        if direction > 0:
            self.up += delta
        else:
            self.down += delta

    @property
    def score(self) -> int:
        return self.up - self.down


class Comment(Votable):
    def __init__(self, cid: str, author: str, text: str, created_at: int) -> None:
        super().__init__()
        self.id, self.author, self.text, self.created_at = cid, author, text, created_at
        self.replies: List[Comment] = []

    def thread_size(self) -> int:
        return 1 + sum(r.thread_size() for r in self.replies)


class Post(Votable):
    def __init__(self, pid: str, seq: int, author: str, text: str, created_at: int) -> None:
        super().__init__()
        self.id, self.seq, self.author, self.text, self.created_at = pid, seq, author, text, created_at
        self.comments: List[Comment] = []
        self.deleted = False

    def comment_count(self) -> int:
        return sum(c.thread_size() for c in self.comments)

    def newest_key(self) -> Tuple[int, int]:
        """Sort ascending by this key = newest first."""
        return (-self.created_at, -self.seq)


@dataclass
class FeedPage:
    items: List[Post]
    next_cursor: Optional[str]


class ManualClock:
    def __init__(self, start: int = 0) -> None:
        self.millis = start

    def now(self) -> int:
        return self.millis

    def advance_minutes(self, m: int) -> None:
        self.millis += m * 60_000


# ---------- stores ----------
class FollowGraph:
    def __init__(self) -> None:
        self._following: Dict[str, Set[str]] = {}
        self._followers: Dict[str, Set[str]] = {}
        self._lock = threading.Lock()

    def follow(self, a: str, b: str) -> None:
        with self._lock:
            self._following.setdefault(a, set()).add(b)
            self._followers.setdefault(b, set()).add(a)

    def unfollow(self, a: str, b: str) -> None:
        with self._lock:
            self._following.get(a, set()).discard(b)
            self._followers.get(b, set()).discard(a)

    def following(self, a: str) -> Set[str]:
        with self._lock:
            return set(self._following.get(a, ()))

    def followers(self, b: str) -> Set[str]:
        with self._lock:
            return set(self._followers.get(b, ()))

    def follows(self, a: str, b: str) -> bool:
        with self._lock:
            return b in self._following.get(a, ())


class PostStore:
    def __init__(self) -> None:
        self._by_id: Dict[str, Post] = {}
        self._timelines: Dict[str, List[Post]] = {}
        self._all: List[Post] = []
        self._lock = threading.Lock()

    def add(self, p: Post) -> None:
        with self._lock:
            self._by_id[p.id] = p
            self._timelines.setdefault(p.author, []).append(p)
            self._all.append(p)

    def get(self, pid: str) -> Post:
        p = self._by_id.get(pid)
        if p is None or p.deleted:
            raise FeedError("no such post: " + pid)
        return p

    def timeline(self, author: str) -> List[Post]:
        with self._lock:
            return list(self._timelines.get(author, ()))

    def recent(self, limit: int) -> List[Post]:
        with self._lock:
            return [p for p in reversed(self._all) if not p.deleted][:limit]


# ---------- Strategy 1: candidate generation ----------
class FeedGenerator(Protocol):
    def candidates(self, user_id: str, limit: int) -> List[Post]: ...


class PullFeedGenerator:
    """Fan-out on read: k-way merge of followees' timelines with a heap."""

    def __init__(self, posts: PostStore, graph: FollowGraph) -> None:
        self.posts, self.graph = posts, graph

    def candidates(self, user_id: str, limit: int) -> List[Post]:
        heap: List[Tuple[Tuple[int, int], int, List[Post]]] = []
        for author in self.graph.following(user_id) | {user_id}:
            t = self.posts.timeline(author)
            if t:
                heapq.heappush(heap, (t[-1].newest_key(), len(t) - 1, t))
        out: List[Post] = []
        while heap and len(out) < limit:
            _, i, t = heapq.heappop(heap)
            if not t[i].deleted:
                out.append(t[i])
            if i > 0:
                heapq.heappush(heap, (t[i - 1].newest_key(), i - 1, t))
        return out


class PushFeedGenerator:
    """Fan-out on write: deliver each new post into followers' inboxes (Observer)."""

    def __init__(self, posts: PostStore, graph: FollowGraph, inbox_cap: int) -> None:
        self.posts, self.graph, self.cap = posts, graph, inbox_cap
        self._inbox: Dict[str, Deque[Post]] = {}
        self._lock = threading.Lock()

    def _deliver(self, user: str, p: Post) -> None:
        box = self._inbox.setdefault(user, deque())
        box.appendleft(p)
        if len(box) > self.cap:
            box.pop()

    def on_post(self, p: Post) -> None:
        with self._lock:
            self._deliver(p.author, p)
            for f in self.graph.followers(p.author):
                self._deliver(f, p)

    def on_follow(self, follower: str, followee: str) -> None:
        with self._lock:
            box = self._inbox.setdefault(follower, deque())
            merged = {id(p): p for p in list(box) + self.posts.timeline(followee)[-self.cap:]}
            ordered = sorted(merged.values(), key=Post.newest_key)[: self.cap]
            self._inbox[follower] = deque(ordered)

    def candidates(self, user_id: str, limit: int) -> List[Post]:
        with self._lock:
            box = list(self._inbox.get(user_id, ()))
        out = [p for p in box
               if not p.deleted and (p.author == user_id or self.graph.follows(user_id, p.author))]
        return out[:limit]


class GlobalFeedGenerator:
    def __init__(self, posts: PostStore) -> None:
        self.posts = posts

    def candidates(self, user_id: str, limit: int) -> List[Post]:
        return self.posts.recent(limit)


# ---------- Strategy 2: ranking (a sort key per viewer) ----------
Ranker = Callable[[str, FollowGraph], Callable[[Post], tuple]]


def recent(viewer: str, g: FollowGraph) -> Callable[[Post], tuple]:
    return lambda p: p.newest_key()


def top(viewer: str, g: FollowGraph) -> Callable[[Post], tuple]:
    return lambda p: (-p.score,) + p.newest_key()


def engagement(viewer: str, g: FollowGraph) -> Callable[[Post], tuple]:
    """Flipkart rule: followed authors first, then score, then comments, then newest."""
    return lambda p: (not g.follows(viewer, p.author), -p.score, -p.comment_count()) + p.newest_key()


RANKERS: Dict[str, Ranker] = {"recent": recent, "top": top, "engagement": engagement}


# ---------- facade ----------
class SocialNetwork:
    CANDIDATE_LIMIT = 500

    def __init__(self, clock: ManualClock, default_ranker: Ranker = recent) -> None:
        self.clock = clock
        self.default_ranker = default_ranker
        self.posts = PostStore()
        self.graph = FollowGraph()
        self.generator: FeedGenerator = PullFeedGenerator(self.posts, self.graph)
        self.listeners: list = []
        self._users: Set[str] = set()
        self._comments: Dict[str, Comment] = {}
        self._ranker_of: Dict[str, Ranker] = {}
        self._post_seq = itertools.count(1)
        self._comment_seq = itertools.count(1)
        self._lock = threading.Lock()

    def use_generator(self, g) -> None:
        self.generator = g
        if hasattr(g, "on_post"):
            self.listeners.append(g)

    def signup(self, uid: str) -> None:
        if not uid:
            raise FeedError("user id required")
        with self._lock:
            if uid in self._users:
                raise FeedError("user exists: " + uid)
            self._users.add(uid)

    def _must_exist(self, uid: str) -> None:
        if uid not in self._users:
            raise FeedError("no such user: " + uid)

    def follow(self, a: str, b: str) -> None:
        self._must_exist(a)
        self._must_exist(b)
        if a == b:
            raise FeedError("cannot follow yourself")
        if self.graph.follows(a, b):
            return
        self.graph.follow(a, b)
        for l in self.listeners:
            l.on_follow(a, b)

    def unfollow(self, a: str, b: str) -> None:
        self._must_exist(a)
        self.graph.unfollow(a, b)

    def set_ranking(self, uid: str, ranker: Ranker) -> None:
        self._must_exist(uid)
        self._ranker_of[uid] = ranker

    def post(self, author: str, text: str) -> Post:
        self._must_exist(author)
        if not text or not text.strip():
            raise FeedError("empty post")
        with self._lock:
            seq = next(self._post_seq)
        p = Post(f"{seq:03d}", seq, author, text, self.clock.now())
        self.posts.add(p)
        for l in self.listeners:
            l.on_post(p)
        return p

    def delete_post(self, uid: str, pid: str) -> None:
        p = self.posts.get(pid)
        if p.author != uid:
            raise FeedError("only the author can delete " + pid)
        p.deleted = True

    def _new_comment(self, uid: str, text: str) -> Comment:
        if not text or not text.strip():
            raise FeedError("empty comment")
        with self._lock:
            c = Comment(f"c{next(self._comment_seq)}", uid, text, self.clock.now())
            self._comments[c.id] = c
        return c

    def comment(self, uid: str, pid: str, text: str) -> Comment:
        self._must_exist(uid)
        post = self.posts.get(pid)
        c = self._new_comment(uid, text)
        post.comments.append(c)
        return c

    def reply(self, uid: str, cid: str, text: str) -> Comment:
        self._must_exist(uid)
        parent = self._comments.get(cid)
        if parent is None:
            raise FeedError("no such comment: " + cid)
        c = self._new_comment(uid, text)
        parent.replies.append(c)
        return c

    def vote_post(self, uid: str, pid: str, direction: int) -> None:
        self._must_exist(uid)
        self.posts.get(pid).vote(uid, direction)

    def feed(self, uid: str, page_size: int, cursor: Optional[str] = None) -> FeedPage:
        self._must_exist(uid)
        if page_size <= 0:
            raise FeedError("page size must be positive")
        try:
            start = int(cursor) if cursor is not None else 0
        except ValueError:
            raise FeedError("bad cursor: " + str(cursor)) from None
        ranked = self.generator.candidates(uid, self.CANDIDATE_LIMIT)
        ranked.sort(key=self._ranker_of.get(uid, self.default_ranker)(uid, self.graph))
        end = min(len(ranked), start + page_size)
        return FeedPage(ranked[start:end], str(end) if end < len(ranked) else None)


def ago(now: int, then: int) -> str:
    s = max(0, (now - then) // 1000)
    if s < 60:
        return "just now"
    if s < 3600:
        return f"{s // 60}m ago"
    if s < 86400:
        return f"{s // 3600} hr ago"
    return f"{s // 86400}d ago"


class Console:
    """Flipkart-style 'command~arg~arg' runner. Command pattern: a dict of handlers."""

    def __init__(self, net: SocialNetwork) -> None:
        self.net = net
        self.current: Optional[str] = None
        self.commands: Dict[str, Callable[[List[str]], str]] = {
            "signup": lambda a: (net.signup(a[1]), "signed up " + a[1])[1],
            "login": self._login,
            "post": lambda a: "posted " + net.post(self._me(), a[1]).id,
            "follow": lambda a: (net.follow(self._me(), a[1]), "following " + a[1])[1],
            "unfollow": lambda a: (net.unfollow(self._me(), a[1]), "unfollowed " + a[1])[1],
            "reply": lambda a: "comment " + net.comment(self._me(), a[1], a[2]).id,
            "upvote": lambda a: (net.vote_post(self._me(), a[1], +1), "upvoted " + a[1])[1],
            "downvote": lambda a: (net.vote_post(self._me(), a[1], -1), "downvoted " + a[1])[1],
            "shownewsfeed": lambda a: self._show(),
            "ranking": lambda a: (net.set_ranking(self._me(), RANKERS[a[1].lower()]), "ranking " + a[1])[1],
        }

    def _me(self) -> str:
        if self.current is None:
            raise FeedError("login first")
        return self.current

    def _login(self, a: List[str]) -> str:
        self.net.feed(a[1], 1)
        self.current = a[1]
        return self._show()

    def _show(self) -> str:
        items = self.net.feed(self._me(), 10).items
        if not items:
            return "(feed is empty)"
        now = self.net.clock.now()
        return "\n".join(
            f"id: {p.id} ({p.up} up, {p.down} down, {p.comment_count()} comments) "
            f"{p.author}: {p.text} [{ago(now, p.created_at)}]" for p in items)

    def run(self, line: str) -> str:
        args = line.strip().split("~")
        handler = self.commands.get(args[0].lower())
        if handler is None:
            return "unknown command: " + args[0]
        try:
            return handler(args)
        except FeedError as e:
            return "error: " + str(e)
        except (IndexError, KeyError):
            return "error: bad arguments for " + args[0]

"""Run: python -m unittest -v test_feed (or pytest)."""
import threading
import unittest

from feed import (Console, FeedError, GlobalFeedGenerator, ManualClock, PushFeedGenerator, SocialNetwork,
                  ago, engagement, recent)


def build(push: bool = False) -> SocialNetwork:
    clock = ManualClock()
    net = SocialNetwork(clock, recent)
    if push:
        net.use_generator(PushFeedGenerator(net.posts, net.graph, 3))
    for u in ("a", "b", "c"):
        net.signup(u)
    return net


class FeedTest(unittest.TestCase):
    def test_pull_merges_followees_newest_first(self) -> None:
        net = build()
        net.follow("a", "b")
        for i, who in enumerate(["b", "c", "a", "b"]):
            net.clock.advance_minutes(1)
            net.post(who, f"p{i}")
        self.assertEqual([p.text for p in net.feed("a", 10).items], ["p3", "p2", "p0"])

    def test_push_inbox_cap_and_lazy_unfollow(self) -> None:
        net = build(push=True)
        net.follow("a", "b")
        for i in range(5):
            net.clock.advance_minutes(1)
            net.post("b", f"p{i}")
        self.assertEqual([p.text for p in net.feed("a", 10).items], ["p4", "p3", "p2"])   # cap 3
        net.unfollow("a", "b")
        self.assertEqual(net.feed("a", 10).items, [])

    def test_same_timestamp_uses_sequence(self) -> None:
        net = build()
        net.post("a", "first")
        net.post("a", "second")
        self.assertEqual([p.text for p in net.feed("a", 10).items], ["second", "first"])

    def test_votes_are_one_per_user(self) -> None:
        net = build()
        p = net.post("a", "x")
        net.vote_post("b", p.id, +1)
        net.vote_post("b", p.id, +1)
        net.vote_post("c", p.id, -1)
        net.vote_post("c", p.id, +1)
        self.assertEqual((p.up, p.down, p.score), (2, 0, 2))
        net.vote_post("c", p.id, 0)
        self.assertEqual(p.score, 1)

    def test_engagement_ranking(self) -> None:
        net = build()
        net.set_ranking("a", engagement)
        net.use_generator(GlobalFeedGenerator(net.posts))
        net.follow("a", "b")
        hot = net.post("c", "hot")              # not followed, but popular
        net.vote_post("b", hot.id, +1)
        net.clock.advance_minutes(1)
        quiet = net.post("b", "quiet")
        net.comment("c", quiet.id, "hi")
        self.assertEqual([p.text for p in net.feed("a", 10).items], ["quiet", "hot"])

    def test_comment_thread_counts(self) -> None:
        net = build()
        p = net.post("a", "x")
        c = net.comment("b", p.id, "one")
        net.reply("c", c.id, "two")
        net.reply("a", c.id, "three")
        self.assertEqual(p.comment_count(), 3)

    def test_pagination_walks_everything_once(self) -> None:
        net = build()
        for i in range(7):
            net.post("a", f"p{i}")
        seen, cursor = [], None
        while True:
            page = net.feed("a", 3, cursor)
            seen += [p.text for p in page.items]
            if page.next_cursor is None:
                break
            cursor = page.next_cursor
        self.assertEqual(seen, [f"p{i}" for i in range(6, -1, -1)])

    def test_errors(self) -> None:
        net = build()
        p = net.post("a", "x")
        for fn in (lambda: net.follow("a", "a"), lambda: net.post("zz", "x"), lambda: net.signup("a"),
                   lambda: net.delete_post("b", p.id), lambda: net.feed("a", 0), lambda: net.feed("a", 5, "q"),
                   lambda: net.post("a", "  ")):
            with self.assertRaises(FeedError):
                fn()
        net.delete_post("a", p.id)
        with self.assertRaises(FeedError):
            net.vote_post("b", p.id, 1)

    def test_ago(self) -> None:
        self.assertEqual(ago(130_000, 10_000), "2m ago")
        self.assertEqual(ago(7_200_000, 0), "2 hr ago")

    def test_console(self) -> None:
        net = build()
        con = Console(net)
        self.assertEqual(con.run("post~hi"), "error: login first")
        con.run("login~a")
        self.assertEqual(con.run("post~hi"), "posted 001")
        self.assertTrue(con.run("shownewsfeed").startswith("id: 001"))
        self.assertEqual(con.run("nope"), "unknown command: nope")
        self.assertEqual(con.run("follow"), "error: bad arguments for follow")

    def test_concurrent_votes_and_fanout(self) -> None:
        net = build(push=False)
        push = PushFeedGenerator(net.posts, net.graph, 10_000)
        net.use_generator(push)
        for i in range(200):
            net.signup(f"u{i}")
            net.follow(f"u{i}", "a")
        p = net.post("a", "hello")

        def worker(k: int) -> None:
            for i in range(k, 200, 4):
                net.vote_post(f"u{i}", p.id, -1)
                net.vote_post(f"u{i}", p.id, +1)
            for i in range(25):
                net.post("a", f"t{k}-{i}")

        threads = [threading.Thread(target=worker, args=(k,)) for k in range(4)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertEqual((p.up, p.down), (200, 0))
        self.assertEqual(len(net.feed("u7", 10_000).items), 101)
