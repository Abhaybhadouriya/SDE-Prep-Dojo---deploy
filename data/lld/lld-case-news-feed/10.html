The hybrid generator, the follow-up that comes up most once scale enters the conversation:

final class HybridFeedGenerator implements FeedGenerator {
    private final PushFeedGenerator push;      // registered as a listener, but skips celebrities
    private final PostStore posts;
    private final FollowGraph graph;
    private final int celebrityThreshold;

    public List<Post> candidates(String user, int limit) {
        List<Post> out = new ArrayList<>(push.candidates(user, limit));       // normal authors
        for (String a : graph.following(user))
            if (graph.followers(a).size() >= celebrityThreshold) {            // pull the big ones
                List<Post> t = posts.timeline(a);
                out.addAll(t.subList(Math.max(0, t.size() - limit), t.size()));
            }
        out.sort(Post::newestFirst);
        return out.subList(0, Math.min(limit, out.size()));
    }
}
