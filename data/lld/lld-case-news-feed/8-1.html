Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was C++ demo OK.

// News feed core in C++14: follow graph, posts with one-vote-per-user, pull vs push generators
// (Strategy), push fan-out as an Observer, per-user ranking, offset-cursor pagination.
// Single-threaded demo: this MinGW g++ 6.3 build (win32 thread model) has no std::thread.
// In a threaded build, guard each store with a mutex and each inbox with its own mutex.
#include <algorithm>
#include <cassert>
#include <cstdio>
#include <deque>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <queue>
#include <set>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>
#include <mutex>
// Locking: one facade lock, a real std::recursive_mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). Recursive because public methods call each other,
// just as Java's synchronized is re-entrant. MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and
// -pthread fails to link there, so the default build uses a no-op Mutex with the same lock()/unlock():
// the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::recursive_mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

using namespace std;

struct FeedError : runtime_error {
    explicit FeedError(const string& m) : runtime_error(m) {}
};

struct Post {
    string id, author, text;
    long seq, createdAt;
    bool deleted = false;
    map<string, int> votes;                      // user -> +1 / -1
    int up = 0, down = 0;
    Post(string i, long s, string a, string t, long c)
        : id(move(i)), author(move(a)), text(move(t)), seq(s), createdAt(c) {}
    void vote(const string& user, int dir) {
        auto it = votes.find(user);
        if (it != votes.end()) { (it->second > 0 ? up : down)--; votes.erase(it); }
        if (dir != 0) { votes[user] = dir; (dir > 0 ? up : down)++; }
    }
    int score() const { return up - down; }
};
using PostPtr = shared_ptr<Post>;

bool newerThan(const PostPtr& a, const PostPtr& b) {          // newest first
    if (a->createdAt != b->createdAt) return a->createdAt > b->createdAt;
    return a->seq > b->seq;
}

class FollowGraph {
    map<string, set<string>> following_, followers_;
public:
    void follow(const string& a, const string& b) { following_[a].insert(b); followers_[b].insert(a); }
    void unfollow(const string& a, const string& b) { following_[a].erase(b); followers_[b].erase(a); }
    set<string> following(const string& a) const {
        auto it = following_.find(a);
        return it == following_.end() ? set<string>() : it->second;
    }
    set<string> followers(const string& b) const {
        auto it = followers_.find(b);
        return it == followers_.end() ? set<string>() : it->second;
    }
    bool follows(const string& a, const string& b) const {
        auto it = following_.find(a);
        return it != following_.end() && it->second.count(b) > 0;
    }
};

class PostStore {
    map<string, PostPtr> byId_;
    map<string, vector<PostPtr>> timelines_;     // per author, oldest first
public:
    void add(const PostPtr& p) { byId_[p->id] = p; timelines_[p->author].push_back(p); }
    PostPtr get(const string& id) const {
        auto it = byId_.find(id);
        if (it == byId_.end() || it->second->deleted) throw FeedError("no such post: " + id);
        return it->second;
    }
    const vector<PostPtr>& timeline(const string& a) const {
        static const vector<PostPtr> empty;
        auto it = timelines_.find(a);
        return it == timelines_.end() ? empty : it->second;
    }
};

// ---------- Strategy 1: candidates; Observer hooks ----------
struct FeedGenerator {
    virtual ~FeedGenerator() {}
    virtual vector<PostPtr> candidates(const string& user, size_t limit) = 0;
    virtual void onPost(const PostPtr&) {}
    virtual void onFollow(const string&, const string&) {}
};

class PullGenerator : public FeedGenerator {
    const PostStore& posts_;
    const FollowGraph& graph_;
public:
    PullGenerator(const PostStore& p, const FollowGraph& g) : posts_(p), graph_(g) {}
    vector<PostPtr> candidates(const string& user, size_t limit) override {
        typedef pair<const vector<PostPtr>*, size_t> Cur;           // timeline, index
        auto later = [](const Cur& a, const Cur& b) { return newerThan((*b.first)[b.second], (*a.first)[a.second]); };
        priority_queue<Cur, vector<Cur>, decltype(later)> heap(later);
        set<string> sources = graph_.following(user);
        sources.insert(user);
        for (const string& a : sources) {
            const auto& t = posts_.timeline(a);
            if (!t.empty()) heap.push(Cur(&t, t.size() - 1));
        }
        vector<PostPtr> out;
        while (!heap.empty() && out.size() < limit) {
            Cur c = heap.top();
            heap.pop();
            const PostPtr& p = (*c.first)[c.second];
            if (!p->deleted) out.push_back(p);
            if (c.second > 0) heap.push(Cur(c.first, c.second - 1));
        }
        return out;
    }
};

class PushGenerator : public FeedGenerator {
    const PostStore& posts_;
    const FollowGraph& graph_;
    size_t cap_;
    map<string, deque<PostPtr>> inbox_;
    void deliver(const string& u, const PostPtr& p) {
        auto& b = inbox_[u];
        b.push_front(p);
        if (b.size() > cap_) b.pop_back();
    }
public:
    PushGenerator(const PostStore& p, const FollowGraph& g, size_t cap) : posts_(p), graph_(g), cap_(cap) {}
    void onPost(const PostPtr& p) override {
        deliver(p->author, p);
        for (const string& f : graph_.followers(p->author)) deliver(f, p);
    }
    void onFollow(const string& follower, const string& followee) override {
        auto& b = inbox_[follower];
        vector<PostPtr> merged(b.begin(), b.end());
        const auto& t = posts_.timeline(followee);
        for (size_t i = t.size() > cap_ ? t.size() - cap_ : 0; i < t.size(); ++i)
            if (find(merged.begin(), merged.end(), t[i]) == merged.end()) merged.push_back(t[i]);
        sort(merged.begin(), merged.end(), newerThan);
        if (merged.size() > cap_) merged.resize(cap_);
        b.assign(merged.begin(), merged.end());
    }
    vector<PostPtr> candidates(const string& user, size_t limit) override {
        vector<PostPtr> out;
        for (const PostPtr& p : inbox_[user]) {
            if (out.size() >= limit) break;
            if (!p->deleted && (p->author == user || graph_.follows(user, p->author))) out.push_back(p);
        }
        return out;
    }
};

// ---------- Strategy 2: ranking, chosen per user ----------
typedef function<bool(const PostPtr&, const PostPtr&)> Order;
typedef function<Order(const string& viewer, const FollowGraph&)> Ranker;

Ranker recentRanker = [](const string&, const FollowGraph&) { return Order(newerThan); };
Ranker topRanker = [](const string&, const FollowGraph&) {
    return Order([](const PostPtr& a, const PostPtr& b) {
        if (a->score() != b->score()) return a->score() > b->score();
        return newerThan(a, b);
    });
};

struct FeedPage {
    vector<PostPtr> items;
    string nextCursor;                            // empty = no more pages
};

class SocialNetwork {
    mutable Mutex mu_;   // the facade lock: every public call holds it
    set<string> users_;
    map<string, Ranker> rankerOf_;
    long seq_ = 0;
    long now_ = 0;
    unique_ptr<FeedGenerator> gen_;
    void mustExist(const string& u) const {
        if (!users_.count(u)) throw FeedError("no such user: " + u);
    }
public:
    PostStore posts;
    FollowGraph graph;
    explicit SocialNetwork(bool push) {
        if (push) gen_.reset(new PushGenerator(posts, graph, 100));
        else gen_.reset(new PullGenerator(posts, graph));
    }
    void tick() { LockGuard g(mu_); now_ += 60000; }
    void signup(const string& u) { LockGuard g(mu_);
        if (!users_.insert(u).second) throw FeedError("user exists: " + u);
    }
    void follow(const string& a, const string& b) { LockGuard g(mu_);
        mustExist(a);
        mustExist(b);
        if (a == b) throw FeedError("cannot follow yourself");
        if (graph.follows(a, b)) return;
        graph.follow(a, b);
        gen_->onFollow(a, b);
    }
    void unfollow(const string& a, const string& b) { LockGuard g(mu_); graph.unfollow(a, b); }
    void setRanking(const string& u, Ranker r) { LockGuard g(mu_); mustExist(u); rankerOf_[u] = r; }
    PostPtr post(const string& a, const string& text) { LockGuard g(mu_);
        mustExist(a);
        ++seq_;
        char id[16];
        snprintf(id, sizeof id, "%03ld", seq_);
        auto p = make_shared<Post>(id, seq_, a, text, now_);
        posts.add(p);
        gen_->onPost(p);                          // Observer hook: push fan-out
        return p;
    }
    void vote(const string& u, const string& pid, int dir) { LockGuard g(mu_); mustExist(u); posts.get(pid)->vote(u, dir); }
    void deletePost(const string& u, const string& pid) { LockGuard g(mu_);
        PostPtr p = posts.get(pid);
        if (p->author != u) throw FeedError("only the author can delete " + pid);
        p->deleted = true;
    }
    FeedPage feed(const string& u, size_t pageSize, const string& cursor) { LockGuard g(mu_);
        mustExist(u);
        size_t from = cursor.empty() ? 0 : (size_t)stoul(cursor);
        vector<PostPtr> ranked = gen_->candidates(u, 500);
        auto it = rankerOf_.find(u);
        Ranker r = it == rankerOf_.end() ? recentRanker : it->second;
        stable_sort(ranked.begin(), ranked.end(), r(u, graph));
        FeedPage page;
        size_t to = min(ranked.size(), from + pageSize);
        for (size_t i = from; i < to; ++i) page.items.push_back(ranked[i]);
        if (to < ranked.size()) page.nextCursor = to_string(to);
        return page;
    }
};

string ids(const FeedPage& p) {
    string s;
    for (size_t i = 0; i < p.items.size(); ++i) s += (i ? "," : "") + p.items[i]->id;
    return s;
}

int main() {
    SocialNetwork pull(false), push(true);
    const char* names[] = {"ana", "bo", "cy", "dee"};
    for (SocialNetwork* n : {&pull, &push}) {
        for (const char* u : names) n->signup(u);
        n->follow("ana", "bo");
        n->follow("ana", "cy");
    }
    const char* authors[] = {"bo", "cy", "dee", "ana"};
    for (int i = 0; i < 12; ++i) {
        for (SocialNetwork* n : {&pull, &push}) { n->tick(); n->post(authors[i % 4], "msg " + to_string(i)); }
    }
    string a = ids(pull.feed("ana", 50, "")), b = ids(push.feed("ana", 50, ""));
    cout << "pull: " << a << "\npush: " << b << "\n";
    assert(a == b);

    FeedPage p1 = pull.feed("ana", 4, "");
    FeedPage p2 = pull.feed("ana", 4, p1.nextCursor);
    cout << "page1 " << ids(p1) << " next=" << p1.nextCursor << "\npage2 " << ids(p2) << " next=" << p2.nextCursor << "\n";

    pull.vote("bo", "005", +1);
    pull.vote("cy", "005", +1);
    pull.vote("cy", "005", +1);                  // same vote twice: still one
    assert(pull.posts.get("005")->score() == 2);
    pull.setRanking("ana", topRanker);
    cout << "ana with TOP: " << ids(pull.feed("ana", 3, "")) << "\n";

    for (SocialNetwork* n : {&pull, &push}) { n->unfollow("ana", "cy"); n->deletePost("bo", "009"); }
    pull.setRanking("ana", recentRanker);
    a = ids(pull.feed("ana", 50, ""));
    b = ids(push.feed("ana", 50, ""));
    cout << "after unfollow cy + delete 009: pull " << a << " | push " << b << "\n";
    assert(a == b && a.find("009") == string::npos);
    push.follow("ana", "dee");
    cout << "push after follow dee (backfill): " << ids(push.feed("ana", 5, "")) << "\n";

    try { pull.follow("ana", "ana"); assert(false); } catch (const FeedError& e) { cout << "error ok: " << e.what() << "\n"; }
    try { pull.deletePost("ana", "001"); assert(false); } catch (const FeedError& e) { cout << "error ok: " << e.what() << "\n"; }
    cout << "C++ demo OK\n";
    return 0;
}
