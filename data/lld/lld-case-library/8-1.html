Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was All 4 checks passed.

// Library management - C++14 version.
// Book vs BookCopy, copy state table, fine Strategy, availability Observer, FIFO reservations.
// The MinGW 6.3 toolchain here has no std::thread / std::mutex, so this version is single-threaded;
// the Java and Python tabs show the locking.
#include <algorithm>
#include <cctype>
#include <cstdio>
#include <cstdlib>
#include <deque>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
#include <mutex>
// Locking: one facade lock, a real std::recursive_mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). Recursive because public methods call each other,
// just as Java's synchronized is re-entrant. MinGW.org g++ 6.3 (win32 thread model) has no std::mutex and
// -pthread fails to link there, so the default build uses a no-op Mutex with the same lock()/unlock():
// the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::recursive_mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

typedef long long Paise;

std::string fmt(Paise p) {
    std::string cents = std::to_string(p % 100);
    return std::to_string(p / 100) + "." + (cents.size() < 2 ? "0" : "") + cents;
}

// ---------- dates as day numbers (civil calendar <-> days since 1970-01-01) ----------
long daysFromCivil(int y, int m, int d) {
    y -= m <= 2;
    const long era = (y >= 0 ? y : y - 399) / 400;
    const unsigned yoe = static_cast<unsigned>(y - era * 400);
    const unsigned doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
    const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    return era * 146097 + static_cast<long>(doe) - 719468;
}
std::string dateStr(long z) {
    z += 719468;
    const long era = (z >= 0 ? z : z - 146096) / 146097;
    const unsigned doe = static_cast<unsigned>(z - era * 146097);
    const unsigned yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
    const unsigned doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    const unsigned mp = (5 * doy + 2) / 153;
    const unsigned d = doy - (153 * mp + 2) / 5 + 1;
    const unsigned m = mp < 10 ? mp + 3 : mp - 9;
    const long y = static_cast<long>(yoe) + era * 400 + (m <= 2);
    char buf[32];
    std::snprintf(buf, sizeof buf, "%04ld-%02u-%02u", y, m, d);
    return buf;
}

struct LibraryError : std::runtime_error {
    explicit LibraryError(const std::string& m) : std::runtime_error(m) {}
};

struct Book { std::string isbn, title, author, subject; };

enum class CopyStatus { AVAILABLE, BORROWED, ON_HOLD, LOST };

bool canMove(CopyStatus from, CopyStatus to) {   // state table
    switch (from) {
        case CopyStatus::AVAILABLE: return to == CopyStatus::BORROWED || to == CopyStatus::ON_HOLD || to == CopyStatus::LOST;
        case CopyStatus::BORROWED:  return to == CopyStatus::AVAILABLE || to == CopyStatus::ON_HOLD || to == CopyStatus::LOST;
        case CopyStatus::ON_HOLD:   return to == CopyStatus::BORROWED || to == CopyStatus::AVAILABLE || to == CopyStatus::ON_HOLD;
        default:                    return false;
    }
}

struct BookCopy {
    std::string barcode, rack;
    const Book* book;
    CopyStatus status = CopyStatus::AVAILABLE;
    std::string holdFor;
    long holdUntil = 0;
    void moveTo(CopyStatus next) {
        if (!canMove(status, next)) throw std::logic_error(barcode + ": illegal transition");
        status = next;
        if (next != CopyStatus::ON_HOLD) { holdFor.clear(); holdUntil = 0; }
    }
};

enum class MemberType { STUDENT, FACULTY };
int maxLoans(MemberType t) { return t == MemberType::STUDENT ? 2 : 5; }
int loanDays(MemberType t) { return t == MemberType::STUDENT ? 14 : 30; }

struct Loan { BookCopy* copy; std::string memberId; long issued, due; bool renewed; };

struct Member {
    std::string id, name;
    MemberType type;
    Paise dues = 0;
    std::map<std::string, Loan*> loans;   // barcode -> loan
};

// ---------- Strategy ----------
struct FineStrategy {
    virtual ~FineStrategy() {}
    virtual Paise fine(long daysLate) const = 0;
};
struct PerDayFine : FineStrategy {
    Paise rate, cap;
    PerDayFine(Paise r, Paise c) : rate(r), cap(c) {}
    Paise fine(long late) const override { return std::min<Paise>(std::max(0L, late) * rate, cap); }
};
struct GraceThenPerDayFine : FineStrategy {
    long grace; Paise rate;
    GraceThenPerDayFine(long g, Paise r) : grace(g), rate(r) {}
    Paise fine(long late) const override { return std::max(0L, late - grace) * rate; }
};

// ---------- Observer ----------
typedef std::function<void(const Member&, const Book&, long)> Listener;

class LibraryService {
    mutable Mutex mu_;   // the facade lock: every public call holds it
    static const int HOLD_DAYS = 3;
    std::vector<std::unique_ptr<Book>> books;                       // stable addresses
    std::map<std::string, Book*> byIsbn;
    std::map<std::string, std::vector<BookCopy*>> copiesByIsbn;
    std::map<std::string, std::unique_ptr<BookCopy>> copies;
    std::map<std::string, Member> members;
    std::map<std::string, std::unique_ptr<Loan>> active;            // barcode -> loan
    std::map<std::string, std::deque<std::string>> waitlist;
    std::map<MemberType, std::unique_ptr<FineStrategy>> fines;
    std::vector<Listener> listeners;

    const Book& book(const std::string& isbn) const {
        auto it = byIsbn.find(isbn);
        if (it == byIsbn.end()) throw LibraryError("unknown book " + isbn);
        return *it->second;
    }
    Member& member(const std::string& id) {
        auto it = members.find(id);
        if (it == members.end()) throw LibraryError("unknown member " + id);
        return it->second;
    }
    BookCopy& copy(const std::string& barcode) {
        auto it = copies.find(barcode);
        if (it == copies.end()) throw LibraryError("unknown copy " + barcode);
        return *it->second;
    }
    bool hasTitle(const Member& m, const Book& b) const {
        for (const auto& kv : m.loans) if (kv.second->copy->book == &b) return true;
        return false;
    }
    void passToNext(BookCopy& c) {
        auto& q = waitlist[c.book->isbn];
        if (q.empty()) return;
        Member& next = member(q.front());
        q.pop_front();
        c.moveTo(CopyStatus::ON_HOLD);
        c.holdFor = next.id;
        c.holdUntil = today + HOLD_DAYS;
        for (const auto& l : listeners) l(next, *c.book, c.holdUntil);
    }

public:
    long today;
    explicit LibraryService(long t) : today(t) {
        fines[MemberType::STUDENT].reset(new PerDayFine(1000, 50000));
        fines[MemberType::FACULTY].reset(new GraceThenPerDayFine(2, 500));
    }
    void subscribe(Listener l) { LockGuard g(mu_); listeners.push_back(l); }
    void addBook(const Book& b) { LockGuard g(mu_); books.emplace_back(new Book(b)); byIsbn[b.isbn] = books.back().get(); copiesByIsbn[b.isbn]; }
    void addCopy(const std::string& barcode, const std::string& isbn, const std::string& rack) { LockGuard g(mu_);
        std::unique_ptr<BookCopy> c(new BookCopy());
        c->barcode = barcode; c->rack = rack; c->book = &book(isbn);
        copiesByIsbn[isbn].push_back(c.get());
        copies[barcode] = std::move(c);
    }
    void addMember(const std::string& id, const std::string& name, MemberType t) { LockGuard g(mu_);
        Member m; m.id = id; m.name = name; m.type = t; members[id] = m;
    }

    std::vector<std::string> search(std::string q) const { LockGuard g(mu_);
        auto lower = [](std::string s) { for (auto& ch : s) ch = static_cast<char>(std::tolower(static_cast<unsigned char>(ch))); return s; };
        q = lower(q);
        std::vector<std::string> out;
        for (const auto& b : books) {
            if (lower(b->title + " " + b->author + " " + b->subject).find(q) == std::string::npos) continue;
            const auto& cs = copiesByIsbn.at(b->isbn);
            int freeN = 0;
            for (const auto* c : cs) if (c->status == CopyStatus::AVAILABLE) ++freeN;
            out.push_back(b->title + " by " + b->author + " [" + std::to_string(freeN) + "/" + std::to_string(cs.size()) + " available]");
        }
        return out;
    }

    Loan borrow(const std::string& memberId, const std::string& isbn) { LockGuard g(mu_);
        Member& m = member(memberId);
        const Book& b = book(isbn);
        if (m.dues > 0) throw LibraryError("pay dues " + fmt(m.dues) + " first");
        if (static_cast<int>(m.loans.size()) >= maxLoans(m.type)) throw LibraryError("loan limit " + std::to_string(maxLoans(m.type)) + " reached");
        if (hasTitle(m, b)) throw LibraryError("already has a copy of " + b.title);
        BookCopy* pick = nullptr;
        bool held = false;
        for (auto* c : copiesByIsbn[isbn]) if (c->status == CopyStatus::ON_HOLD && c->holdFor == memberId) pick = c;
        if (!pick) for (auto* c : copiesByIsbn[isbn]) if (c->status == CopyStatus::AVAILABLE) { pick = c; break; }
        for (auto* c : copiesByIsbn[isbn]) if (c->status == CopyStatus::ON_HOLD) held = true;
        if (!pick) throw LibraryError(held ? "all copies are on hold for other members" : "no copy available; reserve it");
        pick->moveTo(CopyStatus::BORROWED);
        auto& q = waitlist[isbn];
        q.erase(std::remove(q.begin(), q.end(), memberId), q.end());
        std::unique_ptr<Loan> loan(new Loan{pick, memberId, today, today + loanDays(m.type), false});
        m.loans[pick->barcode] = loan.get();
        Loan copyOut = *loan;
        active[pick->barcode] = std::move(loan);
        return copyOut;
    }

    int reserve(const std::string& memberId, const std::string& isbn) { LockGuard g(mu_);
        Member& m = member(memberId);
        const Book& b = book(isbn);
        for (auto* c : copiesByIsbn[isbn]) if (c->status == CopyStatus::AVAILABLE) throw LibraryError("a copy is available; borrow it instead");
        if (hasTitle(m, b)) throw LibraryError("already has a copy of " + b.title);
        auto& q = waitlist[isbn];
        if (std::find(q.begin(), q.end(), memberId) != q.end()) throw LibraryError("already in the queue");
        q.push_back(memberId);
        return static_cast<int>(q.size());
    }

    Paise returnCopy(const std::string& barcode) { LockGuard g(mu_);
        BookCopy& c = copy(barcode);
        if (c.status != CopyStatus::BORROWED) throw LibraryError(barcode + " is not on loan");
        std::unique_ptr<Loan> loan = std::move(active[barcode]);
        active.erase(barcode);
        Member& m = member(loan->memberId);
        m.loans.erase(barcode);
        Paise f = fines[m.type]->fine(today - loan->due);
        m.dues += f;
        c.moveTo(CopyStatus::AVAILABLE);
        passToNext(c);
        return f;
    }

    int expireHolds() { LockGuard g(mu_);
        int n = 0;
        for (auto& kv : copies) {
            BookCopy& c = *kv.second;
            if (c.status == CopyStatus::ON_HOLD && today > c.holdUntil) { ++n; c.moveTo(CopyStatus::AVAILABLE); passToNext(c); }
        }
        return n;
    }

    void renew(const std::string& barcode) { LockGuard g(mu_);
        BookCopy& c = copy(barcode);
        if (c.status != CopyStatus::BORROWED) throw LibraryError(barcode + " is not on loan");
        size_t waiting = waitlist[c.book->isbn].size();
        if (waiting > 0) throw LibraryError("cannot renew: " + std::to_string(waiting) + " member(s) waiting");
        Loan& l = *active[barcode];
        if (l.renewed) throw LibraryError("already renewed once");
        l.due += loanDays(member(l.memberId).type);
        l.renewed = true;
    }

    void payDues(const std::string& memberId, Paise p) { LockGuard g(mu_);
        Member& m = member(memberId);
        if (p <= 0 || p > m.dues) throw LibraryError("invalid payment");
        m.dues -= p;
    }
    CopyStatus statusOf(const std::string& barcode) { LockGuard g(mu_); return copy(barcode).status; }
};

static int checks = 0;
void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "check failed: " << what << "\n"; std::exit(1); }
    ++checks;
}
void attempt(const std::string& label, const std::function<void()>& fn) {
    try { fn(); std::cout << label << " -> ok\n"; }
    catch (const LibraryError& e) { std::cout << label << " -> rejected: " << e.what() << "\n"; }
}
