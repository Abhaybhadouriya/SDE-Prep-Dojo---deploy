-- schema for the design-shape round
CREATE TABLE books   (isbn VARCHAR PRIMARY KEY, title VARCHAR, author VARCHAR, subject VARCHAR, published DATE);
CREATE TABLE copies  (barcode VARCHAR PRIMARY KEY, isbn VARCHAR REFERENCES books, rack VARCHAR,
                      status VARCHAR CHECK (status IN ('AVAILABLE','BORROWED','ON_HOLD','LOST')),
                      hold_for VARCHAR NULL, hold_until DATE NULL);
CREATE TABLE members (id VARCHAR PRIMARY KEY, name VARCHAR, type VARCHAR, dues_paise BIGINT DEFAULT 0);
CREATE TABLE loans   (id BIGSERIAL PRIMARY KEY, barcode VARCHAR REFERENCES copies, member_id VARCHAR REFERENCES members,
                      issued DATE, due DATE, returned DATE NULL, fine_paise BIGINT DEFAULT 0, renewed BOOLEAN DEFAULT FALSE);
CREATE UNIQUE INDEX one_open_loan_per_copy ON loans(barcode) WHERE returned IS NULL;
CREATE TABLE reservations (isbn VARCHAR, member_id VARCHAR, requested_at TIMESTAMP,
                      PRIMARY KEY (isbn, member_id));          -- FIFO = ORDER BY requested_at

The partial unique index is the database's version of our state check: two open loans for one copy are impossible even if application code has a bug.
