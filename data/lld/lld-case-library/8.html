enum CopyStatus {
    AVAILABLE, BORROWED, ON_HOLD, LOST;
    /** Allowed transitions, like a protocol state table. */
    boolean canMoveTo(CopyStatus next) {
        switch (this) {
            case AVAILABLE: return next == BORROWED || next == ON_HOLD || next == LOST;
            case BORROWED:  return next == AVAILABLE || next == ON_HOLD || next == LOST;
            case ON_HOLD:   return next == BORROWED || next == AVAILABLE || next == ON_HOLD;
            default:        return false;                               // LOST is terminal
        }
    }
}

final class BookCopy {
    final String barcode, rack;
    final Book book;
    private CopyStatus status = CopyStatus.AVAILABLE;
    private String holdFor;          // member id while ON_HOLD
    private LocalDate holdUntil;
    BookCopy(String barcode, Book book, String rack) { this.barcode = barcode; this.book = book; this.rack = rack; }

    CopyStatus status() { return status; }
    String holdFor() { return holdFor; }
    LocalDate holdUntil() { return holdUntil; }
    void moveTo(CopyStatus next) {
        if (!status.canMoveTo(next)) throw new IllegalStateException(barcode + ": " + status + " -> " + next);
        status = next;
        if (next != CopyStatus.ON_HOLD) { holdFor = null; holdUntil = null; }
    }
    void holdFor(String memberId, LocalDate until) { moveTo(CopyStatus.ON_HOLD); holdFor = memberId; holdUntil = until; }
}

final class LibraryService {
    static final int HOLD_DAYS = 3;
    private final Map<String, Book> books = new LinkedHashMap<>();
    private final Map<String, List<BookCopy>> copiesByIsbn = new HashMap<>();
    private final Map<String, BookCopy> copies = new HashMap<>();
    private final Map<String, Member> members = new HashMap<>();
    private final Map<String, Loan> activeLoans = new HashMap<>();          // barcode -> loan
    private final Map<String, Deque<String>> waitlist = new HashMap<>();   // isbn -> member ids, FIFO
    private final Map<MemberType, FineStrategy> fines = new EnumMap<>(MemberType.class);
    private final List<AvailabilityListener> listeners = new CopyOnWriteArrayList<>();
    private LocalDate today;

    LibraryService(LocalDate today) {
        this.today = today;
        fines.put(MemberType.STUDENT, new PerDayFine(1000, 50000));        // 10.00 a day, max 500.00
        fines.put(MemberType.FACULTY, new GraceThenPerDayFine(2, 500));    // 2 free days, then 5.00
    }
    synchronized void setToday(LocalDate d) { today = d; }
    void subscribe(AvailabilityListener l) { listeners.add(l); }

    synchronized void addBook(Book b) { books.put(b.isbn, b); copiesByIsbn.putIfAbsent(b.isbn, new ArrayList<>()); }
    synchronized void addCopy(String barcode, String isbn, String rack) {
        BookCopy c = new BookCopy(barcode, book(isbn), rack);
        copies.put(barcode, c);
        copiesByIsbn.get(isbn).add(c);
    }
    synchronized void addMember(Member m) { members.put(m.id, m); }

    private Book book(String isbn) {
        Book b = books.get(isbn);
        if (b == null) throw new LibraryException("unknown book " + isbn);
        return b;
    }
    private Member member(String id) {
        Member m = members.get(id);
        if (m == null) throw new LibraryException("unknown member " + id);
        return m;
    }
    private BookCopy copy(String barcode) {
        BookCopy c = copies.get(barcode);
        if (c == null) throw new LibraryException("unknown copy " + barcode);
        return c;
    }

    synchronized List<String> search(String query) {
        String q = query.toLowerCase(Locale.ROOT);
        List<String> out = new ArrayList<>();
        for (Book b : books.values()) {
            if (!(b.title + " " + b.author + " " + b.subject).toLowerCase(Locale.ROOT).contains(q)) continue;
            long free = copiesByIsbn.get(b.isbn).stream().filter(c -> c.status() == CopyStatus.AVAILABLE).count();   // C: stream() = a for-loop written as filter/map/collect
            out.add(b.title + " by " + b.author + " [" + free + "/" + copiesByIsbn.get(b.isbn).size() + " available]");
        }
        return out;
    }

    Loan borrow(String memberId, String isbn) {
        synchronized (this) {
            Member m = member(memberId);
            Book b = book(isbn);
            if (m.duesPaise > 0) throw new LibraryException("pay dues " + Money.fmt(m.duesPaise) + " first");
            if (m.loans.size() >= m.type.maxLoans) throw new LibraryException("loan limit " + m.type.maxLoans + " reached");
            for (Loan l : m.loans.values())
                if (l.copy.book == b) throw new LibraryException("already has a copy of " + b.title);
            BookCopy pick = null;
            for (BookCopy c : copiesByIsbn.get(isbn))                     // a copy held for me wins
                if (c.status() == CopyStatus.ON_HOLD && memberId.equals(c.holdFor())) pick = c;
            if (pick == null)
                for (BookCopy c : copiesByIsbn.get(isbn))
                    if (c.status() == CopyStatus.AVAILABLE) { pick = c; break; }
            if (pick == null) {
                boolean held = copiesByIsbn.get(isbn).stream().anyMatch(c -> c.status() == CopyStatus.ON_HOLD);
                throw new LibraryException(held ? "all copies are on hold for other members" : "no copy available; reserve it");
            }
            pick.moveTo(CopyStatus.BORROWED);
            Deque<String> q = waitlist.get(isbn);
            if (q != null) q.remove(memberId);
            Loan loan = new Loan(pick, m, today, today.plusDays(m.type.loanDays));
            m.loans.put(pick.barcode, loan);
            activeLoans.put(pick.barcode, loan);
            return loan;
        }
    }

    int reserve(String memberId, String isbn) {
        synchronized (this) {
            Member m = member(memberId);
            Book b = book(isbn);
            if (copiesByIsbn.get(isbn).stream().anyMatch(c -> c.status() == CopyStatus.AVAILABLE))
                throw new LibraryException("a copy is available; borrow it instead");
            for (Loan l : m.loans.values())
                if (l.copy.book == b) throw new LibraryException("already has a copy of " + b.title);
            Deque<String> q = waitlist.computeIfAbsent(isbn, k -> new ArrayDeque<>());   // C: look up k; if missing, allocate, insert and return it
            if (q.contains(memberId)) throw new LibraryException("already in the queue");
            q.addLast(memberId);
            return q.size();
        }
    }

    /** Returns the fine charged. Listeners are told after the lock is released. */
    long returnCopy(String barcode) {
        List<Runnable> events = new ArrayList<>();
        long fine;
        synchronized (this) {
            BookCopy c = copy(barcode);
            if (c.status() != CopyStatus.BORROWED) throw new LibraryException(barcode + " is not on loan");
            Loan loan = activeLoans.remove(barcode);
            loan.member.loans.remove(barcode);
            long late = ChronoUnit.DAYS.between(loan.due, today);
            fine = fines.get(loan.member.type).fine(late);
            loan.member.duesPaise += fine;
            c.moveTo(CopyStatus.AVAILABLE);
            passToNextInQueue(c, events);
        }
        events.forEach(Runnable::run);   // C: Type::method = a pointer to an existing function
        return fine;
    }

    /** Holds not collected in time go to the next member in the queue (or back on the shelf). */
    int expireHolds() {
        List<Runnable> events = new ArrayList<>();
        int expired = 0;
        synchronized (this) {
            for (BookCopy c : copies.values())
                if (c.status() == CopyStatus.ON_HOLD && today.isAfter(c.holdUntil())) {
                    expired++;
                    c.moveTo(CopyStatus.AVAILABLE);
                    passToNextInQueue(c, events);
                }
        }
        events.forEach(Runnable::run);
        return expired;
    }

    private void passToNextInQueue(BookCopy c, List<Runnable> events) {
        Deque<String> q = waitlist.get(c.book.isbn);
        if (q == null || q.isEmpty()) return;
        Member next = member(q.pollFirst());
        LocalDate until = today.plusDays(HOLD_DAYS);
        c.holdFor(next.id, until);
        for (AvailabilityListener l : listeners) events.add(() -> l.onHold(next, c.book, until));
    }

    synchronized void renew(String barcode) {
        BookCopy c = copy(barcode);
        if (c.status() != CopyStatus.BORROWED) throw new LibraryException(barcode + " is not on loan");
        int waiting = waitlist.getOrDefault(c.book.isbn, new ArrayDeque<>()).size();
        if (waiting > 0) throw new LibraryException("cannot renew: " + waiting + " member(s) waiting");
        Loan l = activeLoans.get(barcode);
        if (l.renewed) throw new LibraryException("already renewed once");
        l.due = l.due.plusDays(l.member.type.loanDays);
        l.renewed = true;
    }

    synchronized void payDues(String memberId, long paise) {
        Member m = member(memberId);
        if (paise <= 0 || paise > m.duesPaise) throw new LibraryException("invalid payment");
        m.duesPaise -= paise;
    }
    synchronized long duesOf(String memberId) { return member(memberId).duesPaise; }
    synchronized CopyStatus statusOf(String barcode) { return copy(barcode).status(); }
}

        for (Future<?> f : fs) f.get();
        pool.shutdown();
        System.out.println("borrowed: " + wins.get() + ", turned away: " + (6 - wins.get()));
        check(wins.get() == 2, "never lend more copies than exist");

        System.out.println("All " + checks + " checks passed");

from __future__ import annotations

import threading
from collections import deque
from dataclasses import dataclass, field
from datetime import date, timedelta
from enum import Enum
from typing import Callable, Deque, Dict, List, Optional


def fmt(paise: int) -> str:
    return f"{paise // 100}.{paise % 100:02d}"


class LibraryError(Exception):
    pass


@dataclass(frozen=True)
class Book:
    isbn: str
    title: str
    author: str
    subject: str


class CopyStatus(Enum):
    AVAILABLE = "AVAILABLE"
    BORROWED = "BORROWED"
    ON_HOLD = "ON_HOLD"
    LOST = "LOST"


ALLOWED = {  # state table: from -> allowed next states
    CopyStatus.AVAILABLE: {CopyStatus.BORROWED, CopyStatus.ON_HOLD, CopyStatus.LOST},
    CopyStatus.BORROWED: {CopyStatus.AVAILABLE, CopyStatus.ON_HOLD, CopyStatus.LOST},
    CopyStatus.ON_HOLD: {CopyStatus.BORROWED, CopyStatus.AVAILABLE, CopyStatus.ON_HOLD},
    CopyStatus.LOST: set(),
}


class BookCopy:
    def __init__(self, barcode: str, book: Book, rack: str) -> None:
        self.barcode, self.book, self.rack = barcode, book, rack
        self.status = CopyStatus.AVAILABLE
        self.hold_for: Optional[str] = None
        self.hold_until: Optional[date] = None

    def move_to(self, nxt: CopyStatus) -> None:
        if nxt not in ALLOWED[self.status]:
            raise RuntimeError(f"{self.barcode}: {self.status.value} -> {nxt.value}")
        self.status = nxt
        if nxt != CopyStatus.ON_HOLD:
            self.hold_for = self.hold_until = None

    def hold(self, member_id: str, until: date) -> None:
        self.move_to(CopyStatus.ON_HOLD)
        self.hold_for, self.hold_until = member_id, until


class MemberType(Enum):
    STUDENT = (2, 14)   # max loans, loan days
    FACULTY = (5, 30)

    @property
    def max_loans(self) -> int:
        return self.value[0]

    @property
    def loan_days(self) -> int:
        return self.value[1]


@dataclass
class Member:
    id: str
    name: str
    type: MemberType
    dues: int = 0
    loans: Dict[str, "Loan"] = field(default_factory=dict)  # barcode -> loan


@dataclass
class Loan:
    copy: BookCopy
    member: Member
    issued: date
    due: date
    renewed: bool = False


# ---------- Strategy ----------
class FineStrategy:
    def fine(self, days_late: int) -> int:
        raise NotImplementedError


class PerDayFine(FineStrategy):
    def __init__(self, rate: int, cap: int) -> None:
        self.rate, self.cap = rate, cap

    def fine(self, days_late: int) -> int:
        return min(max(0, days_late) * self.rate, self.cap)


class GraceThenPerDayFine(FineStrategy):
    def __init__(self, grace_days: int, rate: int) -> None:
        self.grace, self.rate = grace_days, rate

    def fine(self, days_late: int) -> int:
        return max(0, days_late - self.grace) * self.rate


# ---------- Observer ----------
Listener = Callable[[Member, Book, date], None]


class LibraryService:
    HOLD_DAYS = 3

    def __init__(self, today: date) -> None:
        self.today = today
        self._books: Dict[str, Book] = {}
        self._copies_by_isbn: Dict[str, List[BookCopy]] = {}
        self._copies: Dict[str, BookCopy] = {}
        self._members: Dict[str, Member] = {}
        self._active: Dict[str, Loan] = {}
        self._waitlist: Dict[str, Deque[str]] = {}
        self._fines: Dict[MemberType, FineStrategy] = {
            MemberType.STUDENT: PerDayFine(1000, 50000),       # 10.00 a day, max 500.00
            MemberType.FACULTY: GraceThenPerDayFine(2, 500),   # 2 free days, then 5.00
        }
        self._listeners: List[Listener] = []
        self._lock = threading.RLock()

    def subscribe(self, listener: Listener) -> None:
        self._listeners.append(listener)

    def add_book(self, b: Book) -> None:
        with self._lock:
            self._books[b.isbn] = b
            self._copies_by_isbn.setdefault(b.isbn, [])

    def add_copy(self, barcode: str, isbn: str, rack: str) -> None:
        with self._lock:
            c = BookCopy(barcode, self._book(isbn), rack)
            self._copies[barcode] = c
            self._copies_by_isbn[isbn].append(c)

    def add_member(self, m: Member) -> None:
        with self._lock:
            self._members[m.id] = m

    def _book(self, isbn: str) -> Book:
        if isbn not in self._books:
            raise LibraryError(f"unknown book {isbn}")
        return self._books[isbn]

    def _member(self, mid: str) -> Member:
        if mid not in self._members:
            raise LibraryError(f"unknown member {mid}")
        return self._members[mid]

    def _copy(self, barcode: str) -> BookCopy:
        if barcode not in self._copies:
            raise LibraryError(f"unknown copy {barcode}")
        return self._copies[barcode]

    def search(self, query: str) -> List[str]:
        q = query.lower()
        out = []
        with self._lock:
            for b in self._books.values():
                if q not in f"{b.title} {b.author} {b.subject}".lower():
                    continue
                cs = self._copies_by_isbn[b.isbn]
                free = sum(1 for c in cs if c.status == CopyStatus.AVAILABLE)
                out.append(f"{b.title} by {b.author} [{free}/{len(cs)} available]")
        return out

    def borrow(self, member_id: str, isbn: str) -> Loan:
        with self._lock:
            m, b = self._member(member_id), self._book(isbn)
            if m.dues > 0:
                raise LibraryError(f"pay dues {fmt(m.dues)} first")
            if len(m.loans) >= m.type.max_loans:
                raise LibraryError(f"loan limit {m.type.max_loans} reached")
            if any(l.copy.book == b for l in m.loans.values()):
                raise LibraryError(f"already has a copy of {b.title}")
            cs = self._copies_by_isbn[isbn]
            pick = next((c for c in cs if c.status == CopyStatus.ON_HOLD and c.hold_for == member_id), None)
            if pick is None:
                pick = next((c for c in cs if c.status == CopyStatus.AVAILABLE), None)
            if pick is None:
                held = any(c.status == CopyStatus.ON_HOLD for c in cs)
                raise LibraryError("all copies are on hold for other members" if held
                                   else "no copy available; reserve it")
            pick.move_to(CopyStatus.BORROWED)
            q = self._waitlist.get(isbn)
            if q and member_id in q:
                q.remove(member_id)
            loan = Loan(pick, m, self.today, self.today + timedelta(days=m.type.loan_days))
            m.loans[pick.barcode] = loan
            self._active[pick.barcode] = loan
            return loan

    def reserve(self, member_id: str, isbn: str) -> int:
        with self._lock:
            m, b = self._member(member_id), self._book(isbn)
            if any(c.status == CopyStatus.AVAILABLE for c in self._copies_by_isbn[isbn]):
                raise LibraryError("a copy is available; borrow it instead")
            if any(l.copy.book == b for l in m.loans.values()):
                raise LibraryError(f"already has a copy of {b.title}")
            q = self._waitlist.setdefault(isbn, deque())
            if member_id in q:
                raise LibraryError("already in the queue")
            q.append(member_id)
            return len(q)

    def return_copy(self, barcode: str) -> int:
        events: List[Callable[[], None]] = []
        with self._lock:
            c = self._copy(barcode)
            if c.status != CopyStatus.BORROWED:
                raise LibraryError(f"{barcode} is not on loan")
            loan = self._active.pop(barcode)
            del loan.member.loans[barcode]
            fine = self._fines[loan.member.type].fine((self.today - loan.due).days)
            loan.member.dues += fine
            c.move_to(CopyStatus.AVAILABLE)
            self._pass_to_next(c, events)
        for e in events:  # notify outside the lock
            e()
        return fine

    def expire_holds(self) -> int:
        events: List[Callable[[], None]] = []
        expired = 0
        with self._lock:
            for c in self._copies.values():
                if c.status == CopyStatus.ON_HOLD and self.today > c.hold_until:
                    expired += 1
                    c.move_to(CopyStatus.AVAILABLE)
                    self._pass_to_next(c, events)
        for e in events:
            e()
        return expired

    def _pass_to_next(self, c: BookCopy, events: List[Callable[[], None]]) -> None:
        q = self._waitlist.get(c.book.isbn)
        if not q:
            return
        nxt = self._member(q.popleft())
        until = self.today + timedelta(days=self.HOLD_DAYS)
        c.hold(nxt.id, until)
        for listener in self._listeners:
            events.append(lambda l=listener: l(nxt, c.book, until))

    def renew(self, barcode: str) -> None:
        with self._lock:
            c = self._copy(barcode)
            if c.status != CopyStatus.BORROWED:
                raise LibraryError(f"{barcode} is not on loan")
            waiting = len(self._waitlist.get(c.book.isbn, ()))
            if waiting:
                raise LibraryError(f"cannot renew: {waiting} member(s) waiting")
            loan = self._active[barcode]
            if loan.renewed:
                raise LibraryError("already renewed once")
            loan.due += timedelta(days=loan.member.type.loan_days)
            loan.renewed = True

    def pay_dues(self, member_id: str, paise: int) -> None:
        with self._lock:
            m = self._member(member_id)
            if paise <= 0 or paise > m.dues:
                raise LibraryError("invalid payment")
            m.dues -= paise

    def status_of(self, barcode: str) -> CopyStatus:
        return self._copy(barcode).status


def build_library() -> LibraryService:
    lib = LibraryService(date(2026, 11, 1))
    lib.add_book(Book("B1", "Clean Code", "Robert Martin", "Software"))
    lib.add_book(Book("B2", "Linux Kernel Development", "Robert Love", "Operating systems"))
    lib.add_book(Book("B3", "TCP/IP Illustrated", "W. Richard Stevens", "Networking"))
    for barcode, isbn, rack in (("C1", "B1", "R1-S2"), ("C2", "B1", "R1-S2"),
                                ("C3", "B2", "R4-S1"), ("C4", "B3", "R5-S3")):
        lib.add_copy(barcode, isbn, rack)
    lib.add_member(Member("m1", "Asha", MemberType.STUDENT))
    lib.add_member(Member("m2", "Ben", MemberType.STUDENT))
    lib.add_member(Member("m3", "Chitra", MemberType.FACULTY))
    return lib


def attempt(label: str, fn: Callable[[], object]) -> None:
    try:
        fn()
        print(f"{label} -> ok")
    except LibraryError as e:
        print(f"{label} -> rejected: {e}")


def main() -> None:
    lib = build_library()
    sent: List[str] = []

    def notify(m: Member, b: Book, until: date) -> None:
        msg = f"notify {m.id}: '{b.title}' is ready for pickup until {until.isoformat()}"
        sent.append(msg)
        print("  " + msg)

    lib.subscribe(notify)
    print("== 2026-11-01 search 'robert'")
    for line in lib.search("robert"):
        print("  " + line)
    l1 = lib.borrow("m1", "B2")
    print(f"m1 borrows B2 -> {l1.copy.barcode} due {l1.due.isoformat()}")
    l2 = lib.borrow("m1", "B1")
    print(f"m1 borrows B1 -> {l2.copy.barcode} due {l2.due.isoformat()}")
    attempt("m1 borrows B3", lambda: lib.borrow("m1", "B3"))
    attempt("m2 borrows B2", lambda: lib.borrow("m2", "B2"))
    print(f"m2 reserves B2 -> position {lib.reserve('m2', 'B2')}")
    print(f"m3 reserves B2 -> position {lib.reserve('m3', 'B2')}")
    attempt("m3 reserves B1", lambda: lib.reserve("m3", "B1"))
    attempt("m1 renews C3", lambda: lib.renew("C3"))

    print("== 2026-11-10 m1 returns C1 early")
    lib.today = date(2026, 11, 10)
    print(f"fine {fmt(lib.return_copy('C1'))}")

    print("== 2026-11-18 m1 returns C3, 3 days late")
    lib.today = date(2026, 11, 18)
    fine = lib.return_copy("C3")
    print(f"fine {fmt(fine)}")
    assert fine == 3000 and lib.status_of("C3") == CopyStatus.ON_HOLD
    attempt("m3 borrows B2", lambda: lib.borrow("m3", "B2"))
    attempt("m1 borrows B3", lambda: lib.borrow("m1", "B3"))
    lib.pay_dues("m1", 3000)
    print("m1 pays 30.00")
    attempt("m1 borrows B3", lambda: lib.borrow("m1", "B3"))

    print("== 2026-11-22 m2 never picked up")
    lib.today = date(2026, 11, 22)
    print(f"expired holds: {lib.expire_holds()}")
    attempt("m2 borrows B2", lambda: lib.borrow("m2", "B2"))
    l3 = lib.borrow("m3", "B2")
    print(f"m3 borrows B2 -> {l3.copy.barcode} due {l3.due.isoformat()}")

    print("== 2026-12-26 m3 returns C3, 4 days late")
    lib.today = date(2026, 12, 26)
    fine3 = lib.return_copy("C3")
    print(f"fine {fmt(fine3)} (2 days grace, then 5.00 a day)")
    assert fine3 == 1000 and lib.status_of("C3") == CopyStatus.AVAILABLE and len(sent) == 2
    print("demo OK")


if __name__ == "__main__":
    main()
import threading
import unittest
from datetime import date

from main import CopyStatus, LibraryError, Member, MemberType, build_library


class LibraryTest(unittest.TestCase):
    def setUp(self):
        self.lib = build_library()
        self.sent = []
        self.lib.subscribe(lambda m, b, until: self.sent.append((m.id, b.isbn, until)))

    def test_book_vs_copy_and_limits(self):
        self.lib.borrow("m1", "B1")
        with self.assertRaisesRegex(LibraryError, "already has a copy"):
            self.lib.borrow("m1", "B1")
        self.lib.borrow("m1", "B2")
        with self.assertRaisesRegex(LibraryError, "loan limit 2"):
            self.lib.borrow("m1", "B3")
        self.assertIn("Clean Code by Robert Martin [1/2 available]", self.lib.search("clean"))

    def test_reserve_only_when_nothing_on_shelf(self):
        with self.assertRaisesRegex(LibraryError, "borrow it instead"):
            self.lib.reserve("m2", "B2")
        self.lib.borrow("m1", "B2")
        self.assertEqual(self.lib.reserve("m2", "B2"), 1)
        with self.assertRaisesRegex(LibraryError, "already in the queue"):
            self.lib.reserve("m2", "B2")
        with self.assertRaisesRegex(LibraryError, "already has a copy"):
            self.lib.reserve("m1", "B2")

    def test_return_goes_to_queue_in_fifo_order_and_notifies(self):
        self.lib.borrow("m1", "B2")
        self.lib.reserve("m2", "B2")
        self.lib.reserve("m3", "B2")
        self.lib.return_copy("C3")
        self.assertEqual(self.lib.status_of("C3"), CopyStatus.ON_HOLD)
        self.assertEqual(self.sent, [("m2", "B2", date(2026, 11, 4))])
        with self.assertRaisesRegex(LibraryError, "on hold for other members"):
            self.lib.borrow("m3", "B2")
        self.lib.borrow("m2", "B2")  # the holder can take it

    def test_expired_hold_moves_on(self):
        self.lib.borrow("m1", "B2")
        self.lib.reserve("m2", "B2")
        self.lib.reserve("m3", "B2")
        self.lib.return_copy("C3")
        self.lib.today = date(2026, 11, 5)
        self.assertEqual(self.lib.expire_holds(), 1)
        self.assertEqual(self.sent[-1][0], "m3")
        self.lib.borrow("m3", "B2")

    def test_fine_strategies_and_dues_block(self):
        self.lib.borrow("m1", "B1")            # student, due 11-15
        self.lib.borrow("m3", "B2")            # faculty, due 12-01
        self.lib.today = date(2026, 12, 4)
        self.assertEqual(self.lib.return_copy("C1"), 19 * 1000)   # 19 days x 10.00
        self.assertEqual(self.lib.return_copy("C3"), 1 * 500)     # 3 late - 2 grace
        with self.assertRaisesRegex(LibraryError, "pay dues 190.00"):
            self.lib.borrow("m1", "B3")
        self.lib.pay_dues("m1", 19000)
        self.lib.borrow("m1", "B3")

    def test_fine_is_capped(self):
        self.lib.borrow("m1", "B1")
        self.lib.today = date(2027, 6, 1)
        self.assertEqual(self.lib.return_copy("C1"), 50000)

    def test_renew_rules(self):
        self.lib.borrow("m1", "B1")
        self.lib.renew("C1")
        with self.assertRaisesRegex(LibraryError, "already renewed"):
            self.lib.renew("C1")
        self.lib.borrow("m1", "B2")
        self.lib.reserve("m2", "B2")
        with self.assertRaisesRegex(LibraryError, "cannot renew: 1"):
            self.lib.renew("C3")

    def test_double_return_rejected(self):
        self.lib.borrow("m1", "B1")
        self.lib.return_copy("C1")
        with self.assertRaisesRegex(LibraryError, "not on loan"):
            self.lib.return_copy("C1")

    def test_concurrent_borrows_never_exceed_copies(self):
        ids = [f"x{i}" for i in range(8)]
        for i in ids:
            self.lib.add_member(Member(i, i, MemberType.STUDENT))
        wins = []
        gate = threading.Barrier(len(ids))

        def go(mid):
            gate.wait()
            try:
                self.lib.borrow(mid, "B1")
                wins.append(mid)
            except LibraryError:
                pass

        ts = [threading.Thread(target=go, args=(i,)) for i in ids]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
        self.assertEqual(len(wins), 2)


if __name__ == "__main__":
    unittest.main()
