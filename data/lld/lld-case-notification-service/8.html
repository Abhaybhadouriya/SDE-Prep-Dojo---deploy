From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

The longer version is OOP for C programmers.

Domain.java - enums and immutable records

import java.util.EnumSet;
import java.util.Map;
import java.util.Set;

enum Channel { EMAIL, SMS, PUSH, IN_APP }

/** Declaration order is queue order: HIGH first among jobs due at the same time. */
enum Priority { HIGH, MEDIUM, LOW }

enum DeliveryStatus { SENT, FAILED, SKIPPED_OPT_OUT, NO_ADDRESS, RATE_LIMITED }

record User(String id, String name, String email, String phone, String deviceToken) {
    /** Where a message for this channel goes; null = user has no address for it. */
    String addressFor(Channel c) {
        return switch (c) {
            case EMAIL -> email;
            case SMS -> phone;
            case PUSH -> deviceToken;
            case IN_APP -> id;                       // in-app inbox is keyed by user id
        };
    }
}

/** What a caller asks for. sendAtMs = 0 means "now". idempotencyKey makes retries by the caller safe. */
record NotificationRequest(String idempotencyKey, String userId, String templateId, Map<String, String> params,
                           Set<Channel> channels, Priority priority, long sendAtMs) {
    static NotificationRequest of(String key, String userId, String templateId, Map<String, String> params,
                                  Priority priority, long sendAtMs, Channel first, Channel... rest) {
        return new NotificationRequest(key, userId, templateId, Map.copyOf(params), EnumSet.of(first, rest),
                                       priority, sendAtMs);
    }
}

/** One rendered message for one channel, handed to a sender. */
record Message(String notificationId, String userId, Channel channel, String to, String body, Priority priority) {}

/** One line of history: what happened to one channel of one notification. */
record DeliveryRecord(String notificationId, String userId, Channel channel, DeliveryStatus status, String detail,
                      long atMs) {
    @Override
    public String toString() {
        return notificationId + " " + channel + " " + status + (detail.isEmpty() ? "" : " (" + detail + ")");
    }
}

Templates.java

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/** One template id, one body per channel: "Hi {name}, order {orderId} is on its way". */
final class TemplateRegistry {
    private static final Pattern VAR = Pattern.compile("\\{(\\w+)}");
    private final Map<String, Map<Channel, String>> bodies = new ConcurrentHashMap<>();

    TemplateRegistry register(String templateId, Channel channel, String body) {
        bodies.computeIfAbsent(templateId, k -> new ConcurrentHashMap<>()).put(channel, body);
        return this;
    }

    boolean exists(String templateId) { return bodies.containsKey(templateId); }

    boolean supports(String templateId, Channel c) {
        Map<Channel, String> m = bodies.get(templateId);
        return m != null && m.containsKey(c);
    }

    String render(String templateId, Channel c, Map<String, String> params) {
        String body = bodies.get(templateId).get(c);
        Matcher mt = VAR.matcher(body);
        StringBuilder out = new StringBuilder();
        while (mt.find()) {
            String v = params.get(mt.group(1));
            if (v == null) throw new IllegalArgumentException("missing {" + mt.group(1) + "} for " + templateId);
            mt.appendReplacement(out, Matcher.quoteReplacement(v));
        }
        return mt.appendTail(out).toString();
    }
}

Senders.java - the Strategy interface and a fake provider

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class DeliveryException extends Exception {
    final boolean retryable;

    DeliveryException(String msg, boolean retryable) {
        super(msg);
        this.retryable = retryable;
    }
}

final class RateLimitedException extends DeliveryException {
    RateLimitedException(String msg) { super(msg, false); }
}

/** Strategy per channel. Returns the name of the provider that actually delivered. */
interface ChannelSender {
    String send(Message m) throws DeliveryException;
}

/** Stand-in for a real gateway (SES, Twilio, FCM ...). Can be told to fail. */
final class FakeProvider implements ChannelSender {
    private final String name;
    private int failNext;                 // fail this many calls with a retryable error
    private boolean down;                 // fail every call
    final List<String> delivered = Collections.synchronizedList(new ArrayList<>());

    FakeProvider(String name) { this.name = name; }

    synchronized FakeProvider failNext(int n) { failNext = n; return this; }

    synchronized FakeProvider down(boolean d) { down = d; return this; }

    @Override
    public String send(Message m) throws DeliveryException {
        synchronized (this) {
            if (down) throw new DeliveryException(name + " is down", true);
            if (failNext > 0) {
                failNext--;
                throw new DeliveryException(name + " timeout", true);
            }
        }
        delivered.add(m.to() + ": " + m.body());
        return name;
    }
}

Decorators.java - retry, failover, rate limit

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.atomic.AtomicInteger;

/** Decorator: same interface, adds retries with exponential backoff for retryable errors. */
final class RetryingSender implements ChannelSender {
    private final ChannelSender inner;
    private final int maxAttempts;
    private final long baseBackoffMs;
    private final Sleeper sleeper;
    final AtomicInteger retries = new AtomicInteger();

    RetryingSender(ChannelSender inner, int maxAttempts, long baseBackoffMs, Sleeper sleeper) {
        this.inner = inner;
        this.maxAttempts = maxAttempts;
        this.baseBackoffMs = baseBackoffMs;
        this.sleeper = sleeper;
    }

    @Override
    public String send(Message m) throws DeliveryException {
        for (int attempt = 1; ; attempt++) {
            try {
                return inner.send(m);
            } catch (DeliveryException e) {
                if (!e.retryable || attempt >= maxAttempts) throw e;
                retries.incrementAndGet();
                try {
                    sleeper.sleep(baseBackoffMs << (attempt - 1));
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    throw new DeliveryException("interrupted", false);
                }
            }
        }
    }
}

/** Tries providers in order; the first success wins (PhonePe: many providers per channel). */
final class FailoverSender implements ChannelSender {
    private final List<ChannelSender> providers;

    FailoverSender(List<ChannelSender> providers) {
        if (providers.isEmpty()) throw new IllegalArgumentException("need at least one provider");
        this.providers = List.copyOf(providers);
    }

    @Override
    public String send(Message m) throws DeliveryException {
        DeliveryException last = null;
        for (ChannelSender p : providers) {
            try {
                return p.send(m);
            } catch (DeliveryException e) {
                last = e;
            }
        }
        throw new DeliveryException("all providers failed, last: " + last.getMessage(), true);
    }
}

/** Decorator: at most maxPerWindow messages per user per window. HIGH priority (OTP) is exempt. */
final class RateLimitingSender implements ChannelSender {
    private final ChannelSender inner;
    private final int maxPerWindow;
    private final long windowMs;
    private final Clock clock;
    private final Map<String, Deque<Long>> sentAt = new HashMap<>();

    RateLimitingSender(ChannelSender inner, int maxPerWindow, long windowMs, Clock clock) {
        this.inner = inner;
        this.maxPerWindow = maxPerWindow;
        this.windowMs = windowMs;
        this.clock = clock;
    }

    @Override
    public String send(Message m) throws DeliveryException {
        if (m.priority() != Priority.HIGH) {
            synchronized (this) {                                    // sliding-window log per user
                long now = clock.nowMs();
                Deque<Long> q = sentAt.computeIfAbsent(m.userId(), k -> new ArrayDeque<>());
                while (!q.isEmpty() && q.peekFirst() <= now - windowMs) q.pollFirst();
                if (q.size() >= maxPerWindow)
                    throw new RateLimitedException(maxPerWindow + " per " + windowMs / 1000 + "s for " + m.userId());
                q.addLast(now);
            }
        }
        return inner.send(m);
    }
}

EventBus.java - Observer

import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;

record DomainEvent(String type, Map<String, String> data) {}

/** Observer: business services publish events; notification rules subscribe to the types they care about. */
interface EventListener {
    void onEvent(DomainEvent e);
}

final class EventBus {
    private final Map<String, List<EventListener>> listeners = new ConcurrentHashMap<>();

    void subscribe(String type, EventListener l) {
        listeners.computeIfAbsent(type, k -> new CopyOnWriteArrayList<>()).add(l);
    }

    void publish(DomainEvent e) {
        for (EventListener l : listeners.getOrDefault(e.type(), List.of())) {
            try {
                l.onEvent(e);
            } catch (RuntimeException ex) {                 // one bad rule must not stop the others
                System.out.println("  [bus] listener failed on " + e.type() + ": " + ex.getMessage());
            }
        }
    }
}

NotificationService.java - the facade, queue and worker

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.ReentrantLock;

/**
 * Facade: accept requests (dedupe, validate), keep them in a time + priority queue, and deliver due ones
 * through one ChannelSender per channel, honouring user preferences. Every outcome goes to history.
 */
final class NotificationService {
    private record Job(String id, NotificationRequest req, long seq) {}

    private static final Comparator<Job> ORDER = Comparator
        .comparingLong((Job j) -> j.req().sendAtMs())        // earliest due first
        .thenComparing(j -> j.req().priority())               // then HIGH before LOW
        .thenComparingLong(Job::seq);                         // then FIFO

    private final Clock clock;
    private final TemplateRegistry templates;
    private final Map<String, User> users = new ConcurrentHashMap<>();
    private final Map<String, Set<Channel>> optedOut = new ConcurrentHashMap<>();
    private final Map<Channel, ChannelSender> senders = new ConcurrentHashMap<>();
    private final Set<String> seenKeys = ConcurrentHashMap.newKeySet();
    private final List<DeliveryRecord> history = Collections.synchronizedList(new ArrayList<>());
    private final PriorityQueue<Job> queue = new PriorityQueue<>(ORDER);
    private final ReentrantLock lock = new ReentrantLock();
    private final Condition changed = lock.newCondition();
    private final AtomicLong seq = new AtomicLong();
    private volatile boolean running;
    private Thread worker;

    NotificationService(Clock clock, TemplateRegistry templates) {
        this.clock = clock;
        this.templates = templates;
    }

    void registerUser(User u) { users.put(u.id(), u); }

    void registerSender(Channel c, ChannelSender s) { senders.put(c, s); }

    void optOut(String userId, Channel c) {
        optedOut.computeIfAbsent(userId, k -> ConcurrentHashMap.newKeySet()).add(c);
    }

    /** Returns the notification id, or empty if this idempotency key was already accepted. */
    Optional<String> submit(NotificationRequest r) {
        if (!users.containsKey(r.userId())) throw new IllegalArgumentException("unknown user " + r.userId());
        if (!templates.exists(r.templateId())) throw new IllegalArgumentException("unknown template " + r.templateId());
        for (Channel c : r.channels())                      // fail fast on missing params, not at 3 a.m.
            if (templates.supports(r.templateId(), c)) templates.render(r.templateId(), c, r.params());
        if (r.idempotencyKey() != null && !seenKeys.add(r.idempotencyKey())) return Optional.empty();
        long s = seq.incrementAndGet();
        Job job = new Job("n" + s, r, s);
        lock.lock();
        try {
            queue.add(job);
            changed.signal();                               // wake the worker: maybe a new earliest job
        } finally {
            lock.unlock();
        }
        return Optional.of(job.id());
    }

    /** Synchronous mode (tests, demos): deliver every job that is due now. */
    int processDue() {
        int n = 0;
        for (Job j; (j = pollDue()) != null; n++) deliver(j);
        return n;
    }

    private Job pollDue() {
        lock.lock();
        try {
            Job head = queue.peek();
            return head != null && head.req().sendAtMs() <= clock.nowMs() ? queue.poll() : null;
        } finally {
            lock.unlock();
        }
    }

    /** Background mode: one worker sleeps until the earliest job is due or a new job arrives. */
    void start() {
        running = true;
        worker = new Thread(() -> {
            while (running) {
                Job j;
                lock.lock();
                try {
                    while (running && (queue.isEmpty() || queue.peek().req().sendAtMs() > clock.nowMs())) {
                        long wait = queue.isEmpty() ? 1000 : queue.peek().req().sendAtMs() - clock.nowMs();
                        changed.await(Math.max(1, wait), TimeUnit.MILLISECONDS);
                    }
                    if (!running) return;
                    j = queue.poll();
                } catch (InterruptedException e) {
                    return;
                } finally {
                    lock.unlock();
                }
                deliver(j);                                  // outside the lock: senders are slow
            }
        }, "notification-worker");
        worker.start();
    }

    void stop() throws InterruptedException {
        running = false;
        lock.lock();
        try { changed.signalAll(); } finally { lock.unlock(); }
        if (worker != null) worker.join(2000);
    }

    int pending() {
        lock.lock();
        try { return queue.size(); } finally { lock.unlock(); }
    }

    private void deliver(Job j) {
        NotificationRequest r = j.req();
        User u = users.get(r.userId());
        for (Channel c : r.channels()) {
            DeliveryStatus status;
            String detail = "";
            String to = u.addressFor(c);
            if (optedOut.getOrDefault(u.id(), Set.of()).contains(c)) {
                status = DeliveryStatus.SKIPPED_OPT_OUT;
            } else if (to == null || !senders.containsKey(c) || !templates.supports(r.templateId(), c)) {
                status = DeliveryStatus.NO_ADDRESS;
                detail = to == null ? "no " + c + " address" : "no sender/template for " + c;
            } else {
                Message m = new Message(j.id(), u.id(), c, to, templates.render(r.templateId(), c, r.params()),
                                        r.priority());
                try {
                    detail = "via " + senders.get(c).send(m);
                    status = DeliveryStatus.SENT;
                } catch (RateLimitedException e) {
                    status = DeliveryStatus.RATE_LIMITED;
                    detail = e.getMessage();
                } catch (DeliveryException e) {
                    status = DeliveryStatus.FAILED;
                    detail = e.getMessage();
                }
            }
            history.add(new DeliveryRecord(j.id(), u.id(), c, status, detail, clock.nowMs()));
        }
    }

    List<DeliveryRecord> historyOf(String userId) {
        synchronized (history) {
            return history.stream().filter(d -> d.userId().equals(userId)).toList();
        }
    }

    List<DeliveryRecord> history() {
        synchronized (history) { return List.copyOf(history); }
    }
}

Main.java - demo driver with assertions

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

/** Demo driver with simple assertions. Build: javac -d out *.java   Run: java -cp out Main */
public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static int printed = 0;

    /** Print history lines added since the last call. */
    static List<DeliveryRecord> newRecords(NotificationService svc) {
        List<DeliveryRecord> all = svc.history();
        List<DeliveryRecord> fresh = all.subList(printed, all.size());
        fresh.forEach(d -> System.out.println("  " + d.userId() + ": " + d));
        printed = all.size();
        return fresh;
    }

    public static void main(String[] args) throws Exception {
        FakeClock clock = new FakeClock(0);
        TemplateRegistry t = new TemplateRegistry()
            .register("order_placed", Channel.EMAIL, "Hi {name}, order {orderId} is confirmed. Total Rs {amount}.")
            .register("order_placed", Channel.SMS, "Order {orderId} confirmed.")
            .register("order_placed", Channel.PUSH, "Order {orderId} confirmed")
            .register("seller_new_order", Channel.IN_APP, "New order {orderId}: please pack it")
            .register("order_shipped", Channel.SMS, "Order {orderId} shipped")
            .register("order_shipped", Channel.PUSH, "Order {orderId} is on its way")
            .register("pickup_request", Channel.IN_APP, "Pick up {orderId} from {seller}")
            .register("rate_order", Channel.EMAIL, "How was order {orderId}?")
            .register("otp", Channel.SMS, "OTP {otp}, valid 5 min")
            .register("promo", Channel.PUSH, "{name}, 20% off today");

        NotificationService svc = new NotificationService(clock, t);
        svc.registerUser(new User("asha", "Asha", "asha@example.com", "+91-90000-00001", "dev-asha"));
        svc.registerUser(new User("ravi", "Ravi", "ravi@example.com", "+91-90000-00002", null));
        svc.registerUser(new User("seller-7", "Kirana 7", null, null, null));
        svc.registerUser(new User("ekart", "Ekart", null, null, null));
        svc.optOut("ravi", Channel.SMS);

        FakeProvider ses = new FakeProvider("ses").failNext(1);              // one transient failure
        FakeProvider smsA = new FakeProvider("sms-A").down(true);            // primary gateway is down
        FakeProvider smsB = new FakeProvider("sms-B");
        FakeProvider fcm = new FakeProvider("fcm"), inbox = new FakeProvider("inbox");
        RetryingSender email = new RetryingSender(ses, 3, 100, Sleeper.NONE);
        ChannelSender sms = new RateLimitingSender(
            new RetryingSender(new FailoverSender(List.of(smsA, smsB)), 3, 100, Sleeper.NONE), 3, 60_000, clock);
        svc.registerSender(Channel.EMAIL, email);
        svc.registerSender(Channel.SMS, sms);
        svc.registerSender(Channel.PUSH, fcm);
        svc.registerSender(Channel.IN_APP, inbox);

        // Observer: who hears about which order event is configuration, not code in OrderService.
        EventBus bus = new EventBus();
        AtomicInteger duplicates = new AtomicInteger();
        java.util.function.Consumer<NotificationRequest> submit =
            r -> { if (svc.submit(r).isEmpty()) duplicates.incrementAndGet(); };
        bus.subscribe("ORDER_PLACED", e -> submit.accept(NotificationRequest.of(
            e.data().get("orderId") + "/placed/customer", e.data().get("customer"), "order_placed", e.data(),
            Priority.MEDIUM, 0, Channel.EMAIL, Channel.SMS, Channel.PUSH)));
        bus.subscribe("ORDER_PLACED", e -> submit.accept(NotificationRequest.of(
            e.data().get("orderId") + "/placed/seller", e.data().get("seller"), "seller_new_order", e.data(),
            Priority.MEDIUM, 0, Channel.IN_APP)));
        bus.subscribe("ORDER_SHIPPED", e -> submit.accept(NotificationRequest.of(
            e.data().get("orderId") + "/shipped/customer", e.data().get("customer"), "order_shipped", e.data(),
            Priority.MEDIUM, 0, Channel.SMS, Channel.PUSH)));
        bus.subscribe("ORDER_SHIPPED", e -> submit.accept(NotificationRequest.of(
            e.data().get("orderId") + "/shipped/logistics", "ekart", "pickup_request", e.data(),
            Priority.MEDIUM, 0, Channel.IN_APP)));
        bus.subscribe("ORDER_DELIVERED", e -> submit.accept(NotificationRequest.of(
            e.data().get("orderId") + "/rate", e.data().get("customer"), "rate_order", e.data(),
            Priority.LOW, clock.nowMs() + 3_600_000, Channel.EMAIL)));          // ask for a rating in 1 hour

        Map<String, String> o101 = Map.of("orderId", "101", "customer", "asha", "name", "Asha",
                                          "seller", "seller-7", "amount", "499");
        System.out.println("== ORDER_PLACED 101 (published twice by an upstream retry) ==");
        bus.publish(new DomainEvent("ORDER_PLACED", o101));
        bus.publish(new DomainEvent("ORDER_PLACED", o101));
        System.out.println("  duplicates suppressed: " + duplicates.get());
        svc.processDue();
        List<DeliveryRecord> r1 = newRecords(svc);
        check(duplicates.get() == 2 && r1.size() == 4, "dedupe: 3 channels + seller, once");
        check(r1.stream().allMatch(d -> d.status() == DeliveryStatus.SENT), "all sent");
        check(email.retries.get() == 1, "email retried once");
        check(smsB.delivered.size() == 1 && smsA.delivered.isEmpty(), "sms failed over to B");

        System.out.println("== ORDER_PLACED 102 for ravi (opted out of SMS, no push device) ==");
        bus.publish(new DomainEvent("ORDER_PLACED", Map.of("orderId", "102", "customer", "ravi", "name", "Ravi",
                                                           "seller", "seller-7", "amount", "120")));
        svc.processDue();
        List<DeliveryRecord> r2 = newRecords(svc);
        check(r2.stream().anyMatch(d -> d.status() == DeliveryStatus.SKIPPED_OPT_OUT), "opt-out honoured");
        check(r2.stream().anyMatch(d -> d.status() == DeliveryStatus.NO_ADDRESS), "no device token");

        System.out.println("== Priority: LOW promo submitted before HIGH OTP, both due now ==");
        svc.submit(NotificationRequest.of(null, "asha", "promo", Map.of("name", "Asha"), Priority.LOW, 0, Channel.PUSH));
        svc.submit(NotificationRequest.of(null, "asha", "otp", Map.of("otp", "4321"), Priority.HIGH, 0, Channel.SMS));
        svc.processDue();
        List<DeliveryRecord> r3 = newRecords(svc);
        check(r3.get(0).channel() == Channel.SMS && r3.get(1).channel() == Channel.PUSH, "HIGH first");

        System.out.println("== Rate limit: 3 non-urgent SMS per minute per user ==");
        bus.publish(new DomainEvent("ORDER_SHIPPED", o101));                  // asha's 2nd non-urgent SMS
        for (String id : List.of("103", "104"))                               // 3rd ok, 4th limited
            svc.submit(NotificationRequest.of(null, "asha", "order_shipped", Map.of("orderId", id),
                                              Priority.MEDIUM, 0, Channel.SMS));
        svc.submit(NotificationRequest.of(null, "asha", "otp", Map.of("otp", "9876"), Priority.HIGH, 0, Channel.SMS));
        svc.processDue();
        List<DeliveryRecord> r4 = newRecords(svc);
        check(r4.stream().filter(d -> d.status() == DeliveryStatus.RATE_LIMITED).count() == 1, "one limited");
        check(r4.get(0).status() == DeliveryStatus.SENT && r4.get(0).detail().equals("via sms-B"), "OTP exempt");

        System.out.println("== Scheduling: ORDER_DELIVERED asks for a rating 1 hour later ==");
        bus.publish(new DomainEvent("ORDER_DELIVERED", o101));
        System.out.println("  now: delivered " + svc.processDue() + ", pending " + svc.pending());
        clock.advance(3_600_000);
        System.out.println("  +1h: delivered " + svc.processDue() + ", pending " + svc.pending());
        List<DeliveryRecord> r5 = newRecords(svc);
        check(r5.size() == 1 && r5.get(0).status() == DeliveryStatus.SENT, "scheduled send");

        System.out.println("== Validation ==");
        try {
            svc.submit(NotificationRequest.of(null, "asha", "order_placed", Map.of("orderId", "105"),
                                              Priority.MEDIUM, 0, Channel.EMAIL));
        } catch (IllegalArgumentException e) {
            System.out.println("  rejected: " + e.getMessage());

"""Notification service LLD: Observer event bus, channel Strategy, Decorator retries / failover / rate limit,
templates, scheduling with a time + priority queue, idempotency, history, background worker."""
from __future__ import annotations

import heapq
import itertools
import re
import threading
import time
from collections import defaultdict, deque
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Dict, FrozenSet, List, Optional

class Channel(Enum):
    EMAIL = "EMAIL"
    SMS = "SMS"
    PUSH = "PUSH"
    IN_APP = "IN_APP"

class Priority(Enum):          # value = queue order
    HIGH = 0
    MEDIUM = 1
    LOW = 2

class Status(Enum):
    SENT = "SENT"
    FAILED = "FAILED"
    SKIPPED_OPT_OUT = "SKIPPED_OPT_OUT"
    NO_ADDRESS = "NO_ADDRESS"
    RATE_LIMITED = "RATE_LIMITED"

@dataclass(frozen=True)
class User:
    id: str
    name: str
    email: Optional[str] = None
    phone: Optional[str] = None
    device_token: Optional[str] = None

    def address_for(self, c: Channel) -> Optional[str]:
        return {Channel.EMAIL: self.email, Channel.SMS: self.phone,
                Channel.PUSH: self.device_token, Channel.IN_APP: self.id}[c]

@dataclass(frozen=True)
class Request:
    key: Optional[str]                     # idempotency key
    user_id: str
    template_id: str
    params: Dict[str, str]
    channels: FrozenSet[Channel]
    priority: Priority = Priority.MEDIUM
    send_at: float = 0.0                   # seconds; 0 = now

@dataclass(frozen=True)
class Message:
    notification_id: str
    user_id: str
    channel: Channel
    to: str
    body: str
    priority: Priority

@dataclass(frozen=True)
class Record:
    notification_id: str
    user_id: str
    channel: Channel
    status: Status
    detail: str = ""

    def __str__(self) -> str:
        return f"{self.notification_id} {self.channel.value} {self.status.value}" + \
            (f" ({self.detail})" if self.detail else "")

class DeliveryError(Exception):
    def __init__(self, msg: str, retryable: bool = True) -> None:
        super().__init__(msg)
        self.retryable = retryable

class RateLimited(DeliveryError):
    def __init__(self, msg: str) -> None:
        super().__init__(msg, retryable=False)

class FakeProvider:
    """Strategy implementation standing in for SES / Twilio / FCM. send() returns the provider name."""

    def __init__(self, name: str, fail_next: int = 0, down: bool = False) -> None:
        self.name, self.fail_next, self.down = name, fail_next, down
        self.delivered: List[str] = []
        self._lock = threading.Lock()

    def send(self, m: Message) -> str:
        with self._lock:
            if self.down:
                raise DeliveryError(f"{self.name} is down")
            if self.fail_next > 0:
                self.fail_next -= 1
                raise DeliveryError(f"{self.name} timeout")
            self.delivered.append(f"{m.to}: {m.body}")
        return self.name

class Retrying:
    """Decorator: retry retryable errors with exponential backoff."""

    def __init__(self, inner, max_attempts: int, base_s: float, sleep: Callable[[float], None] = time.sleep):
        self.inner, self.max_attempts, self.base_s, self.sleep = inner, max_attempts, base_s, sleep
        self.retries = 0

    def send(self, m: Message) -> str:
        attempt = 1
        while True:
            try:
                return self.inner.send(m)
            except DeliveryError as e:
                if not e.retryable or attempt >= self.max_attempts:
                    raise
                self.retries += 1
                self.sleep(self.base_s * 2 ** (attempt - 1))
                attempt += 1

class Failover:
    """Try providers in order; first success wins."""

    def __init__(self, providers) -> None:
        if not providers:
            raise ValueError("need at least one provider")
        self.providers = list(providers)

    def send(self, m: Message) -> str:
        last = None
        for p in self.providers:
            try:
                return p.send(m)
            except DeliveryError as e:
                last = e
        raise DeliveryError(f"all providers failed, last: {last}")

class RateLimiting:
    """Decorator: sliding-window log per user; HIGH priority is exempt."""

    def __init__(self, inner, max_per_window: int, window_s: float, clock: Callable[[], float]) -> None:
        self.inner, self.max, self.window, self.clock = inner, max_per_window, window_s, clock
        self._sent = defaultdict(deque)
        self._lock = threading.Lock()

    def send(self, m: Message) -> str:
        if m.priority is not Priority.HIGH:
            with self._lock:
                now, q = self.clock(), self._sent[m.user_id]
                while q and q[0] <= now - self.window:
                    q.popleft()
                if len(q) >= self.max:
                    raise RateLimited(f"{self.max} per {int(self.window)}s for {m.user_id}")
                q.append(now)
        return self.inner.send(m)

class Templates:
    VAR = re.compile(r"\{(\w+)\}")

    def __init__(self) -> None:
        self._bodies: Dict[str, Dict[Channel, str]] = defaultdict(dict)

    def register(self, tid: str, c: Channel, body: str) -> "Templates":
        self._bodies[tid][c] = body
        return self

    def exists(self, tid: str) -> bool:
        return tid in self._bodies

    def supports(self, tid: str, c: Channel) -> bool:
        return c in self._bodies.get(tid, {})

    def render(self, tid: str, c: Channel, params: Dict[str, str]) -> str:
        def sub(mt):
            if mt.group(1) not in params:
                raise ValueError(f"missing {{{mt.group(1)}}} for {tid}")
            return params[mt.group(1)]
        return self.VAR.sub(sub, self._bodies[tid][c])

class EventBus:
    """Observer: event type -> listeners. A failing listener does not stop the others."""

    def __init__(self) -> None:
        self._listeners: Dict[str, List[Callable[[dict], None]]] = defaultdict(list)

    def subscribe(self, event_type: str, listener: Callable[[dict], None]) -> None:
        self._listeners[event_type].append(listener)

    def publish(self, event_type: str, data: dict) -> None:
        for listener in list(self._listeners[event_type]):
            try:
                listener(data)
            except Exception as e:  # noqa: BLE001
                print(f"  [bus] listener failed on {event_type}: {e}")

@dataclass(order=True)
class _Job:
    send_at: float
    priority: int
    seq: int
    id: str = field(compare=False)
    req: Request = field(compare=False)

class NotificationService:
    def __init__(self, clock: Callable[[], float], templates: Templates) -> None:
        self.clock, self.templates = clock, templates
        self.users: Dict[str, User] = {}
        self.opted_out: Dict[str, set] = defaultdict(set)
        self.senders: Dict[Channel, object] = {}
        self._seen = set()
        self._heap: List[_Job] = []
        self._seq = itertools.count(1)
        self._cv = threading.Condition()       # guards heap, seen-keys and seq
        self.history: List[Record] = []
        self._hist_lock = threading.Lock()
        self._running = False
        self._worker: Optional[threading.Thread] = None

    def register_user(self, u: User) -> None:
        self.users[u.id] = u

    def register_sender(self, c: Channel, s) -> None:
        self.senders[c] = s

    def opt_out(self, user_id: str, c: Channel) -> None:
        self.opted_out[user_id].add(c)

    def submit(self, r: Request) -> Optional[str]:
        if r.user_id not in self.users:
            raise ValueError(f"unknown user {r.user_id}")
        if not self.templates.exists(r.template_id):
            raise ValueError(f"unknown template {r.template_id}")
        for c in sorted(r.channels, key=lambda ch: list(Channel).index(ch)):
            if self.templates.supports(r.template_id, c):
                self.templates.render(r.template_id, c, r.params)     # fail fast
        with self._cv:
            if r.key is not None:
                if r.key in self._seen:
                    return None
                self._seen.add(r.key)
            seq = next(self._seq)
            job = _Job(r.send_at, r.priority.value, seq, f"n{seq}", r)
            heapq.heappush(self._heap, job)
            self._cv.notify()
            return job.id

    def _poll_due(self) -> Optional[_Job]:
        with self._cv:
            if self._heap and self._heap[0].send_at <= self.clock():
                return heapq.heappop(self._heap)
            return None

    def process_due(self) -> int:
        n = 0
        while (job := self._poll_due()) is not None:
            self._deliver(job)
            n += 1
        return n

    def pending(self) -> int:
        with self._cv:
            return len(self._heap)

    def start(self) -> None:
        self._running = True

        def loop():
            while True:
                with self._cv:
                    while self._running and (not self._heap or self._heap[0].send_at > self.clock()):
                        wait = 1.0 if not self._heap else max(0.001, self._heap[0].send_at - self.clock())
                        self._cv.wait(wait)
                    if not self._running:
                        return
                    job = heapq.heappop(self._heap)
                self._deliver(job)                       # outside the lock
        self._worker = threading.Thread(target=loop, name="notification-worker", daemon=True)
        self._worker.start()

    def stop(self) -> None:
        with self._cv:
            self._running = False
            self._cv.notify_all()
        if self._worker:
            self._worker.join(2)

    def _deliver(self, job: _Job) -> None:
        r, u = job.req, self.users[job.req.user_id]
        for c in sorted(r.channels, key=lambda ch: list(Channel).index(ch)):
            to, detail = u.address_for(c), ""
            if c in self.opted_out[u.id]:
                status = Status.SKIPPED_OPT_OUT
            elif to is None or c not in self.senders or not self.templates.supports(r.template_id, c):
                status, detail = Status.NO_ADDRESS, (f"no {c.value} address" if to is None
                                                     else f"no sender/template for {c.value}")
            else:
                msg = Message(job.id, u.id, c, to, self.templates.render(r.template_id, c, r.params), r.priority)
                try:
                    status, detail = Status.SENT, "via " + self.senders[c].send(msg)
                except RateLimited as e:
                    status, detail = Status.RATE_LIMITED, str(e)
                except DeliveryError as e:
                    status, detail = Status.FAILED, str(e)
            with self._hist_lock:
                self.history.append(Record(job.id, u.id, c, status, detail))

    def history_of(self, user_id: str) -> List[Record]:
        with self._hist_lock:
            return [h for h in self.history if h.user_id == user_id]

def request(key, user, template, params, priority=Priority.MEDIUM, send_at=0.0, *channels) -> Request:
    return Request(key, user, template, dict(params), frozenset(channels), priority, send_at)
