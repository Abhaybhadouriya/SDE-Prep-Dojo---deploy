The same design in C++14

One file. A ChannelSender is a struct with one virtual send, the C++ spelling of an ops table; each decorator holds a std::shared_ptr to the sender it wraps (shared so the demo can keep handles to the fake providers and inspect them). The core is single-threaded: processDue() with an injected clock, because MinGW.org g++ 6.3 on this machine has no std::thread. The worker thread would be the same std::condition_variable::wait_until loop as in Java. The demo is shorter (6 checks) but covers dedupe, retry, failover, opt-out, priority, rate limit, scheduling and validation.

// Notification service in C++14 (single-threaded core: synchronous processDue with an injected clock).
// Channel senders are a small virtual interface (like a struct of function pointers); retries, failover
// and rate limiting are decorators that wrap one. Build: g++ -std=c++14 -O2 -Wall -Wextra main.cpp -o notify
#include <cstdio>
#include <cstdlib>
#include <deque>
#include <functional>
#include <map>
#include <memory>
#include <queue>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>

enum class Channel { EMAIL, SMS, PUSH, IN_APP };
enum class Priority { HIGH, MEDIUM, LOW };
enum class Status { SENT, FAILED, SKIPPED_OPT_OUT, NO_ADDRESS, RATE_LIMITED };
static const char* str(Channel c) { static const char* n[] = {"EMAIL", "SMS", "PUSH", "IN_APP"}; return n[int(c)]; }
static const char* str(Status s) {
    static const char* n[] = {"SENT", "FAILED", "SKIPPED_OPT_OUT", "NO_ADDRESS", "RATE_LIMITED"};
    return n[int(s)];
}
using Params = std::map<std::string, std::string>;

struct User {
    std::string id, name, email, phone, device;      // empty = no address
    std::string addressFor(Channel c) const {
        switch (c) {
            case Channel::EMAIL: return email;
            case Channel::SMS: return phone;
            case Channel::PUSH: return device;
            default: return id;
        }
    }
};
struct Request {
    std::string key, userId, templateId;
    Params params;
    std::vector<Channel> channels;
    Priority priority;
    long sendAt;                                      // seconds on the injected clock
};
struct Message { std::string id, userId; Channel channel; std::string to, body; Priority priority; };
struct Record { std::string id, userId; Channel channel; Status status; std::string detail; };

struct DeliveryError : std::runtime_error {
    bool retryable;
    DeliveryError(const std::string& m, bool r = true) : std::runtime_error(m), retryable(r) {}
};
struct RateLimited : DeliveryError { explicit RateLimited(const std::string& m) : DeliveryError(m, false) {} };

struct ChannelSender {                                // Strategy
    virtual ~ChannelSender() = default;
    virtual std::string send(const Message& m) = 0;  // returns the provider that delivered
};
using SenderPtr = std::shared_ptr<ChannelSender>;

struct FakeProvider : ChannelSender {
    std::string name;
    int failNext = 0;
    bool down = false;
    std::vector<std::string> delivered;
    explicit FakeProvider(std::string n) : name(std::move(n)) {}
    std::string send(const Message& m) override {
        if (down) throw DeliveryError(name + " is down");
        if (failNext > 0) { --failNext; throw DeliveryError(name + " timeout"); }
        delivered.push_back(m.to + ": " + m.body);
        return name;
    }
};

struct Retrying : ChannelSender {                     // Decorator
    SenderPtr inner;
    int maxAttempts, retries = 0;
    Retrying(SenderPtr i, int n) : inner(std::move(i)), maxAttempts(n) {}
    std::string send(const Message& m) override {
        for (int attempt = 1;; ++attempt) {
            try { return inner->send(m); }
            catch (const DeliveryError& e) {
                if (!e.retryable || attempt >= maxAttempts) throw;
                ++retries;                            // a real one sleeps base << (attempt-1) here
            }
        }
    }
};

struct Failover : ChannelSender {
    std::vector<SenderPtr> providers;
    explicit Failover(std::vector<SenderPtr> p) : providers(std::move(p)) {}
    std::string send(const Message& m) override {
        std::string last;
        for (auto& p : providers) {
            try { return p->send(m); } catch (const DeliveryError& e) { last = e.what(); }
        }
        throw DeliveryError("all providers failed, last: " + last);
    }
};

struct RateLimiting : ChannelSender {                 // Decorator: sliding-window log per user
    SenderPtr inner;
    size_t maxPerWindow;
    long window;
    std::function<long()> clock;
    std::map<std::string, std::deque<long>> sentAt;
    RateLimiting(SenderPtr i, size_t n, long w, std::function<long()> c)
        : inner(std::move(i)), maxPerWindow(n), window(w), clock(std::move(c)) {}
    std::string send(const Message& m) override {
        if (m.priority != Priority::HIGH) {
            long now = clock();
            auto& q = sentAt[m.userId];
            while (!q.empty() && q.front() <= now - window) q.pop_front();
            if (q.size() >= maxPerWindow)
                throw RateLimited(std::to_string(maxPerWindow) + " per " + std::to_string(window) + "s for " + m.userId);
            q.push_back(now);
        }
        return inner->send(m);
    }
};

class Templates {
public:
    Templates& add(const std::string& id, Channel c, const std::string& body) { t_[id][c] = body; return *this; }
    bool exists(const std::string& id) const { return t_.count(id) > 0; }
    bool supports(const std::string& id, Channel c) const { auto it = t_.find(id); return it != t_.end() && it->second.count(c); }
    std::string render(const std::string& id, Channel c, const Params& p) const {
        const std::string& b = t_.at(id).at(c);
        std::string out;
        for (size_t i = 0; i < b.size(); ++i) {
            if (b[i] != '{') { out += b[i]; continue; }
            size_t j = b.find('}', i);
            std::string var = b.substr(i + 1, j - i - 1);
            auto it = p.find(var);
            if (it == p.end()) throw std::invalid_argument("missing {" + var + "} for " + id);
            out += it->second;
            i = j;
        }
        return out;
    }
private:
    std::map<std::string, std::map<Channel, std::string>> t_;
};

class EventBus {                                      // Observer
public:
    using Listener = std::function<void(const Params&)>;
    void subscribe(const std::string& type, Listener l) { l_[type].push_back(std::move(l)); }
    void publish(const std::string& type, const Params& data) {
        for (auto& l : l_[type]) {
            try { l(data); } catch (const std::exception& e) { std::printf("  [bus] listener failed: %s\n", e.what()); }
        }
    }
private:
    std::map<std::string, std::vector<Listener>> l_;
};

class NotificationService {
public:
    NotificationService(std::function<long()> clock, Templates t) : clock_(std::move(clock)), t_(std::move(t)) {}
    void addUser(const User& u) { users_[u.id] = u; }
    void addSender(Channel c, SenderPtr s) { senders_[c] = std::move(s); }
    void optOut(const std::string& user, Channel c) { optedOut_[user].insert(c); }

    // returns "" when the idempotency key was seen before
    std::string submit(const Request& r) {
        if (!users_.count(r.userId)) throw std::invalid_argument("unknown user " + r.userId);
        if (!t_.exists(r.templateId)) throw std::invalid_argument("unknown template " + r.templateId);
        for (Channel c : r.channels) if (t_.supports(r.templateId, c)) t_.render(r.templateId, c, r.params);
        if (!r.key.empty() && !seen_.insert(r.key).second) return "";
        Job j{"n" + std::to_string(++seq_), r, seq_};
        q_.push(j);
        return j.id;
    }
    int processDue() {
        int n = 0;
        while (!q_.empty() && q_.top().req.sendAt <= clock_()) {
            Job j = q_.top();
            q_.pop();
            deliver(j);
            ++n;
        }
        return n;
    }
    size_t pending() const { return q_.size(); }
    std::vector<Record> history;
private:
    struct Job { std::string id; Request req; long seq; };
    struct Later {                                    // priority_queue is a max-heap: "a after b"
        bool operator()(const Job& a, const Job& b) const {
            if (a.req.sendAt != b.req.sendAt) return a.req.sendAt > b.req.sendAt;
            if (a.req.priority != b.req.priority) return a.req.priority > b.req.priority;
            return a.seq > b.seq;
        }
    };
    void deliver(const Job& j) {
        const User& u = users_[j.req.userId];
        for (Channel c : j.req.channels) {
            Status st;
            std::string detail, to = u.addressFor(c);
            if (optedOut_[u.id].count(c)) st = Status::SKIPPED_OPT_OUT;
            else if (to.empty() || !senders_.count(c) || !t_.supports(j.req.templateId, c)) {
                st = Status::NO_ADDRESS;
                detail = std::string("no ") + str(c) + " address";
            } else {
                Message m{j.id, u.id, c, to, t_.render(j.req.templateId, c, j.req.params), j.req.priority};
                try { detail = "via " + senders_[c]->send(m); st = Status::SENT; }
                catch (const RateLimited& e) { st = Status::RATE_LIMITED; detail = e.what(); }
                catch (const DeliveryError& e) { st = Status::FAILED; detail = e.what(); }
            }
            history.push_back(Record{j.id, u.id, c, st, detail});
        }
    }
    std::function<long()> clock_;
    Templates t_;
    std::map<std::string, User> users_;
    std::map<std::string, std::set<Channel>> optedOut_;
    std::map<Channel, SenderPtr> senders_;
    std::set<std::string> seen_;
    std::priority_queue<Job, std::vector<Job>, Later> q_;
    long seq_ = 0;
};

static int checks = 0;
static void check(bool ok, const char* what) { if (!ok) { std::printf("FAILED: %s\n", what); std::exit(1); } ++checks; }
static size_t printed = 0;
static std::vector<Record> fresh(const NotificationService& s) {
    std::vector<Record> out(s.history.begin() + printed, s.history.end());
    for (auto& r : out)
        std::printf("  %s: %s %s %s%s%s%s\n", r.userId.c_str(), r.id.c_str(), str(r.channel), str(r.status),
                    r.detail.empty() ? "" : " (", r.detail.c_str(), r.detail.empty() ? "" : ")");
    printed = s.history.size();
    return out;
}

int main() {
    long now = 0;
    auto clock = [&now] { return now; };
    Templates t;
    t.add("order_placed", Channel::EMAIL, "Hi {name}, order {orderId} is confirmed. Total Rs {amount}.")
     .add("order_placed", Channel::SMS, "Order {orderId} confirmed.")
     .add("order_placed", Channel::PUSH, "Order {orderId} confirmed")
     .add("seller_new_order", Channel::IN_APP, "New order {orderId}: please pack it")
     .add("otp", Channel::SMS, "OTP {otp}, valid 5 min")
     .add("promo", Channel::PUSH, "{name}, 20% off today")
     .add("rate_order", Channel::EMAIL, "How was order {orderId}?");
    NotificationService svc(clock, t);
    svc.addUser(User{"asha", "Asha", "asha@example.com", "+91-90000-00001", "dev-asha"});
    svc.addUser(User{"ravi", "Ravi", "ravi@example.com", "+91-90000-00002", ""});
    svc.addUser(User{"seller-7", "Kirana 7", "", "", ""});
    svc.optOut("ravi", Channel::SMS);

    auto ses = std::make_shared<FakeProvider>("ses");
    ses->failNext = 1;
    auto smsA = std::make_shared<FakeProvider>("sms-A");
    smsA->down = true;
    auto smsB = std::make_shared<FakeProvider>("sms-B");
    auto email = std::make_shared<Retrying>(ses, 3);
    svc.addSender(Channel::EMAIL, email);
    svc.addSender(Channel::SMS, std::make_shared<RateLimiting>(
        std::make_shared<Retrying>(std::make_shared<Failover>(std::vector<SenderPtr>{smsA, smsB}), 3), 3, 60, clock));
    svc.addSender(Channel::PUSH, std::make_shared<FakeProvider>("fcm"));
    svc.addSender(Channel::IN_APP, std::make_shared<FakeProvider>("inbox"));

    EventBus bus;
    int dupes = 0;
    auto submit = [&](const Request& r) { if (svc.submit(r).empty()) ++dupes; };
    bus.subscribe("ORDER_PLACED", [&](const Params& d) {
        submit(Request{d.at("orderId") + "/placed/customer", d.at("customer"), "order_placed", d,
                       {Channel::EMAIL, Channel::SMS, Channel::PUSH}, Priority::MEDIUM, 0});
    });
    bus.subscribe("ORDER_PLACED", [&](const Params& d) {
        submit(Request{d.at("orderId") + "/placed/seller", d.at("seller"), "seller_new_order", d,
                       {Channel::IN_APP}, Priority::MEDIUM, 0});
    });
    bus.subscribe("ORDER_DELIVERED", [&](const Params& d) {
        submit(Request{d.at("orderId") + "/rate", d.at("customer"), "rate_order", d, {Channel::EMAIL},
                       Priority::LOW, now + 3600});
    });

    Params o101{{"orderId", "101"}, {"customer", "asha"}, {"name", "Asha"}, {"seller", "seller-7"}, {"amount", "499"}};
    std::printf("== ORDER_PLACED 101 (published twice) ==\n");
    bus.publish("ORDER_PLACED", o101);
    bus.publish("ORDER_PLACED", o101);
    std::printf("  duplicates suppressed: %d\n", dupes);
    svc.processDue();
    auto r1 = fresh(svc);
    check(dupes == 2 && r1.size() == 4 && email->retries == 1 && smsB->delivered.size() == 1, "placed");

    std::printf("== ORDER_PLACED 102 for ravi ==\n");
    bus.publish("ORDER_PLACED", Params{{"orderId", "102"}, {"customer", "ravi"}, {"name", "Ravi"},
                                       {"seller", "seller-7"}, {"amount", "120"}});
    svc.processDue();
    auto r2 = fresh(svc);
    check(r2[1].status == Status::SKIPPED_OPT_OUT && r2[2].status == Status::NO_ADDRESS, "prefs");

    std::printf("== Priority and rate limit ==\n");
    svc.submit(Request{"", "asha", "promo", {{"name", "Asha"}}, {Channel::PUSH}, Priority::LOW, 0});
    svc.submit(Request{"", "asha", "otp", {{"otp", "4321"}}, {Channel::SMS}, Priority::HIGH, 0});
    for (int i = 0; i < 3; ++i)
        svc.submit(Request{"", "asha", "order_placed", o101, {Channel::SMS}, Priority::MEDIUM, 0});
    svc.processDue();
    auto r3 = fresh(svc);
    check(r3.front().channel == Channel::SMS && r3.back().channel == Channel::PUSH, "HIGH first, LOW last");
    check(r3[3].status == Status::RATE_LIMITED, "4th non-urgent SMS limited");

    std::printf("== Scheduling ==\n");
    bus.publish("ORDER_DELIVERED", o101);
    int nowCount = svc.processDue();
    std::printf("  now: delivered %d, pending %d\n", nowCount, static_cast<int>(svc.pending()));
    now += 3600;
    int laterCount = svc.processDue();
    std::printf("  +1h: delivered %d, pending %d\n", laterCount, static_cast<int>(svc.pending()));
    check(nowCount == 0 && laterCount == 1, "scheduled");
    fresh(svc);

    try {
        svc.submit(Request{"", "asha", "order_placed", {{"orderId", "105"}}, {Channel::EMAIL}, Priority::MEDIUM, 0});
    } catch (const std::invalid_argument& e) {
        std::printf("  rejected: %s\n", e.what());
        ++checks;
    }
    std::printf("ALL %d CHECKS PASSED\n", checks);
    return 0;
}
