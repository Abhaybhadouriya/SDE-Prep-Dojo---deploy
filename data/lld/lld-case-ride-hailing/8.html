Full code (runs as is)

One file per language. The Java demo includes the rush-hour test: 40 threads released together by a CountDownLatch fight over 5 PREMIUM cabs. The Python test does the same with a Barrier. The C++14 build on our toolchain has no std::thread, so its demo is single-threaded, but the reservation is still an atomic compare-exchange. Source: lld-code/lld-case-ride-hailing/.

Read in this order (the files are long; this is the 10-minute path):

  1. Main.main: the happy path, the receipts observer, the release-on-failure check, then the 40-thread rush hour.
  2. RideService.requestRide: snapshot, rank, CAS claim, release on failure. This is the heart (about 25 lines).
  3. Driver.tryReserve and TripStatus.canMoveTo: the one atomic step and the legal-move table.
  4. Trip.moveTo / complete: state change under the lock, listeners after it.
  5. Then the strategies (NearestDriverFirst, SurgeFare). Skip Location and the enums.

From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import java.util.stream.*;

// ---------- model ----------
enum CabType {
    MINI(4000, 1000, 100), SEDAN(6000, 1400, 150), PREMIUM(10000, 2200, 300);   // paise: base, per km, per minute
    final long base, perKm, perMin;
    CabType(long base, long perKm, long perMin) { this.base = base; this.perKm = perKm; this.perMin = perMin; }
}

final class Location {
    final double x, y;                                   // km on a city grid (real code: lat/lng + haversine)
    Location(double x, double y) { this.x = x; this.y = y; }
    double km(Location o) { return Math.hypot(x - o.x, y - o.y); }
    public String toString() { return "(" + x + "," + y + ")"; }
}

final class Rider {
    final String id, name;
    Rider(String id, String name) { this.id = id; this.name = name; }
}

enum DriverStatus { OFFLINE, AVAILABLE, ON_TRIP }

final class Driver {
    final String id, name; final CabType cab; final double rating;
    private volatile Location location;
    private final AtomicReference<DriverStatus> status = new AtomicReference<>(DriverStatus.AVAILABLE);
    Driver(String id, String name, CabType cab, Location at, double rating) {
        this.id = id; this.name = name; this.cab = cab; this.location = at; this.rating = rating;
    }
    /** The one line that prevents double booking: only one thread can win AVAILABLE -> ON_TRIP. */
    boolean tryReserve() { return status.compareAndSet(DriverStatus.AVAILABLE, DriverStatus.ON_TRIP); }
    void release()       { status.compareAndSet(DriverStatus.ON_TRIP, DriverStatus.AVAILABLE); }
    boolean goOffline()  { return status.compareAndSet(DriverStatus.AVAILABLE, DriverStatus.OFFLINE); }
    boolean goOnline()   { return status.compareAndSet(DriverStatus.OFFLINE, DriverStatus.AVAILABLE); }
    DriverStatus status() { return status.get(); }
    Location location() { return location; }
    void moveTo(Location l) { location = l; }
}

// ---------- trip state machine ----------
enum TripStatus {
    REQUESTED, DRIVER_ASSIGNED, IN_PROGRESS, COMPLETED, CANCELLED;
    boolean canMoveTo(TripStatus next) {
        switch (this) {
            case REQUESTED:       return next == DRIVER_ASSIGNED || next == CANCELLED;
            case DRIVER_ASSIGNED: return next == IN_PROGRESS || next == CANCELLED;
            case IN_PROGRESS:     return next == COMPLETED;
            default:              return false;           // COMPLETED and CANCELLED are terminal
        }
    }
}

interface TripListener { void onChange(Trip t, TripStatus from, TripStatus to); }     // Observer

final class Trip {
    final String id; final Rider rider; final CabType cab; final Location pickup, drop;
    private Driver driver;
    private TripStatus status = TripStatus.REQUESTED;
    private long farePaise;
    private final List<TripListener> listeners;
    Trip(String id, Rider rider, CabType cab, Location pickup, Location drop, List<TripListener> listeners) {
        this.id = id; this.rider = rider; this.cab = cab; this.pickup = pickup; this.drop = drop; this.listeners = listeners;
    }
    /** Caller holds the lock. Checks the transition table and records the move. */
    private TripStatus transition(TripStatus next) {
        if (!status.canMoveTo(next)) throw new IllegalStateException("trip " + id + ": " + status + " -> " + next + " not allowed");
        TripStatus from = status;
        status = next;
        return from;
    }
    /** State changes under the lock; listeners run AFTER it is released (never call out while holding a lock). */
    void moveTo(TripStatus next) {
        TripStatus from;
        synchronized (this) { from = transition(next); }
        fire(from, next);
    }
    void assign(Driver d) {
        TripStatus from;
        synchronized (this) { from = transition(TripStatus.DRIVER_ASSIGNED); driver = d; }
        fire(from, TripStatus.DRIVER_ASSIGNED);
    }
    /** The fare is written in the same critical section as COMPLETED, before anyone is told. */
    void complete(long fare) {
        TripStatus from;
        synchronized (this) { from = transition(TripStatus.COMPLETED); farePaise = fare; }
        fire(from, TripStatus.COMPLETED);
    }
    private void fire(TripStatus from, TripStatus to) { for (TripListener l : listeners) l.onChange(this, from, to); }
    synchronized TripStatus status() { return status; }
    synchronized Driver driver() { return driver; }
    synchronized long fare() { return farePaise; }
}

// ---------- strategies ----------
interface DriverMatchingStrategy { List<Driver> rank(List<Driver> candidates, Location pickup); }

final class NearestDriverFirst implements DriverMatchingStrategy {
    public List<Driver> rank(List<Driver> c, Location p) {
        return c.stream().sorted(Comparator.comparingDouble((Driver d) -> d.location().km(p)).thenComparing(d -> d.id))
                .collect(Collectors.toList());
    }
}

final class BestRatedFirst implements DriverMatchingStrategy {
    public List<Driver> rank(List<Driver> c, Location p) {
        return c.stream().sorted(Comparator.comparingDouble((Driver d) -> -d.rating).thenComparingDouble(d -> d.location().km(p)))
                .collect(Collectors.toList());
    }
}

interface FareStrategy { long fare(CabType cab, double km, long minutes); }

final class StandardFare implements FareStrategy {
    public long fare(CabType c, double km, long min) { return c.base + Math.round(c.perKm * km) + c.perMin * min; }
}

final class SurgeFare implements FareStrategy {                 // Decorator over any other fare rule
    private final FareStrategy inner; private final double multiplier;
    SurgeFare(FareStrategy inner, double multiplier) { this.inner = inner; this.multiplier = multiplier; }
    public long fare(CabType c, double km, long min) { return Math.round(inner.fare(c, km, min) * multiplier); }
}

class NoDriverAvailableException extends RuntimeException {
    private static final long serialVersionUID = 1L;
    NoDriverAvailableException(String m) { super(m); }
}

// ---------- service ----------
final class RideService {
    private final Map<String, Rider> riders = new ConcurrentHashMap<>();
    private final Map<String, Driver> drivers = new ConcurrentHashMap<>();
    private final Map<String, Trip> trips = new ConcurrentHashMap<>();
    private final Set<String> busyRiders = ConcurrentHashMap.newKeySet();   // one active trip per rider
    private final AtomicInteger nextTrip = new AtomicInteger(1);
    private final List<TripListener> listeners = new CopyOnWriteArrayList<>();
    private volatile DriverMatchingStrategy matcher;
    private volatile FareStrategy fareRule;
    private final double radiusKm;

    RideService(DriverMatchingStrategy matcher, FareStrategy fareRule, double radiusKm) {
        this.matcher = matcher; this.fareRule = fareRule; this.radiusKm = radiusKm;
    }
    void addListener(TripListener l) { listeners.add(l); }
    void setMatcher(DriverMatchingStrategy m) { matcher = m; }
    void setFareRule(FareStrategy f) { fareRule = f; }

    void addRider(String id, String name) { riders.put(id, new Rider(id, name)); }
    void addDriver(String id, String name, CabType cab, Location at, double rating) {
        drivers.put(id, new Driver(id, name, cab, at, rating));
    }
    void updateLocation(String driverId, Location at) { driver(driverId).moveTo(at); }
    boolean goOffline(String driverId) { return driver(driverId).goOffline(); }

    Trip requestRide(String riderId, Location pickup, Location drop, CabType cab) {
        Rider r = riders.get(riderId);
        if (r == null) throw new IllegalArgumentException("unknown rider " + riderId);
        if (!busyRiders.add(riderId)) throw new IllegalStateException(riderId + " already has an active trip");
        try {
            List<Driver> nearby = drivers.values().stream()
                    .filter(d -> d.cab == cab && d.status() == DriverStatus.AVAILABLE && d.location().km(pickup) <= radiusKm)
                    .collect(Collectors.toList());                       // a snapshot: may be stale a moment later
            for (Driver d : matcher.rank(nearby, pickup)) {
                if (d.tryReserve()) {                                     // the CAS decides; the snapshot only suggests
                    try {
                        Trip t = new Trip("T" + nextTrip.getAndIncrement(), r, cab, pickup, drop, listeners);
                        t.assign(d);
                        trips.put(t.id, t);
                        return t;
                    } catch (RuntimeException e) {
                        d.release();                                      // release on failure: never leak a claimed driver
                        throw e;
                    }
                }
            }
            throw new NoDriverAvailableException("no " + cab + " within " + radiusKm + " km of " + pickup);
        } catch (RuntimeException e) {
            busyRiders.remove(riderId);
            throw e;
        }
    }

    void startTrip(String tripId) { trip(tripId).moveTo(TripStatus.IN_PROGRESS); }

    long endTrip(String tripId, long minutes) {
        Trip t = trip(tripId);
        t.complete(fareRule.fare(t.cab, t.pickup.km(t.drop), minutes));  // fare first, then COMPLETED, then listeners
        Driver d = t.driver();
        d.moveTo(t.drop);
        d.release();
        busyRiders.remove(t.rider.id);
        return t.fare();
    }

    void cancelTrip(String tripId) {
        Trip t = trip(tripId);
        t.moveTo(TripStatus.CANCELLED);                                   // throws if already started
        if (t.driver() != null) t.driver().release();
        busyRiders.remove(t.rider.id);
    }

    Driver driver(String id) {
        Driver d = drivers.get(id);
        if (d == null) throw new IllegalArgumentException("unknown driver " + id);
        return d;
    }
    Trip trip(String id) {
        Trip t = trips.get(id);
        if (t == null) throw new IllegalArgumentException("unknown trip " + id);
        return t;
    }
}

public class Main {
    static int passed = 0;
    static void check(boolean ok, String what) { if (!ok) throw new AssertionError("FAILED: " + what); passed++; }
    static String rs(long p) { return String.format("%d.%02d", p / 100, p % 100); }

    public static void main(String[] args) throws Exception {
        RideService svc = new RideService(new NearestDriverFirst(), new StandardFare(), 5.0);
        List<String> events = Collections.synchronizedList(new ArrayList<>());
        svc.addListener((t, from, to) -> events.add(t.id + " " + from + " -> " + to));
        List<Long> receipts = Collections.synchronizedList(new ArrayList<>());   // a "receipts" observer
        svc.addListener((t, from, to) -> { if (to == TripStatus.COMPLETED) receipts.add(t.fare()); });

        svc.addDriver("d1", "Arun", CabType.MINI, new Location(1, 1), 4.6);
        svc.addDriver("d2", "Bala", CabType.SEDAN, new Location(2, 2), 4.5);
        svc.addDriver("d3", "Chen", CabType.SEDAN, new Location(3, 3), 4.9);
        for (String r : new String[]{"r1", "r2", "r3", "r4"}) svc.addRider(r, r.toUpperCase());

        Location home = new Location(0, 0), office = new Location(6, 8);
        Trip t1 = svc.requestRide("r1", home, office, CabType.SEDAN);
        System.out.println(t1.id + " assigned " + t1.driver().name + " (nearest SEDAN)");
        check(t1.driver().id.equals("d2"), "nearest sedan is d2");

        Trip t2 = svc.requestRide("r2", home, office, CabType.SEDAN);
        System.out.println(t2.id + " assigned " + t2.driver().name + " (d2 is busy)");
        check(t2.driver().id.equals("d3"), "next sedan is d3");

        try { svc.requestRide("r3", home, office, CabType.SEDAN); check(false, "no sedan left"); }
        catch (NoDriverAvailableException e) { System.out.println("r3: " + e.getMessage()); }

        try { svc.requestRide("r1", home, office, CabType.MINI); check(false, "r1 already riding"); }
        catch (IllegalStateException e) { System.out.println("r1: " + e.getMessage()); }

        svc.startTrip(t1.id);
        long fare = svc.endTrip(t1.id, 25);           // 10 km, 25 min: 60 + 140 + 37.50
        System.out.println(t1.id + " fare " + rs(fare));
        check(fare == 23750, "sedan fare 237.50");
        check(t1.driver().status() == DriverStatus.AVAILABLE, "driver free again");

        svc.cancelTrip(t2.id);
        System.out.println(t2.id + " cancelled; " + t2.driver().name + " is " + t2.driver().status());
        try { svc.startTrip(t2.id); check(false, "cancelled trip cannot start"); }
        catch (IllegalStateException e) { System.out.println("rejected: " + e.getMessage()); }

        svc.setFareRule(new SurgeFare(new StandardFare(), 1.5));
        svc.addDriver("d4", "Dev", CabType.SEDAN, new Location(1, 0), 4.2);   // closer than d3, lower rating
        svc.setMatcher(new BestRatedFirst());
        Trip t3 = svc.requestRide("r3", home, office, CabType.SEDAN);
        System.out.println(t3.id + " assigned " + t3.driver().name + " (best rated)");
        check(t3.driver().id.equals("d3"), "best rated d3 (4.9) beats nearer d4 (4.2)");
        svc.startTrip(t3.id);
        long surge = svc.endTrip(t3.id, 25);
        System.out.println(t3.id + " fare with 1.5x surge " + rs(surge));
        check(surge == 35625, "surge fare 356.25");
        System.out.println("receipts observer saw fares " + receipts);
        check(receipts.equals(List.of(23750L, 35625L)), "a listener sees the final fare on COMPLETED");

        // ---- release on failure: a step after the CAS throws, the driver must not stay ON_TRIP ----
        RideService flaky = new RideService(new NearestDriverFirst(), new StandardFare(), 5.0);
        flaky.addDriver("f1", "Faiz", CabType.MINI, new Location(1, 1), 4.0);
        flaky.addRider("x1", "X1");
        flaky.addListener((t, from, to) -> { if (to == TripStatus.DRIVER_ASSIGNED) throw new IllegalStateException("listener down"); });
        try { flaky.requestRide("x1", home, office, CabType.MINI); check(false, "listener failure surfaces"); }
        catch (IllegalStateException e) { System.out.println("x1: " + e.getMessage() + "; driver f1 is " + flaky.driver("f1").status()); }
        check(flaky.driver("f1").status() == DriverStatus.AVAILABLE, "claimed driver released after a failure");

        // ---- concurrency: 40 riders, 5 premium cabs, all at the same instant ----
        RideService rush = new RideService(new NearestDriverFirst(), new StandardFare(), 50.0);
        for (int i = 1; i <= 5; i++) rush.addDriver("p" + i, "P" + i, CabType.PREMIUM, new Location(i, 0), 4.8);
        for (int i = 1; i <= 40; i++) rush.addRider("u" + i, "U" + i);
        ExecutorService pool = Executors.newFixedThreadPool(16);
        CountDownLatch go = new CountDownLatch(1);
        List<Future<String>> results = new ArrayList<>();
        for (int i = 1; i <= 40; i++) {
            final String rider = "u" + i;
            results.add(pool.submit(() -> {
                go.await();
                try { return rush.requestRide(rider, new Location(0, 0), new Location(9, 9), CabType.PREMIUM).driver().id; }
                catch (NoDriverAvailableException e) { return null; }
            }));
        }
        go.countDown();
        List<String> got = new ArrayList<>();
        for (Future<String> f : results) { String d = f.get(); if (d != null) got.add(d); }
        pool.shutdown();
        System.out.println("rush hour: " + got.size() + " of 40 riders got a cab; drivers " + new TreeSet<>(got));
        check(got.size() == 5, "exactly 5 trips");
        check(new HashSet<>(got).size() == 5, "no driver booked twice");

        System.out.println("events: " + events);
        System.out.println("All " + passed + " checks passed");
    }
}
"""Ride hailing / cab booking - machine-coding solution (in-memory, fares in integer paise)."""
import itertools
import math
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum

class CabType(Enum):
    MINI = (4000, 1000, 100)        # paise: base, per km, per minute
    SEDAN = (6000, 1400, 150)
    PREMIUM = (10000, 2200, 300)

    @property
    def base(self): return self.value[0]
    @property
    def per_km(self): return self.value[1]
    @property
    def per_min(self): return self.value[2]

@dataclass(frozen=True)
class Location:
    x: float
    y: float

    def km(self, o):
        return math.hypot(self.x - o.x, self.y - o.y)

@dataclass(frozen=True)
class Rider:
    id: str
    name: str

class DriverStatus(Enum):
    OFFLINE = "OFFLINE"
    AVAILABLE = "AVAILABLE"
    ON_TRIP = "ON_TRIP"

class Driver:
    def __init__(self, id_, name, cab, at, rating):
        self.id, self.name, self.cab, self.rating = id_, name, cab, rating
        self.location = at
        self._status = DriverStatus.AVAILABLE
        self._lock = threading.Lock()          # Python has no CAS; a tiny per-driver lock does the same job

    def _cas(self, expect, new):
        with self._lock:
            if self._status is not expect:
                return False
            self._status = new
            return True

    def try_reserve(self): return self._cas(DriverStatus.AVAILABLE, DriverStatus.ON_TRIP)
    def release(self): self._cas(DriverStatus.ON_TRIP, DriverStatus.AVAILABLE)
    def go_offline(self): return self._cas(DriverStatus.AVAILABLE, DriverStatus.OFFLINE)

    @property
    def status(self): return self._status

class TripStatus(Enum):
    REQUESTED = "REQUESTED"
    DRIVER_ASSIGNED = "DRIVER_ASSIGNED"
    IN_PROGRESS = "IN_PROGRESS"
    COMPLETED = "COMPLETED"
    CANCELLED = "CANCELLED"

ALLOWED = {
    TripStatus.REQUESTED: {TripStatus.DRIVER_ASSIGNED, TripStatus.CANCELLED},
    TripStatus.DRIVER_ASSIGNED: {TripStatus.IN_PROGRESS, TripStatus.CANCELLED},
    TripStatus.IN_PROGRESS: {TripStatus.COMPLETED},
    TripStatus.COMPLETED: set(),
    TripStatus.CANCELLED: set(),
}

class Trip:
    def __init__(self, id_, rider, cab, pickup, drop, listeners):
        self.id, self.rider, self.cab, self.pickup, self.drop = id_, rider, cab, pickup, drop
        self.driver = None
        self.status = TripStatus.REQUESTED
        self.fare_paise = 0
        self._listeners = listeners
        self._lock = threading.RLock()

    def _transition(self, nxt):                  # caller holds the lock
        if nxt not in ALLOWED[self.status]:
            raise ValueError(f"trip {self.id}: {self.status.name} -> {nxt.name} not allowed")
        frm, self.status = self.status, nxt
        return frm

    def _fire(self, frm, nxt):                   # listeners run after the lock is released
        for listener in self._listeners:
            listener(self, frm, nxt)

    def move_to(self, nxt):
        with self._lock:
            frm = self._transition(nxt)
        self._fire(frm, nxt)

    def assign(self, d):
        with self._lock:
            frm = self._transition(TripStatus.DRIVER_ASSIGNED)
            self.driver = d
        self._fire(frm, TripStatus.DRIVER_ASSIGNED)

    def complete(self, fare_paise):
        """Fare and COMPLETED are set together, before any listener hears about it."""
        with self._lock:
            frm = self._transition(TripStatus.COMPLETED)
            self.fare_paise = fare_paise
        self._fire(frm, TripStatus.COMPLETED)

# ---------- strategies ----------
class MatchingStrategy(ABC):
    @abstractmethod
    def rank(self, candidates, pickup): ...

class NearestDriverFirst(MatchingStrategy):
    def rank(self, candidates, pickup):
        return sorted(candidates, key=lambda d: (d.location.km(pickup), d.id))

class BestRatedFirst(MatchingStrategy):
    def rank(self, candidates, pickup):
        return sorted(candidates, key=lambda d: (-d.rating, d.location.km(pickup)))

class FareStrategy(ABC):
    @abstractmethod
    def fare(self, cab, km, minutes): ...

class StandardFare(FareStrategy):
    def fare(self, cab, km, minutes):
        return cab.base + round(cab.per_km * km) + cab.per_min * minutes

class SurgeFare(FareStrategy):                # decorator over any fare rule
    def __init__(self, inner, multiplier):
        self.inner, self.multiplier = inner, multiplier

    def fare(self, cab, km, minutes):
        return round(self.inner.fare(cab, km, minutes) * self.multiplier)

class NoDriverAvailable(Exception):
    pass

class RideService:
    def __init__(self, matcher, fare_rule, radius_km):
        self.matcher, self.fare_rule, self.radius_km = matcher, fare_rule, radius_km
        self.riders, self.drivers, self.trips = {}, {}, {}
        self.listeners = []
        self._busy = set()
        self._busy_lock = threading.Lock()
        self._ids = itertools.count(1)

    def add_rider(self, id_, name): self.riders[id_] = Rider(id_, name)
    def add_driver(self, id_, name, cab, at, rating): self.drivers[id_] = Driver(id_, name, cab, at, rating)

    def request_ride(self, rider_id, pickup, drop, cab):
        rider = self.riders.get(rider_id)
        if rider is None:
            raise ValueError(f"unknown rider {rider_id}")
        with self._busy_lock:
            if rider_id in self._busy:
                raise ValueError(f"{rider_id} already has an active trip")
            self._busy.add(rider_id)
        try:
            nearby = [d for d in list(self.drivers.values())
                      if d.cab is cab and d.status is DriverStatus.AVAILABLE and d.location.km(pickup) <= self.radius_km]
            for d in self.matcher.rank(nearby, pickup):
                if d.try_reserve():                       # the reservation decides; the snapshot only suggests
                    try:
                        t = Trip(f"T{next(self._ids)}", rider, cab, pickup, drop, self.listeners)
                        t.assign(d)
                        self.trips[t.id] = t
                        return t
                    except Exception:
                        d.release()                       # release on failure: never leak a claimed driver
                        raise
            raise NoDriverAvailable(f"no {cab.name} within {self.radius_km} km")
        except Exception:
            with self._busy_lock:
                self._busy.discard(rider_id)
            raise

    def start_trip(self, trip_id):
        self.trips[trip_id].move_to(TripStatus.IN_PROGRESS)

    def end_trip(self, trip_id, minutes):
        t = self.trips[trip_id]
        t.complete(self.fare_rule.fare(t.cab, t.pickup.km(t.drop), minutes))   # fare first, then COMPLETED
        t.driver.location = t.drop
        t.driver.release()
        self._free(t.rider.id)
        return t.fare_paise

    def cancel_trip(self, trip_id):
        t = self.trips[trip_id]
        t.move_to(TripStatus.CANCELLED)
        if t.driver:
            t.driver.release()
        self._free(t.rider.id)

    def _free(self, rider_id):
        with self._busy_lock:
            self._busy.discard(rider_id)

def rs(p):
    return f"{p // 100}.{p % 100:02d}"

def demo():

// Ride hailing - C++14 version. The demo is single-threaded (this toolchain has no std::thread).
// Driver reservation already uses an atomic compare-exchange (the step that prevents double booking);
// in a threaded build the service maps (drivers_, trips_, busy_) would also need a std::mutex.
#include <algorithm>
#include <atomic>
#include <cmath>
#include <cstdlib>
#include <functional>
#include <iomanip>
#include <iostream>
#include <map>
#include <memory>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>

using std::string; using std::vector;

struct CabSpec { long long base, perKm, perMin; };        // paise
enum class CabType { MINI, SEDAN, PREMIUM };
static CabSpec spec(CabType t) {
    switch (t) { case CabType::MINI: return {4000, 1000, 100}; case CabType::SEDAN: return {6000, 1400, 150}; default: return {10000, 2200, 300}; }
}

struct Location { double x, y; double km(const Location& o) const { return std::hypot(x - o.x, y - o.y); } };

enum DriverStatus { OFFLINE = 0, AVAILABLE = 1, ON_TRIP = 2 };

struct Driver {
    string id, name; CabType cab; Location at; double rating;
    std::atomic<int> status;
    Driver(string i, string n, CabType c, Location l, double r) : id(i), name(n), cab(c), at(l), rating(r), status(AVAILABLE) {}
    bool tryReserve() { int expect = AVAILABLE; return status.compare_exchange_strong(expect, ON_TRIP); }
    void release()    { int expect = ON_TRIP;   status.compare_exchange_strong(expect, AVAILABLE); }
};

enum class TripStatus { REQUESTED, DRIVER_ASSIGNED, IN_PROGRESS, COMPLETED, CANCELLED };
static const char* name(TripStatus s) {
    static const char* n[] = {"REQUESTED", "DRIVER_ASSIGNED", "IN_PROGRESS", "COMPLETED", "CANCELLED"};
    return n[(int)s];
}
static bool canMove(TripStatus a, TripStatus b) {
    switch (a) {
        case TripStatus::REQUESTED:       return b == TripStatus::DRIVER_ASSIGNED || b == TripStatus::CANCELLED;
        case TripStatus::DRIVER_ASSIGNED: return b == TripStatus::IN_PROGRESS || b == TripStatus::CANCELLED;
        case TripStatus::IN_PROGRESS:     return b == TripStatus::COMPLETED;
        default:                          return false;
    }
}

struct Trip;
typedef std::function<void(const Trip&, TripStatus, TripStatus)> Listener;   // Observer

struct Trip {
    string id, rider; CabType cab; Location pickup, drop;
    Driver* driver = nullptr;
    TripStatus status = TripStatus::REQUESTED;
    long long fare = 0;
    const vector<Listener>* listeners = nullptr;
    void moveTo(TripStatus next) {
        if (!canMove(status, next)) throw std::logic_error("trip " + id + ": " + name(status) + " -> " + name(next) + " not allowed");
        TripStatus from = status; status = next;
        for (const auto& l : *listeners) l(*this, from, next);
    }
    // Fare first, then COMPLETED: a listener that hears COMPLETED reads the final fare.
    // (Single-threaded build; the Java version also calls listeners only after releasing the trip lock.)
    void complete(long long finalFare) {
        if (!canMove(status, TripStatus::COMPLETED)) throw std::logic_error("trip " + id + ": cannot complete from " + name(status));
        fare = finalFare;
        moveTo(TripStatus::COMPLETED);
    }
};

// ---------- strategies ----------
struct MatchingStrategy {
    virtual ~MatchingStrategy() {}
    virtual void rank(vector<Driver*>& c, const Location& p) const = 0;
};
struct NearestFirst : MatchingStrategy {
    void rank(vector<Driver*>& c, const Location& p) const override {
        std::sort(c.begin(), c.end(), [&](Driver* a, Driver* b) { return a->at.km(p) < b->at.km(p); });
    }
};
struct BestRatedFirst : MatchingStrategy {
    void rank(vector<Driver*>& c, const Location& p) const override {
        std::sort(c.begin(), c.end(), [&](Driver* a, Driver* b) {
            return a->rating != b->rating ? a->rating > b->rating : a->at.km(p) < b->at.km(p); });
    }
};
struct FareStrategy {
    virtual ~FareStrategy() {}
    virtual long long fare(CabType c, double km, long long min) const = 0;
};
struct StandardFare : FareStrategy {
    long long fare(CabType c, double km, long long min) const override {
        CabSpec s = spec(c); return s.base + std::llround(s.perKm * km) + s.perMin * min;
    }
};
struct SurgeFare : FareStrategy {                          // decorator
    std::shared_ptr<FareStrategy> inner; double mult;
    SurgeFare(std::shared_ptr<FareStrategy> i, double m) : inner(i), mult(m) {}
    long long fare(CabType c, double km, long long min) const override { return std::llround(inner->fare(c, km, min) * mult); }
};

struct NoDriver : std::runtime_error { explicit NoDriver(const string& m) : std::runtime_error(m) {} };

class RideService {
    std::map<string, std::unique_ptr<Driver>> drivers_;
    std::map<string, std::unique_ptr<Trip>> trips_;
    std::map<string, bool> busy_;
    vector<Listener> listeners_;
    std::shared_ptr<MatchingStrategy> matcher_;
    std::shared_ptr<FareStrategy> fare_;
    double radius_;
    int next_ = 1;
public:
    RideService(std::shared_ptr<MatchingStrategy> m, std::shared_ptr<FareStrategy> f, double r) : matcher_(m), fare_(f), radius_(r) {}
    void addListener(Listener l) { listeners_.push_back(l); }
    void setMatcher(std::shared_ptr<MatchingStrategy> m) { matcher_ = m; }
    void setFare(std::shared_ptr<FareStrategy> f) { fare_ = f; }
    void addDriver(const string& id, const string& n, CabType c, Location l, double r) { drivers_[id].reset(new Driver(id, n, c, l, r)); }

    Trip& requestRide(const string& rider, Location pickup, Location drop, CabType cab) {
        if (busy_[rider]) throw std::logic_error(rider + " already has an active trip");
        vector<Driver*> nearby;
        for (auto& kv : drivers_) {
            Driver* d = kv.second.get();
            if (d->cab == cab && d->status.load() == AVAILABLE && d->at.km(pickup) <= radius_) nearby.push_back(d);
        }
        matcher_->rank(nearby, pickup);
        for (Driver* d : nearby) {
            if (!d->tryReserve()) continue;
            try {
                std::unique_ptr<Trip> t(new Trip());
                t->id = "T" + std::to_string(next_++); t->rider = rider; t->cab = cab; t->pickup = pickup; t->drop = drop;
                t->listeners = &listeners_;
                t->driver = d;
                t->moveTo(TripStatus::DRIVER_ASSIGNED);
                busy_[rider] = true;
                Trip& ref = *t;
                trips_[ref.id] = std::move(t);
                return ref;
            } catch (...) {
                d->release();                       // release on failure: never leak a claimed driver
                throw;
            }
        }
        throw NoDriver("no driver of that type nearby");
    }
    void start(const string& id) { trips_.at(id)->moveTo(TripStatus::IN_PROGRESS); }
    long long end(const string& id, long long minutes) {
        Trip& t = *trips_.at(id);
        t.complete(fare_->fare(t.cab, t.pickup.km(t.drop), minutes));
        t.driver->at = t.drop; t.driver->release(); busy_[t.rider] = false;
        return t.fare;
    }
    void cancel(const string& id) {
        Trip& t = *trips_.at(id);
        t.moveTo(TripStatus::CANCELLED);
        if (t.driver) t.driver->release();
        busy_[t.rider] = false;
    }
};

static int passed = 0;
static void check(bool ok, const char* what) { if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); } ++passed; }
static string rs(long long p) { std::ostringstream o; o << p / 100 << '.' << std::setw(2) << std::setfill('0') << p % 100; return o.str(); }

int main() {
    RideService svc(std::make_shared<NearestFirst>(), std::make_shared<StandardFare>(), 5.0);
    vector<string> events;
    svc.addListener([&](const Trip& t, TripStatus a, TripStatus b) { events.push_back(t.id + " " + name(a) + "->" + name(b)); });
    vector<long long> receipts;                                   // a "receipts" observer
    svc.addListener([&](const Trip& t, TripStatus, TripStatus b) { if (b == TripStatus::COMPLETED) receipts.push_back(t.fare); });
    svc.addDriver("d2", "Bala", CabType::SEDAN, {2, 2}, 4.5);
    svc.addDriver("d3", "Chen", CabType::SEDAN, {3, 3}, 4.9);
    Location home{0, 0}, office{6, 8};

    Trip& t1 = svc.requestRide("r1", home, office, CabType::SEDAN);
    std::cout << t1.id << " assigned " << t1.driver->name << "\n";
    check(t1.driver->id == "d2", "nearest is d2");
    Trip& t2 = svc.requestRide("r2", home, office, CabType::SEDAN);
    check(t2.driver->id == "d3", "then d3");
    try { svc.requestRide("r3", home, office, CabType::SEDAN); check(false, "no sedan left"); }
    catch (const NoDriver& e) { std::cout << "r3: " << e.what() << "\n"; }
    check(!t1.driver->tryReserve(), "a reserved driver cannot be reserved again");

    svc.start(t1.id);
    long long f = svc.end(t1.id, 25);
    std::cout << t1.id << " fare " << rs(f) << "\n";
    check(f == 23750, "fare 237.50");

    svc.cancel(t2.id);
    try { svc.start(t2.id); check(false, "cancelled cannot start"); }
    catch (const std::logic_error& e) { std::cout << "rejected: " << e.what() << "\n"; }

    svc.addDriver("d4", "Dev", CabType::SEDAN, {1, 0}, 4.2);

    check(t3.driver->id == "d3", "best rated d3");
    svc.start(t3.id);
    long long s = svc.end(t3.id, 25);
    std::cout << t3.id << " assigned " << t3.driver->name << ", surge fare " << rs(s) << "\n";
    check(s == 35625, "surge 356.25");
    check(receipts.size() == 2 && receipts[0] == 23750 && receipts[1] == 35625, "listener sees the final fare on COMPLETED");
    std::cout << "events:";
    for (const auto& e : events) std::cout << " [" << e << "]";
    std::cout << "\nAll " << passed << " checks passed\n";
}
