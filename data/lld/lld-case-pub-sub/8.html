From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

Partition.java

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.ReentrantLock;

/** Append-only log of one partition. Writers append; readers read by offset and never remove. */
final class Partition {
    private final String topic;
    private final int id;
    private final List<Message> log = new ArrayList<>();
    private final ReentrantLock lock = new ReentrantLock();
    private final Condition grew = lock.newCondition();   // signalled on every append

    Partition(String topic, int id) {
        this.topic = topic;
        this.id = id;
    }

    int id() { return id; }

    Message append(String key, String value) {
        lock.lock();
        try {
            Message m = new Message(topic, id, log.size(), key, value);
            log.add(m);
            grew.signalAll();                              // wake readers waiting at the tail
            return m;
        } finally {
            lock.unlock();
        }
    }

    /** Up to max messages from offset; waits up to timeoutMs when the reader is already at the tail. */
    List<Message> read(long offset, int max, long timeoutMs) throws InterruptedException {
        lock.lock();
        try {
            long nanos = TimeUnit.MILLISECONDS.toNanos(timeoutMs);
            while (log.size() <= offset) {
                if (nanos <= 0) return List.of();
                nanos = grew.awaitNanos(nanos);
            }
            int from = (int) offset;
            int to = (int) Math.min(log.size(), offset + max);
            return new ArrayList<>(log.subList(from, to)); // copy: safe to use after unlock
        } finally {
            lock.unlock();
        }
    }

    long endOffset() {
        lock.lock();
        try {
            return log.size();
        } finally {
            lock.unlock();
        }
    }
}

Partitioner.java

import java.util.concurrent.atomic.AtomicInteger;

/** Strategy: which partition a message goes to. */
interface Partitioner {
    int partitionFor(String key, int numPartitions);
}

/** Same key -> same partition -> per-key order. No key -> round robin. */
final class KeyHashPartitioner implements Partitioner {
    private final AtomicInteger next = new AtomicInteger();

    /** Small stable hash (same result in the Java, Python and C++ versions of this demo). */
    static int stableHash(String s) {
        int h = 0;
        for (char c : s.toCharArray()) h = (h * 31 + c) & 0x7fffffff;
        return h;
    }

    @Override
    public int partitionFor(String key, int numPartitions) {
        if (key == null) return Math.floorMod(next.getAndIncrement(), numPartitions);
        return stableHash(key) % numPartitions;
    }
}

RetryPolicy.java

/** Strategy: how many times to try a delivery and how long to wait between tries. */
interface RetryPolicy {
    int maxAttempts();
    long backoffMs(int failedAttempt);   // 1-based: wait after the first failure, the second, ...
}

final class ExponentialBackoff implements RetryPolicy {
    private final int maxAttempts;
    private final long baseMs;

    ExponentialBackoff(int maxAttempts, long baseMs) {
        if (maxAttempts < 1) throw new IllegalArgumentException("maxAttempts >= 1");
        this.maxAttempts = maxAttempts;
        this.baseMs = baseMs;
    }

    @Override public int maxAttempts() { return maxAttempts; }

    @Override public long backoffMs(int failedAttempt) {
        return baseMs << (failedAttempt - 1);   // base, 2*base, 4*base ...
    }
}

Topic.java

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/** A named set of partitions plus the strategy that picks one per message. */
final class Topic {
    private final String name;
    private final List<Partition> partitions;
    private final Partitioner partitioner;

    Topic(String name, int numPartitions, Partitioner partitioner) {
        if (numPartitions < 1) throw new IllegalArgumentException("partitions >= 1");
        this.name = name;
        this.partitioner = partitioner;
        List<Partition> ps = new ArrayList<>();
        for (int i = 0; i < numPartitions; i++) ps.add(new Partition(name, i));
        this.partitions = Collections.unmodifiableList(ps);
    }

    String name() { return name; }
    List<Partition> partitions() { return partitions; }

    Message publish(String key, String value) {
        int p = partitioner.partitionFor(key, partitions.size());
        return partitions.get(p).append(key, value);
    }
}

ConsumerGroup.java

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;

/**
 * Consumers that share the work of one topic. Each partition is owned by exactly one member and
 * served by one worker thread, so messages of a partition are handled one at a time, in order.
 * Different groups keep their own offsets, so each group sees every message (fan-out).
 */
final class ConsumerGroup {
    private final String groupId;
    private final Topic topic;
    private final RetryPolicy retry;
    private final Broker broker;                                // for dead letters
    private final Map<String, Subscriber> members = new LinkedHashMap<>();
    private final Map<Integer, AtomicLong> nextOffset = new ConcurrentHashMap<>();
    private final Map<Integer, String> owner = new ConcurrentHashMap<>();
    private final List<Thread> workers = new ArrayList<>();
    private volatile boolean running;

    final AtomicInteger delivered = new AtomicInteger();
    final AtomicInteger retries = new AtomicInteger();
    final AtomicInteger deadLettered = new AtomicInteger();

    ConsumerGroup(String groupId, Topic topic, RetryPolicy retry, Broker broker) {
        this.groupId = groupId;
        this.topic = topic;
        this.retry = retry;
        this.broker = broker;
        for (Partition p : topic.partitions()) nextOffset.put(p.id(), new AtomicLong(0));
    }

    String id() { return groupId; }

    synchronized void join(String memberName, Subscriber s) {
        if (running) throw new IllegalStateException("join before start (rebalance is a follow-up)");
        members.put(memberName, s);
    }

    /** Round-robin assignment: partition p -> member p % n. Then one worker thread per partition. */
    synchronized void start() {
        if (running) return;
        if (members.isEmpty()) throw new IllegalStateException("group " + groupId + " has no members");
        running = true;
        List<String> names = new ArrayList<>(members.keySet());
        for (Partition p : topic.partitions()) {
            String name = names.get(p.id() % names.size());
            owner.put(p.id(), name);
            Subscriber s = members.get(name);
            Thread t = new Thread(() -> runWorker(p, s), groupId + "-p" + p.id());
            t.setDaemon(true);
            workers.add(t);
            t.start();
        }
    }

    private void runWorker(Partition p, Subscriber s) {
        AtomicLong next = nextOffset.get(p.id());
        while (running) {
            List<Message> batch;
            try {
                batch = p.read(next.get(), 16, 50);
            } catch (InterruptedException e) {
                return;
            }
            for (Message m : batch) {
                if (!running || next.get() != m.offset()) break;  // stopped, or someone seeked
                deliver(m, s);
                next.compareAndSet(m.offset(), m.offset() + 1);    // commit AFTER processing
            }
        }
    }

    /** At-least-once: retry with backoff; after the last attempt, park it in the DLQ and move on. */
    private void deliver(Message m, Subscriber s) {
        for (int attempt = 1; ; attempt++) {
            try {
                s.onMessage(m);
                delivered.incrementAndGet();
                return;
            } catch (Exception e) {
                if (attempt >= retry.maxAttempts()) {
                    deadLettered.incrementAndGet();
                    broker.deadLetter(groupId, m, e.getMessage());
                    return;
                }
                retries.incrementAndGet();
                try {
                    Thread.sleep(retry.backoffMs(attempt));
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    return;
                }
            }
        }
    }

    /** Replay or skip: move this group's position in one partition. */
    void seek(int partition, long offset) { nextOffset.get(partition).set(offset); }

    long lag() {
        long lag = 0;
        for (Partition p : topic.partitions()) lag += p.endOffset() - nextOffset.get(p.id()).get();
        return lag;
    }

    Map<Integer, String> assignment() { return new java.util.TreeMap<>(owner); }

    void stop() {
        running = false;
        for (Thread t : workers) t.interrupt();
        for (Thread t : workers) {
            try { t.join(1000); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        }
    }
}

Broker.java

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/** Facade the callers use: topics, publish, subscribe, dead letters, shutdown. */
final class Broker {
    private final Map<String, Topic> topics = new ConcurrentHashMap<>();
    private final Map<String, ConsumerGroup> groups = new ConcurrentHashMap<>();

    Topic createTopic(String name, int partitions) {
        return topics.computeIfAbsent(name, n -> new Topic(n, partitions, new KeyHashPartitioner()));
    }

    Topic topic(String name) {
        Topic t = topics.get(name);
        if (t == null) throw new IllegalArgumentException("no topic " + name);
        return t;
    }

    /** Never blocks on consumers: it only appends to a partition log. */
    Message publish(String topic, String key, String value) {
        return topic(topic).publish(key, value);
    }

    ConsumerGroup group(String topic, String groupId, RetryPolicy retry) {
        Topic t = topic(topic);
        return groups.computeIfAbsent(topic + "/" + groupId, k -> new ConsumerGroup(groupId, t, retry, this));
    }

    /** Poison messages go to "<topic>.dlq", keyed by the group that gave up, with the reason. */
    void deadLetter(String groupId, Message m, String reason) {
        createTopic(m.topic() + ".dlq", 1)
            .publish(groupId, m.value() + " (from " + m.topic() + "-" + m.partition() + "@" + m.offset()
                     + ", " + reason + ")");
    }

    void shutdown() {
        groups.values().forEach(ConsumerGroup::stop);
    }
}

import java.util.*;

/** Demo driver with simple assertions. Build: javac -d out *.java   Run: java -cp out Main */
public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static void awaitDrained(ConsumerGroup g, long timeoutMs) throws InterruptedException {
        long end = System.currentTimeMillis() + timeoutMs;
        while (g.lag() > 0 && System.currentTimeMillis() < end) Thread.sleep(5);
        Thread.sleep(20);   // let the last callback finish its bookkeeping
    }

    /** key -> values in the order this group processed them. */
    static Map<String, List<String>> byKey(List<String[]> seen) {
        Map<String, List<String>> m = new TreeMap<>();
        synchronized (seen) {
            for (String[] r : seen) m.computeIfAbsent(r[1], k -> new ArrayList<>()).add(r[2]);
        }
        return m;
    }

    public static void main(String[] args) throws Exception {
        Broker broker = new Broker();
        Topic orders = broker.createTopic("orders", 3);

        // Group 1: billing, two members share the partitions.
        List<String[]> billingSeen = Collections.synchronizedList(new ArrayList<>());
        ConsumerGroup billing = broker.group("orders", "billing", new ExponentialBackoff(3, 5));
        billing.join("billing-A", m -> billingSeen.add(new String[]{"billing-A", m.key(), m.value()}));
        billing.join("billing-B", m -> billingSeen.add(new String[]{"billing-B", m.key(), m.value()}));

        // Group 2: email, one member; fails once on o5 (transient) and always on "poison".
        List<String[]> emailSeen = Collections.synchronizedList(new ArrayList<>());
        Set<String> failedOnce = Collections.synchronizedSet(new HashSet<>());
        ConsumerGroup email = broker.group("orders", "email", new ExponentialBackoff(3, 5));
        email.join("email-1", m -> {
            if (m.value().equals("poison")) throw new IllegalStateException("cannot render template");
            if (m.value().equals("o5") && failedOnce.add("o5")) throw new RuntimeException("SMTP timeout");
            emailSeen.add(new String[]{"email-1", m.key(), m.value()});
        });

        billing.start();
        email.start();

        String[][] events = {{"u1", "o1"}, {"u2", "o2"}, {"u3", "o3"}, {"u1", "o4"}, {"u2", "o5"},
                             {"u3", "o6"}, {"u4", "poison"}, {"u1", "o7"}, {"u2", "o8"}, {"u3", "o9"}};
        Map<String, List<String>> published = new TreeMap<>();
        for (String[] e : events) {
            broker.publish("orders", e[0], e[1]);
            if (!e[1].equals("poison")) published.computeIfAbsent(e[0], k -> new ArrayList<>()).add(e[1]);
        }

        System.out.println("topic orders: " + orders.partitions().size() + " partitions");
        for (Partition p : orders.partitions()) {
            StringBuilder sb = new StringBuilder("  p" + p.id() + ":");
            for (Message m : p.read(0, 100, 0)) sb.append(' ').append(m.key()).append('=').append(m.value());
            System.out.println(sb);
        }

        awaitDrained(billing, 3000);
        awaitDrained(email, 3000);
        System.out.println("billing assignment " + billing.assignment());
        System.out.println("email   assignment " + email.assignment());
        System.out.println("billing: delivered=" + billing.delivered + " retries=" + billing.retries
                           + " dlq=" + billing.deadLettered);
        System.out.println("email:   delivered=" + email.delivered + " retries=" + email.retries
                           + " dlq=" + email.deadLettered);

        Map<String, List<String>> billingOrder = byKey(billingSeen);
        Map<String, List<String>> emailOrder = byKey(emailSeen);
        System.out.println("email per-key order: " + emailOrder);

        published.forEach((k, v) -> check(v.equals(emailOrder.get(k)), "email order for " + k));
        published.forEach((k, v) -> check(v.equals(billingOrder.get(k).subList(0, v.size())),
                                          "billing order for " + k));
        check(billing.delivered.get() == 10, "billing got all 10 (fan-out: own offsets)");
        check(email.delivered.get() == 9 && email.retries.get() == 3, "email: 9 ok, 1 + 2 retries");
        check(email.deadLettered.get() == 1, "one poison message dead-lettered");

        Topic dlq = broker.topic("orders.dlq");
        List<Message> parked = dlq.partitions().get(0).read(0, 10, 0);
        System.out.println("dead letters: " + parked);
        check(parked.size() == 1 && parked.get(0).value().startsWith("poison"), "DLQ content");

        // Replay: billing rewinds partition 1 to offset 0 (e.g. after fixing a bug in billing).
        int before = billing.delivered.get();
        billing.seek(1, 0);
        awaitDrained(billing, 3000);
        System.out.println("billing replayed p1 from 0: +" + (billing.delivered.get() - before)
                           + " messages, lag=" + billing.lag());
        check(billing.delivered.get() - before == 4, "replay of p1 delivers its 4 messages again");

        broker.shutdown();
        System.out.println("ALL " + checks + " CHECKS PASSED");
    }
}

pubsub.py

"""Kafka-lite: topics, partitions, consumer groups with offsets, retries and a dead-letter topic."""
from __future__ import annotations

import itertools
import threading
import time
from dataclasses import dataclass
from typing import Callable, Dict, List, Optional

@dataclass(frozen=True)
class Message:
    topic: str
    partition: int
    offset: int
    key: Optional[str]
    value: str

    def __str__(self) -> str:
        return f"{self.topic}-{self.partition}@{self.offset} {self.key}={self.value}"

class Partition:
    """Append-only log. Readers keep their own offset; nobody removes messages."""

    def __init__(self, topic: str, pid: int) -> None:
        self.topic, self.id = topic, pid
        self._log: List[Message] = []
        self._grew = threading.Condition()          # lock + "log grew" signal

    def append(self, key: Optional[str], value: str) -> Message:
        with self._grew:
            m = Message(self.topic, self.id, len(self._log), key, value)
            self._log.append(m)
            self._grew.notify_all()
            return m

    def read(self, offset: int, max_n: int, timeout_s: float) -> List[Message]:
        with self._grew:
            self._grew.wait_for(lambda: len(self._log) > offset, timeout=timeout_s)
            return self._log[offset:offset + max_n]  # slicing copies

    def end_offset(self) -> int:
        with self._grew:
            return len(self._log)

def stable_hash(s: str) -> int:
    """Same tiny hash as the Java and C++ versions (Python's hash() is randomised per process)."""
    h = 0
    for ch in s:
        h = (h * 31 + ord(ch)) & 0x7FFFFFFF
    return h

class KeyHashPartitioner:
    """Strategy: same key -> same partition; no key -> round robin."""

    def __init__(self) -> None:
        self._rr = itertools.count()

    def partition_for(self, key: Optional[str], n: int) -> int:
        return next(self._rr) % n if key is None else stable_hash(key) % n

class ExponentialBackoff:
    """Strategy: max attempts and the wait after each failed attempt."""

    def __init__(self, max_attempts: int, base_s: float) -> None:
        if max_attempts < 1:
            raise ValueError("max_attempts >= 1")
        self.max_attempts, self.base_s = max_attempts, base_s

    def backoff(self, failed_attempt: int) -> float:
        return self.base_s * (2 ** (failed_attempt - 1))

class Topic:
    def __init__(self, name: str, partitions: int, partitioner: KeyHashPartitioner) -> None:
        if partitions < 1:
            raise ValueError("partitions >= 1")
        self.name = name
        self.partitions = [Partition(name, i) for i in range(partitions)]
        self._partitioner = partitioner

    def publish(self, key: Optional[str], value: str) -> Message:
        p = self._partitioner.partition_for(key, len(self.partitions))
        return self.partitions[p].append(key, value)

Subscriber = Callable[[Message], None]   # Observer: raise to signal failure

class ConsumerGroup:
    """One worker thread per partition; each partition owned by one member -> ordered per partition."""

    def __init__(self, group_id: str, topic: Topic, retry: ExponentialBackoff, broker: "Broker") -> None:
        self.id, self.topic, self.retry, self._broker = group_id, topic, retry, broker
        self._members: Dict[str, Subscriber] = {}
        self._next = {p.id: 0 for p in topic.partitions}
        self._pos_lock = threading.Lock()
        self.owner: Dict[int, str] = {}
        self._workers: List[threading.Thread] = []
        self._running = False
        self._stats_lock = threading.Lock()
        self.delivered = self.retries = self.dead_lettered = 0

    def join(self, name: str, sub: Subscriber) -> None:
        if self._running:
            raise RuntimeError("join before start (rebalance is a follow-up)")
        self._members[name] = sub

    def start(self) -> None:
        if self._running:
            return
        if not self._members:
            raise RuntimeError(f"group {self.id} has no members")
        self._running = True
        names = list(self._members)
        for p in self.topic.partitions:
            name = names[p.id % len(names)]
            self.owner[p.id] = name
            t = threading.Thread(target=self._run, args=(p, self._members[name]),
                                 name=f"{self.id}-p{p.id}", daemon=True)
            self._workers.append(t)
            t.start()

    def _position(self, pid: int) -> int:
        with self._pos_lock:
            return self._next[pid]

    def _commit(self, pid: int, expected: int) -> bool:
        with self._pos_lock:                      # compare-and-set, so a concurrent seek wins
            if self._next[pid] != expected:
                return False
            self._next[pid] = expected + 1
            return True

    def _run(self, p: Partition, sub: Subscriber) -> None:
        while self._running:
            for m in p.read(self._position(p.id), 16, 0.05):
                if not self._running or self._position(p.id) != m.offset:
                    break                             # stopped, or someone seeked
                self._deliver(m, sub)
                self._commit(p.id, m.offset)          # commit AFTER processing: at-least-once

    def _bump(self, field: str) -> None:
        with self._stats_lock:
            setattr(self, field, getattr(self, field) + 1)

    def _deliver(self, m: Message, sub: Subscriber) -> None:
        attempt = 1
        while True:
            try:
                sub(m)
                self._bump("delivered")
                return
            except Exception as e:                   # noqa: BLE001 - any failure means retry
                if attempt >= self.retry.max_attempts:
                    self._bump("dead_lettered")
                    self._broker.dead_letter(self.id, m, str(e))
                    return
                self._bump("retries")
                time.sleep(self.retry.backoff(attempt))
                attempt += 1

    def seek(self, pid: int, offset: int) -> None:
        with self._pos_lock:
            self._next[pid] = offset

    def lag(self) -> int:
        return sum(p.end_offset() - self._position(p.id) for p in self.topic.partitions)

    def stop(self) -> None:
        self._running = False
        for t in self._workers:
            t.join(timeout=1)

class Broker:
    """Facade: create topics, publish, build consumer groups, park dead letters."""

    def __init__(self) -> None:
        self._topics: Dict[str, Topic] = {}
        self._groups: Dict[str, ConsumerGroup] = {}
        self._lock = threading.Lock()

    def create_topic(self, name: str, partitions: int) -> Topic:
        with self._lock:
            if name not in self._topics:
                self._topics[name] = Topic(name, partitions, KeyHashPartitioner())
            return self._topics[name]

    def topic(self, name: str) -> Topic:
        with self._lock:
            if name not in self._topics:
                raise KeyError(f"no topic {name}")
            return self._topics[name]

    def publish(self, topic: str, key: Optional[str], value: str) -> Message:
        return self.topic(topic).publish(key, value)     # never waits for consumers

    def group(self, topic: str, group_id: str, retry: ExponentialBackoff) -> ConsumerGroup:
        t = self.topic(topic)
        with self._lock:
            k = f"{topic}/{group_id}"
            if k not in self._groups:
                self._groups[k] = ConsumerGroup(group_id, t, retry, self)
            return self._groups[k]

    def dead_letter(self, group_id: str, m: Message, reason: str) -> None:
        self.create_topic(m.topic + ".dlq", 1).publish(
            group_id, f"{m.value} (from {m.topic}-{m.partition}@{m.offset}, {reason})")

    def shutdown(self) -> None:
        for g in list(self._groups.values()):
            g.stop()

def await_drained(g: ConsumerGroup, timeout_s: float = 3.0) -> None:
    end = time.time() + timeout_s
    while g.lag() > 0 and time.time() < end:
        time.sleep(0.005)
    time.sleep(0.02)

"""Demo driver: python main.py"""
import threading
from collections import defaultdict

from pubsub import Broker, ExponentialBackoff, await_drained

EVENTS = [("u1", "o1"), ("u2", "o2"), ("u3", "o3"), ("u1", "o4"), ("u2", "o5"),
          ("u3", "o6"), ("u4", "poison"), ("u1", "o7"), ("u2", "o8"), ("u3", "o9")]

def run() -> int:
    checks = 0
    broker = Broker()
    orders = broker.create_topic("orders", 3)

    seen_lock = threading.Lock()
    billing_seen, email_seen = [], []

    billing = broker.group("orders", "billing", ExponentialBackoff(3, 0.005))
    for name in ("billing-A", "billing-B"):
        def on_bill(m, name=name):
            with seen_lock:
                billing_seen.append((name, m.key, m.value))
        billing.join(name, on_bill)

    failed_once = set()
    email = broker.group("orders", "email", ExponentialBackoff(3, 0.005))

    def on_email(m):
        if m.value == "poison":
            raise ValueError("cannot render template")
        if m.value == "o5" and "o5" not in failed_once:
            failed_once.add("o5")
            raise TimeoutError("SMTP timeout")
        with seen_lock:
            email_seen.append(("email-1", m.key, m.value))
    email.join("email-1", on_email)

    billing.start()
    email.start()

    published = defaultdict(list)
    for key, value in EVENTS:
        broker.publish("orders", key, value)
        if value != "poison":
            published[key].append(value)

    print(f"topic orders: {len(orders.partitions)} partitions")
    for p in orders.partitions:
        print(f"  p{p.id}: " + " ".join(f"{m.key}={m.value}" for m in p.read(0, 100, 0)))

    await_drained(billing)
    await_drained(email)
    print("billing assignment", dict(sorted(billing.owner.items())))
    print("email   assignment", dict(sorted(email.owner.items())))
    print(f"billing: delivered={billing.delivered} retries={billing.retries} dlq={billing.dead_lettered}")
    print(f"email:   delivered={email.delivered} retries={email.retries} dlq={email.dead_lettered}")

    def by_key(rows):
        out = defaultdict(list)
        for _, k, v in rows:
            out[k].append(v)
        return dict(sorted(out.items()))

    email_order = by_key(email_seen)
    print("email per-key order:", email_order)
    for k, v in published.items():
        assert email_order[k] == v, f"email order for {k}"
        checks += 1
        assert by_key(billing_seen)[k][:len(v)] == v, f"billing order for {k}"
        checks += 1
    assert billing.delivered == 10
    assert email.delivered == 9 and email.retries == 3 and email.dead_lettered == 1
    checks += 2

    parked = broker.topic("orders.dlq").partitions[0].read(0, 10, 0)
    print("dead letters:", [str(m) for m in parked])
    assert len(parked) == 1 and parked[0].value.startswith("poison")
    checks += 1

    before = billing.delivered
    billing.seek(1, 0)
    await_drained(billing)
    print(f"billing replayed p1 from 0: +{billing.delivered - before} messages, lag={billing.lag()}")
    assert billing.delivered - before == 4
    checks += 1

    broker.shutdown()
    print(f"ALL {checks} CHECKS PASSED")
    return checks

if __name__ == "__main__":
    run()

class PubSubTest(unittest.TestCase):

        self.broker = Broker()
        self.broker.create_topic("t", 4)

    def tearDown(self):
        self.broker.shutdown()

    def test_same_key_same_partition(self):
        p = KeyHashPartitioner()
        self.assertEqual({p.partition_for("user-7", 4) for _ in range(20)}, {p.partition_for("user-7", 4)})

    def test_offsets_are_dense_per_partition(self):
        ms = [self.broker.publish("t", "k", str(i)) for i in range(5)]
        self.assertEqual([m.offset for m in ms], [0, 1, 2, 3, 4])

    def test_each_group_gets_every_message(self):
        got = {"a": [], "b": []}
        lock = threading.Lock()
        for gid in got:
            g = self.broker.group("t", gid, ExponentialBackoff(1, 0))
            g.join(gid, lambda m, gid=gid: (lock.acquire(), got[gid].append(m.value), lock.release()))
            g.start()
        for i in range(20):
            self.broker.publish("t", f"k{i % 3}", str(i))
        for gid in got:
            await_drained(self.broker.group("t", gid, None))
        self.assertEqual(sorted(got["a"], key=int), [str(i) for i in range(20)])
        self.assertEqual(sorted(got["b"], key=int), [str(i) for i in range(20)])

    def test_poison_goes_to_dlq_and_does_not_block(self):
        got = []
        g = self.broker.group("t", "g", ExponentialBackoff(2, 0.001))

        def sub(m):
            if m.value == "bad":
                raise RuntimeError("boom")
            got.append(m.value)
        g.join("only", sub)
        g.start()
        for v in ["1", "bad", "2"]:
            self.broker.publish("t", "same-key", v)
        await_drained(g)
        self.assertEqual(got, ["1", "2"])
        dlq = self.broker.topic("t.dlq").partitions[0].read(0, 10, 0)
        self.assertEqual(len(dlq), 1)
        self.assertEqual(g.retries, 1)

    def test_publisher_never_waits_for_slow_consumer(self):
        g = self.broker.group("t", "slow", ExponentialBackoff(1, 0))
        g.join("s", lambda m: time.sleep(0.2))
        g.start()
        t0 = time.time()
        for i in range(10):
            self.broker.publish("t", "k", str(i))
        self.assertLess(time.time() - t0, 0.1)

// one worker thread per partition, retries with backoff, dead-letter topic.
// Build: g++ -std=c++14 -O2 -Wall -pthread main.cpp -o pubsub && ./pubsub
// (MinGW.org g++ 6.3 uses the win32 thread model and has no std::thread: build in WSL/Linux.)

struct Message {
    std::string topic;
    int partition;
    long offset;
    std::string key, value;
    std::string str() const {
        return topic + "-" + std::to_string(partition) + "@" + std::to_string(offset) + " " + key + "=" + value;
    }
};

class Partition {                                   // like a ring buffer that never overwrites
public:
    Partition(std::string topic, int id) : topic_(std::move(topic)), id_(id) {}
    int id() const { return id_; }
    Message append(const std::string& key, const std::string& value) {
        std::lock_guard<std::mutex> lk(m_);
        Message msg{topic_, id_, static_cast<long>(log_.size()), key, value};
        log_.push_back(msg);
        grew_.notify_all();
        return msg;
    }
    std::vector<Message> read(long offset, size_t max, int timeoutMs) {
        std::unique_lock<std::mutex> lk(m_);
        grew_.wait_for(lk, std::chrono::milliseconds(timeoutMs),
                       [&] { return static_cast<long>(log_.size()) > offset; });
        std::vector<Message> out;
        for (size_t i = offset; i < log_.size() && out.size() < max; ++i) out.push_back(log_[i]);
        return out;                                  // copies: safe after unlock
    }
    long endOffset() {
        std::lock_guard<std::mutex> lk(m_);
        return static_cast<long>(log_.size());
    }
private:
    std::string topic_;
    int id_;
    std::vector<Message> log_;
    std::mutex m_;
    std::condition_variable grew_;
};

static int stableHash(const std::string& s) {      // same as the Java and Python versions
    int h = 0;
    for (char c : s) h = (h * 31 + static_cast<unsigned char>(c)) & 0x7fffffff;
    return h;
}

struct Partitioner {                                // Strategy
    virtual ~Partitioner() = default;
    virtual int partitionFor(const std::string& key, int n) = 0;
};
struct KeyHashPartitioner : Partitioner {
    int partitionFor(const std::string& key, int n) override { return stableHash(key) % n; }
};

struct RetryPolicy {                                // Strategy
    int maxAttempts;
    int baseMs;
    int backoffMs(int failedAttempt) const { return baseMs << (failedAttempt - 1); }
};

class Topic {
public:
    Topic(std::string name, int n) : name_(std::move(name)), partitioner_(new KeyHashPartitioner) {
        for (int i = 0; i < n; ++i) parts_.emplace_back(new Partition(name_, i));
    }
    const std::string& name() const { return name_; }
    std::vector<std::unique_ptr<Partition>>& partitions() { return parts_; }
    Message publish(const std::string& key, const std::string& value) {
        int p = partitioner_->partitionFor(key, static_cast<int>(parts_.size()));
        return parts_[p]->append(key, value);
    }
private:
    std::string name_;
    std::unique_ptr<Partitioner> partitioner_;
    std::vector<std::unique_ptr<Partition>> parts_;
};

using Subscriber = std::function<void(const Message&)>;   // Observer: throw = failure
class Broker;

class ConsumerGroup {
public:
    ConsumerGroup(std::string id, Topic& t, RetryPolicy r, Broker& b)
        : id_(std::move(id)), topic_(t), retry_(r), broker_(b), next_(t.partitions().size()) {
        for (auto& n : next_) n = 0;
    }
    ~ConsumerGroup() { stop(); }
    void join(const std::string& name, Subscriber s) {
        if (running_) throw std::logic_error("join before start");
        names_.push_back(name);
        members_[name] = std::move(s);
    }
    void start() {
        running_ = true;
        for (auto& p : topic_.partitions()) {
            const std::string& name = names_[p->id() % names_.size()];
            owner_[p->id()] = name;
            Partition* part = p.get();
            Subscriber* sub = &members_[name];
            workers_.emplace_back([this, part, sub] { run(*part, *sub); });
        }
    }
    void stop() {
        running_ = false;
        for (auto& t : workers_) if (t.joinable()) t.join();
        workers_.clear();
    }
    void seek(int partition, long offset) { next_[partition] = offset; }
    long lag() {
        long lag = 0;
        for (auto& p : topic_.partitions()) lag += p->endOffset() - next_[p->id()];
        return lag;
    }
    const std::map<int, std::string>& assignment() const { return owner_; }
    std::atomic<int> delivered{0}, retries{0}, deadLettered{0};
private:
    void run(Partition& p, Subscriber& s) {
        while (running_) {
            for (const Message& m : p.read(next_[p.id()], 16, 50)) {
                if (!running_ || next_[p.id()] != m.offset) break;    // stopped or seeked
                deliver(m, s);
                long expected = m.offset;
                next_[p.id()].compare_exchange_strong(expected, m.offset + 1);  // commit after work
            }
        }
    }
    void deliver(const Message& m, Subscriber& s);

    std::string id_;
    Topic& topic_;
    RetryPolicy retry_;
    Broker& broker_;
    std::vector<std::atomic<long>> next_;           // one position per partition
    std::vector<std::string> names_;
    std::map<std::string, Subscriber> members_;
    std::map<int, std::string> owner_;
    std::vector<std::thread> workers_;
    std::atomic<bool> running_{false};
};

class Broker {
public:
    ~Broker() { shutdown(); }
    Topic& createTopic(const std::string& name, int n) {
        std::lock_guard<std::mutex> lk(m_);
        auto it = topics_.find(name);
        if (it == topics_.end()) it = topics_.emplace(name, std::unique_ptr<Topic>(new Topic(name, n))).first;
        return *it->second;
    }
    Topic& topic(const std::string& name) {
        std::lock_guard<std::mutex> lk(m_);
        auto it = topics_.find(name);
        if (it == topics_.end()) throw std::invalid_argument("no topic " + name);
        return *it->second;
    }
    Message publish(const std::string& t, const std::string& k, const std::string& v) {
        return topic(t).publish(k, v);              // never waits on consumers
    }
    ConsumerGroup& group(const std::string& t, const std::string& gid, RetryPolicy r) {
        Topic& top = topic(t);
        std::lock_guard<std::mutex> lk(m_);
        auto& g = groups_[t + "/" + gid];
        if (!g) g.reset(new ConsumerGroup(gid, top, r, *this));
        return *g;
    }
    void deadLetter(const std::string& gid, const Message& m, const std::string& why) {
        createTopic(m.topic + ".dlq", 1).publish(gid, m.value + " (from " + m.topic + "-" +
            std::to_string(m.partition) + "@" + std::to_string(m.offset) + ", " + why + ")");
    }
    void shutdown() { for (auto& g : groups_) g.second->stop(); }
private:
    std::mutex m_;
    std::map<std::string, std::unique_ptr<Topic>> topics_;
    std::map<std::string, std::unique_ptr<ConsumerGroup>> groups_;
};

void ConsumerGroup::deliver(const Message& m, Subscriber& s) {
    for (int attempt = 1;; ++attempt) {
        try {
            s(m);
            ++delivered;
            return;
        } catch (const std::exception& e) {
            if (attempt >= retry_.maxAttempts) {
                ++deadLettered;
                broker_.deadLetter(id_, m, e.what());
                return;
            }
            ++retries;
            std::this_thread::sleep_for(std::chrono::milliseconds(retry_.backoffMs(attempt)));
        }
    }
}
