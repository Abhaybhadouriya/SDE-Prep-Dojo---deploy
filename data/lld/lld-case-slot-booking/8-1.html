"""Doctor appointment / fitness class slot booking - in-memory machine-coding solution."""
from __future__ import annotations

import threading
from abc import ABC, abstractmethod
from collections import deque
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Deque, Dict, List, Optional


class Tier(Enum):
    PLATINUM = 10
    GOLD = 5
    SILVER = 3
    UNLIMITED = 10**9

    @property
    def quota(self) -> int:
        return self.value


class Status(Enum):
    CONFIRMED = "CONFIRMED"
    WAITLISTED = "WAITLISTED"
    CANCELLED = "CANCELLED"


class BookingError(Exception):
    pass


def parse(hhmm: str) -> int:
    h, m = (int(x) for x in hhmm.strip().split(":"))
    if not (0 <= h <= 24 and 0 <= m <= 59):
        raise BookingError(f"bad time {hhmm}")
    return h * 60 + m


def fmt(minutes: int) -> str:
    return f"{minutes // 60:02d}:{minutes % 60:02d}"


@dataclass(frozen=True)
class TimeRange:
    start: int  # minutes since midnight, [start, end)
    end: int

    def __post_init__(self) -> None:
        if self.start < 0 or self.end > 24 * 60 or self.start >= self.end:
            raise BookingError("bad time range")

    def overlaps(self, other: "TimeRange") -> bool:
        return self.start < other.end and other.start < self.end

    def __str__(self) -> str:
        return f"{fmt(self.start)}-{fmt(self.end)}"


@dataclass(frozen=True)
class Provider:  # a doctor, or a gym centre
    name: str
    category: str  # speciality or workout type
    rating: float


@dataclass(frozen=True)
class User:
    name: str
    tier: Tier


@dataclass(eq=False)
class Slot:
    id: str
    provider: Provider
    time: TimeRange
    capacity: int
    waitlist_limit: int
    confirmed: Dict[str, None] = field(default_factory=dict)  # ordered set of booking ids
    waitlist: Deque[str] = field(default_factory=deque)
    cancelled: bool = False

    def free(self) -> int:
        return self.capacity - len(self.confirmed)

    def label(self) -> str:
        return f"{self.provider.name} {self.time}"


@dataclass(eq=False)
class Booking:
    id: str
    user: str
    slot: Slot
    status: Status


# ---------- rules (chain of checks) ----------
class BookingRule(ABC):
    @abstractmethod
    def check(self, user: User, slot: Slot, others: List[Booking]) -> None:
        """Raise BookingError if the user may not hold this slot."""


class NoDuplicateRule(BookingRule):
    def check(self, user, slot, others):
        if any(b.slot is slot for b in others):
            raise BookingError(f"{user.name} already holds or waits for {slot.label()}")


class NoOverlapRule(BookingRule):
    def check(self, user, slot, others):
        for b in others:
            if b.status is Status.CONFIRMED and b.slot.time.overlaps(slot.time):
                raise BookingError(f"{user.name} already has {b.slot.label()} at that time")


class TierQuotaRule(BookingRule):
    def check(self, user, slot, others):
        used = sum(1 for b in others if b.status is Status.CONFIRMED)
        if used >= user.tier.quota:
            raise BookingError(f"{user.tier.name} tier allows {user.tier.quota} bookings")


def default_rules() -> List[BookingRule]:
    return [NoDuplicateRule(), NoOverlapRule(), TierQuotaRule()]


# ---------- ranking strategies ----------
RankingStrategy = Callable[[List[Slot]], List[Slot]]


def by_start_time(slots: List[Slot]) -> List[Slot]:
    return sorted(slots, key=lambda s: (s.time.start, s.provider.name))


def by_rating(slots: List[Slot]) -> List[Slot]:
    return sorted(slots, key=lambda s: (-s.provider.rating, s.time.start))


Listener = Callable[[str, str], None]


# ---------- service (facade) ----------
class BookingService:
    def __init__(self, slot_minutes: int, cancel_cutoff: int, clock: Callable[[], int],
                 rules: Optional[List[BookingRule]] = None) -> None:
        self._slot_minutes = slot_minutes
        self._cutoff = cancel_cutoff
        self._clock = clock
        self._rules = rules if rules is not None else default_rules()
        self._providers: Dict[str, Provider] = {}
        self._users: Dict[str, User] = {}
        self._slots: Dict[str, Slot] = {}
        self._bookings: Dict[str, Booking] = {}
        self._by_user: Dict[str, List[Booking]] = {}
        self._per_provider: Dict[str, int] = {}
        self._listeners: List[Listener] = []
        self._lock = threading.Lock()
        self._next_slot = 1
        self._next_booking = 1

    def add_listener(self, listener: Listener) -> None:
        self._listeners.append(listener)

    def register_provider(self, name: str, category: str, rating: float) -> None:
        with self._lock:
            if name in self._providers:
                raise BookingError(f"{name} already registered")
            self._providers[name] = Provider(name, category, rating)

    def register_user(self, name: str, tier: Tier) -> None:
        with self._lock:
            if name in self._users:
                raise BookingError(f"{name} already registered")
            self._users[name] = User(name, tier)
            self._by_user[name] = []

    def add_slot(self, provider: str, start: str, end: str, capacity: int, waitlist_limit: int) -> str:
        with self._lock:
            p = self._provider(provider)
            t = TimeRange(parse(start), parse(end))
            if t.end - t.start != self._slot_minutes:
                raise BookingError(f"slots are {self._slot_minutes} mins only")
            if capacity < 1 or waitlist_limit < 0:
                raise BookingError("bad capacity")
            for s in self._slots.values():
                if s.provider is p and not s.cancelled and s.time.overlaps(t):
                    raise BookingError(f"{p.name} already has {s.time}")
            sid = f"S{self._next_slot}"
            self._next_slot += 1
            self._slots[sid] = Slot(sid, p, t, capacity, waitlist_limit)
            return sid

    def available(self, category: str, ranking: RankingStrategy = by_start_time) -> List[Slot]:
        with self._lock:
            now = self._clock()
            found = [s for s in self._slots.values()
                     if not s.cancelled and s.free() > 0 and s.time.start > now
                     and s.provider.category == category]
            return ranking(found)

    def book(self, user_name: str, slot_id: str) -> Booking:
        events = []
        with self._lock:
            user = self._user(user_name)
            slot = self._slot(slot_id)
            if slot.cancelled:
                raise BookingError(f"{slot.label()} is cancelled")
            if slot.time.start <= self._clock():
                raise BookingError(f"{slot.label()} has started")
            for rule in self._rules:  # validate first, mutate after
                rule.check(user, slot, self._active(user_name))
            if slot.free() > 0:
                b = self._new_booking(user, slot, Status.CONFIRMED)
                slot.confirmed[b.id] = None
                self._count(slot, +1)
            elif len(slot.waitlist) < slot.waitlist_limit:
                b = self._new_booking(user, slot, Status.WAITLISTED)
                slot.waitlist.append(b.id)
                events.append((user.name, f"waitlisted #{len(slot.waitlist)} for {slot.label()}"))
            else:
                raise BookingError(f"{slot.label()} and its waitlist are full")
        self._fire(events)  # slow I/O outside the lock
        return b

    def cancel(self, booking_id: str) -> None:
        events = []
        with self._lock:
            b = self._bookings.get(booking_id)
            if b is None:
                raise BookingError(f"no booking {booking_id}")
            if b.status is Status.CANCELLED:
                raise BookingError(f"{booking_id} already cancelled")
            slot = b.slot
            if b.status is Status.WAITLISTED:
                slot.waitlist.remove(b.id)
                b.status = Status.CANCELLED
                return
            if self._clock() > slot.time.start - self._cutoff:
                raise BookingError(f"too late: cancel at least {self._cutoff} min before start")
            del slot.confirmed[b.id]
            b.status = Status.CANCELLED
            self._count(slot, -1)
            self._promote(slot, events)
        self._fire(events)

    def cancel_slot(self, slot_id: str) -> None:
        events = []
        with self._lock:
            slot = self._slot(slot_id)
            if slot.cancelled:
                return
            slot.cancelled = True
            for bid in list(slot.confirmed) + list(slot.waitlist):
                b = self._bookings[bid]
                if b.status is Status.CONFIRMED:
                    self._count(slot, -1)
                b.status = Status.CANCELLED
                events.append((b.user, f"{slot.label()} was cancelled by the provider"))
            slot.confirmed.clear()
            slot.waitlist.clear()
        self._fire(events)

    def bookings_of(self, user_name: str) -> List[str]:
        with self._lock:
            self._user(user_name)
            return [f"{b.id} {b.slot.label()} {b.status.value}" for b in self._active(user_name)]

    def trending(self) -> str:
        with self._lock:
            best, best_n = None, 0
            for name in sorted(self._per_provider):
                if self._per_provider[name] > best_n:
                    best, best_n = name, self._per_provider[name]
            return "none" if best is None else f"{best} ({best_n})"

    def status_of(self, booking_id: str) -> Status:
        with self._lock:
            return self._bookings[booking_id].status

    def counts(self, slot_id: str) -> tuple:
        with self._lock:
            s = self._slot(slot_id)
            return len(s.confirmed), len(s.waitlist)

    # ----- helpers (lock held) -----
    def _promote(self, slot: Slot, events: list) -> None:
        while slot.free() > 0 and slot.waitlist:
            nxt = self._bookings[slot.waitlist.popleft()]
            user = self._users[nxt.user]
            others = [b for b in self._active(user.name) if b is not nxt]
            try:
                for rule in self._rules:  # the user's day may have changed since joining
                    rule.check(user, slot, others)
            except BookingError as e:
                nxt.status = Status.CANCELLED
                events.append((user.name, f"dropped from waitlist of {slot.label()}: {e}"))
                continue
            nxt.status = Status.CONFIRMED
            slot.confirmed[nxt.id] = None
            self._count(slot, +1)
            events.append((user.name, f"promoted from waitlist: {slot.label()}"))

    def _count(self, slot: Slot, delta: int) -> None:
        name = slot.provider.name
        self._per_provider[name] = self._per_provider.get(name, 0) + delta

    def _new_booking(self, user: User, slot: Slot, status: Status) -> Booking:
        b = Booking(f"B{self._next_booking}", user.name, slot, status)
        self._next_booking += 1
        self._bookings[b.id] = b
        self._by_user[user.name].append(b)
        return b

    def _active(self, user_name: str) -> List[Booking]:
        return [b for b in self._by_user[user_name] if b.status is not Status.CANCELLED]

    def _fire(self, events: list) -> None:
        for user, msg in events:
            for listener in self._listeners:
                listener(user, msg)

    def _provider(self, name: str) -> Provider:
        if name not in self._providers:
            raise BookingError(f"unknown provider {name}")
        return self._providers[name]

    def _user(self, name: str) -> User:
        if name not in self._users:
            raise BookingError(f"unknown user {name}")
        return self._users[name]

    def _slot(self, sid: str) -> Slot:
        if sid not in self._slots:
            raise BookingError(f"unknown slot {sid}")
        return self._slots[sid]

            self.svc.add_slot("Dr.Curious", "09:30", "10:30", 1, 1)  # 60 min
        with self.assertRaises(BookingError):
            self.svc.add_slot("Dr.Curious", "12:45", "13:15", 1, 1)  # overlaps

    def test_waitlist_promotion_on_cancel(self):
        a = self.svc.book("A", self.c)
        b = self.svc.book("B", self.c)
        self.assertEqual(b.status, Status.WAITLISTED)
        self.svc.cancel(a.id)
        self.assertEqual(self.svc.status_of(b.id), Status.CONFIRMED)
        self.assertIn("B: promoted from waitlist: Dr.Curious 12:30-13:00", self.inbox)

    def test_no_overlap_and_no_duplicate(self):
        self.svc.book("A", self.c)
        with self.assertRaises(BookingError):
            self.svc.book("A", self.d)  # same time, other doctor
        with self.assertRaises(BookingError):
            self.svc.book("A", self.c)  # same slot twice

    def test_promotion_skips_user_who_now_overlaps(self):
        a = self.svc.book("A", self.c)
        b_wait = self.svc.book("B", self.c)  # B waits for Curious 12:30
        self.svc.book("B", self.d)  # ...and meanwhile books Daring 12:30
        c_wait = self.svc.book("C", self.c)
        self.svc.cancel(a.id)
        self.assertEqual(self.svc.status_of(b_wait.id), Status.CANCELLED)
        self.assertEqual(self.svc.status_of(c_wait.id), Status.CONFIRMED)

    def test_ranking_strategy(self):
        names = [s.provider.name for s in self.svc.available("Dermatologist", by_rating)]
        self.assertEqual(names, ["Dr.Daring", "Dr.Dreadful"])

    def test_tier_quota_and_cutoff(self):
        now = [5 * 60]
        gym = BookingService(60, 30, lambda: now[0])
        gym.register_provider("Yoga", "Yoga", 4.5)
        ids = [gym.add_slot("Yoga", f"{h:02d}:00", f"{h + 1:02d}:00", 2, 1) for h in (6, 7, 8, 18)]
        gym.register_user("S", Tier.SILVER)
        bookings = [gym.book("S", sid) for sid in ids[:3]]
        with self.assertRaises(BookingError):
            gym.book("S", ids[3])  # 4th > SILVER quota
        gym.cancel(bookings[2].id)  # quota restored
        gym.book("S", ids[3])
        now[0] = 5 * 60 + 45
        with self.assertRaises(BookingError):
            gym.cancel(bookings[0].id)  # 15 min before start

    def test_concurrent_booking_never_overbooks(self):
        svc = BookingService(60, 30, lambda: 0)
        svc.register_provider("HIIT", "HIIT", 4.0)
        sid = svc.add_slot("HIIT", "07:00", "08:00", 10, 5)
        for i in range(50):
            svc.register_user(f"u{i}", Tier.GOLD)
        rejected = []
        go = threading.Event()

        def worker(name):
            go.wait()
            try:
                svc.book(name, sid)
            except BookingError:
                rejected.append(name)

        threads = [threading.Thread(target=worker, args=(f"u{i}",)) for i in range(50)]
        for t in threads:
            t.start()
        go.set()
        for t in threads:
            t.join()
        self.assertEqual(svc.counts(sid), (10, 5))
        self.assertEqual(len(rejected), 35)

// Doctor appointment / fitness class slot booking - C++14, single file.
// Locking: every public BookingService method takes one lock on the service (std::lock_guard), exactly like
// the Java synchronized methods. Listeners are called after the unlock (events are collected under the lock).
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run-cpp.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <algorithm>
#include <cstdio>
#include <deque>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>

#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif

enum class Tier { PLATINUM, GOLD, SILVER, UNLIMITED };
static int quota(Tier t) {
    switch (t) { case Tier::PLATINUM: return 10; case Tier::GOLD: return 5; case Tier::SILVER: return 3; default: return 1 << 30; }
}
static const char* tierName(Tier t) {
    switch (t) { case Tier::PLATINUM: return "PLATINUM"; case Tier::GOLD: return "GOLD"; case Tier::SILVER: return "SILVER"; default: return "UNLIMITED"; }
}
enum class Status { CONFIRMED, WAITLISTED, CANCELLED };

struct BookingError : std::runtime_error { explicit BookingError(const std::string& m) : std::runtime_error(m) {} };

static int parseTime(const std::string& s) {
    int h = 0, m = 0;
    if (std::sscanf(s.c_str(), "%d:%d", &h, &m) != 2 || h < 0 || h > 24 || m < 0 || m > 59) throw BookingError("bad time " + s);
    return h * 60 + m;
}
static std::string fmt(int t) { char b[16]; std::snprintf(b, sizeof b, "%02d:%02d", t / 60, t % 60); return b; }

struct TimeRange {
    int start, end;  // minutes since midnight, [start, end)
    bool overlaps(const TimeRange& o) const { return start < o.end && o.start < end; }
    std::string str() const { return fmt(start) + "-" + fmt(end); }
};

struct Provider { std::string name, category; double rating; };
struct User { std::string name; Tier tier; };

struct Slot {
    std::string id; const Provider* provider; TimeRange time; int capacity, waitlistLimit;
    std::vector<std::string> confirmed;   // booking ids
    std::deque<std::string> waitlist;     // booking ids, FIFO
    bool cancelled = false;
    int freeSeats() const { return capacity - (int)confirmed.size(); }
    std::string label() const { return provider->name + " " + time.str(); }
};

struct Booking { std::string id, user; Slot* slot; Status status; };

// ----- rules: a chain of checks, each may veto -----
struct BookingRule {
    virtual ~BookingRule() {}
    virtual void check(const User& u, const Slot& s, const std::vector<Booking*>& others) const = 0;
};
struct NoDuplicateRule : BookingRule {
    void check(const User& u, const Slot& s, const std::vector<Booking*>& others) const override {
        for (auto b : others) if (b->slot == &s) throw BookingError(u.name + " already holds or waits for " + s.label());
    }
};
struct NoOverlapRule : BookingRule {
    void check(const User& u, const Slot& s, const std::vector<Booking*>& others) const override {
        for (auto b : others)
            if (b->status == Status::CONFIRMED && b->slot->time.overlaps(s.time))
                throw BookingError(u.name + " already has " + b->slot->label() + " at that time");
    }
};
struct TierQuotaRule : BookingRule {
    void check(const User& u, const Slot&, const std::vector<Booking*>& others) const override {
        int used = (int)std::count_if(others.begin(), others.end(), [](Booking* b) { return b->status == Status::CONFIRMED; });
        if (used >= quota(u.tier)) throw BookingError(std::string(tierName(u.tier)) + " tier allows " + std::to_string(quota(u.tier)) + " bookings");
    }
};

// ----- ranking strategies -----
using Ranking = std::function<bool(const Slot*, const Slot*)>;
static const Ranking byStartTime = [](const Slot* a, const Slot* b) {
    return a->time.start != b->time.start ? a->time.start < b->time.start : a->provider->name < b->provider->name;
};
static const Ranking byRating = [](const Slot* a, const Slot* b) {
    return a->provider->rating != b->provider->rating ? a->provider->rating > b->provider->rating : a->time.start < b->time.start;
};

using Listener = std::function<void(const std::string&, const std::string&)>;

class BookingService {
public:
    BookingService(int slotMinutes, int cutoff, std::function<int()> clock)
        : slotMinutes_(slotMinutes), cutoff_(cutoff), clock_(clock) {
        rules_.emplace_back(new NoDuplicateRule());
        rules_.emplace_back(new NoOverlapRule());
        rules_.emplace_back(new TierQuotaRule());
    }
    void addListener(Listener l) { listeners_.push_back(l); }

    void registerProvider(const std::string& n, const std::string& cat, double rating) {
        std::lock_guard<Mutex> g(mu_);
        if (providers_.count(n)) throw BookingError(n + " already registered");
        providers_[n] = Provider{n, cat, rating};
    }
    void registerUser(const std::string& n, Tier t) {
        std::lock_guard<Mutex> g(mu_);
        if (users_.count(n)) throw BookingError(n + " already registered");
        users_[n] = User{n, t};
        byUser_[n];
    }
    std::string addSlot(const std::string& prov, const std::string& from, const std::string& to, int cap, int wl) {
        std::lock_guard<Mutex> g(mu_);
        const Provider& p = provider(prov);
        TimeRange t{parseTime(from), parseTime(to)};
        if (t.start >= t.end) throw BookingError("bad time range");
        if (t.end - t.start != slotMinutes_) throw BookingError("slots are " + std::to_string(slotMinutes_) + " mins only");
        for (auto& kv : slots_)
            if (kv.second->provider == &p && !kv.second->cancelled && kv.second->time.overlaps(t))
                throw BookingError(p.name + " already has " + kv.second->time.str());
        std::string id = "S" + std::to_string(nextSlot_++);
        slots_[id].reset(new Slot{id, &p, t, cap, wl, {}, {}, false});
        order_.push_back(id);
        return id;
    }
    std::vector<const Slot*> available(const std::string& cat, const Ranking& rank) const {
        std::lock_guard<Mutex> g(mu_);
        std::vector<const Slot*> r;
        for (auto& id : order_) {
            const Slot* s = slots_.at(id).get();
            if (!s->cancelled && s->freeSeats() > 0 && s->time.start > clock_() && s->provider->category == cat) r.push_back(s);
        }
        std::stable_sort(r.begin(), r.end(), rank);
        return r;
    }
    Booking& book(const std::string& userName, const std::string& slotId) {
        std::vector<std::pair<std::string, std::string>> events;
        Booking* b;
        {
            std::lock_guard<Mutex> g(mu_);   // released at the closing brace, also when a rule throws
            User& u = user(userName);
            Slot& s = slot(slotId);
            if (s.cancelled) throw BookingError(s.label() + " is cancelled");
            if (s.time.start <= clock_()) throw BookingError(s.label() + " has started");
            for (auto& r : rules_) r->check(u, s, active(userName));  // validate first
            if (s.freeSeats() > 0) {
                b = &newBooking(u, s, Status::CONFIRMED);
                s.confirmed.push_back(b->id);
                perProvider_[s.provider->name]++;
            } else if ((int)s.waitlist.size() < s.waitlistLimit) {
                b = &newBooking(u, s, Status::WAITLISTED);
                s.waitlist.push_back(b->id);
                events.emplace_back(u.name, "waitlisted #" + std::to_string(s.waitlist.size()) + " for " + s.label());
            } else {
                throw BookingError(s.label() + " and its waitlist are full");
            }
        }
        fire(events);  // after the unlock: a slow listener never blocks other bookings
        return *b;
    }
    void cancel(const std::string& id) {
        std::vector<std::pair<std::string, std::string>> events;
        {
            std::lock_guard<Mutex> g(mu_);
            auto it = bookings_.find(id);
            if (it == bookings_.end()) throw BookingError("no booking " + id);
            Booking& b = *it->second;
            if (b.status == Status::CANCELLED) throw BookingError(id + " already cancelled");
            Slot& s = *b.slot;
            if (b.status == Status::WAITLISTED) {
                s.waitlist.erase(std::find(s.waitlist.begin(), s.waitlist.end(), id));
                b.status = Status::CANCELLED;
                return;
            }
            if (clock_() > s.time.start - cutoff_) throw BookingError("too late: cancel at least " + std::to_string(cutoff_) + " min before start");
            s.confirmed.erase(std::find(s.confirmed.begin(), s.confirmed.end(), id));
            b.status = Status::CANCELLED;
            perProvider_[s.provider->name]--;
            promote(s, events);   // the freed seat goes to the waitlist head in the same critical section
        }
        fire(events);
    }
    Status statusOf(const std::string& id) const { std::lock_guard<Mutex> g(mu_); return bookings_.at(id)->status; }
    std::string trending() const {
        std::lock_guard<Mutex> g(mu_);
        std::string best; int n = 0;
        for (auto& kv : perProvider_) if (kv.second > n) { best = kv.first; n = kv.second; }
        return n == 0 ? "none" : best + " (" + std::to_string(n) + ")";
    }

private:
    void promote(Slot& s, std::vector<std::pair<std::string, std::string>>& events) {
        while (s.freeSeats() > 0 && !s.waitlist.empty()) {
            Booking& next = *bookings_.at(s.waitlist.front());
            s.waitlist.pop_front();
            User& u = users_.at(next.user);
            std::vector<Booking*> others = active(u.name);
            others.erase(std::remove(others.begin(), others.end(), &next), others.end());
            try {
                for (auto& r : rules_) r->check(u, s, others);
            } catch (const BookingError& e) {
                next.status = Status::CANCELLED;
                events.emplace_back(u.name, "dropped from waitlist of " + s.label() + ": " + e.what());
                continue;
            }
            next.status = Status::CONFIRMED;
            s.confirmed.push_back(next.id);
            perProvider_[s.provider->name]++;
            events.emplace_back(u.name, "promoted from waitlist: " + s.label());
        }
    }
    Booking& newBooking(User& u, Slot& s, Status st) {
        std::string id = "B" + std::to_string(nextBooking_++);
        bookings_[id].reset(new Booking{id, u.name, &s, st});
        byUser_[u.name].push_back(bookings_[id].get());
        return *bookings_[id];
    }
    std::vector<Booking*> active(const std::string& n) const {
        std::vector<Booking*> r;
        for (auto b : byUser_.at(n)) if (b->status != Status::CANCELLED) r.push_back(b);
        return r;
    }
    void fire(const std::vector<std::pair<std::string, std::string>>& ev) const {
        for (auto& e : ev) for (auto& l : listeners_) l(e.first, e.second);
    }
    const Provider& provider(const std::string& n) const {
        auto it = providers_.find(n);
        if (it == providers_.end()) throw BookingError("unknown provider " + n);
        return it->second;
    }
    User& user(const std::string& n) {
        auto it = users_.find(n);
        if (it == users_.end()) throw BookingError("unknown user " + n);
        return it->second;
    }
    Slot& slot(const std::string& id) {
        auto it = slots_.find(id);
        if (it == slots_.end()) throw BookingError("unknown slot " + id);
        return *it->second;
    }

    mutable Mutex mu_;   // guards every map below and the Slot seat / waitlist fields
    int slotMinutes_, cutoff_;
    std::function<int()> clock_;
    std::vector<std::unique_ptr<BookingRule>> rules_;
    std::map<std::string, Provider> providers_;   // std::map nodes never move: Provider* stays valid
    std::map<std::string, User> users_;
    std::map<std::string, std::unique_ptr<Slot>> slots_;
    std::vector<std::string> order_;
    std::map<std::string, std::unique_ptr<Booking>> bookings_;
    std::map<std::string, std::vector<Booking*>> byUser_;
    std::map<std::string, int> perProvider_;
    std::vector<Listener> listeners_;
    int nextSlot_ = 1, nextBooking_ = 1;
};

static int checks = 0;
static void check(bool ok, const char* what) {
    if (!ok) { std::cout << "FAILED: " << what << "\n"; std::exit(1); }
    ++checks;
}
static bool rejects(const std::function<void()>& f) {
    try { f(); return false; } catch (const BookingError& e) { std::cout << "  rejected: " << e.what() << "\n"; return true; }
}

int main() {
    int now = 8 * 60;
    BookingService clinic(30, 0, [&now] { return now; });
    clinic.addListener([](const std::string& u, const std::string& m) { std::cout << "  [notify " << u << "] " << m << "\n"; });
    clinic.registerProvider("Dr.Curious", "Cardiologist", 4.2);
    clinic.registerProvider("Dr.Daring", "Dermatologist", 4.8);
    clinic.registerProvider("Dr.Dreadful", "Dermatologist", 3.9);
    check(rejects([&] { clinic.addSlot("Dr.Curious", "09:30", "10:30", 1, 2); }), "30-min slots only");
    std::string c2 = clinic.addSlot("Dr.Curious", "12:30", "13:00", 1, 2);
    clinic.addSlot("Dr.Curious", "09:30", "10:00", 1, 2);
    std::string d2 = clinic.addSlot("Dr.Dreadful", "12:30", "13:00", 1, 2);
    clinic.addSlot("Dr.Daring", "11:30", "12:00", 1, 2);
    check(rejects([&] { clinic.addSlot("Dr.Curious", "09:45", "10:15", 1, 2); }), "no overlapping doctor slots");
    for (const char* p : {"PatientA", "PatientB"}) clinic.registerUser(p, Tier::UNLIMITED);

    std::string a = clinic.book("PatientA", c2).id;
    std::string b = clinic.book("PatientB", c2).id;
    std::cout << "PatientA " << a << " booked, PatientB " << b << " waitlisted\n";
    check(clinic.statusOf(b) == Status::WAITLISTED, "waitlisted");
    check(rejects([&] { clinic.book("PatientA", d2); }), "no two slots at the same time");
    clinic.cancel(a);
    check(clinic.statusOf(b) == Status::CONFIRMED, "promoted on cancel");
    std::cout << "Dermatologist by rating:\n";
    auto derm = clinic.available("Dermatologist", byRating);
    for (auto s : derm) std::cout << "  " << s->id << " " << s->label() << "\n";
    check(derm.front()->provider->name == "Dr.Daring", "rating strategy");
    std::cout << "trending: " << clinic.trending() << "\n";

    now = 5 * 60;
    BookingService gym(60, 30, [&now] { return now; });
    gym.registerProvider("Yoga", "Yoga", 4.5);
    std::vector<std::string> ids;
    for (int h : {6, 7, 8, 18}) ids.push_back(gym.addSlot("Yoga", fmt(h * 60), fmt(h * 60 + 60), 2, 1));
    gym.registerUser("Silvi", Tier::SILVER);
    std::vector<std::string> mine;
    for (int i = 0; i < 3; i++) mine.push_back(gym.book("Silvi", ids[i]).id);
    check(rejects([&] { gym.book("Silvi", ids[3]); }), "SILVER quota is 3");
    gym.cancel(mine[2]);
    gym.book("Silvi", ids[3]);
    now = 5 * 60 + 45;
    check(rejects([&] { gym.cancel(mine[0]); }), "cancel cutoff 30 min");
    std::cout << "All " << checks << " checks passed\n";
    return 0;
}
