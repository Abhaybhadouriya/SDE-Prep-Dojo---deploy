Java builds a custom aggregate with two iterators (one fail-fast), a BST iterator with an explicit stack, and shows JDK fail-fast and snapshot behaviour. Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Iterator<T> / Iterable<T>: the cursor interface (hasNext(), next()) and "can hand out a cursor"; any Iterable works in a for (T x : coll) loop.
  • Generics Foo<T>: a type parameter chosen by the user of the class. Think of a macro that generates one typed struct per element type, but checked by the compiler.
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • Method reference Foo::new, obj::method: shorthand for a lambda that just calls that constructor or method.
  • Python yield: turns a function into a generator: each yield hands out one value and freezes the function until the next one is asked for.
  • Python @dataclass: generates __init__, __eq__ and __repr__ from the field list, like a record.
  • operator++, operator*, operator!= (C++): functions with operator names, so an iterator object can be used like a pointer in a loop.
  • template <class T> (C++): code generated per type at compile time, like a type-safe macro.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;
import java.util.concurrent.CopyOnWriteArrayList;

public class IteratorDemo {

    // ---- Aggregate: a playlist that hides HOW songs are stored (a plain array) ----
    static final class Playlist implements Iterable<String> {
        private String[] songs = new String[2];
        private int size = 0;
        private int modCount = 0;                    // bumped on every structural change

        void add(String s) {
            if (size == songs.length) songs = Arrays.copyOf(songs, size * 2);
            songs[size++] = s;
            modCount++;
        }

        @Override public Iterator<String> iterator() { return new Forward(); }   // factory method
        Iterator<String> shuffled(long seed) { return new Shuffled(seed); }      // second traversal

        // ConcreteIterator 1: forward order, fail-fast
        private final class Forward implements Iterator<String> {
            private int cursor = 0;
            private final int expectedMod = modCount;   // snapshot of the version
            @Override public boolean hasNext() { return cursor < size; }
            @Override public String next() {
                if (modCount != expectedMod) throw new ConcurrentModificationException();
                if (!hasNext()) throw new NoSuchElementException();
                return songs[cursor++];
            }
        }

        // ConcreteIterator 2: a different order over the same data
        private final class Shuffled implements Iterator<String> {
            private final int[] order;
            private int i = 0;
            Shuffled(long seed) {
                List<Integer> idx = new ArrayList<>();
                for (int k = 0; k < size; k++) idx.add(k);
                Collections.shuffle(idx, new Random(seed));
                order = idx.stream().mapToInt(Integer::intValue).toArray();
            }
            @Override public boolean hasNext() { return i < order.length; }
            @Override public String next() {
                if (!hasNext()) throw new NoSuchElementException();
                return songs[order[i++]];
            }
        }
    }

    // ---- A custom iterator over a tree: in-order BST walk with an explicit stack ----
    record Node(int val, Node left, Node right) { }

    static final class InOrder implements Iterator<Integer> {
        private final Deque<Node> stack = new ArrayDeque<>();
        InOrder(Node root) { pushLeft(root); }
        private void pushLeft(Node n) { for (; n != null; n = n.left()) stack.push(n); }
        @Override public boolean hasNext() { return !stack.isEmpty(); }
        @Override public Integer next() {
            if (!hasNext()) throw new NoSuchElementException();
            Node n = stack.pop();
            pushLeft(n.right());                     // O(h) memory, amortised O(1) per next()
            return n.val();
        }
    }

    public static void main(String[] args) {
        Playlist p = new Playlist();
        p.add("Kun Faya Kun"); p.add("Tum Hi Ho"); p.add("Kesariya");

        System.out.print("for-each   :");
        for (String s : p) System.out.print(" [" + s + "]");      // compiler calls p.iterator()
        System.out.println();

        System.out.print("shuffled   :");
        Iterator<String> sh = p.shuffled(7);
        while (sh.hasNext()) System.out.print(" [" + sh.next() + "]");
        System.out.println();

        // Fail-fast: change the playlist while an iterator is open
        try {
            for (String s : p) if (s.startsWith("Tum")) p.add("Channa Mereya");
        } catch (ConcurrentModificationException e) {
            System.out.println("our list   : ConcurrentModificationException (fail-fast)");
        }

        // The JDK does the same with ArrayList...
        List<Integer> nums = new ArrayList<>(List.of(1, 2, 3, 4));
        try {
            for (Integer n : nums) if (n == 2) nums.remove(n);
        } catch (ConcurrentModificationException e) {
            System.out.println("ArrayList  : ConcurrentModificationException");
        }
        // ...and the safe way is to remove THROUGH the iterator
        for (Iterator<Integer> it = nums.iterator(); it.hasNext(); )
            if (it.next() % 2 == 0) it.remove();
        System.out.println("it.remove(): " + nums);

        // Fail-safe (snapshot): CopyOnWriteArrayList iterates a copy, never throws
        List<String> cow = new CopyOnWriteArrayList<>(List.of("a", "b"));
        for (String s : cow) cow.add(s.toUpperCase());
        System.out.println("COW list   : " + cow);

        // Custom tree iterator
        Node root = new Node(4, new Node(2, new Node(1, null, null), new Node(3, null, null)),
                                new Node(6, new Node(5, null, null), null));
        System.out.print("BST inorder:");
        for (Iterator<Integer> it = new InOrder(root); it.hasNext(); ) System.out.print(" " + it.next());
        System.out.println();

        // Internal iteration: the collection drives, we pass the body
        System.out.print("internal   :");
        p.forEach(s -> System.out.print(" <" + s.length() + ">"));
        System.out.println();
    }
}

Two details worth saying out loud. The ArrayList loop threw only after it had already removed 2, which is why the list is [1, 3, 4] before the it.remove() pass: fail-fast catches the bug, it does not undo it. And the BST iterator uses O(h) memory (h = tree height) with amortised O(1) per next(), the answer LeetCode 173 "Binary Search Tree Iterator" expects.

"""Iterator in Python: the iterator protocol, generators, fail-fast dicts, and a paged API."""
from dataclasses import dataclass
from typing import Iterator, Optional


class Playlist:
    """Aggregate. Callers never see the list inside."""

    def __init__(self, *songs: str):
        self._songs = list(songs)

    def __iter__(self) -> "PlaylistIterator":     # called by for-loops
        return PlaylistIterator(self._songs)

    def reversed_order(self) -> Iterator[str]:    # a second traversal, as a generator
        for i in range(len(self._songs) - 1, -1, -1):
            yield self._songs[i]


class PlaylistIterator:
    """A hand-written ConcreteIterator: keeps its own cursor."""

    def __init__(self, songs):
        self._songs, self._i = songs, 0

    def __iter__(self):
        return self

    def __next__(self) -> str:
        if self._i >= len(self._songs):
            raise StopIteration                    # Python's "hasNext() is false"
        song = self._songs[self._i]
        self._i += 1
        return song


@dataclass
class Node:
    val: int
    left: Optional["Node"] = None
    right: Optional["Node"] = None


def inorder(node: Optional[Node]) -> Iterator[int]:
    """A tree iterator in five lines: the generator keeps the stack for us."""
    if node:
        yield from inorder(node.left)
        yield node.val
        yield from inorder(node.right)


def fetch_page(cursor: int):
    """Pretend REST API: 2 orders per page, then a next cursor (None at the end)."""
    data = ["ORD-1", "ORD-2", "ORD-3", "ORD-4", "ORD-5"]
    page = data[cursor:cursor + 2]
    nxt = cursor + 2 if cursor + 2 < len(data) else None
    print(f"   (GET /orders?cursor={cursor})")
    return page, nxt


def all_orders() -> Iterator[str]:
    """Hides pagination: callers see one flat stream and pages load lazily."""
    cursor = 0
    while cursor is not None:
        page, cursor = fetch_page(cursor)
        yield from page


if __name__ == "__main__":
    p = Playlist("Kun Faya Kun", "Tum Hi Ho", "Kesariya")
    print("for-each   :", [s for s in p])
    print("reversed   :", list(p.reversed_order()))

    it = iter(p)                                   # external iteration: we pull
    print("manual next:", next(it), "|", next(it))

    root = Node(4, Node(2, Node(1), Node(3)), Node(6, Node(5)))
    print("BST inorder:", list(inorder(root)))

    print("paged API  :")
    for order in all_orders():
        print("   got", order)
        if order == "ORD-3":
            break                                  # pages 3+ are never fetched

    stock = {"apple": 3, "kiwi": 0}
    try:
        for k in stock:
            if stock[k] == 0:
                stock["mango"] = 5                 # size change during iteration
    except RuntimeError as e:
        print("dict       : RuntimeError:", e)
    for k in list(stock):                          # iterate a copy instead
        if stock[k] == 0:
            del stock[k]
    print("safe delete:", stock)

The paged-API generator is the most useful iterator in real backend code: the caller writes one flat loop, pages are fetched only when needed, and stopping early (break) means later pages are never requested. Note also that the dict really did gain "mango" before Python raised the error.

// Iterator in C++14: an STL-style iterator for a ring buffer, and a kernel-style intrusive list walk.
// Build: g++ -std=c++14 -Wall iterator.cpp -o iterator
#include <cstddef>
#include <cstdio>
#include <iostream>
#include <iterator>
#include <numeric>

// ---- Aggregate: fixed-size ring buffer (like a NIC RX ring). Callers never see head/wrap math. ----
template <typename T, std::size_t N>
class Ring {
public:
    void push(const T& v) {
        buf_[(head_ + count_) % N] = v;
        if (count_ < N) ++count_; else head_ = (head_ + 1) % N;   // overwrite oldest
    }

    class iterator {                       // ConcreteIterator
    public:
        using iterator_category = std::forward_iterator_tag;
        using value_type = T;
        using difference_type = std::ptrdiff_t;
        using pointer = const T*;
        using reference = const T&;

        iterator(const Ring* r, std::size_t i) : r_(r), i_(i) {}
        reference operator*() const { return r_->buf_[(r_->head_ + i_) % N]; }
        iterator& operator++() { ++i_; return *this; }
        bool operator!=(const iterator& o) const { return i_ != o.i_; }
        bool operator==(const iterator& o) const { return i_ == o.i_; }
    private:
        const Ring* r_;
        std::size_t i_;                    // logical position 0..count-1
    };

    iterator begin() const { return iterator(this, 0); }
    iterator end() const { return iterator(this, count_); }

private:
    T buf_[N];
    std::size_t head_ = 0, count_ = 0;
};

// ---- Kernel-style intrusive list: the link lives inside the object ----
struct list_head { list_head *next, *prev; };
#define container_of(ptr, type, member) \
    reinterpret_cast<type*>(reinterpret_cast<char*>(ptr) - offsetof(type, member))

static void list_init(list_head* h) { h->next = h->prev = h; }
static void list_add_tail(list_head* n, list_head* h) {
    n->prev = h->prev; n->next = h; h->prev->next = n; h->prev = n;
}
static void list_del(list_head* n) { n->prev->next = n->next; n->next->prev = n->prev; }

struct Flow { int id; int pkts; list_head link; };

// list_for_each_safe style: remember 'next' BEFORE the body may delete 'pos'
#define for_each_safe(pos, tmp, head) \
    for (pos = (head)->next, tmp = pos->next; pos != (head); pos = tmp, tmp = pos->next)

int main() {
    Ring<int, 4> rx;
    for (int pkt = 1; pkt <= 6; ++pkt) rx.push(pkt * 100);        // 100,200 get overwritten
    std::cout << "ring range-for :";
    for (int v : rx) std::cout << ' ' << v;                        // uses begin()/end()
    std::cout << "\nring std::accumulate = "
              << std::accumulate(rx.begin(), rx.end(), 0) << '\n'; // works with STL algorithms

    list_head flows; list_init(&flows);
    Flow f[4] = {{1, 10, {}}, {2, 0, {}}, {3, 7, {}}, {4, 0, {}}};
    for (auto& x : f) list_add_tail(&x.link, &flows);

    list_head *pos, *tmp;
    for_each_safe(pos, tmp, &flows) {                              // delete idle flows mid-walk
        Flow* fl = container_of(pos, Flow, link);
        if (fl->pkts == 0) { std::printf("evict flow %d\n", fl->id); list_del(pos); }
    }
    std::cout << "flows left     :";
    for (pos = flows.next; pos != &flows; pos = pos->next)
        std::cout << ' ' << container_of(pos, Flow, link)->id;
    std::cout << '\n';
    return 0;
}

Once the ring buffer has begin() and end() with *, ++ and !=, range-for and every STL algorithm work on it for free. The second half is the kernel idiom in miniature: the for_each_safe macro reads pos->next into tmp before the body can unlink pos.
