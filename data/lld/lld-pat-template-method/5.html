Code: importer (Java), game loop (Python), driver probe with NVI and CRTP (C++14)

All three programs were compiled and run; the output under each one is real. Files live in lld-code/lld-pat-template-method/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • abstract class or method: a base type with some slots deliberately left empty; you cannot create one directly, only subclasses that fill the slots.
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • default method in an interface: a slot with a fallback implementation, like an ops table where NULL means "use the generic helper".
  • Python ABC + @abstractmethod: a base class that refuses to be instantiated until a subclass fills the marked methods.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • template <class T> (C++): code generated per type at compile time, like a type-safe macro.
  • CRTP (Curiously Recurring Template Pattern): class Csv : public Importer<Csv>. The base is a template that takes the derived class as its parameter and calls the steps through static_cast<Derived*>(this)->step(), resolved at compile time with no vtable.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

/** Template Method: a data importer. The base class owns the steps and their order;
 *  subclasses fill in parsing, and may override optional hooks.
 *  Build and run: javac -d out ImportDemo.java && java -cp out ImportDemo */
public class ImportDemo {

    record Row(String name, int qty) { }

    abstract static class Importer {
        /** The template method: final, so no subclass can reorder or skip steps. */
        public final int run(String raw) {
            System.out.println("[" + getClass().getSimpleName() + "]");
            List<Row> rows = parse(raw);                           // abstract step: must implement
            List<Row> good = new ArrayList<>();
            for (Row r : rows) {
                if (isValid(r)) good.add(r);                       // hook with a default
                else System.out.println("  skip invalid " + r);
            }
            if (good.isEmpty()) { System.out.println("  nothing to save"); return 0; }
            save(good);                                            // invariant step: same for all
            afterSave(good.size());                                // empty hook: optional
            return good.size();
        }

        protected abstract List<Row> parse(String raw);
        protected boolean isValid(Row r) { return r.qty() >= 0; } // default rule
        protected void afterSave(int n) { }                       // does nothing unless overridden

        private void save(List<Row> rows) {                        // private: subclasses cannot touch it
            System.out.println("  BEGIN; INSERT " + rows.size() + " row(s); COMMIT");
        }
    }

    static final class CsvImporter extends Importer {
        protected List<Row> parse(String raw) {
            List<Row> out = new ArrayList<>();
            for (String line : raw.split("\n")) {
                String[] f = line.split(",");
                out.add(new Row(f[0].trim(), Integer.parseInt(f[1].trim())));
            }
            return out;
        }
    }

    static final class KeyValueImporter extends Importer {     // "name=pen qty=3; name=ink qty=0"
        protected List<Row> parse(String raw) {
            List<Row> out = new ArrayList<>();
            for (String rec : raw.split(";")) {
                Map<String, String> m = new HashMap<>();
                for (String kv : rec.trim().split("\\s+")) { String[] p = kv.split("="); m.put(p[0], p[1]); }
                out.add(new Row(m.get("name"), Integer.parseInt(m.get("qty"))));
            }
            return out;
        }
        @Override protected boolean isValid(Row r) { return r.qty() > 0; }    // stricter rule
        @Override protected void afterSave(int n) { System.out.println("  hook: notify warehouse, " + n + " new SKUs"); }
    }

    public static void main(String[] args) {
        new CsvImporter().run("pen, 10\npencil, -2\neraser, 4");
        new KeyValueImporter().run("name=pen qty=3; name=ink qty=0");
        new KeyValueImporter().run("name=glue qty=0");
    }
}

The Java keywords do the enforcing: final run() (cannot be reordered), abstract parse() (must be written), protected hooks with bodies (may be changed), private save() (cannot be touched). Also notice the early exit "nothing to save" lives in the skeleton, so every importer gets it for free.

"""Template Method in Python: a board-game loop. play() is the skeleton; games fill in the steps.
Run: python game_turns.py
"""
from abc import ABC, abstractmethod


class Game(ABC):
    def play(self, players, max_turns=10):        # the template method (do not override)
        self.setup(players)
        for turn in range(max_turns):
            p = players[turn % len(players)]
            self.take_turn(p)
            if self.winner():
                print(f"  winner: {self.winner()} after {turn + 1} turns")
                break
        else:
            print("  draw")
        self.on_game_over()                       # hook

    @abstractmethod
    def setup(self, players): ...

    @abstractmethod
    def take_turn(self, player): ...

    @abstractmethod
    def winner(self): ...

    def on_game_over(self):                       # hook: default does nothing
        pass


class SnakesAndLadders(Game):
    JUMPS = {3: 22, 17: 4, 20: 29}

    def setup(self, players):
        self.pos = {p: 0 for p in players}
        self.rolls = iter([3, 5, 6, 2, 6, 6, 6, 4, 6, 6])   # fixed dice so output is repeatable

    def take_turn(self, p):
        r = next(self.rolls)
        new = self.pos[p] + r
        new = self.JUMPS.get(new, new)
        self.pos[p] = new
        print(f"  {p} rolls {r} -> {new}")

    def winner(self):
        return next((p for p, at in self.pos.items() if at >= 30), None)


class CoinToss(Game):
    def setup(self, players):
        self.flips, self.score = iter("HTHHT"), {p: 0 for p in players}

    def take_turn(self, p):
        f = next(self.flips, "T")
        self.score[p] += f == "H"
        print(f"  {p} flips {f}")

    def winner(self):
        return next((p for p, s in self.score.items() if s == 2), None)

    def on_game_over(self):
        print(f"  final score {self.score}")


if __name__ == "__main__":
    print("Snakes and ladders:")
    SnakesAndLadders().play(["asha", "ravi"])
    print("Coin toss (first to 2 heads):")
    CoinToss().play(["asha", "ravi"])

Python has no final keyword for methods at runtime (typing.final is only checked by type checkers), so the skeleton is protected by convention and documentation. @abstractmethod still makes Python refuse to create a subclass object that forgot a step. Board-game LLD questions (snakes and ladders, tic-tac-toe) fit this shape well: setup, loop of turns, check winner, game over.

CRTP (Curiously Recurring Template Pattern): the base class is a template that takes the derived class as its parameter, so it calls the steps with static_cast<Derived*>(this)->step(), resolved at compile time. In C this is like a macro that pastes in the driver's own functions instead of calling through an ops pointer. NVI (non-virtual interface) is the other form: a public non-virtual method that calls private virtual steps.

// Template Method in C++14 as the NVI idiom (non-virtual interface):
// the public probe() is NOT virtual and fixes the order; private virtual steps are the hooks.
// A second version shows the compile-time form (CRTP), with no virtual calls at all.
// Build: g++ -std=c++14 -Wall driver_probe.cpp -o driver_probe
#include <iostream>
#include <string>

class NetDriver {
public:
    virtual ~NetDriver() = default;
    int probe(const std::string &dev) {             // template method: fixed skeleton
        std::cout << "probe " << dev << " (" << name() << ")\n";
        if (!map_registers()) { std::cout << "  -ENODEV\n"; return -19; }
        std::cout << "  alloc rings: " << ring_size() << " descriptors\n";
        int err = hw_init();
        if (err) { std::cout << "  hw_init failed " << err << ", unmapping\n"; return err; }   // cleanup is in ONE place
        std::cout << "  register_netdev(" << dev << ")\n";
        return 0;
    }
private:
    virtual std::string name() const = 0;           // primitive steps
    virtual int hw_init() = 0;
    virtual bool map_registers() { return true; }   // hooks with defaults
    virtual int ring_size() const { return 256; }
};

class E1000Like : public NetDriver {
    std::string name() const override { return "e1000-like"; }
    int hw_init() override { std::cout << "  reset MAC, read EEPROM\n"; return 0; }
};

class FlakyNic : public NetDriver {
    std::string name() const override { return "flaky"; }
    int ring_size() const override { return 4096; }
    int hw_init() override { std::cout << "  firmware load...\n"; return -110; }   // -ETIMEDOUT
};

class MissingNic : public NetDriver {
    std::string name() const override { return "missing"; }
    bool map_registers() override { return false; }
    int hw_init() override { return 0; }
};

// Compile-time Template Method: the base calls the derived step through a static cast.
template <typename Derived>
class Checksum {
public:
    unsigned run(const std::string &data) {
        unsigned acc = static_cast<Derived *>(this)->start();
        for (unsigned char c : data) acc = static_cast<Derived *>(this)->step(acc, c);
        return acc;
    }
};
struct Sum8 : Checksum<Sum8> {
    unsigned start() { return 0; }
    unsigned step(unsigned a, unsigned char c) { return (a + c) & 0xFF; }
};
struct Xor8 : Checksum<Xor8> {
    unsigned start() { return 0xFF; }
    unsigned step(unsigned a, unsigned char c) { return a ^ c; }
};

int main() {
    E1000Like a; FlakyNic b; MissingNic c;
    NetDriver *drivers[] = {&a, &b, &c};
    const char *devs[] = {"eth0", "eth1", "eth2"};
    for (int i = 0; i < 3; i++) std::cout << "  => " << drivers[i]->probe(devs[i]) << "\n";

    Sum8 s; Xor8 x;
    std::cout << "sum8(\"kernel\") = " << s.run("kernel") << ", xor8(\"kernel\") = " << x.run("kernel") << "\n";
    return 0;
}

Two C++ forms. NVI (non-virtual interface): the public probe() is not virtual, and the steps are private virtuals; derived classes may override a private virtual but cannot call it, so only the skeleton decides when steps run. CRTP: the base is a class template that calls the derived step through static_cast<Derived*>(this), so the "template method" is resolved at compile time with no vtable, which is how performance-sensitive C++ libraries often do it.
