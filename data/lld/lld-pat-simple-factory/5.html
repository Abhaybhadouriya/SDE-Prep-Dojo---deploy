Code: switch version and registry version

One runnable program per language. Each shows Version 1 (the classic switch) and Version 2 (a registry map), then adds an ElectricCar type without editing either factory function in the registry version. All three print the same six lines.

// Simple Factory: callers ask for a product by key and get back the interface.
// Run: java SimpleFactoryDemo.java   (Java 21)
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Function;

public class SimpleFactoryDemo {

    enum SpotSize { SMALL, MEDIUM, LARGE }

    // ----- Product interface + concrete products -----
    interface Vehicle {
        String plate();
        SpotSize needs();
        default String describe() {
            return getClass().getSimpleName() + "(" + plate() + ") needs a " + needs() + " spot";
        }
    }
    record Bike(String plate)  implements Vehicle { public SpotSize needs() { return SpotSize.SMALL; } }
    record Car(String plate)   implements Vehicle { public SpotSize needs() { return SpotSize.MEDIUM; } }
    record Truck(String plate) implements Vehicle { public SpotSize needs() { return SpotSize.LARGE; } }

    // ----- Version 1: the classic switch (fine while the list is small and stable) -----
    static final class VehicleFactory {
        private VehicleFactory() {}                     // no instances: it is just a function holder
        static Vehicle create(String type, String plate) {
            // <-- the pattern: one place maps a key to a concrete class
            return switch (type.trim().toUpperCase()) {
                case "BIKE"  -> new Bike(plate);
                case "CAR"   -> new Car(plate);
                case "TRUCK" -> new Truck(plate);
                default -> throw new IllegalArgumentException("unknown vehicle type: " + type);
            };
        }
    }

    // ----- Version 2: registry map (new types register themselves, the factory never changes) -----
    static final class VehicleRegistry {
        private static final Map<String, Function<String, Vehicle>> MAKERS = new ConcurrentHashMap<>();
        static {
            register("BIKE", Bike::new);
            register("CAR", Car::new);
            register("TRUCK", Truck::new);
        }
        private VehicleRegistry() {}
        static void register(String type, Function<String, Vehicle> maker) {
            if (MAKERS.putIfAbsent(type.toUpperCase(), maker) != null)
                throw new IllegalStateException("type already registered: " + type);
        }
        static Vehicle create(String type, String plate) {
            Function<String, Vehicle> maker = MAKERS.get(type.trim().toUpperCase());
            if (maker == null) throw new IllegalArgumentException("unknown vehicle type: " + type);
            return maker.apply(plate);
        }
    }

    // A type added later, in a different file in real life. No factory code is edited.
    record ElectricCar(String plate) implements Vehicle { public SpotSize needs() { return SpotSize.MEDIUM; } }

    public static void main(String[] args) {
        // The caller only knows strings (from a config file, CLI or HTTP request) and the Vehicle interface.
        String[][] arrivals = { {"car", "KA-01-1234"}, {"bike", "KA-05-777"}, {"TRUCK", "MH-12-9000"} };
        for (String[] a : arrivals) System.out.println("switch   : " + VehicleFactory.create(a[0], a[1]).describe());

        VehicleRegistry.register("EV", ElectricCar::new);  // plug-in style extension
        System.out.println("registry : " + VehicleRegistry.create("ev", "DL-3C-4242").describe());
        System.out.println("registry : " + VehicleRegistry.create("car", "TN-09-1111").describe());

        try {
            VehicleFactory.create("spaceship", "X-1");
        } catch (IllegalArgumentException e) {
            System.out.println("error    : " + e.getMessage());
        }
    }
}
"""Simple Factory: callers ask for a product by key and get back the interface.
Run: python simple_factory.py   (Python 3.11)"""
from __future__ import annotations

from abc import ABC, abstractmethod
from enum import Enum
from typing import Callable


class SpotSize(Enum):
    SMALL = 1
    MEDIUM = 2
    LARGE = 3


class Vehicle(ABC):
    def __init__(self, plate: str) -> None:
        self.plate = plate

    @abstractmethod
    def needs(self) -> SpotSize: ...

    def describe(self) -> str:
        return f"{type(self).__name__}({self.plate}) needs a {self.needs().name} spot"


# ----- Registry dict filled by a class decorator -----
_MAKERS: dict[str, Callable[[str], Vehicle]] = {}


def vehicle_type(key: str):
    """Class decorator: registers the class under `key`."""
    def wrap(cls):
        k = key.upper()
        if k in _MAKERS:
            raise ValueError(f"type already registered: {key}")
        _MAKERS[k] = cls
        return cls
    return wrap


@vehicle_type("BIKE")
class Bike(Vehicle):
    def needs(self) -> SpotSize:
        return SpotSize.SMALL


@vehicle_type("CAR")
class Car(Vehicle):
    def needs(self) -> SpotSize:
        return SpotSize.MEDIUM


@vehicle_type("TRUCK")
class Truck(Vehicle):
    def needs(self) -> SpotSize:
        return SpotSize.LARGE


def create_vehicle(kind: str, plate: str) -> Vehicle:
    """The registry factory. Callers never write Car(...) themselves."""
    try:
        maker = _MAKERS[kind.strip().upper()]
    except KeyError:
        raise ValueError(f"unknown vehicle type: {kind}") from None
    return maker(plate)


# ----- The plain if/elif version, for comparison -----
def create_vehicle_switch(kind: str, plate: str) -> Vehicle:
    k = kind.strip().upper()
    if k == "BIKE":
        return Bike(plate)
    if k == "CAR":
        return Car(plate)
    if k == "TRUCK":
        return Truck(plate)
    raise ValueError(f"unknown vehicle type: {kind}")


# A type added later. Only this block is new; create_vehicle() is untouched.
@vehicle_type("EV")
class ElectricCar(Vehicle):
    def needs(self) -> SpotSize:
        return SpotSize.MEDIUM


if __name__ == "__main__":
    for kind, plate in [("car", "KA-01-1234"), ("bike", "KA-05-777"), ("TRUCK", "MH-12-9000")]:
        print("switch   :", create_vehicle_switch(kind, plate).describe())
    print("registry :", create_vehicle("ev", "DL-3C-4242").describe())
    print("registry :", create_vehicle("car", "TN-09-1111").describe())
    try:
        create_vehicle("spaceship", "X-1")
    except ValueError as e:
        print("error    :", e)

C++ you need here: std::unique_ptr<T> = an owning pointer that frees its object when it goes out of scope (a kfree() that runs itself on every exit path). A lambda [](args){ body } = an anonymous function: a function pointer plus any variables it captured in the brackets, like a callback with its void *ctx built in. std::function<R(Args)> = a typed slot that can hold any callable (function pointer, lambda, functor). Ownership and RAII in more depth: kref and shared ownership.

// Simple Factory: callers ask for a product by key and get back the interface.
// Build: g++ -std=c++14 -Wall -O2 simple_factory.cpp -o simple_factory && ./simple_factory
#include <algorithm>
#include <cctype>
#include <functional>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <unordered_map>

enum class SpotSize { SMALL, MEDIUM, LARGE };
static const char* toStr(SpotSize s) {
    switch (s) {
        case SpotSize::SMALL: return "SMALL";
        case SpotSize::MEDIUM: return "MEDIUM";
        default: return "LARGE";
    }
}

struct Vehicle {                     // product interface: a vtable, like an ops struct
    explicit Vehicle(std::string p) : plate(std::move(p)) {}
    virtual ~Vehicle() = default;
    virtual SpotSize needs() const = 0;
    virtual const char* kind() const = 0;
    std::string describe() const {
        return std::string(kind()) + "(" + plate + ") needs a " + toStr(needs()) + " spot";
    }
    std::string plate;
};
struct Bike : Vehicle {
    using Vehicle::Vehicle;
    SpotSize needs() const override { return SpotSize::SMALL; }
    const char* kind() const override { return "Bike"; }
};
struct Car : Vehicle {
    using Vehicle::Vehicle;
    SpotSize needs() const override { return SpotSize::MEDIUM; }
    const char* kind() const override { return "Car"; }
};
struct Truck : Vehicle {
    using Vehicle::Vehicle;
    SpotSize needs() const override { return SpotSize::LARGE; }
    const char* kind() const override { return "Truck"; }
};

static std::string upper(std::string s) {
    std::transform(s.begin(), s.end(), s.begin(),
                   [](unsigned char c) { return static_cast<char>(std::toupper(c)); });
    return s;
}

// ----- Version 1: switch-style factory -----
std::unique_ptr<Vehicle> createVehicle(const std::string& type, const std::string& plate) {
    const std::string t = upper(type);
    if (t == "BIKE")  return std::make_unique<Bike>(plate);
    if (t == "CAR")   return std::make_unique<Car>(plate);
    if (t == "TRUCK") return std::make_unique<Truck>(plate);
    throw std::invalid_argument("unknown vehicle type: " + type);
}

// ----- Version 2: registry of maker functions (a table of function pointers keyed by name) -----
class VehicleRegistry {
public:
    using Maker = std::function<std::unique_ptr<Vehicle>(const std::string&)>;
    static VehicleRegistry& instance() { static VehicleRegistry r; return r; }  // thread-safe init (C++11+)
    void add(const std::string& type, Maker m) {
        if (!makers_.emplace(upper(type), std::move(m)).second)
            throw std::logic_error("type already registered: " + type);
    }
    std::unique_ptr<Vehicle> create(const std::string& type, const std::string& plate) const {
        auto it = makers_.find(upper(type));
        if (it == makers_.end()) throw std::invalid_argument("unknown vehicle type: " + type);
        return it->second(plate);
    }
private:
    VehicleRegistry() {
        add("BIKE",  [](const std::string& p) { return std::unique_ptr<Vehicle>(new Bike(p)); });
        add("CAR",   [](const std::string& p) { return std::unique_ptr<Vehicle>(new Car(p)); });
        add("TRUCK", [](const std::string& p) { return std::unique_ptr<Vehicle>(new Truck(p)); });
    }
    std::unordered_map<std::string, Maker> makers_;
};

// Added later: a new product plus one registration line; no factory code edited.
struct ElectricCar : Vehicle {
    using Vehicle::Vehicle;
    SpotSize needs() const override { return SpotSize::MEDIUM; }
    const char* kind() const override { return "ElectricCar"; }
};

int main() {
    const char* arrivals[][2] = { {"car", "KA-01-1234"}, {"bike", "KA-05-777"}, {"TRUCK", "MH-12-9000"} };
    for (auto& a : arrivals) std::cout << "switch   : " << createVehicle(a[0], a[1])->describe() << "\n";

    auto& reg = VehicleRegistry::instance();
    reg.add("EV", [](const std::string& p) { return std::unique_ptr<Vehicle>(new ElectricCar(p)); });
    std::cout << "registry : " << reg.create("ev", "DL-3C-4242")->describe() << "\n";
    std::cout << "registry : " << reg.create("car", "TN-09-1111")->describe() << "\n";
    try {
        createVehicle("spaceship", "X-1");
    } catch (const std::invalid_argument& e) {
        std::cout << "error    : " << e.what() << "\n";
    }
    return 0;
}

Reading the code like a reviewer

  • Return type is the interface. create returns Vehicle, not Car. If it returned the concrete type, callers would start depending on it again.
  • Keys are normalised once. Upper-casing and trimming in the factory removes the "CAR vs car" bug class from the whole system. In a real system, prefer an enum VehicleType over raw strings and parse the string at the edge.
  • Unknown key = exception. Returning null just moves the crash somewhere harder to debug.
  • Registry is thread-safe for registration. Java uses ConcurrentHashMap.putIfAbsent; duplicates are rejected. In practice you register everything at start-up and only read afterwards, so a plain map published safely (for example in a static initialiser) is also fine.
  • Python uses a class decorator to register, so the "one line to add a type" sits right next to the new class. C++ uses a function-local static for the registry, which C++11 and later initialise thread-safely.

From your world

The registry map is a table of constructors keyed by name - the same thing as an array of struct entries { const char *name; struct foo *(*create)(void); } that you walk with strcmp, or a list that modules add to with a register_*() call in their module_init. The C++ std::function map is that table with type-checking.

The same idea in C

Simple Factory in plain C: a name to ops-table registry with a strcmp lookup. The caller never names the real type; it only gets back a const struct vehicle_ops *. The kernel crypto API (crypto_alloc_shash("sha256", 0, 0)) and register_filesystem() / get_fs_type() are this shape. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-simple-factory/factory_sketch.c).

/* Simple Factory in C: a name -> ops registry with a strcmp lookup.
   Illustrative, not real kernel code (the crypto API's crypto_alloc_shash("sha256") works this way).
   Build: gcc -std=c99 -Wall factory_sketch.c -o factory_sketch */
#include <stdio.h>
#include <string.h>

struct vehicle_ops { const char *name; int wheels; int (*fee)(int hours); };

static int bike_fee(int h) { return 10 * h; }
static int car_fee(int h)  { return 30 * h; }

static const struct vehicle_ops registry[] = {      /* the only place that names the types */
    { "bike", 2, bike_fee },
    { "car",  4, car_fee  },
};

/* the factory: callers pass a key and get back an ops pointer, never a concrete type */
static const struct vehicle_ops *vehicle_lookup(const char *name)
{
    for (size_t i = 0; i < sizeof(registry) / sizeof(registry[0]); i++)
        if (strcmp(registry[i].name, name) == 0)
            return &registry[i];
    return NULL;                                   /* unknown key: caller must handle it */
}

int main(void)
{
    const char *asks[] = { "car", "bike", "truck" };
    for (int i = 0; i < 3; i++) {
        const struct vehicle_ops *v = vehicle_lookup(asks[i]);
        if (v) printf("%s: %d wheels, 3h fee %d\n", asks[i], v->wheels, v->fee(3));
        else   printf("%s: no such type\n", asks[i]);
    }
    return 0;
}
