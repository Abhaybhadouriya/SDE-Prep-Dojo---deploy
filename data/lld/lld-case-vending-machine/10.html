x -> x.price (lambda)An anonymous function that can capture local variables: a callback and its void *ctx in one value. (a, b) -> a.x - b.x is the cmp you would hand to qsort.
Type::method (method reference)A pointer to a function that already exists, written without a wrapper: Post::score is &post_score.
Optional<T>A pointer that may be NULL, made explicit in the type, so the caller has to say what happens when it is empty (orElse, isPresent, map).
.stream().filter(..).map(..).collect(..)A for-loop written as a pipeline: filter is if (!pred(x)) continue;, map transforms each element, collect/toList appends the survivors to a new array.
map.computeIfAbsent(k, f)Lookup-or-create in one call: find k; if it is missing, call f, insert the result and return it.
map.merge(k, 1, Integer::sum)counts[k] += 1, inserting the first value when k is absent.
Comparator.comparing(f).reversed().thenComparing(g)Builds the cmp() function field by field: compare by f, negate the result, break ties with g.
default method in an interfaceAn ops-table slot pre-filled with a generic implementation; an implementing class overrides only the slots it cares about.

synchronizedTake the object's built-in mutex on entry and release it on every exit. It is reentrant: the owning thread may enter again.
throw new XException(..)Unwinds to the nearest catch: an error return that a caller cannot silently ignore.

import java.util.*;

// ---------- model ----------
enum Denomination {                       // Indian coins and notes, value in whole rupees
    HUNDRED(100), FIFTY(50), TWENTY(20), TEN(10), FIVE(5), TWO(2), ONE(1);
    final int value;
    Denomination(int value) { this.value = value; }
    @Override public String toString() { return "Rs" + value; }
}

final class Product {
    final String name; final int price;
    Product(String name, int price) {
        if (price <= 0) throw new IllegalArgumentException("price must be positive");
        this.name = name; this.price = price;
    }
}

final class Purchase {                    // what falls out of the machine
    final Product product; final List<Denomination> change;
    Purchase(Product product, List<Denomination> change) { this.product = product; this.change = change; }
}

class VendingException extends RuntimeException {
    private static final long serialVersionUID = 1L;
    VendingException(String msg) { super(msg); }
}

// ---------- inventory: slot code -> product + count ----------
final class Inventory {
    private static final class Slot { Product product; int count; }
    private final Map<String, Slot> slots = new LinkedHashMap<>();

    void restock(String code, Product p, int qty) {
        if (qty <= 0) throw new VendingException("quantity must be positive");
        Slot s = slots.computeIfAbsent(code, k -> new Slot());   // C: a lambda = a function pointer plus its captured context
        if (s.product != null && s.count > 0 && !s.product.name.equals(p.name))
            throw new VendingException("slot " + code + " still holds " + s.product.name);
        s.product = p; s.count += qty;
    }
    Product productAt(String code) {
        Slot s = slots.get(code);
        if (s == null) throw new VendingException("unknown slot " + code);
        if (s.count == 0) throw new VendingException(code + " (" + s.product.name + ") is sold out");
        return s.product;
    }
    void takeOne(String code) { slots.get(code).count--; }
    int count(String code) { Slot s = slots.get(code); return s == null ? 0 : s.count; }
    boolean isEmpty() { return slots.values().stream().allMatch(s -> s.count == 0); }   // C: stream() = a for-loop written as filter/map/collect
}

// ---------- cash box: how many of each coin/note the machine holds ----------
final class CashBox {
    private final EnumMap<Denomination, Integer> counts = new EnumMap<>(Denomination.class);
    CashBox() { for (Denomination d : Denomination.values()) counts.put(d, 0); }
    void add(Denomination d, int n) { counts.merge(d, n, Integer::sum); }   // C: Type::method = a pointer to an existing function
    void addAll(Collection<Denomination> ds) { for (Denomination d : ds) add(d, 1); }
    void removeAll(Collection<Denomination> ds) {
        for (Denomination d : ds) {
            if (counts.get(d) == 0) throw new IllegalStateException("cash box has no " + d);
            counts.merge(d, -1, Integer::sum);
        }
    }
    EnumMap<Denomination, Integer> snapshot() { return new EnumMap<>(counts); }
    int total() { int t = 0; for (Map.Entry<Denomination, Integer> e : counts.entrySet()) t += e.getKey().value * e.getValue(); return t; }
}

// ---------- strategy: how to pay out change from a limited cash box ----------
interface ChangeMaker {
    /** Coins summing exactly to amount using at most the available counts, or empty if impossible. */
    Optional<List<Denomination>> make(int amount, Map<Denomination, Integer> available);   // C: Optional = a pointer that may be NULL, made explicit
}

/** Biggest coin first. Fast, but can fail when counts are limited (8 from {5, 2, 2, 2, 2}). */
final class GreedyChangeMaker implements ChangeMaker {
    public Optional<List<Denomination>> make(int amount, Map<Denomination, Integer> available) {
        List<Denomination> out = new ArrayList<>();
        for (Denomination d : Denomination.values()) {          // declared largest first
            int use = Math.min(amount / d.value, available.getOrDefault(d, 0));
            for (int i = 0; i < use; i++) out.add(d);
            amount -= use * d.value;
        }
        return amount == 0 ? Optional.of(out) : Optional.empty();
    }
}

/** Bounded coin change DP: fewest coins, respects counts. O(types * amount * maxCount). */
final class DpChangeMaker implements ChangeMaker {
    public Optional<List<Denomination>> make(int amount, Map<Denomination, Integer> available) {
        Denomination[] ds = Denomination.values();
        final int INF = Integer.MAX_VALUE / 2;
        int[] best = new int[amount + 1];
        Arrays.fill(best, INF);
        best[0] = 0;
        int[][] used = new int[ds.length][amount + 1];          // used[i][a] = how many ds[i] in best(a) after type i
        for (int i = 0; i < ds.length; i++) {
            int v = ds[i].value, have = available.getOrDefault(ds[i], 0);
            for (int a = amount; a >= v; a--)                    // descending: each type is used up to 'have' times
                for (int k = 1; k <= have && k * v <= a; k++)
                    if (best[a - k * v] + k < best[a]) { best[a] = best[a - k * v] + k; used[i][a] = k; }
        }
        if (best[amount] >= INF) return Optional.empty();
        List<Denomination> out = new ArrayList<>();
        for (int i = ds.length - 1, a = amount; i >= 0; i--) {
            int k = used[i][a];
            for (int j = 0; j < k; j++) out.add(ds[i]);
            a -= k * ds[i].value;
        }
        out.sort(Comparator.comparingInt((Denomination d) -> d.value).reversed());   // C: builds the cmp() function you would pass to qsort
        return Optional.of(out);
    }
}

// ---------- State pattern: one class per machine state ----------
interface State {
    String name();
    default int insert(VendingMachine m, Denomination d) { throw refuse("insert money"); }   // C: default method = an ops-table slot with a generic body
    default Purchase select(VendingMachine m, String code) { throw refuse("select a product"); }
    default List<Denomination> cancel(VendingMachine m) { throw refuse("cancel"); }
    default void enterMaintenance(VendingMachine m) { throw refuse("enter maintenance"); }
    default void restock(VendingMachine m, String code, Product p, int qty) { throw refuse("restock"); }
    default void loadChange(VendingMachine m, Denomination d, int n) { throw refuse("load change"); }
    default void exitMaintenance(VendingMachine m) { throw refuse("exit maintenance"); }
    default VendingException refuse(String what) { return new VendingException("cannot " + what + " while " + name()); }
}

final class IdleState implements State {
    public String name() { return "IDLE"; }
    public int insert(VendingMachine m, Denomination d) { m.escrow.add(d); m.setState(m.hasMoney); return m.balance(); }
    public Purchase select(VendingMachine m, String code) { m.inventory.productAt(code); throw new VendingException("insert money first"); }
    public List<Denomination> cancel(VendingMachine m) { return List.of(); }   // C: immutable literal collection
    public void enterMaintenance(VendingMachine m) { m.setState(m.maintenance); }
}

final class HasMoneyState implements State {
    public String name() { return "HAS_MONEY"; }
    public int insert(VendingMachine m, Denomination d) { m.escrow.add(d); return m.balance(); }
    public List<Denomination> cancel(VendingMachine m) {
        List<Denomination> refund = new ArrayList<>(m.escrow);   // give back exactly what was inserted
        m.escrow.clear();
        m.setState(m.idle);
        return refund;
    }
    public Purchase select(VendingMachine m, String code) {
        Product p = m.inventory.productAt(code);                  // unknown / sold out -> throws, stay HAS_MONEY
        int balance = m.balance();
        if (balance < p.price) throw new VendingException("insert Rs" + (p.price - balance) + " more for " + p.name);
        EnumMap<Denomination, Integer> pool = m.cash.snapshot();  // inserted coins can be used as change
        for (Denomination d : m.escrow) pool.merge(d, 1, Integer::sum);
        List<Denomination> change = m.changeMaker.make(balance - p.price, pool)
            .orElseThrow(() -> new VendingException("cannot return Rs" + (balance - p.price)
                + " change; insert exact amount or cancel"));
        // all checks passed: commit (escrow -> cash box, change out of cash box, one item out)
        m.cash.addAll(m.escrow);
        m.cash.removeAll(change);
        m.escrow.clear();
        m.setState(m.dispensing);
        return m.dispensing.dispense(m, code, p, change);
    }
}

final class DispensingState implements State {
    public String name() { return "DISPENSING"; }
    Purchase dispense(VendingMachine m, String code, Product p, List<Denomination> change) {
        m.inventory.takeOne(code);                                // real machine: motor + drop sensor here
        m.sales += p.price;
        m.setState(m.inventory.isEmpty() ? m.soldOut : m.idle);
        return new Purchase(p, change);
    }
}

final class SoldOutState implements State {
    public String name() { return "SOLD_OUT"; }
    public int insert(VendingMachine m, Denomination d) { throw new VendingException("machine is sold out, " + d + " returned"); }
    public List<Denomination> cancel(VendingMachine m) { return List.of(); }
    public void enterMaintenance(VendingMachine m) { m.setState(m.maintenance); }
}

final class MaintenanceState implements State {
    public String name() { return "MAINTENANCE"; }
    public void restock(VendingMachine m, String code, Product p, int qty) { m.inventory.restock(code, p, qty); }
    public void loadChange(VendingMachine m, Denomination d, int n) { m.cash.add(d, n); }
    public void exitMaintenance(VendingMachine m) { m.setState(m.inventory.isEmpty() ? m.soldOut : m.idle); }
}

"""Vending machine LLD: State pattern, inventory, change-making (greedy vs bounded DP), refunds."""
import threading
from dataclasses import dataclass
from enum import Enum
from typing import Dict, List, Optional


class Denomination(Enum):          # whole rupees, declared largest first
    HUNDRED = 100
    FIFTY = 50
    TWENTY = 20
    TEN = 10
    FIVE = 5
    TWO = 2
    ONE = 1

    def __repr__(self) -> str:
        return f"Rs{self.value}"


@dataclass(frozen=True)
class Product:
    name: str
    price: int

    def __post_init__(self):
        if self.price <= 0:
            raise ValueError("price must be positive")


@dataclass(frozen=True)
class Purchase:
    product: Product
    change: List[Denomination]


class VendingError(Exception):
    pass


class Inventory:
    def __init__(self):
        self._slots: Dict[str, list] = {}           # code -> [product, count]

    def restock(self, code: str, p: Product, qty: int) -> None:
        if qty <= 0:
            raise VendingError("quantity must be positive")
        slot = self._slots.setdefault(code, [p, 0])
        if slot[1] > 0 and slot[0].name != p.name:
            raise VendingError(f"slot {code} still holds {slot[0].name}")
        slot[0] = p
        slot[1] += qty

    def product_at(self, code: str) -> Product:
        if code not in self._slots:
            raise VendingError(f"unknown slot {code}")
        p, n = self._slots[code]
        if n == 0:
            raise VendingError(f"{code} ({p.name}) is sold out")
        return p

    def take_one(self, code: str) -> None:
        self._slots[code][1] -= 1

    def count(self, code: str) -> int:
        return self._slots.get(code, [None, 0])[1]

    def is_empty(self) -> bool:
        return all(n == 0 for _, n in self._slots.values())


class CashBox:
    def __init__(self):
        self._counts = {d: 0 for d in Denomination}

    def add(self, d: Denomination, n: int = 1) -> None:
        self._counts[d] += n

    def add_all(self, ds: List[Denomination]) -> None:
        for d in ds:
            self.add(d)

    def remove_all(self, ds: List[Denomination]) -> None:
        for d in ds:
            if self._counts[d] == 0:
                raise RuntimeError(f"cash box has no {d!r}")
            self._counts[d] -= 1

    def snapshot(self) -> Dict[Denomination, int]:
        return dict(self._counts)

    def total(self) -> int:
        return sum(d.value * n for d, n in self._counts.items())


class ChangeMaker:
    def make(self, amount: int, available: Dict[Denomination, int]) -> Optional[List[Denomination]]:
        raise NotImplementedError


class GreedyChangeMaker(ChangeMaker):
    """Biggest coin first; can fail with limited counts."""

    def make(self, amount, available):
        out = []
        for d in Denomination:
            use = min(amount // d.value, available.get(d, 0))
            out += [d] * use
            amount -= use * d.value
        return out if amount == 0 else None


class DpChangeMaker(ChangeMaker):
    """Bounded coin change: fewest coins that respect the counts."""

    def make(self, amount, available):
        ds = list(Denomination)
        inf = float("inf")
        best = [0] + [inf] * amount
        used = [[0] * (amount + 1) for _ in ds]
        for i, d in enumerate(ds):
            have, v = available.get(d, 0), d.value
            for a in range(amount, v - 1, -1):
                for k in range(1, min(have, a // v) + 1):
                    if best[a - k * v] + k < best[a]:
                        best[a] = best[a - k * v] + k
                        used[i][a] = k
        if best[amount] == inf:
            return None
        out, a = [], amount
        for i in range(len(ds) - 1, -1, -1):
            out += [ds[i]] * used[i][a]
            a -= used[i][a] * ds[i].value
        return sorted(out, key=lambda d: -d.value)

class State:
    name = "?"

    def _refuse(self, what: str) -> VendingError:
        return VendingError(f"cannot {what} while {self.name}")

    def insert(self, m, d):
        raise self._refuse("insert money")

    def select(self, m, code):
        raise self._refuse("select a product")

    def cancel(self, m):
        raise self._refuse("cancel")

    def enter_maintenance(self, m):
        raise self._refuse("enter maintenance")

    def restock(self, m, code, p, qty):
        raise self._refuse("restock")

    def load_change(self, m, d, n):
        raise self._refuse("load change")

    def exit_maintenance(self, m):
        raise self._refuse("exit maintenance")


class IdleState(State):
    name = "IDLE"

    def insert(self, m, d):
        m.escrow.append(d)
        m.state = m.HAS_MONEY
        return m.balance()

    def select(self, m, code):
        m.inventory.product_at(code)
        raise VendingError("insert money first")

    def cancel(self, m):
        return []

    def enter_maintenance(self, m):
        m.state = m.MAINTENANCE


class HasMoneyState(State):
    name = "HAS_MONEY"

    def insert(self, m, d):
        m.escrow.append(d)
        return m.balance()

    def cancel(self, m):
        refund, m.escrow = m.escrow, []
        m.state = m.IDLE
        return refund

    def select(self, m, code):
        p = m.inventory.product_at(code)
        balance = m.balance()
        if balance < p.price:
            raise VendingError(f"insert Rs{p.price - balance} more for {p.name}")
        pool = m.cash.snapshot()
        for d in m.escrow:
            pool[d] += 1
        change = m.change_maker.make(balance - p.price, pool)
        if change is None:
            raise VendingError(f"cannot return Rs{balance - p.price} change; insert exact amount or cancel")
        m.cash.add_all(m.escrow)
        m.cash.remove_all(change)
        m.escrow = []
        m.state = m.DISPENSING
        return m.DISPENSING.dispense(m, code, p, change)


class DispensingState(State):
    name = "DISPENSING"

    def dispense(self, m, code, p, change):
        m.inventory.take_one(code)
        m.sales += p.price
        m.state = m.SOLD_OUT if m.inventory.is_empty() else m.IDLE
        return Purchase(p, change)


class SoldOutState(State):
    name = "SOLD_OUT"

    def insert(self, m, d):
        raise VendingError(f"machine is sold out, {d!r} returned")

    def cancel(self, m):
        return []

    def enter_maintenance(self, m):
        m.state = m.MAINTENANCE


class MaintenanceState(State):
    name = "MAINTENANCE"

    def restock(self, m, code, p, qty):
        m.inventory.restock(code, p, qty)

    def load_change(self, m, d, n):
        m.cash.add(d, n)

    def exit_maintenance(self, m):
        m.state = m.SOLD_OUT if m.inventory.is_empty() else m.IDLE


class VendingMachine:
    IDLE, HAS_MONEY, DISPENSING = IdleState(), HasMoneyState(), DispensingState()
    SOLD_OUT, MAINTENANCE = SoldOutState(), MaintenanceState()

    def __init__(self, change_maker: ChangeMaker):
        self.inventory = Inventory()
        self.cash = CashBox()
        self.escrow: List[Denomination] = []
        self.change_maker = change_maker
        self.sales = 0
        self.state: State = self.SOLD_OUT
        self._lock = threading.Lock()

    def balance(self) -> int:
        return sum(d.value for d in self.escrow)

    def _do(self, fn, *args):
        with self._lock:                       # one event at a time, like one keypad
            return fn(self, *args)

    def insert(self, d):
        return self._do(lambda m, x: m.state.insert(m, x), d)

    def select(self, code):
        return self._do(lambda m, c: m.state.select(m, c), code)

    def cancel(self):
        return self._do(lambda m: m.state.cancel(m))

    def enter_maintenance(self):
        self._do(lambda m: m.state.enter_maintenance(m))

    def restock(self, code, p, qty):
        self._do(lambda m: m.state.restock(m, code, p, qty))

    def load_change(self, d, n):
        self._do(lambda m: m.state.load_change(m, d, n))

    def exit_maintenance(self):
        self._do(lambda m: m.state.exit_maintenance(m))

Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

// ---------- change-making strategies ----------
struct ChangeMaker {
    virtual ~ChangeMaker() = default;
    virtual bool make(int amount, const Pool& avail, Coins& out) const = 0;   // false = impossible
};
struct GreedyChangeMaker : ChangeMaker {
    bool make(int amount, const Pool& avail, Coins& out) const override {
        out.clear();
        for (int d : DENOMS) {
            auto it = avail.find(d);
            int use = std::min(amount / d, it == avail.end() ? 0 : it->second);
            out.insert(out.end(), use, d);
            amount -= use * d;
        }
        return amount == 0;
    }
};
struct DpChangeMaker : ChangeMaker {
    bool make(int amount, const Pool& avail, Coins& out) const override {
        const int INF = INT_MAX / 2, T = 7;
        std::vector<int> best(amount + 1, INF);
        std::vector<std::vector<int>> used(T, std::vector<int>(amount + 1, 0));
        best[0] = 0;
        for (int i = 0; i < T; ++i) {
            int v = DENOMS[i];
            auto it = avail.find(v);
            int have = it == avail.end() ? 0 : it->second;
            for (int a = amount; a >= v; --a)
                for (int k = 1; k <= have && k * v <= a; ++k)
                    if (best[a - k * v] + k < best[a]) { best[a] = best[a - k * v] + k; used[i][a] = k; }
        }
        out.clear();
        if (best[amount] >= INF) return false;
        for (int i = T - 1, a = amount; i >= 0; --i) { out.insert(out.end(), used[i][a], DENOMS[i]); a -= used[i][a] * DENOMS[i]; }
        std::sort(out.rbegin(), out.rend());
        return true;
    }
};

class Inventory {
public:
    void restock(const std::string& code, const Product& p, int qty) {
        if (qty <= 0) throw VendingError("quantity must be positive");
        auto& s = slots_[code];
        if (s.count > 0 && s.product.name != p.name) throw VendingError("slot " + code + " still holds " + s.product.name);
        s.product = p; s.count += qty;

        auto it = slots_.find(code);
        if (it == slots_.end()) throw VendingError("unknown slot " + code);
        if (it->second.count == 0) throw VendingError(code + " (" + it->second.product.name + ") is sold out");
        return it->second.product;
    }
    void takeOne(const std::string& code) { --slots_[code].count; }
    bool isEmpty() const { for (const auto& kv : slots_) if (kv.second.count) return false; return true; }
private:
    struct Slot { Product product{"", 0}; int count = 0; };

int HasMoneyState::insert(VendingMachine& m, int d) { m.escrow.push_back(d); return m.balance(); }
Coins HasMoneyState::cancel(VendingMachine& m) { Coins r; r.swap(m.escrow); m.state = &m.idle; return r; }
Purchase HasMoneyState::select(VendingMachine& m, const std::string& code) {
    Product p = m.inventory.productAt(code);
    int balance = m.balance();
    if (balance < p.price) throw VendingError("insert Rs" + std::to_string(p.price - balance) + " more for " + p.name);
    Pool pool = m.cash.snapshot();
    for (int d : m.escrow) ++pool[d];
    Coins change;
    if (!m.changeMaker.make(balance - p.price, pool, change))
        throw VendingError("cannot return Rs" + std::to_string(balance - p.price) + " change; insert exact amount or cancel");
    m.cash.addAll(m.escrow);
    m.cash.removeAll(change);
    m.escrow.clear();
    m.state = &m.dispensing;
    return m.dispensing.dispense(m, code, p, change);
}
Purchase DispensingState::dispense(VendingMachine& m, const std::string& code, const Product& p, const Coins& change) {
    m.inventory.takeOne(code);
    m.sales += p.price;
    m.state = m.inventory.isEmpty() ? static_cast<State*>(&m.soldOut) : &m.idle;
    return Purchase{p, change};
}
void SoldOutState::enterMaintenance(VendingMachine& m) { m.state = &m.maintenance; }
void MaintenanceState::restock(VendingMachine& m, const std::string& c, const Product& p, int q) { m.inventory.restock(c, p, q); }
void MaintenanceState::loadChange(VendingMachine& m, int d, int n) { m.cash.add(d, n); }
void MaintenanceState::exitMaintenance(VendingMachine& m) {
    m.state = m.inventory.isEmpty() ? static_cast<State*>(&m.soldOut) : &m.idle;
}

static int passed = 0;
static void check(bool ok, const char* what) { if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); } ++passed; }
template <class F> static std::string error(F f) { try { f(); return "no error"; } catch (const VendingError& e) { return e.what(); } }

int main() {
    DpChangeMaker dp;
    VendingMachine vm(dp);
    vm.enterMaintenance();
    vm.restock("A1", {"Chips", 20}, 5); vm.restock("A2", {"Soda", 35}, 3);

    vm.loadChange(20, 1); vm.loadChange(10, 2); vm.loadChange(5, 1); vm.loadChange(2, 4);
    vm.exitMaintenance();
    int float0 = vm.cashTotal();
    std::cout << "ready: state=" << vm.stateName() << ", float Rs" << float0 << "\n";
    check(std::string(vm.stateName()) == "IDLE" && float0 == 53, "stocked and idle");

    vm.insert(50);
    Purchase p1 = vm.select("A1");
    std::cout << "1) Chips with Rs50, change " << str(p1.change) << "\n";
    check(p1.change == Coins({20, 10}), "change 30");

    vm.insert(10);
    std::string e2 = error([&] { vm.select("A2"); });
    Coins refund = vm.cancel();
    std::cout << "2) " << e2 << "; refund " << str(refund) << "\n";
    check(e2.find("Rs25 more") != std::string::npos && refund == Coins({10}) && std::string(vm.stateName()) == "IDLE", "refund");

    Pool pool = vm.cash.snapshot(); ++pool[20];
    Coins g;
    bool greedyOk = GreedyChangeMaker().make(8, pool, g);
    vm.insert(20);
    Purchase p3 = vm.select("B2");
    std::cout << "3) change 8: greedy " << (greedyOk ? str(g) : "impossible") << ", DP " << str(p3.change) << "\n";
    check(!greedyOk && p3.change == Coins({2, 2, 2, 2}), "dp beats greedy");

    vm.insert(10); vm.insert(5); vm.select("B1");
    vm.insert(20);
    std::string e4 = error([&] { vm.select("B1"); });
    std::string e4b = error([&] { vm.select("Z9"); });
    check(e4.find("sold out") != std::string::npos && e4b.find("unknown slot") != std::string::npos, "sold out + unknown");
    std::string e5 = error([&] { vm.select("B2"); });
    vm.insert(2);
    Purchase p5 = vm.select("B2");
    std::cout << "4) " << e4 << "\n5) " << e5 << "; with Rs2 more, change " << str(p5.change) << "\n";
    check(e5.find("cannot return Rs8") != std::string::npos && p5.change == Coins({10}), "recover by adding a coin");

    vm.insert(5);
    std::string e6 = error([&] { vm.enterMaintenance(); });
    vm.cancel();
    std::cout << "6) " << e6 << "\n";
    check(e6.find("while HAS_MONEY") != std::string::npos, "no maintenance mid-transaction");
    check(vm.cashTotal() == float0 + 59 && vm.sales == 59, "money conserved");

    VendingMachine tiny(dp);                                   // 7) sold out, then restocked
    tiny.enterMaintenance(); tiny.restock("C1", {"Gum", 5}, 1); tiny.exitMaintenance();
    tiny.insert(5); tiny.select("C1");
    std::string e7 = error([&] { tiny.insert(10); });
    std::string soldOut = tiny.stateName();
    tiny.enterMaintenance(); tiny.restock("C1", {"Gum", 5}, 10); tiny.exitMaintenance();
    std::cout << "7) last Gum sold: state " << soldOut << ", " << e7 << "; after restock " << tiny.stateName() << "\n";
    check(soldOut == "SOLD_OUT" && e7.find("sold out") != std::string::npos && std::string(tiny.stateName()) == "IDLE",
          "SOLD_OUT -> MAINTENANCE -> IDLE");

    std::cout << "All " << passed << " checks passed\n";
    return 0;
}
