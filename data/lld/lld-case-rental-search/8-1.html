"""Rental search: listings + inverted text index + location index, Specification filters,
ranking strategies and pagination. Same design as the Java version."""
from __future__ import annotations

from enum import Enum
from typing import Callable, Dict, FrozenSet, Iterable, List, Optional, Set, Tuple


class ListingType(Enum):
    RENT = "RENT"
    SELL = "SELL"


class Status(Enum):
    AVAILABLE = "AVAILABLE"
    SOLD = "SOLD"


# ---------------- units ----------------
def parse_price(s: str) -> int:
    t = s.strip().lower().replace(",", "")
    mult = 1
    for suffix, m in (("cr", 10_000_000), ("l", 100_000), ("k", 1_000)):
        if t.endswith(suffix):
            t, mult = t[: -len(suffix)], m
            break
    try:
        v = float(t)
    except ValueError:
        raise ValueError(f"bad price: {s}") from None
    if v <= 0:
        raise ValueError(f"price must be > 0: {s}")
    return round(v * mult)


def parse_size_sqft(s: str) -> int:
    t = s.strip().lower().replace(" ", "").replace(".", "")
    factor = 1.0
    for suffix, f in (("sqm", 10.7639), ("sqyd", 9.0), ("sqft", 1.0)):
        if t.endswith(suffix):
            t, factor = t[: -len(suffix)], f
            break
    try:
        return round(float(t) * factor)
    except ValueError:
        raise ValueError(f"bad size: {s}") from None


def parse_price_range(s: str) -> Tuple[int, float]:
    parts = s.split("-")
    if len(parts) == 1:
        return parse_price(parts[0]), float("inf")
    lo, hi = parse_price(parts[0]), parse_price(parts[1])
    if lo > hi:
        raise ValueError(f"empty range: {s}")
    return lo, hi


# ---------------- model ----------------
@dataclass(frozen=True)
class Listing:
    id: int
    owner: str
    title: str
    description: str
    location: str
    nearby: FrozenSet[str]
    type: ListingType
    price: int
    size_sqft: int
    bhk: int
    status: Status = Status.AVAILABLE
    created_at: int = 0
    updated_at: int = 0

    def __post_init__(self):
        if self.price <= 0 or self.size_sqft <= 0 or self.bhk <= 0:
            raise ValueError("price, size and bhk must be > 0")
        object.__setattr__(self, "location", self.location.strip().lower())
        object.__setattr__(self, "nearby", frozenset(n.strip().lower() for n in self.nearby))


@dataclass(frozen=True)
class Hit:
    listing: Listing
    score: float
    nearby_only: bool


@dataclass(frozen=True)
class Page:
    items: List[Hit]
    total: int
    page: int
    size: int

    @property
    def has_next(self) -> bool:
        return (self.page + 1) * self.size < self.total


# ---------------- specification ----------------
class Spec:
    def __init__(self, fn: Callable[[Listing], bool]):
        self.fn = fn

    def __call__(self, x) -> bool:
        return self.fn(x)

    def __and__(self, o: "Spec") -> "Spec":
        return Spec(lambda x: self(x) and o(x))

    def __or__(self, o: "Spec") -> "Spec":
        return Spec(lambda x: self(x) or o(x))

    def __invert__(self) -> "Spec":
        return Spec(lambda x: not self(x))


def available() -> Spec: return Spec(lambda l: l.status is Status.AVAILABLE)
def of_type(t: ListingType) -> Spec: return Spec(lambda l: l.type is t)
def price_between(lo, hi) -> Spec: return Spec(lambda l: lo <= l.price <= hi)
def size_between(lo, hi) -> Spec: return Spec(lambda l: lo <= l.size_sqft <= hi)
def bhk_in(bhks) -> Spec: return Spec(lambda l: l.bhk in bhks)


def location_in(locs, include_nearby) -> Spec:
    return Spec(lambda l: l.location in locs or (include_nearby and bool(l.nearby & locs)))


# ---------------- ranking ----------------
SORTS: Dict[str, Callable[[Hit], tuple]] = {
    "relevance": lambda h: (-h.score, -h.listing.updated_at),
    "price": lambda h: (h.listing.price,),
    "size": lambda h: (h.listing.size_sqft,),
    "newest": lambda h: (-h.listing.updated_at,),
}


@dataclass
class SearchQuery:
    text: str = ""
    prefix_last: bool = False
    locations: Set[str] = field(default_factory=set)
    include_nearby: bool = False
    type: Optional[ListingType] = None
    price: Tuple[float, float] = (0, float("inf"))
    size: Tuple[float, float] = (0, float("inf"))
    bhks: Set[int] = field(default_factory=set)
    sort_by: str = "relevance"
    page: int = 0
    page_size: int = 10

    def __post_init__(self):
        if self.page < 0 or not 0 < self.page_size <= 100:
            raise ValueError(f"bad page {self.page}/{self.page_size}")
        if self.sort_by not in SORTS:
            raise ValueError(f"unknown sort {self.sort_by}")
        self.locations = {l.strip().lower() for l in self.locations}

    def to_spec(self) -> Spec:
        s = available()
        if self.type:
            s = s & of_type(self.type)
        if self.locations:
            s = s & location_in(self.locations, self.include_nearby)
        if self.price != (0, float("inf")):
            s = s & price_between(*self.price)
        if self.size != (0, float("inf")):
            s = s & size_between(*self.size)
        if self.bhks:
            s = s & bhk_in(self.bhks)
        return s

    def order(self) -> Callable[[Hit], tuple]:
        key = SORTS[self.sort_by]
        return lambda h: (h.nearby_only,) + key(h) + (h.listing.id,)


# ---------------- inverted index ----------------
TITLE_WEIGHT, BODY_WEIGHT = 3, 1


def tokenize(text: str) -> List[str]:
    return [t for t in re.split(r"[^a-z0-9]+", text.lower()) if t]


class TextIndex:
    """term -> {doc_id: weighted tf}, plus doc_id -> terms for exact removal. Not thread-safe alone."""

    def __init__(self):
        self.postings: Dict[str, Dict[int, int]] = {}
        self.forward: Dict[int, Dict[str, int]] = {}
        self._sorted_terms: Optional[List[str]] = None        # built lazily for prefix scans

    def add(self, doc_id: int, title: str, body: str = "") -> None:
        if doc_id in self.forward:
            self.remove(doc_id)
        tf: Dict[str, int] = {}
        for t in tokenize(title):
            tf[t] = tf.get(t, 0) + TITLE_WEIGHT
        for t in tokenize(body):
            tf[t] = tf.get(t, 0) + BODY_WEIGHT
        self.forward[doc_id] = tf
        for term, w in tf.items():
            self.postings.setdefault(term, {})[doc_id] = w
        self._sorted_terms = None

    def remove(self, doc_id: int) -> None:
        tf = self.forward.pop(doc_id, None)
        if tf is None:
            return
        for term in tf:
            p = self.postings[term]
            del p[doc_id]
            if not p:
                del self.postings[term]
        self._sorted_terms = None

    def _prefix(self, prefix: str) -> Dict[int, int]:
        if self._sorted_terms is None:
            self._sorted_terms = sorted(self.postings)
        merged: Dict[int, int] = {}
        i = bisect_left(self._sorted_terms, prefix)
        while i < len(self._sorted_terms) and self._sorted_terms[i].startswith(prefix):
            for d, w in self.postings[self._sorted_terms[i]].items():
                merged[d] = merged.get(d, 0) + w
            i += 1
        return merged

    def match_all(self, terms: List[str], prefix_last: bool = False) -> Dict[int, float]:
        if not terms:
            return {}
        lists = []
        for i, t in enumerate(terms):
            p = self._prefix(t) if prefix_last and i == len(terms) - 1 else self.postings.get(t)
            if not p:
                return {}
            lists.append(p)
        lists.sort(key=len)                                   # intersect from the rarest term
        out = {}
        for d, w in lists[0].items():
            ws = [p.get(d) for p in lists[1:]]
            if all(x is not None for x in ws):
                out[d] = float(w + sum(ws))
        return out


# ---------------- engine ----------------
class RWLock:
    """Many readers or one writer, writer-preferring: once a writer waits, new readers queue behind it,
    so a steady stream of readers cannot starve writers (Java's ReentrantReadWriteLock does the same)."""

    def __init__(self):
        self._cond = threading.Condition()
        self._readers = 0
        self._writer = False
        self._waiting_writers = 0                 # new readers yield to a queued writer

    def acquire_read(self):
        with self._cond:
            while self._writer or self._waiting_writers:
                self._cond.wait()
            self._readers += 1

    def release_read(self):
        with self._cond:
            self._readers -= 1
            if self._readers == 0:
                self._cond.notify_all()

    def acquire_write(self):
        with self._cond:
            self._waiting_writers += 1
            try:
                while self._writer or self._readers:
                    self._cond.wait()
            finally:
                self._waiting_writers -= 1
            self._writer = True

    def release_write(self):
        with self._cond:
            self._writer = False
            self._cond.notify_all()


class ListingSearchEngine:
    def __init__(self):
        self._rw = RWLock()
        self._by_id: Dict[int, Listing] = {}
        self._text = TextIndex()
        self._by_location: Dict[str, Set[int]] = {}

    def upsert(self, l: Listing) -> None:
        self._rw.acquire_write()
        try:
            old = self._by_id.get(l.id)
            if old:
                self._unindex_location(old)
            self._by_id[l.id] = l
            for k in {l.location} | l.nearby:
                self._by_location.setdefault(k, set()).add(l.id)
            self._text.add(l.id, l.title, l.description)
        finally:
            self._rw.release_write()

    def remove(self, listing_id: int) -> bool:
        self._rw.acquire_write()
        try:
            old = self._by_id.pop(listing_id, None)
            if old is None:
                return False
            self._unindex_location(old)
            self._text.remove(listing_id)
            return True
        finally:
            self._rw.release_write()

    def get(self, listing_id: int) -> Optional[Listing]:
        self._rw.acquire_read()
        try:
            return self._by_id.get(listing_id)
        finally:
            self._rw.release_read()

    def find(self, spec: Callable[[Listing], bool]) -> List[Listing]:
        self._rw.acquire_read()
        try:
            return sorted((l for l in self._by_id.values() if spec(l)), key=lambda l: l.id)
        finally:
            self._rw.release_read()

    def search(self, q: SearchQuery) -> Page:
        self._rw.acquire_read()
        try:
            terms = tokenize(q.text)
            scores = self._text.match_all(terms, q.prefix_last) if terms else None
            if scores is not None:
                candidates: Iterable[int] = scores.keys()
            elif q.locations:
                candidates = set().union(*(self._by_location.get(loc, set()) for loc in q.locations))
            else:
                candidates = self._by_id.keys()
            spec = q.to_spec()
            hits = [Hit(l, scores[i] if scores else 0.0, bool(q.locations) and l.location not in q.locations)
                    for i in candidates for l in (self._by_id[i],) if spec(l)]
        finally:
            self._rw.release_read()
        hits.sort(key=q.order())
        start = q.page * q.page_size
        return Page(hits[start:start + q.page_size], len(hits), q.page, q.page_size)

    def __len__(self) -> int:
        return len(self._by_id)

    def _unindex_location(self, l: Listing) -> None:
        for k in {l.location} | l.nearby:
            s = self._by_location.get(k)
            if s is not None:
                s.discard(l.id)
                if not s:
                    del self._by_location[k]


# ---------------- facade ----------------
class RentalPlatform:
    def __init__(self):
        self.engine = ListingSearchEngine()
        self._users: Set[str] = set()
        self._active: Set[str] = set()
        self._shortlists: Dict[str, List[int]] = {}
        self._ids = itertools.count(1)
        self._clock = itertools.count(1)
        self._lock = threading.Lock()

    def register(self, user):
        with self._lock:
            if user in self._users:
                raise ValueError(f"user exists: {user}")
            self._users.add(user)

    def login(self, user):
        if user not in self._users:
            raise ValueError(f"unknown user {user}")
        self._active.add(user)

    def logout(self, user):
        self._active.discard(user)

    def list(self, owner, title, description, location, nearby_csv, type_, price, size, bhk) -> Listing:
        self._require_active(owner)
        now = next(self._clock)
        l = Listing(next(self._ids), owner, title, description, location,
                    frozenset(n for n in nearby_csv.split(",") if n.strip()), type_,
                    parse_price(price), parse_size_sqft(size), bhk, Status.AVAILABLE, now, now)
        self.engine.upsert(l)
        return l

    def mark_sold(self, owner, listing_id):
        self.engine.upsert(replace(self._owned(owner, listing_id), status=Status.SOLD, updated_at=next(self._clock)))

    def change_price(self, owner, listing_id, price):
        self.engine.upsert(replace(self._owned(owner, listing_id), price=parse_price(price), updated_at=next(self._clock)))

    def search(self, q: SearchQuery) -> Page:
        return self.engine.search(q)

    def shortlist(self, user, listing_id):
        self._require_active(user)
        if self.engine.get(listing_id) is None:
            raise ValueError(f"no listing {listing_id}")
        with self._lock:
            s = self._shortlists.setdefault(user, [])
            if listing_id not in s:
                s.append(listing_id)

    def shortlisted(self, user) -> List[Listing]:
        with self._lock:
            ids = list(self._shortlists.get(user, []))
        return [l for l in (self.engine.get(i) for i in ids) if l]

    def listed_by(self, owner) -> List[Listing]:
        return self.engine.find(lambda l: l.owner == owner)

    def _owned(self, owner, listing_id) -> Listing:
        self._require_active(owner)
        l = self.engine.get(listing_id)
        if l is None:
            raise ValueError(f"no listing {listing_id}")
        if l.owner != owner:
            raise ValueError(f"{owner} does not own listing {listing_id}")
        return l

    def _require_active(self, user):
        if user not in self._active:
            raise PermissionError(f"{user} is not logged in")

def show(title, page):
    print(f"== {title} ({page.total} found) ==")
    for h in page.items:
        l = h.listing
        extra = (f"  score={h.score:.0f}" if h.score else "") + ("  (nearby)" if h.nearby_only else "")
        print(f"  #{l.id} {l.title:<32} {l.location:<11} {l.type.value} {l.price:>10,} {l.size_sqft:>5} sqft "
              f"{l.bhk}BHK{'  [SOLD]' if l.status.value == 'SOLD' else ''}{extra}")

class RentalTests(unittest.TestCase):
    def setUp(self):
        self.app = RentalPlatform()
        for u in ("asha", "ravi", "meera"):
            self.app.register(u)
            self.app.login(u)
        L = self.app.list
        self.a = L("asha", "2BHK near lake with balcony", "gym, pool", "Bellandur", "HSR Layout", R, "32k", "1100 sqft", 2)
        self.b = L("asha", "Spacious 3BHK", "gym, kids pool, pool view", "Bellandur", "", R, "45k", "1650", 3)
        self.c = L("ravi", "Cosy 1BHK", "metro", "HSR Layout", "", R, "18000", "550", 1)
        self.d = L("ravi", "2BHK budget flat", "no gym", "Bellandur", "", R, "24k", "100 sqm", 2)
        self.e = L("meera", "Villa", "garden, pool", "Whitefield", "", S, "1.8Cr", "2400", 3)

    def ids(self, page):
        return [h.listing.id for h in page.items]

    def test_units(self):
        self.assertEqual(parse_price("40k"), 40_000)
        self.assertEqual(parse_price("60L"), 6_000_000)
        self.assertEqual(parse_price("2Cr"), 20_000_000)
        self.assertEqual(parse_size_sqft("100 sqm"), 1076)
        self.assertEqual(parse_size_sqft("150 sqyd"), 1350)
        self.assertEqual(parse_price_range("50L"), (5_000_000, float("inf")))
        for bad in ("abc", "-5", "0"):
            with self.assertRaises(ValueError):
                parse_price(bad)
        with self.assertRaises(ValueError):
            parse_price_range("40k-20k")

    def test_combined_filters_and_price_sort(self):
        q = SearchQuery(locations={"Bellandur"}, type=R, price=parse_price_range("20k-40k"), bhks={2, 3}, sort_by="price")
        self.assertEqual(self.ids(self.app.search(q)), [self.d.id, self.a.id])

    def test_and_text_search_and_relevance(self):
        page = self.app.search(SearchQuery(text="pool gym"))
        self.assertEqual(set(self.ids(page)), {self.a.id, self.b.id})
        self.assertEqual(page.items[0].listing.id, self.b.id)          # pool twice

    def test_nearby_ranked_last(self):
        page = self.app.search(SearchQuery(locations={"hsr layout"}, include_nearby=True))
        self.assertEqual(self.ids(page), [self.c.id, self.a.id])
        self.assertTrue(page.items[1].nearby_only)

    def test_prefix_typeahead(self):
        self.assertEqual(self.ids(self.app.search(SearchQuery(text="balc", prefix_last=True))), [self.a.id])
        self.assertEqual(self.app.search(SearchQuery(text="balc")).total, 0)   # exact word only

    def test_pagination_is_stable_and_complete(self):
        seen, page = [], 0
        while True:
            p = self.app.search(SearchQuery(type=R, sort_by="newest", page=page, page_size=2))
            seen += self.ids(p)
            if not p.has_next:
                break
            page += 1
        self.assertEqual(seen, [self.d.id, self.c.id, self.b.id, self.a.id])
        with self.assertRaises(ValueError):
            SearchQuery(page=-1)

    def test_sold_hidden_from_search_but_kept_in_shortlist(self):
        self.app.shortlist("meera", self.a.id)
        self.app.mark_sold("asha", self.a.id)
        self.assertNotIn(self.a.id, self.ids(self.app.search(SearchQuery(locations={"bellandur"}))))
        self.assertEqual(self.app.shortlisted("meera")[0].status.value, "SOLD")
        self.assertEqual(len(self.app.listed_by("asha")), 2)

    def test_price_change_reindexes(self):
        self.app.change_price("ravi", self.d.id, "28k")
        self.assertEqual(self.app.search(SearchQuery(price=(20_000, 25_000))).total, 0)
        self.assertEqual(self.ids(self.app.search(SearchQuery(price=(26_000, 30_000)))), [self.d.id])

    def test_remove_cleans_every_index(self):
        self.assertTrue(self.app.engine.remove(self.c.id))
        self.assertEqual(self.app.search(SearchQuery(text="metro")).total, 0)
        self.assertEqual(self.app.search(SearchQuery(locations={"hsr layout"})).total, 0)
        self.assertFalse(self.app.engine.remove(self.c.id))

    def test_permissions(self):
        with self.assertRaises(ValueError):
            self.app.mark_sold("asha", self.c.id)
        self.app.logout("ravi")
        with self.assertRaises(PermissionError):
            self.app.mark_sold("ravi", self.c.id)

    def test_blog_variant_frequency_and_delete(self):
        idx = TextIndex()
        idx.add(1, "Kafka at scale", "kafka partitions and kafka consumers")
        idx.add(2, "Intro to Java", "java streams and kafka clients")
        r = idx.match_all(tokenize("kafka"))
        self.assertEqual(sorted(r, key=lambda d: -r[d]), [1, 2])
        self.assertEqual(set(idx.match_all(tokenize("java kafka"))), {2})
        idx.remove(1)
        self.assertEqual(idx.match_all(tokenize("partitions")), {})
        self.assertNotIn("partitions", idx.postings)

    def test_readers_never_see_half_updates(self):
        eng = ListingSearchEngine()
        for i in range(1, 101):
            eng.upsert(Listing(i, "o", f"flat {i}", "gym", "bellandur", frozenset(), R, 10_000 + (i * 397) % 80_000, 900, 2))
        stop, bad = threading.Event(), []

        def writer(seed):
            rnd, tick = random.Random(seed), 1000
            while not stop.is_set():
                old = eng.get(rnd.randint(1, 100))
                eng.upsert(replace(old, price=rnd.randint(10_000, 90_000), updated_at=tick))
                tick += 1

        def reader():
            q = SearchQuery(text="gym", price=(20_000, 40_000), page_size=100)
            while not stop.is_set():

class RWLockTests(unittest.TestCase):
    def test_new_reader_waits_behind_a_queued_writer(self):
        import time
        rw = RWLock()
        order = []
        rw.acquire_read()                                   # reader 1 holds the lock

        def writer():
            rw.acquire_write()
            order.append("writer")
            rw.release_write()

        def reader2():
            rw.acquire_read()
            order.append("reader2")
            rw.release_read()

        w = threading.Thread(target=writer)
        w.start()
        while rw._waiting_writers == 0:                     # the writer is queued
            time.sleep(0.001)
        r = threading.Thread(target=reader2)
        r.start()
        time.sleep(0.05)
        self.assertEqual(order, [])                         # reader 2 did not overtake the writer
        rw.release_read()
        w.join(2)
        r.join(2)
        self.assertEqual(order, ["writer", "reader2"])

enum class Type { RENT, SELL };

static std::string lower(std::string s) {
    for (auto& c : s) c = (char)std::tolower((unsigned char)c);
    return s;
}

static long long parsePrice(const std::string& in) {
    std::string t = lower(in);
    t.erase(std::remove(t.begin(), t.end(), ','), t.end());
    double mult = 1;
    auto ends = [&](const std::string& suf) { return t.size() > suf.size() && t.compare(t.size() - suf.size(), suf.size(), suf) == 0; };
    if (ends("cr")) { mult = 1e7; t.resize(t.size() - 2); }
    else if (ends("l")) { mult = 1e5; t.resize(t.size() - 1); }
    else if (ends("k")) { mult = 1e3; t.resize(t.size() - 1); }
    size_t used = 0;
    double v = 0;
    try { v = std::stod(t, &used); } catch (...) { throw std::invalid_argument("bad price: " + in); }
    if (used != t.size() || v <= 0) throw std::invalid_argument("bad price: " + in);
    return (long long)std::llround(v * mult);
}

static int parseSizeSqft(const std::string& in) {
    std::string t;
    for (char c : lower(in)) if (c != ' ' && c != '.') t += c;
    double f = 1;
    auto strip = [&](const std::string& suf, double fac) {
        if (t.size() > suf.size() && t.compare(t.size() - suf.size(), suf.size(), suf) == 0) { t.resize(t.size() - suf.size()); f = fac; return true; }
        return false;
    };
    if (!strip("sqm", 10.7639) && !strip("sqyd", 9.0)) strip("sqft", 1.0);
    return (int)std::lround(std::stod(t) * f);
}

struct Listing {
    long long id; std::string owner, title, description, location; std::set<std::string> nearby;

struct Hit { ListingPtr l; double score; bool nearbyOnly; };

// ---- Specification ----
using Spec = std::function<bool(const Listing&)>;
static Spec operator&&(Spec a, Spec b) { return [a, b](const Listing& l) { return a(l) && b(l); }; }

// ---- Inverted index ----
static std::vector<std::string> tokenize(const std::string& s) {
    std::vector<std::string> out; std::string cur;
    for (char c : lower(s)) {
        if (std::isalnum((unsigned char)c)) cur += c;
        else if (!cur.empty()) { out.push_back(cur); cur.clear(); }
    }
    if (!cur.empty()) out.push_back(cur);
    return out;
}

class TextIndex {
public:
    void add(long long id, const std::string& title, const std::string& body) {
        remove(id);
        std::map<std::string, int> tf;
        for (auto& t : tokenize(title)) tf[t] += 3;
        for (auto& t : tokenize(body)) tf[t] += 1;
        for (auto& kv : tf) postings_[kv.first][id] = kv.second;
        forward_[id] = tf;
    }
    void remove(long long id) {
        auto it = forward_.find(id);
        if (it == forward_.end()) return;
        for (auto& kv : it->second) {
            auto p = postings_.find(kv.first);
            p->second.erase(id);
            if (p->second.empty()) postings_.erase(p);
        }
        forward_.erase(it);
    }
    // AND of all terms; the last one may be a prefix. Empty result if any term is missing.
    std::unordered_map<long long, double> matchAll(const std::vector<std::string>& terms, bool prefixLast) const {
        std::vector<std::unordered_map<long long, int>> lists;
        for (size_t i = 0; i < terms.size(); ++i) {
            std::unordered_map<long long, int> p;
            if (prefixLast && i + 1 == terms.size()) {
                for (auto it = postings_.lower_bound(terms[i]); it != postings_.end() && it->first.compare(0, terms[i].size(), terms[i]) == 0; ++it)
                    for (auto& d : it->second) p[d.first] += d.second;
            } else {
                auto it = postings_.find(terms[i]);
                if (it != postings_.end()) p.insert(it->second.begin(), it->second.end());
            }
            if (p.empty()) return {};
            lists.push_back(std::move(p));
        }
        std::unordered_map<long long, double> out;
        if (lists.empty()) return out;
        std::sort(lists.begin(), lists.end(), [](const std::unordered_map<long long, int>& a, const std::unordered_map<long long, int>& b) { return a.size() < b.size(); });
        for (auto& d : lists[0]) {
            double s = d.second; bool all = true;
            for (size_t i = 1; i < lists.size() && all; ++i) {
                auto f = lists[i].find(d.first);
                if (f == lists[i].end()) all = false; else s += f->second;
            }
            if (all) out[d.first] = s;
        }
        return out;
    }
    size_t terms() const { return postings_.size(); }
private:
    std::map<std::string, std::map<long long, int>> postings_;
    std::unordered_map<long long, std::map<std::string, int>> forward_;
};

struct Query {
    std::string text; bool prefixLast = false;
    std::set<std::string> locations; bool includeNearby = false;
    bool anyType = true; Type type = Type::RENT;
    long long minPrice = 0, maxPrice = (1LL << 62);
    std::set<int> bhks;
    std::string sortBy = "relevance";
    int page = 0, pageSize = 10;
    Spec toSpec() const {
        Spec s = [](const Listing& l) { return l.status == Status::AVAILABLE; };
        if (!anyType) { Type t = type; s = s && Spec([t](const Listing& l) { return l.type == t; }); }
        if (!locations.empty()) {
            auto locs = locations; bool nb = includeNearby;
            s = s && Spec([locs, nb](const Listing& l) {
                if (locs.count(l.location)) return true;
                if (nb) for (auto& n : l.nearby) if (locs.count(n)) return true;
                return false;
            });
        }
        long long lo = minPrice, hi = maxPrice;
        s = s && Spec([lo, hi](const Listing& l) { return l.price >= lo && l.price <= hi; });
        if (!bhks.empty()) { auto b = bhks; s = s && Spec([b](const Listing& l) { return b.count(l.bhk) > 0; }); }
        return s;
    }
};

struct Page { std::vector<Hit> items; size_t total; bool hasNext; };

class Engine {
public:
    void upsert(const Listing& in) {                              // WRITE
        auto l = std::make_shared<const Listing>(in);
        auto old = byId_.find(l->id);
        if (old != byId_.end()) unindex(*old->second);
        byId_[l->id] = l;
        byLoc_[l->location].insert(l->id);
        for (auto& n : l->nearby) byLoc_[n].insert(l->id);
        text_.add(l->id, l->title, l->description);
    }
    bool remove(long long id) {                                   // WRITE
        auto it = byId_.find(id);
        if (it == byId_.end()) return false;
        unindex(*it->second);
        text_.remove(id);
        byId_.erase(it);
        return true;
    }
    ListingPtr get(long long id) const { auto it = byId_.find(id); return it == byId_.end() ? nullptr : it->second; }   // READ
    Page search(const Query& q) const {                           // READ
        if (q.page < 0 || q.pageSize <= 0) throw std::invalid_argument("bad page");
        auto terms = tokenize(q.text);
        bool useText = !terms.empty();
        auto scores = useText ? text_.matchAll(terms, q.prefixLast) : std::unordered_map<long long, double>();
        std::set<long long> cand;
        if (useText) for (auto& s : scores) cand.insert(s.first);
        else if (!q.locations.empty()) { for (auto& loc : q.locations) { auto f = byLoc_.find(loc); if (f != byLoc_.end()) cand.insert(f->second.begin(), f->second.end()); } }
        else for (auto& kv : byId_) cand.insert(kv.first);
        Spec spec = q.toSpec();
        std::vector<Hit> hits;
        for (long long id : cand) {
            ListingPtr l = byId_.at(id);
            if (!spec(*l)) continue;
            hits.push_back(Hit{l, useText ? scores.at(id) : 0.0, !q.locations.empty() && !q.locations.count(l->location)});
        }
        std::string by = q.sortBy;
        std::sort(hits.begin(), hits.end(), [&by](const Hit& a, const Hit& b) {   // nearby last, then the strategy, then id
            if (a.nearbyOnly != b.nearbyOnly) return !a.nearbyOnly;
            if (by == "price" && a.l->price != b.l->price) return a.l->price < b.l->price;
            if (by == "size" && a.l->sizeSqft != b.l->sizeSqft) return a.l->sizeSqft < b.l->sizeSqft;
            if (by == "newest" && a.l->updatedAt != b.l->updatedAt) return a.l->updatedAt > b.l->updatedAt;
            if (by == "relevance") {
                if (a.score != b.score) return a.score > b.score;
                if (a.l->updatedAt != b.l->updatedAt) return a.l->updatedAt > b.l->updatedAt;
            }
            return a.l->id < b.l->id;
        });
        size_t from = std::min(hits.size(), (size_t)q.page * q.pageSize), to = std::min(hits.size(), from + q.pageSize);
        return Page{std::vector<Hit>(hits.begin() + from, hits.begin() + to), hits.size(), to < hits.size()};
    }
    size_t terms() const { return text_.terms(); }
private:
    void unindex(const Listing& l) {
        std::set<std::string> keys(l.nearby);
        keys.insert(l.location);
        for (auto& k : keys) { auto f = byLoc_.find(k); if (f != byLoc_.end()) { f->second.erase(l.id); if (f->second.empty()) byLoc_.erase(f); } }
    }
    std::map<long long, ListingPtr> byId_;
    TextIndex text_;
    std::unordered_map<std::string, std::set<long long>> byLoc_;
};

static std::vector<long long> ids(const Page& p) {

}

From your world

The C++ Spec is a std::function and && builds a new one that calls both: a linked chain of predicate callbacks, like a netfilter hook list where each hook can return "drop" and the packet stops there. Using a std::map for postings buys sorted keys, so a prefix lookup is lower_bound(prefix) plus a forward walk, the same way a routing table finds the first entry that can match.
