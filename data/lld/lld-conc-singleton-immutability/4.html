Four correct fixes

class Config {
    private static volatile Config instance;     // the fix
    private final int port;
    private Config() { port = 8080; }

    static Config get() {
        Config local = instance;                 // read the volatile once (cheaper)
        if (local == null) {
            synchronized (Config.class) {
                local = instance;
                if (local == null) {
                    instance = local = new Config();
                }
            }
        }
        return local;
    }
}
class Config {
    private Config() { /* load */ }

    private static class Holder {                // not initialised until first use
        static final Config INSTANCE = new Config();
    }
    static Config get() { return Holder.INSTANCE; }
}
// The JVM initialises Holder exactly once, under its own class-init lock,
// the first time get() touches it. Lazy, thread-safe, no volatile, no lock after that.
enum Config {
    INSTANCE;
    private final int port = 8080;
    int port() { return port; }
}
// Thread-safe by class initialisation, and also safe against reflection
// and serialization making a second copy. Not lazy beyond class loading.
// (a) Simplest: a function-local static. Since C++11 its initialisation is thread-safe.
Registry& Registry::get() { static Registry r; return r; }

// (b) std::call_once: one-time init of something that is not a local static (like pthread_once).
std::once_flag driver_once;
Driver* driver = nullptr;
Driver& get_driver() {
    std::call_once(driver_once, [] { driver = new Driver{42}; });
    return *driver;
}

// (c) Hand-written DCL: atomic pointer with acquire/release, the C++ spelling of volatile.
std::atomic<Cache*> cache_ptr(nullptr);
std::mutex cache_mu;
Cache* get_cache() {
    Cache* p = cache_ptr.load(std::memory_order_acquire);
    if (!p) {
        std::lock_guard<std::mutex> g(cache_mu);
        p = cache_ptr.load(std::memory_order_relaxed);
        if (!p) {
            p = new Cache();                                // build fully ...
            cache_ptr.store(p, std::memory_order_release);  // ... then publish
        }
    }
    return p;
}
