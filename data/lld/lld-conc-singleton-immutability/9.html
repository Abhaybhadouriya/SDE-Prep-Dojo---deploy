public class SafePublicationDemo {

    static final AtomicInteger constructed = new AtomicInteger();

    // ---- BROKEN: double-checked locking WITHOUT volatile (shown for teaching, never use) ----
    static class BrokenDcl {
        private static BrokenDcl instance;          // not volatile: the bug
        private int port;
        private BrokenDcl() { port = 8080; }
        static BrokenDcl get() {
            if (instance == null) {                 // 1st check, no lock: may see a half-built object
                synchronized (BrokenDcl.class) {
                    if (instance == null) instance = new BrokenDcl();
                }
            }
            return instance;
        }
    }

    // ---- FIX 1: DCL with volatile ----
    static class VolatileDcl {
        private static volatile VolatileDcl instance;
        private final int port;
        private VolatileDcl() { constructed.incrementAndGet(); port = 8080; }
        static VolatileDcl get() {
            VolatileDcl local = instance;           // one volatile read on the fast path
            if (local == null) {
                synchronized (VolatileDcl.class) {
                    local = instance;
                    if (local == null) instance = local = new VolatileDcl();
                }
            }
            return local;
        }
    }

    // ---- FIX 2: initialization-on-demand holder ----
    static class HolderSingleton {
        private HolderSingleton() { constructed.incrementAndGet(); }
        private static class Holder { static final HolderSingleton INSTANCE = new HolderSingleton(); }
        static HolderSingleton get() { return Holder.INSTANCE; }   // class init runs once, under a JVM lock
    }

    // ---- FIX 3: enum ----
    enum EnumSingleton {
        INSTANCE;
        EnumSingleton() { constructed.incrementAndGet(); }
    }

    // ---- Immutable value object ----
    record Money(long paise, String currency) {
        Money {                                    // compact constructor validates once
            if (paise < 0) throw new IllegalArgumentException("negative");
            if (currency == null || currency.length() != 3) throw new IllegalArgumentException("currency");
        }
        Money plus(Money o) {                      // "modify" = return a new object
            if (!currency.equals(o.currency)) throw new IllegalArgumentException("mixed currency");
            return new Money(paise + o.paise, currency);
        }
    }

    // ---- Immutable class with a collection: defensive copies in and out ----
    static final class Route {
        private final String id;
        private final List<String> stops;
        Route(String id, List<String> stops) {
            this.id = id;
            this.stops = List.copyOf(stops);        // copy in: caller's list can change later, ours cannot
        }
        List<String> stops() { return stops; }     // already unmodifiable, safe to hand out
        Route withStop(String s) {
            List<String> next = new ArrayList<>(stops);
            next.add(s);
            return new Route(id, next);
        }
    }

    // ---- Copy-on-write config: readers never lock, writers swap a new immutable snapshot ----
    record Config(int version, Map<String, String> values) {
        Config { values = Map.copyOf(values); }
        Config with(String k, String v) {
            Map<String, String> m = new HashMap<>(values);
            m.put(k, v);
            return new Config(version + 1, m);
        }
    }
    static final class ConfigStore {
        private final AtomicReference<Config> current = new AtomicReference<>(new Config(0, Map.of()));
        Config snapshot() { return current.get(); }                    // one volatile read, no lock
        void put(String k, String v) { current.updateAndGet(c -> c.with(k, v)); }   // CAS loop, retries on clash
    }

    static int distinct(Supplier<Object> s) throws InterruptedException {
        Set<Object> seen = ConcurrentHashMap.newKeySet();
        CountDownLatch go = new CountDownLatch(1);
        List<Thread> ts = new ArrayList<>();
        for (int i = 0; i < 32; i++) ts.add(Thread.ofPlatform().start(() -> {
            try { go.await(); } catch (InterruptedException e) { return; }
            seen.add(s.get());
        }));
        go.countDown();
        for (Thread t : ts) t.join();
        return seen.size();
    }

                + " (the bug is a reordering the JMM allows, not something a quick test proves)");
        System.out.println("volatile DCL distinct objects = " + distinct(VolatileDcl::get));
        System.out.println("holder distinct objects       = " + distinct(HolderSingleton::get));
        System.out.println("enum distinct objects         = " + distinct(() -> EnumSingleton.INSTANCE));
        System.out.println("constructors run in total     = " + constructed.get() + " (expect 3: one per idiom)");

        Money a = new Money(10_000, "INR");
        Money b = a.plus(new Money(2_500, "INR"));
        System.out.println("Money a=" + a.paise() + " b=" + b.paise() + " (a unchanged)");

        List<String> mine = new ArrayList<>(List.of("BLR", "HYD"));
        Route r = new Route("R1", mine);
        mine.add("DEL");                              // caller mutates its own list afterwards
        System.out.println("route stops after caller edit = " + r.stops() + " (still 2: we copied)");
        try { r.stops().add("PUNE"); } catch (UnsupportedOperationException e) {
            System.out.println("route.stops().add -> UnsupportedOperationException (cannot mutate)");
        }
        System.out.println("withStop gives a new route: " + r.withStop("PUNE").stops());

        ConfigStore store = new ConfigStore();
        AtomicBoolean torn = new AtomicBoolean();
        AtomicBoolean stop = new AtomicBoolean();
        Thread reader = Thread.ofPlatform().start(() -> {
            while (!stop.get()) {
                Config c = store.snapshot();
                if (c.version() != c.values().size()) torn.set(true);   // invariant of every snapshot
            }
        });
        List<Thread> writers = new ArrayList<>();
        for (int w = 0; w < 4; w++) {
            final int id = w;
            writers.add(Thread.ofPlatform().start(() -> {
                for (int i = 0; i < 250; i++) store.put("k" + id + "-" + i, "v");
            }));
        }
        for (Thread t : writers) t.join();
        stop.set(true);
        reader.join();
        Config fin = store.snapshot();
        System.out.println("COW config: version=" + fin.version() + " keys=" + fin.values().size()
                + " (expect 1000, no lost update), reader saw a torn snapshot = " + torn.get());

        List<String> plain = new ArrayList<>(List.of("a", "b"));
        try {
            for (String s : plain) if (s.equals("a")) plain.add("c");
        } catch (ConcurrentModificationException e) {
            System.out.println("ArrayList: add during iteration -> ConcurrentModificationException");
        }
        List<Runnable> listeners = new CopyOnWriteArrayList<>();
        AtomicInteger calls = new AtomicInteger();
        listeners.add(calls::incrementAndGet);
        listeners.add(calls::incrementAndGet);
        for (Runnable l : listeners) { l.run(); listeners.add(calls::incrementAndGet); }  // iterates the old snapshot
        System.out.println("CopyOnWriteArrayList: iterated 2 listeners while adding, now size=" + listeners.size()
                + ", calls=" + calls.get());
    }

// Build and run:  g++ -std=c++14 -O2 -Wall -pthread once.cpp -o once && ./once
// Needs a POSIX-threads g++ (Linux, WSL, or MinGW-w64 "posix"). We built and ran it with g++ 13.3 in
// WSL Ubuntu 24.04. The MinGW.org g++ 6.3 on Windows (win32 thread model) has no std::thread or std::mutex.

#include <cstdio>
#include <map>
#include <memory>
#include <mutex>
#include <set>

#include <vector>

std::atomic<int> built(0);

// 1. Meyers singleton: C++11 guarantees a function-local static is initialised once ("magic statics").
struct Registry {
    static Registry& get() { static Registry r; return r; }
    Registry(const Registry&) = delete;
    Registry& operator=(const Registry&) = delete;
private:
    Registry() { ++built; }
};

// 2. std::call_once: one-time init of something that is not a local static (like pthread_once).
struct Driver { int irq; };
std::once_flag driver_once;
Driver* driver = nullptr;
Driver& get_driver() {
    std::call_once(driver_once, [] { driver = new Driver{42}; ++built; });
    return *driver;
}

// 3. DCL done right in C++: an atomic pointer with acquire/release ordering.
struct Cache { int size = 128; };
std::atomic<Cache*> cache_ptr(nullptr);
std::mutex cache_mu;
Cache* get_cache() {
    Cache* p = cache_ptr.load(std::memory_order_acquire);   // pairs with the release store
    if (!p) {
        std::lock_guard<std::mutex> g(cache_mu);
        p = cache_ptr.load(std::memory_order_relaxed);
        if (!p) {
            p = new Cache();                                // fully build first ...
            ++built;
            cache_ptr.store(p, std::memory_order_release);  // ... then publish
        }
    }
    return p;
}

// 4. Copy-on-write config: readers grab a shared_ptr<const Config>, writers swap in a new one.
//    std::atomic_load/atomic_store on shared_ptr exist since C++11 (C++20 adds atomic<shared_ptr>).
struct Config {
    int version;
    std::map<std::string, std::string> values;
};
std::shared_ptr<const Config> current = std::make_shared<const Config>(Config{0, {}});
std::mutex writer_mu;   // writers serialise among themselves; readers never take it

std::shared_ptr<const Config> snapshot() { return std::atomic_load(&current); }
void put(const std::string& k, const std::string& v) {
    std::lock_guard<std::mutex> g(writer_mu);
    auto old = std::atomic_load(&current);
    auto next = std::make_shared<Config>(*old);   // copy ...
    next->values[k] = v;                          // ... modify the copy ...
    next->version = old->version + 1;
    std::atomic_store(&current, std::shared_ptr<const Config>(next));   // ... publish
}

"""Thread-safe singleton, immutable values and copy-on-write config in Python 3.11.
Run:  python immutability.py
"""
import threading
from dataclasses import dataclass, FrozenInstanceError, replace
from types import MappingProxyType


class Settings:
    """Double-checked singleton. In CPython the GIL hides most reordering, but the lock is still
    needed: two threads can both pass the first check before either assigns."""
    _instance = None
    _lock = threading.Lock()
    built = 0

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    obj = super().__new__(cls)
                    Settings.built += 1
                    cls._instance = obj
        return cls._instance


@dataclass(frozen=True)
class Money:
    paise: int
    currency: str

    def __post_init__(self):
        if self.paise < 0 or len(self.currency) != 3:
            raise ValueError("bad money")

    def plus(self, other: "Money") -> "Money":
        return replace(self, paise=self.paise + other.paise)


@dataclass(frozen=True)
class Config:
    version: int
    values: MappingProxyType          # read-only view over a private dict copy

    @staticmethod
    def empty():
        return Config(0, MappingProxyType({}))

    def with_(self, k, v):
        d = dict(self.values)
        d[k] = v
        return Config(self.version + 1, MappingProxyType(d))


class ConfigStore:
    def __init__(self):
        self._current = Config.empty()
        self._write = threading.Lock()

    def snapshot(self):
        return self._current            # a single reference read: atomic in CPython

    def put(self, k, v):
        with self._write:               # writers serialise; readers never lock
            self._current = self._current.with_(k, v)


def main():
    seen = set()
    guard = threading.Lock()

    def grab():
        s = Settings()
        with guard:
            seen.add(id(s))

    ts = [threading.Thread(target=grab) for _ in range(32)]
    for t in ts:
        t.start()
    for t in ts:
        t.join()
    print("singleton distinct =", len(seen), "built =", Settings.built)

    a = Money(10_000, "INR")
    b = a.plus(Money(2_500, "INR"))
    try:
        a.paise = 1
    except FrozenInstanceError:
        print("frozen dataclass: assignment raises FrozenInstanceError")
    print("a =", a.paise, "b =", b.paise)

    store = ConfigStore()
    stop = threading.Event()
    torn = []

    def reader():
        while not stop.is_set():
            c = store.snapshot()
            if c.version != len(c.values):
                torn.append(c.version)

    r = threading.Thread(target=reader)
    r.start()
    ws = [threading.Thread(target=lambda w=w: [store.put(f"k{w}-{i}", "v") for i in range(250)]) for w in range(4)]
    for t in ws:
        t.start()
    for t in ws:
        t.join()
    stop.set()
    r.join()
    fin = store.snapshot()
    print("COW config version =", fin.version, "keys =", len(fin.values), "torn =", bool(torn))
    try:
        fin.values["x"] = "y"
    except TypeError:
        print("MappingProxyType: item assignment raises TypeError")


if __name__ == "__main__":
    main()
