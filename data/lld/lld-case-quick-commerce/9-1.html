Build it two ways from the same file. The output shown on this page is from Windows MinGW.org g++ 6.3, where the lock class is a no-op with the same lock()/unlock(); on Linux, define HAVE_STD_MUTEX and the same lock sites use a real std::mutex through a scoped guard. Where two locks are held at once they are taken in one fixed order (sorted ids, or outer object first), which is the rule std::lock would otherwise enforce for you:

$ g++ -std=c++14 -Wall -Wextra main.cpp -o main            # MinGW g++ 6.3: single-threaded build
$ g++ -std=c++14 -Wall -Wextra -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o main && ./main   # Linux: real locks

Say it plainly in an interview: MinGW.org g++ 6.3 uses the "win32" thread model, ships no std::mutex or std::thread, and -pthread fails to link there. The -DHAVE_STD_MUTEX build of this file was compiled and run with g++ 13.3 on Linux (WSL Ubuntu 24.04); its last line was ALL 11 CHECKS PASSED.

// Quick-commerce order + inventory (Zepto style) - C++14.
// Windows (MinGW g++ 6.3):  g++ -std=c++14 -Wall -O2 main.cpp -o qc && ./qc
// Linux (real locks + race): g++ -std=c++14 -Wall -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o qc && ./qc
// MinGW.org g++ 6.3 uses the win32 thread model: it has no std::mutex/std::thread and -pthread fails to
// link. Without HAVE_STD_MUTEX, Mutex below is a no-op with the same lock()/unlock() interface, so the
// SAME locking code compiles both ways; only the Linux build runs the 20-thread race.
#include <algorithm>
#include <cmath>
#include <cstdlib>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>
#ifdef HAVE_STD_MUTEX
#include <thread>
typedef std::mutex Mutex;
typedef std::recursive_mutex RecursiveMutex;           // the order lock is re-taken by expire()
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
typedef Mutex RecursiveMutex;
#endif
typedef std::lock_guard<Mutex> LockGuard;

using std::map; using std::string; using std::vector;
typedef map<string, int> Lines;                         // sku -> qty; std::map iterates in sorted order

struct QcError : std::runtime_error { explicit QcError(const string& m) : std::runtime_error(m) {} };

// One SKU in one store. Invariant: 0 <= reserved <= onHand. Fields guarded by mu.
struct StockItem {
    Mutex mu; int onHand = 0, reserved = 0;
    int available() { LockGuard g(mu); return onHand - reserved; }
    void restock(int q) { LockGuard g(mu); onHand += q; }
};

class DarkStore {
public:
    const string id; const double x, y;
    DarkStore(string i, double px, double py) : id(std::move(i)), x(px), y(py) {}
    StockItem& item(const string& sku) {                // the map itself has its own lock (like ConcurrentHashMap)
        LockGuard g(mapMu_);
        std::unique_ptr<StockItem>& p = stock_[sku];
        if (!p) p.reset(new StockItem());
        return *p;
    }
    double distanceTo(double cx, double cy) const { return std::hypot(x - cx, y - cy); }
    bool hasAll(const Lines& l) {                       // a hint only; reserveAll re-checks under the locks
        for (const auto& kv : l) if (item(kv.first).available() < kv.second) return false;
        return true;
    }
    bool reserveAll(const Lines& l) {
        vector<StockItem*> items = lockSorted(l);
        bool ok = true;
        for (const auto& kv : l) if (item(kv.first).onHand - item(kv.first).reserved < kv.second) ok = false;
        if (ok) for (const auto& kv : l) item(kv.first).reserved += kv.second;
        unlock(items);
        return ok;
    }
    void releaseAll(const Lines& l) {
        vector<StockItem*> items = lockSorted(l);
        for (const auto& kv : l) item(kv.first).reserved -= kv.second;
        unlock(items);
    }
    void commitAll(const Lines& l) {
        vector<StockItem*> items = lockSorted(l);
        for (const auto& kv : l) { StockItem& s = item(kv.first); s.reserved -= kv.second; s.onHand -= kv.second; }
        unlock(items);
    }
private:
    Mutex mapMu_;
    map<string, std::unique_ptr<StockItem>> stock_;
    // One global order (SKU name, which is std::map order) means two baskets never deadlock.
    vector<StockItem*> lockSorted(const Lines& l) {
        vector<StockItem*> items;
        for (const auto& kv : l) items.push_back(&item(kv.first));
        for (StockItem* s : items) s->mu.lock();
        return items;
    }
    static void unlock(vector<StockItem*>& items) { for (auto it = items.rbegin(); it != items.rend(); ++it) (*it)->mu.unlock(); }
};

enum class St { CREATED, CONFIRMED, PACKED, OUT_FOR_DELIVERY, DELIVERED, CANCELLED, EXPIRED };
const char* name(St s) {
    static const char* n[] = {"CREATED", "CONFIRMED", "PACKED", "OUT_FOR_DELIVERY", "DELIVERED", "CANCELLED", "EXPIRED"};
    return n[static_cast<int>(s)];
}
bool canMove(St a, St b) {
    switch (a) {
        case St::CREATED:          return b == St::CONFIRMED || b == St::CANCELLED || b == St::EXPIRED;
        case St::CONFIRMED:        return b == St::PACKED || b == St::CANCELLED;
        case St::PACKED:           return b == St::OUT_FOR_DELIVERY || b == St::CANCELLED;
        case St::OUT_FOR_DELIVERY: return b == St::DELIVERED;
        default:                   return false;
    }
}

struct Order {
    string id, customer; DarkStore* store; Lines lines; int total; long long expiresAt;
    St status = St::CREATED;
    RecursiveMutex mu;                                  // per-order lock: state change + its side effect
    Order(string i, string c, DarkStore* s, Lines l, int t, long long e)
        : id(std::move(i)), customer(std::move(c)), store(s), lines(std::move(l)), total(t), expiresAt(e) {}
    St moveTo(St n) {                                   // caller holds mu
        if (!canMove(status, n)) throw QcError("order " + id + ": " + name(status) + " -> " + name(n) + " not allowed");
        St old = status; status = n; return old;
    }
};
typedef std::lock_guard<RecursiveMutex> OrderGuard;

class QuickCommerce {
public:
    QuickCommerce(double maxKm, long long holdMs) : maxKm_(maxKm), holdMs_(holdMs) {}
    void addProduct(const string& sku, int price) { LockGuard g(regMu_); price_[sku] = price; }
    void addStore(const string& id, double x, double y) { LockGuard g(regMu_); stores_.emplace_back(new DarkStore(id, x, y)); }
    void restock(const string& sid, const string& sku, int q) {
        { LockGuard g(regMu_); if (!price_.count(sku) || q <= 0) throw QcError("bad restock"); }
        store(sid).item(sku).restock(q);
    }
    void advance(long long ms) { LockGuard g(regMu_); now_ += ms; }

    Order& createOrder(const string& cust, double cx, double cy, const Lines& lines) {
        if (lines.empty()) throw QcError("empty basket");
        int total = 0;
        vector<DarkStore*> cand;
        long long now;
        {
            LockGuard g(regMu_);
            for (const auto& kv : lines) {
                if (!price_.count(kv.first) || kv.second <= 0) throw QcError("bad line " + kv.first);
                total += price_[kv.first] * kv.second;
            }
            for (auto& s : stores_) cand.push_back(s.get());
            now = now_;
        }
        for (;;) {
            DarkStore* best = nullptr;                 // Strategy: nearest store with the full basket in range
            for (DarkStore* s : cand)
                if (s->distanceTo(cx, cy) <= maxKm_ && s->hasAll(lines) &&
                    (!best || s->distanceTo(cx, cy) < best->distanceTo(cx, cy))) best = s;
            if (!best) throw QcError("out of stock nearby");
            if (best->reserveAll(lines)) {             // the real check, under the SKU locks
                LockGuard g(regMu_);
                string id = "q" + std::to_string(++seq_);
                orders_[id].reset(new Order(id, cust, best, lines, total, now + holdMs_));
                return *orders_[id];
            }
            cand.erase(std::find(cand.begin(), cand.end(), best));   // stock moved: try the next store
        }
    }
    // Acts only while CREATED. A duplicate or late webhook changes nothing: a failure that arrives
    // after success must not release stock that was already committed.
    St onPayment(const string& oid, bool ok) {
        Order& o = order(oid);
        OrderGuard g(o.mu);                            // lock order: order, then SKUs
        if (o.status == St::CREATED && now() > o.expiresAt) expire(o);
        if (o.status != St::CREATED) {                 // duplicate or out-of-order webhook
            if (ok && (o.status == St::EXPIRED || o.status == St::CANCELLED))
                throw QcError("order " + oid + " is " + name(o.status) + ": refund the payment");
            return o.status;
        }
        if (ok) { o.moveTo(St::CONFIRMED); o.store->commitAll(o.lines); }
        else    { o.moveTo(St::CANCELLED); o.store->releaseAll(o.lines); }
        return o.status;
    }
    void pack(const string& oid) { Order& o = order(oid); OrderGuard g(o.mu); o.moveTo(St::PACKED); }
    void dispatch(const string& oid) { Order& o = order(oid); OrderGuard g(o.mu); o.moveTo(St::OUT_FOR_DELIVERY); }
    void cancel(const string& oid) {
        Order& o = order(oid);
        OrderGuard g(o.mu);                            // state change and side effect in one critical section
        St old = o.moveTo(St::CANCELLED);
        if (old == St::CREATED) o.store->releaseAll(o.lines);
        else if (old == St::CONFIRMED || old == St::PACKED)
            for (const auto& kv : o.lines) o.store->item(kv.first).restock(kv.second);
    }
    int sweepExpired() {
        vector<Order*> snap;
        { LockGuard g(regMu_); for (auto& kv : orders_) snap.push_back(kv.second.get()); }
        int n = 0;
        for (Order* o : snap) if (expireIfDue(*o)) ++n;
        return n;
    }
    Order& order(const string& id) {
        LockGuard g(regMu_);
        auto it = orders_.find(id); if (it == orders_.end()) throw QcError("unknown order " + id); return *it->second;
    }
    DarkStore& store(const string& id) {
        LockGuard g(regMu_);
        for (auto& s : stores_) if (s->id == id) return *s;
        throw QcError("unknown store " + id);
    }
private:
    double maxKm_; long long holdMs_, now_ = 0; int seq_ = 0;
    Mutex regMu_;                                      // guards the registries, the clock and seq_
    map<string, int> price_;
    vector<std::unique_ptr<DarkStore>> stores_;
    map<string, std::unique_ptr<Order>> orders_;
    long long now() { LockGuard g(regMu_); return now_; }
    bool expireIfDue(Order& o) {
        OrderGuard g(o.mu);
        return o.status == St::CREATED && now() > o.expiresAt && expire(o);
    }
    bool expire(Order& o) {                            // caller holds o.mu
        if (!canMove(o.status, St::EXPIRED)) return false;
        o.moveTo(St::EXPIRED);
        o.store->releaseAll(o.lines);
        return true;
    }
};

static int checks = 0;
static void check(bool ok, const char* what) { if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); } ++checks; }
template <class F> static void expectFail(F f) {
    try { f(); check(false, "should fail"); } catch (const QcError& e) { std::cout << "rejected: " << e.what() << "\n"; ++checks; }
}

int main() {
    QuickCommerce qc(3.0, 5 * 60000);
    qc.addProduct("milk", 60); qc.addProduct("bread", 40); qc.addProduct("eggs", 50);
    qc.addStore("DS-near", 0.5, 0.5); qc.addStore("DS-far", 2.0, 1.5);
    qc.restock("DS-near", "milk", 2); qc.restock("DS-near", "bread", 10);
    qc.restock("DS-far", "milk", 10); qc.restock("DS-far", "bread", 10); qc.restock("DS-far", "eggs", 10);

    Order& q1 = qc.createOrder("asha", 0, 0, {{"milk", 2}, {"bread", 1}});
    StockItem& milk = q1.store->item("milk");
    std::cout << "CREATE " << q1.id << " at " << q1.store->id << " total=" << q1.total
              << " milk onHand=" << milk.onHand << " reserved=" << milk.reserved << "\n";
    check(q1.store->id == "DS-near" && q1.total == 160 && milk.available() == 0, "reserved at nearest");
    Order& q2 = qc.createOrder("ravi", 0, 0, {{"milk", 1}, {"eggs", 1}});
    std::cout << "CREATE " << q2.id << " at " << q2.store->id << "\n";
    check(q2.store->id == "DS-far", "fallback store");
    expectFail([&] { qc.createOrder("far-away", 9, 9, {{"milk", 1}}); });
    expectFail([&] { qc.createOrder("asha", 0, 0, {{"caviar", 1}}); });
    qc.onPayment(q1.id, true);
    check(qc.onPayment(q1.id, true) == St::CONFIRMED && milk.onHand == 0 && milk.reserved == 0, "deduct once");
    check(qc.onPayment(q1.id, false) == St::CONFIRMED && milk.onHand == 0 && milk.reserved == 0,
          "late failure webhook after success leaves counters unchanged");
    std::cout << "LATE failure webhook for " << q1.id << " -> " << name(q1.status) << "\n";
    qc.onPayment(q2.id, false);
    check(q2.store->item("eggs").available() == 10, "failed payment releases");
    qc.pack(q1.id);
    qc.cancel(q1.id);
    check(milk.available() == 2, "cancel after pack puts milk back on the shelf");
    std::cout << "CANCEL " << q1.id << " after packing -> milk available=" << milk.available() << "\n";
    Order& q3 = qc.createOrder("meera", 0, 0, {{"bread", 4}});
    qc.advance(6 * 60000);
    check(qc.sweepExpired() == 1 && q3.status == St::EXPIRED, "expired");
    std::cout << "after 6 min: " << q3.id << "=" << name(q3.status) << " bread available="
              << q3.store->item("bread").available() << "\n";
    expectFail([&] { qc.onPayment(q3.id, true); });
#ifdef HAVE_STD_MUTEX
    {   // 20 shoppers, 5 milk; half the baskets list bread first (std::map sorts them anyway)
        QuickCommerce race(5, 60000);
        race.addProduct("milk", 60); race.addProduct("bread", 40);
        race.addStore("DS-race", 0, 0);
        race.restock("DS-race", "milk", 5); race.restock("DS-race", "bread", 100);
        Mutex wonMu; int won = 0;
        vector<std::thread> ts;
        for (int i = 0; i < 20; ++i)
            ts.emplace_back([&, i] {
                try { race.createOrder("u" + std::to_string(i), 0, 0, {{"milk", 1}, {"bread", 1}}); LockGuard g(wonMu); ++won; }
                catch (const QcError&) {}
            });
        for (auto& t : ts) t.join();
        StockItem& m = race.store("DS-race").item("milk");
        std::cout << "RACE 20 threads: orders=" << won << " milk reserved=" << m.reserved << "\n";
        check(won == 5 && m.available() == 0 && m.reserved == 5, "never oversold");
    }
#endif
    std::cout << "ALL " << checks << " CHECKS PASSED\n";
    return 0;
}
