Before and after: splitting into role interfaces

The fix is to cut the fat interface along the lines of who calls what. Each piece is called a role interface: an interface that describes one role a client needs, not everything an object can do.

// One small interface per role a client needs.
interface Printer { void print(String doc); }
interface Scanner { String scan(String doc); }
interface Fax     { void fax(String doc, String number); }

// A cheap printer implements only what it can really do.
static class SimplePrinter implements Printer {
    public void print(String doc) { System.out.println("[simple] printing " + doc); }
}

// The office machine plays all three roles.
static class OfficeMachine implements Printer, Scanner, Fax {
    public void print(String doc) { System.out.println("[office] printing " + doc); }
    public String scan(String doc) { return "scan-of-" + doc; }
    public void fax(String doc, String number) { System.out.println("[office] fax " + doc + " -> " + number); }
}

// Clients ask only for the role they use.
static class PrintQueue {
    private final Printer printer;
    private final List<String> jobs = new ArrayList<>();
    PrintQueue(Printer printer) { this.printer = printer; }
    void add(String doc) { jobs.add(doc); }
    int flush() { int n = jobs.size(); jobs.forEach(printer::print); jobs.clear(); return n; }
}

static class Archiver {
    private final Scanner scanner;
    Archiver(Scanner scanner) { this.scanner = scanner; }
    String archive(String doc) { return "archived:" + scanner.scan(doc); }
}

// The same OfficeMachine object can be handed to both:
static void wireBoth() {
    OfficeMachine office = new OfficeMachine();
    PrintQueue queue = new PrintQueue(office);   // sees only Printer
    Archiver archiver = new Archiver(office);    // sees only Scanner
    queue.add("d.pdf"); queue.flush();
    System.out.println(archiver.archive("e.pdf"));
}
interface Machine {
    void print(String doc);
    String scan(String doc);
    void fax(String doc, String number);
}

static class CheapPrinter implements Machine {
    public void print(String doc) { System.out.println("[fat] printing " + doc); }
    public String scan(String doc) { throw new UnsupportedOperationException("no scanner"); }
    public void fax(String doc, String number) { throw new UnsupportedOperationException("no fax"); }
}

static class FatPrintQueue {
    private final Machine machine;      // sees scan() and fax() it never calls
    FatPrintQueue(Machine m) { this.machine = m; }
    void print(String doc) { machine.print(doc); }
}
class Printer(Protocol):
    def print_doc(self, doc: str) -> None: ...

class Scanner(Protocol):
    def scan(self, doc: str) -> str: ...

class Fax(Protocol):
    def fax(self, doc: str, number: str) -> None: ...

class SimplePrinter:                      # structurally a Printer
    def print_doc(self, doc: str) -> None:
        print("printing", doc)

class PrintQueue:
    def __init__(self, printer: Printer) -> None:   # asks only for the role
        self._printer = printer
        self._jobs: list[str] = []

    def add(self, doc: str) -> None:
        self._jobs.append(doc)

    def flush(self) -> int:
        n = len(self._jobs)
        for doc in self._jobs:
            self._printer.print_doc(doc)
        self._jobs.clear()
        return n

The signatures are the same on every principles page: print(doc) returns nothing, scan(doc) returns the scanned text, fax(doc, number) returns nothing. It prints the runtime surprise from the fat version and then shows the role-interface version where the impossible call simply does not compile.

Say it like this: "I split the interface by client, not by implementer. The print queue only needs print, so it depends on Printer. The office machine implements all three roles, so nothing is lost - one object, three narrow views."
