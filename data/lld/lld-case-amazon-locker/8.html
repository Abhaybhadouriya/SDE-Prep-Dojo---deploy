  1. Enums and data classes. Size knows what fits where; Slot is one physical box.
  2. One interface, one implementation. SlotAllocationStrategy picks a slot from the free lists; BestFitStrategy upgrades only when the exact size is full.
  3. The core. deposit is one synchronized method on the station: pick a slot, mark it, issue a code, index it.
  4. The driver. main uses a fake clock and an inbox listener so every rule is checkable.

import java.util.*;

/** Package and slot sizes. A parcel fits any slot of its own size or bigger. */
enum Size {
    SMALL, MEDIUM, LARGE;

    boolean fitsIn(Size slot) { return slot.ordinal() >= ordinal(); }

    /** Classify a box by its longest side in cm (an assumption to agree with the interviewer). */
    static Size forBox(int lengthCm, int widthCm, int heightCm) {
        int longest = Math.max(lengthCm, Math.max(widthCm, heightCm));
        if (longest <= 30) return SMALL;
        if (longest <= 50) return MEDIUM;
        if (longest <= 80) return LARGE;
        throw new LockerException(LockerException.Reason.TOO_BIG, longest + " cm does not fit any slot");
    }
}

enum SlotState { FREE, OCCUPIED, OUT_OF_SERVICE }

enum DeliveryState { DEPOSITED, PICKED_UP, EXPIRED, RETURNED }

/** One physical compartment with a door. Mutable; guarded by its station's lock. */
final class Slot {
    final String id;
    final Size size;
    SlotState state = SlotState.FREE;

    Slot(String id, Size size) { this.id = id; this.size = size; }

    @Override public String toString() { return id + "(" + size + "," + state + ")"; }
}

/** What the courier brings. Immutable. */
final class Parcel {
    final String id;
    final String customerId;
    final Size size;

    Parcel(String id, String customerId, Size size) {
        this.id = Objects.requireNonNull(id);
        this.customerId = Objects.requireNonNull(customerId);
        this.size = Objects.requireNonNull(size);
    }

    @Override public String toString() { return id + "/" + size; }
}

/** The run-time record that ties a parcel to a slot and a pickup code. Guarded by the station lock. */
final class Delivery {
    final Parcel parcel;
    final Slot slot;
    final String code;
    final long depositedAt;
    final long expiresAt;
    DeliveryState state = DeliveryState.DEPOSITED;

    Delivery(Parcel parcel, Slot slot, String code, long depositedAt, long expiresAt) {
        this.parcel = parcel; this.slot = slot; this.code = code;
        this.depositedAt = depositedAt; this.expiresAt = expiresAt;
    }
}

/** What the courier sees after a deposit: which door opened. The code goes to the customer, not here. */
final class DepositReceipt {
    final String stationId, slotId, parcelId;
    final Size slotSize;
    final long expiresAt;

    DepositReceipt(String stationId, String slotId, Size slotSize, String parcelId, long expiresAt) {
        this.stationId = stationId; this.slotId = slotId; this.slotSize = slotSize;
        this.parcelId = parcelId; this.expiresAt = expiresAt;
    }
}

/** What the customer sees after a pickup: which door opened and which parcel is inside. */
final class PickupResult {
    final String slotId;
    final Parcel parcel;

    PickupResult(String slotId, Parcel parcel) { this.slotId = slotId; this.parcel = parcel; }
}

final class LockerException extends RuntimeException {
    enum Reason { UNKNOWN_STATION, UNKNOWN_SLOT, NO_SLOT, TOO_BIG, DUPLICATE_PARCEL, INVALID_CODE, CODE_EXPIRED, KEYPAD_LOCKED, SLOT_BUSY }

    final Reason reason;

    LockerException(Reason reason, String message) { super(reason + ": " + message); this.reason = reason; }
}

/** Time source. Inject a fake in tests; milliseconds. */
interface Clock { long now(); }

/** Strategy: how a pickup code is produced. SecureRandom in production, a seeded one in tests. */
interface CodeGenerator { String next(); }

final class RandomCodeGenerator implements CodeGenerator {
    private final Random random;
    private final int digits;

    RandomCodeGenerator(Random random, int digits) { this.random = random; this.digits = digits; }

    @Override public String next() {
        StringBuilder sb = new StringBuilder(digits);
        for (int i = 0; i < digits; i++) sb.append((char) ('0' + random.nextInt(10)));
        return sb.toString();
    }
}

/** Observer: notifications, metrics and audit hang off these events. Called outside every lock. */
interface LockerListener {
    default void onDeposited(String stationId, Delivery d) {}   // C: default method = an ops-table slot with a generic body
    default void onPickedUp(String stationId, Delivery d) {}
    default void onExpired(String stationId, Delivery d) {}
}
import java.util.*;

/** Read-only view of a station's free slots, handed to a strategy while the station lock is held. */
interface FreeSlots {
    int count(Size size);
}

/** Strategy: which slot size should this parcel get? Returns null when nothing suitable is free. */
interface SlotAllocationStrategy {
    Size choose(Size parcel, FreeSlots free);
}

/** Smallest free size that fits. A small parcel uses a medium slot only when no small one is free. */
final class BestFitStrategy implements SlotAllocationStrategy {
    @Override public Size choose(Size parcel, FreeSlots free) {
        for (Size s : Size.values())
            if (parcel.fitsIn(s) && free.count(s) > 0) return s;
        return null;
    }
}

/** Only the exact size. Simple and predictable; wastes nothing, but rejects more parcels. */
final class ExactFitStrategy implements SlotAllocationStrategy {
    @Override public Size choose(Size parcel, FreeSlots free) {
        return free.count(parcel) > 0 ? parcel : null;
    }
}

/**
 * Best fit, but a parcel may be upgraded into a bigger size only while more than reserve(size) slots of
 * that size stay free. Keeps the last large slots for large parcels, which nothing else can hold.
 */
final class ReservingBestFitStrategy implements SlotAllocationStrategy {
    private final EnumMap<Size, Integer> reserve;

    ReservingBestFitStrategy(EnumMap<Size, Integer> reserve) { this.reserve = new EnumMap<>(reserve); }

    @Override public Size choose(Size parcel, FreeSlots free) {
        for (Size s : Size.values()) {
            if (!parcel.fitsIn(s)) continue;
            int keep = s == parcel ? 0 : reserve.getOrDefault(s, 0);
            if (free.count(s) > keep) return s;
        }
        return null;
    }
}
import java.util.*;

/**
 * One locker location: its slots, free lists per size, and live deliveries.
 * Every public method is synchronized: one lock per station, so two stations never block each other,
 * and two couriers at the same station can never be given the same slot.
 */
final class LockerStation implements FreeSlots {
    static final int MAX_BAD_ATTEMPTS = 5;
    static final long KEYPAD_LOCK_MS = 60_000;

    final String id;
    private final int maxBadAttempts;
    private final Map<String, Slot> slots = new LinkedHashMap<>();
    private final EnumMap<Size, ArrayDeque<Slot>> free = new EnumMap<>(Size.class);
    private final Map<String, Delivery> byCode = new HashMap<>();     // DEPOSITED or EXPIRED (awaiting return)
    private final Map<String, Delivery> byParcel = new HashMap<>();
    private int badAttempts;
    private long keypadLockedUntil;

    LockerStation(String id, EnumMap<Size, Integer> counts, int maxBadAttempts) {
        this.id = id;
        this.maxBadAttempts = maxBadAttempts;
        for (Size s : Size.values()) free.put(s, new ArrayDeque<>());
        for (Size s : Size.values()) {
            int n = counts.getOrDefault(s, 0);
            for (int i = 1; i <= n; i++) {
                Slot slot = new Slot(String.format("%c%02d", s.name().charAt(0), i), s);
                slots.put(slot.id, slot);
                free.get(s).addLast(slot);
            }
        }
    }

    /** FreeSlots view for the strategy; only called while this station's lock is held. */
    @Override public int count(Size size) { return free.get(size).size(); }

    synchronized Delivery deposit(Parcel parcel, SlotAllocationStrategy strategy, CodeGenerator codes,
                                  long now, long holdMs) {
        if (byParcel.containsKey(parcel.id))
            throw new LockerException(LockerException.Reason.DUPLICATE_PARCEL, parcel.id + " is already here");
        Size size = strategy.choose(parcel.size, this);
        if (size == null)
            throw new LockerException(LockerException.Reason.NO_SLOT, "no free slot for " + parcel.size + " at " + id);
        Slot slot = free.get(size).pollFirst();                    // strategy chose a size with count > 0
        String code = uniqueCode(codes);
        slot.state = SlotState.OCCUPIED;
        Delivery d = new Delivery(parcel, slot, code, now, now + holdMs);
        byCode.put(code, d);
        byParcel.put(parcel.id, d);
        return d;
    }

    synchronized Delivery pickup(String code, long now) {
        if (now < keypadLockedUntil)
            throw new LockerException(LockerException.Reason.KEYPAD_LOCKED, "try again in " + (keypadLockedUntil - now) + " ms");
        Delivery d = byCode.get(code);
        if (d == null) {
            if (maxBadAttempts > 0 && ++badAttempts >= maxBadAttempts) { keypadLockedUntil = now + KEYPAD_LOCK_MS; badAttempts = 0; }
            throw new LockerException(LockerException.Reason.INVALID_CODE, "wrong code");
        }
        if (d.state == DeliveryState.DEPOSITED && now >= d.expiresAt) d.state = DeliveryState.EXPIRED;  // lazy expiry
        if (d.state == DeliveryState.EXPIRED)
            throw new LockerException(LockerException.Reason.CODE_EXPIRED, "parcel " + d.parcel.id + " went back to the courier queue");
        badAttempts = 0;
        d.state = DeliveryState.PICKED_UP;
        byCode.remove(code);                                       // single use
        byParcel.remove(d.parcel.id);
        release(d.slot);
        return d;
    }

    /** Marks overdue deliveries EXPIRED. Their slots stay OCCUPIED until a courier collects them. */
    synchronized List<Delivery> expireDue(long now) {
        List<Delivery> out = new ArrayList<>();
        for (Delivery d : byParcel.values())
            if (d.state == DeliveryState.DEPOSITED && now >= d.expiresAt) { d.state = DeliveryState.EXPIRED; out.add(d); }
        return out;
    }

    /** Courier takes expired parcels back to the warehouse: codes die, slots become free. */
    synchronized List<Delivery> collectExpired() {
        List<Delivery> out = new ArrayList<>();
        for (Iterator<Delivery> it = byParcel.values().iterator(); it.hasNext(); ) {
            Delivery d = it.next();
            if (d.state != DeliveryState.EXPIRED) continue;
            it.remove();
            byCode.remove(d.code);
            d.state = DeliveryState.RETURNED;
            release(d.slot);
            out.add(d);
        }
        return out;
    }

    synchronized void setOutOfService(String slotId, boolean broken) {
        Slot slot = slots.get(slotId);
        if (slot == null) throw new LockerException(LockerException.Reason.UNKNOWN_SLOT, slotId);
        if (broken) {
            if (slot.state == SlotState.OCCUPIED)
                throw new LockerException(LockerException.Reason.SLOT_BUSY, slotId + " holds a parcel");
            if (slot.state == SlotState.FREE) free.get(slot.size).remove(slot);
            slot.state = SlotState.OUT_OF_SERVICE;
        } else if (slot.state == SlotState.OUT_OF_SERVICE) {
            release(slot);
        }
    }

    synchronized EnumMap<Size, Integer> availability() {
        EnumMap<Size, Integer> m = new EnumMap<>(Size.class);
        for (Size s : Size.values()) m.put(s, free.get(s).size());
        return m;
    }

    synchronized Optional<DeliveryState> stateOf(String parcelId) {
        Delivery d = byParcel.get(parcelId);
        return d == null ? Optional.empty() : Optional.of(d.state);
    }

    synchronized int occupied() {
        int n = 0;
        for (Slot s : slots.values()) if (s.state == SlotState.OCCUPIED) n++;
        return n;
    }

    private void release(Slot slot) {
        slot.state = SlotState.FREE;
        free.get(slot.size).addLast(slot);                         // FIFO reuse spreads wear across doors
    }

    private String uniqueCode(CodeGenerator codes) {
        for (int i = 0; i < 20; i++) {
            String c = codes.next();
            if (!byCode.containsKey(c)) return c;
        }
        throw new IllegalStateException("code space exhausted at " + id);
    }
}
import java.util.*;
import java.util.concurrent.*;

/** Facade: the only class couriers, customers, the expiry job and admins talk to. */
final class LockerService {
    private final Map<String, LockerStation> stations = new ConcurrentHashMap<>();
    private final List<LockerListener> listeners = new CopyOnWriteArrayList<>();
    private final SlotAllocationStrategy strategy;
    private final CodeGenerator codes;
    private final Clock clock;
    private final long holdMs;

    LockerService(SlotAllocationStrategy strategy, CodeGenerator codes, Clock clock, long holdMs) {
        this.strategy = strategy; this.codes = codes; this.clock = clock; this.holdMs = holdMs;
    }

    void addListener(LockerListener l) { listeners.add(l); }

    void addStation(String stationId, int small, int medium, int large) {
        addStation(stationId, small, medium, large, LockerStation.MAX_BAD_ATTEMPTS);
    }

    /** maxBadAttempts = 0 disables the keypad lockout. */
    void addStation(String stationId, int small, int medium, int large, int maxBadAttempts) {
        EnumMap<Size, Integer> counts = new EnumMap<>(Size.class);
        counts.put(Size.SMALL, small); counts.put(Size.MEDIUM, medium); counts.put(Size.LARGE, large);
        if (stations.putIfAbsent(stationId, new LockerStation(stationId, counts, maxBadAttempts)) != null)
            throw new IllegalArgumentException("station exists: " + stationId);
    }

    /** Courier flow. Returns the door that opened; the pickup code is sent to the customer by a listener. */
    DepositReceipt deposit(String stationId, Parcel parcel) {
        LockerStation st = station(stationId);
        Delivery d = st.deposit(parcel, strategy, codes, clock.now(), holdMs);
        for (LockerListener l : listeners) l.onDeposited(stationId, d);          // outside the lock
        return new DepositReceipt(stationId, d.slot.id, d.slot.size, parcel.id, d.expiresAt);
    }

    /** Customer flow: type the code on the station keypad. */
    PickupResult pickup(String stationId, String code) {
        LockerStation st = station(stationId);
        Delivery d = st.pickup(code, clock.now());
        for (LockerListener l : listeners) l.onPickedUp(stationId, d);
        return new PickupResult(d.slot.id, d.parcel);
    }

    /** Called by a periodic job (see the task-scheduler case). Returns how many deliveries expired. */
    int expireOverdue() {
        long now = clock.now();
        int n = 0;
        for (LockerStation st : stations.values()) {
            List<Delivery> expired = st.expireDue(now);
            for (Delivery d : expired) for (LockerListener l : listeners) l.onExpired(st.id, d);
            n += expired.size();
        }
        return n;
    }

    /** Courier return run: take every expired parcel out of this station. */
    List<Parcel> collectExpired(String stationId) {
        List<Parcel> out = new ArrayList<>();
        for (Delivery d : station(stationId).collectExpired()) out.add(d.parcel);
        return out;
    }

    void setOutOfService(String stationId, String slotId, boolean broken) { station(stationId).setOutOfService(slotId, broken); }

    EnumMap<Size, Integer> availability(String stationId) { return station(stationId).availability(); }

    Optional<DeliveryState> stateOf(String stationId, String parcelId) { return station(stationId).stateOf(parcelId); }   // C: Optional = a pointer that may be NULL, made explicit

    int occupied(String stationId) { return station(stationId).occupied(); }

    private LockerStation station(String id) {
        LockerStation st = stations.get(id);
        if (st == null) throw new LockerException(LockerException.Reason.UNKNOWN_STATION, id);
        return st;
    }
}
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

/** Demo driver with simple checks. Run: javac -d out *.java && java -cp out Main */
public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static LockerException.Reason failure(Runnable r) {
        try { r.run(); } catch (LockerException e) { return e.reason; }
        return null;
    }

    static final class FakeClock implements Clock {
        final AtomicLong t = new AtomicLong();
        public long now() { return t.get(); }
        void advance(long ms) { t.addAndGet(ms); }
    }

    /** Plays the SMS service: remembers the code each customer was sent. */
    static final class Inbox implements LockerListener {
        final Map<String, String> codeFor = new ConcurrentHashMap<>();
        final List<String> log = Collections.synchronizedList(new ArrayList<>());
        public void onDeposited(String st, Delivery d) { codeFor.put(d.parcel.id, d.code); }
        public void onPickedUp(String st, Delivery d) { log.add("picked " + d.parcel.id); }
        public void onExpired(String st, Delivery d) { log.add("expired " + d.parcel.id); }
    }

    static final long DAY = 24L * 3600 * 1000;

    public static void main(String[] args) throws Exception {
        FakeClock clock = new FakeClock();
        Inbox inbox = new Inbox();
        LockerService svc = new LockerService(new BestFitStrategy(), new RandomCodeGenerator(new Random(42), 6), clock, 3 * DAY);
        svc.addListener(inbox);
        svc.addStation("BLR-01", 2, 2, 1);

        System.out.println("== 1. best-fit allocation (2 small, 2 medium, 1 large) ==");
        String[][] arrivals = {{"p1", "SMALL"}, {"p2", "SMALL"}, {"p3", "SMALL"}, {"p4", "MEDIUM"}, {"p5", "MEDIUM"}};
        List<String> got = new ArrayList<>();
        for (String[] a : arrivals) {
            DepositReceipt r = svc.deposit("BLR-01", new Parcel(a[0], "c-" + a[0], Size.valueOf(a[1])));
            got.add(r.slotId);
            System.out.printf("  %s %-6s -> slot %s%n", a[0], a[1], r.slotId);
        }
        check(got.equals(Arrays.asList("S01", "S02", "M01", "M02", "L01")), "upgrade only when the exact size is full");
        LockerException.Reason r6 = failure(() -> svc.deposit("BLR-01", new Parcel("p6", "c6", Size.SMALL)));   // C: a lambda = a function pointer plus its captured context
        System.out.println("  p6 SMALL  -> " + r6);
        check(r6 == LockerException.Reason.NO_SLOT, "full station rejects");
        check(failure(() -> svc.deposit("BLR-01", new Parcel("p1", "c1", Size.SMALL))) == LockerException.Reason.DUPLICATE_PARCEL, "duplicate parcel");

        System.out.println("== 2. pickup with a one-time code ==");
        String code1 = inbox.codeFor.get("p1");
        check(code1 != null && code1.length() == 6, "customer got a 6-digit code");
        String wrong = code1.equals("000000") ? "111111" : "000000";
        System.out.println("  wrong code -> " + failure(() -> svc.pickup("BLR-01", wrong)));
        PickupResult pr = svc.pickup("BLR-01", code1);
        System.out.println("  right code -> door " + pr.slotId + " opens, parcel " + pr.parcel.id);
        check(pr.slotId.equals("S01") && pr.parcel.id.equals("p1"), "right door");
        LockerException.Reason again = failure(() -> svc.pickup("BLR-01", code1));
        System.out.println("  same code again -> " + again);
        check(again == LockerException.Reason.INVALID_CODE, "code is single use");
        DepositReceipt p6 = svc.deposit("BLR-01", new Parcel("p6", "c6", Size.SMALL));
        System.out.println("  freed S01 is reused: p6 -> " + p6.slotId);
        check(p6.slotId.equals("S01"), "freed slot reused");

        System.out.println("== 3. keypad lockout after 5 wrong codes ==");
        for (int i = 0; i < 5; i++) failure(() -> svc.pickup("BLR-01", "999999".equals(inbox.codeFor.get("p2")) ? "888888" : "999999"));
        String code2 = inbox.codeFor.get("p2");
        LockerException.Reason locked = failure(() -> svc.pickup("BLR-01", code2));
        System.out.println("  6th try, even with the right code -> " + locked);
        check(locked == LockerException.Reason.KEYPAD_LOCKED, "keypad locked");
        clock.advance(LockerStation.KEYPAD_LOCK_MS);
        System.out.println("  60 s later -> door " + svc.pickup("BLR-01", code2).slotId);

        System.out.println("== 4. expiry after 3 days, courier collects ==");
        clock.advance(3 * DAY);
        int expired = svc.expireOverdue();
        System.out.println("  expireOverdue() -> " + expired + " deliveries expired");
        check(expired == 4, "p3 p4 p5 p6 expired");
        String code3 = inbox.codeFor.get("p3");
        LockerException.Reason late = failure(() -> svc.pickup("BLR-01", code3));
        System.out.println("  customer tries p3 -> " + late);
        check(late == LockerException.Reason.CODE_EXPIRED, "expired code refused");
        check(svc.availability("BLR-01").get(Size.MEDIUM) == 0, "expired parcels still hold their slots");
        List<Parcel> back = svc.collectExpired("BLR-01");
        System.out.println("  courier collects " + back + " ; free now " + svc.availability("BLR-01"));
        check(back.size() == 4 && svc.occupied("BLR-01") == 0, "all slots free after collection");
        check(svc.stateOf("BLR-01", "p3").equals(Optional.empty()), "returned delivery is gone from the station");

        System.out.println("== 5. swapping the strategy ==");
        EnumMap<Size, Integer> keep = new EnumMap<>(Size.class);
        keep.put(Size.LARGE, 1);
        LockerService reserving = new LockerService(new ReservingBestFitStrategy(keep), new RandomCodeGenerator(new Random(7), 6), clock, DAY);
        reserving.addStation("HYD-02", 1, 0, 2);
        List<String> res = new ArrayList<>();
        for (int i = 1; i <= 3; i++) {
            try { res.add(reserving.deposit("HYD-02", new Parcel("s" + i, "c", Size.SMALL)).slotId); }
            catch (LockerException e) { res.add(e.reason.name()); }
        }
        DepositReceipt big = reserving.deposit("HYD-02", new Parcel("big", "c", Size.LARGE));
        System.out.println("  reserve 1 large: smalls -> " + res + " ; LARGE -> " + big.slotId);
        check(res.equals(Arrays.asList("S01", "L01", "NO_SLOT")) && big.slotId.equals("L02"), "last large slot kept for a large parcel");
        LockerService exact = new LockerService(new ExactFitStrategy(), new RandomCodeGenerator(new Random(1), 6), clock, DAY);
        exact.addStation("X", 0, 0, 3);
        LockerException.Reason ex = failure(() -> exact.deposit("X", new Parcel("e1", "c", Size.SMALL)));
        System.out.println("  exact fit, small parcel, only large slots -> " + ex);
        check(ex == LockerException.Reason.NO_SLOT, "exact fit never upgrades");

        System.out.println("== 6. maintenance and sizing ==");
        svc.setOutOfService("BLR-01", "L01", true);
        System.out.println("  L01 out of service; free " + svc.availability("BLR-01"));
        check(svc.availability("BLR-01").get(Size.LARGE) == 0, "broken slot is not offered");
        DepositReceipt m = svc.deposit("BLR-01", new Parcel("m1", "c", Size.MEDIUM));
        check(failure(() -> svc.setOutOfService("BLR-01", m.slotId, true)) == LockerException.Reason.SLOT_BUSY, "cannot break an occupied slot");
        svc.setOutOfService("BLR-01", "L01", false);
        check(svc.availability("BLR-01").get(Size.LARGE) == 1, "repaired slot is back");
        System.out.println("  box 25x20x10 -> " + Size.forBox(25, 20, 10) + ", 60x40x40 -> " + Size.forBox(60, 40, 40)
                + ", 90x10x10 -> " + failure(() -> Size.forBox(90, 10, 10)));
        check(Size.forBox(25, 20, 10) == Size.SMALL && Size.forBox(60, 40, 40) == Size.LARGE, "size by longest side");
        check(failure(() -> svc.deposit("NOPE", new Parcel("z", "c", Size.SMALL))) == LockerException.Reason.UNKNOWN_STATION, "unknown station");

        System.out.println("== 7. stress: 8 couriers x 100 parcels into 600 slots, then 2 racing pickers ==");
        Inbox sms = new Inbox();
        LockerService big600 = new LockerService(new BestFitStrategy(), new RandomCodeGenerator(new Random(99), 6), clock, DAY);
        big600.addListener(sms);
        big600.addStation("DEL-03", 200, 200, 200, 0);   // 0 = no keypad lockout: this test races an app API
        ExecutorService pool = Executors.newFixedThreadPool(8);
        Set<String> slotsUsed = ConcurrentHashMap.newKeySet();
        AtomicInteger ok = new AtomicInteger(), full = new AtomicInteger(), dupSlot = new AtomicInteger();
        CountDownLatch start = new CountDownLatch(1);
        List<Future<?>> fs = new ArrayList<>();
        for (int t = 0; t < 8; t++) {
            final int courier = t;
            fs.add(pool.submit(() -> {
                start.await();
                for (int i = 0; i < 100; i++) {
                    try {
                        DepositReceipt rc = big600.deposit("DEL-03", new Parcel("c" + courier + "-" + i, "cust", Size.SMALL));
                        if (!slotsUsed.add(rc.slotId)) dupSlot.incrementAndGet();
                        ok.incrementAndGet();
                    } catch (LockerException e) { full.incrementAndGet(); }
                }
                return null;
            }));
        }
        start.countDown();
        for (Future<?> f : fs) f.get();
        System.out.println("  deposited " + ok + ", rejected " + full + ", slot given twice " + dupSlot);
        check(ok.get() == 600 && full.get() == 200 && dupSlot.get() == 0 && big600.occupied("DEL-03") == 600, "no slot handed out twice");
        List<String> codes = new ArrayList<>(sms.codeFor.values());
        check(new HashSet<>(codes).size() == 600, "600 distinct live codes");
        AtomicInteger opened = new AtomicInteger(), refused = new AtomicInteger();
        CountDownLatch go = new CountDownLatch(1);
        List<Future<?>> pickers = new ArrayList<>();
        for (int t = 0; t < 2; t++) {
            pickers.add(pool.submit(() -> {
                go.await();
                for (String c : codes) {
                    try { big600.pickup("DEL-03", c); opened.incrementAndGet(); }
                    catch (LockerException e) { refused.incrementAndGet(); }
                }
                return null;
            }));
        }
        go.countDown();
        for (Future<?> f : pickers) f.get();
        pool.shutdown();
        System.out.println("  two pickers typed all 600 codes: opened " + opened + ", refused " + refused);
        check(opened.get() == 600 && big600.occupied("DEL-03") == 0, "every code opened exactly one door, once");
        System.out.println("== all " + checks + " checks passed ==");
    }
