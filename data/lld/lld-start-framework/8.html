public class BeforeGodClass {
    static String[] roomIds = {"R-small", "R-mid", "R-big"};
    static int[] capacity = {4, 8, 20};
    static List<String[]> bookings = new ArrayList<>();        // {roomId, start, end}

    static String book(int start, int end, int people, String mode) {
        for (int i = 0; i < roomIds.length; i++) {
            if (mode.equals("smallest") && capacity[i] < people) continue;
            if (mode.equals("floor2") && i == 0) continue;      // hard-coded floor knowledge
            boolean free = true;
            for (String[] b : bookings)
                if (b[0].equals(roomIds[i]) && start < Integer.parseInt(b[2])
                        && Integer.parseInt(b[1]) < end) free = false;
            if (free) {                                           // check ... then act: race
                bookings.add(new String[]{roomIds[i], "" + start, "" + end});
                return roomIds[i];
            }
        }
        return null;                                              // caller must remember null
    }
}

It runs. But: a new selection rule means another if (mode.equals(...)); "floor2" mode even skips the capacity check (a real bug); bookings are string arrays; failure is a silent null; and two threads can both see "free" and both add.

record Interval(int start, int end) {                  // minutes since midnight, [start, end)
    Interval {
        if (start < 0 || end <= start) throw new IllegalArgumentException("bad interval " + start + "-" + end);
    }
    boolean overlaps(Interval o) { return start < o.end && o.start < end; }
}

record Booking(int id, String roomId, String organizer, Interval slot, int attendees) {}

final class Room {
    final String id; final int capacity; final int floor;
    private final List<Booking> bookings = new ArrayList<>();
    Room(String id, int capacity, int floor) { this.id = id; this.capacity = capacity; this.floor = floor; }

    // the room owns its own rule: "no two bookings overlap"
    synchronized boolean isFree(Interval slot) {
        for (Booking b : bookings) if (b.slot().overlaps(slot)) return false;
        return true;
    }
    synchronized boolean tryAdd(Booking b) {            // check-then-act under ONE lock
        if (!isFree(b.slot())) return false;
        bookings.add(b);
        return true;
    }
    synchronized boolean remove(int bookingId) { return bookings.removeIf(b -> b.id() == bookingId); }
}

interface RoomSelectionStrategy {                     // step 5: the change point
    List<Room> rank(List<Room> candidates, int attendees);
}

final class SmallestFitFirst implements RoomSelectionStrategy {
    public List<Room> rank(List<Room> rooms, int attendees) {
        List<Room> fit = new ArrayList<>();
        for (Room r : rooms) if (r.capacity >= attendees) fit.add(r);
        fit.sort(Comparator.comparingInt((Room r) -> r.capacity).thenComparing(r -> r.id));
        return fit;
    }
}

final class PreferFloor implements RoomSelectionStrategy {   // step 7: extension = new class only
    private final int floor; private final RoomSelectionStrategy fallback = new SmallestFitFirst();
    PreferFloor(int floor) { this.floor = floor; }
    public List<Room> rank(List<Room> rooms, int attendees) {
        List<Room> ranked = new ArrayList<>(fallback.rank(rooms, attendees));
        ranked.sort(Comparator.comparingInt((Room r) -> r.floor == floor ? 0 : 1));   // stable sort keeps size order
        return ranked;
    }
}

class NoRoomAvailableException extends RuntimeException {
    NoRoomAvailableException(String msg) { super(msg); }
}

final class BookingService {                          // step 4: the orchestrator and its API
    private final Map<String, Room> rooms = new ConcurrentHashMap<>();   // addRoom may run during traffic
    private final Map<Integer, Booking> byId = new ConcurrentHashMap<>();
    private final AtomicInteger nextId = new AtomicInteger(1);
    private volatile RoomSelectionStrategy strategy;

    BookingService(RoomSelectionStrategy strategy) { this.strategy = strategy; }
    void setStrategy(RoomSelectionStrategy s) { this.strategy = s; }
    void addRoom(Room r) { rooms.put(r.id, r); }

    Booking book(String organizer, Interval slot, int attendees) {
        if (attendees <= 0) throw new IllegalArgumentException("attendees must be > 0");
        for (Room r : strategy.rank(new ArrayList<>(rooms.values()), attendees)) {
            // a failed tryAdd uses up an id: ids stay unique but can have gaps (fine for ids)
            Booking b = new Booking(nextId.getAndIncrement(), r.id, organizer, slot, attendees);
            if (r.tryAdd(b)) { byId.put(b.id(), b); return b; }
        }
        throw new NoRoomAvailableException("no room for " + attendees + " at " + slot);
    }

    void cancel(int bookingId) {
        Booking b = byId.remove(bookingId);
        if (b == null) throw new NoSuchElementException("no booking " + bookingId);
        rooms.get(b.roomId()).remove(bookingId);
    }
}
@dataclass(frozen=True)
class Interval:
    start: int
    end: int

    def __post_init__(self) -> None:
        if self.start < 0 or self.end <= self.start:
            raise ValueError(f"bad interval {self.start}-{self.end}")

    def overlaps(self, o: "Interval") -> bool:
        return self.start < o.end and o.start < self.end


class Room:
    def __init__(self, room_id: str, capacity: int, floor: int) -> None:
        self.id, self.capacity, self.floor = room_id, capacity, floor
        self._bookings: list[Booking] = []
        self._lock = threading.Lock()

    def try_add(self, b: Booking) -> bool:
        with self._lock:                                  # check-then-act under one lock
            if any(x.slot.overlaps(b.slot) for x in self._bookings):
                return False
            self._bookings.append(b)
            return True


class RoomSelectionStrategy(ABC):
    @abstractmethod
    def rank(self, rooms: list[Room], attendees: int) -> list[Room]: ...


class SmallestFitFirst(RoomSelectionStrategy):
    def rank(self, rooms, attendees):
        return sorted((r for r in rooms if r.capacity >= attendees), key=lambda r: (r.capacity, r.id))


class BookingService:
    def book(self, organizer: str, slot: Interval, attendees: int) -> Booking:
        if attendees <= 0:
            raise ValueError("attendees must be > 0")
        for room in self.strategy.rank(list(self._rooms.values()), attendees):
            with self._id_lock:
                bid = next(self._ids)
            b = Booking(bid, room.id, organizer, slot, attendees)
            if room.try_add(b):
                self._by_id[b.id] = b
                return b
        raise NoRoomAvailable(f"no room for {attendees} at {slot}")
