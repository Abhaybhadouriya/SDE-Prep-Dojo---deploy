After: one class per actor

The refactor uses one move from Martin Fowler's refactoring catalogue again and again: Extract Class. Take a group of methods (and the fields only they use) and move them into a new class. Click through the stepper to see the order of moves.

  1. Start: one class with four kinds of methods. Tests for the tax rule need a database and an SMTP stub.
  2. Write the actor next to each method. Methods that answer to the same person stay together.
  3. Extract Class: move rendering into InvoiceFormatter. It reads the invoice; it never changes it.
  4. Move storage behind an InvoiceRepository interface, so the database can change without touching money code.
  5. Move customer messages behind a Notifier interface. Email today, WhatsApp tomorrow: one new class.
  6. Add a thin InvoiceService that calls them in order. Invoice is now small, pure and easy to test.
import java.util.ArrayList;
import java.util.List;

// AFTER: each class serves one actor. InvoiceService only wires them together.
public class InvoiceAfter {
    record Line(String item, int qty, long unitPaise) {}

    // Domain object: data + the money rule finance owns.
    static final class Invoice {
        final String id, customerEmail;
        final List<Line> lines = new ArrayList<>();
        private final TaxPolicy tax;
        Invoice(String id, String customerEmail, TaxPolicy tax) { this.id = id; this.customerEmail = customerEmail; this.tax = tax; }
        void add(String item, int qty, long unitPaise) {
            if (qty <= 0) throw new IllegalArgumentException("qty must be > 0");
            lines.add(new Line(item, qty, unitPaise));
        }
        long subtotalPaise() { long s = 0; for (Line l : lines) s += l.qty() * l.unitPaise(); return s; }
        long totalPaise() { long s = subtotalPaise(); return s + tax.taxOn(s); }
    }

    interface TaxPolicy { long taxOn(long subtotalPaise); }                 // finance
    static final class Gst18 implements TaxPolicy { public long taxOn(long s) { return Math.round(s * 0.18); } }

    static final class InvoiceFormatter {                                     // support / design
        String render(Invoice inv) {
            StringBuilder sb = new StringBuilder("INVOICE " + inv.id + "\n");
            for (Line l : inv.lines) sb.append(String.format("  %-8s x%d  %8.2f%n", l.item(), l.qty(), l.unitPaise() / 100.0));
            return sb.append(String.format("  TOTAL      %8.2f%n", inv.totalPaise() / 100.0)).toString();
        }
    }

    interface InvoiceRepository { void save(Invoice inv); }                  // ops / DBA
    static final class SqlInvoiceRepository implements InvoiceRepository {
        public void save(Invoice inv) {
            System.out.println("INSERT INTO invoices(id,total_paise) VALUES('" + inv.id + "'," + inv.totalPaise() + ")");
        }
    }

    interface Notifier { void invoiceReady(Invoice inv); }                   // customer comms
    static final class EmailNotifier implements Notifier {
        public void invoiceReady(Invoice inv) { System.out.println("to=" + inv.customerEmail + " subject=Invoice " + inv.id); }
    }

    // Use-case class: knows the ORDER of steps, not how each step works.
    static final class InvoiceService {
        private final InvoiceFormatter fmt; private final InvoiceRepository repo; private final Notifier notifier;
        InvoiceService(InvoiceFormatter f, InvoiceRepository r, Notifier n) { fmt = f; repo = r; notifier = n; }
        String issue(Invoice inv) { repo.save(inv); notifier.invoiceReady(inv); return fmt.render(inv); }
    }

    public static void main(String[] args) {
        Invoice inv = new Invoice("INV-7", "asha@example.com", new Gst18());
        inv.add("dosa", 2, 6000);
        inv.add("coffee", 1, 3000);
        InvoiceService svc = new InvoiceService(new InvoiceFormatter(), new SqlInvoiceRepository(), new EmailNotifier());
        System.out.print(svc.issue(inv));

        // Finance test needs no DB, no SMTP: the payoff of SRP.
        Invoice t = new Invoice("T", "x@y", new Gst18());
        t.add("a", 1, 10000);
        if (t.totalPaise() != 11800) throw new AssertionError("tax math broke");
        System.out.println("tax test ok: " + t.totalPaise());
    }
}
from dataclasses import dataclass, field
from typing import List, Protocol


@dataclass(frozen=True)
class Line:
    item: str
    qty: int
    unit_paise: int


class TaxPolicy(Protocol):
    def tax_on(self, subtotal: int) -> int: ...


class Gst18:
    def tax_on(self, subtotal: int) -> int:
        return round(subtotal * 0.18)


@dataclass
class Invoice:                      # finance: data + money rule
    id: str
    email: str
    tax: TaxPolicy
    lines: List[Line] = field(default_factory=list)

    def add(self, item: str, qty: int, unit_paise: int) -> None:
        if qty <= 0:
            raise ValueError("qty must be > 0")
        self.lines.append(Line(item, qty, unit_paise))

    def total(self) -> int:
        sub = sum(l.qty * l.unit_paise for l in self.lines)
        return sub + self.tax.tax_on(sub)


class InvoiceFormatter:             # support / design
    def render(self, inv: Invoice) -> str:
        rows = [f"  {l.item:<8} x{l.qty}  {l.unit_paise / 100:8.2f}" for l in inv.lines]
        return "\n".join([f"INVOICE {inv.id}", *rows, f"  TOTAL      {inv.total() / 100:8.2f}"])


class SqlInvoiceRepository:         # ops / DBA
    def save(self, inv: Invoice) -> None:
        print(f"INSERT INTO invoices(id,total_paise) VALUES('{inv.id}',{inv.total()})")


class EmailNotifier:                # customer comms
    def invoice_ready(self, inv: Invoice) -> None:
        print(f"to={inv.email} subject=Invoice {inv.id}")


class InvoiceService:               # wiring only
    def __init__(self, fmt, repo, notifier):
        self.fmt, self.repo, self.notifier = fmt, repo, notifier

    def issue(self, inv: Invoice) -> str:
        self.repo.save(inv)
        self.notifier.invoice_ready(inv)
        return self.fmt.render(inv)


if __name__ == "__main__":
    inv = Invoice("INV-7", "asha@example.com", Gst18())
    inv.add("dosa", 2, 6000)
    inv.add("coffee", 1, 3000)
    print(InvoiceService(InvoiceFormatter(), SqlInvoiceRepository(), EmailNotifier()).issue(inv))
    t = Invoice("T", "x@y", Gst18())
    t.add("a", 1, 10000)
    assert t.total() == 11800
    print("tax test ok:", t.total())

What changed, and why each choice matters:

  • Invoice keeps the data and the money rule. It also gained a check (qty > 0), because a class that owns its data should guard it.
  • TaxPolicy is an interface. Finance's rule now lives in one tiny class. (Making it swappable is a bonus that belongs to the next letter, Open/Closed.)
  • InvoiceFormatter has no state. It only reads an invoice and returns text.
  • InvoiceRepository and Notifier are interfaces, so storage and messaging can change without the domain code knowing.
  • InvoiceService is a thin coordinator. Its single job is the order of steps for the "issue invoice" use case. That order is itself a responsibility, owned by the product flow.
  • The last lines of main test the tax rule with no database and no email. That is the clearest sign the split worked.

When you present this in a round, say the actor out loud: "I split it because finance, support and the DBA change it for different reasons. Now a GST change touches one class, and I can unit-test it without a database." Interviewers look for the reason, not just for more classes.
