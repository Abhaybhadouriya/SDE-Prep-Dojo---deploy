Code: ATM and logger (Java), middleware (Python), netfilter-style hooks (C++14)

All three programs were compiled and run; the output under each one is real. Files live in lld-code/lld-pat-chain-of-responsibility/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • abstract class or method: a base type with some slots deliberately left empty; you cannot create one directly, only subclasses that fill the slots.
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • Comparator.comparingLong(f).reversed(): builds a compare function (like the one you pass to qsort) from a key extractor, then flips its order.
  • Python ABC + @abstractmethod: a base class that refuses to be instantiated until a subclass fills the marked methods.
  • std::function<R(Args)>: holds any callable: a plain function pointer, a lambda with captured state, or an object with operator().
  • C++ lambda [&](int x) { ... }: an inline function; the brackets list what it captures from the enclosing scope (& = by reference, = = by copy).

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

/** Chain of Responsibility: ATM cash dispenser (2000 -> 500 -> 200 -> 100) with limited notes,
 *  plus a logger chain where a record can travel past several handlers.
 *  Build and run: javac -d out AtmDemo.java && java -cp out AtmDemo */
public class AtmDemo {

    // ---- Handler: one note denomination. Knows only the NEXT handler. ----
    static final class NoteDispenser {
        private final int note;
        private int count;
        private NoteDispenser next;
        NoteDispenser(int note, int count) { this.note = note; this.count = count; }
        NoteDispenser then(NoteDispenser n) { this.next = n; return n; }   // fluent chain building

        /** Plan how many of each note to give. Returns false if this handler and the rest of the
         *  chain cannot make the amount. Tries the most notes first, then fewer (backtracking),
         *  because plain greedy fails when notes run out. Nothing is changed until commit(). */
        boolean plan(int amount, Map<Integer, Integer> out) {
            int max = Math.min(amount / note, count);
            for (int k = max; k >= 0; k--) {
                int rest = amount - k * note;
                boolean ok = (rest == 0) || (next != null && next.plan(rest, out));
                if (ok) { if (k > 0) out.put(note, k); return true; }
            }
            return false;
        }
        void commit(Map<Integer, Integer> plan) {
            count -= plan.getOrDefault(note, 0);
            if (next != null) next.commit(plan);
        }
        String stock() { return note + "x" + count + (next == null ? "" : " " + next.stock()); }
    }

    static final class Atm {
        private final NoteDispenser head;
        Atm(NoteDispenser head) { this.head = head; }
        void withdraw(int amount) {
            Map<Integer, Integer> plan = new TreeMap<>(Comparator.reverseOrder());
            if (amount <= 0 || amount % 100 != 0 || !head.plan(amount, plan)) {
                System.out.println("  " + amount + ": cannot dispense, nothing given   [stock " + head.stock() + "]");
                return;
            }
            head.commit(plan);
            System.out.println("  " + amount + ": " + plan + "   [stock " + head.stock() + "]");
        }
    }

    // ---- Logger chain: classic CoR where several handlers may act on one request ----
    enum Level { DEBUG, INFO, ERROR }

    abstract static class LogHandler {
        private LogHandler next;
        private final Level min;
        LogHandler(Level min) { this.min = min; }
        LogHandler then(LogHandler n) { next = n; return n; }
        final void log(Level lvl, String msg) {                 // the chain walk is fixed here
            if (lvl.compareTo(min) >= 0) write(lvl, msg);
            if (next != null) next.log(lvl, msg);
        }
        protected abstract void write(Level lvl, String msg);
    }
    static final class Console extends LogHandler {
        Console() { super(Level.DEBUG); }
        protected void write(Level l, String m) { System.out.println("  console  " + l + " " + m); }
    }
    static final class FileLog extends LogHandler {
        FileLog() { super(Level.INFO); }
        protected void write(Level l, String m) { System.out.println("  app.log  " + l + " " + m); }
    }
    static final class Pager extends LogHandler {
        Pager() { super(Level.ERROR); }
        protected void write(Level l, String m) { System.out.println("  PAGER    " + l + " " + m); }
    }

    public static void main(String[] args) {
        System.out.println("ATM with notes 2000x1, 500x1, 200x4, 100x0:");
        NoteDispenser head = new NoteDispenser(2000, 1);
        head.then(new NoteDispenser(500, 1)).then(new NoteDispenser(200, 4)).then(new NoteDispenser(100, 0));
        Atm atm = new Atm(head);
        atm.withdraw(600);    // greedy would take 500 and get stuck on 100; backtracking gives 200x3
        atm.withdraw(2700);   // 2000 + 500 + 200
        atm.withdraw(300);    // only 200s left (one) and no 100s: impossible
        atm.withdraw(250);    // not a multiple of 100

        System.out.println("Logger chain console(DEBUG+) -> file(INFO+) -> pager(ERROR):");
        LogHandler log = new Console();
        log.then(new FileLog()).then(new Pager());
        log.log(Level.DEBUG, "cache miss key=user:42");
        log.log(Level.INFO, "order ORD-77 placed");
        log.log(Level.ERROR, "payment gateway timeout");
    }
}

Two chain styles in one file. The ATM is a pipeline where every handler contributes, and any one failing makes the whole request fail. The logger is the pass-through style: every handler sees every record and decides only whether to write it. Note log() is final: the walk itself is fixed in the base class, and subclasses only supply write(). That is a Template Method inside a chain.

"""Chain of Responsibility in Python: an HTTP-style middleware pipeline.
Each handler may stop the request (return a response) or pass it to the next one.
Run: python middleware.py
"""
from abc import ABC


class Handler(ABC):
    def __init__(self):
        self._next = None

    def then(self, nxt):
        self._next = nxt
        return nxt

    def handle(self, req):
        return self._next.handle(req) if self._next else (500, "no handler")


class Auth(Handler):
    def handle(self, req):
        if req.get("token") != "good-token":
            return 401, "unauthorized"            # stop here
        req["user"] = "ravi"
        return super().handle(req)                # pass along


class RateLimit(Handler):
    def __init__(self, budget):
        super().__init__()
        self.budget, self.used = budget, {}      # calls allowed per user (per window, simplified)

    def handle(self, req):
        u = req["user"]
        self.used[u] = self.used.get(u, 0) + 1
        if self.used[u] > self.budget:
            return 429, "slow down"
        return super().handle(req)


class Validate(Handler):
    def handle(self, req):
        if req.get("amount", 0) <= 0:
            return 400, "amount must be positive"
        return super().handle(req)


class PayEndpoint(Handler):                       # end of chain: the real work
    def handle(self, req):
        return 200, f"paid {req['amount']} for {req['user']}"


if __name__ == "__main__":
    chain = Auth()
    chain.then(RateLimit(budget=2)).then(Validate()).then(PayEndpoint())

    requests = [
        {"token": "bad", "amount": 100},
        {"token": "good-token", "amount": -5},
        {"token": "good-token", "amount": 100},
        {"token": "good-token", "amount": 200},   # 3rd call from ravi: over budget
    ]
    for r in requests:
        print(chain.handle(dict(r)))

Each handler returns early to stop, or calls super().handle() to pass on. The base Handler.handle() also covers the "fell off the end" case with a 500, so a misbuilt chain fails loudly instead of returning None. Python web frameworks build middleware the same way: each layer wraps the next one and decides whether to call it.

// Chain of Responsibility in C++14, netfilter flavour: hooks sorted by priority, each returns a
// verdict. ACCEPT = keep walking the chain, DROP or STOLEN = stop.
// Build: g++ -std=c++14 -Wall nfhooks.cpp -o nfhooks
#include <algorithm>
#include <functional>
#include <iostream>
#include <string>
#include <vector>

enum Verdict { NF_DROP, NF_ACCEPT, NF_STOLEN };
static const char *vname(Verdict v) { return v == NF_DROP ? "DROP" : v == NF_ACCEPT ? "ACCEPT" : "STOLEN"; }

struct Packet { std::string src; int dport; int len; };

struct HookOps {                               // like struct nf_hook_ops
    std::string name;
    int priority;                              // lower runs first
    std::function<Verdict(Packet &)> hook;
};

class HookChain {
    std::vector<HookOps> hooks_;
public:
    void register_hook(HookOps h) {
        hooks_.push_back(std::move(h));
        std::stable_sort(hooks_.begin(), hooks_.end(),
                         [](const HookOps &a, const HookOps &b) { return a.priority < b.priority; });
    }
    Verdict run(Packet &p) {                   // like nf_hook_slow(): walk until someone stops it
        for (auto &h : hooks_) {
            Verdict v = h.hook(p);
            std::cout << "    " << h.name << " -> " << vname(v) << "\n";
            if (v != NF_ACCEPT) return v;
        }
        return NF_ACCEPT;                      // fell off the end: default policy
    }
};

int main() {
    HookChain prerouting;
    // registered out of order on purpose; priority decides the walk order
    prerouting.register_hook({"filter: block port 23", 0, [](Packet &p) {
        return p.dport == 23 ? NF_DROP : NF_ACCEPT; }});
    prerouting.register_hook({"conntrack", -200, [](Packet &) { return NF_ACCEPT; }});
    prerouting.register_hook({"queue big packets to userspace", 100, [](Packet &p) {
        return p.len > 1400 ? NF_STOLEN : NF_ACCEPT; }});
    prerouting.register_hook({"mangle: tag", -150, [](Packet &p) { p.src += "#tagged"; return NF_ACCEPT; }});

    std::vector<Packet> pkts = {{"10.0.0.5", 443, 600}, {"10.0.0.9", 23, 80}, {"10.0.0.7", 80, 1500}};
    for (auto &p : pkts) {
        std::cout << "packet " << p.src << ":" << p.dport << " len " << p.len << "\n";
        Verdict v = prerouting.run(p);
        std::cout << "  final verdict " << vname(v) << " (src now " << p.src << ")\n";
    }
    return 0;
}

Hooks were registered in a random order; the priority field decides the walk order, as in netfilter, where lower priority numbers run first. A hook can also change the packet (mangle tags it) and still pass it on.
