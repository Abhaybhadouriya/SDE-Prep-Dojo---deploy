After: an interface, implementations and a registry

Three moves turn the switch inside out:

  1. Name the variation. What changes per method is "how the fee is computed". That becomes an interface, FeeRule.
  2. One class per branch. Each case becomes a small class (Upi, Card...). Each can be tested alone.
  3. Replace the switch with a lookup. A FeeRegistry maps the method name to its rule. The calculator asks the registry and calls the rule through the interface.
import java.util.LinkedHashMap;
import java.util.Map;

// AFTER: the calculator is closed for modification; new methods arrive as new classes.
public class FeeAfter {
    interface FeeRule {                       // the abstraction = the extension point
        String method();
        long feePaise(long amountPaise);
    }

    static final class FeeRegistry {          // plays the role of a kernel "register_*_ops" list
        private final Map<String, FeeRule> rules = new LinkedHashMap<>();
        void register(FeeRule r) {
            if (rules.putIfAbsent(r.method(), r) != null) throw new IllegalStateException("duplicate " + r.method());
        }
        FeeRule find(String method) {
            FeeRule r = rules.get(method);
            if (r == null) throw new IllegalArgumentException("no fee rule for " + method);
            return r;
        }
        Iterable<String> methods() { return rules.keySet(); }
    }

    static final class FeeCalculator {        // CLOSED: never edited when a method is added
        private final FeeRegistry registry;
        FeeCalculator(FeeRegistry registry) { this.registry = registry; }
        long feePaise(String method, long amountPaise) {
            if (amountPaise <= 0) throw new IllegalArgumentException("amount must be > 0");
            return registry.find(method).feePaise(amountPaise);
        }
    }

    // ---- existing rules (made-up demo rates) ----
    static final class Upi implements FeeRule {
        public String method() { return "UPI"; }
        public long feePaise(long a) { return 0; }
    }
    static final class Card implements FeeRule {
        public String method() { return "CARD"; }
        public long feePaise(long a) { return Math.round(a * 0.02); }
    }
    static final class NetBanking implements FeeRule {
        public String method() { return "NETBANKING"; }
        public long feePaise(long a) { return 1000; }
    }
    static final class Wallet implements FeeRule {
        public String method() { return "WALLET"; }
        public long feePaise(long a) { return Math.max(500, Math.round(a * 0.015)); }
    }

    // ---- the NEW requirement: one new class, zero edits above ----
    static final class Bnpl implements FeeRule {
        public String method() { return "BNPL"; }
        public long feePaise(long a) { return Math.min(5000, Math.round(a * 0.03)); }   // 3%, capped at Rs 50
    }

    public static void main(String[] args) {
        FeeRegistry reg = new FeeRegistry();          // composition root: the ONE place that knows all rules
        reg.register(new Upi());
        reg.register(new Card());
        reg.register(new NetBanking());
        reg.register(new Wallet());
        reg.register(new Bnpl());                     // the only line added for BNPL

        FeeCalculator calc = new FeeCalculator(reg);
        for (String m : reg.methods())
            System.out.println(m + " fee on Rs 1000 = " + calc.feePaise(m, 100_000) / 100.0);

        try { calc.feePaise("CRYPTO", 100_000); }
        catch (IllegalArgumentException e) { System.out.println("rejected: " + e.getMessage()); }
        try { reg.register(new Card()); }
        catch (IllegalStateException e) { System.out.println("rejected: " + e.getMessage()); }
    }
}
from typing import Callable, Dict

FeeRule = Callable[[int], int]          # amount_paise -> fee_paise
_REGISTRY: Dict[str, FeeRule] = {}


def fee_rule(method: str):
    """Decorator that registers a rule, like module_init() calling a register function."""
    def wrap(fn: FeeRule) -> FeeRule:
        if method in _REGISTRY:
            raise ValueError(f"duplicate rule {method}")
        _REGISTRY[method] = fn
        return fn
    return wrap


def fee_paise(method: str, amount_paise: int) -> int:   # CLOSED: never edited
    if amount_paise <= 0:
        raise ValueError("amount must be > 0")
    try:
        return _REGISTRY[method](amount_paise)
    except KeyError:
        raise ValueError(f"no fee rule for {method}") from None


@fee_rule("UPI")
def _upi(a: int) -> int: return 0

@fee_rule("CARD")
def _card(a: int) -> int: return round(a * 0.02)

@fee_rule("NETBANKING")
def _nb(a: int) -> int: return 1000

@fee_rule("WALLET")
def _wallet(a: int) -> int: return max(500, round(a * 0.015))

# New requirement: one new function (it could live in a new file) is enough.
@fee_rule("BNPL")
def _bnpl(a: int) -> int: return min(5000, round(a * 0.03))


if __name__ == "__main__":
    for m in _REGISTRY:
        print(f"{m} fee on Rs 1000 = {fee_paise(m, 100_000) / 100}")
    try:
        fee_paise("CRYPTO", 100_000)
    except ValueError as e:
        print("rejected:", e)

Now watch what happens when BNPL arrives:

  1. The calculator, the registry and the four existing rules are written and tested. A new request arrives.
  2. You add one new class, Bnpl, that implements FeeRule. It lives in its own file with its own tests.
  3. The composition root (Main, or your DI config) registers it. This is the only existing file that changes, by one line.
  4. At runtime the calculator asks the registry for "BNPL" and calls the rule through the interface.
  5. Calculator, registry and old rules have no diff, so nothing old needs re-testing. That is "closed for modification".

Notice what did not disappear: somebody still decides which rules exist. OCP does not delete the decision, it moves it to one place, the composition root (usually Main, a factory, or a dependency-injection config). That one place is allowed to change. Everything that does real work is not.

Say this sentence in the round: "The switch does not vanish; it moves to the composition root, which is the only place that knows every concrete rule. The calculator only knows the interface." It shows you understand OCP, not just the slogan.
