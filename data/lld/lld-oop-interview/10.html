// lld-oop-interview: the small Java programs behind the classic OOP interview questions.
// Build and run (Java 21):  javac -d out OopPuzzles.java && java -cp out OopPuzzles
import java.util.*;

public class OopPuzzles {

    // Q: overloading is chosen at compile time, overriding at run time
    static class Animal { String sound() { return "..."; } static String kind() { return "Animal"; } }
    static class Dog extends Animal { @Override String sound() { return "woof"; } static String kind() { return "Dog"; } }
    static String describe(Animal a) { return "an animal"; }
    static String describe(Dog d) { return "a dog"; }

    // Q: diamond with default methods - the class must resolve the conflict
    interface Camera { default String start() { return "camera on"; } }
    interface Phone { default String start() { return "phone on"; } }
    static class SmartPhone implements Camera, Phone {
        @Override public String start() { return Camera.super.start() + " + " + Phone.super.start(); }
    }

    // Q: constructor order and calling an overridable method from a constructor
    static class Base {
        Base() { System.out.println("  Base() sees name = " + name()); }
        String name() { return "base"; }
    }
    static class Derived extends Base {
        private final String n = new String("derived");   // not a compile-time constant
        Derived() { super(); System.out.println("  Derived() sees name = " + name()); }
        @Override String name() { return n; }          // runs before n is assigned when called from Base()
    }

    // Q: shallow vs deep copy
    static class Team implements Cloneable {
        List<String> members = new ArrayList<>(List.of("asha"));
        @Override public Team clone() {
            try { return (Team) super.clone(); } catch (CloneNotSupportedException e) { throw new AssertionError(e); }
        }
        Team deepCopy() { Team t = new Team(); t.members = new ArrayList<>(members); return t; }
    }

    // Q: composition vs inheritance - an always-sorted list (asked at Walmart)
    static final class SortedList<T extends Comparable<T>> implements Iterable<T> {
        private final List<T> items = new ArrayList<>();      // HAS-A list; does not inherit add(int, T) or set()
        void add(T x) {
            int i = Collections.binarySearch(items, x);
            items.add(i < 0 ? -i - 1 : i, x);
        }
        T get(int i) { return items.get(i); }
        int size() { return items.size(); }
        @Override public Iterator<T> iterator() { return Collections.unmodifiableList(items).iterator(); }
        @Override public String toString() { return items.toString(); }
    }

    // Q: why must equals and hashCode agree? (see the toolkit page for the full demo)
    record Seat(String row, int number) {}

    // Q: String immutability and the pool
    static void strings() {
        String a = "dojo", b = "dojo", c = new String("dojo");
        System.out.println("  a == b: " + (a == b) + ", a == c: " + (a == c) + ", a.equals(c): " + a.equals(c));
    }

    public static void main(String[] args) throws Exception {
        System.out.println("1. overload vs override");
        Animal pet = new Dog();
        System.out.println("  pet.sound() = " + pet.sound() + "   (runtime type wins: overriding)");
        System.out.println("  describe(pet) = " + describe(pet) + "   (static type wins: overloading)");
        System.out.println("  Animal.kind() = " + Animal.kind() + ", Dog.kind() = " + Dog.kind() + "   (static methods are hidden, not overridden)");

        System.out.println("2. diamond via default methods");
        System.out.println("  " + new SmartPhone().start());

        System.out.println("3. constructor order");
        new Derived();

        System.out.println("4. shallow vs deep copy");
        Team t1 = new Team(); Team shallow = t1.clone(); Team deep = t1.deepCopy();
        t1.members.add("ravi");
        System.out.println("  shallow sees " + shallow.members + ", deep sees " + deep.members);

        System.out.println("5. composition: always-sorted list");
        SortedList<Integer> s = new SortedList<>();
        for (int x : new int[]{5, 1, 4, 1, 9}) s.add(x);
        System.out.println("  " + s + " size=" + s.size());

        System.out.println("6. records give value equality");
        Set<Seat> booked = new HashSet<>(List.of(new Seat("A", 1)));
        System.out.println("  booked.contains(A1) = " + booked.contains(new Seat("A", 1)));

        System.out.println("7. strings");
        strings();

        System.out.println("8. singleton survives serialization (readResolve)");
        Config before = Config.get();
        java.io.ByteArrayOutputStream bytes = new java.io.ByteArrayOutputStream();
        try (java.io.ObjectOutputStream out = new java.io.ObjectOutputStream(bytes)) { out.writeObject(before); }
        Object after;
        try (java.io.ObjectInputStream in = new java.io.ObjectInputStream(new java.io.ByteArrayInputStream(bytes.toByteArray()))) {
            after = in.readObject();
        }
        System.out.println("  same instance after deserialize: " + (after == before));
        if (after != before) throw new AssertionError("readResolve did not return the singleton");
        try {
            java.lang.reflect.Constructor<Config> ctor = Config.class.getDeclaredConstructor();
            ctor.setAccessible(true);
            ctor.newInstance();
            System.out.println("  reflection made a second Config");
        } catch (java.lang.reflect.InvocationTargetException e) {
            System.out.println("  reflection blocked: " + e.getCause().getMessage());
        }
    }

    // Q: thread-safe Singleton, and how else it can break (serialization, reflection)
    public static final class Config implements java.io.Serializable {
        private static final long serialVersionUID = 1L;
        private static volatile Config instance;          // volatile: no half-built object seen by other threads
        private Config() {
            if (instance != null) throw new IllegalStateException("use Config.get()");  // partial reflection guard
        }
        public static Config get() {
            Config c = instance;
            if (c == null) {
                synchronized (Config.class) {
                    c = instance;
                    if (c == null) instance = c = new Config();
                }
            }
            return c;
        }
        private Object readResolve() { return get(); }     // deserialization returns the one instance
    }
    // simplest complete answer: enum Config { INSTANCE; } (serialization and reflection safe by the JVM)
}
// lld-oop-interview: C++ questions (virtual destructor, diamond, object slicing).
// Build and run (C++14):  g++ -std=c++14 -Wall -o cpp_puzzles cpp_puzzles.cpp && ./cpp_puzzles
#include <iostream>
#include <memory>
#include <string>

// 1. virtual destructor
struct BufNoVirt { ~BufNoVirt() { std::cout << "  ~BufNoVirt\n"; } };
struct RingNoVirt : BufNoVirt { ~RingNoVirt() { std::cout << "  ~RingNoVirt (frees ring memory)\n"; } };

struct Buf { virtual ~Buf() { std::cout << "  ~Buf\n"; } };
struct Ring : Buf { ~Ring() override { std::cout << "  ~Ring (frees ring memory)\n"; } };

// 2. diamond: without virtual inheritance there are two Device sub-objects
struct Device { int id = 0; };
struct NetDev : Device {};
struct BlockDev : Device {};
struct Combo : NetDev, BlockDev {};                   // c.id is ambiguous: which Device?

struct VDevice { int id = 0; };
struct VNetDev : virtual VDevice {};
struct VBlockDev : virtual VDevice {};
struct VCombo : VNetDev, VBlockDev {};                // one shared VDevice

// 3. object slicing
struct Shape { virtual std::string name() const { return "shape"; } virtual ~Shape() = default; };
struct Circle : Shape { std::string name() const override { return "circle"; } };
std::string byValue(Shape s) { return s.name(); }     // copies only the Shape part
std::string byRef(const Shape& s) { return s.name(); }

int main() {
    std::cout << "1. delete through base pointer, no virtual dtor:\n";
    BufNoVirt* a = new RingNoVirt();
    delete a;                                          // undefined behaviour; in practice only ~BufNoVirt runs
    std::cout << "   with virtual dtor:\n";
    std::unique_ptr<Buf> b(new Ring());
    b.reset();

    std::cout << "2. diamond\n";
    Combo c;
    c.NetDev::id = 1; c.BlockDev::id = 2;
    std::cout << "  plain: two ids " << c.NetDev::id << " and " << c.BlockDev::id << ", sizeof(Combo)=" << sizeof(Combo) << "\n";
    VCombo v;
    v.id = 7;                                          // unambiguous
    std::cout << "  virtual: one id " << v.VNetDev::id << " == " << v.VBlockDev::id << "\n";

    std::cout << "3. slicing\n";
    Circle circle;
    std::cout << "  by value: " << byValue(circle) << ", by reference: " << byRef(circle) << "\n";
    return 0;
}
