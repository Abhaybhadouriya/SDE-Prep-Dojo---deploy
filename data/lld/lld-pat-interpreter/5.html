Code: rule engine (Java), pricing formula (Python), packet filter (C++14)

All three programs were compiled and run; output is from the real runs. The Java tab includes a small recursive-descent parser and an operator registry so a new operator needs no new class. Files live in lld-code/lld-pat-interpreter/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • Lambda x -> x * 2: a function pointer plus the variables it captured, packed into one object (like a callback with a void *ctx).
  • Map.of(k1, v1, ...), List.of(...): build a small read-only map or list in one expression.
  • static { ... }: a block that runs once when the class is loaded, like a module init function.
  • var (Java): let the compiler infer a local variable's type, like C23 auto.
  • Python ABC + @abstractmethod: a base class that refuses to be instantiated until a subclass fills the marked methods.
  • Python lambda x: ...: a one-expression anonymous function.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • std::shared_ptr<T>: a refcounted pointer, the kref of C++: the object is freed when the last owner drops it.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;
import java.util.function.BiPredicate;

// Interpreter: a tiny rule language, e.g.  age >= 18 && (city == BLR || vip == true) && !blocked == true
public class RuleDemo {

    // ---- AbstractExpression ----
    interface Expr { boolean interpret(Map<String, Object> ctx); }

    // ---- TerminalExpression: one comparison. Operators come from a registry (open for extension) ----
    static final Map<String, BiPredicate<Object, Object>> OPS = new LinkedHashMap<>();
    static {
        OPS.put("==", (a, b) -> String.valueOf(a).equals(String.valueOf(b)));
        OPS.put("!=", (a, b) -> !String.valueOf(a).equals(String.valueOf(b)));
        OPS.put(">=", (a, b) -> num(a) >= num(b));
        OPS.put("<=", (a, b) -> num(a) <= num(b));
        OPS.put(">",  (a, b) -> num(a) > num(b));
        OPS.put("<",  (a, b) -> num(a) < num(b));
    }
    static double num(Object o) { return Double.parseDouble(String.valueOf(o)); }

    record Compare(String var, String op, String literal) implements Expr {
        public boolean interpret(Map<String, Object> ctx) {
            Object actual = ctx.get(var);
            if (actual == null) return false;                       // unknown field: rule does not match
            return OPS.get(op).test(actual, literal);
        }
        public String toString() { return var + " " + op + " " + literal; }
    }

    // ---- NonterminalExpressions: one class per grammar rule, each interprets its children ----
    record And(Expr l, Expr r) implements Expr {
        public boolean interpret(Map<String, Object> c) { return l.interpret(c) && r.interpret(c); }
        public String toString() { return "AND(" + l + ", " + r + ")"; }
    }
    record Or(Expr l, Expr r) implements Expr {
        public boolean interpret(Map<String, Object> c) { return l.interpret(c) || r.interpret(c); }
        public String toString() { return "OR(" + l + ", " + r + ")"; }
    }
    record Not(Expr e) implements Expr {
        public boolean interpret(Map<String, Object> c) { return !e.interpret(c); }
        public String toString() { return "NOT(" + e + ")"; }
    }

    // ---- Parser (not part of the GoF pattern, but you always need one) ----
    //   or    := and ( "||" and )*
    //   and   := unary ( "&&" unary )*
    //   unary := "!" unary | "(" or ")" | IDENT OP VALUE
    static final class Parser {
        private final List<String> toks = new ArrayList<>();
        private int pos = 0;

        Parser(String src) {
            int i = 0;
            while (i < src.length()) {
                char ch = src.charAt(i);
                if (Character.isWhitespace(ch)) { i++; continue; }
                String two = i + 1 < src.length() ? src.substring(i, i + 2) : "";
                if (two.equals("&&") || two.equals("||") || OPS.containsKey(two)) { toks.add(two); i += 2; continue; }
                if ("()!<>".indexOf(ch) >= 0) { toks.add(String.valueOf(ch)); i++; continue; }
                int j = i;
                while (j < src.length() && (Character.isLetterOrDigit(src.charAt(j)) || src.charAt(j) == '.' || src.charAt(j) == '_')) j++;
                if (j == i) throw new IllegalArgumentException("bad char '" + ch + "' at " + i);
                toks.add(src.substring(i, j));
                i = j;
            }
        }
        private String peek() { return pos < toks.size() ? toks.get(pos) : ""; }
        private String next() {
            if (pos >= toks.size()) throw new IllegalArgumentException("unexpected end of rule");
            return toks.get(pos++);
        }
        private void expect(String t) {
            if (!peek().equals(t)) throw new IllegalArgumentException("expected '" + t + "' but got '" + peek() + "'");
            pos++;
        }

        Expr parse() {
            Expr e = or();
            if (pos != toks.size()) throw new IllegalArgumentException("unexpected '" + peek() + "'");
            return e;
        }
        private Expr or()  { Expr e = and();   while (peek().equals("||")) { next(); e = new Or(e, and()); }  return e; }
        private Expr and() { Expr e = unary(); while (peek().equals("&&")) { next(); e = new And(e, unary()); } return e; }
        private Expr unary() {
            if (peek().equals("!")) { next(); return new Not(unary()); }
            if (peek().equals("(")) { next(); Expr e = or(); expect(")"); return e; }
            String var = next();
            String op = next();
            if (!OPS.containsKey(op)) throw new IllegalArgumentException("unknown operator '" + op + "'");
            return new Compare(var, op, next());
        }
    }

    public static void main(String[] args) {
        String rule = "age >= 18 && (city == BLR || vip == true) && !(blocked == true)";
        Expr tree = new Parser(rule).parse();                     // parse once ...
        System.out.println("rule : " + rule);
        System.out.println("tree : " + tree);

        List<Map<String, Object>> users = List.of(
            Map.of("name", "Asha", "age", 25, "city", "BLR", "vip", false, "blocked", false),
            Map.of("name", "Ravi", "age", 31, "city", "DEL", "vip", true,  "blocked", false),
            Map.of("name", "Meena", "age", 16, "city", "BLR", "vip", true, "blocked", false),
            Map.of("name", "Kabir", "age", 40, "city", "BLR", "vip", false, "blocked", true));
        for (Map<String, Object> u : users)                       // ... interpret many times
            System.out.printf("  %-6s -> %b%n", u.get("name"), tree.interpret(u));

        // Extending the language: a new operator is one registry entry, no parser or class change.
        OPS.put("in", (a, b) -> Arrays.asList(String.valueOf(b).split("_")).contains(String.valueOf(a)));
        Expr metro = new Parser("city in BLR_MUM_DEL && age < 30").parse();
        System.out.println("tree : " + metro);
        System.out.println("  Asha   -> " + metro.interpret(users.get(0)));

        try { new Parser("age >= 18 &&").parse(); }
        catch (RuntimeException e) { System.out.println("error: " + e.getMessage()); }
    }
}

Two design choices matter in an interview. Precedence lives in the grammar: && binds tighter than || because or() calls and(). Extensibility: operators sit in a map, so adding in was one line, with no parser or class change.

"""Interpreter: a pricing formula language, e.g.  "base * qty - discount" with variables."""
from abc import ABC, abstractmethod


class Expr(ABC):                                 # AbstractExpression
    @abstractmethod
    def interpret(self, ctx: dict) -> float: ...


class Num(Expr):                                 # TerminalExpression
    def __init__(self, v):
        self.v = float(v)

    def interpret(self, ctx):
        return self.v

    def __repr__(self):
        return f"{self.v:g}"


class Var(Expr):                                 # TerminalExpression
    def __init__(self, name):
        self.name = name

    def interpret(self, ctx):
        if self.name not in ctx:
            raise KeyError(f"unknown variable '{self.name}'")
        return float(ctx[self.name])

    def __repr__(self):
        return self.name


class BinOp(Expr):                               # NonterminalExpression
    FUNCS = {"+": lambda a, b: a + b, "-": lambda a, b: a - b,
             "*": lambda a, b: a * b, "/": lambda a, b: a / b,
             "max": max, "min": min}

    def __init__(self, op, left, right):
        self.op, self.left, self.right = op, left, right

    def interpret(self, ctx):
        return self.FUNCS[self.op](self.left.interpret(ctx), self.right.interpret(ctx))

    def __repr__(self):
        return f"({self.left} {self.op} {self.right})"


def parse_rpn(text: str) -> Expr:
    """Reverse Polish notation needs no precedence rules: 'base qty * discount -'."""
    stack = []
    for tok in text.split():
        if tok in BinOp.FUNCS:
            right, left = stack.pop(), stack.pop()
            stack.append(BinOp(tok, left, right))
        elif tok.replace(".", "", 1).isdigit():
            stack.append(Num(tok))
        else:
            stack.append(Var(tok))
    if len(stack) != 1:
        raise ValueError("malformed expression")
    return stack[0]


if __name__ == "__main__":
    # "subtotal minus discount, but never below the minimum order fee"
    formula = parse_rpn("base qty * discount - fee max")
    print("tree :", formula)
    for order in ({"base": 120, "qty": 3, "discount": 50, "fee": 99},
                  {"base": 40, "qty": 2, "discount": 30, "fee": 99}):
        print("price:", formula.interpret(order), "for", order)
    try:
        formula.interpret({"base": 1})
    except KeyError as e:
        print("error:", e)

Reverse Polish notation keeps the parser to a dozen lines (no brackets, no precedence), which makes it a good trick when time is short. A missing variable fails loudly instead of silently pricing at zero.

// Interpreter in C++14: a tcpdump-like packet filter language, as a tree of expression objects.
// Filter: (tcp && dport == 22) || icmp
// Build: g++ -std=c++14 -Wall interpreter.cpp -o interpreter
#include <cstdint>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

struct Packet {            // the "context" every expression is evaluated against
    std::uint8_t proto;    // 1 = ICMP, 6 = TCP, 17 = UDP (IP protocol numbers)
    std::uint16_t dport;
    std::string desc;
};

class Expr {                                     // AbstractExpression
public:
    virtual ~Expr() = default;
    virtual bool interpret(const Packet& p) const = 0;
    virtual std::string show() const = 0;
};
using ExprPtr = std::shared_ptr<const Expr>;

class ProtoIs : public Expr {                    // TerminalExpression: "tcp", "icmp"
    std::uint8_t proto_; std::string name_;
public:
    ProtoIs(std::uint8_t p, std::string n) : proto_(p), name_(std::move(n)) {}
    bool interpret(const Packet& p) const override { return p.proto == proto_; }
    std::string show() const override { return name_; }
};

class DportEq : public Expr {                    // TerminalExpression: "dport == N"
    std::uint16_t port_;
public:
    explicit DportEq(std::uint16_t port) : port_(port) {}
    bool interpret(const Packet& p) const override { return p.dport == port_; }
    std::string show() const override { return "dport == " + std::to_string(port_); }
};

class And : public Expr {                        // NonterminalExpression
    ExprPtr l_, r_;
public:
    And(ExprPtr l, ExprPtr r) : l_(std::move(l)), r_(std::move(r)) {}
    bool interpret(const Packet& p) const override { return l_->interpret(p) && r_->interpret(p); }
    std::string show() const override { return "(" + l_->show() + " && " + r_->show() + ")"; }
};

class Or : public Expr {                         // NonterminalExpression
    ExprPtr l_, r_;
public:
    Or(ExprPtr l, ExprPtr r) : l_(std::move(l)), r_(std::move(r)) {}
    bool interpret(const Packet& p) const override { return l_->interpret(p) || r_->interpret(p); }
    std::string show() const override { return "(" + l_->show() + " || " + r_->show() + ")"; }
};

int main() {
    // Built by hand here; a real tool would parse the text into this tree.
    ExprPtr tcp  = std::make_shared<ProtoIs>(6, "tcp");
    ExprPtr icmp = std::make_shared<ProtoIs>(1, "icmp");
    ExprPtr filter = std::make_shared<Or>(std::make_shared<And>(tcp, std::make_shared<DportEq>(22)), icmp);

    std::cout << "filter: " << filter->show() << "\n";
    std::vector<Packet> pkts = {
        {6, 22, "TCP to :22 (ssh)"},
        {6, 443, "TCP to :443"},
        {1, 0, "ICMP echo"},
        {17, 22, "UDP to :22"},
    };
    int accepted = 0;
    for (const auto& p : pkts) {
        bool ok = filter->interpret(p);
        accepted += ok;
        std::cout << "  " << (ok ? "ACCEPT " : "drop   ") << p.desc << "\n";
    }
    std::cout << accepted << " of " << pkts.size() << " packets matched\n";
    return 0;
}
$ g++ -std=c++14 -Wall interpreter.cpp -o interpreter && ./interpreter
filter: ((tcp && dport == 22) || icmp)
  ACCEPT TCP to :22 (ssh)
  drop   TCP to :443
  ACCEPT ICMP echo
  drop   UDP to :22
2 of 4 packets matched

The same filter that tcpdump would compile to BPF, here walked as a tree of objects. Sub-trees are shared_ptr<const Expr>, so a common piece (tcp) can be reused in many filters, and the tree is immutable, so many threads can interpret it at once.
