import java.util.*;
import java.util.function.Predicate;
import java.util.stream.Collectors;

public class SpecificationDemo {

    // ---- The Specification interface: one question, plus combinators ----
    interface Spec<T> {
        boolean isSatisfiedBy(T t);
        // <-- the pattern: combinators build a Composite of rules
        default Spec<T> and(Spec<T> o) { return new AndSpec<>(this, o); }
        default Spec<T> or(Spec<T> o)  { return new OrSpec<>(this, o); }
        default Spec<T> not()          { return new NotSpec<>(this); }
        default Predicate<T> asPredicate() { return this::isSatisfiedBy; }   // interop with streams
        String describe();
    }
    // Composite specs: Spec objects that hold other Spec objects (the Composite pattern).
    record AndSpec<T>(Spec<T> a, Spec<T> b) implements Spec<T> {
        public boolean isSatisfiedBy(T t) { return a.isSatisfiedBy(t) && b.isSatisfiedBy(t); }  // short-circuits
        public String describe() { return "(" + a.describe() + " AND " + b.describe() + ")"; }
    }
    record OrSpec<T>(Spec<T> a, Spec<T> b) implements Spec<T> {
        public boolean isSatisfiedBy(T t) { return a.isSatisfiedBy(t) || b.isSatisfiedBy(t); }
        public String describe() { return "(" + a.describe() + " OR " + b.describe() + ")"; }
    }
    record NotSpec<T>(Spec<T> s) implements Spec<T> {
        public boolean isSatisfiedBy(T t) { return !s.isSatisfiedBy(t); }
        public String describe() { return "NOT " + s.describe(); }
    }

    // ---- Domain 1: car rental search ----
    enum CarType { HATCHBACK, SEDAN, SUV }
    record Car(String id, CarType type, int seats, int pricePerDay, boolean automatic, String city) { }

    record TypeIs(CarType type) implements Spec<Car> {
        public boolean isSatisfiedBy(Car c) { return c.type() == type; }
        public String describe() { return "type=" + type; }
    }
    record SeatsAtLeast(int n) implements Spec<Car> {
        public boolean isSatisfiedBy(Car c) { return c.seats() >= n; }
        public String describe() { return "seats>=" + n; }
    }
    record PriceAtMost(int max) implements Spec<Car> {
        public boolean isSatisfiedBy(Car c) { return c.pricePerDay() <= max; }
        public String describe() { return "price<=" + max; }
    }
    record InCity(String city) implements Spec<Car> {
        public boolean isSatisfiedBy(Car c) { return c.city().equals(city); }
        public String describe() { return "city=" + city; }
    }
    static final Spec<Car> AUTOMATIC = new Spec<>() {
        public boolean isSatisfiedBy(Car c) { return c.automatic(); }
        public String describe() { return "automatic"; }
    };

    static <T> List<T> filter(Collection<T> items, Spec<T> spec) {     // repository.findAll(spec) in memory
        List<T> out = new ArrayList<>();
        for (T t : items) if (spec.isSatisfiedBy(t)) out.add(t);
        return out;
    }

    // ---- Domain 2: coupon eligibility rules ("age > 18 and cart value > 1000") ----
    record Ctx(int age, int cartValue, boolean firstOrder, String paymentMode) { }
    static Spec<Ctx> rule(String name, Predicate<Ctx> p) {             // quick leaf from a lambda
        return new Spec<>() {
            public boolean isSatisfiedBy(Ctx c) { return p.test(c); }
            public String describe() { return name; }
        };
    }

    public static void main(String[] args) {
        List<Car> fleet = List.of(
            new Car("C1", CarType.HATCHBACK, 5, 1500, false, "BLR"),
            new Car("C2", CarType.SEDAN, 5, 2500, true, "BLR"),
            new Car("C3", CarType.SUV, 7, 4000, true, "BLR"),
            new Car("C4", CarType.SUV, 7, 3200, false, "BLR"),
            new Car("C5", CarType.SUV, 7, 3000, true, "DEL"));

        Spec<Car> familyTrip = new InCity("BLR").and(new SeatsAtLeast(7)).and(new PriceAtMost(3500));
        Spec<Car> cheapOrAuto = new InCity("BLR").and(new PriceAtMost(2000).or(AUTOMATIC));
        Spec<Car> notSuv = new TypeIs(CarType.SUV).not();

        for (Spec<Car> s : List.of(familyTrip, cheapOrAuto, notSuv))
            System.out.println(s.describe() + " -> " +
                filter(fleet, s).stream().map(Car::id).collect(Collectors.joining(",")));

        // Same spec object used in a stream, through the Predicate bridge:
        long n = fleet.stream().filter(familyTrip.asPredicate()).count();
        System.out.println("stream count for familyTrip = " + n);

        // Coupon rules built by an admin, stored as data, evaluated at checkout.
        Spec<Ctx> adult = rule("age>18", c -> c.age() > 18);
        Spec<Ctx> bigCart = rule("cart>1000", c -> c.cartValue() > 1000);
        Spec<Ctx> upi = rule("pay=UPI", c -> c.paymentMode().equals("UPI"));
        Spec<Ctx> first = rule("firstOrder", c -> c.firstOrder());
        Spec<Ctx> coupon = adult.and(bigCart).and(upi.or(first));
        System.out.println("coupon rule: " + coupon.describe());
        for (Ctx c : List.of(new Ctx(25, 1500, false, "UPI"), new Ctx(17, 5000, true, "CARD"),
                             new Ctx(30, 1200, true, "CARD"), new Ctx(30, 800, false, "UPI")))
            System.out.println("  " + c + " eligible=" + coupon.isSatisfiedBy(c));
    }
}

describe() is not decoration. In a review call, printing the rule tree shows the interviewer that rules are data: you can log which rule rejected a coupon, show it on an admin screen, or serialise it to JSON and load it back.

"""Specification with operator overloading: spec_a & spec_b, spec_a | spec_b, ~spec."""
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable, Generic, Iterable, TypeVar

T = TypeVar("T")


class Spec(Generic[T]):
    def is_satisfied_by(self, item: T) -> bool:
        raise NotImplementedError

    def __and__(self, other: Spec[T]) -> Spec[T]:
        return AndSpec(self, other)

    def __or__(self, other: Spec[T]) -> Spec[T]:
        return OrSpec(self, other)

    def __invert__(self) -> Spec[T]:
        return NotSpec(self)

    def __call__(self, item: T) -> bool:          # a spec can be passed anywhere a predicate is expected
        return self.is_satisfied_by(item)


@dataclass(frozen=True)
class AndSpec(Spec[T]):
    a: Spec[T]
    b: Spec[T]

    def is_satisfied_by(self, item):
        return self.a.is_satisfied_by(item) and self.b.is_satisfied_by(item)

    def __repr__(self):
        return f"({self.a!r} & {self.b!r})"


@dataclass(frozen=True)
class OrSpec(Spec[T]):
    a: Spec[T]
    b: Spec[T]

    def is_satisfied_by(self, item):
        return self.a.is_satisfied_by(item) or self.b.is_satisfied_by(item)

    def __repr__(self):
        return f"({self.a!r} | {self.b!r})"


@dataclass(frozen=True)
class NotSpec(Spec[T]):
    s: Spec[T]

    def is_satisfied_by(self, item):
        return not self.s.is_satisfied_by(item)

    def __repr__(self):
        return f"~{self.s!r}"


class Leaf(Spec[T]):
    """A named leaf built from a plain function."""
    def __init__(self, name: str, fn: Callable[[T], bool]):
        self.name, self.fn = name, fn

    def is_satisfied_by(self, item):
        return self.fn(item)

    def __repr__(self):
        return self.name


# ---- a tiny "find": files as data, filters as specs ----
@dataclass(frozen=True)
class File:
    path: str
    size_kb: int
    owner: str
    days_old: int


def name_ends(ext: str) -> Spec[File]:
    return Leaf(f"-name '*{ext}'", lambda f: f.path.endswith(ext))


def size_over(kb: int) -> Spec[File]:
    return Leaf(f"-size +{kb}k", lambda f: f.size_kb > kb)


def owner(user: str) -> Spec[File]:
    return Leaf(f"-user {user}", lambda f: f.owner == user)


def older_than(days: int) -> Spec[File]:
    return Leaf(f"-mtime +{days}", lambda f: f.days_old > days)


def find(files: Iterable[File], spec: Spec[File]) -> list[str]:
    return [f.path for f in files if spec.is_satisfied_by(f)]


if __name__ == "__main__":
    files = [
        File("/var/log/syslog", 900, "root", 1),
        File("/var/log/kern.log", 4200, "root", 12),
        File("/home/asha/trace.pcap", 52000, "asha", 40),
        File("/home/asha/notes.txt", 3, "asha", 2),
        File("/tmp/core.1234", 12000, "asha", 9),
    ]
    big_old_logs = name_ends(".log") & size_over(1000) & older_than(7)
    ashas_big = owner("asha") & (size_over(10000) | older_than(30))
    not_root = ~owner("root")

    for spec in (big_old_logs, ashas_big, not_root):
        print(spec, "->", find(files, spec))

    # Specs are callables, so they work with built-ins too:
    print("filter():", [f.path for f in filter(ashas_big, files)])

Python's operator overloading makes specs read like the rule itself. Making Spec callable (__call__) lets the same object go into filter(), sorted(key=...) helpers or any function that expects a predicate. Django's Q objects use the same operators for database filters.

C++ you need here: std::unique_ptr<T> = an owning pointer that frees its object when it goes out of scope (a kfree() that runs itself on every exit path). A lambda [](args){ body } = an anonymous function: a function pointer plus any variables it captured in the brackets, like a callback with its void *ctx built in. std::function<R(Args)> = a typed slot that can hold any callable (function pointer, lambda, functor). Ownership and RAII in more depth: kref and shared ownership.

// Specification in C++14: a small tree of predicate objects, like a packet-filter rule.
// Build: g++ -std=c++14 -Wall specification.cpp -o specification
#include <functional>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

template <class T>
class Spec {
public:
    virtual ~Spec() = default;
    virtual bool isSatisfiedBy(const T& t) const = 0;
    virtual std::string describe() const = 0;
};
template <class T> using SpecPtr = std::shared_ptr<const Spec<T>>;

template <class T>
class AndSpec : public Spec<T> {
    SpecPtr<T> a_, b_;
public:
    AndSpec(SpecPtr<T> a, SpecPtr<T> b) : a_(std::move(a)), b_(std::move(b)) {}
    bool isSatisfiedBy(const T& t) const override { return a_->isSatisfiedBy(t) && b_->isSatisfiedBy(t); }
    std::string describe() const override { return "(" + a_->describe() + " and " + b_->describe() + ")"; }
};
template <class T>
class OrSpec : public Spec<T> {
    SpecPtr<T> a_, b_;
public:
    OrSpec(SpecPtr<T> a, SpecPtr<T> b) : a_(std::move(a)), b_(std::move(b)) {}
    bool isSatisfiedBy(const T& t) const override { return a_->isSatisfiedBy(t) || b_->isSatisfiedBy(t); }
    std::string describe() const override { return "(" + a_->describe() + " or " + b_->describe() + ")"; }
};
template <class T>
class NotSpec : public Spec<T> {
    SpecPtr<T> s_;
public:
    explicit NotSpec(SpecPtr<T> s) : s_(std::move(s)) {}
    bool isSatisfiedBy(const T& t) const override { return !s_->isSatisfiedBy(t); }
    std::string describe() const override { return "not " + s_->describe(); }
};
template <class T>
class Leaf : public Spec<T> {                      // a leaf wraps any callable
    std::string name_;
    std::function<bool(const T&)> fn_;
public:
    Leaf(std::string n, std::function<bool(const T&)> f) : name_(std::move(n)), fn_(std::move(f)) {}
    bool isSatisfiedBy(const T& t) const override { return fn_(t); }
    std::string describe() const override { return name_; }
};

// Operators make rules read like a filter expression.
template <class T> SpecPtr<T> operator&&(SpecPtr<T> a, SpecPtr<T> b) { return std::make_shared<AndSpec<T>>(a, b); }
template <class T> SpecPtr<T> operator||(SpecPtr<T> a, SpecPtr<T> b) { return std::make_shared<OrSpec<T>>(a, b); }
template <class T> SpecPtr<T> operator!(SpecPtr<T> s) { return std::make_shared<NotSpec<T>>(s); }

// ---- domain: packets, like a tcpdump / BPF filter ----
struct Packet { std::string proto; int dport; std::string src; int len; };
using P = SpecPtr<Packet>;

P proto(const std::string& p) { return std::make_shared<Leaf<Packet>>(p, [p](const Packet& k) { return k.proto == p; }); }
P port(int n) { return std::make_shared<Leaf<Packet>>("port " + std::to_string(n), [n](const Packet& k) { return k.dport == n; }); }
P host(const std::string& h) { return std::make_shared<Leaf<Packet>>("host " + h, [h](const Packet& k) { return k.src == h; }); }
P greater(int n) { return std::make_shared<Leaf<Packet>>("greater " + std::to_string(n), [n](const Packet& k) { return k.len > n; }); }

int main() {
    std::vector<Packet> pkts = {
        {"tcp", 443, "10.0.0.5", 1400},
        {"tcp", 80, "10.0.0.9", 60},
        {"udp", 53, "10.0.0.5", 80},
        {"tcp", 22, "192.168.1.7", 120},
        {"tcp", 443, "192.168.1.7", 1500},
    };
    P web = proto("tcp") && (port(80) || port(443));
    P bigWebNotLocal = web && greater(1000) && !host("10.0.0.5");
    P dns = proto("udp") && port(53);

    for (const P& f : {web, bigWebNotLocal, dns}) {
        std::cout << f->describe() << " ->";
        for (size_t i = 0; i < pkts.size(); ++i)
            if (f->isSatisfiedBy(pkts[i])) std::cout << " #" << i;
        std::cout << "\n";
    }
    return 0;
}

The overloaded &&, || and ! build the tree; short-circuiting happens later, inside AndSpec::isSatisfiedBy. (Overloaded && itself never short-circuits in C++, which is fine here because building has no side effects.) Specs are immutable and shared through shared_ptr<const Spec>, so one leaf can sit in many trees.
