"""Autocomplete (trie + top-k + ranking Strategy) and a wildcard path router (trie by segment)."""
from __future__ import annotations

import heapq
import re
import threading
from typing import Dict, Generic, List, Optional, Tuple, TypeVar


# ---------------- Part 1: autocomplete ----------------

class Entry:
    __slots__ = ("text", "count", "last_used")

    def __init__(self, text: str):
        self.text, self.count, self.last_used = text, 0, 0


class RankingStrategy:
    """Strategy: a sort key, smaller is better. Scores may only go UP when a phrase is recorded."""
    name = "abstract"

    def key(self, e: Entry) -> Tuple:
        raise NotImplementedError


class ByFrequency(RankingStrategy):
    name = "frequency"

    def key(self, e):
        return (-e.count, e.text)          # ties: alphabetical (LeetCode 642 rule)


class ByRecency(RankingStrategy):
    name = "recency"

    def key(self, e):
        return (-e.last_used, e.text)


class TrieNode:
    __slots__ = ("children", "entry", "top_k")

    def __init__(self):
        self.children: Dict[str, TrieNode] = {}
        self.entry: Optional[Entry] = None
        self.top_k: Optional[List[Entry]] = None


def normalize(s: Optional[str], trim_end: bool) -> str:
    t = re.sub(r"\s+", " ", (s or "").lower()).lstrip(" ")
    return t.strip() if trim_end else t


class RWLock:
    """Many readers or one writer (the stdlib has none). Writer-preferring: once a writer waits,
    new readers queue behind it, so a steady stream of readers cannot starve writers."""

    def __init__(self):
        self._cond = threading.Condition()
        self._readers = 0
        self._writer = False
        self._waiting_writers = 0                 # new readers yield to a queued writer

    def acquire_read(self):
        with self._cond:
            while self._writer or self._waiting_writers:
                self._cond.wait()
            self._readers += 1

    def release_read(self):
        with self._cond:
            self._readers -= 1
            if self._readers == 0:
                self._cond.notify_all()

    def acquire_write(self):
        with self._cond:
            self._waiting_writers += 1
            try:
                while self._writer or self._readers:
                    self._cond.wait()
            finally:
                self._waiting_writers -= 1
            self._writer = True

    def release_write(self):
        with self._cond:
            self._writer = False
            self._cond.notify_all()


class AutocompleteEngine:
    def __init__(self, rank: RankingStrategy, k: int, cache_top_k: bool):
        if k < 1:
            raise ValueError("k must be >= 1")
        self.rank, self.k, self.cache = rank, k, cache_top_k
        self._root = TrieNode()
        self._lock = RWLock()
        self._clock = 0

    def describe(self) -> str:
        return f"{self.rank.name}, k={self.k}, " + ("cached top-k" if self.cache else "scan")

    def record(self, phrase: str, times: int = 1) -> None:
        p = normalize(phrase, True)
        if not p:
            raise ValueError("empty phrase")
        if times < 1:
            raise ValueError("times must be >= 1")
        self._lock.acquire_write()
        try:
            n, path = self._root, [self._root]
            for ch in p:
                n = n.children.setdefault(ch, TrieNode())
                path.append(n)
            if n.entry is None:
                n.entry = Entry(p)
            n.entry.count += times
            self._clock += 1
            n.entry.last_used = self._clock
            if self.cache:
                for node in path:
                    self._offer(node, n.entry)
        finally:
            self._lock.release_write()

    def _offer(self, node: TrieNode, e: Entry) -> None:
        # e's score only went up and nobody else's changed: move it in, re-sort, drop the (k+1)th.
        lst = node.top_k if node.top_k is not None else []
        lst = [x for x in lst if x is not e] + [e]
        lst.sort(key=self.rank.key)
        node.top_k = lst[: self.k]

    def remove(self, phrase: str) -> bool:
        p = normalize(phrase, True)
        self._lock.acquire_write()
        try:
            n, path = self._root, [self._root]
            for ch in p:
                n = n.children.get(ch)
                if n is None:
                    return False
                path.append(n)
            if n.entry is None:
                return False
            n.entry = None
            for i in range(len(path) - 1, 0, -1):          # prune empty leaves
                child = path[i]
                if child.entry is None and not child.children:
                    del path[i - 1].children[p[i - 1]]
                else:
                    break
            if self.cache:                                 # removal breaks "only up": rebuild bottom-up
                for node in reversed(path):
                    self._rebuild(node)
            return True
        finally:
            self._lock.release_write()

    def _rebuild(self, node: TrieNode) -> None:
        pool = [node.entry] if node.entry else []
        for ch in node.children.values():
            pool.extend(ch.top_k or [])
        pool.sort(key=self.rank.key)
        node.top_k = pool[: self.k]

    def suggest(self, prefix: str) -> List[Tuple[str, int]]:
        p = normalize(prefix, False)
        self._lock.acquire_read()
        try:
            n = self._root
            for ch in p:
                n = n.children.get(ch)
                if n is None:
                    return []
            best = (n.top_k or []) if self.cache else self._scan(n)
            return [(e.text, e.count) for e in best]
        finally:
            self._lock.release_read()

    def _scan(self, start: TrieNode) -> List[Entry]:
        found, stack = [], [start]
        while stack:
            n = stack.pop()
            if n.entry is not None:
                found.append(n.entry)
            stack.extend(n.children.values())
        return heapq.nsmallest(self.k, found, key=self.rank.key)

    def session(self) -> "TypingSession":
        return TypingSession(self)


class TypingSession:
    """LeetCode 642 style: characters one by one; '#' ends the sentence and records it."""

    def __init__(self, engine: AutocompleteEngine):
        self._engine, self._typed = engine, []

    def type(self, c: str) -> List[Tuple[str, int]]:
        if c == "#":
            self._engine.record("".join(self._typed))
            self._typed.clear()
            return []
        self._typed.append(c)
        return self._engine.suggest("".join(self._typed))


def fmt(sugs: List[Tuple[str, int]]) -> str:
    return "[" + ", ".join(f"{t}({c})" for t, c in sugs) + "]"


# ---------------- Part 2: wildcard router ----------------

V = TypeVar("V")


class RouteMatch(Generic[V]):
    def __init__(self, value: V, pattern: str, params: Dict[str, str]):
        self.value, self.pattern, self.params = value, pattern, params

    def __repr__(self) -> str:
        ps = "" if not self.params else " {" + ", ".join(f"{k}={v}" for k, v in self.params.items()) + "}"
        return f"{self.value} via {self.pattern}{ps}"


class _Node:
    __slots__ = ("literal", "param", "param_name", "star", "globstar", "value", "pattern")

    def __init__(self):
        self.literal: Dict[str, _Node] = {}
        self.param: Optional[_Node] = None
        self.param_name: Optional[str] = None
        self.star: Optional[_Node] = None
        self.globstar: Optional[_Node] = None
        self.value = None
        self.pattern: Optional[str] = None


def segments(path: Optional[str]) -> List[str]:
    if not path or not path.startswith("/"):
        raise ValueError(f"path must start with /: {path}")
    return [s for s in path.split("/") if s]            # "/a//b/" == "/a/b"


class PathTrie(Generic[V]):
    """put(pattern): literal, ':name', '*' (one segment), '**' (rest, last only).
    match(path): literal beats :param beats * beats **, with backtracking.
    find_all(query): '*' in the QUERY over plain stored keys (the KV follow-up)."""

    def __init__(self):
        self._root = _Node()
        self._lock = RWLock()

    def put(self, pattern: str, value: V) -> None:
        segs = segments(pattern)
        norm = "/" + "/".join(segs)
        self._lock.acquire_write()
        try:
            n = self._root
            for i, s in enumerate(segs):
                if s == "**":
                    if i != len(segs) - 1:
                        raise ValueError(f"** must be the last segment: {pattern}")
                    n.globstar = n.globstar or _Node()
                    n = n.globstar
                elif s == "*":
                    n.star = n.star or _Node()
                    n = n.star
                elif s.startswith(":"):
                    name = s[1:]
                    if not name:
                        raise ValueError(f"empty parameter name: {pattern}")
                    if n.param is None:
                        n.param, n.param_name = _Node(), name
                    elif n.param_name != name:
                        raise ValueError(f"conflict: :{name} vs :{n.param_name} in {pattern}")
                    n = n.param
                else:
                    n = n.literal.setdefault(s, _Node())
            if n.value is not None:
                raise ValueError(f"duplicate route {norm} (already {n.pattern})")
            n.value, n.pattern = value, norm
        finally:
            self._lock.release_write()

    def match(self, path: str) -> Optional[RouteMatch[V]]:
        segs = segments(path)
        self._lock.acquire_read()
        try:
            params: Dict[str, str] = {}
            hit = self._match(self._root, segs, 0, params)
            return None if hit is None else RouteMatch(hit.value, hit.pattern, params)
        finally:
            self._lock.release_read()

    def _match(self, n: _Node, segs: List[str], i: int, params: Dict[str, str]) -> Optional[_Node]:
        if i == len(segs):
            if n.value is not None:
                return n
            if n.globstar is not None and n.globstar.value is not None:
                params["**"] = ""
                return n.globstar
            return None
        s = segs[i]
        lit = n.literal.get(s)
        if lit is not None:
            r = self._match(lit, segs, i + 1, params)
            if r is not None:
                return r
        if n.param is not None:
            params[n.param_name] = s
            r = self._match(n.param, segs, i + 1, params)
            if r is not None:
                return r
            del params[n.param_name]
        if n.star is not None:
            r = self._match(n.star, segs, i + 1, params)
            if r is not None:
                return r
        if n.globstar is not None and n.globstar.value is not None:
            params["**"] = "/".join(segs[i:])
            return n.globstar
        return None

    def find_all(self, query: str) -> List[str]:
        segs = segments(query)
        out: List[str] = []
        self._lock.acquire_read()
        try:
            self._collect(self._root, segs, 0, out)
        finally:
            self._lock.release_read()
        return out

    def _collect(self, n: _Node, segs: List[str], i: int, out: List[str]) -> None:
        if i == len(segs):
            if n.value is not None:
                out.append(f"{n.pattern}={n.value}")
            return
        s = segs[i]
        if s == "*":
            for key in sorted(n.literal):
                self._collect(n.literal[key], segs, i + 1, out)
        elif s in n.literal:
            self._collect(n.literal[s], segs, i + 1, out)

"""Demo driver: the same story as the Java Main."""
import random
import threading

from autocomplete import AutocompleteEngine, ByFrequency, ByRecency, PathTrie, fmt


def check(cond, msg="check failed"):
    """Like assert, but never stripped by python -O: the call inside it always runs."""
    if not cond:
        raise AssertionError(msg)


def expect_error(fn, contains):
    try:
        fn()
    except ValueError as e:
        print("  rejected:", e)
        check(contains in str(e), e)
        return
    raise AssertionError("expected error: " + contains)


def seed(e):
    e.record("linux kernel", 5)
    e.record("linked list", 3)
    e.record("lint rules", 3)
    e.record("linus torvalds", 2)
    e.record("LRU   Cache", 4)              # normalised to "lru cache"
    e.record("linux kernel modules", 1)


def show(m):
    return "404" if m is None else repr(m)


def main():
    print("== autocomplete")
    freq = AutocompleteEngine(ByFrequency(), 3, True)
    recent = AutocompleteEngine(ByRecency(), 3, True)
    seed(freq)
    seed(recent)
    print(f"  [{freq.describe()}] 'lin' -> {fmt(freq.suggest('lin'))}")
    print(f"  [{recent.describe()}] 'lin' -> {fmt(recent.suggest('lin'))}")
    check(fmt(freq.suggest("lin")) == "[linux kernel(5), linked list(3), lint rules(3)]")
    check(recent.suggest("lin")[0][0] == "linux kernel modules")
    print(f"  'LI' -> {fmt(freq.suggest('LI'))}   'xyz' -> {fmt(freq.suggest('xyz'))}")
    check(freq.suggest("xyz") == [])

    print("== typing session (LeetCode 642 style)")
    t = freq.session()
    for c in "lru cache#":
        s = t.type(c)
        if c in "lr#":
            print(f"  typed '{c}' -> {fmt(s)}")
    print(f"  'l' now -> {fmt(freq.suggest('l'))}")
    check(freq.suggest("l")[1] == ("lru cache", 5))

    print("== remove 'linux kernel'")
    check(freq.remove("linux kernel"))
    check(not freq.remove("linux kernel"))
    print(f"  'lin' -> {fmt(freq.suggest('lin'))}   'linux' -> {fmt(freq.suggest('linux'))}")
    check(fmt(freq.suggest("linux")) == "[linux kernel modules(1)]")

    print("== cached top-k equals full scan (random workload)")
    rnd = random.Random(7)
    vocab = ["a", "ab", "abc", "abd", "b", "ba", "bab", "bad", "c", "ca", "cab", "cad", "dab", "dad"]
    cached = AutocompleteEngine(ByFrequency(), 3, True)
    scan = AutocompleteEngine(ByFrequency(), 3, False)
    mismatches, ops = 0, 2000
    for _ in range(ops):
        w = rnd.choice(vocab)
        if rnd.randrange(10) == 0:
            cached.remove(w)
            scan.remove(w)
        else:
            cached.record(w)
            scan.record(w)
        for p in ("", "a", "b", "ba", "c", "d"):
            if cached.suggest(p) != scan.suggest(p):
                mismatches += 1
    print(f"  {ops} ops, {mismatches} mismatches")
    check(mismatches == 0)

    print("== router")
    r = PathTrie()
    r.put("/users", "listUsers")
    r.put("/users/:id", "getUser")
    r.put("/users/me", "currentUser")
    r.put("/users/:id/orders/*", "ordersByStatus")
    r.put("/static/**", "staticFiles")
    r.put("/a/b/x", "abx")
    r.put("/a/*/c", "anyC")
    for p in ["/users/me", "/users/42", "/users/42/orders/open", "/static/css/site.css", "/static",
              "/a/b/c", "//users//", "/nope"]:
        print(f"  {p} -> {show(r.match(p))}")
    check(r.match("/users/me").value == "currentUser")
    check(r.match("/users/42").params["id"] == "42")
    check(r.match("/a/b/c").value == "anyC")
    check(r.match("/static").value == "staticFiles")
    check(r.match("/nope") is None)
    expect_error(lambda: r.put("/users/:id", "again"), "duplicate")
    expect_error(lambda: r.put("/users/:uid/cart", "cart"), "conflict")
    expect_error(lambda: r.put("/x/**/y", "bad"), "last segment")
    expect_error(lambda: r.match("users"), "must start with /")

    print("== wildcard get over plain keys (KV follow-up)")
    kv = PathTrie()
    kv.put("/team/alpha/lead", "asha")
    kv.put("/team/alpha/dev", "ben")
    kv.put("/team/beta/lead", "chen")
    print("  /team/*/lead  ->", "[" + ", ".join(kv.find_all("/team/*/lead")) + "]")
    print("  /team/alpha/* ->", "[" + ", ".join(kv.find_all("/team/alpha/*")) + "]")
    print(f"  /*/*/*        -> {len(kv.find_all('/*/*/*'))} keys")
    check(len(kv.find_all("/team/*/lead")) == 2 and len(kv.find_all("/*/*/*")) == 3)

    print("== 4 writers x 500 records while 2 readers suggest")
    shared = AutocompleteEngine(ByFrequency(), 3, True)

    def writer():
        for _ in range(500):
            shared.record("load test")

    def reader():
        for _ in range(500):
            shared.suggest("lo")

    ts = [threading.Thread(target=writer) for _ in range(4)] + [threading.Thread(target=reader) for _ in range(2)]
    for th in ts:
        th.start()
    for th in ts:
        th.join()
    print(f"  'lo' -> {fmt(shared.suggest('lo'))}")
    check(shared.suggest("lo")[0][1] == 2000)
    print("All checks passed")


if __name__ == "__main__":
    main()

class AutocompleteTests(unittest.TestCase):
    def engine(self, cache=True, rank=None, k=3):
        return AutocompleteEngine(rank or ByFrequency(), k, cache)

    def test_top_k_by_frequency_ties_alphabetical(self):
        e = self.engine()
        e.record("bb", 2)
        e.record("ba", 2)
        e.record("bc", 5)
        e.record("bd", 1)
        self.assertEqual(e.suggest("b"), [("bc", 5), ("ba", 2), ("bb", 2)])

    def test_recency_strategy(self):
        e = self.engine(rank=ByRecency())
        for w in ("x1", "x2", "x3", "x1"):
            e.record(w)
        self.assertEqual([t for t, _ in e.suggest("x")], ["x1", "x3", "x2"])

    def test_normalisation_and_case(self):
        e = self.engine()
        e.record("  Hello   World ")
        self.assertEqual(e.suggest("HELLO w"), [("hello world", 1)])

    def test_unknown_prefix_and_empty_phrase(self):
        e = self.engine()
        self.assertEqual(e.suggest("zz"), [])
        with self.assertRaises(ValueError):
            e.record("   ")

    def test_remove_rebuilds_cache(self):
        e = self.engine(k=1)
        e.record("abc", 3)
        e.record("abd", 2)
        self.assertEqual(e.suggest("ab"), [("abc", 3)])
        self.assertTrue(e.remove("abc"))
        self.assertEqual(e.suggest("ab"), [("abd", 2)])
        self.assertFalse(e.remove("abc"))

    def test_prefix_that_is_itself_a_phrase(self):
        e = self.engine()
        e.record("ab")
        e.record("abc", 2)
        self.assertEqual(e.suggest("ab"), [("abc", 2), ("ab", 1)])

    def test_typing_session_records_on_hash(self):
        e = self.engine()
        s = e.session()
        for c in "hi#":
            s.type(c)
        self.assertEqual(e.suggest("h"), [("hi", 1)])

class RWLockTests(unittest.TestCase):
    def test_new_reader_waits_behind_a_queued_writer(self):
        import time
        rw = RWLock()
        order = []
        rw.acquire_read()                                   # reader 1 holds the lock

        def writer():
            rw.acquire_write()
            order.append("writer")
            rw.release_write()

        def reader2():
            rw.acquire_read()
            order.append("reader2")
            rw.release_read()

        w = threading.Thread(target=writer)
        w.start()
        while rw._waiting_writers == 0:                     # the writer is queued
            time.sleep(0.001)
        r = threading.Thread(target=reader2)
        r.start()
        time.sleep(0.05)
        self.assertEqual(order, [])                         # reader 2 did not overtake the writer
        rw.release_read()
        w.join(2)
        r.join(2)
        self.assertEqual(order, ["writer", "reader2"])

// Autocomplete (trie + cached top-k + ranking Strategy) and a wildcard path router, C++14.
// Locking: one mutex per AutocompleteEngine and per PathTrie, taken by every public method (std::lock_guard).
// Java uses a read-write lock so suggest()/match() run in parallel; the C++14 equivalent is
// std::shared_timed_mutex (std::shared_mutex is C++17) with std::shared_lock for readers. Pick a
// writer-preferring lock, or a steady stream of suggest() calls can starve record().
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run-cpp.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp

// ---------------- Part 1: autocomplete ----------------
struct Entry {
    std::string text;
    long count = 0;
    long lastUsed = 0;
};
using EntryPtr = std::shared_ptr<Entry>;

// Strategy: "a is better than b". Scores may only go UP when a phrase is recorded.
struct RankingStrategy {
    virtual ~RankingStrategy() = default;
    virtual std::string name() const = 0;
    virtual bool better(const Entry& a, const Entry& b) const = 0;
};
struct ByFrequency : RankingStrategy {
    std::string name() const override { return "frequency"; }
    bool better(const Entry& a, const Entry& b) const override {
        if (a.count != b.count) return a.count > b.count;
        return a.text < b.text;                       // ties: alphabetical (LeetCode 642 rule)
    }
};
struct ByRecency : RankingStrategy {
    std::string name() const override { return "recency"; }
    bool better(const Entry& a, const Entry& b) const override {
        if (a.lastUsed != b.lastUsed) return a.lastUsed > b.lastUsed;
        return a.text < b.text;
    }
};

struct TrieNode {
    std::map<char, std::unique_ptr<TrieNode>> children;
    EntryPtr entry;                // set when a phrase ends here
    std::vector<EntryPtr> topK;    // cached best k of this subtree
};

std::string normalize(const std::string& s, bool trimEnd) {
    std::string out;
    for (char c : s) {
        char l = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
        if (std::isspace(static_cast<unsigned char>(l))) {
            if (!out.empty() && out.back() != ' ') out += ' ';
        } else {
            out += l;
        }
    }
    if (!s.empty() && std::isspace(static_cast<unsigned char>(s.back())) && !out.empty() && out.back() != ' ') out += ' ';
    if (trimEnd && !out.empty() && out.back() == ' ') out.pop_back();
    return out;
}

using Suggestions = std::vector<std::pair<std::string, long>>;

std::string fmt(const Suggestions& v) {
    std::string s = "[";
    for (size_t i = 0; i < v.size(); i++) s += (i ? ", " : "") + v[i].first + "(" + std::to_string(v[i].second) + ")";
    return s + "]";
}

class AutocompleteEngine {
public:
    AutocompleteEngine(std::shared_ptr<RankingStrategy> r, size_t k, bool cache) : rank_(std::move(r)), k_(k), cache_(cache) {
        if (k < 1) throw std::invalid_argument("k must be >= 1");
    }
    std::string describe() const {
        std::lock_guard<Mutex> g(mu_);
        return rank_->name() + ", k=" + std::to_string(k_) + (cache_ ? ", cached top-k" : ", scan");
    }
    void record(const std::string& phrase, long times = 1) {
        std::lock_guard<Mutex> g(mu_);
        std::string p = normalize(phrase, true);
        if (p.empty()) throw std::invalid_argument("empty phrase");
        if (times < 1) throw std::invalid_argument("times must be >= 1");
        std::vector<TrieNode*> path{&root_};
        TrieNode* n = &root_;
        for (char c : p) {
            auto& ch = n->children[c];
            if (!ch) ch.reset(new TrieNode());
            n = ch.get();
            path.push_back(n);
        }
        if (!n->entry) { n->entry = std::make_shared<Entry>(); n->entry->text = p; }
        n->entry->count += times;
        n->entry->lastUsed = ++clock_;
        if (cache_) for (TrieNode* node : path) offer(*node, n->entry);
    }
    bool remove(const std::string& phrase) {
        std::lock_guard<Mutex> g(mu_);
        std::string p = normalize(phrase, true);
        std::vector<TrieNode*> path{&root_};
        TrieNode* n = &root_;
        for (char c : p) {
            auto it = n->children.find(c);
            if (it == n->children.end()) return false;
            n = it->second.get();
            path.push_back(n);
        }
        if (!n->entry) return false;
        n->entry.reset();
        size_t keep = path.size();
        for (size_t i = path.size() - 1; i >= 1; i--) {          // prune empty leaves
            if (!path[i]->entry && path[i]->children.empty()) { path[i - 1]->children.erase(p[i - 1]); keep = i; }
            else break;
        }
        if (cache_) for (size_t i = keep; i-- > 0;) rebuild(*path[i]);   // removal breaks "only up"
        return true;
    }
    Suggestions suggest(const std::string& prefix) const {
        std::lock_guard<Mutex> g(mu_);
        std::string p = normalize(prefix, false);
        const TrieNode* n = &root_;
        for (char c : p) {
            auto it = n->children.find(c);
            if (it == n->children.end()) return {};
            n = it->second.get();
        }
        std::vector<EntryPtr> best = cache_ ? n->topK : scan(*n);
        Suggestions out;
        for (const auto& e : best) out.emplace_back(e->text, e->count);
        return out;
    }

private:
    bool cmp(const EntryPtr& a, const EntryPtr& b) const { return rank_->better(*a, *b); }
    void offer(TrieNode& node, const EntryPtr& e) {
        auto& v = node.topK;
        v.erase(std::remove(v.begin(), v.end(), e), v.end());
        v.push_back(e);
        std::sort(v.begin(), v.end(), [this](const EntryPtr& a, const EntryPtr& b) { return cmp(a, b); });
        if (v.size() > k_) v.resize(k_);
    }
    void rebuild(TrieNode& node) {
        std::vector<EntryPtr> pool;
        if (node.entry) pool.push_back(node.entry);
        for (const auto& kv : node.children) pool.insert(pool.end(), kv.second->topK.begin(), kv.second->topK.end());
        std::sort(pool.begin(), pool.end(), [this](const EntryPtr& a, const EntryPtr& b) { return cmp(a, b); });
        if (pool.size() > k_) pool.resize(k_);
        node.topK = pool;
    }
    std::vector<EntryPtr> scan(const TrieNode& start) const {
        std::vector<EntryPtr> found;
        std::vector<const TrieNode*> stack{&start};
        while (!stack.empty()) {
            const TrieNode* n = stack.back();
            stack.pop_back();
            if (n->entry) found.push_back(n->entry);
            for (const auto& kv : n->children) stack.push_back(kv.second.get());
        }
        auto c = [this](const EntryPtr& a, const EntryPtr& b) { return cmp(a, b); };
        size_t m = std::min(k_, found.size());
        std::partial_sort(found.begin(), found.begin() + m, found.end(), c);
        found.resize(m);
        return found;
    }
    TrieNode root_;
    std::shared_ptr<RankingStrategy> rank_;
    size_t k_;
    bool cache_;
    long clock_ = 0;
    mutable Mutex mu_;                     // guards the whole trie (children, entries, cached top-k) and clock_
};

class TypingSession {   // LeetCode 642 style: '#' ends the sentence and records it
public:
    explicit TypingSession(AutocompleteEngine& e) : engine_(e) {}
    Suggestions type(char c) {
        if (c == '#') { engine_.record(typed_); typed_.clear(); return {}; }
        typed_ += c;
        return engine_.suggest(typed_);
    }
private:
    AutocompleteEngine& engine_;
    std::string typed_;
};

// ---------------- Part 2: wildcard router ----------------
std::vector<std::string> segments(const std::string& path) {
    if (path.empty() || path[0] != '/') throw std::invalid_argument("path must start with /: " + path);
    std::vector<std::string> out;
    std::stringstream ss(path);
    std::string s;
    while (std::getline(ss, s, '/')) if (!s.empty()) out.push_back(s);   // "/a//b/" == "/a/b"
    return out;
}
std::string join(const std::vector<std::string>& v, size_t from) {
    std::string s;
    for (size_t i = from; i < v.size(); i++) s += (i > from ? "/" : "") + v[i];
    return s;
}

struct RouteMatch {
    std::string value, pattern;
    std::vector<std::pair<std::string, std::string>> params;
    std::string str() const {
        std::string s = value + " via " + pattern;
        if (!params.empty()) {
            s += " {";
            for (size_t i = 0; i < params.size(); i++) s += (i ? ", " : "") + params[i].first + "=" + params[i].second;
            s += "}";
        }
        return s;
    }
};

class PathTrie {
    struct Node {
        std::map<std::string, std::unique_ptr<Node>> literal;
        std::unique_ptr<Node> param, star, globstar;
        std::string paramName, value, pattern;
        bool has = false;
    };

public:
    void put(const std::string& pattern, const std::string& value) {
        std::lock_guard<Mutex> g(mu_);
        auto segs = segments(pattern);
        std::string norm = "/" + join(segs, 0);
        Node* n = &root_;
        for (size_t i = 0; i < segs.size(); i++) {
            const std::string& s = segs[i];
            if (s == "**") {
                if (i != segs.size() - 1) throw std::invalid_argument("** must be the last segment: " + pattern);
                if (!n->globstar) n->globstar.reset(new Node());
                n = n->globstar.get();
            } else if (s == "*") {
                if (!n->star) n->star.reset(new Node());
                n = n->star.get();
            } else if (s[0] == ':') {
                std::string name = s.substr(1);
                if (name.empty()) throw std::invalid_argument("empty parameter name: " + pattern);
                if (!n->param) { n->param.reset(new Node()); n->paramName = name; }
                else if (n->paramName != name)
                    throw std::invalid_argument("conflict: :" + name + " vs :" + n->paramName + " in " + pattern);
                n = n->param.get();
            } else {
                auto& ch = n->literal[s];
                if (!ch) ch.reset(new Node());
                n = ch.get();
            }
        }
        if (n->has) throw std::invalid_argument("duplicate route " + norm + " (already " + n->pattern + ")");
        n->has = true; n->value = value; n->pattern = norm;
    }
    bool match(const std::string& path, RouteMatch& out) const {
        std::lock_guard<Mutex> g(mu_);
        auto segs = segments(path);
        out.params.clear();
        const Node* hit = match(&root_, segs, 0, out.params);
        if (!hit) return false;
        out.value = hit->value; out.pattern = hit->pattern;
        return true;
    }
    std::vector<std::string> findAll(const std::string& query) const {
        std::lock_guard<Mutex> g(mu_);
        std::vector<std::string> out;
        collect(&root_, segments(query), 0, out);
        return out;
    }

private:
    using Params = std::vector<std::pair<std::string, std::string>>;
    const Node* match(const Node* n, const std::vector<std::string>& segs, size_t i, Params& ps) const {
        if (i == segs.size()) {
            if (n->has) return n;
            if (n->globstar && n->globstar->has) { ps.emplace_back("**", ""); return n->globstar.get(); }
            return nullptr;
        }
        const std::string& s = segs[i];
        auto it = n->literal.find(s);
        if (it != n->literal.end()) if (const Node* r = match(it->second.get(), segs, i + 1, ps)) return r;
        if (n->param) {
            ps.emplace_back(n->paramName, s);
            if (const Node* r = match(n->param.get(), segs, i + 1, ps)) return r;
            ps.pop_back();
        }
        if (n->star) if (const Node* r = match(n->star.get(), segs, i + 1, ps)) return r;
        if (n->globstar && n->globstar->has) { ps.emplace_back("**", join(segs, i)); return n->globstar.get(); }
        return nullptr;
    }
    void collect(const Node* n, const std::vector<std::string>& segs, size_t i, std::vector<std::string>& out) const {
        if (i == segs.size()) { if (n->has) out.push_back(n->pattern + "=" + n->value); return; }
        if (segs[i] == "*") {
            for (const auto& kv : n->literal) collect(kv.second.get(), segs, i + 1, out);
        } else {
            auto it = n->literal.find(segs[i]);
            if (it != n->literal.end()) collect(it->second.get(), segs, i + 1, out);
        }
    }
    Node root_;
    mutable Mutex mu_;                     // guards every node
};

template <class F>
void expectError(F f, const std::string& contains) {
    try { f(); } catch (const std::invalid_argument& e) {
        std::cout << "  rejected: " << e.what() << "\n";
        CHECK(std::string(e.what()).find(contains) != std::string::npos);
        return;
    }
    throw std::logic_error("expected error: " + contains);
}

void seed(AutocompleteEngine& e) {
    e.record("linux kernel", 5);
    e.record("linked list", 3);
    e.record("lint rules", 3);
    e.record("linus torvalds", 2);
    e.record("LRU   Cache", 4);
    e.record("linux kernel modules", 1);
}

std::string list(const std::vector<std::string>& v) {
    std::string s = "[";
    for (size_t i = 0; i < v.size(); i++) s += (i ? ", " : "") + v[i];
    return s + "]";
}

int main() {
    std::cout << "== autocomplete\n";
    AutocompleteEngine freq(std::make_shared<ByFrequency>(), 3, true);
    AutocompleteEngine recent(std::make_shared<ByRecency>(), 3, true);
    seed(freq);
    seed(recent);
    std::cout << "  [" << freq.describe() << "] 'lin' -> " << fmt(freq.suggest("lin")) << "\n";
    std::cout << "  [" << recent.describe() << "] 'lin' -> " << fmt(recent.suggest("lin")) << "\n";
    CHECK(fmt(freq.suggest("lin")) == "[linux kernel(5), linked list(3), lint rules(3)]");
    CHECK(recent.suggest("lin")[0].first == "linux kernel modules");
    std::cout << "  'LI' -> " << fmt(freq.suggest("LI")) << "   'xyz' -> " << fmt(freq.suggest("xyz")) << "\n";
    CHECK(freq.suggest("xyz").empty());

    std::cout << "== typing session (LeetCode 642 style)\n";
    TypingSession t(freq);
    for (char c : std::string("lru cache#")) {
        Suggestions s = t.type(c);
        if (c == 'l' || c == 'r' || c == '#') std::cout << "  typed '" << c << "' -> " << fmt(s) << "\n";
    }
    std::cout << "  'l' now -> " << fmt(freq.suggest("l")) << "\n";
    CHECK(freq.suggest("l")[1] == std::make_pair(std::string("lru cache"), 5L));

    std::cout << "== remove 'linux kernel'\n";
    CHECK(freq.remove("linux kernel"));
    CHECK(!freq.remove("linux kernel"));
    std::cout << "  'lin' -> " << fmt(freq.suggest("lin")) << "   'linux' -> " << fmt(freq.suggest("linux")) << "\n";
    CHECK(fmt(freq.suggest("linux")) == "[linux kernel modules(1)]");

    std::cout << "== cached top-k equals full scan (random workload)\n";
    std::mt19937 rnd(7);
    std::vector<std::string> vocab{"a", "ab", "abc", "abd", "b", "ba", "bab", "bad", "c", "ca", "cab", "cad", "dab", "dad"};
    AutocompleteEngine cached(std::make_shared<ByFrequency>(), 3, true), scan(std::make_shared<ByFrequency>(), 3, false);
    int mismatches = 0, ops = 2000;
    for (int i = 0; i < ops; i++) {
        const std::string& w = vocab[rnd() % vocab.size()];
        if (rnd() % 10 == 0) { cached.remove(w); scan.remove(w); }
        else { cached.record(w); scan.record(w); }
        for (const char* p : {"", "a", "b", "ba", "c", "d"})
            if (cached.suggest(p) != scan.suggest(p)) mismatches++;
    }
    std::cout << "  " << ops << " ops, " << mismatches << " mismatches\n";
    CHECK(mismatches == 0);

    std::cout << "== router\n";
    PathTrie r;
    r.put("/users", "listUsers");
    r.put("/users/:id", "getUser");
    r.put("/users/me", "currentUser");
    r.put("/users/:id/orders/*", "ordersByStatus");
    r.put("/static/**", "staticFiles");
    r.put("/a/b/x", "abx");
    r.put("/a/*/c", "anyC");
    RouteMatch m;
    for (const char* p : {"/users/me", "/users/42", "/users/42/orders/open", "/static/css/site.css", "/static",
                          "/a/b/c", "//users//", "/nope"})
        std::cout << "  " << p << " -> " << (r.match(p, m) ? m.str() : "404") << "\n";
    CHECK(r.match("/users/me", m) && m.value == "currentUser");
    CHECK(r.match("/users/42", m) && m.params[0].second == "42");
    CHECK(r.match("/a/b/c", m) && m.value == "anyC");
    CHECK(r.match("/static", m) && m.value == "staticFiles");
    CHECK(!r.match("/nope", m));
    expectError([&] { r.put("/users/:id", "again"); }, "duplicate");
    expectError([&] { r.put("/users/:uid/cart", "cart"); }, "conflict");
    expectError([&] { r.put("/x/**/y", "bad"); }, "last segment");
    expectError([&] { r.match("users", m); }, "must start with /");

    std::cout << "== wildcard get over plain keys (KV follow-up)\n";
    PathTrie kv;
    kv.put("/team/alpha/lead", "asha");
    kv.put("/team/alpha/dev", "ben");
    kv.put("/team/beta/lead", "chen");
    std::cout << "  /team/*/lead  -> " << list(kv.findAll("/team/*/lead")) << "\n";
    std::cout << "  /team/alpha/* -> " << list(kv.findAll("/team/alpha/*")) << "\n";
    std::cout << "  /*/*/*        -> " << kv.findAll("/*/*/*").size() << " keys\n";
    CHECK(kv.findAll("/team/*/lead").size() == 2 && kv.findAll("/*/*/*").size() == 3);

    std::cout << "== 2000 records of one phrase (single thread here; Java runs 4 writers + 2 readers)\n";
    AutocompleteEngine shared(std::make_shared<ByFrequency>(), 3, true);
    for (int i = 0; i < 2000; i++) shared.record("load test");
    std::cout << "  'lo' -> " << fmt(shared.suggest("lo")) << "\n";
    CHECK(shared.suggest("lo")[0].second == 2000);
    std::cout << "All checks passed\n";
    return 0;
}
