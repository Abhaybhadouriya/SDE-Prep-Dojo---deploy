"""ATM - Python version of the Java design: State machine, Chain of Responsibility dispenser, bank Adapter."""
from __future__ import annotations

import hashlib
import itertools
import threading
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional


class InvalidOperation(Exception): pass
class TransactionDeclined(Exception): pass


class DebitResult(Enum):
    APPROVED = "APPROVED"
    INSUFFICIENT_FUNDS = "INSUFFICIENT_FUNDS"
    DAILY_LIMIT_EXCEEDED = "DAILY_LIMIT_EXCEEDED"
    CARD_BLOCKED = "CARD_BLOCKED"
    UNKNOWN_CARD = "UNKNOWN_CARD"
    ALREADY_REVERSED = "ALREADY_REVERSED"
    BANK_ERROR = "BANK_ERROR"


# ---------------------------------------------------------------- bank side
class LegacyCoreBanking:
    """Pretend vendor library: paise, two-character response codes, hashed PIN blocks. We adapt to it."""

    def __init__(self):
        self._acc: dict = {}
        # ref -> [pan, paise, code, reversed]: kept for approvals AND declines and never removed;
        # a reversal marks it reversed (a tombstone) instead of deleting it.
        self._outcomes: dict = {}
        self._lock = threading.Lock()

    @staticmethod
    def pin_block(pan: str, pin: str) -> str:
        return hashlib.sha256(f"{pan}:{pin}".encode()).hexdigest()[:16]

    def open_account(self, pan, pin, rupees, daily_limit_rupees):
        with self._lock:
            self._acc[pan] = {"pin": self.pin_block(pan, pin), "paise": rupees * 100,
                              "limit": daily_limit_rupees * 100, "today": 0, "blocked": False}

    def authorize(self, pan, pin_block) -> str:
        with self._lock:
            a = self._acc.get(pan)
            if a is None:
                return "14"
            if a["blocked"]:
                return "62"
            return "00" if a["pin"] == pin_block else "55"

    def ledger_paise(self, pan) -> int:
        with self._lock:
            return self._acc[pan]["paise"]

    def post_debit(self, pan, paise, ref) -> str:
        """Same ref twice = the first answer again, and at most one debit. "94" = ref already reversed."""
        with self._lock:
            known = self._outcomes.get(ref)
            if known is not None:        # retransmit: replay the stored answer, never re-post
                return "94" if known[3] else known[2]
            code = self._decide(pan, paise)
            if code == "00":
                a = self._acc[pan]
                a["paise"] -= paise
                a["today"] += paise
            self._outcomes[ref] = [pan, paise, code, False]
            return code

    def _decide(self, pan, paise) -> str:
        a = self._acc.get(pan)
        if a is None:
            return "14"
        if a["blocked"]:
            return "62"
        if a["today"] + paise > a["limit"]:
            return "61"
        if a["paise"] < paise:
            return "51"
        return "00"

    def reversal(self, ref) -> str:
        """Idempotent. If the debit was never seen, store a tombstone so a late debit is declined."""
        with self._lock:
            o = self._outcomes.get(ref)
            if o is None:                # the reversal overtook its debit
                self._outcomes[ref] = [None, 0, "25", True]
                return "25"
            pan, paise, code, reversed_ = o
            if not reversed_ and code == "00":
                self._acc[pan]["paise"] += paise
                self._acc[pan]["today"] -= paise
            o[3] = True
            return "00"

    def block(self, pan):
        with self._lock:
            self._acc[pan]["blocked"] = True


class CoreBankingAdapter:
    """Adapter: the BankService the ATM expects (rupees, enums) on top of the legacy core."""
    _CODES = {"00": DebitResult.APPROVED, "51": DebitResult.INSUFFICIENT_FUNDS, "61": DebitResult.DAILY_LIMIT_EXCEEDED,
              "62": DebitResult.CARD_BLOCKED, "14": DebitResult.UNKNOWN_CARD, "94": DebitResult.ALREADY_REVERSED}

    def __init__(self, core: LegacyCoreBanking):
        self.core = core

    def verify_pin(self, card, pin) -> bool:
        return self.core.authorize(card, LegacyCoreBanking.pin_block(card, pin)) == "00"

    def balance(self, card) -> int:
        return self.core.ledger_paise(card) // 100

    def debit(self, card, rupees, txn_id) -> DebitResult:
        # an unknown code is a bank error, never a guess like "blocked"
        return self._CODES.get(self.core.post_debit(card, rupees * 100, txn_id), DebitResult.BANK_ERROR)

    def reverse(self, txn_id):
        self.core.reversal(txn_id)

    def block_card(self, card):
        self.core.block(card)


# ---------------------------------------------------------------- cash side (Chain of Responsibility)
class NoteHandler:
    def __init__(self, denomination: int, count: int, nxt: Optional["NoteHandler"]):
        self.denomination, self.count, self.next = denomination, count, nxt

    def plan(self, amount: int, notes_left: int) -> Optional[dict]:
        """Greedy first, then one note fewer at a time (bounded backtracking). Does not change stock."""
        for take in range(min(self.count, amount // self.denomination, notes_left), -1, -1):
            rest = amount - take * self.denomination
            if rest == 0:
                tail = {}
            elif self.next is None:
                continue
            else:
                tail = self.next.plan(rest, notes_left - take)
            if tail is None:
                continue
            return ({self.denomination: take} if take else {}) | tail
        return None

    def apply(self, notes: dict, sign: int):
        self.count += sign * notes.get(self.denomination, 0)
        if self.next:
            self.next.apply(notes, sign)

    def walk(self):
        h = self
        while h:
            yield h
            h = h.next


class CashDispenser:
    MAX_NOTES = 40

    def __init__(self, stock: dict):
        head = None
        for d in sorted(stock):                    # build smallest -> largest so head is the largest
            head = NoteHandler(d, stock[d], head)
        self.head = head
        self._lock = threading.Lock()
        self._jam = False

    def reserve(self, amount: int) -> Optional[dict]:
        with self._lock:                           # plan + set aside in one step
            plan = self.head.plan(amount, self.MAX_NOTES)
            if plan is not None:
                self.head.apply(plan, -1)
            return plan

    def release(self, notes: dict):
        with self._lock:
            self.head.apply(notes, +1)

    def load(self, denom: int, n: int):
        with self._lock:
            for h in self.head.walk():
                if h.denomination == denom:
                    h.count += n
                    return
            raise ValueError(f"no cassette for {denom}")

    def stock(self) -> dict:
        with self._lock:
            return {h.denomination: h.count for h in self.head.walk()}

    def total(self) -> int:
        return sum(d * c for d, c in self.stock().items())

    def jam_next(self):
        self._jam = True

    def eject(self, notes: dict) -> bool:
        if self._jam:
            self._jam = False
            return False
        return True


# ---------------------------------------------------------------- states
class StateName(Enum):
    IDLE = "IDLE"
    CARD_INSERTED = "CARD_INSERTED"
    AUTHENTICATED = "AUTHENTICATED"
    DISPENSING = "DISPENSING"
    OUT_OF_SERVICE = "OUT_OF_SERVICE"


@dataclass(frozen=True)
class Receipt:
    txn_id: str
    amount: int
    balance_after: int
    notes: dict = field(default_factory=dict)
    status: str = "OK"


class AtmState:
    name = None

    def _no(self, what):
        raise InvalidOperation(f"cannot {what} while {self.name.value}")

    def insert_card(self, atm, card): self._no("insert a card")
    def enter_pin(self, atm, pin): self._no("enter a PIN")
    def balance(self, atm): self._no("check balance")
    def withdraw(self, atm, amount): self._no("withdraw")
    def eject_card(self, atm): self._no("eject a card")


class Idle(AtmState):
    name = StateName.IDLE

    def insert_card(self, atm, card):
        if card in atm.retained:
            raise InvalidOperation(f"card {card} is blocked")
        atm.card = card
        atm.move_to(CardInserted())


class CardInserted(AtmState):
    name = StateName.CARD_INSERTED
    MAX_TRIES = 3

    def __init__(self):
        self.tries = 0

    def enter_pin(self, atm, pin):
        if atm.bank.verify_pin(atm.card, pin):
            atm.move_to(Authenticated())
            return
        self.tries += 1
        if self.tries >= self.MAX_TRIES:
            atm.retain_card()
            atm.move_to(Idle())
            raise InvalidOperation(f"wrong PIN {self.tries} times: card retained")
        raise InvalidOperation(f"wrong PIN, tries left: {self.MAX_TRIES - self.tries}")

    def eject_card(self, atm):
        atm.end_session()


class Authenticated(AtmState):
    name = StateName.AUTHENTICATED
    MAX_PER_TXN = 20_000

    def balance(self, atm):
        return atm.bank.balance(atm.card)

    def withdraw(self, atm, amount):
        if amount <= 0 or amount % 100:
            raise InvalidOperation("amount must be a positive multiple of 100")
        if amount > self.MAX_PER_TXN:
            raise InvalidOperation(f"max {self.MAX_PER_TXN} per withdrawal")
        d = Dispensing()
        atm.move_to(d)
        try:
            return d.run(atm, amount)
        finally:
            atm.move_to(self)

    def eject_card(self, atm):
        atm.end_session()


class Dispensing(AtmState):
    name = StateName.DISPENSING

    def run(self, atm, amount):
        txn = atm.next_txn_id()
        notes = atm.dispenser.reserve(amount)
        if notes is None:
            raise TransactionDeclined(f"ATM cannot dispense {amount} with the notes it has")
        try:
            result = atm.bank.debit(atm.card, amount, txn)
        except Exception:                # timeout: did the debit land? unknown
            atm.dispenser.release(notes)
            try:
                atm.bank.reverse(txn)    # safe either way: a tombstone if the bank never saw the debit
            except Exception:
                pass                     # a real ATM queues the reversal and retries it (store-and-forward)
            raise TransactionDeclined(f"bank unavailable for {txn}: nothing dispensed, reversal sent")
        if result is not DebitResult.APPROVED:
            atm.dispenser.release(notes)
            raise TransactionDeclined(f"bank declined {txn}: {result.value}")
        if not atm.dispenser.eject(notes):
            atm.bank.reverse(txn)
            atm.dispenser.release(notes)
            return Receipt(txn, amount, atm.bank.balance(atm.card), {}, "REVERSED (dispense fault)")
        return Receipt(txn, amount, atm.bank.balance(atm.card), notes)


class OutOfService(AtmState):
    name = StateName.OUT_OF_SERVICE


# ---------------------------------------------------------------- context
class Atm:
    _txn = itertools.count(1)
    _txn_lock = threading.Lock()

    def __init__(self, atm_id: str, bank: CoreBankingAdapter, dispenser: CashDispenser, min_cash: int = 100):
        self.id, self.bank, self.dispenser, self.min_cash = atm_id, bank, dispenser, min_cash
        self.card: Optional[str] = None
        self.retained: set = set()
        self.history: list = []
        self._state: AtmState = Idle() if dispenser.total() >= min_cash else OutOfService()
        self._lock = threading.RLock()            # one customer at a time

    # user actions
    def insert_card(self, card):
        with self._lock: self._state.insert_card(self, card)
    def enter_pin(self, pin):
        with self._lock: self._state.enter_pin(self, pin)
    def balance(self):
        with self._lock: return self._state.balance(self)
    def withdraw(self, amount):
        with self._lock: return self._state.withdraw(self, amount)
    def eject_card(self):
        with self._lock: self._state.eject_card(self)

    # operator
    def load_cash(self, denom, n):
        with self._lock:
            self.dispenser.load(denom, n)
            if self._state.name is StateName.OUT_OF_SERVICE and self.dispenser.total() >= self.min_cash:
                self.move_to(Idle())

    # used by states
    @property
    def state(self) -> StateName:
        return self._state.name

    def next_txn_id(self):
        with Atm._txn_lock:
            return f"{self.id}-T{next(Atm._txn)}"

    def end_session(self):
        self.card = None
        self.move_to(OutOfService() if self.dispenser.total() < self.min_cash else Idle())

    def retain_card(self):
        self.retained.add(self.card)
        self.bank.block_card(self.card)
        self.card = None

    def move_to(self, nxt: AtmState):
        if nxt.name is not self._state.name:
            self.history.append(f"{self._state.name.value} -> {nxt.name.value}")
        self._state = nxt
"""Demo driver: the same story as Java Main, with simple asserts."""
from atm import (Atm, CashDispenser, CoreBankingAdapter, DebitResult, InvalidOperation, LegacyCoreBanking, StateName,
                 TransactionDeclined)


def expect(err, fn):
    try:
        fn()
    except err as e:
        print("  rejected as expected:", e)
        return
    raise AssertionError(f"expected {err.__name__}")


def main():
    core = LegacyCoreBanking()
    core.open_account("4111-0001", "1234", 10_000, 20_000)
    core.open_account("4111-0002", "4321", 500, 20_000)
    bank = CoreBankingAdapter(core)
    atm = Atm("ATM1", bank, CashDispenser({2000: 1, 500: 5, 200: 4, 100: 4}))

    print("== 1. wrong-state actions ==")
    expect(InvalidOperation, lambda: atm.withdraw(500))

    print("== 2. card, PIN, withdraw 1700 ==")
    atm.insert_card("4111-0001")
    expect(InvalidOperation, lambda: atm.enter_pin("9999"))
    atm.enter_pin("1234")
    r = atm.withdraw(1700)
    print(" ", r)
    assert r.notes == {500: 3, 200: 1} and r.balance_after == 8_300
    expect(InvalidOperation, lambda: atm.withdraw(250))
    expect(TransactionDeclined, lambda: atm.withdraw(4500))
    assert bank.balance("4111-0001") == 8_300
    atm.eject_card()

    print("== 3. greedy is not enough: 600 from 2000x1, 500x1, 200x4, no 100s ==")
    atm2 = Atm("ATM2", bank, CashDispenser({2000: 1, 500: 1, 200: 4, 100: 0}))
    atm2.insert_card("4111-0001")
    atm2.enter_pin("1234")
    r2 = atm2.withdraw(600)
    print(" ", r2)
    assert r2.notes == {200: 3}
    atm2.eject_card()

    print("== 4. bank declines, notes go back ==")
    before = atm.dispenser.stock()
    atm.insert_card("4111-0002")
    atm.enter_pin("4321")
    expect(TransactionDeclined, lambda: atm.withdraw(1000))
    assert atm.dispenser.stock() == before
    atm.eject_card()

    print("== 5. jam after debit: reversed ==")
    atm.insert_card("4111-0001")
    atm.enter_pin("1234")
    atm.dispenser.jam_next()
    r3 = atm.withdraw(1000)
    print(" ", r3)
    assert r3.status.startswith("REVERSED") and r3.balance_after == 7_700
    atm.eject_card()

    print("== 5b. retries and the network: one durable answer per txn id ==")
    assert bank.debit("4111-0001", 1000, "RETRY-1") is DebitResult.APPROVED
    assert bank.debit("4111-0001", 1000, "RETRY-1") is DebitResult.APPROVED      # retransmit: replayed
    bank.reverse("RETRY-1")
    assert bank.debit("4111-0001", 1000, "RETRY-1") is DebitResult.ALREADY_REVERSED
    bank.reverse("LATE-1")                                                       # reversal overtakes its debit
    assert bank.debit("4111-0001", 1000, "LATE-1") is DebitResult.ALREADY_REVERSED
    assert bank.balance("4111-0001") == 7_700
    print("  reverse-then-retry and reversal-before-debit both declined; balance", bank.balance("4111-0001"))

    print("== 6. three wrong PINs ==")
    atm.insert_card("4111-0002")
    for p in ("1111", "2222", "3333"):
        expect(InvalidOperation, lambda: atm.enter_pin(p))
    assert atm.state is StateName.IDLE
    expect(InvalidOperation, lambda: atm.insert_card("4111-0002"))
    print("  history:", atm.history[:4], "...")
    print("ALL DEMO ASSERTS PASSED")


if __name__ == "__main__":
    main()
import threading
import unittest
from concurrent.futures import ThreadPoolExecutor

from atm import (Atm, CashDispenser, CoreBankingAdapter, DebitResult, InvalidOperation, LegacyCoreBanking, StateName,
                 TransactionDeclined)


def bank_with(*accounts):
    core = LegacyCoreBanking()
    for pan, pin, bal, limit in accounts:
        core.open_account(pan, pin, bal, limit)
    return CoreBankingAdapter(core)


class DispenserTest(unittest.TestCase):
    def test_greedy_when_possible(self):
        d = CashDispenser({2000: 1, 500: 5, 200: 4, 100: 4})
        self.assertEqual(d.reserve(1700), {500: 3, 200: 1})

    def test_backtracks_when_greedy_gets_stuck(self):
        d = CashDispenser({2000: 1, 500: 1, 200: 4, 100: 0})
        self.assertEqual(d.reserve(600), {200: 3})

    def test_impossible_amount_leaves_stock_alone(self):
        d = CashDispenser({500: 1, 200: 1})
        self.assertIsNone(d.reserve(600))
        self.assertEqual(d.stock(), {500: 1, 200: 1})

    def test_reserve_then_release(self):
        d = CashDispenser({500: 2})
        notes = d.reserve(1000)
        self.assertEqual(d.total(), 0)
        d.release(notes)
        self.assertEqual(d.total(), 1000)


class AtmTest(unittest.TestCase):
    def setUp(self):
        self.bank = bank_with(("c1", "1111", 5_000, 3_000), ("c2", "2222", 300, 10_000))
        self.atm = Atm("A", self.bank, CashDispenser({500: 10, 100: 10}))

    def login(self, card="c1", pin="1111"):
        self.atm.insert_card(card)
        self.atm.enter_pin(pin)

    def test_state_guards(self):
        with self.assertRaises(InvalidOperation):
            self.atm.balance()
        self.atm.insert_card("c1")
        with self.assertRaises(InvalidOperation):
            self.atm.withdraw(100)
        with self.assertRaises(InvalidOperation):
            self.atm.insert_card("c2")

    def test_daily_limit_from_core(self):
        self.login()
        self.atm.withdraw(2_000)
        with self.assertRaises(TransactionDeclined) as e:
            self.atm.withdraw(1_500)
        self.assertIn("DAILY_LIMIT_EXCEEDED", str(e.exception))

    def test_insufficient_funds_returns_notes(self):
        self.login("c2", "2222")
        before = self.atm.dispenser.stock()
        with self.assertRaises(TransactionDeclined):
            self.atm.withdraw(500)
        self.assertEqual(self.atm.dispenser.stock(), before)

    def test_jam_reverses_debit(self):
        self.login()
        self.atm.dispenser.jam_next()
        r = self.atm.withdraw(1_000)
        self.assertTrue(r.status.startswith("REVERSED"))
        self.assertEqual(self.bank.balance("c1"), 5_000)

    def test_three_wrong_pins_retain_card(self):
        self.atm.insert_card("c1")
        for _ in range(3):
            with self.assertRaises(InvalidOperation):
                self.atm.enter_pin("0000")
        self.assertIs(self.atm.state, StateName.IDLE)
        self.assertFalse(self.bank.verify_pin("c1", "1111"))

    def test_out_of_service_and_refill(self):
        atm = Atm("S", self.bank, CashDispenser({500: 2}), min_cash=500)
        atm.insert_card("c1"); atm.enter_pin("1111"); atm.withdraw(1_000); atm.eject_card()
        self.assertIs(atm.state, StateName.OUT_OF_SERVICE)
        atm.load_cash(500, 4)
        self.assertIs(atm.state, StateName.IDLE)

    def test_idempotent_debit(self):
        self.assertIs(self.bank.debit("c1", 100, "X"), DebitResult.APPROVED)
        self.assertIs(self.bank.debit("c1", 100, "X"), DebitResult.APPROVED)
        self.assertEqual(self.bank.balance("c1"), 4_900)

    def test_reverse_then_retry_debit_takes_nothing(self):
        before = self.bank.balance("c1")
        self.assertEqual(self.bank.debit("c1", 1000, "R-1"), DebitResult.APPROVED)
        self.bank.reverse("R-1")
        self.assertEqual(self.bank.debit("c1", 1000, "R-1"), DebitResult.ALREADY_REVERSED)
        self.assertEqual(self.bank.balance("c1"), before)

    def test_reversal_before_debit_declines_the_late_debit(self):
        before = self.bank.balance("c1")
        self.bank.reverse("LATE-1")
        self.assertEqual(self.bank.debit("c1", 1000, "LATE-1"), DebitResult.ALREADY_REVERSED)
        self.assertEqual(self.bank.balance("c1"), before)

    def test_declines_are_remembered_and_unknown_card_is_not_blocked(self):
        self.assertEqual(self.bank.debit("c2", 1000, "NF-1"), DebitResult.INSUFFICIENT_FUNDS)
        self.assertEqual(self.bank.debit("c2", 1000, "NF-1"), DebitResult.INSUFFICIENT_FUNDS)
        self.assertEqual(self.bank.debit("nope", 100, "NC-1"), DebitResult.UNKNOWN_CARD)

    def test_bank_timeout_after_posting_releases_notes_and_reverses(self):
        class LostAnswerBank(CoreBankingAdapter):
            def debit(self, card, rupees, txn_id):
                super().debit(card, rupees, txn_id)
                raise TimeoutError("timeout waiting for the bank")
        before = self.bank.balance("c1")
        atm = Atm("F", LostAnswerBank(self.bank.core), CashDispenser({500: 4}), 100)
        full = atm.dispenser.stock()
        atm.insert_card("c1")
        atm.enter_pin("1111")
        with self.assertRaises(TransactionDeclined):
            atm.withdraw(1000)
        self.assertEqual(atm.dispenser.stock(), full)
        self.assertEqual(self.bank.balance("c1"), before)

    def test_joint_account_never_overdrawn(self):
        bank = bank_with(("j", "9", 10_000, 100_000))
        go = threading.Event()

        def one(i):
            a = Atm(f"J{i}", bank, CashDispenser({500: 4}))
            go.wait()
            a.insert_card("j"); a.enter_pin("9")
            try:
                a.withdraw(1_000)
                return True
            except TransactionDeclined:
                return False
            finally:
                a.eject_card()

        with ThreadPoolExecutor(8) as pool:
            futs = [pool.submit(one, i) for i in range(20)]
            go.set()
            wins = sum(f.result() for f in futs)
        self.assertEqual(wins, 10)
        self.assertEqual(bank.balance("j"), 0)

    unittest.main()

// Linux (real locks):       g++ -std=c++14 -Wall -O2 -pthread -DHAVE_STD_MUTEX main.cpp -o atm

// ------------------------------------------------------------ bank: legacy core + adapter
class LegacyCore {                       // paise, response codes; we cannot change this class
    struct Acc { size_t pinHash; long paise; long limit; long today; bool blocked; };
    // The durable answer per ref, kept for approvals AND declines and never erased: a reversal
    // marks it reversed (a tombstone), so a retransmitted or late debit cannot post again.
    struct Outcome { string pan; long paise; string code; bool reversed; };
    Mutex mu_;                                           // one lock for the whole core (fine for a demo)
    map<string, Acc> acc_;
    map<string, Outcome> outcomes_;
    string decide(const string& pan, long paise) {
        auto it = acc_.find(pan);
        if (it == acc_.end()) return "14";
        const Acc& a = it->second;
        if (a.blocked) return "62";
        if (a.today + paise > a.limit) return "61";
        if (a.paise < paise) return "51";
        return "00";
    }
public:
    static size_t pinBlock(const string& pan, const string& pin) { return hash<string>()(pan + ":" + pin); }
    void open(const string& pan, const string& pin, long rupees, long limit) {
        LockGuard g(mu_); acc_[pan] = Acc{pinBlock(pan, pin), rupees * 100, limit * 100, 0, false};
    }
    string authorize(const string& pan, size_t block) {
        LockGuard g(mu_);
        auto it = acc_.find(pan);
        if (it == acc_.end()) return "14";
        if (it->second.blocked) return "62";
        return it->second.pinHash == block ? "00" : "55";
    }
    long ledger(const string& pan) { LockGuard g(mu_); return acc_.at(pan).paise; }
    // Same ref twice = the first answer again, at most one debit. "94" = this ref was reversed.
    string postDebit(const string& pan, long paise, const string& ref) {
        LockGuard g(mu_);
        auto known = outcomes_.find(ref);
        if (known != outcomes_.end()) return known->second.reversed ? "94" : known->second.code;
        string code = decide(pan, paise);
        if (code == "00") { Acc& a = acc_.at(pan); a.paise -= paise; a.today += paise; }
        outcomes_[ref] = Outcome{pan, paise, code, false};
        return code;
    }
    // Idempotent. A reversal that overtakes its debit leaves a tombstone ("25") so the late debit is declined.
    string reversal(const string& ref) {
        LockGuard g(mu_);
        auto it = outcomes_.find(ref);
        if (it == outcomes_.end()) { outcomes_[ref] = Outcome{"", 0, "25", true}; return "25"; }
        Outcome& o = it->second;
        if (!o.reversed && o.code == "00") { Acc& a = acc_.at(o.pan); a.paise += o.paise; a.today -= o.paise; }
        o.reversed = true;
        return "00";
    }
    void block(const string& pan) { LockGuard g(mu_); acc_.at(pan).blocked = true; }
};

struct BankService {                     // what the ATM was written against
    virtual ~BankService() {}
    virtual bool verifyPin(const string& card, const string& pin) = 0;
    virtual long balance(const string& card) = 0;
    virtual DebitResult debit(const string& card, long rupees, const string& txn) = 0;
    virtual void reverse(const string& txn) = 0;
    virtual void blockCard(const string& card) = 0;
};

class CoreAdapter : public BankService {
    LegacyCore& core_;
public:
    explicit CoreAdapter(LegacyCore& c) : core_(c) {}
    bool verifyPin(const string& c, const string& p) override { return core_.authorize(c, LegacyCore::pinBlock(c, p)) == "00"; }
    long balance(const string& c) override { return core_.ledger(c) / 100; }
    DebitResult debit(const string& c, long r, const string& t) override {
        string code = core_.postDebit(c, r * 100, t);
        if (code == "00") return DebitResult::APPROVED;
        if (code == "51") return DebitResult::INSUFFICIENT_FUNDS;
        if (code == "61") return DebitResult::DAILY_LIMIT;
        if (code == "62") return DebitResult::BLOCKED;
        if (code == "14") return DebitResult::UNKNOWN_CARD;
        if (code == "94") return DebitResult::ALREADY_REVERSED;
        return DebitResult::BANK_ERROR;                  // never guess "blocked" for an unknown code
    }
    void reverse(const string& t) override { core_.reversal(t); }
    void blockCard(const string& c) override { core_.block(c); }
};

// ------------------------------------------------------------ cash: chain of note handlers
class NoteHandler {
    int denom_, count_;
    unique_ptr<NoteHandler> next_;
public:
    NoteHandler(int d, int c, unique_ptr<NoteHandler> n) : denom_(d), count_(c), next_(move(n)) {}
    // greedy first, then one note fewer (bounded backtracking); returns false if impossible
    bool plan(int amount, int notesLeft, Notes& out) const {
        int mx = min(min(count_, amount / denom_), notesLeft);
        for (int take = mx; take >= 0; take--) {
            int rest = amount - take * denom_;
            Notes tail;
            if (rest != 0 && (!next_ || !next_->plan(rest, notesLeft - take, tail))) continue;
            if (take) tail[denom_] = take;
            out = tail;
            return true;
        }
        return false;
    }
    void apply(const Notes& n, int sign) {
        auto it = n.find(denom_);
        if (it != n.end()) count_ += sign * it->second;
        if (next_) next_->apply(n, sign);
    }
    long total() const { return (long)denom_ * count_ + (next_ ? next_->total() : 0); }
};

class CashDispenser {
    unique_ptr<NoteHandler> head_;
    bool jam_ = false;
    mutable Mutex mu_;                                   // the cassettes are shared hardware
public:
    explicit CashDispenser(const Notes& stock) {
        for (auto it = stock.rbegin(); it != stock.rend(); ++it)      // smallest first, so head = largest
            head_.reset(new NoteHandler(it->first, it->second, move(head_)));
    }
    bool reserve(int amount, Notes& out) {
        LockGuard g(mu_);
        if (!head_->plan(amount, 40, out)) return false;
        head_->apply(out, -1);
        return true;
    }
    void release(const Notes& n) { LockGuard g(mu_); head_->apply(n, +1); }
    long total() const { LockGuard g(mu_); return head_->total(); }
    void jamNext() { LockGuard g(mu_); jam_ = true; }
    bool eject(const Notes&) { LockGuard g(mu_); if (jam_) { jam_ = false; return false; } return true; }
};

class Atm;
struct AtmState {
    virtual ~AtmState() {}
    virtual const char* name() const = 0;
    virtual void insertCard(Atm&, const string&) { no("insert a card"); }
    virtual void enterPin(Atm&, const string&) { no("enter a PIN"); }
    virtual long balance(Atm&) { no("check balance"); return 0; }
    virtual string withdraw(Atm&, long) { no("withdraw"); return ""; }
    virtual void ejectCard(Atm&) { no("eject a card"); }
protected:
    void no(const string& what) const { throw InvalidOperation("cannot " + what + " while " + name()); }
};

class Atm {
    unique_ptr<AtmState> state_;
    BankService& bank_;
    CashDispenser& cash_;
    string id_;                          // txn ids are "<atmId>-T<n>": globally unique, so idempotency keys never collide
    set<string> retained_;
    long txn_ = 0;
    Mutex mu_;                           // one customer, one keypad: every user action holds it
public:
    string card;
    string history;
    Atm(const string& id, BankService& b, CashDispenser& c);
    void moveTo(AtmState* s) {
        if (string(s->name()) != state_->name()) history += string(state_->name()) + "->" + s->name() + " ";
        state_.reset(s);
    }
    BankService& bank() { return bank_; }
    CashDispenser& cash() { return cash_; }
    string nextTxn() { return id_ + "-T" + to_string(++txn_); }
    bool isRetained(const string& c) const { return retained_.count(c) > 0; }
    void retain() { retained_.insert(card); bank_.blockCard(card); card.clear(); }
    void endSession();
    const char* state() const { return state_->name(); }
    // user actions: each takes the ATM lock, then delegates to the current state object
    void insertCard(const string& c) { LockGuard g(mu_); state_->insertCard(*this, c); }
    void enterPin(const string& p) { LockGuard g(mu_); state_->enterPin(*this, p); }
    long balance() { LockGuard g(mu_); return state_->balance(*this); }
    string withdraw(long a) { LockGuard g(mu_); return state_->withdraw(*this, a); }
    void ejectCard() { LockGuard g(mu_); state_->ejectCard(*this); }
};

struct Idle : AtmState {
    const char* name() const override { return "IDLE"; }
    void insertCard(Atm& atm, const string& c) override;
};
struct OutOfService : AtmState { const char* name() const override { return "OUT_OF_SERVICE"; } };

struct Dispensing : AtmState {
    const char* name() const override { return "DISPENSING"; }
    static string run(Atm& atm, long amount) {
        string txn = atm.nextTxn();
        Notes notes;
        if (!atm.cash().reserve((int)amount, notes)) throw Declined("ATM cannot dispense " + to_string(amount));
        DebitResult r;
        try {
            r = atm.bank().debit(atm.card, amount, txn);
        } catch (const exception&) {                    // timeout: did the debit land? unknown
            atm.cash().release(notes);
            try { atm.bank().reverse(txn); }            // safe either way: tombstone if never seen
            catch (const exception&) { /* a real ATM queues it and retries (store-and-forward) */ }
            throw Declined("bank unavailable for " + txn + ": nothing dispensed, reversal sent");
        }
        if (r != DebitResult::APPROVED) {
            atm.cash().release(notes);
            throw Declined("bank declined " + txn);
        }
        if (!atm.cash().eject(notes)) {
            atm.bank().reverse(txn);
            atm.cash().release(notes);
            return txn + " REVERSED, balance " + to_string(atm.bank().balance(atm.card));
        }
        return txn + " OK " + show(notes) + ", balance " + to_string(atm.bank().balance(atm.card));
    }
};

struct Authenticated : AtmState {
    const char* name() const override { return "AUTHENTICATED"; }
    long balance(Atm& atm) override { return atm.bank().balance(atm.card); }
    string withdraw(Atm& atm, long amount) override {
        if (amount <= 0 || amount % 100) throw InvalidOperation("amount must be a positive multiple of 100");
        atm.moveTo(new Dispensing());            // 'this' is destroyed here: do not touch members after
        string r;
        try { r = Dispensing::run(atm, amount); }
        catch (...) { atm.moveTo(new Authenticated()); throw; }
        atm.moveTo(new Authenticated());
        return r;
    }
    void ejectCard(Atm& atm) override { atm.endSession(); }
};

struct CardInserted : AtmState {
    int tries = 0;
    const char* name() const override { return "CARD_INSERTED"; }
    void enterPin(Atm& atm, const string& pin) override {
        if (atm.bank().verifyPin(atm.card, pin)) { atm.moveTo(new Authenticated()); return; }
        if (++tries >= 3) {
            atm.retain();
            atm.moveTo(new Idle());              // 'this' is gone after this line
            throw InvalidOperation("wrong PIN 3 times: card retained");
        }
        throw InvalidOperation("wrong PIN, tries left: " + to_string(3 - tries));
    }
    void ejectCard(Atm& atm) override { atm.endSession(); }
};

void Idle::insertCard(Atm& atm, const string& c) {
    if (atm.isRetained(c)) throw InvalidOperation("card " + c + " is blocked");
    atm.card = c;
    atm.moveTo(new CardInserted());
}

}

A real bug we hit while writing the C++ version. Each C++ ATM first numbered its transactions T1, T2, ... on its own. The second ATM's first withdrawal was then sent as "T1", which the bank had already seen from the first ATM. The bank's idempotency check answered "approved" from memory and debited nothing, while the ATM handed out cash. The assertion on the balance caught it. The fix is the rule every idempotency key needs: it must be unique across all senders, so the id is now <atmId>-T<n>. Java and Python use the ATM id plus a global counter for the same reason.

A small C++ detail worth knowing: after atm.moveTo(new Dispensing()) the old state object is destroyed, so the code that called it must not touch its own members afterwards. The comments mark those lines. It is the C++ version of using a struct after kfree(); Java and Python hide it with garbage collection.
