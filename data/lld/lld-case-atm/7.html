APIs and interfaces

Write these first; they are what the interviewer grades against.

// the ATM's public API (one customer at a time)
void    insertCard(String card);      // IDLE only
void    enterPin(String pin);         // CARD_INSERTED only; 3rd wrong PIN retains the card
long    balance();                    // AUTHENTICATED only
Receipt withdraw(long amount);        // AUTHENTICATED only; throws InvalidOperationException, TransactionDeclinedException
void    ejectCard();                  // CARD_INSERTED or AUTHENTICATED
void    loadCash(int denomination, int count);   // operator action; may leave OUT_OF_SERVICE

// one table of actions, every slot defaults to "not allowed now"
interface AtmState {
    StateName name();
    default void insertCard(Atm atm, String card) { reject("insert a card"); }
    default void enterPin(Atm atm, String pin)    { reject("enter a PIN"); }
    default long balance(Atm atm)                 { reject("check balance"); return 0; }
    default Receipt withdraw(Atm atm, long amt)   { reject("withdraw"); return null; }
    default void ejectCard(Atm atm)               { reject("eject a card"); }
}

// what the ATM needs from a bank, in the ATM's words
interface BankService {
    boolean     verifyPin(String card, String pin);
    long        balance(String card);
    DebitResult debit(String card, long rupees, String txnId);   // idempotent per txnId
    void        reverse(String txnId);                           // idempotent compensation
}

// the cash side
Map<Integer, Integer> reserve(int amount);   // plan + set aside, atomically; null if impossible
void    release(Map<Integer, Integer> notes);
boolean eject(Map<Integer, Integer> notes);     // false = hardware fault, notes still inside
