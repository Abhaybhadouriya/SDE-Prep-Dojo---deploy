C++ and Python versions

C++ has no interface keyword. A class with at least one pure virtual function (= 0) is abstract. By convention, a class with only pure virtual functions and a virtual destructor, and no data, is called an interface. C++ allows multiple inheritance of classes, so "implementing several interfaces" is just inheriting from several such classes.

Python has two flavours. abc.ABC with @abstractmethod is nominal: a class counts only if it inherits, and Python refuses to instantiate it while a method is missing. typing.Protocol (PEP 544) is structural: any class with matching methods fits, no inheritance needed, which is duck typing that a type checker can verify.

// Interfaces and abstract classes in C++14: both are "classes with pure virtual functions".
// Build + run:  g++ -std=c++14 -Wall -o interfaces interfaces.cpp && ./interfaces
#include <cstdio>
#include <memory>
#include <string>
#include <vector>

// "Interface" by convention: only pure virtual functions + virtual destructor, no data.
class RateLimiter {
public:
    virtual ~RateLimiter() = default;
    virtual bool allow(const std::string& client) = 0;
};

// "Abstract class": has data and real code, but at least one pure virtual hole.
class WindowLimiter : public RateLimiter {
public:
    explicit WindowLimiter(int limit) : limit_(limit) {}
    bool allow(const std::string& client) final {        // template method
        if (count(client) >= limit_) return false;
        record(client);
        return true;
    }
protected:
    virtual int  count(const std::string& client) = 0;   // subclasses choose the storage
    virtual void record(const std::string& client) = 0;
private:
    int limit_;
};

class InMemoryLimiter : public WindowLimiter {
public:
    using WindowLimiter::WindowLimiter;
protected:
    int count(const std::string& c) override {
        int n = 0;
        for (auto& s : seen_) if (s == c) n++;
        return n;
    }
    void record(const std::string& c) override { seen_.push_back(c); }
private:
    std::vector<std::string> seen_;
};

class AllowAll : public RateLimiter {                    // implements only the interface
public:
    bool allow(const std::string&) override { return true; }
};

int main() {
    std::vector<std::unique_ptr<RateLimiter>> ls;
    ls.push_back(std::unique_ptr<RateLimiter>(new InMemoryLimiter(2)));
    ls.push_back(std::unique_ptr<RateLimiter>(new AllowAll()));
    for (auto& l : ls) {
        std::printf("limiter:");
        for (int i = 0; i < 3; i++) std::printf(" %s", l->allow("10.0.0.7") ? "allow" : "deny");
        std::printf("\n");
    }
    // RateLimiter r;   // error: cannot declare variable of abstract type
    return 0;
}
$ g++ -std=c++14 -Wall -o interfaces interfaces.cpp && ./interfaces
limiter: allow allow deny
limiter: allow allow allow
"""Interfaces in Python: nominal (abc.ABC) and structural (typing.Protocol).
Run:  python interfaces.py
"""
from abc import ABC, abstractmethod
from typing import Protocol, runtime_checkable


class Notifier(ABC):                      # nominal: you must inherit to count
    def __init__(self, sender: str):
        self.sender = sender
        self.sent = 0

    def notify(self, user: str, msg: str) -> None:   # shared template
        self.deliver(user, f"[{self.sender}] {msg}")
        self.sent += 1

    @abstractmethod
    def deliver(self, user: str, body: str) -> None: ...


class Sms(Notifier):
    def deliver(self, user, body):
        print(f"  SMS to {user}: {body}")


@runtime_checkable
class PaymentGateway(Protocol):           # structural: any class with charge() fits
    def charge(self, order_id: str, paise: int) -> str: ...


class FakeGateway:                        # note: does NOT inherit PaymentGateway
    def charge(self, order_id, paise):
        return f"OK {order_id}"


def checkout(gw: PaymentGateway, order_id: str) -> str:
    return gw.charge(order_id, 100)


if __name__ == "__main__":
    print("1. ABC: shared code + abstract hole")
    s = Sms("ACME")
    s.notify("asha", "order shipped")
    print("  sent =", s.sent)
    try:
        Notifier("x")
    except TypeError as e:
        print("  Notifier('x') ->", type(e).__name__ + ":", str(e).split(" with")[0])

    print("2. Protocol: fits by shape, no inheritance")
    print("  ", checkout(FakeGateway(), "ord-9"))
    print("   isinstance(FakeGateway(), PaymentGateway) =", isinstance(FakeGateway(), PaymentGateway))
$ python interfaces.py

Tip: the C++ WindowLimiter::allow is marked final and calls protected pure virtual steps. That is the C++ spelling of a template method. Note the virtual destructor on the interface: always add one (see Polymorphism & vtables).
