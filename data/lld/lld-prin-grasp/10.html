import java.util.*;

public class GraspParking {
    enum Size { SMALL, MEDIUM, LARGE }

    enum VehicleType {                                     // Polymorphism
        BIKE(Size.SMALL), CAR(Size.MEDIUM), TRUCK(Size.LARGE);
        final Size size;
        VehicleType(Size s) { size = s; }
    }
    record Vehicle(String plate, VehicleType type) {}

    static final class ParkingSpot {                       // Expert: size + occupant
        final String id; final Size size; private Vehicle parked;
        ParkingSpot(String id, Size size) { this.id = id; this.size = size; }
        boolean isFree() { return parked == null; }
        boolean canFit(Vehicle v) { return isFree() && v.type().size.compareTo(size) <= 0; }
        void park(Vehicle v) { if (!canFit(v)) throw new IllegalStateException("spot " + id); parked = v; }
        void free() { parked = null; }
    }

    static final class Floor {                             // Expert: owns its spots
        final int number; final List<ParkingSpot> spots = new ArrayList<>();
        Floor(int number) { this.number = number; }
        Optional<ParkingSpot> findSpot(Vehicle v) { return spots.stream().filter(s -> s.canFit(v)).findFirst(); }
    }

    record Ticket(String id, Vehicle vehicle, ParkingSpot spot, long entryMinute) {
        long minutesParked(long exitMinute) { return Math.max(0, exitMinute - entryMinute); }   // Expert
    }

    interface PricingPolicy { long feeInPaise(Ticket t, long exitMinute); }   // Protected Variations
    static final class HourlyPricing implements PricingPolicy {
        private final Map<VehicleType, Long> perHour;
        HourlyPricing(Map<VehicleType, Long> perHour) { this.perHour = perHour; }
        public long feeInPaise(Ticket t, long exit) {
            return Math.max(1, (t.minutesParked(exit) + 59) / 60) * perHour.get(t.vehicle().type());
        }
    }
    static final class FlatPricing implements PricingPolicy {
        private final long flat;
        FlatPricing(long flat) { this.flat = flat; }
        public long feeInPaise(Ticket t, long exit) { return flat; }
    }

    interface PaymentGateway { boolean charge(String ref, long paise); }     // Indirection
    static final class FakeUpiGateway implements PaymentGateway {
        public boolean charge(String ref, long paise) { System.out.println("  [upi] " + ref + " " + paise); return true; }
    }
    interface Clock { long nowMinute(); }

    static final class TicketRepository {                  // Pure Fabrication
        private final Map<String, Ticket> open = new HashMap<>();
        void save(Ticket t) { open.put(t.id(), t); }
        Optional<Ticket> find(String id) { return Optional.ofNullable(open.get(id)); }
        void close(String id) { open.remove(id); }
        int openCount() { return open.size(); }
    }

    static final class ParkingLot {                        // Creator of Ticket
        final List<Floor> floors = new ArrayList<>();
        private int nextTicket = 1;
        Optional<Ticket> admit(Vehicle v, long now) {
            for (Floor f : floors) {
                Optional<ParkingSpot> s = f.findSpot(v);
                if (s.isPresent()) { s.get().park(v); return Optional.of(new Ticket("T" + nextTicket++, v, s.get(), now)); }
            }
            return Optional.empty();
        }
    }

    static final class ParkingController {                 // Controller
        private final ParkingLot lot; private final TicketRepository tickets;
        private final PricingPolicy pricing; private final PaymentGateway payments; private final Clock clock;
        ParkingController(ParkingLot l, TicketRepository r, PricingPolicy p, PaymentGateway g, Clock c) {
            lot = l; tickets = r; pricing = p; payments = g; clock = c;
        }
        String onEntry(String plate, VehicleType type) {
            Optional<Ticket> t = lot.admit(new Vehicle(plate, type), clock.nowMinute());
            if (t.isEmpty()) return "FULL";
            tickets.save(t.get());
            return t.get().id() + "@" + t.get().spot().id;
        }
        String onExit(String ticketId) {
            Ticket t = tickets.find(ticketId).orElseThrow();
            long fee = pricing.feeInPaise(t, clock.nowMinute());
            if (!payments.charge(t.id(), fee)) return "PAYMENT_FAILED";
            t.spot().free(); tickets.close(t.id());
            return "PAID " + fee;
        }
    }

    static long now = 0;
    public static void main(String[] args) {
        ParkingLot lot = new ParkingLot();
        Floor f0 = new Floor(0);
        f0.spots.add(new ParkingSpot("F0-S1", Size.SMALL)); f0.spots.add(new ParkingSpot("F0-M1", Size.MEDIUM));
        Floor f1 = new Floor(1); f1.spots.add(new ParkingSpot("F1-L1", Size.LARGE));
        lot.floors.add(f0); lot.floors.add(f1);
        TicketRepository repo = new TicketRepository();
        ParkingController gate = new ParkingController(lot, repo,
            new HourlyPricing(Map.of(VehicleType.BIKE, 2000L, VehicleType.CAR, 5000L, VehicleType.TRUCK, 10000L)),
            new FakeUpiGateway(), () -> now);
        System.out.println(gate.onEntry("KA01-B-1", VehicleType.BIKE));   // T1@F0-S1
        System.out.println(gate.onEntry("KA01-C-2", VehicleType.CAR));    // T2@F0-M1
        System.out.println(gate.onEntry("KA01-C-3", VehicleType.CAR));    // T3@F1-L1 (medium full)
        System.out.println(gate.onEntry("KA01-T-4", VehicleType.TRUCK));  // FULL
        now = 130;
        System.out.println(gate.onExit("T2"));                            // PAID 15000 (3 h x Rs 50)
    }
}
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Optional, Protocol

class Size(Enum):
    SMALL = 1; MEDIUM = 2; LARGE = 3

class VehicleType(Enum):                  # Polymorphism: the type knows its size
    BIKE = Size.SMALL; CAR = Size.MEDIUM; TRUCK = Size.LARGE

@dataclass(frozen=True)
class Vehicle:
    plate: str
    type: VehicleType

@dataclass
class ParkingSpot:                        # Information Expert
    id: str
    size: Size
    parked: Optional[Vehicle] = None
    def can_fit(self, v): return self.parked is None and v.type.value.value <= self.size.value
    def park(self, v):
        if not self.can_fit(v): raise ValueError(self.id)
        self.parked = v
    def free(self): self.parked = None

@dataclass
class Floor:                              # Information Expert
    number: int
    spots: list = field(default_factory=list)
    def find_spot(self, v): return next((s for s in self.spots if s.can_fit(v)), None)

@dataclass(frozen=True)
class Ticket:
    id: str; vehicle: Vehicle; spot: ParkingSpot; entry_minute: int
    def minutes_parked(self, exit_minute): return max(0, exit_minute - self.entry_minute)

class PricingPolicy(Protocol):            # Protected Variations
    def fee_in_paise(self, t: Ticket, exit_minute: int) -> int: ...

class HourlyPricing:
    def __init__(self, per_hour): self.per_hour = per_hour
    def fee_in_paise(self, t, exit_minute):
        return max(1, (t.minutes_parked(exit_minute) + 59) // 60) * self.per_hour[t.vehicle.type]

class FakeUpiGateway:                     # Indirection to a vendor
    def charge(self, ref, paise): print("  [upi]", ref, paise); return True

class TicketRepository:                   # Pure Fabrication
    def __init__(self): self._open = {}
    def save(self, t): self._open[t.id] = t
    def find(self, tid): return self._open.get(tid)
    def close(self, tid): self._open.pop(tid, None)

class ParkingLot:                         # Creator
    def __init__(self): self.floors, self._next = [], 1
    def admit(self, v, now):
        for f in self.floors:
            if (spot := f.find_spot(v)):
                spot.park(v); t = Ticket(f"T{self._next}", v, spot, now); self._next += 1
                return t
        return None

class ParkingController:                  # Controller
    def __init__(self, lot, repo, pricing, payments, clock: Callable[[], int]):
        self.lot, self.repo, self.pricing, self.payments, self.clock = lot, repo, pricing, payments, clock
    def on_entry(self, plate, vtype):
        t = self.lot.admit(Vehicle(plate, vtype), self.clock())
        if t is None: return "FULL"
        self.repo.save(t); return f"{t.id}@{t.spot.id}"
    def on_exit(self, tid):
        t = self.repo.find(tid)
        fee = self.pricing.fee_in_paise(t, self.clock())
        if not self.payments.charge(t.id, fee): return "PAYMENT_FAILED"
        t.spot.free(); self.repo.close(t.id)
        return f"PAID {fee}"
