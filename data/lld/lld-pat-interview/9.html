// Strategy: the CALLER picks how the fee is computed.
interface FeeStrategy { long fee(long minutes); }
static final FeeStrategy HOURLY   = m -> ((m + 59) / 60) * 40;   // 40 per started hour
static final FeeStrategy FLAT_DAY = m -> 300;
static final class ParkingTicket {
    private final FeeStrategy strategy;
    ParkingTicket(FeeStrategy s) { strategy = s; }
    long pay(long minutes) { return strategy.fee(minutes); }
}

// State: the OBJECT moves itself. Each state returns the next state.
interface VendState { VendState insertCoin(Vending v); VendState pressButton(Vending v); }
static final VendState IDLE = new VendState() {
    public VendState insertCoin(Vending v)  { v.log.add("coin accepted"); return HAS_COIN; }
    public VendState pressButton(Vending v) { v.log.add("insert a coin first"); return this; }
};
static final VendState HAS_COIN = new VendState() {
    public VendState insertCoin(Vending v)  { v.log.add("coin returned (already have one)"); return this; }
    public VendState pressButton(Vending v) {
        v.stock--; v.log.add("item dispensed");
        return v.stock > 0 ? IDLE : SOLD_OUT;
    }
};
// SOLD_OUT returns coins and stays SOLD_OUT. The machine just does: state = state.pressButton(this);
interface DataStore { String read(String key); }

// Decorator: ADDS behaviour; every call still reaches the inner object.
static final class UpperCaseDecorator implements DataStore {
    private final DataStore inner;
    UpperCaseDecorator(DataStore inner) { this.inner = inner; }
    public String read(String key) { return inner.read(key).toUpperCase(); }
}

// Proxy: CONTROLS access; it may answer without calling the real object at all.
static final class CachingAuthProxy implements DataStore {
    private final DataStore real; private final Set<String> allowed;
    private final Map<String, String> cache = new HashMap<>();
    CachingAuthProxy(DataStore real, Set<String> allowed) { this.real = real; this.allowed = allowed; }
    public String read(String key) {
        if (!allowed.contains(key)) throw new SecurityException("no access to " + key);
        return cache.computeIfAbsent(key, real::read);
    }
}
interface Command { void execute(); void undo(); }
static final class History {
    private final Deque<Command> undo = new ArrayDeque<>(), redo = new ArrayDeque<>();
    void run(Command c) { c.execute(); undo.push(c); redo.clear(); }   // a new action kills redo
    void undo() { if (!undo.isEmpty()) { Command c = undo.pop(); c.undo(); redo.push(c); } }
    void redo() { if (!redo.isEmpty()) { Command c = redo.pop(); c.execute(); undo.push(c); } }
}

static final class PriceFeed {   // Observer with an unsubscribe handle, safe to iterate while others subscribe
    private final List<Consumer<Integer>> observers = new CopyOnWriteArrayList<>();
    Runnable subscribe(Consumer<Integer> o) { observers.add(o); return () -> observers.remove(o); }
    void publish(int price) { for (Consumer<Integer> o : observers) o.accept(price); }
}
class Logger {
public:
    static Logger& get() {
        static Logger instance;   // built on the first call, destroyed at program exit
        return instance;
    }
    Logger(const Logger&) = delete;             // no copies: a copy would be a second "singleton"
    Logger& operator=(const Logger&) = delete;
    Logger(Logger&&) = delete;
    Logger& operator=(Logger&&) = delete;

    void log(const std::string& msg) { ++lines_; last_ = msg; }
    int lines() const { return lines_; }
    static int constructed;

private:
    Logger() { ++constructed; }
    ~Logger() { std::printf("Logger destroyed at exit after %d line(s)\n", lines_); }
    int lines_ = 0;
    std::string last_;
};
int Logger::constructed = 0;

// The follow-up the interviewer asked next: a tiny ref-counted handle (what shared_ptr does).
// Same idea as kref / refcount_t in the kernel: get() bumps the count, put() drops it, the last put frees.
// Single-threaded counter. For threads: std::atomic<int>, ++ is fetch_add(1, relaxed),
// and the put is: if (count_->fetch_sub(1, std::memory_order_acq_rel) == 1) { delete p_; ... }
template <typename T>
class RefPtr {
public:
    explicit RefPtr(T* p) : p_(p), count_(nullptr) {
        try { count_ = new int(1); }
        catch (...) { delete p; throw; }         // like shared_ptr: never leak p if the counter fails
    }
    RefPtr(const RefPtr& o) : p_(o.p_), count_(o.count_) { if (count_) ++*count_; }
    RefPtr(RefPtr&& o) noexcept : p_(o.p_), count_(o.count_) { o.p_ = nullptr; o.count_ = nullptr; }
    RefPtr& operator=(const RefPtr& o) {
        if (this != &o) { release(); p_ = o.p_; count_ = o.count_; if (count_) ++*count_; }
        return *this;
    }
    RefPtr& operator=(RefPtr&& o) noexcept {
        if (this != &o) { release(); p_ = o.p_; count_ = o.count_; o.p_ = nullptr; o.count_ = nullptr; }
        return *this;
    }
    ~RefPtr() { release(); }
    T* operator->() const { return p_; }
    int use_count() const { return count_ ? *count_ : 0; }

private:
    void release() {                             // kref_put: the last one frees
        if (count_ && --*count_ == 0) { delete p_; delete count_; }
        p_ = nullptr; count_ = nullptr;
    }
    T* p_;
    int* count_;
};
# 1. Strategy: in Python a strategy is often just a function in a dict (a registry).
FeeFn = Callable[[int], int]
FEE_STRATEGIES: Dict[str, FeeFn] = {
    "hourly": lambda minutes: ((minutes + 59) // 60) * 40,
    "flat_day": lambda minutes: 300,
}


def register_strategy(name: str, fn: FeeFn) -> None:
    FEE_STRATEGIES[name] = fn          # open for extension: no if/else to edit


def fee(kind: str, minutes: int) -> int:
    return FEE_STRATEGIES[kind](minutes)


# 2. Observer with weak references, so a forgotten subscriber does not leak.
#    Trade-off: the caller must keep each observer alive (a lambda or an object held
#    only here vanishes at once: the "vanishing listener"), and bound methods need
#    weakref.WeakMethod. The explicit alternative: attach() returns an unsubscribe
#    handle, as the Java tab does.
class Subject:
    def __init__(self) -> None:
        self._observers: "weakref.WeakSet[Watcher]" = weakref.WeakSet()

    def attach(self, o: "Watcher") -> None:
        self._observers.add(o)

    def notify(self, event: str) -> int:
        for o in list(self._observers):
            o.update(event)
        return len(self._observers)


class Watcher:
    def __init__(self, name: str, inbox: List[str]) -> None:
        self.name, self.inbox = name, inbox

    def update(self, event: str) -> None:
        self.inbox.append(f"{self.name}:{event}")


# 3. Decorator: same interface, wraps one object, adds behaviour; stack them in any order.
class Coffee:
    def cost(self) -> int:
        return 100

    def describe(self) -> str:
        return "coffee"


class AddOn:
    def __init__(self, inner, price: int, name: str) -> None:
        self.inner, self.price, self.name = inner, price, name

    def cost(self) -> int:
        return self.inner.cost() + self.price

    def describe(self) -> str:
        return f"{self.inner.describe()} + {self.name}"
