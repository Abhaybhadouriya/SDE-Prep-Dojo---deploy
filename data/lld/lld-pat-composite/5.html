import java.util.*;

public class CompositeDemo {

    // ---- Component: what clients see for BOTH files and directories ----
    static abstract class Node {
        private final String name;
        private Directory parent;                  // back pointer, like dentry->d_parent
        Node(String name) { this.name = name; }
        String name() { return name; }
        Directory parent() { return parent; }
        void setParent(Directory p) { parent = p; }

        abstract long size();                      // the uniform operation
        abstract void print(String indent);        // another uniform operation
        String path() { return parent == null ? name : parent.path() + "/" + name; }
    }

    // ---- Leaf: has no children ----
    static final class File extends Node {
        private final long bytes;
        File(String name, long bytes) { super(name); this.bytes = bytes; }
        @Override long size() { return bytes; }
        @Override void print(String indent) {
            System.out.println(indent + "- " + name() + " (" + bytes + " B)");
        }
    }

    // ---- Composite: holds children (any Node) and forwards operations to them ----
    static final class Directory extends Node {
        private final Map<String, Node> children = new LinkedHashMap<>();
        Directory(String name) { super(name); }

        // "Safety" design: add/remove live only on Directory, not on Node.
        Directory add(Node child) {
            // Invariant: a failed add leaves the tree unchanged, so check everything first.
            for (Directory d = this; d != null; d = d.parent())     // refuse cycles
                if (d == child) throw new IllegalArgumentException("cycle: " + child.name());
            if (child.parent() == this) return this;                 // already here
            if (children.containsKey(child.name()))                  // clash: refuse BEFORE detaching
                throw new IllegalArgumentException("exists: " + child.name());
            if (child.parent() != null) child.parent().children.remove(child.name());
            children.put(child.name(), child);                       // <-- the pattern: a child is any Node
            child.setParent(this);
            return this;
        }
        Node remove(String name) {
            Node n = children.remove(name);
            if (n != null) n.setParent(null);
            return n;
        }
        Node child(String name) { return children.get(name); }

        @Override long size() {                     // recursion does the tree walk
            long total = 0;
            for (Node c : children.values()) total += c.size();
            return total;
        }
        @Override void print(String indent) {
            System.out.println(indent + "+ " + name() + "/ (" + size() + " B)");
            for (Node c : children.values()) c.print(indent + "  ");
        }
    }

    // Client code: works on any Node, never asks "is this a file or a folder?"
    static void report(Node n) {
        System.out.println(n.path() + " -> " + n.size() + " bytes");
    }

    public static void main(String[] args) {
        Directory root = new Directory("root");
        Directory src = new Directory("src");
        Directory net = new Directory("net");
        net.add(new File("tcp.c", 4200)).add(new File("udp.c", 1800));
        src.add(net).add(new File("main.c", 900));
        root.add(src).add(new File("README", 300));

        root.print("");
        report(root);                               // a composite
        report(net);                                // a smaller composite
        report(net.child("tcp.c"));                 // a leaf, same call

        try { net.add(root); }                      // root is net's ancestor
        catch (IllegalArgumentException e) { System.out.println("refused: " + e.getMessage()); }

        Directory archive = new Directory("archive");
        root.add(archive);
        archive.add(net);                           // move = re-parent (like rename(2))
        System.out.println("after move: " + net.path() + ", src now " + src.size() + " B");

        File clash = new File("README", 1);           // same name as root/README
        archive.add(clash);
        try { root.add(clash); }                      // refused: root already has a README
        catch (IllegalArgumentException e) { System.out.println("refused: " + e.getMessage()); }
        System.out.println("failed add left tree unchanged: " + clash.path() + ", archive " + archive.size() + " B");
    }
}

Three details worth saying in an interview: the children map is keyed by name, so duplicate names are refused (like EEXIST); add() walks up the parent chain to refuse cycles (like rename(2) refusing to move a directory into its own subtree with EINVAL); and a move is just "remove from old parent, add to new parent". Reading like a reviewer: add() does every check before it changes anything (cycle, then name clash, then detach and attach). An earlier draft detached the child from its old parent before the clash check, so a refused move silently lost the file. The invariant to state: a failed add leaves the tree unchanged; the last two output lines prove it.

"""Composite: an expression tree. Numbers are leaves, operators are composites."""
from __future__ import annotations
from abc import ABC, abstractmethod
import operator


class Expr(ABC):                                   # Component
    @abstractmethod
    def evaluate(self) -> float: ...

    @abstractmethod
    def render(self) -> str: ...


class Num(Expr):                                   # Leaf
    def __init__(self, value: float):
        self.value = value

    def evaluate(self) -> float:
        return self.value

    def render(self) -> str:
        return f"{self.value:g}"


class BinOp(Expr):                                 # Composite with exactly two children
    OPS = {"+": operator.add, "-": operator.sub, "*": operator.mul, "/": operator.truediv}

    def __init__(self, op: str, left: Expr, right: Expr):
        if op not in self.OPS:
            raise ValueError(f"unknown operator {op}")
        self.op, self.left, self.right = op, left, right

    def evaluate(self) -> float:
        return self.OPS[self.op](self.left.evaluate(), self.right.evaluate())

    def render(self) -> str:
        return f"({self.left.render()} {self.op} {self.right.render()})"


class Sum(Expr):                                   # Composite with any number of children
    def __init__(self, *parts: Expr):
        self.parts = list(parts)

    def add(self, part: Expr) -> "Sum":
        self.parts.append(part)
        return self

    def evaluate(self) -> float:
        return sum(p.evaluate() for p in self.parts)

    def render(self) -> str:
        return "sum[" + ", ".join(p.render() for p in self.parts) + "]"


def show(e: Expr) -> None:                         # client: one call for leaf or tree
    print(f"{e.render()} = {e.evaluate():g}")


if __name__ == "__main__":
    show(Num(7))
    expr = BinOp("+", Num(2), BinOp("*", Num(3), Num(4)))      # 2 + (3 * 4)
    show(expr)
    total = Sum(expr, Num(10)).add(BinOp("/", Num(9), Num(3)))
    show(total)

    # The standard library has composites too: an XML element holds elements.
    import xml.etree.ElementTree as ET
    menu = ET.fromstring("<menu><item>Home</item><menu><item>Docs</item><item>API</item></menu></menu>")
    print("items in nested menu:", [i.text for i in menu.iter("item")])

BinOp is a composite with exactly two children and Sum has any number; both are just Expr to the client. This is also the Interpreter pattern: an Interpreter is a Composite whose operation is "evaluate".

// Composite in C++14: a menu tree. MenuItem is a leaf, Menu is a composite.
// "Transparent" design: add() is declared on the Component; leaves reject it.
// Build: g++ -std=c++14 -Wall composite.cpp -o composite
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

class MenuComponent {
public:
    explicit MenuComponent(std::string name) : name_(std::move(name)) {}
    virtual ~MenuComponent() = default;
    const std::string& name() const { return name_; }

    virtual int price() const = 0;                         // uniform operation
    virtual void print(int depth) const = 0;               // uniform operation
    virtual void add(std::unique_ptr<MenuComponent>) {     // transparency: on the base
        throw std::logic_error(name_ + " is a leaf; cannot add children");
    }
    virtual bool isComposite() const { return false; }
protected:
    std::string name_;
};

class MenuItem : public MenuComponent {                    // Leaf
public:
    MenuItem(std::string n, int p) : MenuComponent(std::move(n)), price_(p) {}
    int price() const override { return price_; }
    void print(int depth) const override {
        std::cout << std::string(depth * 2, ' ') << "- " << name_ << " Rs " << price_ << "\n";
    }
private:
    int price_;
};

class Menu : public MenuComponent {                        // Composite
public:
    using MenuComponent::MenuComponent;
    void add(std::unique_ptr<MenuComponent> c) override { kids_.push_back(std::move(c)); }
    bool isComposite() const override { return true; }
    int price() const override {                           // a combo = sum of its parts
        int total = 0;
        for (const auto& k : kids_) total += k->price();
        return total;
    }
    void print(int depth) const override {
        std::cout << std::string(depth * 2, ' ') << "+ " << name_ << " (Rs " << price() << ")\n";
        for (const auto& k : kids_) k->print(depth + 1);
    }
private:
    std::vector<std::unique_ptr<MenuComponent>> kids_;     // the tree owns its children
};

int main() {
    auto root = std::make_unique<Menu>("Lunch menu");
    auto combo = std::make_unique<Menu>("Thali combo");
    combo->add(std::make_unique<MenuItem>("Dal", 80));
    combo->add(std::make_unique<MenuItem>("Rice", 60));
    combo->add(std::make_unique<MenuItem>("Roti x2", 40));
    root->add(std::move(combo));
    root->add(std::make_unique<MenuItem>("Lassi", 50));
    root->print(0);

    MenuItem chai("Masala chai", 30);
    MenuComponent& c = chai;                               // client holds the base type
    try { c.add(std::make_unique<MenuItem>("Sugar", 0)); }
    catch (const std::logic_error& e) { std::cout << "error: " << e.what() << "\n"; }
    std::cout << "root is composite? " << std::boolalpha << root->isComposite() << "\n";
    return 0;
}

Ownership is the C++ question: a composite owns its children through std::unique_ptr, so destroying the root frees the whole tree, depth first, with no manual walk. If nodes must be shared between parents (a DAG, not a tree), you need shared_ptr or, like the kernel's dentries, a reference count; and parent back-pointers should then be non-owning to avoid cycles.

The same idea in C

Composite in plain C: each node embeds a list of children (a tiny list_head) and one recursive node_size() works for files and directories alike. Illustrative, not real kernel code; it compiles and runs (lld-code/lld-pat-composite/composite_sketch.c).

/* Composite in C: a node with an embedded list of children and a recursive size().
   Illustrative, not real kernel code (a tiny list_head and container_of, as in <linux/list.h>).
   Build: gcc -std=c99 -Wall composite_sketch.c -o composite_sketch */
#include <stddef.h>
#include <stdio.h>

struct list_head { struct list_head *next, *prev; };
#define container_of(p, type, member) ((type *)((char *)(p) - offsetof(type, member)))
static void list_add_tail(struct list_head *n, struct list_head *h)
{ n->prev = h->prev; n->next = h; h->prev->next = n; h->prev = n; }

struct node {
    const char *name;
    long bytes;                     /* a file's own size; 0 for a directory */
    struct list_head children;      /* empty list = a leaf */
    struct list_head sibling;       /* our link in the parent's children list */
};

static long node_size(struct node *n)          /* one function for leaf and composite */
{
    long total = n->bytes;
    for (struct list_head *p = n->children.next; p != &n->children; p = p->next)
        total += node_size(container_of(p, struct node, sibling));   /* recurse */
    return total;
}

int main(void)
{
    struct node root = { "root", 0 }, net = { "net", 0 }, tcp = { "tcp.c", 4200 }, readme = { "README", 300 };
    struct node *all[] = { &root, &net, &tcp, &readme };
    for (int i = 0; i < 4; i++) all[i]->children.next = all[i]->children.prev = &all[i]->children;
    list_add_tail(&tcp.sibling, &net.children);
    list_add_tail(&net.sibling, &root.children);
    list_add_tail(&readme.sibling, &root.children);
    printf("size(tcp.c)=%ld size(net)=%ld size(root)=%ld\n", node_size(&tcp), node_size(&net), node_size(&root));
    return 0;
}
