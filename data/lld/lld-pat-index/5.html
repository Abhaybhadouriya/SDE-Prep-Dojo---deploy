Code: three patterns, three kinds of change

A tiny parking lot that keeps the three kinds of change from the "problem" section apart: a Factory for what is created, a Strategy for how the fee is computed (swapped at run time), and Observer listeners for who reacts to spot changes. Each language version is complete and prints the same output.

Map before you read: which class plays which role in the Java tab (PatternMapDemo.java). The Python and C++ tabs use the same roles under their own names. Key lines carry a // <-- the pattern comment.

// Three tier-1 patterns in one tiny parking lot, each placed where a different kind of change lives.
//   WHAT object to build varies   -> Factory   (VehicleFactory)
//   HOW a rule is computed varies -> Strategy  (FeePolicy)
//   WHO must react to an event    -> Observer  (LotListener)
// Run: java PatternMapDemo.java   (Java 21)
import java.util.ArrayList;
import java.util.List;

public class PatternMapDemo {

    // ---------- Factory: creation varies ----------
    interface Vehicle { String plate(); int wheels(); }
    record Bike(String plate) implements Vehicle { public int wheels() { return 2; } }
    record Car(String plate)  implements Vehicle { public int wheels() { return 4; } }

    static final class VehicleFactory {
        static Vehicle create(String type, String plate) {
            // <-- the pattern: the factory is the only place that names Bike/Car
            return switch (type) {
                case "bike" -> new Bike(plate);
                case "car"  -> new Car(plate);
                default -> throw new IllegalArgumentException("unknown type " + type);
            };
        }
    }

    // ---------- Strategy: an algorithm varies ----------
    interface FeePolicy { int fee(Vehicle v, int hours); }
    static final class FlatFee implements FeePolicy {
        public int fee(Vehicle v, int hours) { return v.wheels() == 2 ? 20 : 50; }
    }
    static final class HourlyFee implements FeePolicy {
        public int fee(Vehicle v, int hours) { return hours * (v.wheels() == 2 ? 10 : 30); }
    }

    // ---------- Observer: who reacts to an event varies ----------
    interface LotListener { void onChange(int free); }

    static final class ParkingLot {
        private int free;
        private FeePolicy policy;
        private final List<LotListener> listeners = new ArrayList<>();
        ParkingLot(int capacity, FeePolicy policy) { this.free = capacity; this.policy = policy; }
        void setPolicy(FeePolicy p) { this.policy = p; }            // swap the strategy at runtime
        void subscribe(LotListener l) { listeners.add(l); }
        void enter(Vehicle v) { free--; publish(); }
        int exit(Vehicle v, int hours) { free++; publish(); return policy.fee(v, hours); }
        private void publish() { for (LotListener l : listeners) l.onChange(free); }
    }

    public static void main(String[] args) {
        ParkingLot lot = new ParkingLot(3, new FlatFee());
        lot.subscribe(free -> System.out.println("  [board] free spots: " + free));
        lot.subscribe(free -> { if (free == 0) System.out.println("  [ops] lot FULL - turn on red sign"); });

        Vehicle a = VehicleFactory.create("car", "KA-01");
        Vehicle b = VehicleFactory.create("bike", "KA-02");
        Vehicle c = VehicleFactory.create("car", "KA-03");
        lot.enter(a); lot.enter(b); lot.enter(c);

        System.out.println("flat fee, car 3h   = " + lot.exit(a, 3));
        lot.setPolicy(new HourlyFee());
        System.out.println("hourly fee, bike 3h = " + lot.exit(b, 3));
    }
}
"""Three tier-1 patterns in one tiny parking lot, each placed where a different kind of change lives.
  WHAT object to build varies   -> Factory   (create_vehicle)
  HOW a rule is computed varies -> Strategy  (fee policies are plain functions here)
  WHO must react to an event    -> Observer  (listeners)
Run: python pattern_map.py   (Python 3.11)"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable


# ---------- Factory ----------
@dataclass(frozen=True)
class Vehicle:
    plate: str
    wheels: int


def create_vehicle(kind: str, plate: str) -> Vehicle:
    makers = {"bike": lambda p: Vehicle(p, 2), "car": lambda p: Vehicle(p, 4)}
    if kind not in makers:
        raise ValueError(f"unknown type {kind}")
    return makers[kind](plate)


# ---------- Strategy (a function is the lightest strategy object) ----------
FeePolicy = Callable[[Vehicle, int], int]


def flat_fee(v: Vehicle, hours: int) -> int:
    return 20 if v.wheels == 2 else 50


def hourly_fee(v: Vehicle, hours: int) -> int:
    return hours * (10 if v.wheels == 2 else 30)


# ---------- Observer ----------
class ParkingLot:
    def __init__(self, capacity: int, policy: FeePolicy) -> None:
        self.free = capacity
        self.policy = policy
        self._listeners: list[Callable[[int], None]] = []

    def subscribe(self, fn: Callable[[int], None]) -> None:
        self._listeners.append(fn)

    def enter(self, v: Vehicle) -> None:
        self.free -= 1
        self._publish()

    def exit(self, v: Vehicle, hours: int) -> int:
        self.free += 1
        self._publish()
        return self.policy(v, hours)

    def _publish(self) -> None:
        for fn in self._listeners:
            fn(self.free)


if __name__ == "__main__":
    lot = ParkingLot(3, flat_fee)
    lot.subscribe(lambda free: print(f"  [board] free spots: {free}"))
    lot.subscribe(lambda free: free == 0 and print("  [ops] lot FULL - turn on red sign"))

    a, b, c = create_vehicle("car", "KA-01"), create_vehicle("bike", "KA-02"), create_vehicle("car", "KA-03")
    for v in (a, b, c):
        lot.enter(v)
    print("flat fee, car 3h   =", lot.exit(a, 3))
    lot.policy = hourly_fee  # swap the strategy at runtime
    print("hourly fee, bike 3h =", lot.exit(b, 3))
// Three tier-1 patterns in one tiny parking lot, each placed where a different kind of change lives.
//   WHAT object to build varies   -> Factory   (createVehicle)
//   HOW a rule is computed varies -> Strategy  (FeePolicy)
//   WHO must react to an event    -> Observer  (listeners)
// Build: g++ -std=c++14 -Wall -O2 pattern_map.cpp -o pattern_map && ./pattern_map
#include <functional>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

// ---------- Factory ----------
struct Vehicle { std::string plate; int wheels; };

std::unique_ptr<Vehicle> createVehicle(const std::string& type, const std::string& plate) {
    if (type == "bike") return std::unique_ptr<Vehicle>(new Vehicle{plate, 2});
    if (type == "car")  return std::unique_ptr<Vehicle>(new Vehicle{plate, 4});
    throw std::invalid_argument("unknown type " + type);
}

// ---------- Strategy ----------
struct FeePolicy {
    virtual ~FeePolicy() = default;
    virtual int fee(const Vehicle& v, int hours) const = 0;
};
struct FlatFee : FeePolicy {
    int fee(const Vehicle& v, int) const override { return v.wheels == 2 ? 20 : 50; }
};
struct HourlyFee : FeePolicy {
    int fee(const Vehicle& v, int hours) const override { return hours * (v.wheels == 2 ? 10 : 30); }
};

// ---------- Observer ----------
class ParkingLot {
public:
    using Listener = std::function<void(int)>;
    ParkingLot(int capacity, std::unique_ptr<FeePolicy> p) : free_(capacity), policy_(std::move(p)) {}
    void setPolicy(std::unique_ptr<FeePolicy> p) { policy_ = std::move(p); }
    void subscribe(Listener l) { listeners_.push_back(std::move(l)); }
    void enter(const Vehicle&) { --free_; publish(); }
    int exit(const Vehicle& v, int hours) { ++free_; publish(); return policy_->fee(v, hours); }
private:
    void publish() { for (auto& l : listeners_) l(free_); }
    int free_;
    std::unique_ptr<FeePolicy> policy_;
    std::vector<Listener> listeners_;
};

int main() {
    ParkingLot lot(3, std::unique_ptr<FeePolicy>(new FlatFee));
    lot.subscribe([](int free) { std::cout << "  [board] free spots: " << free << "\n"; });
    lot.subscribe([](int free) { if (free == 0) std::cout << "  [ops] lot FULL - turn on red sign\n"; });

    auto a = createVehicle("car", "KA-01");
    auto b = createVehicle("bike", "KA-02");
    auto c = createVehicle("car", "KA-03");
    lot.enter(*a); lot.enter(*b); lot.enter(*c);

    std::cout << "flat fee, car 3h   = " << lot.exit(*a, 3) << "\n";
    lot.setPolicy(std::unique_ptr<FeePolicy>(new HourlyFee));
    std::cout << "hourly fee, bike 3h = " << lot.exit(*b, 3) << "\n";
    return 0;
}

Compare it with the knotted exit() at the top of the page. Adding an EV is one new class and one factory entry. Adding a weekend plan is one new FeePolicy. Adding an SMS alert is one more subscribe call. None of them edits ParkingLot. Notice that the Python version uses plain functions as strategies and listeners: the pattern is the same, the ceremony is smaller.
