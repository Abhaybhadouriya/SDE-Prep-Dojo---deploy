"""Chat application: 1:1 and group chats, per-recipient receipts (State), server as Mediator, devices as Observers."""
from __future__ import annotations

import itertools
import threading
from enum import Enum
from typing import Dict, List, Optional, Protocol, Tuple


class ChatError(Exception):
    pass


class Receipt(Enum):
    """Per-recipient state. Each state decides what an event does; receipts only move forward."""
    SENT = 0
    DELIVERED = 1
    READ = 2

    def on_delivered(self) -> "Receipt":
        return Receipt.DELIVERED if self is Receipt.SENT else self

    def on_read(self) -> "Receipt":
        return Receipt.READ


class Message:
    def __init__(self, mid: str, chat_id: str, sender: str, seq: int, text: str, recipients: List[str]):
        self.id, self.chat_id, self.sender, self.seq, self.text = mid, chat_id, sender, seq, text
        self._receipts: Dict[str, Receipt] = {r: Receipt.SENT for r in recipients}
        self._pushed_in: Dict[str, int] = {}       # recipient -> connection that already got a push
        self._lock = threading.Lock()

    def claim_push(self, user: str, connection: int) -> bool:
        """Avoids a duplicate push inside one connection; it is NOT the DELIVERED tick."""
        with self._lock:
            if self._receipts.get(user) is not Receipt.SENT:
                return False                       # already acknowledged
            prev = self._pushed_in.get(user)
            self._pushed_in[user] = connection
            return prev != connection

    def apply(self, user: str, read: bool) -> bool:
        """One event for one recipient; True only if the state really changed."""
        with self._lock:
            old = self._receipts.get(user)
            if old is None:
                return False
            new = old.on_read() if read else old.on_delivered()
            if new is old:
                return False
            self._receipts[user] = new
            return True

    def receipt_of(self, user: str) -> Optional[Receipt]:
        with self._lock:
            return self._receipts.get(user)

    def forget(self, user: str) -> None:
        with self._lock:
            self._receipts.pop(user, None)

    def status(self) -> Receipt:
        """What the sender sees: the lowest receipt over all recipients."""
        with self._lock:
            if not self._receipts:
                return Receipt.SENT
            return min(self._receipts.values(), key=lambda r: r.value)

    def recipients(self) -> List[str]:
        with self._lock:
            return list(self._receipts)

    def __repr__(self) -> str:
        return f"#{self.seq} {self.sender}: {self.text}"


class Chat:
    def __init__(self, cid: str):
        self.id = cid
        self.lock = threading.Lock()
        self.members: Dict[str, int] = {}   # user -> first seq they may see (dict keeps join order)
        self.messages: List[Message] = []
        self._next_seq = 1

    def title(self) -> str:
        raise NotImplementedError

    def check_can_change_members(self, by: str, target: str, adding: bool) -> None:
        raise NotImplementedError

    def is_member(self, user: str) -> bool:
        return user in self.members

    def last_seq(self) -> int:
        return self._next_seq - 1

    def join(self, user: str) -> None:
        self.members.setdefault(user, self._next_seq)

    def leave(self, user: str) -> None:
        self.members.pop(user, None)

    def append(self, mid: str, sender: str, text: str) -> Message:
        """Caller holds lock: one seq per message, no gaps, no duplicates."""
        to = [m for m in self.members if m != sender]
        msg = Message(mid, self.id, sender, self._next_seq, text, to)
        self._next_seq += 1
        self.messages.append(msg)
        return msg

    def by_seq(self, seq: int) -> Optional[Message]:
        """Caller holds lock: seqs are 1..last with no gaps, so this is an index."""
        return self.messages[seq - 1] if 1 <= seq <= len(self.messages) else None

    def visible_to(self, user: str) -> List[Message]:
        start = self.members[user]
        return [m for m in self.messages if m.seq >= start]


class DirectChat(Chat):
    def __init__(self, cid: str, a: str, b: str):
        super().__init__(cid)
        self.join(a)
        self.join(b)

    def title(self) -> str:
        return "direct " + "+".join(self.members)

    def check_can_change_members(self, by, target, adding):
        raise ChatError("a direct chat has exactly two members")


class GroupChat(Chat):
    MAX_MEMBERS = 256

    def __init__(self, cid: str, name: str, creator: str):
        super().__init__(cid)
        self.name = name
        self.admins = {creator}
        self.join(creator)

    def title(self) -> str:
        return f"group '{self.name}'"

    def check_can_change_members(self, by, target, adding):
        self_leave = not adding and by == target
        if not self_leave and by not in self.admins:
            raise ChatError(f"{by} is not an admin of {self.name}")
        if adding and len(self.members) >= self.MAX_MEMBERS:
            raise ChatError("group is full")

    def leave(self, user: str) -> None:
        super().leave(user)
        self.admins.discard(user)
        if not self.admins and self.members:
            self.admins.add(next(iter(self.members)))   # oldest member becomes admin


class ChatClient(Protocol):
    def on_message(self, m: Message) -> None: ...
    def on_receipt(self, m: Message, by: str, now: Receipt) -> None: ...


class ChatServer:
    """Mediator: users never reference each other; every message goes through here."""

    def __init__(self) -> None:
        self._users: Dict[str, str] = {}
        self._chats: Dict[str, Chat] = {}
        self._chats_of: Dict[str, set] = {}
        self._online: Dict[str, Tuple[int, ChatClient]] = {}   # uid -> (connection id, client)
        self._conn_ids = itertools.count(1)
        self._registry = threading.Lock()            # guards the dicts above
        self._msg_ids = itertools.count(1)
        self._group_ids = itertools.count(1)

    def register(self, uid: str, name: str) -> None:
        with self._registry:
            if uid in self._users:
                raise ChatError(f"user {uid} exists")
            self._users[uid] = name
            self._chats_of[uid] = set()

    def _need_user(self, uid: str) -> None:
        if uid not in self._users:
            raise ChatError(f"no user {uid}")

    def chat(self, cid: str) -> Chat:
        c = self._chats.get(cid)
        if c is None:
            raise ChatError(f"no chat {cid}")
        return c

    def direct(self, a: str, b: str) -> DirectChat:
        self._need_user(a)
        self._need_user(b)
        if a == b:
            raise ChatError("cannot chat with yourself")
        key = "d:" + ":".join(sorted((a, b)))
        with self._registry:
            c = self._chats.get(key)
            if c is None:
                c = self._chats[key] = DirectChat(key, a, b)
            self._chats_of[a].add(key)
            self._chats_of[b].add(key)
        return c

    def group(self, creator: str, name: str, *others: str) -> GroupChat:
        self._need_user(creator)
        with self._registry:
            g = GroupChat(f"g{next(self._group_ids)}", name, creator)
            self._chats[g.id] = g
            self._chats_of[creator].add(g.id)
        for o in others:
            self.add_member(creator, g.id, o)
        return g

    def add_member(self, by: str, cid: str, user: str) -> None:
        self._need_user(user)
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(by):
                raise ChatError(f"{by} is not in {c.title()}")
            c.check_can_change_members(by, user, True)   # permission first, even for a no-op
            if c.is_member(user):
                return
            c.join(user)
        with self._registry:
            self._chats_of[user].add(cid)

    def remove_member(self, by: str, cid: str, user: str) -> None:
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(by) or not c.is_member(user):
                raise ChatError("not a member")
            c.check_can_change_members(by, user, False)
            c.leave(user)
            for m in c.messages:
                if m.receipt_of(user) is not Receipt.READ:
                    m.forget(user)
        with self._registry:
            self._chats_of[user].discard(cid)

    def connect(self, uid: str, client: ChatClient) -> None:
        """Re-pushes every message this user has not ACKNOWLEDGED (still SENT): at-least-once."""
        self._need_user(uid)
        with self._registry:
            conn = (next(self._conn_ids), client)
            self._online[uid] = conn
            my_chats = list(self._chats_of[uid])
        for cid in my_chats:
            c = self._chats[cid]
            with c.lock:
                if not c.is_member(uid):
                    continue
                pending = [m for m in c.visible_to(uid) if m.receipt_of(uid) is Receipt.SENT]
            for m in pending:                 # outside the lock
                self._deliver(uid, conn, m)

    def disconnect(self, uid: str) -> None:
        with self._registry:
            self._online.pop(uid, None)

    def send(self, sender: str, cid: str, text: str) -> Message:
        if not text or not text.strip():
            raise ChatError("empty message")
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(sender):
                raise ChatError(f"{sender} is not in {c.title()}")
            m = c.append(f"m{next(self._msg_ids)}", sender, text.strip())
        for r in m.recipients():              # fan-out after unlock
            cn = self._online.get(r)
            if cn is not None:
                self._deliver(r, cn, m)
        return m

    def _deliver(self, uid: str, conn: Tuple[int, ChatClient], m: Message) -> None:
        """Push only; the receipt stays SENT until the device acks. A lost push is re-sent on reconnect."""
        if m.claim_push(uid, conn[0]):
            conn[1].on_message(m)

    def ack(self, uid: str, cid: str, seq: int) -> bool:
        """The device stored message seq: only this moves SENT -> DELIVERED. Idempotent."""
        c = self.chat(cid)
        with c.lock:
            m = c.by_seq(seq) if c.is_member(uid) else None
            if m is None or not m.apply(uid, read=False):
                return False
        self._tell_sender(m, uid)
        return True

    def _tell_sender(self, m: Message, by: str) -> None:
        s = self._online.get(m.sender)
        if s is not None:
            s[1].on_receipt(m, by, m.receipt_of(by))

    def mark_read(self, uid: str, cid: str, upto_seq: int) -> int:
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(uid):
                raise ChatError(f"{uid} is not in {c.title()}")
            changed = [m for m in c.visible_to(uid)
                       if m.seq <= upto_seq and m.sender != uid and m.apply(uid, read=True)]
        for m in changed:
            self._tell_sender(m, uid)
        return len(changed)

    def unread(self, uid: str, cid: str) -> int:
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(uid):
                raise ChatError(f"{uid} is not in {c.title()}")
            return sum(1 for m in c.visible_to(uid) if m.sender != uid and m.receipt_of(uid) is not Receipt.READ)

    def history(self, uid: str, cid: str, before_seq: float = float("inf"), limit: int = 50) -> List[Message]:
        c = self.chat(cid)
        with c.lock:
            if not c.is_member(uid):
                raise ChatError(f"{uid} is not in {c.title()}")
            older = [m for m in c.visible_to(uid) if m.seq < before_seq]
            return older[-limit:] if limit > 0 else []


class Device:
    """A client device: stores a message, drops copies it already has (dedupe by chat + seq), then acks."""

    def __init__(self, server: "ChatServer", owner: str) -> None:
        self.server, self.owner = server, owner
        self._have: set = set()
        self._have_lock = threading.Lock()

    def on_message(self, m: Message) -> None:
        if not self.arrives(m):               # lost on the wire
            return
        with self._have_lock:
            fresh = (m.chat_id, m.seq) not in self._have
            self._have.add((m.chat_id, m.seq))
        if fresh:
            self.store(m)
        else:
            self.duplicate(m)
        if self.send_ack(m):                  # ack a duplicate too: the first ack may be the lost one
            self.server.ack(self.owner, m.chat_id, m.seq)

    def arrives(self, m: Message) -> bool:
        return True

    def store(self, m: Message) -> None:
        pass

    def duplicate(self, m: Message) -> None:
        pass

    def send_ack(self, m: Message) -> bool:
        return True

    def on_receipt(self, m: Message, by: str, now: Receipt) -> None:
        pass

import threading

from chat import ChatError, ChatServer, Device, Receipt


def check(cond, msg="check failed"):
    """Like assert, but never stripped by python -O: the call inside it always runs."""
    if not cond:
        raise AssertionError(msg)


class ConsoleClient(Device):
    def store(self, m):
        print(f"  [{self.owner} device] got {m}")

    def duplicate(self, m):
        print(f"  [{self.owner} device] duplicate #{m.seq} dropped, acked again")

    def on_receipt(self, m, by, now):
        print(f"  [{self.owner} device] {m.id} {now.name} by {by} (overall {m.status().name})")


class CountingClient(Device):
    def __init__(self, server, owner):
        super().__init__(server, owner)
        self.seen, self.pushes, self._lock = set(), 0, threading.Lock()

    def arrives(self, m):
        with self._lock:
            self.pushes += 1
        return True

    def store(self, m):
        with self._lock:
            self.seen.add(m.id)


class LossyClient(ConsoleClient):
    """Loses pushes (drop_push) or acks (drop_ack) on purpose."""

    def __init__(self, server, owner, drop_push, drop_ack):
        super().__init__(server, owner)
        self.drop_push, self.drop_ack = drop_push, drop_ack

    def arrives(self, m):
        return not self.drop_push

    def send_ack(self, m):
        return not self.drop_ack


def expect_error(fn, contains):
    try:
        fn()
    except ChatError as e:
        print("  rejected:", e)
        check(contains in str(e), e)
        return
    raise AssertionError("expected error: " + contains)


def main():
    s = ChatServer()
    for uid in ("asha", "ben", "chen", "dev"):
        s.register(uid, uid.title())

    print("== 1:1 chat, Ben offline")
    s.connect("asha", ConsoleClient(s, "asha"))
    d = s.direct("asha", "ben")
    check(s.direct("ben", "asha") is d)
    hi = s.send("asha", d.id, "hi Ben, free for a review?")
    print(f"  {hi.id} status {hi.status().name}")
    check(hi.status() is Receipt.SENT)

    print("== Ben connects")
    s.connect("ben", ConsoleClient(s, "ben"))
    check(hi.status() is Receipt.DELIVERED)
    print("  ben unread", s.unread("ben", d.id))
    s.mark_read("ben", d.id, hi.seq)
    check(hi.status() is Receipt.READ)
    check(s.mark_read("ben", d.id, hi.seq) == 0)

    print("== group chat, Chen offline")
    g = s.group("asha", "kernel-hackers", "ben", "chen")
    patch = s.send("ben", g.id, "patch v2 is up")
    print(f"  {patch.id} overall {patch.status().name} (chen has not received it)")
    check(patch.status() is Receipt.SENT)
    s.connect("chen", ConsoleClient(s, "chen"))
    s.mark_read("asha", g.id, patch.seq)
    print("  after asha read: overall", patch.status().name)
    check(patch.status() is Receipt.DELIVERED)
    s.mark_read("chen", g.id, patch.seq)
    check(patch.status() is Receipt.READ)

    print("== Dev joins late, sees only new messages")
    s.add_member("asha", g.id, "dev")
    s.send("asha", g.id, "welcome Dev")
    dev_sees = s.history("dev", g.id)
    print("  dev history", dev_sees)
    check(len(dev_sees) == 1)

    print("== rules")
    expect_error(lambda: s.add_member("ben", g.id, "dev"), "not an admin")
    expect_error(lambda: s.add_member("asha", d.id, "chen"), "exactly two")
    expect_error(lambda: s.send("dev", d.id, "hello?"), "not in")
    expect_error(lambda: s.send("asha", g.id, "   "), "empty")
    s.remove_member("asha", g.id, "ben")
    expect_error(lambda: s.send("ben", g.id, "still here?"), "not in")
    s.remove_member("asha", g.id, "asha")       # the only admin leaves
    s.add_member("chen", g.id, "ben")           # chen, the oldest member, is admin now
    check(g.is_member("ben"))

    print("== paging (Ben offline again)")
    s.disconnect("ben")
    for i in range(1, 6):
        s.send("asha", d.id, f"note {i}")
    print("  ben unread", s.unread("ben", d.id))
    check(s.unread("ben", d.id) == 5)
    page = s.history("ben", d.id, limit=2)
    older = s.history("ben", d.id, before_seq=page[0].seq, limit=2)
    print("  newest", page, "older", older)
    check(page[1].seq == d.last_seq() and older[1].seq == page[0].seq - 1)

    print("== lost push and lost ack: DELIVERED only after a device ack")
    s.mark_read("ben", d.id, d.last_seq())
    s.connect("ben", LossyClient(s, "ben", drop_push=True, drop_ack=False))
    lost = s.send("asha", d.id, "did this arrive?")
    print("  push lost:", lost.id, "is", lost.receipt_of("ben").name)
    check(lost.receipt_of("ben") is Receipt.SENT)
    s.disconnect("ben")
    dev = LossyClient(s, "ben", drop_push=False, drop_ack=True)
    s.connect("ben", dev)
    check(lost.receipt_of("ben") is Receipt.SENT)
    s.disconnect("ben")
    dev.drop_ack = False
    s.connect("ben", dev)
    print("  after reconnect + ack:", lost.receipt_of("ben").name)
    check(lost.receipt_of("ben") is Receipt.DELIVERED and not s.ack("ben", d.id, lost.seq))

    print("== 4 senders x 250 messages into one group")
    s2 = ChatServer()
    for uid in ("u1", "u2", "u3", "u4", "watcher"):
        s2.register(uid, uid)
    big = s2.group("u1", "load", "u2", "u3", "u4", "watcher")
    watcher = CountingClient(s2, "watcher")
    s2.connect("watcher", watcher)

    def spam(sender):
        for i in range(250):
            s2.send(sender, big.id, f"{sender}-{i}")

    threads = [threading.Thread(target=spam, args=(f"u{t}",)) for t in range(1, 5)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    msgs = s2.history("watcher", big.id, limit=2000)
    seqs = {m.seq for m in msgs}
    print(f"  messages {len(msgs)}, distinct seqs {len(seqs)}, max seq {big.last_seq()}, watcher pushes {watcher.pushes}")
    check(len(msgs) == 1000 and len(seqs) == 1000 and big.last_seq() == 1000)
    check(watcher.pushes == 1000 and len(watcher.seen) == 1000)
    print("All checks passed")


if __name__ == "__main__":
    main()

import threading
import unittest

from chat import ChatError, ChatServer, Device, Receipt


class Sink(Device):
    def __init__(self, server, owner, lose_pushes=False):
        super().__init__(server, owner)
        self.got, self.receipts, self.lose_pushes = [], [], lose_pushes

    def arrives(self, m):
        return not self.lose_pushes

    def store(self, m):
        self.got.append(m.seq)

    def on_receipt(self, m, by, now):
        self.receipts.append((m.id, by, now))


def server(*users):
    s = ChatServer()
    for u in users:
        s.register(u, u)
    return s


class ChatTests(unittest.TestCase):
    def test_receipt_state_only_moves_forward(self):
        self.assertIs(Receipt.READ.on_delivered(), Receipt.READ)
        self.assertIs(Receipt.SENT.on_read(), Receipt.READ)
        self.assertIs(Receipt.DELIVERED.on_delivered(), Receipt.DELIVERED)

    def test_offline_messages_delivered_on_connect_in_order(self):
        s = server("a", "b")
        d = s.direct("a", "b")
        for i in range(3):
            s.send("a", d.id, f"x{i}")
        sink = Sink(s, "b")
        s.connect("b", sink)
        self.assertEqual(sink.got, [1, 2, 3])

    def test_sender_gets_delivered_then_read(self):
        s = server("a", "b")
        a, b = Sink(s, "a"), Sink(s, "b")
        s.connect("a", a)
        s.connect("b", b)
        d = s.direct("a", "b")
        m = s.send("a", d.id, "hi")
        s.mark_read("b", d.id, m.seq)
        self.assertEqual([r[2] for r in a.receipts], [Receipt.DELIVERED, Receipt.READ])

    def test_group_status_is_lowest_receipt(self):
        s = server("a", "b", "c")
        g = s.group("a", "g", "b", "c")
        s.connect("b", Sink(s, "b"))
        m = s.send("a", g.id, "hello")
        self.assertIs(m.status(), Receipt.SENT)          # c is offline
        s.connect("c", Sink(s, "c"))
        self.assertIs(m.status(), Receipt.DELIVERED)

    def test_removed_member_does_not_block_ticks(self):
        s = server("a", "b", "c")
        g = s.group("a", "g", "b", "c")
        s.connect("b", Sink(s, "b"))
        m = s.send("a", g.id, "hello")
        s.remove_member("a", g.id, "c")
        self.assertIs(m.status(), Receipt.DELIVERED)

    def test_delivered_needs_an_ack_and_lost_push_is_resent(self):
        s = server("a", "b")
        d = s.direct("a", "b")
        s.connect("b", Sink(s, "b", lose_pushes=True))
        m = s.send("a", d.id, "hi")
        self.assertIs(m.receipt_of("b"), Receipt.SENT)    # pushed, never acked: not delivered
        s.disconnect("b")
        sink = Sink(s, "b")
        s.connect("b", sink)                               # reconnect re-sends what was never acked
        self.assertEqual(sink.got, [1])
        self.assertIs(m.receipt_of("b"), Receipt.DELIVERED)
        self.assertFalse(s.ack("b", d.id, m.seq))          # duplicate ack changes nothing

    def test_device_dedupes_a_resent_message(self):
        s = server("a", "b")
        d = s.direct("a", "b")
        s.send("a", d.id, "hi")
        sink = Sink(s, "b")
        sink.send_ack = lambda m: False                    # the ack is lost on the way back
        s.connect("b", sink)
        s.disconnect("b")
        del sink.send_ack
        s.connect("b", sink)                               # same message pushed again
        self.assertEqual(sink.got, [1])                    # stored once
        self.assertIs(d.messages[0].receipt_of("b"), Receipt.DELIVERED)

    def test_late_joiner_sees_no_old_history(self):
        s = server("a", "b")
        g = s.group("a", "g")
        s.send("a", g.id, "secret")
        s.add_member("a", g.id, "b")
        self.assertEqual(s.history("b", g.id), [])

    def test_only_admin_adds_and_direct_is_closed(self):
        s = server("a", "b", "c")
        g = s.group("a", "g", "b")
        with self.assertRaises(ChatError):
            s.add_member("b", g.id, "c")
        with self.assertRaises(ChatError):
            s.add_member("a", s.direct("a", "b").id, "c")

// Chat application in C++14: same design as the Java version.
// Locking (three levels, always taken in this order, never the reverse):
//   ChatServer::mu_  guards users_, chats_, chatsOf_, online_   (Java: ConcurrentHashMap)
//   Chat::mu         guards members, messages, nextSeq_          (Java: synchronized on the chat)
//   Message::mu      guards receipts, pushedIn                   (Java: synchronized methods)
// Every call out to a device (onMessage, onReceipt) happens with NO lock held: a device acks from inside
// onMessage(), which takes the chat lock again. Id counters are std::atomic<long>.
// Mutex is std::mutex when the toolchain has threads (Linux/WSL g++, MSVC, MinGW-w64 posix). The MinGW.org
// g++ 6.3 used for run-cpp.txt has no std::mutex, so there Mutex is a no-op with the same lock()/unlock():
// the locking code is the same in both builds. Linux build (run-cpp-linux.txt): g++ -std=c++14 -pthread main.cpp
#include <algorithm>
#include <atomic>
#include <iostream>
#include <limits>
#include <map>
#include <memory>
#include <mutex>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>
#include <cstdio>
#include <cstdlib>
#if defined(_GLIBCXX_HAS_GTHREADS) || !defined(__GLIBCXX__)
using Mutex = std::mutex;
#else
struct Mutex { void lock() {} void unlock() {} };   // MinGW.org g++ 6.3: no std::mutex, single-threaded demo
#endif
using Guard = std::lock_guard<Mutex>;
// CHECK, not assert: assert() and any call inside it vanish under -DNDEBUG; CHECK always runs.
#define CHECK(x) do { if (!(x)) { std::fprintf(stderr, "CHECK failed (line %d): %s\n", __LINE__, #x); std::exit(1); } } while (0)

struct ChatError : std::runtime_error {
    explicit ChatError(const std::string& m) : std::runtime_error(m) {}
};

// Per-recipient receipt. The transition table plays the role of the State pattern; receipts move forward only.
enum class Receipt { SENT = 0, DELIVERED = 1, READ = 2 };
const char* name(Receipt r) { return r == Receipt::SENT ? "SENT" : r == Receipt::DELIVERED ? "DELIVERED" : "READ"; }
Receipt onDelivered(Receipt r) { return r == Receipt::SENT ? Receipt::DELIVERED : r; }
Receipt onRead(Receipt) { return Receipt::READ; }

struct Message {
    std::string id, chatId, sender, text;
    long seq;
    std::map<std::string, Receipt> receipts;   // recipient -> state
    std::map<std::string, long> pushedIn;      // recipient -> connection that already got a push
    mutable Mutex mu;                          // guards receipts and pushedIn

    // Avoids a second push on the same connection; it is NOT the DELIVERED tick.
    bool claimPush(const std::string& user, long connection) {
        Guard g(mu);
        auto it = receipts.find(user);
        if (it == receipts.end() || it->second != Receipt::SENT) return false;   // already acked
        auto p = pushedIn.find(user);
        if (p != pushedIn.end() && p->second == connection) return false;
        pushedIn[user] = connection;
        return true;
    }

    bool apply(const std::string& user, bool read) {
        Guard g(mu);
        auto it = receipts.find(user);
        if (it == receipts.end()) return false;
        Receipt next = read ? onRead(it->second) : onDelivered(it->second);
        if (next == it->second) return false;
        it->second = next;
        return true;
    }
    bool has(const std::string& u) const { Guard g(mu); return receipts.count(u) > 0; }
    Receipt receiptOf(const std::string& u) const { Guard g(mu); return receipts.at(u); }
    bool pendingFor(const std::string& u) const {   // a recipient that has not acked yet
        Guard g(mu);
        auto it = receipts.find(u);
        return it != receipts.end() && it->second == Receipt::SENT;
    }
    void dropUnread(const std::string& u) {         // a removed member loses unread receipts
        Guard g(mu);
        auto it = receipts.find(u);
        if (it != receipts.end() && it->second != Receipt::READ) receipts.erase(it);
    }
    std::vector<std::string> recipients() const {
        Guard g(mu);
        std::vector<std::string> v;
        for (const auto& kv : receipts) v.push_back(kv.first);
        return v;
    }
    Receipt status() const {                    // the lowest receipt over all recipients
        Guard g(mu);
        if (receipts.empty()) return Receipt::SENT;
        Receipt m = Receipt::READ;
        for (const auto& kv : receipts) if (kv.second < m) m = kv.second;
        return m;
    }
    std::string str() const { return "#" + std::to_string(seq) + " " + sender + ": " + text; }
};
using MsgPtr = std::shared_ptr<Message>;
const long LATEST = std::numeric_limits<long>::max();   // "no upper bound" for history paging

class Chat {
public:
    explicit Chat(std::string id) : id(std::move(id)) {}
    virtual ~Chat() = default;
    virtual std::string title() const = 0;
    virtual void checkCanChangeMembers(const std::string& by, const std::string& target, bool adding) const = 0;

    const std::string id;
    mutable Mutex mu;                                    // the methods below expect the caller to hold it
    std::vector<std::pair<std::string, long>> members;   // join order kept; second = first visible seq
    std::vector<MsgPtr> messages;

    bool isMember(const std::string& u) const {
        for (const auto& m : members) if (m.first == u) return true;
        return false;
    }
    long lastSeq() const { return nextSeq_ - 1; }
    void join(const std::string& u) { if (!isMember(u)) members.emplace_back(u, nextSeq_); }
    virtual void leave(const std::string& u) {
        members.erase(std::remove_if(members.begin(), members.end(),
                                     [&](const std::pair<std::string, long>& m) { return m.first == u; }),
                      members.end());
    }
    MsgPtr append(const std::string& mid, const std::string& sender, const std::string& text) {
        auto m = std::make_shared<Message>();
        m->id = mid; m->chatId = id; m->sender = sender; m->text = text; m->seq = nextSeq_++;
        for (const auto& mem : members) if (mem.first != sender) m->receipts[mem.first] = Receipt::SENT;
        messages.push_back(m);
        return m;
    }
    std::vector<MsgPtr> visibleTo(const std::string& u) const {
        long from = 0;
        for (const auto& m : members) if (m.first == u) from = m.second;
        std::vector<MsgPtr> out;
        for (const auto& m : messages) if (m->seq >= from) out.push_back(m);
        return out;
    }

private:
    long nextSeq_ = 1;
};

class DirectChat : public Chat {
public:
    DirectChat(const std::string& id, const std::string& a, const std::string& b) : Chat(id) { join(a); join(b); }
    std::string title() const override { return "direct " + members[0].first + "+" + members[1].first; }
    void checkCanChangeMembers(const std::string&, const std::string&, bool) const override {
        throw ChatError("a direct chat has exactly two members");
    }
};

class GroupChat : public Chat {
public:
    GroupChat(const std::string& id, std::string n, const std::string& creator) : Chat(id), name(std::move(n)) {
        join(creator);
        admins.insert(creator);
    }
    std::string title() const override { return "group '" + name + "'"; }
    void checkCanChangeMembers(const std::string& by, const std::string& target, bool adding) const override {
        bool selfLeave = !adding && by == target;
        if (!selfLeave && !admins.count(by)) throw ChatError(by + " is not an admin of " + name);
        if (adding && members.size() >= 256) throw ChatError("group is full");
    }
    void leave(const std::string& u) override {
        Chat::leave(u);
        admins.erase(u);
        if (admins.empty() && !members.empty()) admins.insert(members.front().first);   // oldest member
    }
    const std::string name;

private:
    std::set<std::string> admins;
};

// Observer: a connected device.
struct ChatClient {
    virtual ~ChatClient() = default;
    virtual void onMessage(const Message& m) = 0;
    virtual void onReceipt(const Message& m, const std::string& by, Receipt now) = 0;
};

// Mediator: users never reference each other.
class ChatServer {
public:
    void registerUser(const std::string& id) {
        Guard g(mu_);
        if (users_.count(id)) throw ChatError("user " + id + " exists");
        users_.insert(id);
        chatsOf_[id];
    }
    Chat& chat(const std::string& id) {                      // chats are never erased: the reference stays valid
        Guard g(mu_);
        auto it = chats_.find(id);
        if (it == chats_.end()) throw ChatError("no chat " + id);
        return *it->second;
    }
    DirectChat& direct(const std::string& a, const std::string& b) {
        needUser(a); needUser(b);
        if (a == b) throw ChatError("cannot chat with yourself");
        std::string key = a < b ? "d:" + a + ":" + b : "d:" + b + ":" + a;
        Guard g(mu_);
        if (!chats_.count(key)) chats_[key] = std::make_unique<DirectChat>(key, a, b);
        chatsOf_[a].insert(key); chatsOf_[b].insert(key);
        return static_cast<DirectChat&>(*chats_[key]);
    }
    GroupChat& group(const std::string& creator, const std::string& name, const std::vector<std::string>& others) {
        needUser(creator);
        std::string id = "g" + std::to_string(++groupIds_);
        GroupChat* gc;
        {
            Guard g(mu_);
            auto made = std::make_unique<GroupChat>(id, name, creator);
            gc = made.get();
            chats_[id] = std::move(made);
            chatsOf_[creator].insert(id);
        }
        for (const auto& o : others) addMember(creator, id, o);   // takes the locks itself
        return *gc;
    }
    void addMember(const std::string& by, const std::string& cid, const std::string& user) {
        needUser(user);
        Chat& c = chat(cid);
        {
            Guard g(c.mu);
            if (!c.isMember(by)) throw ChatError(by + " is not in " + c.title());
            c.checkCanChangeMembers(by, user, true);
            if (c.isMember(user)) return;
            c.join(user);
        }
        Guard g(mu_);
        chatsOf_[user].insert(cid);
    }
    void removeMember(const std::string& by, const std::string& cid, const std::string& user) {
        Chat& c = chat(cid);
        {
            Guard g(c.mu);
            if (!c.isMember(by) || !c.isMember(user)) throw ChatError("not a member");
            c.checkCanChangeMembers(by, user, false);
            c.leave(user);
            for (auto& m : c.messages) m->dropUnread(user);     // chat lock, then message lock
        }
        Guard g(mu_);
        chatsOf_[user].erase(cid);
    }
    // Re-pushes every message this user has not ACKNOWLEDGED (still SENT): at-least-once.
    void connect(const std::string& uid, ChatClient* client) {
        needUser(uid);
        Conn conn;
        std::set<std::string> cids;
        {
            Guard g(mu_);
            online_[uid] = std::make_pair(++connIds_, client);
            conn = online_[uid];
            cids = chatsOf_[uid];
        }
        for (const auto& cid : cids) {
            Chat& c = chat(cid);
            std::vector<MsgPtr> pending;
            {
                Guard g(c.mu);                                   // collect under the chat lock
                if (!c.isMember(uid)) continue;
                for (const auto& m : c.visibleTo(uid))
                    if (m->pendingFor(uid)) pending.push_back(m);
            }
            for (const auto& m : pending) deliver(uid, conn, *m);     // push with no lock held
        }
    }
    void disconnect(const std::string& uid) { Guard g(mu_); online_.erase(uid); }

    MsgPtr send(const std::string& sender, const std::string& cid, const std::string& text) {
        if (text.find_first_not_of(' ') == std::string::npos) throw ChatError("empty message");
        Chat& c = chat(cid);
        MsgPtr m;
        {
            Guard g(c.mu);                                       // member check + seq++ + append: one step
            if (!c.isMember(sender)) throw ChatError(sender + " is not in " + c.title());
            m = c.append("m" + std::to_string(++msgIds_), sender, text);
        }
        for (const auto& r : m->recipients()) {                 // fan-out after unlock: O(members)
            Conn conn;
            if (onlineConn(r, &conn)) deliver(r, conn, *m);
        }
        return m;
    }
    // The device stored message seq: only this moves SENT -> DELIVERED. Idempotent.
    bool ack(const std::string& uid, const std::string& cid, long seq) {
        Chat& c = chat(cid);
        MsgPtr m;
        {
            Guard g(c.mu);
            if (!c.isMember(uid) || seq < 1 || seq > static_cast<long>(c.messages.size())) return false;
            m = c.messages[seq - 1];                            // seqs are 1..n with no gaps
            if (!m->apply(uid, false)) return false;
        }
        tellSender(*m, uid);                                    // after unlock
        return true;
    }
    int markRead(const std::string& uid, const std::string& cid, long upto) {
        Chat& c = chat(cid);
        std::vector<MsgPtr> changed;
        {
            Guard g(c.mu);
            if (!c.isMember(uid)) throw ChatError(uid + " is not in " + c.title());
            for (const auto& m : c.visibleTo(uid))
                if (m->seq <= upto && m->sender != uid && m->apply(uid, true)) changed.push_back(m);
        }
        for (const auto& m : changed) tellSender(*m, uid);
        return static_cast<int>(changed.size());
    }
    int unread(const std::string& uid, const std::string& cid) {
        Chat& c = chat(cid);
        Guard g(c.mu);
        if (!c.isMember(uid)) throw ChatError(uid + " is not in " + c.title());
        int n = 0;
        for (const auto& m : c.visibleTo(uid))
            if (m->sender != uid && m->has(uid) && m->receiptOf(uid) != Receipt::READ) n++;
        return n;
    }
    std::vector<MsgPtr> history(const std::string& uid, const std::string& cid, long before, size_t limit) {
        Chat& c = chat(cid);
        Guard g(c.mu);
        if (!c.isMember(uid)) throw ChatError(uid + " is not in " + c.title());
        std::vector<MsgPtr> vis = c.visibleTo(uid), out;
        for (auto it = vis.rbegin(); it != vis.rend() && out.size() < limit; ++it)
            if ((*it)->seq < before) out.push_back(*it);
        std::reverse(out.begin(), out.end());
        return out;
    }

private:
    void needUser(const std::string& id) const { Guard g(mu_); if (!users_.count(id)) throw ChatError("no user " + id); }
    bool onlineConn(const std::string& uid, std::pair<long, ChatClient*>* out) const {
        Guard g(mu_);
        auto it = online_.find(uid);
        if (it == online_.end()) return false;
        *out = it->second;
        return true;
    }
    using Conn = std::pair<long, ChatClient*>;                 // connection id, device
    // Push only: the receipt stays SENT until the device acks, so a lost push is re-sent on reconnect.
    void deliver(const std::string& uid, const Conn& conn, Message& m) {
        if (m.claimPush(uid, conn.first)) conn.second->onMessage(m);
    }
    void tellSender(const Message& m, const std::string& by) {
        Conn conn;
        if (onlineConn(m.sender, &conn)) conn.second->onReceipt(m, by, m.receiptOf(by));   // no lock held
    }
    std::set<std::string> users_;
    std::map<std::string, std::unique_ptr<Chat>> chats_;
    std::map<std::string, std::set<std::string>> chatsOf_;
    std::map<std::string, Conn> online_;
    std::atomic<long> msgIds_{0}, groupIds_{0}, connIds_{0};
    mutable Mutex mu_;                                          // guards the four maps above
};

// A device stores a message, drops copies it already has (dedupe by chat + seq), then acks.
struct Device : ChatClient {
    Device(ChatServer& s, std::string o) : server(s), owner(std::move(o)) {}
    void onMessage(const Message& m) override {
        if (dropPush) return;                                   // lost on the wire
        bool fresh = have.insert(m.chatId + "#" + std::to_string(m.seq)).second;
        if (fresh) store(m); else duplicate(m);
        if (!dropAck) server.ack(owner, m.chatId, m.seq);       // ack a duplicate too
    }
    void onReceipt(const Message&, const std::string&, Receipt) override {}
    virtual void store(const Message&) {}
    virtual void duplicate(const Message&) {}
    ChatServer& server;
    std::string owner;
    std::set<std::string> have;
    bool dropPush = false, dropAck = false;
};

struct ConsoleClient : Device {
    ConsoleClient(ChatServer& s, std::string o) : Device(s, std::move(o)) {}
    void store(const Message& m) override { std::cout << "  [" << owner << " device] got " << m.str() << "\n"; }
    void duplicate(const Message& m) override {
        std::cout << "  [" << owner << " device] duplicate #" << m.seq << " dropped, acked again\n";
    }
    void onReceipt(const Message& m, const std::string& by, Receipt now) override {
        std::cout << "  [" << owner << " device] " << m.id << " " << name(now) << " by " << by
                  << " (overall " << name(m.status()) << ")\n";
    }
};

struct CountingClient : Device {
    CountingClient(ChatServer& s, std::string o) : Device(s, std::move(o)) {}
    std::set<std::string> seen;
    long pushes = 0;
    void store(const Message& m) override { pushes++; seen.insert(m.id); }
};

template <class F>
void expectError(F f, const std::string& contains) {
    try { f(); } catch (const ChatError& e) {
        std::cout << "  rejected: " << e.what() << "\n";
        CHECK(std::string(e.what()).find(contains) != std::string::npos);
        return;
    }
    throw std::logic_error("expected error: " + contains);
}

std::string list(const std::vector<MsgPtr>& v) {
    std::string s = "[";
    for (size_t i = 0; i < v.size(); i++) s += (i ? ", " : "") + v[i]->str();
    return s + "]";
}
