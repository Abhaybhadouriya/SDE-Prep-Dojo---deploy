import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicLong;

// ---------- errors ----------
class ChatException extends RuntimeException {
    ChatException(String msg) { super(msg); }
}

// ---------- per-recipient receipt: a State machine written as an enum ----------
// Each constant decides what an event does to it. Receipts only move forward.
enum Receipt {
    SENT      { Receipt onDelivered() { return DELIVERED; } Receipt onRead() { return READ; } },
    DELIVERED { Receipt onDelivered() { return this; }      Receipt onRead() { return READ; } },
    READ      { Receipt onDelivered() { return this; }      Receipt onRead() { return this; } };

    abstract Receipt onDelivered();
    abstract Receipt onRead();
}

// ---------- model ----------
final class User {
    final String id, name;
    User(String id, String name) { this.id = id; this.name = name; }
}

final class Message {
    final String id, chatId, senderId, text;
    final long seq;                                              // order inside the chat
    private final Map<String, Receipt> receipts = new LinkedHashMap<>();  // recipient -> state
    private final Map<String, Long> pushedIn = new HashMap<>();             // recipient -> connection that got a push

    Message(String id, String chatId, String senderId, long seq, String text, Collection<String> recipients) {
        this.id = id; this.chatId = chatId; this.senderId = senderId; this.seq = seq; this.text = text;
        for (String r : recipients) receipts.put(r, Receipt.SENT);
    }

    /** Applies one event for one recipient; true only if the state really changed. */
    synchronized boolean apply(String userId, boolean read) {
        Receipt old = receipts.get(userId);
        if (old == null) return false;
        Receipt next = read ? old.onRead() : old.onDelivered();
        if (next == old) return false;
        receipts.put(userId, next);
        return true;
    }

    synchronized Receipt receiptOf(String userId) { return receipts.get(userId); }

    /** CAS that only avoids a duplicate push inside one connection; it is NOT the DELIVERED tick. */
    synchronized boolean claimPush(String userId, long connection) {
        if (receipts.get(userId) != Receipt.SENT) return false;        // already acknowledged
        Long prev = pushedIn.put(userId, connection);
        return prev == null || prev != connection;
    }

    /** A removed member must not hold the group's double tick hostage. */
    synchronized void forget(String userId) { receipts.remove(userId); }

    /** What the sender sees: the lowest state over all recipients (WhatsApp-style ticks). */
    synchronized Receipt status() {
        if (receipts.isEmpty()) return Receipt.SENT;
        Receipt min = Receipt.READ;
        for (Receipt r : receipts.values()) if (r.ordinal() < min.ordinal()) min = r;
        return min;
    }

    synchronized List<String> recipients() { return new ArrayList<>(receipts.keySet()); }

    @Override public String toString() { return "#" + seq + " " + senderId + ": " + text; }
}

abstract class Chat {
    final String id;
    final Object lock = new Object();
    protected final Map<String, Long> members = new LinkedHashMap<>();  // userId -> first seq they may see
    protected final List<Message> messages = new ArrayList<>();
    private long nextSeq = 1;

    Chat(String id) { this.id = id; }

    abstract String title();
    abstract void checkCanChangeMembers(String byUser, String target, boolean adding);

    boolean isMember(String userId) { return members.containsKey(userId); }
    long lastSeq() { return nextSeq - 1; }
    void join(String userId) { members.putIfAbsent(userId, nextSeq); }   // sees only what comes next
    void leave(String userId) { members.remove(userId); }

    /** Caller holds lock: one sequence number per message, no gaps, no duplicates. */
    Message append(String msgId, String senderId, String text) {
        List<String> to = new ArrayList<>();
        for (String m : members.keySet()) if (!m.equals(senderId)) to.add(m);
        Message msg = new Message(msgId, id, senderId, nextSeq++, text, to);
        messages.add(msg);
        return msg;
    }

    /** Caller holds lock: seq numbers are 1..lastSeq with no gaps, so this is an index. */
    Message bySeq(long seq) { return seq >= 1 && seq <= messages.size() ? messages.get((int) (seq - 1)) : null; }

    /** Caller holds lock: messages this member may see (from the seq they joined at). */
    List<Message> visibleTo(String userId) {
        long from = members.get(userId);
        List<Message> out = new ArrayList<>();
        for (Message m : messages) if (m.seq >= from) out.add(m);
        return out;
    }
}

final class DirectChat extends Chat {
    DirectChat(String id, String a, String b) { super(id); join(a); join(b); }
    String title() { return "direct " + String.join("+", members.keySet()); }
    void checkCanChangeMembers(String byUser, String target, boolean adding) {
        throw new ChatException("a direct chat has exactly two members");
    }
}

final class GroupChat extends Chat {
    final String name;
    private final Set<String> admins = new HashSet<>();
    static final int MAX_MEMBERS = 256;

    GroupChat(String id, String name, String creator) { super(id); this.name = name; join(creator); admins.add(creator); }
    String title() { return "group '" + name + "'"; }

    void checkCanChangeMembers(String byUser, String target, boolean adding) {
        boolean selfLeave = !adding && byUser.equals(target);
        if (!selfLeave && !admins.contains(byUser)) throw new ChatException(byUser + " is not an admin of " + name);
        if (adding && members.size() >= MAX_MEMBERS) throw new ChatException("group is full");
    }

    @Override void leave(String userId) {
        super.leave(userId);
        admins.remove(userId);
        if (admins.isEmpty() && !members.isEmpty()) admins.add(members.keySet().iterator().next()); // oldest member
    }
}

// ---------- Observer: a connected device ----------
interface ChatClient {
    void onMessage(Message m);
    void onReceipt(Message m, String byUser, Receipt now);
}

// ---------- Mediator: users never talk to each other, only through the server ----------
final class ChatServer {
    private final Map<String, User> users = new ConcurrentHashMap<>();
    private final Map<String, Chat> chats = new ConcurrentHashMap<>();
    private final Map<String, Set<String>> chatsOf = new ConcurrentHashMap<>();
    private static final class Connection {
        final ChatClient client; final long id;
        Connection(ChatClient client, long id) { this.client = client; this.id = id; }
    }
    private final Map<String, Connection> online = new ConcurrentHashMap<>();
    private final AtomicLong msgIds = new AtomicLong(), groupIds = new AtomicLong(), connIds = new AtomicLong();

    void register(User u) {
        if (users.putIfAbsent(u.id, u) != null) throw new ChatException("user " + u.id + " exists");
        chatsOf.put(u.id, ConcurrentHashMap.newKeySet());
    }

    private void needUser(String id) { if (!users.containsKey(id)) throw new ChatException("no user " + id); }

    Chat chat(String chatId) {
        Chat c = chats.get(chatId);
        if (c == null) throw new ChatException("no chat " + chatId);
        return c;
    }

    /** Idempotent: the same pair always gets the same chat. */
    DirectChat direct(String a, String b) {
        needUser(a); needUser(b);
        if (a.equals(b)) throw new ChatException("cannot chat with yourself");
        String key = a.compareTo(b) < 0 ? "d:" + a + ":" + b : "d:" + b + ":" + a;
        DirectChat c = (DirectChat) chats.computeIfAbsent(key, k -> new DirectChat(k, a, b));
        chatsOf.get(a).add(key); chatsOf.get(b).add(key);
        return c;
    }

    GroupChat group(String creator, String name, String... others) {
        needUser(creator);
        GroupChat g = new GroupChat("g" + groupIds.incrementAndGet(), name, creator);
        chats.put(g.id, g);
        chatsOf.get(creator).add(g.id);
        for (String o : others) addMember(creator, g.id, o);
        return g;
    }

    void addMember(String by, String chatId, String user) {
        needUser(user);
        Chat c = chat(chatId);
        synchronized (c.lock) {
            if (!c.isMember(by)) throw new ChatException(by + " is not in " + c.title());
            c.checkCanChangeMembers(by, user, true);   // permission first, even for a no-op
            if (c.isMember(user)) return;
            c.join(user);
        }
        chatsOf.get(user).add(chatId);
    }

    void removeMember(String by, String chatId, String user) {
        Chat c = chat(chatId);
        synchronized (c.lock) {
            if (!c.isMember(by) || !c.isMember(user)) throw new ChatException("not a member");
            c.checkCanChangeMembers(by, user, false);
            c.leave(user);
            for (Message m : c.messages) if (m.receiptOf(user) != Receipt.READ) m.forget(user);
        }
        chatsOf.get(user).remove(chatId);
    }

    /** A device comes online: every message this user has not ACKNOWLEDGED (still SENT) is pushed again,
     *  in seq order per chat. A push that was lost last time is therefore re-sent: at-least-once. */
    void connect(String userId, ChatClient client) {
        needUser(userId);
        Connection conn = new Connection(client, connIds.incrementAndGet());
        online.put(userId, conn);
        for (String chatId : chatsOf.get(userId)) {
            Chat c = chats.get(chatId);
            List<Message> pending = new ArrayList<>();
            synchronized (c.lock) {
                if (!c.isMember(userId)) continue;
                for (Message m : c.visibleTo(userId)) if (m.receiptOf(userId) == Receipt.SENT) pending.add(m);
            }
            for (Message m : pending) deliver(userId, conn, m);     // outside the lock
        }
    }

    void disconnect(String userId) { online.remove(userId); }

    Message send(String senderId, String chatId, String text) {
        if (text == null || text.trim().isEmpty()) throw new ChatException("empty message");
        Chat c = chat(chatId);
        Message m;
        synchronized (c.lock) {
            if (!c.isMember(senderId)) throw new ChatException(senderId + " is not in " + c.title());
            m = c.append("m" + msgIds.incrementAndGet(), senderId, text.trim());
        }
        for (String r : m.recipients()) {          // fan-out after unlock
            Connection cn = online.get(r);
            if (cn != null) deliver(r, cn, m);
        }
        return m;
    }

    /** Push; the state does NOT change here. The CAS only stops a second push on the same connection
     *  (send's fan-out racing connect's catch-up). If the push is lost, the receipt stays SENT. */
    private void deliver(String userId, Connection conn, Message m) {
        if (!m.claimPush(userId, conn.id)) return;
        conn.client.onMessage(m);
    }

    /** The device confirms it stored message seq of chatId: only this moves SENT -> DELIVERED.
     *  Idempotent, so a duplicate ack (or an ack for a re-sent copy) changes nothing. */
    boolean ack(String userId, String chatId, long seq) {
        Chat c = chat(chatId);
        Message m;
        synchronized (c.lock) {
            if (!c.isMember(userId)) return false;
            m = c.bySeq(seq);
            if (m == null || !m.apply(userId, false)) return false;
        }
        tellSender(m, userId);
        return true;
    }

    private void tellSender(Message m, String byUser) {
        Connection s = online.get(m.senderId);
        if (s != null) s.client.onReceipt(m, byUser, m.receiptOf(byUser));
    }

    /** Marks everything up to and including uptoSeq as read; returns how many changed. */
    int markRead(String userId, String chatId, long uptoSeq) {
        Chat c = chat(chatId);
        List<Message> changed = new ArrayList<>();
        synchronized (c.lock) {
            if (!c.isMember(userId)) throw new ChatException(userId + " is not in " + c.title());
            for (Message m : c.visibleTo(userId))
                if (m.seq <= uptoSeq && !m.senderId.equals(userId) && m.apply(userId, true)) changed.add(m);
        }
        for (Message m : changed) tellSender(m, userId);
        return changed.size();
    }

    int unread(String userId, String chatId) {
        Chat c = chat(chatId);
        synchronized (c.lock) {
            if (!c.isMember(userId)) throw new ChatException(userId + " is not in " + c.title());
            int n = 0;
            for (Message m : c.visibleTo(userId)) if (!m.senderId.equals(userId) && m.receiptOf(userId) != Receipt.READ) n++;
            return n;
        }
    }

    /** Newest page first in time order: up to limit messages with seq < beforeSeq. */
    List<Message> history(String userId, String chatId, long beforeSeq, int limit) {
        Chat c = chat(chatId);
        synchronized (c.lock) {
            if (!c.isMember(userId)) throw new ChatException(userId + " is not in " + c.title());
            List<Message> vis = c.visibleTo(userId), out = new ArrayList<>();
            for (int i = vis.size() - 1; i >= 0 && out.size() < limit; i--) if (vis.get(i).seq < beforeSeq) out.add(vis.get(i));
            Collections.reverse(out);
            return out;
        }
    }
}

// ---------- demo devices ----------
// A device stores a message, then acknowledges it; it drops copies it already has (dedupe by chat + seq).
abstract class Device implements ChatClient {
    protected final ChatServer server; protected final String owner;
    private final Set<String> have = ConcurrentHashMap.newKeySet();   // "chatId#seq" already stored
    Device(ChatServer server, String owner) { this.server = server; this.owner = owner; }
    public final void onMessage(Message m) {
        if (!arrives(m)) return;                                  // lost on the wire
        if (have.add(m.chatId + "#" + m.seq)) store(m); else duplicate(m);
        if (sendAck(m)) server.ack(owner, m.chatId, m.seq);       // ack a duplicate too: the first ack may be the lost one
    }
    protected boolean arrives(Message m) { return true; }
    protected abstract void store(Message m);
    protected void duplicate(Message m) { }
    protected boolean sendAck(Message m) { return true; }
}

final class ConsoleClient extends Device {
    ConsoleClient(ChatServer server, String owner) { super(server, owner); }
    protected void store(Message m) { System.out.println("  [" + owner + " device] got " + m); }
    protected void duplicate(Message m) { System.out.println("  [" + owner + " device] duplicate " + m.id + " dropped, acked again"); }
    public void onReceipt(Message m, String byUser, Receipt now) {
        System.out.println("  [" + owner + " device] " + m.id + " " + now + " by " + byUser + " (overall " + m.status() + ")");
    }
}

final class CountingClient extends Device {
    final Set<String> seen = ConcurrentHashMap.newKeySet();
    final AtomicLong pushes = new AtomicLong();
    CountingClient(ChatServer server, String owner) { super(server, owner); }
    @Override protected boolean arrives(Message m) { pushes.incrementAndGet(); return true; }
    protected void store(Message m) { seen.add(m.id); }
    public void onReceipt(Message m, String byUser, Receipt now) { }
}

// Loses things on purpose: pushes never arrive (dropPush), or acks never reach the server (dropAck).
final class LossyClient extends Device {
    final Set<Long> stored = ConcurrentHashMap.newKeySet();
    volatile boolean dropPush, dropAck;
    LossyClient(ChatServer server, String owner, boolean dropPush, boolean dropAck) {
        super(server, owner); this.dropPush = dropPush; this.dropAck = dropAck;
    }
    @Override protected boolean arrives(Message m) { return !dropPush; }
    protected void store(Message m) { stored.add(m.seq); System.out.println("  [" + owner + " device] stored " + m); }
    @Override protected void duplicate(Message m) { System.out.println("  [" + owner + " device] duplicate #" + m.seq + " dropped, acked again"); }
    @Override protected boolean sendAck(Message m) { return !dropAck; }
    public void onReceipt(Message m, String byUser, Receipt now) { }
}

public class Main {
    static int checks = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }
    static void expectError(Runnable r, String contains) {
        try { r.run(); } catch (ChatException e) {
            System.out.println("  rejected: " + e.getMessage());
            check(e.getMessage().contains(contains), "error mentions " + contains);
            return;
        }
        throw new AssertionError("expected error: " + contains);
    }

    public static void main(String[] args) throws Exception {
        ChatServer s = new ChatServer();
        for (String[] u : new String[][]{{"asha", "Asha"}, {"ben", "Ben"}, {"chen", "Chen"}, {"dev", "Dev"}})
            s.register(new User(u[0], u[1]));

        System.out.println("== 1:1 chat, Ben offline");
        s.connect("asha", new ConsoleClient(s, "asha"));
        DirectChat d = s.direct("asha", "ben");
        check(s.direct("ben", "asha") == d, "direct chat is idempotent");
        Message hi = s.send("asha", d.id, "hi Ben, free for a review?");
        System.out.println("  " + hi.id + " status " + hi.status());
        check(hi.status() == Receipt.SENT, "offline recipient keeps SENT");

        System.out.println("== Ben connects");
        s.connect("ben", new ConsoleClient(s, "ben"));
        check(hi.status() == Receipt.DELIVERED, "delivered on connect (the device acked)");
        System.out.println("  ben unread " + s.unread("ben", d.id));
        s.markRead("ben", d.id, hi.seq);
        check(hi.status() == Receipt.READ, "read after markRead");
        check(s.markRead("ben", d.id, hi.seq) == 0, "markRead is idempotent");

        System.out.println("== group chat, Chen offline");
        GroupChat g = s.group("asha", "kernel-hackers", "ben", "chen");
        Message patch = s.send("ben", g.id, "patch v2 is up");
        System.out.println("  " + patch.id + " overall " + patch.status() + " (chen has not received it)");
        check(patch.status() == Receipt.SENT, "group status is the lowest receipt");
        s.connect("chen", new ConsoleClient(s, "chen"));
        s.markRead("asha", g.id, patch.seq);
        System.out.println("  after asha read: overall " + patch.status());
        check(patch.status() == Receipt.DELIVERED, "one reader is not enough");
        s.markRead("chen", g.id, patch.seq);
        check(patch.status() == Receipt.READ, "all read -> READ");

        System.out.println("== Dev joins late, sees only new messages");
        s.addMember("asha", g.id, "dev");
        s.send("asha", g.id, "welcome Dev");
        List<String> devSees = new ArrayList<>();
        for (Message m : s.history("dev", g.id, Long.MAX_VALUE, 50)) devSees.add(m.toString());
        System.out.println("  dev history " + devSees);
        check(devSees.size() == 1, "no history before joining");

        System.out.println("== rules");
        expectError(() -> s.addMember("ben", g.id, "dev"), "not an admin");
        expectError(() -> s.addMember("asha", d.id, "chen"), "exactly two");
        expectError(() -> s.send("dev", d.id, "hello?"), "not in");
        expectError(() -> s.send("asha", g.id, "   "), "empty");
        s.removeMember("asha", g.id, "ben");
        expectError(() -> s.send("ben", g.id, "still here?"), "not in");
        s.removeMember("asha", g.id, "asha");                 // the only admin leaves
        s.addMember("chen", g.id, "ben");                      // chen, the oldest member, is admin now
        check(g.isMember("ben"), "admin handed over to oldest member");

        System.out.println("== paging (Ben offline again)");
        s.disconnect("ben");
        for (int i = 1; i <= 5; i++) s.send("asha", d.id, "note " + i);
        System.out.println("  ben unread " + s.unread("ben", d.id));
        check(s.unread("ben", d.id) == 5, "five unread notes");
        List<Message> page = s.history("ben", d.id, Long.MAX_VALUE, 2);
        List<Message> older = s.history("ben", d.id, page.get(0).seq, 2);
        System.out.println("  newest " + page + " older " + older);
        check(page.get(1).seq == d.lastSeq() && older.get(1).seq == page.get(0).seq - 1, "paging by seq");

        System.out.println("== lost push and lost ack: DELIVERED only after a device ack");
        s.markRead("ben", d.id, d.lastSeq());                         // ben read the notes on another device
        s.connect("ben", new LossyClient(s, "ben", true, false));     // the socket write is lost
        Message lost = s.send("asha", d.id, "did this arrive?");
        System.out.println("  push lost: " + lost.id + " is " + lost.receiptOf("ben"));
        check(lost.receiptOf("ben") == Receipt.SENT, "a lost push stays SENT");
        s.disconnect("ben");
        LossyClient noAck = new LossyClient(s, "ben", false, true);    // stored, but the ack is lost
        s.connect("ben", noAck);
        check(noAck.stored.contains(lost.seq) && lost.receiptOf("ben") == Receipt.SENT, "re-sent on reconnect, ack lost");
        s.disconnect("ben");
        noAck.dropAck = false;                                         // same device, network healthy again
        s.connect("ben", noAck);
        System.out.println("  after reconnect + ack: " + lost.receiptOf("ben"));
        check(lost.receiptOf("ben") == Receipt.DELIVERED, "at-least-once push + ack = delivered");
        check(!s.ack("ben", d.id, lost.seq), "a second ack changes nothing");

        System.out.println("== 4 senders x 250 messages into one group");
        ChatServer s2 = new ChatServer();
        String[] ids = {"u1", "u2", "u3", "u4", "watcher"};
        for (String id : ids) s2.register(new User(id, id));
        GroupChat big = s2.group("u1", "load", "u2", "u3", "u4", "watcher");
        CountingClient watcher = new CountingClient(s2, "watcher");
        s2.connect("watcher", watcher);
        ExecutorService pool = Executors.newFixedThreadPool(4);
        List<Future<?>> fs = new ArrayList<>();
        for (int t = 1; t <= 4; t++) {
            final String sender = "u" + t;
            fs.add(pool.submit(() -> { for (int i = 0; i < 250; i++) s2.send(sender, big.id, sender + "-" + i); }));
        }
        for (Future<?> f : fs) f.get();
        pool.shutdown();
        List<Message> all = s2.history("watcher", big.id, Long.MAX_VALUE, 2000);
        Set<Long> seqs = new HashSet<>();
        for (Message m : all) seqs.add(m.seq);
        System.out.println("  messages " + all.size() + ", distinct seqs " + seqs.size() + ", max seq " + big.lastSeq()
                + ", watcher pushes " + watcher.pushes.get());
        check(all.size() == 1000 && seqs.size() == 1000 && big.lastSeq() == 1000, "seq 1..1000 with no gaps or duplicates");
        check(watcher.pushes.get() == 1000 && watcher.seen.size() == 1000, "each message pushed exactly once");

        System.out.println("All " + checks + " checks passed");
    }
}
