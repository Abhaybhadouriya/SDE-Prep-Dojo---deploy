import java.net.URI;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.Duration;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.LongAdder;
import java.util.regex.Pattern;
import java.util.stream.Collectors;

// URL shortener, LLD flavour: in-memory, pluggable code generation, collisions, expiry, analytics.
// Run: javac -d out Main.java && java -cp out Main
public class Main {

    // ---------- errors ----------
    static class ShortenerException extends RuntimeException { ShortenerException(String m) { super(m); } }
    static final class InvalidUrlException extends ShortenerException { InvalidUrlException(String m) { super(m); } }
    static final class AliasTakenException extends ShortenerException { AliasTakenException(String m) { super(m); } }
    static final class NotOwnerException extends ShortenerException { NotOwnerException(String m) { super(m); } }

    // ---------- base62 ----------
    static final class Base62 {
        static final String ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
        static String encode(long n) {
            if (n < 0) throw new IllegalArgumentException("negative");
            if (n == 0) return "0";
            StringBuilder sb = new StringBuilder();
            while (n > 0) { sb.append(ALPHABET.charAt((int) (n % 62))); n /= 62; }
            return sb.reverse().toString();
        }
        static long decode(String s) {
            long n = 0;
            for (char ch : s.toCharArray()) {
                int d = ALPHABET.indexOf(ch);
                if (d < 0) throw new IllegalArgumentException("not base62: " + ch);
                n = n * 62 + d;
            }
            return n;
        }
    }

    // ---------- time is injected so expiry is testable ----------
    interface Clock { Instant now(); }
    static final class FakeClock implements Clock {
        private volatile Instant t;
        FakeClock(Instant t) { this.t = t; }
        public Instant now() { return t; }
        void advance(Duration d) { t = t.plus(d); }
    }

    // ---------- entity ----------
    static final class ShortUrl {
        final String code, longUrl, owner;
        final Instant createdAt, expiresAt;     // expiresAt null = never
        final boolean custom;
        final LongAdder clicks = new LongAdder();
        ShortUrl(String code, String longUrl, String owner, Instant createdAt, Instant expiresAt, boolean custom) {
            this.code = code; this.longUrl = longUrl; this.owner = owner;
            this.createdAt = createdAt; this.expiresAt = expiresAt; this.custom = custom;
        }
        boolean expired(Instant now) { return expiresAt != null && !now.isBefore(expiresAt); }
    }

    // ---------- id generation: Strategy ----------
    interface CodeGenerator {
        // attempt = 0, 1, 2 ... ; a generator that can collide must return a different code for a new attempt
        String generate(String longUrl, int attempt);
    }
    // Counter + base62: never collides with itself, short, but guessable (see the scramble follow-up).
    static final class CounterGenerator implements CodeGenerator {
        private final AtomicLong next;
        CounterGenerator(long start) { next = new AtomicLong(start); }
        public String generate(String longUrl, int attempt) { return Base62.encode(next.getAndIncrement()); }
    }
    // Hash of (url, attempt): same URL -> same first code; collisions resolved by re-hashing with attempt + 1.
    // Keep the LOW-order base62 digits (n mod 62^length, left-padded). A prefix of a most-significant-first
    // encoding is not uniform: the leading digit of a 63-bit number is only ever 1..10.
    static final class HashGenerator implements CodeGenerator {
        private final int length;
        HashGenerator(int length) { this.length = length; }
        public String generate(String longUrl, int attempt) {
            try {
                byte[] h = MessageDigest.getInstance("SHA-256")
                        .digest((longUrl + "#" + attempt).getBytes(StandardCharsets.UTF_8));
                long n = 0;
                for (int i = 0; i < 8; i++) n = (n << 8) | (h[i] & 0xff);
                n &= Long.MAX_VALUE;
                char[] c = new char[length];
                for (int i = length - 1; i >= 0; i--) { c[i] = Base62.ALPHABET.charAt((int) (n % 62)); n /= 62; }
                return new String(c);
            } catch (java.security.NoSuchAlgorithmException e) {
                throw new IllegalStateException(e);
            }
        }
    }
    // Random code: unguessable; collisions resolved by drawing again.
    static final class RandomGenerator implements CodeGenerator {
        private final Random rnd; private final int length;
        RandomGenerator(Random rnd, int length) { this.rnd = rnd; this.length = length; }
        public String generate(String longUrl, int attempt) {
            char[] c = new char[length];
            for (int i = 0; i < length; i++) c[i] = Base62.ALPHABET.charAt(rnd.nextInt(62));
            return new String(c);
        }
    }

    // ---------- storage: Repository ----------
    interface UrlRepository {
        boolean saveIfAbsent(ShortUrl u);           // atomic claim of the code
        Optional<ShortUrl> find(String code);
        boolean delete(String code);
        Collection<ShortUrl> all();
    }
    static final class InMemoryUrlRepository implements UrlRepository {
        private final ConcurrentHashMap<String, ShortUrl> byCode = new ConcurrentHashMap<>();
        public boolean saveIfAbsent(ShortUrl u) { return byCode.putIfAbsent(u.code, u) == null; }
        public Optional<ShortUrl> find(String code) { return Optional.ofNullable(byCode.get(code)); }
        public boolean delete(String code) { return byCode.remove(code) != null; }
        public Collection<ShortUrl> all() { return byCode.values(); }
    }

    // ---------- validation ----------
    static final class UrlValidator {
        private static final Pattern ALIAS = Pattern.compile("[A-Za-z0-9_-]{3,30}");
        private static final Set<String> RESERVED = Set.of("api", "admin", "login", "stats", "health");
        private final String ownHost;
        UrlValidator(String ownHost) { this.ownHost = ownHost; }
        String normalize(String url) {
            if (url == null || url.isBlank() || url.length() > 2048) throw new InvalidUrlException("empty or too long");
            URI u;
            try { u = new URI(url.trim()); } catch (Exception e) { throw new InvalidUrlException("not a URL: " + url); }
            String scheme = u.getScheme() == null ? "" : u.getScheme().toLowerCase(Locale.ROOT);
            if (!scheme.equals("http") && !scheme.equals("https")) throw new InvalidUrlException("only http/https: " + url);
            if (u.getHost() == null) throw new InvalidUrlException("no host: " + url);
            String host = u.getHost().toLowerCase(Locale.ROOT);
            if (host.equals(ownHost)) throw new InvalidUrlException("would redirect to ourselves: " + url);
            return scheme + "://" + host + (u.getPort() == -1 ? "" : ":" + u.getPort())
                    + (u.getRawPath() == null ? "" : u.getRawPath())
                    + (u.getRawQuery() == null ? "" : "?" + u.getRawQuery());
        }
        void checkAlias(String alias) {
            if (!ALIAS.matcher(alias).matches()) throw new InvalidUrlException("alias must be 3-30 of [A-Za-z0-9_-]: " + alias);
            if (RESERVED.contains(alias.toLowerCase(Locale.ROOT))) throw new InvalidUrlException("reserved alias: " + alias);
        }
    }

    // ---------- analytics: Observer ----------
    interface ClickListener { void onClick(ShortUrl u, Instant at); }

    enum Outcome { FOUND, NOT_FOUND, EXPIRED }
    static final class Resolution {
        final Outcome outcome; final String longUrl;
        Resolution(Outcome o, String url) { outcome = o; longUrl = url; }
        int httpStatus() { return outcome == Outcome.FOUND ? 302 : outcome == Outcome.EXPIRED ? 410 : 404; }
    }

    static final class ShortenRequest {
        final String longUrl; String alias; Duration ttl; String owner = "anonymous";
        ShortenRequest(String longUrl) { this.longUrl = longUrl; }
        ShortenRequest alias(String a) { alias = a; return this; }
        ShortenRequest ttl(Duration d) { ttl = d; return this; }
        ShortenRequest owner(String o) { owner = o; return this; }
    }

    // ---------- the service: Facade over generator, repository, validator, clock, listeners ----------
    static final class UrlShortener {
        static final int MAX_ATTEMPTS = 5;
        private final String baseUrl;
        private final CodeGenerator generator;
        private final UrlRepository repo;
        private final UrlValidator validator;
        private final Clock clock;
        private final boolean dedupe;
        // dedupe index keyed by (owner, url): two owners never share one ShortUrl, so a delete by one
        // cannot kill the other's link and click counts stay per owner
        private final ConcurrentHashMap<String, String> codeByUrl = new ConcurrentHashMap<>();
        private final List<ClickListener> listeners = new CopyOnWriteArrayList<>();
        final LongAdder collisions = new LongAdder();

        UrlShortener(String ownHost, CodeGenerator g, UrlRepository r, Clock c, boolean dedupe) {
            this.baseUrl = "https://" + ownHost + "/"; this.generator = g; this.repo = r;
            this.validator = new UrlValidator(ownHost); this.clock = c; this.dedupe = dedupe;
        }
        void addListener(ClickListener l) { listeners.add(l); }
        String link(ShortUrl u) { return baseUrl + u.code; }

        ShortUrl shorten(ShortenRequest req) {
            String url = validator.normalize(req.longUrl);
            Instant now = clock.now();
            Instant exp = req.ttl == null ? null : now.plus(req.ttl);
            if (req.alias != null) {                                   // custom alias: claim or fail, never retry
                validator.checkAlias(req.alias);
                ShortUrl u = new ShortUrl(req.alias, url, req.owner, now, exp, true);
                if (!repo.saveIfAbsent(u)) throw new AliasTakenException("alias taken: " + req.alias);
                return u;
            }
            if (dedupe && req.ttl == null) {                           // same URL -> same code (one winner under races)
                String key = dedupeKey(req.owner, url);
                while (true) {
                    String code = codeByUrl.computeIfAbsent(key, k -> claim(url, req.owner, now, null).code);
                    Optional<ShortUrl> found = repo.find(code);
                    if (found.isPresent()) return found.get();
                    codeByUrl.remove(key, code);   // a delete() removed the link but not yet this entry: drop it, retry
                }
            }
            return claim(url, req.owner, now, exp);
        }

        private static String dedupeKey(String owner, String url) { return owner + "\0" + url; }

        private ShortUrl claim(String url, String owner, Instant now, Instant exp) {
            for (int attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
                ShortUrl u = new ShortUrl(generator.generate(url, attempt), url, owner, now, exp, false);
                if (repo.saveIfAbsent(u)) return u;                     // atomic: no check-then-act race
                collisions.increment();
            }
            throw new ShortenerException("could not find a free code after " + MAX_ATTEMPTS + " attempts");
        }

        Resolution resolve(String code) {
            Optional<ShortUrl> found = repo.find(code);
            if (!found.isPresent()) return new Resolution(Outcome.NOT_FOUND, null);
            ShortUrl u = found.get();
            Instant now = clock.now();
            if (u.expired(now)) return new Resolution(Outcome.EXPIRED, null);   // lazy expiry
            u.clicks.increment();
            for (ClickListener l : listeners) l.onClick(u, now);
            return new Resolution(Outcome.FOUND, u.longUrl);
        }

        void delete(String code, String owner) {
            ShortUrl u = repo.find(code).orElseThrow(() -> new ShortenerException("no such code: " + code));
            if (!u.owner.equals(owner)) throw new NotOwnerException(owner + " does not own " + code);
            codeByUrl.remove(dedupeKey(u.owner, u.longUrl), code);    // dedupe entry first, so no shorten() sees a dead code
            repo.delete(code);
        }

        int purgeExpired() {                                           // eager sweep, e.g. from a scheduled job
            Instant now = clock.now();
            int n = 0;
            for (ShortUrl u : new ArrayList<>(repo.all()))
                if (u.expired(now) && repo.delete(u.code)) n++;
            return n;
        }

        long clicks(String code) { return repo.find(code).map(u -> u.clicks.sum()).orElse(0L); }

        List<String> topK(int k) {
            return repo.all().stream()
                    .sorted(Comparator.comparingLong((ShortUrl u) -> u.clicks.sum()).reversed().thenComparing(u -> u.code))
                    .limit(k).map(u -> u.code + "=" + u.clicks.sum()).collect(Collectors.toList());
        }
    }

    // ---------- demo driver with simple checks ----------
    static int checks = 0;
    static void check(boolean ok, String what) { if (!ok) throw new AssertionError("FAILED: " + what); checks++; }
    static void expectThrows(Class<? extends Exception> type, Runnable r, String what) {
        try { r.run(); } catch (Exception e) { check(type.isInstance(e), what + " threw " + e.getClass().getSimpleName()); return; }
        check(false, what + " did not throw");
    }

    public static void main(String[] args) throws Exception {
        FakeClock clock = new FakeClock(Instant.parse("2026-01-01T00:00:00Z"));
        String host = "sho.rt";

        // 1. counter + base62, dedupe on
        UrlShortener s = new UrlShortener(host, new CounterGenerator(1_000_000), new InMemoryUrlRepository(), clock, true);
        ShortUrl a = s.shorten(new ShortenRequest("https://Example.com/docs?id=42"));
        ShortUrl b = s.shorten(new ShortenRequest("https://example.com/blog"));
        ShortUrl a2 = s.shorten(new ShortenRequest("https://EXAMPLE.com/docs?id=42"));
        System.out.println("1. counter: " + s.link(a) + " -> " + a.longUrl + " | " + s.link(b) + " | same URL again -> " + a2.code);
        check(a.code.equals("4c92") && Base62.decode(a.code) == 1_000_000, "base62 of 1,000,000");
        check(a2.code.equals(a.code), "dedupe returns the same code");

        // 2. custom alias, taken alias, bad alias, reserved alias
        ShortUrl c = s.shorten(new ShortenRequest("https://example.com/handbook").alias("team-docs").owner("asha"));
        expectThrows(AliasTakenException.class, () -> s.shorten(new ShortenRequest("https://other.org").alias("team-docs")), "taken alias");
        expectThrows(InvalidUrlException.class, () -> s.shorten(new ShortenRequest("https://other.org").alias("a b")), "bad alias");
        expectThrows(InvalidUrlException.class, () -> s.shorten(new ShortenRequest("https://other.org").alias("admin")), "reserved alias");
        System.out.println("2. alias: " + s.link(c) + "; taken / bad / reserved aliases rejected");

        // 3. URL validation
        expectThrows(InvalidUrlException.class, () -> s.shorten(new ShortenRequest("ftp://files.example.com/x")), "ftp");
        expectThrows(InvalidUrlException.class, () -> s.shorten(new ShortenRequest("not a url")), "garbage");
        expectThrows(InvalidUrlException.class, () -> s.shorten(new ShortenRequest("https://sho.rt/4c92")), "self loop");
        System.out.println("3. validation: ftp, garbage and self-links rejected");

        // 4. resolve + analytics (observer) + top-k
        Map<String, Integer> perDay = new TreeMap<>();
        s.addListener((u, at) -> perDay.merge(u.code + "@" + at.toString().substring(0, 10), 1, Integer::sum));
        for (int i = 0; i < 3; i++) s.resolve(a.code);
        s.resolve(c.code);
        Resolution miss = s.resolve("zzzz");
        System.out.println("4. resolve: " + a.code + " -> " + s.resolve(a.code).httpStatus() + " " + s.resolve(a.code).longUrl
                + " | zzzz -> " + miss.httpStatus() + " | top2 " + s.topK(2) + " | per day " + perDay);
        check(s.clicks(a.code) == 5 && miss.outcome == Outcome.NOT_FOUND, "clicks and 404");

        // 5. expiry with an injected clock: lazy on read, eager purge
        ShortUrl t = s.shorten(new ShortenRequest("https://example.com/sale").ttl(Duration.ofSeconds(60)));
        check(s.resolve(t.code).outcome == Outcome.FOUND, "before expiry");
        clock.advance(Duration.ofSeconds(61));
        Resolution gone = s.resolve(t.code);
        int purged = s.purgeExpired();
        System.out.println("5. expiry: " + t.code + " after 61 s -> " + gone.httpStatus() + " " + gone.outcome + ", purged " + purged);
        check(gone.outcome == Outcome.EXPIRED && purged == 1 && s.resolve(t.code).outcome == Outcome.NOT_FOUND, "expiry");

        // 6. owner-only delete
        expectThrows(NotOwnerException.class, () -> s.delete(c.code, "ben"), "delete by non-owner");
        s.delete(c.code, "asha");
        check(s.resolve(c.code).outcome == Outcome.NOT_FOUND, "deleted");
        System.out.println("6. delete: ben refused, asha deleted team-docs");

        // 6b. dedupe is per owner: asha and ben shorten the same URL, asha deletes hers, ben's still resolves
        ShortUrl ra = s.shorten(new ShortenRequest("https://example.com/shared").owner("asha"));
        ShortUrl rb = s.shorten(new ShortenRequest("https://example.com/shared").owner("ben"));
        check(!ra.code.equals(rb.code) && rb.owner.equals("ben"), "per-owner dedupe gives ben his own code");
        check(s.shorten(new ShortenRequest("https://example.com/shared").owner("ben")).code.equals(rb.code), "ben dedupes onto his own code");
        s.delete(ra.code, "asha");
        check(s.resolve(rb.code).outcome == Outcome.FOUND, "ben's link survives asha's delete");
        System.out.println("6b. per-owner dedupe: asha " + ra.code + ", ben " + rb.code + "; asha deleted hers, ben's still -> 302");

        // 6c. delete vs dedupe: the link row is gone but the dedupe entry still points at it (the race window)
        InMemoryUrlRepository r6 = new InMemoryUrlRepository();
        UrlShortener s6 = new UrlShortener(host, new CounterGenerator(5_000), r6, clock, true);
        ShortUrl old6 = s6.shorten(new ShortenRequest("https://example.com/race"));
        r6.delete(old6.code);                                          // as if delete() paused between its two steps
        ShortUrl new6 = s6.shorten(new ShortenRequest("https://example.com/race"));
        check(!new6.code.equals(old6.code) && s6.resolve(new6.code).outcome == Outcome.FOUND, "stale dedupe entry is dropped, not a 500");
        System.out.println("6c. delete vs dedupe: stale code " + old6.code + " dropped, new code " + new6.code);

        // 7. hash strategy with a forced collision: another URL already holds the first hash code
        InMemoryUrlRepository repo = new InMemoryUrlRepository();
        HashGenerator hg = new HashGenerator(7);
        String target = "https://example.com/very/long/path";
        String firstCode = hg.generate(target, 0);
        repo.saveIfAbsent(new ShortUrl(firstCode, "https://squatter.example/", "x", clock.now(), null, false));
        UrlShortener hs = new UrlShortener(host, hg, repo, clock, false);
        ShortUrl h = hs.shorten(new ShortenRequest(target));
        System.out.println("7. hash: first code " + firstCode + " was taken -> got " + h.code + " after " + hs.collisions.sum() + " collision");
        check(!h.code.equals(firstCode) && h.code.equals(hg.generate(target, 1)) && hs.collisions.sum() == 1, "re-hash on collision");

        // 7b. the hash code is spread over the whole alphabet, first character included
        Set<Character> firstChars = new HashSet<>();
        for (int i = 0; i < 10_000; i++) firstChars.add(hg.generate("https://spread.example/" + i, 0).charAt(0));
        System.out.println("7b. hash spread: first character took " + firstChars.size() + " of 62 symbols over 10000 URLs");
        check(firstChars.size() == 62, "first character uniform over base62");

        // 8. a generator that always collides must fail loudly, not loop forever
        UrlShortener stuck = new UrlShortener(host, (url, attempt) -> "same", new InMemoryUrlRepository(), clock, false);
        stuck.shorten(new ShortenRequest("https://one.example/"));
        expectThrows(ShortenerException.class, () -> stuck.shorten(new ShortenRequest("https://two.example/")), "bounded retries");
        System.out.println("8. bounded retries: gave up after " + UrlShortener.MAX_ATTEMPTS + " attempts");

        // 9. concurrency: 8 threads x 1000 distinct URLs (random codes), 8 threads on one URL (dedupe), 8 x 1000 clicks
        UrlShortener cs = new UrlShortener(host, new RandomGenerator(new Random(42), 3), new InMemoryUrlRepository(), clock, true);
        ExecutorService pool = Executors.newFixedThreadPool(8);
        Set<String> codes = ConcurrentHashMap.newKeySet();
        List<Future<?>> fs = new ArrayList<>();
        for (int t8 = 0; t8 < 8; t8++) {
            final int id = t8;
            fs.add(pool.submit(() -> { for (int i = 0; i < 1000; i++) codes.add(cs.shorten(new ShortenRequest("https://site.example/" + id + "/" + i)).code); }));
        }
        for (Future<?> f : fs) f.get();
        Set<String> same = ConcurrentHashMap.newKeySet();
        fs.clear();
        for (int t8 = 0; t8 < 8; t8++) fs.add(pool.submit(() -> same.add(cs.shorten(new ShortenRequest("https://hot.example/launch")).code)));
        for (Future<?> f : fs) f.get();
        String hot = same.iterator().next();
        fs.clear();
        for (int t8 = 0; t8 < 8; t8++) fs.add(pool.submit(() -> { for (int i = 0; i < 1000; i++) cs.resolve(hot); }));
        for (Future<?> f : fs) f.get();
        pool.shutdown();
        System.out.println("9. concurrency: " + codes.size() + " unique codes, " + same.size() + " code for 8 racing shortens of one URL, "
                + cs.clicks(hot) + " clicks counted, " + (cs.collisions.sum() > 0 ? "collisions retried" : "no collisions"));
        check(codes.size() == 8000 && same.size() == 1 && cs.clicks(hot) == 8000, "concurrency");

        System.out.println("All " + checks + " checks passed");
    }
}
"""URL shortener, LLD flavour: pluggable code generation, collision retry, expiry, analytics.

Run: python main.py        Tests: python -m unittest test_main
"""
from __future__ import annotations

import hashlib
import random
import re
import threading
from dataclasses import dataclass, field
from datetime import datetime, timedelta, timezone
from enum import Enum
from typing import Callable, Dict, List, Optional, Protocol, Tuple
from urllib.parse import urlsplit

ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"


class ShortenerError(Exception): ...
class InvalidUrlError(ShortenerError): ...
class AliasTakenError(ShortenerError): ...
class NotOwnerError(ShortenerError): ...


def b62_encode(n: int) -> str:
    if n < 0:
        raise ValueError("negative")
    if n == 0:
        return "0"
    out = []
    while n:
        n, r = divmod(n, 62)
        out.append(ALPHABET[r])
    return "".join(reversed(out))


def b62_decode(s: str) -> int:
    n = 0
    for ch in s:
        d = ALPHABET.find(ch)
        if d < 0:
            raise ValueError(f"not base62: {ch}")
        n = n * 62 + d
    return n


class Clock(Protocol):
    def now(self) -> datetime: ...


class FakeClock:
    def __init__(self, t: datetime) -> None:
        self.t = t

    def now(self) -> datetime:
        return self.t

    def advance(self, d: timedelta) -> None:
        self.t += d


@dataclass
class ShortUrl:
    code: str
    long_url: str
    owner: str
    created_at: datetime
    expires_at: Optional[datetime]
    custom: bool = False
    clicks: int = 0
    _lock: threading.Lock = field(default_factory=threading.Lock, repr=False, compare=False)

    def expired(self, now: datetime) -> bool:
        return self.expires_at is not None and now >= self.expires_at

    def click(self) -> None:
        with self._lock:              # += is not atomic across threads
            self.clicks += 1


# ---------- id generation: Strategy (any callable (url, attempt) -> code works) ----------
CodeGenerator = Callable[[str, int], str]


class CounterGenerator:
    def __init__(self, start: int) -> None:
        self._next, self._lock = start, threading.Lock()

    def __call__(self, url: str, attempt: int) -> str:
        with self._lock:
            n, self._next = self._next, self._next + 1
        return b62_encode(n)


class HashGenerator:
    def __init__(self, length: int) -> None:
        self.length = length

    def __call__(self, url: str, attempt: int) -> str:
        h = hashlib.sha256(f"{url}#{attempt}".encode()).digest()
        n = int.from_bytes(h[:8], "big") & ((1 << 63) - 1)
        # low-order digits, left-padded: a prefix of b62_encode(n) is not uniform (its first digit is 1..10)
        return b62_encode(n % 62 ** self.length).rjust(self.length, ALPHABET[0])


class RandomGenerator:
    def __init__(self, rnd: random.Random, length: int) -> None:
        self.rnd, self.length, self._lock = rnd, length, threading.Lock()

    def __call__(self, url: str, attempt: int) -> str:
        with self._lock:
            return "".join(self.rnd.choice(ALPHABET) for _ in range(self.length))


# ---------- storage: Repository ----------
class InMemoryUrlRepository:
    def __init__(self) -> None:
        self._by_code: Dict[str, ShortUrl] = {}
        self._lock = threading.Lock()

    def save_if_absent(self, u: ShortUrl) -> bool:      # atomic claim
        with self._lock:
            if u.code in self._by_code:
                return False
            self._by_code[u.code] = u
            return True

    def find(self, code: str) -> Optional[ShortUrl]:
        with self._lock:
            return self._by_code.get(code)

    def delete(self, code: str) -> bool:
        with self._lock:
            return self._by_code.pop(code, None) is not None

    def all(self) -> List[ShortUrl]:
        with self._lock:
            return list(self._by_code.values())


class UrlValidator:
    ALIAS = re.compile(r"[A-Za-z0-9_-]{3,30}")
    RESERVED = {"api", "admin", "login", "stats", "health"}

    def __init__(self, own_host: str) -> None:
        self.own_host = own_host

    def normalize(self, url: str) -> str:
        if not url or not url.strip() or len(url) > 2048:
            raise InvalidUrlError("empty or too long")
        if any(ch.isspace() for ch in url.strip()):
            raise InvalidUrlError(f"not a URL: {url}")
        parts = urlsplit(url.strip())
        scheme = parts.scheme.lower()
        if scheme not in ("http", "https"):
            raise InvalidUrlError(f"only http/https: {url}")
        if not parts.hostname:
            raise InvalidUrlError(f"no host: {url}")
        host = parts.hostname.lower()
        if host == self.own_host:
            raise InvalidUrlError(f"would redirect to ourselves: {url}")
        port = f":{parts.port}" if parts.port else ""
        query = f"?{parts.query}" if parts.query else ""
        return f"{scheme}://{host}{port}{parts.path}{query}"

    def check_alias(self, alias: str) -> None:
        if not self.ALIAS.fullmatch(alias):
            raise InvalidUrlError(f"alias must be 3-30 of [A-Za-z0-9_-]: {alias}")
        if alias.lower() in self.RESERVED:
            raise InvalidUrlError(f"reserved alias: {alias}")


class Outcome(Enum):
    FOUND = 302
    NOT_FOUND = 404
    EXPIRED = 410


@dataclass
class Resolution:
    outcome: Outcome
    long_url: Optional[str] = None


ClickListener = Callable[[ShortUrl, datetime], None]


class UrlShortener:
    MAX_ATTEMPTS = 5

    def __init__(self, own_host: str, generator: CodeGenerator, repo: InMemoryUrlRepository,
                 clock: Clock, dedupe: bool = True) -> None:
        self.base = f"https://{own_host}/"
        self.generator, self.repo, self.clock, self.dedupe = generator, repo, clock, dedupe
        self.validator = UrlValidator(own_host)
        self._code_by_url: Dict[Tuple[str, str], str] = {}
        self._dedupe_lock = threading.Lock()
        self._listeners: List[ClickListener] = []
        self.collisions = 0
        self._stats_lock = threading.Lock()

    def add_listener(self, l: ClickListener) -> None:
        self._listeners.append(l)

    def link(self, u: ShortUrl) -> str:
        return self.base + u.code

    def shorten(self, long_url: str, alias: Optional[str] = None, ttl: Optional[timedelta] = None,
                owner: str = "anonymous") -> ShortUrl:
        url = self.validator.normalize(long_url)
        now = self.clock.now()
        exp = now + ttl if ttl else None
        if alias is not None:
            self.validator.check_alias(alias)
            u = ShortUrl(alias, url, owner, now, exp, custom=True)
            if not self.repo.save_if_absent(u):
                raise AliasTakenError(f"alias taken: {alias}")
            return u
        if self.dedupe and ttl is None:
            key = (owner, url)                            # per owner: a delete by one owner never kills another's link
            with self._dedupe_lock:                       # one winner when many threads shorten the same URL
                code = self._code_by_url.get(key)
                found = self.repo.find(code) if code is not None else None
                if found is None:                         # no entry, or a stale one whose link was deleted
                    found = self._claim(url, owner, now, None)
                    self._code_by_url[key] = found.code
            return found
        return self._claim(url, owner, now, exp)

    def _claim(self, url: str, owner: str, now: datetime, exp: Optional[datetime]) -> ShortUrl:
        for attempt in range(self.MAX_ATTEMPTS):
            u = ShortUrl(self.generator(url, attempt), url, owner, now, exp)
            if self.repo.save_if_absent(u):
                return u
            with self._stats_lock:
                self.collisions += 1
        raise ShortenerError(f"could not find a free code after {self.MAX_ATTEMPTS} attempts")

    def resolve(self, code: str) -> Resolution:
        u = self.repo.find(code)
        if u is None:
            return Resolution(Outcome.NOT_FOUND)
        now = self.clock.now()
        if u.expired(now):
            return Resolution(Outcome.EXPIRED)
        u.click()
        for l in self._listeners:
            l(u, now)
        return Resolution(Outcome.FOUND, u.long_url)

    def delete(self, code: str, owner: str) -> None:
        u = self.repo.find(code)
        if u is None:
            raise ShortenerError(f"no such code: {code}")
        if u.owner != owner:
            raise NotOwnerError(f"{owner} does not own {code}")
        with self._dedupe_lock:                           # entry and row go together: shorten() never sees a dead code
            if self._code_by_url.get((u.owner, u.long_url)) == code:
                del self._code_by_url[(u.owner, u.long_url)]
            self.repo.delete(code)

    def purge_expired(self) -> int:
        now = self.clock.now()
        return sum(1 for u in self.repo.all() if u.expired(now) and self.repo.delete(u.code))

    def clicks(self, code: str) -> int:
        u = self.repo.find(code)
        return u.clicks if u else 0

    def top_k(self, k: int) -> List[str]:
        ranked = sorted(self.repo.all(), key=lambda u: (-u.clicks, u.code))
        return [f"{u.code}={u.clicks}" for u in ranked[:k]]

// URL shortener (LLD) in C++14: Strategy for code generation, Repository, lazy expiry, click analytics.
// Build: g++ -std=c++14 -Wall -Wextra main.cpp -o main && ./main
// Single-threaded demo: MinGW g++ 6.3 has no std::mutex. In a threaded build, the repository's saveIfAbsent /
// find / remove take one std::mutex (or use a concurrent map), the counter is std::atomic<long long>, and clicks
// are std::atomic<long long> per entry. The C++ standard library has no SHA-256, so HashGenerator uses FNV-1a 64.

static const std::string ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

struct ShortenerError : std::runtime_error { using std::runtime_error::runtime_error; };
struct InvalidUrl : ShortenerError { using ShortenerError::ShortenerError; };
struct AliasTaken : ShortenerError { using ShortenerError::ShortenerError; };
struct NotOwner : ShortenerError { using ShortenerError::ShortenerError; };

static std::string b62encode(unsigned long long n) {
    if (n == 0) return "0";
    std::string s;
    while (n) { s.push_back(ALPHABET[n % 62]); n /= 62; }
    std::reverse(s.begin(), s.end());
    return s;
}
static unsigned long long b62decode(const std::string& s) {
    unsigned long long n = 0;
    for (char ch : s) {
        size_t d = ALPHABET.find(ch);
        if (d == std::string::npos) throw std::invalid_argument("not base62");
        n = n * 62 + d;
    }
    return n;
}

// time in seconds, injected
struct Clock { virtual ~Clock() {} virtual long long now() const = 0; };
struct FakeClock : Clock { long long t = 0; long long now() const override { return t; } };

struct ShortUrl {
    std::string code, longUrl, owner;
    long long createdAt = 0, expiresAt = -1;   // -1 = never
    bool custom = false;
    long long clicks = 0;
    bool expired(long long now) const { return expiresAt >= 0 && now >= expiresAt; }
};

// ---- Strategy: code generation ----
struct CodeGenerator {
    virtual ~CodeGenerator() {}
    virtual std::string generate(const std::string& url, int attempt) = 0;
};
struct CounterGenerator : CodeGenerator {
    unsigned long long next;
    explicit CounterGenerator(unsigned long long start) : next(start) {}
    std::string generate(const std::string&, int) override { return b62encode(next++); }
};
struct HashGenerator : CodeGenerator {
    size_t length;
    explicit HashGenerator(size_t len) : length(len) {}
    static std::uint64_t fnv1a(const std::string& s) {
        std::uint64_t h = 1469598103934665603ULL;
        for (unsigned char c : s) { h ^= c; h *= 1099511628211ULL; }
        return h;
    }
    std::string generate(const std::string& url, int attempt) override {
        // low-order base62 digits, left-padded: a prefix of the full encoding is not uniform
        std::uint64_t n = fnv1a(url + "#" + std::to_string(attempt)) & 0x7fffffffffffffffULL;
        std::string code(length, ALPHABET[0]);
        for (size_t i = length; i-- > 0;) { code[i] = ALPHABET[n % 62]; n /= 62; }
        return code;
    }
};
struct RandomGenerator : CodeGenerator {
    std::mt19937 rng; size_t length;
    RandomGenerator(unsigned seed, size_t len) : rng(seed), length(len) {}
    std::string generate(const std::string&, int) override {
        std::uniform_int_distribution<int> d(0, 61);
        std::string s;
        for (size_t i = 0; i < length; ++i) s.push_back(ALPHABET[d(rng)]);
        return s;
    }
};
struct FixedGenerator : CodeGenerator {   // always the same code: proves retries are bounded
    std::string generate(const std::string&, int) override { return "same"; }
};

struct UrlRepository {
    virtual ~UrlRepository() {}
    virtual bool saveIfAbsent(const ShortUrl& u) = 0;
    virtual ShortUrl* find(const std::string& code) = 0;   // nullptr = missing
    virtual bool remove(const std::string& code) = 0;
    virtual std::vector<ShortUrl*> all() = 0;
};
struct InMemoryRepository : UrlRepository {
    std::unordered_map<std::string, ShortUrl> byCode;
    bool saveIfAbsent(const ShortUrl& u) override { return byCode.emplace(u.code, u).second; }
    ShortUrl* find(const std::string& c) override { auto it = byCode.find(c); return it == byCode.end() ? nullptr : &it->second; }
    bool remove(const std::string& c) override { return byCode.erase(c) > 0; }
    std::vector<ShortUrl*> all() override { std::vector<ShortUrl*> v; for (auto& kv : byCode) v.push_back(&kv.second); return v; }
};

// ---- validation ----
static std::string lower(std::string s) { for (char& c : s) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c))); return s; }
struct UrlValidator {
    std::string ownHost;
    std::string normalize(const std::string& url) const {
        if (url.empty() || url.size() > 2048) throw InvalidUrl("empty or too long");
        if (std::any_of(url.begin(), url.end(), [](char c) { return std::isspace(static_cast<unsigned char>(c)) != 0; }))
            throw InvalidUrl("not a URL: " + url);
        size_t sep = url.find("://");
        if (sep == std::string::npos) throw InvalidUrl("not a URL: " + url);
        std::string scheme = lower(url.substr(0, sep));
        if (scheme != "http" && scheme != "https") throw InvalidUrl("only http/https: " + url);
        size_t hostEnd = url.find_first_of("/?#", sep + 3);
        std::string host = lower(url.substr(sep + 3, hostEnd == std::string::npos ? std::string::npos : hostEnd - sep - 3));
        if (host.empty()) throw InvalidUrl("no host: " + url);
        if (host == ownHost) throw InvalidUrl("would redirect to ourselves: " + url);
        std::string rest = hostEnd == std::string::npos ? "" : url.substr(hostEnd);
        size_t hash = rest.find('#');
        if (hash != std::string::npos) rest = rest.substr(0, hash);   // fragments never reach the server
        return scheme + "://" + host + rest;
    }
    void checkAlias(const std::string& a) const {
        static const std::set<std::string> reserved = {"api", "admin", "login", "stats", "health"};
        if (a.size() < 3 || a.size() > 30) throw InvalidUrl("alias must be 3-30 chars");
        for (char c : a) if (!(std::isalnum(static_cast<unsigned char>(c)) || c == '_' || c == '-')) throw InvalidUrl("bad alias char");
        if (reserved.count(lower(a))) throw InvalidUrl("reserved alias: " + a);
    }
};

enum class Outcome { FOUND = 302, NOT_FOUND = 404, EXPIRED = 410 };
struct Resolution { Outcome outcome; std::string longUrl; };

class UrlShortener {
public:
    static const int MAX_ATTEMPTS = 5;
    long long collisions = 0;
    std::function<void(const ShortUrl&, long long)> onClick;   // one-listener Observer

    UrlShortener(std::string host, std::unique_ptr<CodeGenerator> g, std::unique_ptr<UrlRepository> r,
                 const Clock& c, bool dedupe)
        : base_("https://" + host + "/"), gen_(std::move(g)), repo_(std::move(r)), clock_(c), dedupe_(dedupe) {
        validator_.ownHost = host;
    }
    UrlRepository& repo() { return *repo_; }
    std::string link(const ShortUrl& u) const { return base_ + u.code; }

    ShortUrl shorten(const std::string& longUrl, const std::string& alias = "", long long ttl = -1,
                     const std::string& owner = "anonymous") {
        std::string url = validator_.normalize(longUrl);
        long long now = clock_.now(), exp = ttl < 0 ? -1 : now + ttl;
        if (!alias.empty()) {
            validator_.checkAlias(alias);
            ShortUrl u{alias, url, owner, now, exp, true, 0};
            if (!repo_->saveIfAbsent(u)) throw AliasTaken("alias taken: " + alias);
            return u;
        }
        if (dedupe_ && ttl < 0) {
            std::string key = owner + '\0' + url;              // dedupe per owner
            auto it = codeByUrl_.find(key);
            if (it != codeByUrl_.end()) {
                if (ShortUrl* found = repo_->find(it->second)) return *found;
                codeByUrl_.erase(it);                          // stale: its link was deleted underneath
            }
            ShortUrl u = claim(url, owner, now, -1);
            codeByUrl_[key] = u.code;
            return u;
        }
        return claim(url, owner, now, exp);
    }

    Resolution resolve(const std::string& code) {
        ShortUrl* u = repo_->find(code);
        if (!u) return Resolution{Outcome::NOT_FOUND, ""};
        long long now = clock_.now();
        if (u->expired(now)) return Resolution{Outcome::EXPIRED, ""};
        ++u->clicks;
        if (onClick) onClick(*u, now);
        return Resolution{Outcome::FOUND, u->longUrl};
    }

    void remove(const std::string& code, const std::string& owner) {
        ShortUrl* u = repo_->find(code);
        if (!u) throw ShortenerError("no such code: " + code);
        if (u->owner != owner) throw NotOwner(owner + " does not own " + code);
        auto it = codeByUrl_.find(u->owner + '\0' + u->longUrl);
        if (it != codeByUrl_.end() && it->second == code) codeByUrl_.erase(it);
        repo_->remove(code);
    }

    int purgeExpired() {
        long long now = clock_.now();
        std::vector<std::string> dead;
        for (ShortUrl* u : repo_->all()) if (u->expired(now)) dead.push_back(u->code);
        for (const auto& c : dead) repo_->remove(c);
        return static_cast<int>(dead.size());
    }

    std::vector<std::string> topK(size_t k) {
        std::vector<ShortUrl*> v = repo_->all();
        std::sort(v.begin(), v.end(), [](const ShortUrl* a, const ShortUrl* b) {
            return a->clicks != b->clicks ? a->clicks > b->clicks : a->code < b->code;
        });
        std::vector<std::string> out;
        for (size_t i = 0; i < v.size() && i < k; ++i) out.push_back(v[i]->code + "=" + std::to_string(v[i]->clicks));
        return out;
    }

private:
    std::string base_;
    std::unique_ptr<CodeGenerator> gen_;
    std::unique_ptr<UrlRepository> repo_;
    const Clock& clock_;
    bool dedupe_;
    UrlValidator validator_;
    std::unordered_map<std::string, std::string> codeByUrl_;

    ShortUrl claim(const std::string& url, const std::string& owner, long long now, long long exp) {
        for (int attempt = 0; attempt < MAX_ATTEMPTS; ++attempt) {
            ShortUrl u{gen_->generate(url, attempt), url, owner, now, exp, false, 0};
            if (repo_->saveIfAbsent(u)) return u;
            ++collisions;
        }
        throw ShortenerError("no free code after retries");
    }
};

static int checks = 0;
static void check(bool ok, const char* what) { if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); } ++checks; }
template <class E, class F> static void expectThrows(F f, const char* what) {
    try { f(); } catch (const E&) { ++checks; return; } catch (...) { check(false, what); }
    check(false, what);
}
template <class G> static std::unique_ptr<CodeGenerator> gen(G* g) { return std::unique_ptr<CodeGenerator>(g); }
static std::unique_ptr<UrlRepository> repo() { return std::unique_ptr<UrlRepository>(new InMemoryRepository()); }

int main() {
    FakeClock clock;
    UrlShortener s("sho.rt", gen(new CounterGenerator(1000000)), repo(), clock, true);
    ShortUrl a = s.shorten("https://Example.com/docs?id=42#top");
    ShortUrl b = s.shorten("https://example.com/blog");
    check(a.code == "4c92" && b.code == "4c93" && b62decode(a.code) == 1000000, "counter + base62");
    check(s.shorten("https://EXAMPLE.com/docs?id=42").code == a.code, "dedupe ignores host case and fragment");
    std::cout << "1. counter: " << s.link(a) << " -> " << a.longUrl << " | " << s.link(b) << "\n";

    ShortUrl c = s.shorten("https://example.com/handbook", "team-docs", -1, "asha");
    expectThrows<AliasTaken>([&] { s.shorten("https://other.org", "team-docs"); }, "taken alias");
    expectThrows<InvalidUrl>([&] { s.shorten("https://other.org", "admin"); }, "reserved alias");
    expectThrows<InvalidUrl>([&] { s.shorten("ftp://x.example/a"); }, "ftp");
    expectThrows<InvalidUrl>([&] { s.shorten("not a url"); }, "garbage");
    expectThrows<InvalidUrl>([&] { s.shorten("https://sho.rt/4c92"); }, "self loop");

    int listenerCalls = 0;
    s.onClick = [&listenerCalls](const ShortUrl&, long long) { ++listenerCalls; };
    for (int i = 0; i < 3; ++i) s.resolve(a.code);
    s.resolve(c.code);
    check(s.resolve("zzzz").outcome == Outcome::NOT_FOUND && listenerCalls == 4, "resolve + observer");
    std::vector<std::string> top = s.topK(2);
    std::cout << "4. top2: " << top[0] << " " << top[1] << "\n";

    ShortUrl t = s.shorten("https://example.com/sale", "", 60);
    check(s.resolve(t.code).outcome == Outcome::FOUND, "before expiry");
    clock.t += 61;
    check(s.resolve(t.code).outcome == Outcome::EXPIRED && s.purgeExpired() == 1, "expiry + purge");
    std::cout << "5. expiry: " << t.code << " after 61 s -> " << static_cast<int>(Outcome::EXPIRED) << ", purged 1\n";

    expectThrows<NotOwner>([&] { s.remove(c.code, "ben"); }, "non-owner delete");
    s.remove(c.code, "asha");
    check(s.resolve(c.code).outcome == Outcome::NOT_FOUND, "deleted");
    ShortUrl ra = s.shorten("https://example.com/shared", "", -1, "asha");
    ShortUrl rb = s.shorten("https://example.com/shared", "", -1, "ben");
    s.remove(ra.code, "asha");
    check(ra.code != rb.code && s.resolve(rb.code).outcome == Outcome::FOUND, "per-owner dedupe survives other owner's delete");
    std::cout << "6b. per-owner dedupe: asha " << ra.code << ", ben " << rb.code << "; asha deleted hers, ben's still resolves\n";

    HashGenerator probe(7);
    std::string target = "https://example.com/very/long/path";
    std::string first = probe.generate(target, 0);
    UrlShortener hs("sho.rt", gen(new HashGenerator(7)), repo(), clock, false);
    hs.repo().saveIfAbsent(ShortUrl{first, "https://squatter.example/", "x", 0, -1, false, 0});
    ShortUrl h = hs.shorten(target);
    check(h.code == probe.generate(target, 1) && hs.collisions == 1, "re-hash on collision");
    std::cout << "7. hash (FNV-1a): first code " << first << " was taken -> got " << h.code << "\n";
    std::set<char> firsts;
    for (int i = 0; i < 10000; ++i) firsts.insert(probe.generate("https://spread.example/" + std::to_string(i), 0)[0]);
    check(firsts.size() == 62, "first character uniform over base62");
    std::cout << "7b. hash spread: first character took " << firsts.size() << " of 62 symbols over 10000 URLs\n";

    UrlShortener stuck("sho.rt", gen(new FixedGenerator()), repo(), clock, false);
    stuck.shorten("https://one.example/");
    expectThrows<ShortenerError>([&] { stuck.shorten("https://two.example/"); }, "bounded retries");

    UrlShortener rs("sho.rt", gen(new RandomGenerator(42, 3)), repo(), clock, false);
    std::set<std::string> codes;
    for (int i = 0; i < 8000; ++i) codes.insert(rs.shorten("https://site.example/" + std::to_string(i)).code);
    check(codes.size() == 8000, "8000 unique random codes");
    std::cout << "9. random 3-char codes: 8000 unique, " << rs.collisions << " collisions retried\n";

    std::cout << "All " << checks << " checks passed\n";
    return 0;
}
