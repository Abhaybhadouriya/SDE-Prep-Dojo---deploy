Before and after: the code layout every case uses

Here is the same tiny problem (rent a bike, pay per half hour) written twice. Both print exactly the same output. The first is how many candidates start; the second is the layout every case study on this site uses: model, repository, strategy, service, driver.

import java.util.HashMap;
import java.util.Map;

// BEFORE: everything in main. It works for the first demo and fights every follow-up.
public class Main {
    public static void main(String[] args) {
        Map<String, String> bikeType = new HashMap<>();      // bikeId -> "CITY" / "EBIKE"
        Map<String, String> rentedBy = new HashMap<>();      // bikeId -> user
        bikeType.put("b1", "CITY");
        bikeType.put("b2", "EBIKE");
        String[] commands = {"rent u1 b1", "rent u2 b1", "return b1 90", "rent u2 b2", "return b2 30"};
        for (String line : commands) {
            String[] p = line.split(" ");
            if (p[0].equals("rent")) {
                if (rentedBy.containsKey(p[2])) System.out.println(p[2] + " is busy");
                else { rentedBy.put(p[2], p[1]); System.out.println(p[1] + " rented " + p[2]); }
            } else if (p[0].equals("return")) {
                int minutes = Integer.parseInt(p[2]);
                int fee;
                if (bikeType.get(p[1]).equals("CITY")) fee = 10 * ((minutes + 29) / 30);  // pricing rule 1
                else fee = 25 * ((minutes + 29) / 30);                                     // pricing rule 2
                rentedBy.remove(p[1]);
                System.out.println(p[1] + " returned, fee " + fee);
            }
        }
    }
}
import java.util.HashMap;
import java.util.Map;
import java.util.Optional;

// AFTER: the same behaviour, laid out the way every case study on this site is laid out.
enum BikeType { CITY, EBIKE }                                         // model: fixed vocabulary

record Bike(String id, BikeType type) {}                              // model: entity

class BikeBusyException extends RuntimeException {                    // model: domain error
    BikeBusyException(String id) { super(id + " is busy"); }
}

interface BikeRepository {                                            // repository: storage behind an interface
    void save(Bike b);
    Optional<Bike> find(String id);
}

class InMemoryBikeRepository implements BikeRepository {
    private final Map<String, Bike> bikes = new HashMap<>();
    public void save(Bike b) { bikes.put(b.id(), b); }
    public Optional<Bike> find(String id) { return Optional.ofNullable(bikes.get(id)); }
}

interface PricingStrategy { int fee(Bike bike, int minutes); }        // strategy: the rule that changes

class PerHalfHourPricing implements PricingStrategy {
    private final Map<BikeType, Integer> rate;
    PerHalfHourPricing(Map<BikeType, Integer> rate) { this.rate = rate; }
    public int fee(Bike bike, int minutes) { return rate.get(bike.type()) * ((minutes + 29) / 30); }
}

class RentalService {                                                 // service: the use cases
    private final BikeRepository bikes;
    private final PricingStrategy pricing;
    private final Map<String, String> rentedBy = new HashMap<>();

    RentalService(BikeRepository bikes, PricingStrategy pricing) { this.bikes = bikes; this.pricing = pricing; }

    void rent(String user, String bikeId) {
        bikes.find(bikeId).orElseThrow(() -> new IllegalArgumentException("no bike " + bikeId));
        if (rentedBy.putIfAbsent(bikeId, user) != null) throw new BikeBusyException(bikeId);
    }

    int giveBack(String bikeId, int minutes) {
        Bike bike = bikes.find(bikeId).orElseThrow(() -> new IllegalArgumentException("no bike " + bikeId));
        if (rentedBy.remove(bikeId) == null) throw new IllegalStateException(bikeId + " was not rented");
        return pricing.fee(bike, minutes);
    }
}

public class Main {                                                   // driver: parsing and printing only
    public static void main(String[] args) {
        BikeRepository repo = new InMemoryBikeRepository();
        repo.save(new Bike("b1", BikeType.CITY));
        repo.save(new Bike("b2", BikeType.EBIKE));
        RentalService svc = new RentalService(repo, new PerHalfHourPricing(Map.of(BikeType.CITY, 10, BikeType.EBIKE, 25)));
        String[] commands = {"rent u1 b1", "rent u2 b1", "return b1 90", "rent u2 b2", "return b2 30"};
        for (String line : commands) {
            String[] p = line.split(" ");
            try {
                switch (p[0]) {
                    case "rent" -> { svc.rent(p[1], p[2]); System.out.println(p[1] + " rented " + p[2]); }
                    case "return" -> System.out.println(p[1] + " returned, fee " + svc.giveBack(p[1], Integer.parseInt(p[2])));
                    default -> System.out.println("unknown command " + p[0]);
                }
            } catch (RuntimeException e) {
                System.out.println(e.getMessage());
            }
        }
    }
}

Same output, very different futures:

Before
  • A new bike type means editing the pricing if/else inside the command loop.
  • Nothing can be tested without parsing strings.
  • Storage is two loose maps; moving to a database touches every line.
  • Errors are printed, not raised, so a caller cannot react.
After
  • New pricing = new PricingStrategy class.
  • RentalService is testable without any input parsing.
  • A database means a new BikeRepository implementation.
  • Domain exceptions; the driver decides how to print them.

When you present a machine-coding solution, start with this layout in one sentence: "Models in one place, storage behind a repository interface, the rules that vary behind strategies, use cases in a service, and a driver that only parses and prints."
