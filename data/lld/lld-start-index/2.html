A taste: one design, four languages

Here is the smallest useful LLD idea. An order service must send a confirmation. Today by email and SMS; next month maybe WhatsApp. If OrderService calls sendEmail() and sendSms() directly, every new channel means editing OrderService. Instead, we put an interface between them. The service only knows "something that can send".

interface Notifier {
    void send(String user, String text);
}

final class EmailNotifier implements Notifier {
    public void send(String user, String text) {
        System.out.println("[email] to " + user + ": " + text);
    }
}

final class SmsNotifier implements Notifier {
    public void send(String user, String text) {
        System.out.println("[sms]   to " + user + ": " + text);
    }
}

final class OrderService {
    private final List<Notifier> notifiers;      // knows the interface only

    OrderService(List<Notifier> notifiers) { this.notifiers = notifiers; }

    void placeOrder(String user, String item) {
        System.out.println("order placed: " + item);
        for (Notifier n : notifiers) n.send(user, "your " + item + " is confirmed");
    }
}

from abc import ABC, abstractmethod

class Notifier(ABC):
    @abstractmethod
    def send(self, user: str, text: str) -> None: ...

class EmailNotifier(Notifier):
    def send(self, user: str, text: str) -> None:
        print(f"[email] to {user}: {text}")

class SmsNotifier(Notifier):
    def send(self, user: str, text: str) -> None:
        print(f"[sms]   to {user}: {text}")

class OrderService:
    def __init__(self, notifiers: list[Notifier]) -> None:
        self._notifiers = notifiers          # depends on the abstraction only

    def place_order(self, user: str, item: str) -> None:
        print(f"order placed: {item}")
        for n in self._notifiers:
            n.send(user, f"your {item} is confirmed")

OrderService([EmailNotifier(), SmsNotifier()]).place_order("asha", "keyboard")
struct Notifier {                          // abstract base = a vtable with one slot
    virtual ~Notifier() = default;
    virtual void send(const std::string& user, const std::string& text) = 0;
};

struct EmailNotifier : Notifier {
    void send(const std::string& user, const std::string& text) override {
        std::cout << "[email] to " << user << ": " << text << "\n";
    }
};

struct SmsNotifier : Notifier {
    void send(const std::string& user, const std::string& text) override {
        std::cout << "[sms]   to " << user << ": " << text << "\n";
    }
};

class OrderService {
    std::vector<Notifier*> notifiers_;     // holds, does not own: the caller keeps them alive
public:                                    // (aggregation, the hollow diamond in the diagram)
    explicit OrderService(std::vector<Notifier*> n) : notifiers_(std::move(n)) {}
    void placeOrder(const std::string& user, const std::string& item) {
        std::cout << "order placed: " << item << "\n";
        for (Notifier* n : notifiers_) n->send(user, "your " + item + " is confirmed");
    }
};
struct notifier_ops {                           /* the same idea, by hand */
    void (*send)(const char *user, const char *text);
};

static void sms_send(const char *user, const char *text) { printf("[sms] to %s: %s\n", user, text); }
static const struct notifier_ops sms_ops = { .send = sms_send };

static void place_order(const struct notifier_ops *const *ops, int n,
                        const char *user, const char *item)
{
    char msg[128];
    snprintf(msg, sizeof msg, "your %s is confirmed", item);
    for (int i = 0; i < n; i++)
        ops[i]->send(user, msg);                /* indirect call = virtual dispatch */
}

All four compile and run; the full files are in lld-code/lld-start-index/ (Main.java, notifier.py, notifier.cpp, notifier.c). Look at the C tab: it is the same design. Java just writes the function-pointer table for you and checks at compile time that every slot is filled.

Interviewers love this move: "I put an interface here because channels are the part most likely to change." Naming where change will come from and putting an interface exactly there is most of LLD. The principle has a name: Open/Closed. The shape has a name: Strategy.
