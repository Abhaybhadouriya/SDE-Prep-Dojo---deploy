  1. Enums and data classes. JobState with its terminal states, and Task, the unit of work (a Command).
  2. One interface, its implementations. Schedule answers "when is the next run?"; once, fixed delay and fixed rate implement it.
  3. The core. submit admits a job under the lock and signals the dispatcher; nextReadyJob waits with a deadline until the head of the time queue is due.
  4. The driver. main runs priorities, delays, retries, a DAG, cancellation and the stress test.

How to read the full code

Reading order: Read Model.java (Task, Schedule, JobState, JobSpec), then TaskScheduler.submit, dispatchLoop, nextReadyJob and execute. Skip WorkerPool's shutdown and topologicalOrder on a first read.

import java.util.*;

/** What a job does. Command pattern: the scheduler only knows "run it". */
@FunctionalInterface
interface Task {
    void run() throws Exception;
}

/** Monotonic milliseconds. Injected so tests can control or fake time. */
@FunctionalInterface
interface Clock {
    long millis();
    Clock SYSTEM = () -> System.nanoTime() / 1_000_000;   // monotonic, never jumps back
}

/** When should the job run next? Strategy. Returns -1 for "never again". */
@FunctionalInterface

/** How many times to try, and how long to wait between tries (exponential backoff). */
record RetryPolicy(int maxAttempts, long baseBackoffMs) {   // C: record = an immutable struct (equals/hashCode/toString generated)
    RetryPolicy {
        if (maxAttempts < 1 || baseBackoffMs < 0) throw new IllegalArgumentException("bad retry policy");
    }
    static RetryPolicy none() { return new RetryPolicy(1, 0); }
    long backoffMs(int failedAttempt) { return baseBackoffMs << Math.min(failedAttempt - 1, 20); }
}

/** Immutable description of a job, built with a Builder. */
final class JobSpec {
    final String id;
    final Task task;
    final int priority;            // higher runs first among jobs that are due
    final long delayMs;            // not before now + delay
    final Schedule schedule;
    final Set<String> dependsOn;
    final RetryPolicy retry;

    private JobSpec(Builder b) {
        id = b.id; task = b.task; priority = b.priority; delayMs = b.delayMs;
        schedule = b.schedule; dependsOn = Set.copyOf(b.dependsOn); retry = b.retry;
    }

    static Builder of(String id, Task task) { return new Builder(id, task); }

    static final class Builder {
        private final String id;
        private final Task task;
        private int priority = 5;
        private long delayMs = 0;
        private Schedule schedule = Schedule.once();
        private final Set<String> dependsOn = new LinkedHashSet<>();
        private RetryPolicy retry = RetryPolicy.none();

        private Builder(String id, Task task) {
            if (id == null || id.isBlank()) throw new IllegalArgumentException("job id is required");
            this.id = id;
            this.task = Objects.requireNonNull(task, "task");
        }
        Builder priority(int p) { priority = p; return this; }
        Builder delay(long ms) { if (ms < 0) throw new IllegalArgumentException("delay < 0"); delayMs = ms; return this; }
        Builder schedule(Schedule s) { schedule = Objects.requireNonNull(s); return this; }
        Builder dependsOn(String... ids) { dependsOn.addAll(Arrays.asList(ids)); return this; }
        Builder retry(int maxAttempts, long baseBackoffMs) { retry = new RetryPolicy(maxAttempts, baseBackoffMs); return this; }
        JobSpec build() { return new JobSpec(this); }
    }
}

/** Mutable run-time record of one job. Every field is guarded by the scheduler's lock. */
final class Job {
    final JobSpec spec;
    final long seq;                 // submission order, the last tie-breaker
    final long notBefore;           // submit time + delay
    JobState state;
    long runAt;                     // when it may run (valid while SCHEDULED)
    int attempt;                    // attempts in the current run
    int runs;                       // finished executions (success or failure)
    int pendingDeps;
    String lastError;
    final List<Job> dependents = new ArrayList<>();

    Job(JobSpec spec, long seq, long notBefore) {
        this.spec = spec; this.seq = seq; this.notBefore = notBefore;
    }
}

/** Read-only snapshot handed to callers. */
record JobInfo(String id, JobState state, int runs, int attempt, String lastError) {}

/** Observer: hears every state change. Called outside the scheduler lock. */
@FunctionalInterface
interface JobListener {
    void onStateChange(String jobId, JobState from, JobState to);
}
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.locks.*;

/** A fixed set of worker threads fed by a queue. The scheduler never gives it more than n jobs at once. */
final class WorkerPool {
    private static final Runnable STOP = () -> { };
    private final BlockingQueue<Runnable> queue = new LinkedBlockingQueue<>();
    private final List<Thread> threads = new ArrayList<>();

    WorkerPool(int n, String name) {
        for (int i = 1; i <= n; i++) {
            Thread t = new Thread(this::loop, name + "-" + i);   // C: Type::method = a pointer to an existing function
            threads.add(t);
            t.start();
        }
    }

    void execute(Runnable r) { queue.add(r); }

    private void loop() {
        try {
            while (true) {
                Runnable r = queue.take();
                if (r == STOP) return;
                r.run();
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    /** FIFO queue: work already handed over finishes before the STOP markers are seen. */
    void shutdownAndJoin() throws InterruptedException {
        for (int i = 0; i < threads.size(); i++) queue.add(STOP);
        for (Thread t : threads) t.join();
    }
}

/**
 * In-memory task scheduler: delayed, periodic and dependent (DAG) jobs, priorities, retries,
 * cancellation, a bounded worker pool. One lock guards all job state; one dispatcher thread
 * moves due jobs from the time queue to the ready queue and hands them to free workers.
 */
final class TaskScheduler implements AutoCloseable {
    /** Not yet due: earliest run time first. */
    private static final Comparator<Job> BY_TIME =
            Comparator.comparingLong((Job j) -> j.runAt).thenComparingLong(j -> j.seq);   // C: builds the cmp() function you would pass to qsort
    /** Due: highest priority first, then the one that has waited longest, then submission order. */
    private static final Comparator<Job> BY_PRIORITY =
            Comparator.comparingInt((Job j) -> -j.spec.priority)
                      .thenComparingLong(j -> j.runAt)
                      .thenComparingLong(j -> j.seq);

    private final Clock clock;
    private final ReentrantLock lock = new ReentrantLock();
    private final Condition changed = lock.newCondition();   // dispatcher waits here
    private final Condition settled = lock.newCondition();   // awaitTerminal waits here
    private final PriorityQueue<Job> timeQueue = new PriorityQueue<>(BY_TIME);
    private final PriorityQueue<Job> readyQueue = new PriorityQueue<>(BY_PRIORITY);
    private final Map<String, Job> jobs = new HashMap<>();
    private final List<JobListener> listeners = new CopyOnWriteArrayList<>();
    private final Semaphore freeWorkers;
    private final WorkerPool pool;
    private final Thread dispatcher;
    private boolean running = true;
    private long nextSeq = 0;

    TaskScheduler(int workers, Clock clock) {
        if (workers < 1) throw new IllegalArgumentException("need at least one worker");
        this.clock = clock;
        this.freeWorkers = new Semaphore(workers);
        this.pool = new WorkerPool(workers, "worker");
        this.dispatcher = new Thread(this::dispatchLoop, "dispatcher");
        dispatcher.start();
    }

    TaskScheduler(int workers) { this(workers, Clock.SYSTEM); }

    void addListener(JobListener l) { listeners.add(l); }

    // ---------------------------------------------------------------- public API

    /** Admit one job. Its dependencies must already be known to the scheduler. */
    void submit(JobSpec spec) {
        List<Runnable> events = new ArrayList<>();
        lock.lock();
        try {
            checkOpen();
            validate(spec, Set.of());
            admit(spec, events);
        } finally {   // C: finally runs on every exit path: the goto out_unlock label
            lock.unlock();
        }
        fire(events);
    }

    /** Admit a whole graph, all or nothing. Rejects unknown ids, duplicates and cycles. */
    void submitGraph(List<JobSpec> specs) {
        List<Runnable> events = new ArrayList<>();
        lock.lock();
        try {
            checkOpen();
            Map<String, JobSpec> batch = new LinkedHashMap<>();
            for (JobSpec s : specs) {
                if (batch.put(s.id, s) != null) throw new IllegalArgumentException("duplicate job id " + s.id);
            }
            for (JobSpec s : specs) validate(s, batch.keySet());
            for (JobSpec s : topologicalOrder(batch)) admit(s, events);
        } finally {
            lock.unlock();
        }
        fire(events);
    }

    /** True if the job will not run (again). A running one-shot job cannot be stopped. */
    boolean cancel(String id) {
        List<Runnable> events = new ArrayList<>();
        boolean done;
        lock.lock();
        try {
            Job job = jobs.get(id);
            if (job == null || job.state.isTerminal()) {
                done = false;
            } else if (job.state == JobState.RUNNING && !job.spec.schedule.isRecurring()) {
                done = false;
            } else {
                move(job, JobState.CANCELLED, events);   // lazy removal: the queues skip it later
                skipDependents(job, events);
                changed.signal();
                done = true;
            }
        } finally {
            lock.unlock();
        }
        fire(events);
        return done;
    }

    JobInfo info(String id) {
        lock.lock();
        try {
            Job j = jobs.get(id);
            if (j == null) throw new NoSuchElementException("unknown job " + id);
            return new JobInfo(j.spec.id, j.state, j.runs, j.attempt, j.lastError);
        } finally {
            lock.unlock();
        }
    }

    /** Wait until every listed job is in a terminal state. */
    boolean awaitTerminal(Collection<String> ids, long timeoutMs) throws InterruptedException {
        long deadline = System.nanoTime() + TimeUnit.MILLISECONDS.toNanos(timeoutMs);
        lock.lock();
        try {
            while (true) {
                boolean all = true;
                for (String id : ids) {
                    Job j = jobs.get(id);
                    if (j == null || !j.state.isTerminal()) { all = false; break; }
                }
                if (all) return true;
                long left = deadline - System.nanoTime();
                if (left <= 0) return false;
                settled.awaitNanos(left);
            }
        } finally {
            lock.unlock();
        }
    }

    /** Stop dispatching, let running jobs finish, stop the workers. Queued jobs stay queued. */
    void shutdown() throws InterruptedException {
        lock.lock();
        try {
            running = false;
            changed.signalAll();
        } finally {
            lock.unlock();
        }
        dispatcher.interrupt();          // it may be blocked waiting for a free worker
        dispatcher.join();
        pool.shutdownAndJoin();
    }

    @Override public void close() throws InterruptedException { shutdown(); }

    // ---------------------------------------------------------------- admission (lock held)

    private void checkOpen() {
        if (!running) throw new IllegalStateException("scheduler is shut down");
    }

    private void validate(JobSpec s, Set<String> batchIds) {
        if (jobs.containsKey(s.id)) throw new IllegalArgumentException("duplicate job id " + s.id);
        if (s.schedule.isRecurring() && !s.dependsOn.isEmpty())
            throw new IllegalArgumentException("recurring job " + s.id + " cannot have dependencies");
        for (String d : s.dependsOn) {
            Job parent = jobs.get(d);
            if (parent == null && !batchIds.contains(d))
                throw new IllegalArgumentException("unknown dependency " + d + " of " + s.id);
            if (parent != null && parent.spec.schedule.isRecurring())
                throw new IllegalArgumentException(s.id + " cannot depend on recurring job " + d);
        }
    }

    /** Kahn's algorithm over the batch; edges to already-admitted jobs are ignored here. */
    private List<JobSpec> topologicalOrder(Map<String, JobSpec> batch) {
        Map<String, Integer> indegree = new HashMap<>();
        Map<String, List<String>> children = new HashMap<>();
        for (JobSpec s : batch.values()) {
            indegree.putIfAbsent(s.id, 0);
            for (String d : s.dependsOn) {
                if (!batch.containsKey(d)) continue;
                indegree.merge(s.id, 1, Integer::sum);
                children.computeIfAbsent(d, k -> new ArrayList<>()).add(s.id);   // C: look up k; if missing, allocate, insert and return it
            }
        }
        Deque<String> ready = new ArrayDeque<>();
        for (JobSpec s : batch.values()) if (indegree.get(s.id) == 0) ready.add(s.id);
        List<JobSpec> order = new ArrayList<>();
        while (!ready.isEmpty()) {
            String id = ready.poll();
            order.add(batch.get(id));
            for (String c : children.getOrDefault(id, List.of())) {
                if (indegree.merge(c, -1, Integer::sum) == 0) ready.add(c);
            }
        }
        if (order.size() != batch.size()) {
            List<String> stuck = new ArrayList<>();
            for (String id : batch.keySet()) if (indegree.get(id) > 0) stuck.add(id);
            throw new IllegalArgumentException("cycle among " + stuck);
        }
        return order;
    }

    private void admit(JobSpec spec, List<Runnable> events) {
        Job job = new Job(spec, ++nextSeq, clock.millis() + spec.delayMs);
        jobs.put(spec.id, job);
        boolean doomed = false;
        for (String d : spec.dependsOn) {
            JobState ps = jobs.get(d).state;
            if (ps == JobState.FAILED || ps == JobState.CANCELLED || ps == JobState.SKIPPED) doomed = true;
        }
        if (doomed) { move(job, JobState.SKIPPED, events); return; }
        for (String d : spec.dependsOn) {
            Job parent = jobs.get(d);
            if (parent.state != JobState.SUCCEEDED) { parent.dependents.add(job); job.pendingDeps++; }
        }
        if (job.pendingDeps > 0) move(job, JobState.WAITING, events);
        else schedule(job, job.notBefore, events);
    }

    // ---------------------------------------------------------------- state changes (lock held)

    private void schedule(Job job, long at, List<Runnable> events) {
        job.runAt = at;
        move(job, JobState.SCHEDULED, events);
        timeQueue.add(job);
        changed.signal();
    }

    private void move(Job job, JobState to, List<Runnable> events) {
        JobState from = job.state;
        job.state = to;
        if (to.isTerminal()) settled.signalAll();
        String id = job.spec.id;
        events.add(() -> { for (JobListener l : listeners) l.onStateChange(id, from, to); });
    }

    private void releaseDependents(Job done, long now, List<Runnable> events) {
        for (Job d : done.dependents) {
            if (d.state == JobState.WAITING && --d.pendingDeps == 0) {
                schedule(d, Math.max(now, d.notBefore), events);
            }
        }
    }

    private void skipDependents(Job root, List<Runnable> events) {
        for (Job d : root.dependents) {
            if (d.state == JobState.WAITING) {
                move(d, JobState.SKIPPED, events);
                skipDependents(d, events);
            }
        }
    }

    // ---------------------------------------------------------------- dispatch and execution

    private void dispatchLoop() {
        try {
            while (true) {
                freeWorkers.acquire();             // backpressure: only pick a job when a worker is free
                Job job;
                List<Runnable> events = new ArrayList<>();
                lock.lock();
                try {
                    job = nextReadyJob();
                    if (job == null) { freeWorkers.release(); return; }
                    job.attempt++;
                    move(job, JobState.RUNNING, events);
                } finally {
                    lock.unlock();
                }
                fire(events);
                pool.execute(() -> execute(job));
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();    // shutdown
        }
    }

    /** Lock held. Blocks until a job is due and ready, or returns null on shutdown. */
    private Job nextReadyJob() throws InterruptedException {
        while (running) {
            long now = clock.millis();
            while (!timeQueue.isEmpty() && timeQueue.peek().runAt <= now) {
                Job j = timeQueue.poll();
                if (j.state == JobState.SCHEDULED) readyQueue.add(j);   // drop cancelled ones
            }
            Job j = readyQueue.poll();
            if (j != null) {
                if (j.state == JobState.SCHEDULED) return j;
                continue;                                             // cancelled after it became ready
            }
            if (timeQueue.isEmpty()) changed.await();
            else changed.await(timeQueue.peek().runAt - now, TimeUnit.MILLISECONDS);
        }
        return null;
    }

    /** Runs on a worker thread. The task body runs without the lock. */
    private void execute(Job job) {
        Exception error = null;
        try {
            job.spec.task.run();
        } catch (Exception e) {
            error = e;
        }
        List<Runnable> events = new ArrayList<>();
        lock.lock();
        try {
            long now = clock.millis();
            job.runs++;
            if (job.state == JobState.CANCELLED) {
                // cancelled while running: keep CANCELLED, do not reschedule
            } else if (error == null) {
                long next = job.spec.schedule.next(job.runAt, now);
                job.attempt = 0;
                if (next >= 0) {
                    schedule(job, next, events);
                } else {
                    move(job, JobState.SUCCEEDED, events);
                    releaseDependents(job, now, events);
                }
            } else {
                job.lastError = error.getMessage();
                if (job.attempt < job.spec.retry.maxAttempts()) {
                    schedule(job, now + job.spec.retry.backoffMs(job.attempt), events);
                } else {
                    move(job, JobState.FAILED, events);
                    skipDependents(job, events);
                }
            }
        } finally {
            lock.unlock();
            freeWorkers.release();
        }
        fire(events);
    }

    private void fire(List<Runnable> events) {
        for (Runnable e : events) e.run();
    }
}

public class Main {
    static int checks = 0;

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    static Task record(List<String> log, String name) {
        return () -> log.add(name);
    }

    public static void main(String[] args) throws Exception {
        delayedJobs();
        priorities();
        periodic();

        stress();

    static void delayedJobs() throws Exception {
        System.out.println("== 1. delayed jobs run in time order ==");
        List<String> ran = Collections.synchronizedList(new ArrayList<>());
        try (TaskScheduler s = new TaskScheduler(2)) {
            s.submit(JobSpec.of("late", record(ran, "late")).delay(300).build());
            s.submit(JobSpec.of("early", record(ran, "early")).delay(100).build());
            s.submit(JobSpec.of("mid", record(ran, "mid")).delay(200).build());
            check(s.info("late").state() == JobState.SCHEDULED, "late is SCHEDULED before its time");
            check(s.awaitTerminal(List.of("late", "early", "mid"), 3000), "delayed jobs finish");
        }
        System.out.println("  submitted late(+300ms), early(+100ms), mid(+200ms)");
        System.out.println("  ran: " + String.join(", ", ran));
        check(ran.equals(List.of("early", "mid", "late")), "time order");
    }

    static void priorities() throws Exception {
        System.out.println("== 2. priority decides among due jobs (1 worker) ==");
        List<String> ran = Collections.synchronizedList(new ArrayList<>());
        CountDownLatch gate = new CountDownLatch(1);
        try (TaskScheduler s = new TaskScheduler(1)) {
            s.submit(JobSpec.of("gate", gate::await).build());          // keeps the only worker busy
            s.submit(JobSpec.of("bulk", record(ran, "bulk(p1)")).priority(1).build());
            s.submit(JobSpec.of("normal", record(ran, "normal(p5)")).priority(5).build());
            s.submit(JobSpec.of("urgent", record(ran, "urgent(p9)")).priority(9).build());
            gate.countDown();
            check(s.awaitTerminal(List.of("bulk", "normal", "urgent"), 3000), "priority jobs finish");
        }
        System.out.println("  ran: " + String.join(", ", ran));
        check(ran.equals(List.of("urgent(p9)", "normal(p5)", "bulk(p1)")), "priority order");
    }

    static void periodic() throws Exception {
        System.out.println("== 3. fixed-delay periodic job, cancelled after 3 runs ==");
        AtomicInteger beats = new AtomicInteger();
        CountDownLatch three = new CountDownLatch(3);
        try (TaskScheduler s = new TaskScheduler(2)) {
            s.submit(JobSpec.of("heartbeat", () -> { beats.incrementAndGet(); three.countDown(); })
                    .schedule(Schedule.fixedDelay(150)).build());
            check(three.await(3, TimeUnit.SECONDS), "three heartbeats");
            check(s.cancel("heartbeat"), "cancel a periodic job");
            Thread.sleep(400);                                        // would allow 2 more beats
            check(beats.get() == 3, "no run after cancel");
            check(s.info("heartbeat").state() == JobState.CANCELLED, "state CANCELLED");
            check(!s.cancel("heartbeat"), "second cancel returns false");
            System.out.println("  heartbeat ran " + beats.get() + " times, state " + s.info("heartbeat").state());
        }
    }

    static void retries() throws Exception {
        System.out.println("== 4. retry with exponential backoff, failure skips dependents ==");
        List<String> flakyStates = Collections.synchronizedList(new ArrayList<>());
        AtomicInteger tries = new AtomicInteger();
        try (TaskScheduler s = new TaskScheduler(2)) {
            s.addListener((id, from, to) -> { if (id.equals("flaky")) flakyStates.add(to.name()); });
            s.submit(JobSpec.of("flaky", () -> {
                if (tries.incrementAndGet() < 3) throw new IllegalStateException("timeout #" + tries.get());
            }).retry(3, 20).build());
            s.submit(JobSpec.of("broken", () -> { throw new IllegalStateException("disk full"); })
                    .retry(2, 10).build());
            s.submit(JobSpec.of("report", () -> { }).dependsOn("broken").build());
            s.submit(JobSpec.of("email", () -> { }).dependsOn("report").build());
            check(s.awaitTerminal(List.of("flaky", "broken", "report", "email"), 3000), "retry jobs settle");
            System.out.println("  flaky: " + String.join(" -> ", flakyStates));
            JobInfo b = s.info("broken");
            System.out.println("  broken: " + b.state() + " after " + b.runs() + " attempts, last error '" + b.lastError() + "'");
            System.out.println("  report: " + s.info("report").state() + ", email: " + s.info("email").state());
            check(tries.get() == 3 && s.info("flaky").state() == JobState.SUCCEEDED, "flaky succeeds on try 3");
            check(b.state() == JobState.FAILED && b.runs() == 2, "broken fails after 2 attempts");
            check(s.info("report").state() == JobState.SKIPPED, "dependent skipped");
            check(s.info("email").state() == JobState.SKIPPED, "skip is transitive");
            s.submit(JobSpec.of("late-child", () -> { }).dependsOn("broken").build());
            check(s.info("late-child").state() == JobState.SKIPPED, "child of a failed job is skipped at once");
        }
    }

    static void dag() throws Exception {
        System.out.println("== 5. DAG pipeline (Rubrik-style) on 1 worker ==");
        // A -> B -> C -> G,  A -> E -> F -> G,  K -> L -> M
        String[][] edges = {{"A"}, {"B", "A"}, {"C", "B"}, {"E", "A"}, {"F", "E"}, {"G", "C", "F"},
                            {"K"}, {"L", "K"}, {"M", "L"}};
        List<String> order = Collections.synchronizedList(new ArrayList<>());
        List<JobSpec> specs = new ArrayList<>();
        for (String[] e : edges) {
            String id = e[0];
            specs.add(JobSpec.of(id, () -> { Thread.sleep(5); order.add(id); })
                    .dependsOn(Arrays.copyOfRange(e, 1, e.length)).build());
        }
        try (TaskScheduler s = new TaskScheduler(1)) {
            s.submitGraph(specs);
            check(s.info("G").state() == JobState.WAITING || s.info("G").state() == JobState.SCHEDULED, "G waits");
            List<String> ids = new ArrayList<>();
            for (String[] e : edges) ids.add(e[0]);
            check(s.awaitTerminal(ids, 5000), "pipeline finishes");
        }
        System.out.println("  order: " + String.join(" ", order));
        boolean valid = true;
        for (String[] e : edges)
            for (int i = 1; i < e.length; i++)
                if (order.indexOf(e[i]) > order.indexOf(e[0])) valid = false;
        System.out.println("  every job ran after all its parents: " + valid);
        check(valid && order.size() == 9, "topological order");
        check(order.equals(List.of("A", "K", "B", "E", "L", "C", "F", "M", "G")), "FIFO among equal priority");
    }

    static void validation() throws Exception {
        System.out.println("== 6. validation ==");
        try (TaskScheduler s = new TaskScheduler(1)) {
            s.submit(JobSpec.of("x", () -> { }).delay(10_000).build());
            String[] cases = {"cycle", "duplicate", "unknown", "recurring-with-deps", "depends-on-recurring"};
            for (String c : cases) {
                try {
                    switch (c) {
                        case "cycle" -> s.submitGraph(List.of(   // C: case X -> ... is a switch arm with no fall-through
                                JobSpec.of("P", () -> { }).dependsOn("R").build(),
                                JobSpec.of("Q", () -> { }).dependsOn("P").build(),
                                JobSpec.of("R", () -> { }).dependsOn("Q").build()));
                        case "duplicate" -> s.submit(JobSpec.of("x", () -> { }).build());
                        case "unknown" -> s.submit(JobSpec.of("y", () -> { }).dependsOn("nope").build());
                        case "recurring-with-deps" -> s.submit(JobSpec.of("z", () -> { })
                                .schedule(Schedule.fixedRate(100)).dependsOn("x").build());
                        default -> {
                            s.submit(JobSpec.of("tick", () -> { }).delay(10_000).schedule(Schedule.fixedRate(100)).build());
                            s.submit(JobSpec.of("w", () -> { }).dependsOn("tick").build());
                        }
                    }
                    check(false, c + " should be rejected");
                } catch (IllegalArgumentException e) {
                    System.out.println("  rejected " + c + ": " + e.getMessage());
                    checks++;
                }
            }
            try {
                s.info("P");
                check(false, "graph with a cycle must not be half-admitted");
            } catch (NoSuchElementException e) {
                checks++;
            }
            check(s.cancel("x") && s.info("x").state() == JobState.CANCELLED, "cancel a scheduled job");
        }
    }

    static void stress() throws Exception {
        System.out.println("== 7. stress: 8 threads x 1000 jobs, 4 workers ==");
        int producers = 8, perProducer = 1000, workers = 4;
        ConcurrentHashMap<String, Integer> runs = new ConcurrentHashMap<>();
        AtomicInteger inFlight = new AtomicInteger(), maxInFlight = new AtomicInteger();
        List<String> ids = Collections.synchronizedList(new ArrayList<>());
        long start = System.nanoTime();
        try (TaskScheduler s = new TaskScheduler(workers)) {
            ExecutorService submitters = Executors.newFixedThreadPool(producers);
            for (int p = 0; p < producers; p++) {
                int pid = p;
                submitters.execute(() -> {
                    Random rnd = new Random(pid);
                    for (int i = 0; i < perProducer; i++) {
                        String id = "p" + pid + "-" + i;
                        ids.add(id);
                        s.submit(JobSpec.of(id, () -> {
                            int now = inFlight.incrementAndGet();
                            maxInFlight.accumulateAndGet(now, Math::max);
                            runs.merge(id, 1, Integer::sum);   // C: counts[k] += v, inserting v if k is absent
                            inFlight.decrementAndGet();
                        }).priority(rnd.nextInt(10)).delay(rnd.nextInt(5)).build());
                    }
                });
            }
            submitters.shutdown();
            check(submitters.awaitTermination(10, TimeUnit.SECONDS), "submitters finish");
            check(s.awaitTerminal(ids, 20_000), "all stress jobs finish");
        }
        long ms = (System.nanoTime() - start) / 1_000_000;
        boolean once = runs.size() == producers * perProducer && runs.values().stream().allMatch(v -> v == 1);   // C: stream() = a for-loop written as filter/map/collect
        System.out.println("  " + runs.size() + " jobs ran exactly once: " + once
                + "; max in flight " + maxInFlight.get() + " (limit " + workers + ")");
        System.out.println("  (took " + (ms < 2000 ? "under 2 s" : ms + " ms") + ")");
        check(once, "every job ran exactly once");
        check(maxInFlight.get() <= workers, "never more than " + workers + " jobs at once");
    }
}
