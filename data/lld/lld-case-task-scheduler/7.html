APIs and interfaces

Write these on the board before any body. Note what is missing: there is no getJob() returning the live object, and no setter on a job. Clients get immutable snapshots.

// service API
void    submit(JobSpec spec);                         // IllegalArgumentException: duplicate id, unknown dep, recurring + deps
void    submitGraph(List<JobSpec> specs);            // all or nothing; rejects cycles
boolean cancel(String id);                            // true if it will not run (again)
JobInfo info(String id);                              // snapshot: state, runs, attempt, lastError
boolean awaitTerminal(Collection<String> ids, long timeoutMs);
void    shutdown();                                   // stop dispatching, finish running jobs
void    addListener(JobListener l);

// building a job
JobSpec.of("backup", task).priority(9).delay(100)
       .schedule(Schedule.fixedRate(60_000)).retry(3, 500).dependsOn("snapshot").build();

// extension points
interface Task        { void run() throws Exception; }
interface Schedule    { long next(long plannedStart, long finishedAt); default boolean isRecurring() { return true; } }
interface JobListener { void onStateChange(String id, JobState from, JobState to); }
interface Clock       { long millis(); }
