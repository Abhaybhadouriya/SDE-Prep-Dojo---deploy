"""In-memory task scheduler: delayed, periodic and dependent (DAG) jobs, priorities, retries,
cancellation and a bounded worker pool. Mirrors the Java version.
"""
from __future__ import annotations

import enum
import heapq
import queue
import threading
import time
from collections import deque
from dataclasses import dataclass, field
from typing import Callable, Dict, Iterable, List, Optional, Set


def monotonic_ms() -> int:
    return time.monotonic_ns() // 1_000_000


class JobState(enum.Enum):
    WAITING = "WAITING"
    SCHEDULED = "SCHEDULED"
    RUNNING = "RUNNING"
    SUCCEEDED = "SUCCEEDED"
    FAILED = "FAILED"
    CANCELLED = "CANCELLED"
    SKIPPED = "SKIPPED"

    @property
    def terminal(self) -> bool:
        return self in (JobState.SUCCEEDED, JobState.FAILED, JobState.CANCELLED, JobState.SKIPPED)


class Schedule:
    """Strategy: when should the job run next? None means never again."""
    recurring = True

    def next(self, planned_start: int, finished_at: int) -> Optional[int]:
        raise NotImplementedError


class Once(Schedule):
    recurring = False

    def next(self, planned_start, finished_at):
        return None


class FixedRate(Schedule):
    def __init__(self, period_ms: int):
        if period_ms <= 0:
            raise ValueError("period must be > 0")
        self.period = period_ms

    def next(self, planned_start, finished_at):
        return max(planned_start + self.period, finished_at)


class FixedDelay(Schedule):
    def __init__(self, delay_ms: int):
        if delay_ms <= 0:
            raise ValueError("delay must be > 0")
        self.delay = delay_ms

    def next(self, planned_start, finished_at):
        return finished_at + self.delay


@dataclass(frozen=True)
class RetryPolicy:
    max_attempts: int = 1
    base_backoff_ms: int = 0

    def __post_init__(self):
        if self.max_attempts < 1 or self.base_backoff_ms < 0:
            raise ValueError("bad retry policy")

    def backoff_ms(self, failed_attempt: int) -> int:
        return self.base_backoff_ms << min(failed_attempt - 1, 20)


@dataclass(frozen=True)
class JobSpec:
    id: str
    task: Callable[[], None]
    priority: int = 5                      # higher runs first among due jobs
    delay_ms: int = 0
    schedule: Schedule = field(default_factory=Once)
    depends_on: frozenset = frozenset()
    retry: RetryPolicy = RetryPolicy()

    def __post_init__(self):
        if not self.id:
            raise ValueError("job id is required")
        if self.delay_ms < 0:
            raise ValueError("delay < 0")
        object.__setattr__(self, "depends_on", frozenset(self.depends_on))


class _Job:
    """Run-time record. Every field is guarded by the scheduler lock."""

    def __init__(self, spec: JobSpec, seq: int, not_before: int):
        self.spec = spec
        self.seq = seq
        self.not_before = not_before
        self.state: Optional[JobState] = None
        self.run_at = 0
        self.attempt = 0
        self.runs = 0
        self.pending_deps = 0
        self.last_error: Optional[str] = None
        self.dependents: List[_Job] = []


@dataclass(frozen=True)
class JobInfo:
    id: str
    state: JobState
    runs: int
    attempt: int
    last_error: Optional[str]


Listener = Callable[[str, Optional[JobState], JobState], None]


class _WorkerPool:
    _STOP = object()

    def __init__(self, n: int):
        self._q: "queue.Queue" = queue.Queue()
        self._threads = [threading.Thread(target=self._loop, name=f"worker-{i + 1}", daemon=True)
                         for i in range(n)]
        for t in self._threads:
            t.start()

    def execute(self, fn: Callable[[], None]) -> None:
        self._q.put(fn)

    def _loop(self) -> None:
        while True:
            fn = self._q.get()
            if fn is self._STOP:
                return
            fn()

    def shutdown_and_join(self) -> None:
        for _ in self._threads:
            self._q.put(self._STOP)
        for t in self._threads:
            t.join()


class TaskScheduler:
    def __init__(self, workers: int, clock: Callable[[], int] = monotonic_ms):
        if workers < 1:
            raise ValueError("need at least one worker")
        self._clock = clock
        self._lock = threading.Lock()
        self._changed = threading.Condition(self._lock)    # dispatcher waits here
        self._settled = threading.Condition(self._lock)    # await_terminal waits here
        self._time_q: list = []      # (run_at, seq, job)
        self._ready_q: list = []     # (-priority, run_at, seq, job)
        self._jobs: Dict[str, _Job] = {}
        self._listeners: List[Listener] = []
        self._free_workers = threading.Semaphore(workers)
        self._pool = _WorkerPool(workers)
        self._running = True
        self._seq = 0
        self._dispatcher = threading.Thread(target=self._dispatch_loop, name="dispatcher", daemon=True)
        self._dispatcher.start()

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.shutdown()

    def add_listener(self, listener: Listener) -> None:
        self._listeners.append(listener)

    # ------------------------------------------------------------ public API
    def submit(self, spec: JobSpec) -> None:
        events: list = []
        with self._lock:
            self._check_open()
            self._validate(spec, set())
            self._admit(spec, events)
        self._fire(events)

    def submit_graph(self, specs: Iterable[JobSpec]) -> None:
        events: list = []
        with self._lock:
            self._check_open()
            batch: Dict[str, JobSpec] = {}
            for s in specs:
                if s.id in batch:
                    raise ValueError(f"duplicate job id {s.id}")
                batch[s.id] = s
            for s in batch.values():
                self._validate(s, set(batch))
            for s in self._topological_order(batch):
                self._admit(s, events)
        self._fire(events)

    def cancel(self, job_id: str) -> bool:
        events: list = []
        with self._lock:
            job = self._jobs.get(job_id)
            if job is None or job.state.terminal:
                done = False
            elif job.state is JobState.RUNNING and not job.spec.schedule.recurring:
                done = False
            else:
                self._move(job, JobState.CANCELLED, events)
                self._skip_dependents(job, events)
                self._changed.notify()
                done = True
        self._fire(events)
        return done

    def info(self, job_id: str) -> JobInfo:
        with self._lock:
            j = self._jobs.get(job_id)
            if j is None:
                raise KeyError(f"unknown job {job_id}")
            return JobInfo(j.spec.id, j.state, j.runs, j.attempt, j.last_error)

    def await_terminal(self, ids: Iterable[str], timeout_s: float) -> bool:
        ids = list(ids)
        deadline = time.monotonic() + timeout_s
        with self._lock:
            while True:
                if all(i in self._jobs and self._jobs[i].state.terminal for i in ids):
                    return True
                left = deadline - time.monotonic()
                if left <= 0:
                    return False
                self._settled.wait(left)

    def shutdown(self) -> None:
        with self._lock:
            self._running = False
            self._changed.notify_all()
        self._free_workers.release()      # unblock a dispatcher waiting for a worker
        self._dispatcher.join()
        self._pool.shutdown_and_join()

    # ------------------------------------------------------------ admission (lock held)
    def _check_open(self) -> None:
        if not self._running:
            raise RuntimeError("scheduler is shut down")

    def _validate(self, s: JobSpec, batch_ids: Set[str]) -> None:
        if s.id in self._jobs:
            raise ValueError(f"duplicate job id {s.id}")
        if s.schedule.recurring and s.depends_on:
            raise ValueError(f"recurring job {s.id} cannot have dependencies")
        for d in s.depends_on:
            parent = self._jobs.get(d)
            if parent is None and d not in batch_ids:
                raise ValueError(f"unknown dependency {d} of {s.id}")
            if parent is not None and parent.spec.schedule.recurring:
                raise ValueError(f"{s.id} cannot depend on recurring job {d}")

    @staticmethod
    def _topological_order(batch: Dict[str, JobSpec]) -> List[JobSpec]:
        indegree = {i: 0 for i in batch}
        children: Dict[str, List[str]] = {}
        for s in batch.values():
            for d in sorted(s.depends_on):
                if d in batch:
                    indegree[s.id] += 1
                    children.setdefault(d, []).append(s.id)
        ready = deque(i for i in batch if indegree[i] == 0)
        order: List[JobSpec] = []
        while ready:
            i = ready.popleft()
            order.append(batch[i])
            for c in children.get(i, []):
                indegree[c] -= 1
                if indegree[c] == 0:
                    ready.append(c)
        if len(order) != len(batch):
            stuck = [i for i in batch if indegree[i] > 0]
            raise ValueError(f"cycle among {stuck}")
        return order

    def _admit(self, spec: JobSpec, events: list) -> None:
        self._seq += 1
        job = _Job(spec, self._seq, self._clock() + spec.delay_ms)
        self._jobs[spec.id] = job
        bad = (JobState.FAILED, JobState.CANCELLED, JobState.SKIPPED)
        if any(self._jobs[d].state in bad for d in spec.depends_on):
            self._move(job, JobState.SKIPPED, events)
            return
        for d in spec.depends_on:
            parent = self._jobs[d]
            if parent.state is not JobState.SUCCEEDED:
                parent.dependents.append(job)
                job.pending_deps += 1
        if job.pending_deps:
            self._move(job, JobState.WAITING, events)
        else:
            self._schedule(job, job.not_before, events)

    # ------------------------------------------------------------ state changes (lock held)
    def _schedule(self, job: _Job, at: int, events: list) -> None:
        job.run_at = at
        self._move(job, JobState.SCHEDULED, events)
        heapq.heappush(self._time_q, (at, job.seq, job))
        self._changed.notify()

    def _move(self, job: _Job, to: JobState, events: list) -> None:
        frm, job.state = job.state, to
        if to.terminal:
            self._settled.notify_all()
        events.append((job.spec.id, frm, to))

    def _release_dependents(self, done: _Job, now: int, events: list) -> None:
        for d in done.dependents:
            if d.state is JobState.WAITING:
                d.pending_deps -= 1
                if d.pending_deps == 0:
                    self._schedule(d, max(now, d.not_before), events)

    def _skip_dependents(self, root: _Job, events: list) -> None:
        for d in root.dependents:
            if d.state is JobState.WAITING:
                self._move(d, JobState.SKIPPED, events)
                self._skip_dependents(d, events)

    # ------------------------------------------------------------ dispatch and execution
    def _dispatch_loop(self) -> None:
        while True:
            self._free_workers.acquire()          # backpressure: pick only when a worker is free
            events: list = []
            with self._lock:
                job = self._next_ready_job()
                if job is None:
                    return
                job.attempt += 1
                self._move(job, JobState.RUNNING, events)
            self._fire(events)
            self._pool.execute(lambda j=job: self._execute(j))

    def _next_ready_job(self) -> Optional[_Job]:
        while self._running:
            now = self._clock()
            while self._time_q and self._time_q[0][0] <= now:
                _, seq, j = heapq.heappop(self._time_q)
                if j.state is JobState.SCHEDULED:
                    heapq.heappush(self._ready_q, (-j.spec.priority, j.run_at, seq, j))
            if self._ready_q:
                j = heapq.heappop(self._ready_q)[3]
                if j.state is JobState.SCHEDULED:
                    return j
                continue
            if self._time_q:
                self._changed.wait((self._time_q[0][0] - now) / 1000.0)
            else:
                self._changed.wait()
        return None

    def _execute(self, job: _Job) -> None:
        error: Optional[BaseException] = None
        try:
            job.spec.task()
        except Exception as e:  # noqa: BLE001 - a job may fail in any way
            error = e
        events: list = []
        try:
            with self._lock:
                now = self._clock()
                job.runs += 1
                if job.state is JobState.CANCELLED:
                    pass                                   # cancelled while running
                elif error is None:
                    nxt = job.spec.schedule.next(job.run_at, now)
                    job.attempt = 0
                    if nxt is not None:
                        self._schedule(job, nxt, events)
                    else:
                        self._move(job, JobState.SUCCEEDED, events)
                        self._release_dependents(job, now, events)
                else:
                    job.last_error = str(error)
                    if job.attempt < job.spec.retry.max_attempts:
                        self._schedule(job, now + job.spec.retry.backoff_ms(job.attempt), events)
                    else:
                        self._move(job, JobState.FAILED, events)
                        self._skip_dependents(job, events)
        finally:
            self._free_workers.release()
        self._fire(events)

    def _fire(self, events: list) -> None:
        for job_id, frm, to in events:
            for listener in list(self._listeners):
                listener(job_id, frm, to)

import random
import threading
import time

from task_scheduler import FixedDelay, JobSpec, JobState, RetryPolicy, TaskScheduler

checks = 0


def check(ok: bool, what: str) -> None:
    global checks
    if not ok:
        raise AssertionError("FAILED: " + what)
    checks += 1


def delayed_jobs():
    print("== 1. delayed jobs run in time order ==")
    ran = []
    with TaskScheduler(2) as s:
        for name, delay in (("late", 300), ("early", 100), ("mid", 200)):
            s.submit(JobSpec(name, lambda n=name: ran.append(n), delay_ms=delay))
        check(s.info("late").state is JobState.SCHEDULED, "late is SCHEDULED before its time")
        check(s.await_terminal(["late", "early", "mid"], 3), "delayed jobs finish")
    print("  ran:", ", ".join(ran))
    check(ran == ["early", "mid", "late"], "time order")


def priorities():
    print("== 2. priority decides among due jobs (1 worker) ==")
    ran = []
    gate = threading.Event()
    with TaskScheduler(1) as s:
        s.submit(JobSpec("gate", gate.wait))
        for name, p in (("bulk", 1), ("normal", 5), ("urgent", 9)):
            s.submit(JobSpec(name, lambda n=f"{name}(p{p})": ran.append(n), priority=p))
        gate.set()
        check(s.await_terminal(["bulk", "normal", "urgent"], 3), "priority jobs finish")
    print("  ran:", ", ".join(ran))
    check(ran == ["urgent(p9)", "normal(p5)", "bulk(p1)"], "priority order")


def periodic():
    print("== 3. fixed-delay periodic job, cancelled after 3 runs ==")
    beats = []
    three = threading.Event()

    def beat():
        beats.append(1)
        if len(beats) == 3:
            three.set()

    with TaskScheduler(2) as s:
        s.submit(JobSpec("heartbeat", beat, schedule=FixedDelay(150)))
        check(three.wait(3), "three heartbeats")
        check(s.cancel("heartbeat"), "cancel a periodic job")
        time.sleep(0.4)
        check(len(beats) == 3, "no run after cancel")
        check(s.info("heartbeat").state is JobState.CANCELLED, "state CANCELLED")
        print(f"  heartbeat ran {len(beats)} times, state {s.info('heartbeat').state.value}")


def retries():
    print("== 4. retry with exponential backoff, failure skips dependents ==")
    states = []
    tries = [0]

    def flaky():
        tries[0] += 1
        if tries[0] < 3:
            raise RuntimeError(f"timeout #{tries[0]}")

    def broken():
        raise RuntimeError("disk full")

    with TaskScheduler(2) as s:
        s.add_listener(lambda i, frm, to: states.append(to.value) if i == "flaky" else None)
        s.submit(JobSpec("flaky", flaky, retry=RetryPolicy(3, 20)))
        s.submit(JobSpec("broken", broken, retry=RetryPolicy(2, 10)))
        s.submit(JobSpec("report", lambda: None, depends_on={"broken"}))
        s.submit(JobSpec("email", lambda: None, depends_on={"report"}))
        check(s.await_terminal(["flaky", "broken", "report", "email"], 3), "retry jobs settle")
        b = s.info("broken")
        print("  flaky:", " -> ".join(states))
        print(f"  broken: {b.state.value} after {b.runs} attempts, last error '{b.last_error}'")
        print(f"  report: {s.info('report').state.value}, email: {s.info('email').state.value}")
        check(tries[0] == 3 and s.info("flaky").state is JobState.SUCCEEDED, "flaky succeeds on try 3")
        check(b.state is JobState.FAILED and b.runs == 2, "broken fails after 2 attempts")
        check(s.info("email").state is JobState.SKIPPED, "skip is transitive")


def dag():
    print("== 5. DAG pipeline (Rubrik-style) on 1 worker ==")
    edges = [("A",), ("B", "A"), ("C", "B"), ("E", "A"), ("F", "E"), ("G", "C", "F"),
             ("K",), ("L", "K"), ("M", "L")]
    order = []

    def work(name):
        time.sleep(0.005)
        order.append(name)

    specs = [JobSpec(e[0], lambda n=e[0]: work(n), depends_on=set(e[1:])) for e in edges]
    with TaskScheduler(1) as s:
        s.submit_graph(specs)
        check(s.await_terminal([e[0] for e in edges], 5), "pipeline finishes")
    valid = all(order.index(p) < order.index(e[0]) for e in edges for p in e[1:])
    print("  order:", " ".join(order))
    print("  every job ran after all its parents:", valid)
    check(valid and len(order) == 9, "topological order")


def validation():
    print("== 6. validation ==")
    with TaskScheduler(1) as s:
        s.submit(JobSpec("x", lambda: None, delay_ms=10_000))
        cases = {
            "cycle": lambda: s.submit_graph([JobSpec("P", lambda: None, depends_on={"R"}),
                                             JobSpec("Q", lambda: None, depends_on={"P"}),
                                             JobSpec("R", lambda: None, depends_on={"Q"})]),
            "duplicate": lambda: s.submit(JobSpec("x", lambda: None)),
            "unknown": lambda: s.submit(JobSpec("y", lambda: None, depends_on={"nope"})),
            "recurring-with-deps": lambda: s.submit(
                JobSpec("z", lambda: None, schedule=FixedDelay(100), depends_on={"x"})),
        }
        for name, call in cases.items():
            try:
                call()
                check(False, name + " should be rejected")
            except ValueError as e:
                print(f"  rejected {name}: {e}")
                check(True, name)
        try:
            s.info("P")
            check(False, "no half-admitted graph")
        except KeyError:
            check(True, "no half-admitted graph")


def stress():
    print("== 7. stress: 8 threads x 250 jobs, 4 workers ==")
    producers, per, workers = 8, 250, 4
    runs = {}
    runs_lock = threading.Lock()
    in_flight = [0, 0]           # current, max
    ids = [f"p{p}-{i}" for p in range(producers) for i in range(per)]

    def body(job_id):
        with runs_lock:
            in_flight[0] += 1
            in_flight[1] = max(in_flight[1], in_flight[0])
            runs[job_id] = runs.get(job_id, 0) + 1
        time.sleep(0.001)            # a little work, outside the lock
        with runs_lock:
            in_flight[0] -= 1

    with TaskScheduler(workers) as s:
        def produce(pid):
            rnd = random.Random(pid)
            for i in range(per):
                jid = f"p{pid}-{i}"
                s.submit(JobSpec(jid, lambda j=jid: body(j), priority=rnd.randrange(10),
                                 delay_ms=rnd.randrange(5)))
        threads = [threading.Thread(target=produce, args=(p,)) for p in range(producers)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        check(s.await_terminal(ids, 30), "all stress jobs finish")
    once = len(runs) == producers * per and all(v == 1 for v in runs.values())
    print(f"  {len(runs)} jobs ran exactly once: {once}; max in flight {in_flight[1]} (limit {workers})")
    check(once, "every job ran exactly once")
    check(in_flight[1] <= workers, "bounded concurrency")

"""Unit tests. Run: python -m unittest -v test_task_scheduler"""
import threading
import time
import unittest

from task_scheduler import FixedDelay, FixedRate, JobSpec, JobState, RetryPolicy, TaskScheduler


class ScheduleTest(unittest.TestCase):
    def test_fixed_rate_keeps_the_beat_and_catches_up(self):
        self.assertEqual(FixedRate(100).next(1000, 1030), 1100)
        self.assertEqual(FixedRate(100).next(1000, 1250), 1250)   # overran: start at once

    def test_fixed_delay_counts_from_finish(self):
        self.assertEqual(FixedDelay(100).next(1000, 1030), 1130)

    def test_backoff_doubles(self):
        p = RetryPolicy(5, 10)
        self.assertEqual([p.backoff_ms(i) for i in (1, 2, 3, 4)], [10, 20, 40, 80])


class SchedulerTest(unittest.TestCase):
    def setUp(self):
        self.s = TaskScheduler(2)

    def tearDown(self):
        self.s.shutdown()

    def test_delay_is_respected(self):
        started = time.monotonic()
        ran_at = []
        self.s.submit(JobSpec("d", lambda: ran_at.append(time.monotonic()), delay_ms=120))
        self.assertTrue(self.s.await_terminal(["d"], 2))
        self.assertGreaterEqual(ran_at[0] - started, 0.11)

    def test_cancel_before_run(self):
        ran = []
        self.s.submit(JobSpec("c", lambda: ran.append(1), delay_ms=200))
        self.assertTrue(self.s.cancel("c"))
        time.sleep(0.3)
        self.assertEqual(ran, [])
        self.assertIs(self.s.info("c").state, JobState.CANCELLED)
        self.assertFalse(self.s.cancel("c"))

    def test_cancel_running_one_shot_is_refused(self):
        started, release = threading.Event(), threading.Event()

        def long_job():
            started.set()
            release.wait(2)

        self.s.submit(JobSpec("long", long_job))
        self.assertTrue(started.wait(2))
        self.assertFalse(self.s.cancel("long"))
        release.set()
        self.assertTrue(self.s.await_terminal(["long"], 2))
        self.assertIs(self.s.info("long").state, JobState.SUCCEEDED)

    def test_cancel_upstream_skips_downstream(self):
        self.s.submit(JobSpec("up", lambda: None, delay_ms=500))
        self.s.submit(JobSpec("down", lambda: None, depends_on={"up"}))
        self.assertIs(self.s.info("down").state, JobState.WAITING)
        self.s.cancel("up")
        self.assertIs(self.s.info("down").state, JobState.SKIPPED)

    def test_dependency_already_done_is_satisfied(self):
        self.s.submit(JobSpec("first", lambda: None))
        self.assertTrue(self.s.await_terminal(["first"], 2))
        self.s.submit(JobSpec("second", lambda: None, depends_on={"first"}))
        self.assertTrue(self.s.await_terminal(["second"], 2))
        self.assertIs(self.s.info("second").state, JobState.SUCCEEDED)

    def test_diamond_runs_join_once_after_both(self):
        order = []
        lock = threading.Lock()

        def rec(n):
            with lock:
                order.append(n)

        self.s.submit_graph([JobSpec("a", lambda: rec("a")),
                             JobSpec("b", lambda: rec("b"), depends_on={"a"}),
                             JobSpec("c", lambda: rec("c"), depends_on={"a"}),
                             JobSpec("d", lambda: rec("d"), depends_on={"b", "c"})])
        self.assertTrue(self.s.await_terminal("abcd", 2))
        self.assertEqual(order.count("d"), 1)
        self.assertEqual(order[0], "a")
        self.assertEqual(order[-1], "d")

    def test_cycle_rejected_and_nothing_admitted(self):
        with self.assertRaises(ValueError):
            self.s.submit_graph([JobSpec("p", lambda: None, depends_on={"q"}),
                                 JobSpec("q", lambda: None, depends_on={"p"})])
        with self.assertRaises(KeyError):
            self.s.info("p")

    def test_failure_after_retries(self):

// Task scheduler in C++14 as a deterministic simulation with a virtual clock.
// The MinGW g++ 6.3 used here has no std::thread, so instead of real worker threads we
// simulate N workers: each job has a duration, and time jumps from event to event.
// The data structures (time queue, ready queue, in-degree counts) are the same as in
// the threaded Java/Python versions. One scheduler mutex guards every public call, so another thread
// may submit() or cancel() while runUntil() advances time (the simulation itself has no worker threads).
// Windows (MinGW g++ 6.3): g++ -std=c++14 -O2 -Wall -o scheduler main.cpp
// Linux (real lock):      g++ -std=c++14 -O2 -Wall -pthread -DHAVE_STD_MUTEX -o scheduler main.cpp
#include <cassert>
#include <cstdio>
#include <functional>
#include <map>
#include <memory>
#include <queue>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>
#include <mutex>
// Locking: a real std::mutex when built with -DHAVE_STD_MUTEX (Linux:
//   g++ -std=c++14 -pthread -DHAVE_STD_MUTEX main.cpp). MinGW.org g++ 6.3 (win32 thread model) has no
// std::mutex and -pthread fails to link there, so the default build uses a no-op Mutex with the same
// lock()/unlock(): the SAME locking code compiles both ways and simply runs single-threaded on MinGW.
#ifdef HAVE_STD_MUTEX
typedef std::mutex Mutex;
#else
struct Mutex { void lock() {} void unlock() {} bool try_lock() { return true; } };
#endif
typedef std::lock_guard<Mutex> LockGuard;

enum class State { WAITING, SCHEDULED, RUNNING, SUCCEEDED, FAILED, CANCELLED, SKIPPED };

static const char* name(State s) {
    static const char* n[] = {"WAITING", "SCHEDULED", "RUNNING", "SUCCEEDED", "FAILED", "CANCELLED", "SKIPPED"};
    return n[static_cast<int>(s)];
}
static bool terminal(State s) { return s >= State::SUCCEEDED; }

// Strategy: when does the job run next? -1 = never.
struct Schedule {
    virtual ~Schedule() {}
    virtual long long next(long long planned, long long finished) const = 0;
    virtual bool recurring() const { return true; }
};
struct Once : Schedule {
    long long next(long long, long long) const override { return -1; }
    bool recurring() const override { return false; }
};
struct FixedRate : Schedule {
    long long period;
    explicit FixedRate(long long p) : period(p) {}
    long long next(long long planned, long long finished) const override {
        return planned + period > finished ? planned + period : finished;
    }
};
struct FixedDelay : Schedule {
    long long delay;
    explicit FixedDelay(long long d) : delay(d) {}
    long long next(long long, long long finished) const override { return finished + delay; }
};

struct JobSpec {
    std::string id;
    std::function<bool(int)> task;      // the Command; gets the attempt number, returns success
    long long durationMs = 10;          // simulated run time
    int priority = 5;
    long long delayMs = 0;
    std::shared_ptr<Schedule> schedule = std::make_shared<Once>();
    std::vector<std::string> dependsOn;
    int maxAttempts = 1;
    long long backoffMs = 0;
};

struct Job {
    JobSpec spec;
    long long seq, notBefore, runAt = 0;
    State state = State::WAITING;
    int attempt = 0, runs = 0, pendingDeps = 0;
    std::vector<Job*> dependents;
};

class SimScheduler {
    struct ByTime {
        bool operator()(const Job* a, const Job* b) const {
            return a->runAt != b->runAt ? a->runAt > b->runAt : a->seq > b->seq;
        }
    };
    struct ByPriority {
        bool operator()(const Job* a, const Job* b) const {
            if (a->spec.priority != b->spec.priority) return a->spec.priority < b->spec.priority;
            return a->runAt != b->runAt ? a->runAt > b->runAt : a->seq > b->seq;
        }
    };
    struct Done {
        long long at, seq;
        Job* job;
        bool success;
        bool operator>(const Done& o) const { return at != o.at ? at > o.at : seq > o.seq; }
    };

    mutable Mutex mu_;                                          // guards everything below
    long long now_ = 0, seq_ = 0;
    int freeWorkers_;
    std::map<std::string, std::unique_ptr<Job>> jobs_;
    std::priority_queue<Job*, std::vector<Job*>, ByTime> timeQ_;
    std::priority_queue<Job*, std::vector<Job*>, ByPriority> readyQ_;
    std::priority_queue<Done, std::vector<Done>, std::greater<Done>> running_;

    void log(const std::string& what, const Job* j) {
        std::printf("t=%4ld  %-10s %s\n", static_cast<long>(now_), what.c_str(), j->spec.id.c_str());
    }
    void schedule(Job* j, long long at) {
        j->runAt = at;
        j->state = State::SCHEDULED;
        timeQ_.push(j);
    }
    void skipDependents(Job* root) {
        for (Job* d : root->dependents)
            if (d->state == State::WAITING) { d->state = State::SKIPPED; log("skipped", d); skipDependents(d); }
    }
    void validate(const JobSpec& s, const std::set<std::string>& batch) {
        if (jobs_.count(s.id)) throw std::invalid_argument("duplicate job id " + s.id);
        if (s.schedule->recurring() && !s.dependsOn.empty())
            throw std::invalid_argument("recurring job " + s.id + " cannot have dependencies");
        for (const auto& d : s.dependsOn)
            if (!jobs_.count(d) && !batch.count(d)) throw std::invalid_argument("unknown dependency " + d);
    }
    void admit(const JobSpec& s) {
        std::unique_ptr<Job> j(new Job{s, ++seq_, now_ + s.delayMs});
        Job* job = j.get();
        jobs_[s.id] = std::move(j);
        for (const auto& d : s.dependsOn) {
            Job* p = jobs_[d].get();
            if (p->state == State::FAILED || p->state == State::CANCELLED || p->state == State::SKIPPED) {
                job->state = State::SKIPPED;
                return;
            }
        }
        for (const auto& d : s.dependsOn) {
            Job* p = jobs_[d].get();
            if (p->state != State::SUCCEEDED) { p->dependents.push_back(job); job->pendingDeps++; }
        }
        if (job->pendingDeps == 0) schedule(job, job->notBefore);
    }
    void dispatch() {
        while (!timeQ_.empty() && timeQ_.top()->runAt <= now_) {
            Job* j = timeQ_.top();
            timeQ_.pop();
            if (j->state == State::SCHEDULED) readyQ_.push(j);   // lazily drop cancelled jobs
        }
        while (freeWorkers_ > 0 && !readyQ_.empty()) {
            Job* j = readyQ_.top();
            readyQ_.pop();
            if (j->state != State::SCHEDULED) continue;
            j->state = State::RUNNING;
            j->attempt++;
            freeWorkers_--;
            log("start", j);
            bool ok = j->spec.task(j->attempt);                 // real version: runs on a worker with mu_ released
            running_.push(Done{now_ + j->spec.durationMs, j->seq, j, ok});
        }
    }
    void complete(const Done& d) {                              // caller holds mu_ (real version: re-takes it here)
        Job* j = d.job;
        freeWorkers_++;
        j->runs++;
        if (j->state == State::CANCELLED) { log("stopped", j); return; }
        if (d.success) {
            log("done", j);
            j->attempt = 0;
            long long nxt = j->spec.schedule->next(j->runAt, now_);
            if (nxt >= 0) { schedule(j, nxt); return; }
            j->state = State::SUCCEEDED;
            for (Job* c : j->dependents)
                if (c->state == State::WAITING && --c->pendingDeps == 0)
                    schedule(c, now_ > c->notBefore ? now_ : c->notBefore);
        } else if (j->attempt < j->spec.maxAttempts) {
            long long wait = j->spec.backoffMs << (j->attempt - 1);
            log("fail-retry", j);
            schedule(j, now_ + wait);
        } else {
            log("FAILED", j);
            j->state = State::FAILED;
            skipDependents(j);
        }
    }

public:
    explicit SimScheduler(int workers) : freeWorkers_(workers) {}
    long long now() const { LockGuard g(mu_); return now_; }

    void submit(const JobSpec& s) { LockGuard g(mu_); validate(s, {}); admit(s); }

    // All or nothing; Kahn's algorithm inside the batch.
    void submitGraph(const std::vector<JobSpec>& specs) {
        LockGuard g(mu_);                                       // all or nothing, also against other threads
        std::map<std::string, const JobSpec*> batch;
        std::set<std::string> ids;
        for (const auto& s : specs) {
            if (!ids.insert(s.id).second) throw std::invalid_argument("duplicate job id " + s.id);
            batch[s.id] = &s;
        }
        for (const auto& s : specs) validate(s, ids);
        std::map<std::string, int> indeg;
        std::map<std::string, std::vector<std::string>> kids;
        for (const auto& s : specs) {
            indeg[s.id] += 0;
            for (const auto& d : s.dependsOn)
                if (ids.count(d)) { indeg[s.id]++; kids[d].push_back(s.id); }
        }
        std::queue<std::string> ready;
        for (const auto& s : specs) if (indeg[s.id] == 0) ready.push(s.id);
        std::vector<const JobSpec*> order;
        while (!ready.empty()) {
            std::string id = ready.front();
            ready.pop();
            order.push_back(batch[id]);
            for (const auto& c : kids[id]) if (--indeg[c] == 0) ready.push(c);
        }
        if (order.size() != specs.size()) throw std::invalid_argument("cycle in job graph");
        for (const JobSpec* s : order) admit(*s);
    }

    bool cancel(const std::string& id) {
        LockGuard g(mu_);
        auto it = jobs_.find(id);
        if (it == jobs_.end()) return false;
        Job* j = it->second.get();
        if (terminal(j->state)) return false;
        if (j->state == State::RUNNING && !j->spec.schedule->recurring()) return false;
        j->state = State::CANCELLED;
        log("cancel", j);
        skipDependents(j);
        return true;
    }

    State state(const std::string& id) const { LockGuard g(mu_); return jobs_.at(id)->state; }
    int runs(const std::string& id) const { LockGuard g(mu_); return jobs_.at(id)->runs; }

    // Advance virtual time, processing events in order, up to and including `until`.
    void runUntil(long long until) {
        LockGuard g(mu_);
        while (true) {
            dispatch();
            long long next = -1;
            if (!running_.empty()) next = running_.top().at;
            if (!timeQ_.empty() && (next < 0 || timeQ_.top()->runAt < next)) next = timeQ_.top()->runAt;
            if (next < 0 || next > until) { now_ = until; return; }
            now_ = next;
            while (!running_.empty() && running_.top().at == now_) {
                Done d = running_.top();
                running_.pop();
                complete(d);
            }
        }
    }
};

static JobSpec job(const std::string& id, long long dur, int prio = 5) {
    JobSpec s;
    s.id = id;
    s.durationMs = dur;
    s.priority = prio;
    s.task = [](int) { return true; };
    return s;
}

int main() {
    SimScheduler s(2);   // two simulated workers
    std::printf("== timeline (2 workers, virtual ms) ==\n");

    JobSpec backup = job("backup", 30);
    backup.schedule = std::make_shared<FixedRate>(100);           // every 100 ms

What this C++ version leaves out: it is a single-threaded simulation on a virtual clock with 2 simulated workers, not the real thread pool: one facade lock (real with -DHAVE_STD_MUTEX) but no worker threads, no condition variables, no stress test, and no named check count (it ends with "C++ demo OK" after a 37-line timeline). It does replay priority, no preemption, retry backoff, the DAG and cancellation.

Read the C++ timeline slowly, it teaches three rules. At t=60 urgent (priority 9) is due, but both workers are busy, so it waits until t=70: no preemption. At t=70 the retry of flaky is also due, yet urgent wins: priority beats arrival among due jobs. And backup starts at 0, 100, 200, 300 regardless of its 30 ms run time: fixed rate is anchored to the planned start, not the finish.
