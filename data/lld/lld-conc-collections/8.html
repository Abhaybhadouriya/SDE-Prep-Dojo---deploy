import java.util.concurrent.*;
import java.util.concurrent.atomic.*;

/**
 * Concurrent collections and atomics, for the SDE Prep Dojo page
 * "Concurrent Collections & Atomics (CAS)".
 * Run: javac CollectionsDemo.java && java CollectionsDemo

public class CollectionsDemo {
    static final int THREADS = 8, PER_THREAD = 100_000;

    /** Start THREADS threads at the same moment, each running body PER_THREAD times. */
    static long race(Runnable body) throws InterruptedException {
        CountDownLatch go = new CountDownLatch(1);
        List<Thread> ts = new ArrayList<>();
        for (int t = 0; t < THREADS; t++) {
            Thread th = new Thread(() -> {
                try { go.await(); } catch (InterruptedException e) { return; }
                for (int i = 0; i < PER_THREAD; i++) body.run();
            });
            ts.add(th); th.start();
        }
        long t0 = System.nanoTime();
        go.countDown();
        for (Thread th : ts) th.join();
        return (System.nanoTime() - t0) / 1_000_000;
    }

    public static void main(String[] args) throws Exception {
        int expected = THREADS * PER_THREAD;

        System.out.println("1) check-then-act on a ConcurrentHashMap is still a race");
        ConcurrentHashMap<String, Integer> chm = new ConcurrentHashMap<>();
        chm.put("views", 0);
        race(() -> chm.put("views", chm.get("views") + 1));        // get, add, put: three steps
        System.out.println("  get+put : " + chm.get("views") + " / " + expected);
        chm.put("views", 0);
        race(() -> chm.merge("views", 1, Integer::sum));            // one atomic step per key
        System.out.println("  merge   : " + chm.get("views") + " / " + expected);

    /** Classic CAS loop: read, compute, compareAndSet, retry on conflict. */
    static boolean tryTake(AtomicInteger stock, int n) {
        while (true) {
            int cur = stock.get();
            if (cur < n) return false;                    // not enough: give up, no write
            if (stock.compareAndSet(cur, cur - n)) return true;
            // someone else changed it between get and CAS: loop and re-read
        }
    }

    static void abaDemo() throws InterruptedException {
        AtomicReference<String> top = new AtomicReference<>("A");
        AtomicStampedReference<String> stamped = new AtomicStampedReference<>("A", 0);
        CountDownLatch read = new CountDownLatch(1), changed = new CountDownLatch(1);

        Thread slow = new Thread(() -> {
            String seen = top.get();
            int[] stampHolder = new int[1];
            String seenS = stamped.get(stampHolder);
            int seenStamp = stampHolder[0];
            read.countDown();
            try { changed.await(); } catch (InterruptedException e) { return; }
            System.out.println("  plain CAS(A -> C) succeeded: " + top.compareAndSet(seen, "C"));
            System.out.println("  stamped CAS(A,0 -> C) succeeded: " + stamped.compareAndSet(seenS, "C", seenStamp, seenStamp + 1)
                    + " (stamp is now " + stamped.getStamp() + ")");
        });
        slow.start();
        read.await();
        // the fast thread: A -> B -> A
        top.set("B"); top.set("A");
        int s = stamped.getStamp();
        stamped.compareAndSet("A", "B", s, s + 1);
        stamped.compareAndSet("B", "A", s + 1, s + 2);
        changed.countDown();
        slow.join();
    }

    /** The shape often posted for review: an atomic counter plus a plain field, reset under a lock. */
    static final class BuggyLimiter {
        static final class Window { final AtomicInteger left = new AtomicInteger(); long start; }   // start is not volatile
        final ConcurrentHashMap<String, Window> windows = new ConcurrentHashMap<>();
        final int limit; final long windowMs;
        BuggyLimiter(int limit, long windowMs) { this.limit = limit; this.windowMs = windowMs; }
        boolean allow(String key) {
            Window w = windows.computeIfAbsent(key, k -> new Window());
            long now = System.currentTimeMillis();
            if (now - w.start >= windowMs) {                 // unsynchronised read of start
                synchronized (w) {
                    if (now - w.start >= windowMs) { w.start = now; w.left.set(limit); }
                }
            }
            return w.left.getAndDecrement() >= 0;            // >= 0 lets limit + 1 through
        }
    }

    /** Fix: the whole window is one immutable value, replaced atomically per key by compute(). */
    static final class FixedLimiter {
        record Window(long start, int used) {}
        final ConcurrentHashMap<String, Window> windows = new ConcurrentHashMap<>();
        final int limit; final long windowMs;
        FixedLimiter(int limit, long windowMs) { this.limit = limit; this.windowMs = windowMs; }
        boolean allow(String key) {
            long now = System.currentTimeMillis();
            boolean[] ok = new boolean[1];
            windows.compute(key, (k, w) -> {
                if (w == null || now - w.start() >= windowMs) w = new Window(now, 0);
                ok[0] = w.used() < limit;
                return ok[0] ? new Window(w.start(), w.used() + 1) : w;
            });
            return ok[0];
        }
    }
}

// Atomics, CAS loops and a per-thread sharded counter in C++14, for the SDE Prep Dojo page
// "Concurrent Collections & Atomics (CAS)".
// Build: g++ -std=c++14 -O2 -Wall -Wextra -pthread atomics.cpp -o atomics && ./atomics
// Needs a POSIX-threads g++ (Linux, WSL, or MinGW-w64 "posix"). We built and ran it with g++ 13.3 in
// WSL Ubuntu 24.04. The MinGW.org g++ 6.3 on Windows (win32 thread model) has no std::thread or std::mutex.
#include <atomic>
#include <cstdint>
#include <iostream>
#include <thread>
#include <vector>

constexpr int kThreads = 8;
constexpr int kPerThread = 200000;

template <class F>
void race(F body) {
    std::vector<std::thread> ts;
    for (int t = 0; t < kThreads; ++t)
        ts.emplace_back([&body, t] { for (int i = 0; i < kPerThread; ++i) body(t); });
    for (auto& th : ts) th.join();
}

// CAS loop: decrement only while the value stays >= n.
bool try_take(std::atomic<int>& stock, int n) {
    int cur = stock.load(std::memory_order_relaxed);
    while (cur >= n) {
        // On failure compare_exchange_weak reloads cur for us; "weak" may also fail spuriously, so loop.
        if (stock.compare_exchange_weak(cur, cur - n, std::memory_order_acq_rel, std::memory_order_relaxed))
            return true;
    }
    return false;
}

// Like the kernel's percpu_counter / Java's LongAdder: one padded slot per thread, sum on read.
struct alignas(64) Slot { std::atomic<long> v{0}; };   // 64 = typical cache line; avoids false sharing
struct ShardedCounter {
    Slot slots[kThreads];
    void add(int shard) { slots[shard].v.fetch_add(1, std::memory_order_relaxed); }
    long sum() const {
        long s = 0;
        for (const auto& sl : slots) s += sl.v.load(std::memory_order_relaxed);
        return s;                                       // not a snapshot while writers run
    }
};

int main() {
    const long expected = 1L * kThreads * kPerThread;

    // DATA RACE on purpose: undefined behaviour in C++ (TSan reports this line). It is here only to
    // show the symptom (lost updates). volatile is NOT atomic and gives no ordering either.
    volatile long racy = 0;
    race([&](int) { racy = racy + 1; });
    std::cout << "plain ++     : " << racy << " / " << expected << "\n";

    std::atomic<long> counter{0};
    race([&](int) { counter.fetch_add(1, std::memory_order_relaxed); });   // lock xadd on x86
    std::cout << "atomic fetch_add: " << counter.load() << " / " << expected << "\n";

    ShardedCounter sharded;
    race([&](int t) { sharded.add(t); });
    std::cout << "sharded sum  : " << sharded.sum() << " / " << expected << "\n";

    std::atomic<int> stock{1000};
    std::atomic<int> sold{0};
    race([&](int) { if (try_take(stock, 1)) sold.fetch_add(1); });
    std::cout << "CAS take     : sold=" << sold.load() << " left=" << stock.load() << "\n";

    // Tagged value: pack a 32-bit version next to a 32-bit value so A->B->A is visible.
    std::atomic<std::uint64_t> tagged{0};
    auto pack = [](std::uint32_t ver, std::uint32_t val) { return (std::uint64_t(ver) << 32) | val; };
    tagged = pack(0, 'A');
    std::uint64_t seen = tagged.load();
    tagged = pack(1, 'B');
    tagged = pack(2, 'A');                              // same value, new version
    bool ok = tagged.compare_exchange_strong(seen, pack(3, 'C'));
    std::cout << "tagged CAS after A->B->A succeeded: " << std::boolalpha << ok << "\n";
    std::cout << "lock-free 64-bit atomic: " << tagged.is_lock_free() << "\n";
}

"""Thread-safe counting in Python, for the SDE Prep Dojo page
"Concurrent Collections & Atomics (CAS)".  Run: python counters.py

Python has no user-level CAS. The GIL makes single bytecodes atomic, but `d[k] += 1`
is several bytecodes (load, add, store), so a thread switch in the middle can lose an update.
Free-threaded builds (PEP 703, Python 3.13+) remove the GIL, which makes locks matter even more.
"""
import threading
from collections import Counter

THREADS, PER_THREAD = 8, 50_000


def race(body):
    ts = [threading.Thread(target=lambda: [body() for _ in range(PER_THREAD)]) for _ in range(THREADS)]
    for t in ts:
        t.start()
    for t in ts:
        t.join()


class LockedCounter:
    def __init__(self):
        self._lock = threading.Lock()
        self._c = Counter()

    def incr(self, key, n=1):
        with self._lock:              # read-modify-write as one critical section
            self._c[key] += n

    def get(self, key):
        with self._lock:
            return self._c[key]


if __name__ == "__main__":
    expected = THREADS * PER_THREAD
    lc = LockedCounter()
    race(lambda: lc.incr("views"))
    print(f"LockedCounter: {lc.get('views')} / {expected}")

    # setdefault is a single C call on a dict, so "create once" is safe under the GIL,
    # but do not rely on that for anything bigger than one call.
    groups = {}
    race(lambda: groups.setdefault("asha", set()).add(threading.get_ident()))
    print(f"setdefault built one set with {len(groups['asha'])} thread ids")

The C++ "plain ++" line is undefined behaviour, not just lost updates. Two threads writing a volatile long with no synchronisation is a data race, and a data race is UB in C++ (ThreadSanitizer flags that line). volatile gives neither atomicity nor ordering; the kernel has the same rule, which is why it uses READ_ONCE/WRITE_ONCE and atomics. The line is there only to show the symptom. To show lost updates without UB, use a std::atomic<long> with a separate load() and store(): each access is atomic, the increment is not.
