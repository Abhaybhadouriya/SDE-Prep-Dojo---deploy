The same design in three languages. Java is the main version (one file, compile with javac -d out Main.java && java -cp out Main). It ends with a demo that also checks itself: if any check fails it throws. Python ships a module, a demo and a unittest suite. The C++14 version is single-threaded because the MinGW 6.3 toolchain on this machine has no std::thread; it replaces the async queue with a printk-style ring buffer.

Read in this order (the files are long; this is the 10-minute path):

  1. Main.main (bottom of Main.java): the demo steps.
  2. Logger.log and LogHandler.handle: the cheap level check, then the chain of handlers.
  3. AsyncSink: bounded queue, writer thread, poison pill, drop counter, and the read-write gate on close. This is the heart.
  4. The formatters and sinks: small, one job each.
  5. Skip on first read: LogLevel, LogMessage, the builder.

From your world: reading this Java as a C programmer

  • interface Foo { int bar(int x); } is a struct of function pointers (an ops table like struct file_operations); a class that implements Foo fills the table in.
  • x -> expr is a lambda, not a pointer dereference: an anonymous function. Read it as "a function pointer plus the context it captured" (the void *data you pass with a callback). k -> new ArrayDeque<>() means "given k, make an empty deque". Inside a switch, case X -> ... is a case with no fallthrough, so no break.
  • Foo::bar is a method reference: a function pointer to bar (obj::bar also binds the object, like passing obj as the first argument).
  • record Point(int x, int y) is a const struct whose constructor, equals, hashCode and toString are generated.
  • Generics such as Map<K, V> are a type-checked void *: one implementation, and the compiler checks what goes in and comes out.
  • Optional<T> is a "may be NULL" return made explicit in the type. EnumMap is an array indexed by an enum. A sealed interface allows only the listed implementations (a closed tagged union).
  • final on a field means "assigned once", like const after init. There is no free(): the garbage collector reclaims objects nothing points to.
  • volatile in Java is not C's volatile (see the concurrency section), and synchronized is a mutex built into every object.

// ---------- model ----------
enum LogLevel {
    DEBUG(10), INFO(20), WARN(30), ERROR(40), FATAL(50);
    final int weight;
    LogLevel(int weight) { this.weight = weight; }
    boolean atLeast(LogLevel other) { return weight >= other.weight; }
}

final class LogMessage {                       // immutable: safe to hand to another thread
    final long epochMillis;
    final LogLevel level;
    final String logger, thread, text;
    LogMessage(long epochMillis, LogLevel level, String logger, String thread, String text) {
        this.epochMillis = epochMillis; this.level = level;
        this.logger = logger; this.thread = thread; this.text = text;
    }
}

// ---------- Strategy 1: how a line looks ----------
interface LogFormatter { String format(LogMessage m); }

final class PlainFormatter implements LogFormatter {
    public String format(LogMessage m) {
        return String.format("%d [%-5s] %s - %s", m.epochMillis, m.level, m.logger, m.text);
    }
}

final class JsonFormatter implements LogFormatter {
    private static String esc(String s) { return s.replace("\\", "\\\\").replace("\"", "\\\""); }
    public String format(LogMessage m) {
        return "{\"ts\":" + m.epochMillis + ",\"level\":\"" + m.level + "\",\"logger\":\""
                + esc(m.logger) + "\",\"msg\":\"" + esc(m.text) + "\"}";
    }
}

// ---------- Strategy 2: where a line goes ----------
interface LogSink extends Closeable {
    void write(String line);
    default void close() {}
}

final class ConsoleSink implements LogSink {
    public synchronized void write(String line) { System.out.println(line); }
}

final class MemorySink implements LogSink {      // test double; also a "ring buffer" style sink
    private final List<String> lines = new ArrayList<>();
    public synchronized void write(String line) { lines.add(line); }
    public synchronized List<String> lines() { return new ArrayList<>(lines); }
}

final class FileSink implements LogSink {
    private final BufferedWriter out;
    FileSink(Path path) throws IOException {
        out = Files.newBufferedWriter(path, StandardOpenOption.CREATE, StandardOpenOption.APPEND);
    }
    public synchronized void write(String line) {
        try { out.write(line); out.newLine(); } catch (IOException e) { System.err.println("log write failed: " + e); }
    }
    public synchronized void close() {
        try { out.close(); } catch (IOException e) { System.err.println("log close failed: " + e); }
    }
}

// Follow-up sink (Swiggy asked "search logs by a word"): an inverted index word -> line numbers.
final class SearchIndexSink implements LogSink {
    private final List<String> lines = new ArrayList<>();
    private final Map<String, Set<Integer>> index = new HashMap<>();
    public synchronized void write(String line) {
        int no = lines.size();
        lines.add(line);
        for (String w : line.toLowerCase().split("[^a-z0-9]+"))
            if (!w.isEmpty()) index.computeIfAbsent(w, k -> new TreeSet<>()).add(no);
    }
    public synchronized List<String> search(String word) {
        List<String> hits = new ArrayList<>();
        for (int no : index.getOrDefault(word.toLowerCase(), Collections.emptySet())) hits.add(lines.get(no));
        return hits;
    }
}

// Decorator: same LogSink interface, but writes happen on a background thread.
final class AsyncSink implements LogSink {
    private static final String POISON = "\u0000STOP";
    private final LogSink target;
    private final BlockingQueue<String> queue;
    private final AtomicLong dropped = new AtomicLong();
    private final Thread worker;
    private volatile boolean closed;
    // write() holds the read side, close() the write side: close cannot slip in between a writer's
    // "closed?" check and its offer(), so no line can land behind the POISON pill and vanish.
    private final ReentrantReadWriteLock gate = new ReentrantReadWriteLock();

    AsyncSink(LogSink target, int capacity) {
        this.target = target;
        this.queue = new ArrayBlockingQueue<>(capacity);
        this.worker = new Thread(this::drain, "log-writer");
        this.worker.setDaemon(true);
        this.worker.start();
    }
    public void write(String line) {
        gate.readLock().lock();                 // shared: writers never wait for each other
        try {
            if (closed || !queue.offer(line)) dropped.incrementAndGet();   // never block on a full queue
        } finally { gate.readLock().unlock(); }
    }
    private void drain() {
        try {
            while (true) {
                String line = queue.take();
                if (line == POISON) break;
                target.write(line);
            }
        } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
    public synchronized void close() {
        if (closed) return;                      // idempotent: safe to close twice
        gate.writeLock().lock();                 // waits only for writers already inside write()
        try { closed = true; } finally { gate.writeLock().unlock(); }
        try {
            queue.put(POISON);          // everything queued before this is written first
            worker.join();
        } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        target.close();
    }
    long dropped() { return dropped.get(); }
}

// ---------- Chain of Responsibility: one handler per destination ----------
abstract class LogHandler {
    private final LogLevel threshold;
    private LogHandler next;
    protected LogHandler(LogLevel threshold) { this.threshold = threshold; }
    LogHandler setNext(LogHandler next) { this.next = next; return next; }   // returns next for chaining
    final void handle(LogMessage m) {
        if (m.level.atLeast(threshold)) emit(m);
        if (next != null) next.handle(m);       // pass-through chain: later handlers also see it
    }
    protected abstract void emit(LogMessage m);
    void closeAll() { close(); if (next != null) next.closeAll(); }
    protected void close() {}
}

final class SinkHandler extends LogHandler {
    private final LogFormatter formatter;
    private final LogSink sink;
    SinkHandler(LogLevel threshold, LogFormatter formatter, LogSink sink) {
        super(threshold); this.formatter = formatter; this.sink = sink;
    }
    protected void emit(LogMessage m) { sink.write(formatter.format(m)); }
    protected void close() { sink.close(); }
}

// ---------- configuration (Builder) ----------
final class LoggerConfig {
    final LogLevel minLevel;
    final LogHandler head;
    final LongSupplier clock;
    private LoggerConfig(Builder b) { minLevel = b.minLevel; head = b.head; clock = b.clock; }

    static Builder builder() { return new Builder(); }
    static final class Builder {
        private LogLevel minLevel = LogLevel.INFO;
        private LogHandler head, tail;
        private LongSupplier clock = System::currentTimeMillis;
        Builder minLevel(LogLevel l) { minLevel = l; return this; }
        Builder clock(LongSupplier c) { clock = c; return this; }
        Builder handler(LogLevel threshold, LogFormatter f, LogSink s) {
            LogHandler h = new SinkHandler(threshold, f, s);
            if (head == null) head = tail = h; else tail = tail.setNext(h);
            return this;
        }
        LoggerConfig build() {
            if (head == null) throw new IllegalStateException("at least one handler needed");
            return new LoggerConfig(this);
        }
    }
}

// ---------- the API callers see ----------
final class Logger {
    private final String name;
    private final LogManager manager;
    Logger(String name, LogManager manager) { this.name = name; this.manager = manager; }

    void log(LogLevel level, String text) {
        LoggerConfig cfg = manager.config();               // one volatile read per call
        if (!level.atLeast(cfg.minLevel)) return;          // cheap early exit, no allocation
        cfg.head.handle(new LogMessage(cfg.clock.getAsLong(), level, name,
                Thread.currentThread().getName(), text));
    }
    void debug(String t) { log(LogLevel.DEBUG, t); }
    void info(String t)  { log(LogLevel.INFO, t); }
    void warn(String t)  { log(LogLevel.WARN, t); }
    void error(String t) { log(LogLevel.ERROR, t); }
}

// Singleton + registry of named loggers (a "multiton").
final class LogManager {
    private static final LogManager INSTANCE = new LogManager();   // eager: JVM class init is thread-safe
    private final ConcurrentHashMap<String, Logger> loggers = new ConcurrentHashMap<>();
    private volatile LoggerConfig config = LoggerConfig.builder()
            .handler(LogLevel.INFO, new PlainFormatter(), new ConsoleSink()).build();

    private LogManager() {}
    static LogManager get() { return INSTANCE; }

    Logger getLogger(String name) { return loggers.computeIfAbsent(name, n -> new Logger(n, this)); }
    LoggerConfig config() { return config; }
    void configure(LoggerConfig next) {                   // swap atomically; old chain is closed
        LoggerConfig old = config;
        config = next;
        if (old != next) old.head.closeAll();
    }
    void shutdown() { config.head.closeAll(); }
}

// ---------- demo + checks ----------
public class Main {
    static int checks = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    public static void main(String[] args) throws Exception {
        AtomicLong fakeTime = new AtomicLong(1_700_000_000_000L);
        MemorySink all = new MemorySink();
        MemorySink alerts = new MemorySink();

        LogManager lm = LogManager.get();
        lm.configure(LoggerConfig.builder()
                .minLevel(LogLevel.DEBUG)
                .clock(fakeTime::incrementAndGet)                         // deterministic timestamps
                .handler(LogLevel.INFO,  new PlainFormatter(), new ConsoleSink())
                .handler(LogLevel.DEBUG, new JsonFormatter(),  all)
                .handler(LogLevel.ERROR, new PlainFormatter(), alerts)
                .build());

        Logger pay = lm.getLogger("payments");
        check(pay == lm.getLogger("payments"), "same name gives same logger");

        System.out.println("== 1. levels walk the chain ==");
        pay.debug("cache warm");
        pay.info("charge started id=42");
        pay.warn("retrying gateway");
        pay.error("gateway timeout id=42");
        check(all.lines().size() == 4, "json sink saw all 4");
        check(alerts.lines().size() == 1, "alert sink saw only ERROR");
        check(alerts.lines().get(0).contains("[ERROR] payments - gateway timeout"), "alert text");
        System.out.println("json sink line 1: " + all.lines().get(0));

        System.out.println("== 2. raise the level at runtime ==");
        MemorySink quiet = new MemorySink();
        lm.configure(LoggerConfig.builder().minLevel(LogLevel.WARN)
                .clock(fakeTime::incrementAndGet)
                .handler(LogLevel.DEBUG, new PlainFormatter(), quiet).build());
        pay.info("hidden");
        pay.warn("shown");
        check(quiet.lines().size() == 1 && quiet.lines().get(0).contains("shown"), "min level filter");
        System.out.println("after WARN filter: " + quiet.lines());

        System.out.println("== 3. async sink, 4 threads x 2500 lines ==");
        MemorySink slow = new MemorySink();
        AsyncSink async = new AsyncSink(slow, 20_000);
        lm.configure(LoggerConfig.builder().minLevel(LogLevel.INFO)
                .clock(fakeTime::incrementAndGet)
                .handler(LogLevel.INFO, new PlainFormatter(), async).build());
        ExecutorService pool = Executors.newFixedThreadPool(4);
        for (int t = 0; t < 4; t++) {
            final int id = t;
            pool.submit(() -> {
                Logger l = lm.getLogger("worker-" + id);
                for (int i = 0; i < 2500; i++) l.info("job " + i);
            });
        }
        pool.shutdown();
        check(pool.awaitTermination(10, TimeUnit.SECONDS), "workers finished");
        lm.shutdown();                                   // drains the queue before returning
        check(slow.lines().size() + async.dropped() == 10_000, "every line written or counted as dropped");
        check(async.dropped() == 0, "queue was big enough: nothing dropped");
        System.out.println("async lines written: " + slow.lines().size() + ", dropped: " + async.dropped());

        System.out.println("== 4. tiny queue drops instead of blocking ==");
        MemorySink sink2 = new MemorySink();
        AsyncSink tiny = new AsyncSink(line -> { sleepQuietly(1); sink2.write(line); }, 4);
        for (int i = 0; i < 200; i++) tiny.write("burst " + i);
        tiny.close();
        check(sink2.lines().size() + tiny.dropped() == 200, "written + dropped == sent");
        check(tiny.dropped() > 0, "some lines dropped under burst");
        System.out.println("burst of 200: written + dropped = " + (sink2.lines().size() + tiny.dropped()) + ", some dropped = " + (tiny.dropped() > 0));

        System.out.println("== 4b. close() races with writers ==");
        MemorySink sink3 = new MemorySink();
        AsyncSink racing = new AsyncSink(sink3, 1_000);
        AtomicLong attempted = new AtomicLong();
        ExecutorService writers = Executors.newFixedThreadPool(4);
        CountDownLatch started = new CountDownLatch(4);
        for (int t = 0; t < 4; t++) writers.submit(() -> {
            started.countDown();
            for (int i = 0; i < 5_000; i++) { racing.write("w " + i); attempted.incrementAndGet(); }
        });
        started.await();
        racing.close();                                  // lands in the middle of the writes
        writers.shutdown();
        check(writers.awaitTermination(10, TimeUnit.SECONDS), "racing writers finished");
        check(sink3.lines().size() + racing.dropped() == attempted.get(), "close during writes: written + dropped == attempted");
        System.out.println("close during writes: written + dropped == attempted (" + attempted.get() + "): "
                + (sink3.lines().size() + racing.dropped() == attempted.get()));

        System.out.println("== 5. file sink ==");
        Path tmp = Files.createTempFile("lld-logger", ".log");
        lm.configure(LoggerConfig.builder().clock(fakeTime::incrementAndGet)
                .handler(LogLevel.INFO, new PlainFormatter(), new FileSink(tmp)).build());
        lm.getLogger("orders").info("order placed");
        lm.getLogger("orders").error("order failed");
        lm.shutdown();
        List<String> fileLines = Files.readAllLines(tmp);
        check(fileLines.size() == 2, "two lines in file");
        Files.delete(tmp);
        System.out.println("file had " + fileLines.size() + " lines");

        System.out.println("== 6. follow-up: search logs by word ==");
        SearchIndexSink idx = new SearchIndexSink();
        lm.configure(LoggerConfig.builder().clock(fakeTime::incrementAndGet)
                .handler(LogLevel.INFO, new PlainFormatter(), idx).build());
        Logger api = lm.getLogger("api");
        api.info("user 7 login ok");
        api.error("user 9 login failed");
        api.info("cart updated");
        check(idx.search("login").size() == 2, "two lines contain login");
        check(idx.search("FAILED").size() == 1, "search ignores case");
        System.out.println("search 'login' -> " + idx.search("login").size() + " hits; 'failed' -> " + idx.search("failed"));

        System.out.println("All " + checks + " checks passed");
    }

    static void sleepQuietly(long ms) {
        try { Thread.sleep(ms); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }
}

logger.py

"""Logging framework: levels, Chain of Responsibility handlers, formatter/sink Strategy,
async sink Decorator, and a Singleton LogManager. Standard library only."""
from __future__ import annotations

import enum
import itertools
import json
import queue
import threading
import time
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Callable, List, Optional


class LogLevel(enum.IntEnum):
    DEBUG = 10
    INFO = 20
    WARN = 30
    ERROR = 40
    FATAL = 50


@dataclass(frozen=True)
class LogMessage:            # immutable: safe to pass to the writer thread
    ts: int
    level: LogLevel
    logger: str
    thread: str
    text: str


# ---- Strategy 1: formatting ----
class Formatter(ABC):
    @abstractmethod
    def format(self, m: LogMessage) -> str: ...


class PlainFormatter(Formatter):
    def format(self, m: LogMessage) -> str:
        return f"{m.ts} [{m.level.name:<5}] {m.logger} - {m.text}"


class JsonFormatter(Formatter):
    def format(self, m: LogMessage) -> str:
        return json.dumps({"ts": m.ts, "level": m.level.name, "logger": m.logger, "msg": m.text},
                          separators=(",", ":"))


# ---- Strategy 2: destination ----
class Sink(ABC):
    @abstractmethod
    def write(self, line: str) -> None: ...

    def close(self) -> None:
        pass


class ConsoleSink(Sink):
    def __init__(self) -> None:
        self._lock = threading.Lock()

    def write(self, line: str) -> None:
        with self._lock:
            print(line)


class MemorySink(Sink):
    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._lines: List[str] = []

    def write(self, line: str) -> None:
        with self._lock:
            self._lines.append(line)

    @property
    def lines(self) -> List[str]:
        with self._lock:
            return list(self._lines)


class FileSink(Sink):
    def __init__(self, path: str) -> None:
        self._lock = threading.Lock()
        self._f = open(path, "a", encoding="utf-8")

    def write(self, line: str) -> None:
        with self._lock:
            self._f.write(line + "\n")

    def close(self) -> None:
        with self._lock:
            if not self._f.closed:
                self._f.close()


class AsyncSink(Sink):
    """Decorator: same Sink interface, the real write happens on a background thread."""
    _STOP = object()

    def __init__(self, target: Sink, capacity: int = 10_000) -> None:
        self._target = target
        self._q: "queue.Queue[object]" = queue.Queue(maxsize=capacity)
        self._dropped = 0
        self._drop_lock = threading.Lock()
        self._closed = False
        # write() and close() share this lock, so close() cannot slip in between a writer's
        # "closed?" check and its put: no line can land behind the stop marker and vanish.
        self._gate = threading.Lock()
        self._worker = threading.Thread(target=self._drain, name="log-writer", daemon=True)
        self._worker.start()

    def write(self, line: str) -> None:
        with self._gate:                    # put_nowait never blocks, so the lock is held only briefly
            try:
                if self._closed:
                    raise queue.Full
                self._q.put_nowait(line)    # never block the caller
            except queue.Full:
                with self._drop_lock:
                    self._dropped += 1

    def _drain(self) -> None:
        while True:
            item = self._q.get()
            if item is self._STOP:
                return
            self._target.write(item)  # type: ignore[arg-type]

    def close(self) -> None:
        with self._gate:
            if self._closed:
                return
            self._closed = True             # from here every write() counts as dropped
        self._q.put(self._STOP)             # queued lines are written first
        self._worker.join()
        self._target.close()

    @property
    def dropped(self) -> int:
        with self._drop_lock:
            return self._dropped


# ---- Chain of Responsibility ----
class Handler:
    def __init__(self, threshold: LogLevel, formatter: Formatter, sink: Sink) -> None:
        self.threshold, self.formatter, self.sink = threshold, formatter, sink
        self.next: Optional[Handler] = None

    def set_next(self, h: "Handler") -> "Handler":
        self.next = h
        return h

    def handle(self, m: LogMessage) -> None:
        if m.level >= self.threshold:
            self.sink.write(self.formatter.format(m))
        if self.next:                       # pass-through: later handlers also see it
            self.next.handle(m)

    def close_all(self) -> None:
        h: Optional[Handler] = self
        while h:
            h.sink.close()
            h = h.next


class LoggerConfig:
    def __init__(self, min_level: LogLevel, head: Handler, clock: Callable[[], int]) -> None:
        self.min_level, self.head, self.clock = min_level, head, clock

    class Builder:
        def __init__(self) -> None:
            self._min = LogLevel.INFO
            self._head: Optional[Handler] = None
            self._tail: Optional[Handler] = None
            self._clock: Callable[[], int] = lambda: int(time.time() * 1000)

        def min_level(self, lvl: LogLevel) -> "LoggerConfig.Builder":
            self._min = lvl
            return self

        def clock(self, c: Callable[[], int]) -> "LoggerConfig.Builder":
            self._clock = c
            return self

        def handler(self, threshold: LogLevel, fmt: Formatter, sink: Sink) -> "LoggerConfig.Builder":
            h = Handler(threshold, fmt, sink)
            if self._head is None:
                self._head = self._tail = h
            else:
                self._tail = self._tail.set_next(h)  # type: ignore[union-attr]
            return self

        def build(self) -> "LoggerConfig":
            if self._head is None:
                raise ValueError("at least one handler needed")
            return LoggerConfig(self._min, self._head, self._clock)


class Logger:
    def __init__(self, name: str, manager: "LogManager") -> None:
        self.name, self._manager = name, manager

    def log(self, level: LogLevel, text: str) -> None:
        cfg = self._manager.config          # one attribute read; swaps are atomic
        if level < cfg.min_level:
            return
        cfg.head.handle(LogMessage(cfg.clock(), level, self.name,
                                   threading.current_thread().name, text))

    def debug(self, t: str) -> None: self.log(LogLevel.DEBUG, t)
    def info(self, t: str) -> None: self.log(LogLevel.INFO, t)
    def warn(self, t: str) -> None: self.log(LogLevel.WARN, t)
    def error(self, t: str) -> None: self.log(LogLevel.ERROR, t)


class LogManager:
    """Singleton registry of named loggers."""
    _instance: Optional["LogManager"] = None
    _instance_lock = threading.Lock()

    def __init__(self) -> None:
        self._loggers: dict = {}
        self._lock = threading.Lock()
        self.config = LoggerConfig.Builder().handler(LogLevel.INFO, PlainFormatter(), ConsoleSink()).build()

    @classmethod
    def get(cls) -> "LogManager":
        if cls._instance is None:
            with cls._instance_lock:          # double-checked locking
                if cls._instance is None:
                    cls._instance = LogManager()
        return cls._instance

    def get_logger(self, name: str) -> Logger:
        with self._lock:
            if name not in self._loggers:
                self._loggers[name] = Logger(name, self)
            return self._loggers[name]

    def configure(self, cfg: LoggerConfig) -> None:
        old, self.config = self.config, cfg
        if old is not cfg:
            old.head.close_all()

    def shutdown(self) -> None:
        self.config.head.close_all()


def counter_clock(start: int = 1_700_000_000_000) -> Callable[[], int]:
    c = itertools.count(start + 1)
    lock = threading.Lock()

    def tick() -> int:
        with lock:
            return next(c)
    return tick

from concurrent.futures import ThreadPoolExecutor

from logger import (AsyncSink, ConsoleSink, JsonFormatter, LogLevel, LoggerConfig, LogManager,
                    MemorySink, PlainFormatter, counter_clock)


def main() -> None:
    clock = counter_clock()
    all_sink, alerts = MemorySink(), MemorySink()

    async_sink = AsyncSink(slow, capacity=20_000)
    lm.configure(LoggerConfig.Builder().clock(clock)
                 .handler(LogLevel.INFO, PlainFormatter(), async_sink).build())

    def work(i: int) -> None:
        log = lm.get_logger(f"worker-{i}")
        for n in range(2500):
            log.info(f"job {n}")

    with ThreadPoolExecutor(4) as ex:
        list(ex.map(work, range(4)))
    lm.shutdown()
    assert len(slow.lines) + async_sink.dropped == 10_000
    print("async lines written:", len(slow.lines), "dropped:", async_sink.dropped)
    print("All checks passed")

class SlowSink(Sink):
    def __init__(self, inner: Sink) -> None:
        self.inner = inner

    def write(self, line: str) -> None:
        time.sleep(0.001)
        self.inner.write(line)


class LoggerTest(unittest.TestCase):
    def setUp(self) -> None:
        self.lm = LogManager.get()
        self.clock = counter_clock()

    def test_singleton_and_named_loggers(self):
        self.assertIs(LogManager.get(), LogManager.get())
        self.assertIs(self.lm.get_logger("a"), self.lm.get_logger("a"))
        self.assertIsNot(self.lm.get_logger("a"), self.lm.get_logger("b"))

    def test_handler_thresholds(self):
        everything, errors = MemorySink(), MemorySink()
        self.lm.configure(LoggerConfig.Builder().min_level(LogLevel.DEBUG).clock(self.clock)
                          .handler(LogLevel.DEBUG, PlainFormatter(), everything)
                          .handler(LogLevel.ERROR, PlainFormatter(), errors).build())
        log = self.lm.get_logger("t")
        log.debug("d"); log.info("i"); log.error("e")
        self.assertEqual(len(everything.lines), 3)
        self.assertEqual(len(errors.lines), 1)
        self.assertIn("[ERROR] t - e", errors.lines[0])

    def test_min_level_short_circuits(self):
        sink = MemorySink()
        self.lm.configure(LoggerConfig.Builder().min_level(LogLevel.WARN).clock(self.clock)
                          .handler(LogLevel.DEBUG, PlainFormatter(), sink).build())
        self.lm.get_logger("t").info("hidden")
        self.assertEqual(sink.lines, [])

    def test_json_escapes_quotes(self):
        sink = MemorySink()
        self.lm.configure(LoggerConfig.Builder().clock(self.clock)
                          .handler(LogLevel.INFO, JsonFormatter(), sink).build())
        self.lm.get_logger("t").info('say "hi"')
        self.assertIn('"msg":"say \\"hi\\""', sink.lines[0])

    def test_async_drops_instead_of_blocking(self):
        out = MemorySink()
        a = AsyncSink(SlowSink(out), capacity=4)
        for i in range(200):
            a.write(f"b{i}")
        a.close()
        self.assertEqual(len(out.lines) + a.dropped, 200)
        self.assertGreater(a.dropped, 0)

    def test_close_is_idempotent_and_drains(self):
        out = MemorySink()
        a = AsyncSink(out, capacity=100)
        for i in range(50):
            a.write(str(i))
        a.close(); a.close()
        self.assertEqual(out.lines, [str(i) for i in range(50)])   # order kept
        a.write("late")
        self.assertEqual(a.dropped, 1)

    def test_close_racing_writers_loses_nothing(self):
        out = MemorySink()
        a = AsyncSink(out, capacity=500)
        attempted = [0] * 4
        started = threading.Barrier(5)

        def writer(k):
            started.wait()
            for i in range(3000):
                a.write(f"w{k} {i}")
                attempted[k] += 1
        threads = [threading.Thread(target=writer, args=(k,)) for k in range(4)]
        for th in threads:
            th.start()
        started.wait()
        a.close()                             # lands in the middle of the writes
        for th in threads:
            th.join()
        self.assertEqual(len(out.lines) + a.dropped, sum(attempted))

    def test_file_sink(self):
        fd, path = tempfile.mkstemp(suffix=".log")
        os.close(fd)
        try:
            self.lm.configure(LoggerConfig.Builder().clock(self.clock)
                              .handler(LogLevel.INFO, PlainFormatter(), FileSink(path)).build())
            self.lm.get_logger("f").info("one")
            self.lm.shutdown()
            with open(path, encoding="utf-8") as f:
                self.assertEqual(len(f.read().splitlines()), 1)
        finally:
            os.remove(path)

    def test_builder_requires_handler(self):
        with self.assertRaises(ValueError):
            LoggerConfig.Builder().build()

// Logging framework in C++14: levels, CoR handlers, formatter/sink Strategy,
// a ring-buffer sink Decorator and a Meyers Singleton. Build: g++ -std=c++14 -O2 main.cpp
// Note: the MinGW 6.3 toolchain here has no std::thread, so this demo is single-threaded.
// The buffered sink keeps the printk shape: writers append to a bounded ring, a "klogd"
// step (flush) drains it. The Java and Python tabs show the real background-thread version.
#include <cassert>
#include <iostream>
#include <memory>
#include <sstream>
#include <string>
#include <vector>

enum class Level { DEBUG = 10, INFO = 20, WARN = 30, ERROR = 40 };
static const char* name(Level l) {
    switch (l) { case Level::DEBUG: return "DEBUG"; case Level::INFO: return "INFO ";
                 case Level::WARN: return "WARN "; default: return "ERROR"; }
}

struct LogMessage { long long ts; Level level; std::string logger, text; };

// Strategy 1: the "how it looks" vtable
struct Formatter { virtual ~Formatter() = default; virtual std::string format(const LogMessage&) const = 0; };
struct PlainFormatter : Formatter {
    std::string format(const LogMessage& m) const override {
        std::ostringstream o; o << m.ts << " [" << name(m.level) << "] " << m.logger << " - " << m.text;
        return o.str();
    }
};

// Strategy 2: the "where it goes" vtable (think struct file_operations: write + release)
struct Sink { virtual ~Sink() = default; virtual void write(const std::string&) = 0; virtual void close() {} };
struct ConsoleSink : Sink { void write(const std::string& s) override { std::cout << s << "\n"; } };
struct MemorySink : Sink { std::vector<std::string> lines; void write(const std::string& s) override { lines.push_back(s); } };

// Decorator: bounded ring of lines; when full, the oldest line is overwritten and counted,
// exactly like the kernel log buffer. flush() is the reader that drains it.
class RingBufferSink : public Sink {
    std::shared_ptr<Sink> target_;
    std::vector<std::string> ring_;
    size_t head_ = 0, count_ = 0;
    long lost_ = 0;
public:
    RingBufferSink(std::shared_ptr<Sink> t, size_t cap) : target_(std::move(t)), ring_(cap) {}
    void write(const std::string& s) override {
        size_t tail = (head_ + count_) % ring_.size();
        ring_[tail] = s;
        if (count_ == ring_.size()) { head_ = (head_ + 1) % ring_.size(); ++lost_; }   // overwrite oldest
        else ++count_;
    }
    void flush() {
        while (count_) { target_->write(ring_[head_]); head_ = (head_ + 1) % ring_.size(); --count_; }
    }
    void close() override { flush(); target_->close(); }
    long lost() const { return lost_; }
};

// Chain of Responsibility node
class Handler {
    Level threshold_; std::shared_ptr<Formatter> fmt_; std::shared_ptr<Sink> sink_;
    std::unique_ptr<Handler> next_;
public:
    Handler(Level t, std::shared_ptr<Formatter> f, std::shared_ptr<Sink> s)
        : threshold_(t), fmt_(std::move(f)), sink_(std::move(s)) {}
    Handler* setNext(std::unique_ptr<Handler> n) { next_ = std::move(n); return next_.get(); }
    void handle(const LogMessage& m) {
        if (static_cast<int>(m.level) >= static_cast<int>(threshold_)) sink_->write(fmt_->format(m));
        if (next_) next_->handle(m);          // pass-through: the next handler also sees it
    }
    void closeAll() { sink_->close(); if (next_) next_->closeAll(); }
};

class LogManager {
    Level min_ = Level::INFO;
    std::shared_ptr<Handler> head_;
    long long clock_ = 1700000000000LL;
    LogManager() = default;
public:
    static LogManager& get() { static LogManager inst; return inst; }   // init is thread-safe since C++11
    LogManager(const LogManager&) = delete;
    LogManager& operator=(const LogManager&) = delete;

    void configure(Level min, std::shared_ptr<Handler> head) {
        if (head_) head_->closeAll();
        min_ = min; head_ = std::move(head);
    }
    bool enabled(Level l) const { return head_ && static_cast<int>(l) >= static_cast<int>(min_); }
    void log(Level l, const std::string& logger, const std::string& text) {
        if (!enabled(l)) return;
        head_->handle(LogMessage{++clock_, l, logger, text});
    }
    void shutdown() { if (head_) head_->closeAll(); }
};

// printk-style macro: the message is only formatted when the level is enabled
#define LOG(lvl, logger, expr) do { if (LogManager::get().enabled(Level::lvl)) { \
    std::ostringstream o_; o_ << expr; LogManager::get().log(Level::lvl, logger, o_.str()); } } while (0)

int main() {
    auto plain = std::make_shared<PlainFormatter>();
    auto all = std::make_shared<MemorySink>();
    auto alerts = std::make_shared<MemorySink>();
    auto head = std::make_shared<Handler>(Level::INFO, plain, std::make_shared<ConsoleSink>());
    head->setNext(std::unique_ptr<Handler>(new Handler(Level::DEBUG, plain, all)))
        ->setNext(std::unique_ptr<Handler>(new Handler(Level::ERROR, plain, alerts)));
    LogManager& lm = LogManager::get();
    lm.configure(Level::DEBUG, head);

    std::cout << "== 1. levels walk the chain ==\n";
    LOG(DEBUG, "payments", "cache warm");
    LOG(INFO, "payments", "charge started id=" << 42);
    LOG(ERROR, "payments", "gateway timeout id=" << 42);
    assert(all->lines.size() == 3 && alerts->lines.size() == 1);

    std::cout << "== 2. min level WARN hides INFO ==\n";
    auto quiet = std::make_shared<MemorySink>();
    lm.configure(Level::WARN, std::make_shared<Handler>(Level::DEBUG, plain, quiet));
    int built = 0;
    LOG(INFO, "payments", "hidden " << ++built);   // expression never evaluated
    LOG(WARN, "payments", "shown");
    assert(quiet->lines.size() == 1 && built == 0);
    std::cout << "INFO skipped without formatting: " << (built == 0 ? "yes" : "no") << "\n";

    std::cout << "== 3. ring buffer of 8 lines, 20 writes ==\n";
    auto out = std::make_shared<MemorySink>();
    auto ring = std::make_shared<RingBufferSink>(out, 8);
    lm.configure(Level::INFO, std::make_shared<Handler>(Level::INFO, plain, ring));
    for (int i = 0; i < 20; ++i) LOG(INFO, "net", "pkt " << i);
    lm.shutdown();                                // flushes the ring
    assert(out->lines.size() == 8 && ring->lost() == 12);
    std::cout << "kept " << out->lines.size() << ", lost " << ring->lost()
              << ", oldest kept: " << out->lines.front() << "\n";
    std::cout << "All checks passed\n";
    return 0;
}
