// Prototype demo: build a cinema seat layout ONCE, then clone it for every show.
// Shows why the clone must be DEEP. Run: javac -d out SeatLayoutDemo.java && java -cp out SeatLayoutDemo
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

final class Seat {
    final String id; final String type; final int pricePaise;
    boolean booked;                                          // mutable per-show state
    Seat(String id, String type, int pricePaise) { this.id = id; this.type = type; this.pricePaise = pricePaise; }
    Seat(Seat other) { this(other.id, other.type, other.pricePaise); this.booked = other.booked; }  // copy ctor
}

final class SeatLayout {
    static int builds = 0;                                   // counts the expensive path
    private final String screen;
    private final List<Seat> seats;

    SeatLayout(String screen, int rows, int cols) {          // EXPENSIVE: imagine a DB/file read + pricing rules
        builds++;
        this.screen = screen;
        this.seats = new ArrayList<>();
        for (int r = 0; r < rows; r++)
            for (int c = 1; c <= cols; c++) {
                String type = r >= rows - 2 ? "RECLINER" : "REGULAR";
                seats.add(new Seat((char) ('A' + r) + "" + c, type, type.equals("RECLINER") ? 45_000 : 25_000));
            }
    }
    private SeatLayout(SeatLayout src, boolean deep) {       // the cloning constructor
        this.screen = src.screen;
        this.seats = new ArrayList<>();
        // <-- the pattern: copy the parts too (deep) or share them (shallow)
        for (Seat s : src.seats) seats.add(deep ? new Seat(s) : s);   // shallow = SAME Seat objects
    }
    SeatLayout deepCopy()    { return new SeatLayout(this, true); }
    SeatLayout shallowCopy() { return new SeatLayout(this, false); }

    void book(String id) { find(id).booked = true; }
    boolean isBooked(String id) { return find(id).booked; }
    long free() { return seats.stream().filter(s -> !s.booked).count(); }
    private Seat find(String id) {
        for (Seat s : seats) if (s.id.equals(id)) return s;
        throw new IllegalArgumentException(id);
    }
    @Override public String toString() { return screen + " (" + seats.size() + " seats, " + free() + " free)"; }
}

final class LayoutRegistry {                                 // prototype registry
    private final Map<String, SeatLayout> prototypes = new HashMap<>();
    void register(String key, SeatLayout proto) { prototypes.put(key, proto); }
    SeatLayout newShow(String key) {
        SeatLayout p = prototypes.get(key);
        if (p == null) throw new IllegalArgumentException("no layout " + key);
        return p.deepCopy();
    }
}

public class SeatLayoutDemo {
    public static void main(String[] args) {
        LayoutRegistry registry = new LayoutRegistry();
        registry.register("AUDI-1", new SeatLayout("AUDI-1", 10, 12));
        registry.register("IMAX",   new SeatLayout("IMAX", 20, 24));

        SeatLayout mon = registry.newShow("AUDI-1");
        SeatLayout tue = registry.newShow("AUDI-1");
        mon.book("A1"); mon.book("J5");
        System.out.println("monday  : " + mon + ", A1 booked=" + mon.isBooked("A1"));
        System.out.println("tuesday : " + tue + ", A1 booked=" + tue.isBooked("A1"));
        System.out.println("expensive builds: " + SeatLayout.builds + " (2 prototypes), shows cloned: 2");

        SeatLayout proto = new SeatLayout("AUDI-2", 5, 10);
        SeatLayout wed = proto.shallowCopy();
        wed.book("B2");
        System.out.println("shallow bug: booked B2 on wednesday, prototype B2 booked=" + proto.isBooked("B2"));
    }
}
"""Prototype demo: build a cinema seat layout ONCE, then clone it for every show.
Shows copy.copy (shallow) vs copy.deepcopy (deep). Run: python seat_layout_demo.py
"""
import copy
from dataclasses import dataclass, field


@dataclass
class Seat:
    id: str
    type: str
    price_paise: int
    booked: bool = False                      # mutable per-show state


@dataclass
class SeatLayout:
    screen: str
    seats: list = field(default_factory=list)
    builds = 0                                # class attribute: counts the expensive path

    @classmethod
    def load(cls, screen, rows, cols):        # EXPENSIVE: imagine a DB/file read + pricing rules
        cls.builds += 1
        seats = []
        for r in range(rows):
            for c in range(1, cols + 1):
                t = "RECLINER" if r >= rows - 2 else "REGULAR"
                seats.append(Seat(f"{chr(65 + r)}{c}", t, 45_000 if t == "RECLINER" else 25_000))
        return cls(screen, seats)

    def clone(self):                          # the prototype operation
        return copy.deepcopy(self)

    def book(self, seat_id):
        self._find(seat_id).booked = True

    def is_booked(self, seat_id):
        return self._find(seat_id).booked

    def free(self):
        return sum(not s.booked for s in self.seats)

    def _find(self, seat_id):
        return next(s for s in self.seats if s.id == seat_id)

    def __str__(self):
        return f"{self.screen} ({len(self.seats)} seats, {self.free()} free)"


class LayoutRegistry:                         # prototype registry
    def __init__(self):
        self._protos = {}

    def register(self, key, proto):
        self._protos[key] = proto

    def new_show(self, key):
        return self._protos[key].clone()


if __name__ == "__main__":
    reg = LayoutRegistry()
    reg.register("AUDI-1", SeatLayout.load("AUDI-1", 10, 12))
    reg.register("IMAX", SeatLayout.load("IMAX", 20, 24))

    mon, tue = reg.new_show("AUDI-1"), reg.new_show("AUDI-1")
    mon.book("A1"); mon.book("J5")
    print(f"monday  : {mon}, A1 booked={mon.is_booked('A1')}")
    print(f"tuesday : {tue}, A1 booked={tue.is_booked('A1')}")
    print(f"expensive builds: {SeatLayout.builds} (2 prototypes), shows cloned: 2")

    proto = SeatLayout.load("AUDI-2", 5, 10)
    wed = copy.copy(proto)                    # shallow: new SeatLayout, SAME list of seats
    wed.book("B2")
    print(f"shallow bug: booked B2 on wednesday, prototype B2 booked={proto.is_booked('B2')}")
// Prototype demo (C++14): polymorphic clone() through a base pointer, plus a prototype registry.
// Build: g++ -std=c++14 -Wall -o seat_layout_demo seat_layout_demo.cpp && ./seat_layout_demo
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

struct Seat { std::string id; int pricePaise; bool booked = false; };

class SeatLayout {                                     // the prototype interface + a concrete prototype
public:
    static int builds;
    SeatLayout(std::string screen, int rows, int cols) : screen_(std::move(screen)) {   // EXPENSIVE path
        ++builds;
        for (int r = 0; r < rows; ++r)
            for (int c = 1; c <= cols; ++c)
                seats_.push_back({std::string(1, char('A' + r)) + std::to_string(c), r >= rows - 2 ? 45000 : 25000});
    }
    virtual ~SeatLayout() = default;
    virtual std::unique_ptr<SeatLayout> clone() const {          // "virtual copy constructor"
        return std::make_unique<SeatLayout>(*this);              // vector<Seat> by value => deep copy
    }
    virtual std::string describe() const { return screen_ + " (" + std::to_string(seats_.size()) + " seats, " + std::to_string(free()) + " free)"; }
    void book(const std::string& id) { find(id).booked = true; }
    bool isBooked(const std::string& id) { return find(id).booked; }
    int free() const { int n = 0; for (const auto& s : seats_) n += !s.booked; return n; }
protected:
    std::string screen_;
    std::vector<Seat> seats_;
private:
    Seat& find(const std::string& id) {
        for (auto& s : seats_) if (s.id == id) return s;
        throw std::invalid_argument(id);
    }
};
int SeatLayout::builds = 0;

class ImaxLayout : public SeatLayout {                  // a second concrete prototype with extra state
    int glasses_;
public:
    ImaxLayout(int rows, int cols, int glasses) : SeatLayout("IMAX", rows, cols), glasses_(glasses) {}
    std::unique_ptr<SeatLayout> clone() const override { return std::make_unique<ImaxLayout>(*this); }
    std::string describe() const override { return SeatLayout::describe() + " + " + std::to_string(glasses_) + " 3D glasses"; }
};

class LayoutRegistry {                                  // prototype registry
    std::map<std::string, std::unique_ptr<SeatLayout>> protos_;
public:
    void add(const std::string& key, std::unique_ptr<SeatLayout> p) { protos_[key] = std::move(p); }
    std::unique_ptr<SeatLayout> newShow(const std::string& key) const {
        auto it = protos_.find(key);
        if (it == protos_.end()) throw std::invalid_argument("no layout " + key);
        return it->second->clone();                     // caller never names the concrete class
    }
};

int main() {
    LayoutRegistry reg;
    reg.add("AUDI-1", std::make_unique<SeatLayout>("AUDI-1", 10, 12));
    reg.add("IMAX", std::make_unique<ImaxLayout>(20, 24, 480));

    auto mon = reg.newShow("AUDI-1");
    auto tue = reg.newShow("AUDI-1");
    mon->book("A1"); mon->book("J5");
    std::cout << "monday  : " << mon->describe() << ", A1 booked=" << mon->isBooked("A1") << "\n";
    std::cout << "tuesday : " << tue->describe() << ", A1 booked=" << tue->isBooked("A1") << "\n";
    std::cout << "imax    : " << reg.newShow("IMAX")->describe() << "\n";
    std::cout << "expensive builds: " << SeatLayout::builds << " (2 prototypes), shows cloned: 3\n";

    // Shallow copies come from POINTERS: copying a vector of shared_ptr shares the seats.
    std::vector<std::shared_ptr<Seat>> proto{std::make_shared<Seat>(Seat{"B2", 25000})};
    auto wed = proto;                                   // copies pointers, not seats
    wed[0]->booked = true;
    std::cout << "shallow bug: booked B2 on wednesday, prototype B2 booked=" << proto[0]->booked << "\n";
    return 0;
}

  • Java uses a private copying constructor plus deepCopy(), not Cloneable. That is the modern advice; see the next section for why.
  • Python gives you both copies for free: copy.copy() is shallow and copy.deepcopy() is deep and handles cycles. Classes can customise them with __copy__ and __deepcopy__ (for example to skip a lock or a socket).
  • C++ shows the polymorphic part: the registry holds unique_ptr<SeatLayout>, and clone() returns an ImaxLayout when the stored object is one. A std::vector<Seat> member is copied by value, so C++ value semantics make the deep copy automatic; shallow copies come from pointers, as the last lines show.
