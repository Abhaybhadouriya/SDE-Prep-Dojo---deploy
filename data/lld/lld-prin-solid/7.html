How SOLID maps to design patterns

Patterns are SOLID applied to a recurring problem. When you name a pattern in an interview, also name the principle it serves:

PrinciplePatterns that put it into practice
SRPFacade (one entry point, work delegated), Repository (storage pulled out of the domain), Command (each action its own class)
OCPStrategy, Decorator, Observer, Chain of Responsibility, Template Method, Factory with a registry
LSPEvery pattern that relies on polymorphism assumes it; Null Object is LSP-friendly "do nothing" behaviour
ISPRole interfaces, Adapter (expose only what the client needs)
DIPDependency Injection, Abstract Factory, Repository, Observer (subject depends on the listener interface)

All five, runnable

Every before/after on this page lives in one file with a driver. Compiled and run with Java 21 and Python 3.11.

// lld-prin-solid: one small "before" and "after" for each SOLID letter, plus a driver.
// Build and run (Java 21):  javac -d out Solid.java && java -cp out Solid
import java.util.*;

public class Solid {

    // ===== S: Single Responsibility =====
    // BEFORE: one class with three reasons to change (pricing rules, output format, storage)
    static class InvoiceGod {
        final Map<String, Long> lines = new LinkedHashMap<>();
        long total() { return lines.values().stream().mapToLong(Long::longValue).sum(); }
        String print() { return "INVOICE total=" + total(); }               // format changes -> edit here
        void save(List<String> db) { db.add("invoice," + total()); }        // storage changes -> edit here too
    }
    // AFTER: each class has one reason to change
    record Invoice(Map<String, Long> lines) {
        Invoice { lines = Map.copyOf(lines); }
        long total() { return lines.values().stream().mapToLong(Long::longValue).sum(); }
    }
    static class InvoicePrinter { String print(Invoice i) { return "INVOICE total=" + i.total(); } }
    static class InvoiceRepository {
        final List<String> rows = new ArrayList<>();
        void save(Invoice i) { rows.add("invoice," + i.total()); }
    }

    // ===== O: Open/Closed =====
    // BEFORE: every new customer type edits this method
    static long discountBefore(String type, long amount) {
        if (type.equals("REGULAR")) return 0;
        else if (type.equals("PREMIUM")) return amount / 10;
        else if (type.equals("EMPLOYEE")) return amount / 5;
        throw new IllegalArgumentException(type);
    }
    // AFTER: new rule = new class; the calculator is closed for modification
    interface DiscountRule { boolean applies(String customerType); long discount(long amount); }
    record PercentOff(String type, int percent) implements DiscountRule {
        public boolean applies(String t) { return t.equals(type); }
        public long discount(long amount) { return amount * percent / 100; }
    }
    static class DiscountCalculator {
        private final List<DiscountRule> rules;
        DiscountCalculator(List<DiscountRule> rules) { this.rules = List.copyOf(rules); }
        long discount(String type, long amount) {
            return rules.stream().filter(r -> r.applies(type)).mapToLong(r -> r.discount(amount)).max().orElse(0);
        }
    }

    // ===== L: Liskov Substitution =====
    // BEFORE: a subtype that cannot keep the parent's promise
    static class BirdBefore { String fly() { return "flap flap"; } }
    static class PenguinBefore extends BirdBefore {
        @Override String fly() { throw new UnsupportedOperationException("penguins cannot fly"); }
    }
    // AFTER: only promise what every subtype can do
    interface Bird { String name(); }
    interface FlyingBird extends Bird { String fly(); }
    record Sparrow(String name) implements FlyingBird { public String fly() { return name + " flies"; } }
    record Penguin(String name) implements Bird {}

    // ===== I: Interface Segregation =====
    // BEFORE: a fat interface forces dummy methods
    interface MachineBefore { void print(String doc); String scan(String doc); void fax(String doc, String number); }
    static class CheapPrinterBefore implements MachineBefore {
        public void print(String d) { System.out.println("printed " + d); }
        public String scan(String d) { throw new UnsupportedOperationException(); }   // forced to lie
        public void fax(String d, String number) { throw new UnsupportedOperationException(); }
    }
    // AFTER: small role interfaces; a class picks the roles it really has
    interface Printer { void print(String doc); }
    interface Scanner { String scan(String doc); }
    static class CheapPrinter implements Printer { public void print(String d) { System.out.println("printed " + d); } }
    static class OfficeMachine implements Printer, Scanner {
        public void print(String d) { System.out.println("printed " + d); }
        public String scan(String d) { return "scanned " + d; }
    }

    // ===== D: Dependency Inversion =====
    // BEFORE: high-level policy creates a low-level detail itself
    static class EmailSenderBefore { void send(String to, String m) { System.out.println("EMAIL to " + to + ": " + m); } }
    static class OrderServiceBefore {
        private final EmailSenderBefore email = new EmailSenderBefore();   // hard-wired, untestable
        void place(String user) { email.send(user, "order placed"); }
    }
    // AFTER: both sides depend on an abstraction owned by the policy layer
    // (named send, not notify, so it cannot be confused with Object.notify())
    interface Notifier { void send(String to, String message); }
    static class EmailNotifier implements Notifier { public void send(String to, String m) { System.out.println("EMAIL to " + to + ": " + m); } }
    static class SmsNotifier implements Notifier { public void send(String to, String m) { System.out.println("SMS to " + to + ": " + m); } }
    static class FakeNotifier implements Notifier {                          // for tests
        final List<String> sent = new ArrayList<>();
        public void send(String to, String m) { sent.add(to + "|" + m); }
    }
    static class OrderService {
        private final Notifier notifier;
        OrderService(Notifier notifier) { this.notifier = notifier; }      // constructor injection
        void place(String user) { notifier.send(user, "order placed"); }
    }

    public static void main(String[] args) {
        System.out.println("== S");
        Invoice inv = new Invoice(Map.of("dosa", 6000L, "coffee", 2500L));
        InvoiceRepository repo = new InvoiceRepository();
        repo.save(inv);
        System.out.println(new InvoicePrinter().print(inv) + " | stored: " + repo.rows);

        System.out.println("== O");
        DiscountCalculator calc = new DiscountCalculator(List.of(new PercentOff("PREMIUM", 10), new PercentOff("EMPLOYEE", 20)));
        System.out.println("before PREMIUM 1000 -> " + discountBefore("PREMIUM", 1000) + ", after -> " + calc.discount("PREMIUM", 1000));
        DiscountCalculator festive = new DiscountCalculator(List.of(new PercentOff("PREMIUM", 10), new PercentOff("EMPLOYEE", 20),
                new PercentOff("STUDENT", 15)));                             // new rule, zero edits to the calculator
        System.out.println("new STUDENT rule -> " + festive.discount("STUDENT", 1000) + ", unknown type -> " + festive.discount("GUEST", 1000));

        System.out.println("== L");
        List<BirdBefore> before = List.of(new BirdBefore(), new PenguinBefore());
        for (BirdBefore b : before) {
            try { System.out.println("before: " + b.fly()); }
            catch (UnsupportedOperationException e) { System.out.println("before: surprise! " + e.getMessage()); }
        }
        List<FlyingBird> flyers = List.of(new Sparrow("sparrow"));            // a Penguin cannot even be added here
        flyers.forEach(f -> System.out.println("after: " + f.fly()));
        List<Bird> all = List.of(new Sparrow("sparrow"), new Penguin("penguin"));
        System.out.println("after: all birds = " + all.stream().map(Bird::name).toList());

        System.out.println("== I");
        List<Printer> printers = List.of(new CheapPrinter(), new OfficeMachine());
        printers.forEach(p -> p.print("report.pdf"));
        Scanner s = new OfficeMachine();
        System.out.println(s.scan("id-card.png"));
        try { new CheapPrinterBefore().scan("x"); } catch (UnsupportedOperationException e) { System.out.println("before: fat interface forced a fake scan()"); }

        System.out.println("== D");
        System.out.print("before: "); new OrderServiceBefore().place("asha");
        System.out.print("after:  "); new OrderService(new EmailNotifier()).place("asha");
        System.out.print("after:  "); new OrderService(new SmsNotifier()).place("asha");
        FakeNotifier fake = new FakeNotifier();
        new OrderService(fake).place("ravi");
        System.out.println("test double recorded: " + fake.sent);
    }
}

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Protocol


# S: one reason to change per class
@dataclass(frozen=True)
class Invoice:
    lines: tuple[tuple[str, int], ...]

    def total(self) -> int:
        return sum(p for _, p in self.lines)


class InvoicePrinter:
    def print(self, inv: Invoice) -> str:
        return f"INVOICE total={inv.total()}"


class InvoiceRepository:
    def __init__(self) -> None:
        self.rows: list[str] = []

    def save(self, inv: Invoice) -> None:
        self.rows.append(f"invoice,{inv.total()}")


# O: add a rule by adding a class
class DiscountRule(ABC):
    @abstractmethod
    def applies(self, customer_type: str) -> bool: ...

    @abstractmethod
    def discount(self, amount: int) -> int: ...


@dataclass(frozen=True)
class PercentOff(DiscountRule):
    type: str
    percent: int

    def applies(self, customer_type: str) -> bool:
        return customer_type == self.type

    def discount(self, amount: int) -> int:
        return amount * self.percent // 100


class DiscountCalculator:
    def __init__(self, rules: list[DiscountRule]) -> None:
        self._rules = tuple(rules)

    def discount(self, customer_type: str, amount: int) -> int:
        return max((r.discount(amount) for r in self._rules if r.applies(customer_type)), default=0)


# L: only promise what every subtype can keep
class Bird(Protocol):
    name: str


class FlyingBird(Bird, Protocol):
    def fly(self) -> str: ...


@dataclass
class Sparrow:
    name: str

    def fly(self) -> str:
        return f"{self.name} flies"


@dataclass
class Penguin:
    name: str


# I: small role interfaces
class Printer(Protocol):
    def print_doc(self, doc: str) -> None: ...


class Scanner(Protocol):
    def scan(self, doc: str) -> str: ...


class CheapPrinter:
    def print_doc(self, doc: str) -> None:
        print(f"printed {doc}")


class OfficeMachine:
    def print_doc(self, doc: str) -> None:
        print(f"printed {doc}")

    def scan(self, doc: str) -> str:
        return f"scanned {doc}"


# D: depend on an abstraction, inject the detail
class Notifier(Protocol):
    def send(self, to: str, message: str) -> None: ...


class SmsNotifier:
    def send(self, to: str, message: str) -> None:
        print(f"SMS to {to}: {message}")


class FakeNotifier:
    def __init__(self) -> None:
        self.sent: list[str] = []

    def send(self, to: str, message: str) -> None:
        self.sent.append(f"{to}|{message}")


class OrderService:
    def __init__(self, notifier: Notifier) -> None:
        self._notifier = notifier

    def place(self, user: str) -> None:
        self._notifier.send(user, "order placed")


if __name__ == "__main__":
    inv = Invoice((("dosa", 6000), ("coffee", 2500)))
    repo = InvoiceRepository()
    repo.save(inv)
    print(InvoicePrinter().print(inv), repo.rows)

    print("STUDENT 1000 ->", calc.discount("STUDENT", 1000), "GUEST ->", calc.discount("GUEST", 1000))
    flyers: list[FlyingBird] = [Sparrow("sparrow")]
    print([f.fly() for f in flyers], [b.name for b in (Sparrow("sparrow"), Penguin("penguin"))])
    printers: list[Printer] = [CheapPrinter(), OfficeMachine()]
    for p in printers:
        p.print_doc("report.pdf")
    print(OfficeMachine().scan("id.png"))
