// lld-oop-toolkit: the Java language tools every machine-coding answer leans on.
// Build and run (Java 21):  javac -d out Toolkit.java && java -cp out Toolkit
import java.util.*;
import java.util.function.*;

public class Toolkit {

    // ---------- 1. Rich enums: data + behaviour + a tiny state machine ----------
    enum VehicleType {
        BIKE(1, 20), CAR(2, 50), TRUCK(4, 120);          // slots needed, rupees per hour

        private final int slots;
        private final int ratePerHour;
        VehicleType(int slots, int ratePerHour) { this.slots = slots; this.ratePerHour = ratePerHour; }
        int slots() { return slots; }
        long fee(int hours) { return (long) Math.max(1, hours) * ratePerHour; }
    }

    enum OrderStatus {
        CREATED, PAID, SHIPPED, DELIVERED, CANCELLED;

        // allowed moves live next to the states, not in a switch spread over the codebase
        private static final Map<OrderStatus, Set<OrderStatus>> NEXT = new EnumMap<>(OrderStatus.class);
        static {
            NEXT.put(CREATED, EnumSet.of(PAID, CANCELLED));
            NEXT.put(PAID, EnumSet.of(SHIPPED, CANCELLED));
            NEXT.put(SHIPPED, EnumSet.of(DELIVERED));
            NEXT.put(DELIVERED, EnumSet.noneOf(OrderStatus.class));
            NEXT.put(CANCELLED, EnumSet.noneOf(OrderStatus.class));
        }
        boolean canMoveTo(OrderStatus to) { return NEXT.get(this).contains(to); }
    }

    // constant-specific bodies: each constant overrides the abstract method (an enum as a closed Strategy set)
    enum SplitType {
        EQUAL { long[] split(long total, int n) { long[] r = new long[n]; Arrays.fill(r, total / n); r[0] += total % n; return r; } },
        FIRST_PAYS_ALL { long[] split(long total, int n) { long[] r = new long[n]; r[0] = total; return r; } };
        abstract long[] split(long total, int n);
    }

    // ---------- 2. A custom exception hierarchy ----------
    static class DomainException extends RuntimeException {
        private final String code;
        DomainException(String code, String msg) { super(msg); this.code = code; }
        String code() { return code; }
    }
    static class NotFoundException extends DomainException {
        NotFoundException(String what, Object id) { super("NOT_FOUND", what + " " + id + " not found"); }
    }
    static class InvalidStateException extends DomainException {
        InvalidStateException(OrderStatus from, OrderStatus to) { super("BAD_STATE", "cannot move " + from + " -> " + to); }
    }
    static class InsufficientFundsException extends DomainException {
        InsufficientFundsException(Money need, Money have) { super("NO_FUNDS", "need " + need + ", have " + have); }
    }

    // ---------- 3. Records = immutable value objects ----------
    record Money(long paise, String currency) implements Comparable<Money> {
        Money {                                             // compact constructor: validate once, forever true
            Objects.requireNonNull(currency, "currency");
            if (paise < 0) throw new IllegalArgumentException("negative money: " + paise);
        }
        static Money inr(long rupees) { return new Money(rupees * 100, "INR"); }
        Money plus(Money o) { same(o); return new Money(paise + o.paise, currency); }     // returns a NEW object
        Money minus(Money o) { same(o); return new Money(paise - o.paise, currency); }
        private void same(Money o) { if (!currency.equals(o.currency)) throw new IllegalArgumentException("currency mismatch"); }
        @Override public int compareTo(Money o) { same(o); return Long.compare(paise, o.paise); }
        @Override public String toString() { return currency + " " + paise / 100 + "." + String.format("%02d", paise % 100); }
    }

    // a hand-written immutable class (what a record saves you from typing) with a defensive copy
    static final class Order {
        private final String id;
        private final List<String> items;                   // never exposed as a mutable list
        private final OrderStatus status;
        Order(String id, List<String> items, OrderStatus status) {
            this.id = id; this.items = List.copyOf(items); this.status = status;
        }
        String id() { return id; }
        List<String> items() { return items; }               // already unmodifiable
        OrderStatus status() { return status; }
        Order moveTo(OrderStatus to) {                        // "with-er": a new Order, the old one is untouched
            if (!status.canMoveTo(to)) throw new InvalidStateException(status, to);
            return new Order(id, items, to);
        }
    }

    // ---------- 4. Generics: one repository for every entity ----------
    interface Identifiable<ID> { ID id(); }
    record User(String id, String name, int age) implements Identifiable<String> {}
    record Spot(Integer id, int floor, int distance) implements Identifiable<Integer> {}

    static class InMemoryRepo<ID, T extends Identifiable<ID>> {
        private final Map<ID, T> rows = new LinkedHashMap<>();
        void save(T t) { rows.put(t.id(), t); }
        Optional<T> findById(ID id) { return Optional.ofNullable(rows.get(id)); }   // absence is part of the type
        T get(ID id, String what) { return findById(id).orElseThrow(() -> new NotFoundException(what, id)); }
        List<T> findAll(Predicate<? super T> p) { return rows.values().stream().filter(p).toList(); }
    }

    // bounded + wildcard generic method: PECS (producer extends, consumer super)
    static <T> T best(Collection<? extends T> items, Comparator<? super T> order) {
        if (items.isEmpty()) throw new NoSuchElementException("empty");
        return Collections.max(items, order);
    }

    // ---------- 5. equals/hashCode: the HashSet trap ----------
    static final class BadPoint {                           // equals WITHOUT hashCode
        final int x, y;
        BadPoint(int x, int y) { this.x = x; this.y = y; }
        @Override public boolean equals(Object o) { return o instanceof BadPoint p && p.x == x && p.y == y; }
    }
    static final class GoodPoint {
        final int x, y;
        GoodPoint(int x, int y) { this.x = x; this.y = y; }
        @Override public boolean equals(Object o) { return o instanceof GoodPoint p && p.x == x && p.y == y; }
        @Override public int hashCode() { return Objects.hash(x, y); }
    }

    // ---------- driver ----------
    public static void main(String[] args) {
        System.out.println("== 1. enums");
        for (VehicleType v : VehicleType.values()) System.out.println(v + " slots=" + v.slots() + " fee(3h)=" + v.fee(3));
        System.out.println("CREATED->PAID? " + OrderStatus.CREATED.canMoveTo(OrderStatus.PAID)
                + "  DELIVERED->CANCELLED? " + OrderStatus.DELIVERED.canMoveTo(OrderStatus.CANCELLED));
        System.out.println("EQUAL split of 100 over 3: " + Arrays.toString(SplitType.EQUAL.split(100, 3)));
        EnumMap<VehicleType, Integer> free = new EnumMap<>(VehicleType.class);
        free.put(VehicleType.CAR, 10); free.put(VehicleType.BIKE, 25);
        System.out.println("EnumMap keeps declaration order: " + free);

        System.out.println("== 2. records and immutability");
        Money a = Money.inr(150), b = Money.inr(150);
        System.out.println(a + " equals " + b + "? " + a.equals(b) + "  same hash? " + (a.hashCode() == b.hashCode()));
        Money c = a.plus(Money.inr(50));
        System.out.println("a.plus(50) = " + c + ", a is still " + a);
        try { new Money(-1, "INR"); } catch (IllegalArgumentException e) { System.out.println("rejected: " + e.getMessage()); }
        List<String> cart = new ArrayList<>(List.of("milk", "bread"));
        Order o1 = new Order("o1", cart, OrderStatus.CREATED);
        cart.add("caviar");                                  // caller mutates its own list later...
        System.out.println("order items after caller mutation: " + o1.items());
        try { o1.items().add("x"); } catch (UnsupportedOperationException e) { System.out.println("items() is read-only"); }
        Order o2 = o1.moveTo(OrderStatus.PAID);
        System.out.println("o1=" + o1.status() + " o2=" + o2.status());

        System.out.println("== 3. exceptions");
        try { o2.moveTo(OrderStatus.DELIVERED); }
        catch (DomainException e) { System.out.println("[" + e.code() + "] " + e.getMessage()); }
        try { throw new InsufficientFundsException(Money.inr(500), Money.inr(120)); }
        catch (DomainException e) { System.out.println("[" + e.code() + "] " + e.getMessage()); }

        System.out.println("== 4. generics + Optional");
        InMemoryRepo<String, User> users = new InMemoryRepo<>();
        users.save(new User("u1", "Asha", 31)); users.save(new User("u2", "Ravi", 27));
        System.out.println("find u1: " + users.findById("u1").map(User::name).orElse("nobody"));
        System.out.println("find u9: " + users.findById("u9").map(User::name).orElse("nobody"));
        try { users.get("u9", "user"); } catch (NotFoundException e) { System.out.println("[" + e.code() + "] " + e.getMessage()); }
        System.out.println("oldest: " + best(users.findAll(u -> true), Comparator.comparingInt(User::age)).name());

        System.out.println("== 5. equals/hashCode");
        Set<BadPoint> bad = new HashSet<>(); bad.add(new BadPoint(1, 2));
        Set<GoodPoint> good = new HashSet<>(); good.add(new GoodPoint(1, 2));
        System.out.println("BadPoint set contains (1,2)? " + bad.contains(new BadPoint(1, 2)));
        System.out.println("GoodPoint set contains (1,2)? " + good.contains(new GoodPoint(1, 2)));

        System.out.println("== 6. Comparator chains");
        InMemoryRepo<Integer, Spot> spots = new InMemoryRepo<>();
        spots.save(new Spot(1, 2, 5)); spots.save(new Spot(2, 0, 40)); spots.save(new Spot(3, 0, 10)); spots.save(new Spot(4, 1, 1));
        Comparator<Spot> nearest = Comparator.comparingInt(Spot::floor).thenComparingInt(Spot::distance);
        List<Spot> sorted = new ArrayList<>(spots.findAll(s -> true));
        sorted.sort(nearest);
        System.out.println("by floor, then distance: " + sorted.stream().map(Spot::id).toList());
        sorted.sort(nearest.reversed());
        System.out.println("reversed: " + sorted.stream().map(Spot::id).toList());
        PriorityQueue<Spot> pq = new PriorityQueue<>(nearest);
        pq.addAll(spots.findAll(s -> true));
        System.out.println("PriorityQueue head (allocate next): spot " + pq.peek().id());
    }
}

from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Generic, Hashable, Optional, Protocol, TypeVar


class VehicleType(Enum):
    BIKE = (1, 20)
    CAR = (2, 50)
    TRUCK = (4, 120)

    def __init__(self, slots: int, rate: int):
        self.slots = slots
        self.rate = rate

    def fee(self, hours: int) -> int:
        return max(1, hours) * self.rate


class OrderStatus(Enum):
    CREATED = "created"
    PAID = "paid"
    SHIPPED = "shipped"
    DELIVERED = "delivered"
    CANCELLED = "cancelled"

    def can_move_to(self, to: "OrderStatus") -> bool:
        return to in _NEXT[self]


_NEXT = {
    OrderStatus.CREATED: {OrderStatus.PAID, OrderStatus.CANCELLED},
    OrderStatus.PAID: {OrderStatus.SHIPPED, OrderStatus.CANCELLED},
    OrderStatus.SHIPPED: {OrderStatus.DELIVERED},
    OrderStatus.DELIVERED: set(),
    OrderStatus.CANCELLED: set(),
}


class DomainError(Exception):
    code = "DOMAIN"


class NotFoundError(DomainError):
    code = "NOT_FOUND"


class InvalidStateError(DomainError):
    code = "BAD_STATE"


@dataclass(frozen=True)          # frozen = immutable; eq and hash are generated from the fields
class Money:
    paise: int
    currency: str = "INR"

    def __post_init__(self):
        if self.paise < 0:
            raise ValueError(f"negative money: {self.paise}")

    def plus(self, other: "Money") -> "Money":
        assert self.currency == other.currency, "currency mismatch"
        return Money(self.paise + other.paise, self.currency)


@dataclass(frozen=True)
class Order:
    id: str
    items: tuple[str, ...]        # a tuple, not a list: nobody can append later
    status: OrderStatus = OrderStatus.CREATED

    def move_to(self, to: OrderStatus) -> "Order":
        if not self.status.can_move_to(to):
            raise InvalidStateError(f"cannot move {self.status.name} -> {to.name}")
        return Order(self.id, self.items, to)


class HasId(Protocol):
    @property
    def id(self) -> Hashable: ...


T = TypeVar("T", bound=HasId)


class InMemoryRepo(Generic[T]):
    def __init__(self) -> None:
        self._rows: dict[Hashable, T] = {}

    def save(self, row: T) -> None:
        self._rows[row.id] = row

    def find_by_id(self, id_: Hashable) -> Optional[T]:
        return self._rows.get(id_)

    def get(self, id_: Hashable) -> T:
        row = self.find_by_id(id_)
        if row is None:
            raise NotFoundError(f"{id_} not found")
        return row

    def find_all(self, pred: Callable[[T], bool] = lambda r: True) -> list[T]:
        return [r for r in self._rows.values() if pred(r)]


@dataclass(frozen=True)
class Spot:
    id: int
    floor: int
    distance: int
