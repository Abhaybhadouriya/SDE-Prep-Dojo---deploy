import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

// Snake game (Nokia style). One file for the interview; split into model/, rules/, game/ in a real repo.
// Run: javac -d out Main.java && java -cp out Main
public class Main {

    // ---------- model ----------
    enum Direction {
        UP(-1, 0), DOWN(1, 0), LEFT(0, -1), RIGHT(0, 1);
        final int dr, dc;
        Direction(int dr, int dc) { this.dr = dr; this.dc = dc; }
        boolean isOpposite(Direction o) { return dr == -o.dr && dc == -o.dc; }
    }

    static final class Position {
        final int row, col;
        Position(int row, int col) { this.row = row; this.col = col; }
        @Override public boolean equals(Object o) {
            if (!(o instanceof Position)) return false;
            Position p = (Position) o;
            return row == p.row && col == p.col;
        }
        @Override public int hashCode() { return row * 31 + col; }
        @Override public String toString() { return "(" + row + "," + col + ")"; }
    }

    enum Status { RUNNING, WON, OVER }
    enum Cause { NONE, HIT_WALL, HIT_SELF, BOARD_FULL }

    static final class Board {
        final int rows, cols;
        Board(int rows, int cols) {
            if (rows < 1 || cols < 1) throw new IllegalArgumentException("board must be at least 1 x 1");
            this.rows = rows; this.cols = cols;
        }
        boolean inside(int r, int c) { return r >= 0 && r < rows && c >= 0 && c < cols; }
        int cells() { return rows * cols; }
    }

    // ---------- the snake: deque for order + hash set for O(1) "is this cell mine?" ----------
    static final class Snake {
        private final Deque<Position> body = new ArrayDeque<>();   // head first
        private final Set<Position> cells = new HashSet<>();
        private int pendingGrowth = 0;

        Snake(Position start) { body.addFirst(start); cells.add(start); }
        Position head() { return body.peekFirst(); }
        Position tail() { return body.peekLast(); }
        int length() { return body.size(); }
        boolean occupies(Position p) { return cells.contains(p); }
        void grow(int n) { pendingGrowth += n; }
        List<Position> bodyHeadFirst() { return new ArrayList<>(body); }

        // Moving into the cell the tail is leaving is legal, unless the tail stays this tick (growing).
        boolean wouldBiteItself(Position next) {
            if (!cells.contains(next)) return false;
            return !(next.equals(tail()) && pendingGrowth == 0);
        }

        // O(1): add a head; drop the tail unless we owe growth.
        void advance(Position next) {
            if (pendingGrowth > 0) {
                pendingGrowth--;
            } else {
                cells.remove(body.removeLast());   // remove first: the head may enter the old tail cell
            }
            body.addFirst(next);
            cells.add(next);
        }

        boolean consistent() { return body.size() == cells.size() && cells.containsAll(body); }
    }

    // ---------- strategies ----------
    interface BoundaryPolicy {   // where does the head go? null = it hit a wall
        Position next(Position head, Direction d, Board b);
    }
    static final class WallBoundary implements BoundaryPolicy {
        public Position next(Position h, Direction d, Board b) {
            int r = h.row + d.dr, c = h.col + d.dc;
            return b.inside(r, c) ? new Position(r, c) : null;
        }
    }
    static final class WrapBoundary implements BoundaryPolicy {   // torus: leave right, enter left
        public Position next(Position h, Direction d, Board b) {
            return new Position(Math.floorMod(h.row + d.dr, b.rows), Math.floorMod(h.col + d.dc, b.cols));
        }
    }

    interface FoodSpawner {   // null = no food to place
        Position next(Board b, Snake s);
    }
    // LeetCode 353 style: food comes from a fixed list, in order. A cell under the snake is skipped.
    static final class FixedFoodSpawner implements FoodSpawner {
        private final Deque<Position> queue;
        FixedFoodSpawner(List<Position> foods) { queue = new ArrayDeque<>(foods); }
        public Position next(Board b, Snake s) {
            while (!queue.isEmpty()) {
                Position p = queue.pollFirst();
                if (!s.occupies(p)) return p;
            }
            return null;
        }
    }
    // Random free cell: a few random tries (fast while the board is empty), then scan the free cells.
    static final class RandomFoodSpawner implements FoodSpawner {
        private final Random rnd;
        RandomFoodSpawner(long seed) { rnd = new Random(seed); }
        public Position next(Board b, Snake s) {
            if (s.length() >= b.cells()) return null;
            for (int i = 0; i < 8; i++) {
                Position p = new Position(rnd.nextInt(b.rows), rnd.nextInt(b.cols));
                if (!s.occupies(p)) return p;
            }
            List<Position> free = new ArrayList<>();
            for (int r = 0; r < b.rows; r++)
                for (int c = 0; c < b.cols; c++) {
                    Position p = new Position(r, c);
                    if (!s.occupies(p)) free.add(p);
                }
            return free.get(rnd.nextInt(free.size()));
        }
    }

    interface GrowthRule {   // how many cells to add this tick
        int growth(boolean ateFood, long tick);
    }
    static final class FoodGrowth implements GrowthRule {
        public int growth(boolean ate, long tick) { return ate ? 1 : 0; }
    }
    static final class EveryNTicksGrowth implements GrowthRule {   // follow-up: "grow every N moves"
        private final int n;
        EveryNTicksGrowth(int n) { this.n = n; }
        public int growth(boolean ate, long tick) { return (ate ? 1 : 0) + (tick % n == 0 ? 1 : 0); }
    }

    // ---------- observer ----------
    interface GameListener {
        default void onFoodEaten(int score, Position at) {}
        default void onGameEnd(Status status, Cause cause, int score) {}
    }

    // ---------- the game: owns state, rules and the input buffer ----------
    static final class Game {
        private static final int MAX_BUFFERED_TURNS = 2;
        private final Board board;
        private final Snake snake;
        private final BoundaryPolicy boundary;
        private final FoodSpawner spawner;
        private final GrowthRule growth;
        private final List<GameListener> listeners = new CopyOnWriteArrayList<>();
        private final Deque<Direction> turns = new ArrayDeque<>();
        private Direction direction;
        private Position food;
        private Status status = Status.RUNNING;
        private Cause cause = Cause.NONE;
        private int score = 0;
        private long tick = 0;

        Game(Board board, Position start, Direction dir, BoundaryPolicy boundary, FoodSpawner spawner, GrowthRule growth) {
            if (!board.inside(start.row, start.col)) throw new IllegalArgumentException("start " + start + " is off the board");
            this.board = board; this.snake = new Snake(start); this.direction = dir;
            this.boundary = boundary; this.spawner = spawner; this.growth = growth;
            this.food = spawner.next(board, snake);
        }

        // LeetCode 353 API: height x width, start (0,0) going right, walls, fixed food list.
        static Game classic(int height, int width, int[][] food) {
            List<Position> f = new ArrayList<>();
            for (int[] p : food) f.add(new Position(p[0], p[1]));
            return new Game(new Board(height, width), new Position(0, 0), Direction.RIGHT,
                    new WallBoundary(), new FixedFoodSpawner(f), new FoodGrowth());
        }

        void addListener(GameListener l) { listeners.add(l); }

        // Input thread calls this. A reverse turn (or a repeat) is ignored; at most 2 turns are buffered.
        synchronized boolean turn(Direction d) {
            Direction last = turns.isEmpty() ? direction : turns.peekLast();
            if (status != Status.RUNNING || d == last || turns.size() >= MAX_BUFFERED_TURNS) return false;
            if (d.isOpposite(last) && snake.length() > 1) return false;
            turns.addLast(d);
            return true;
        }

        // Game-loop thread calls this once per tick. Returns the status after the tick.
        synchronized Status tick() {
            if (status != Status.RUNNING) return status;
            tick++;
            if (!turns.isEmpty()) direction = turns.pollFirst();
            Position next = boundary.next(snake.head(), direction, board);
            if (next == null) return end(Status.OVER, Cause.HIT_WALL);
            boolean ate = next.equals(food);
            snake.grow(growth.growth(ate, tick));
            if (snake.wouldBiteItself(next)) return end(Status.OVER, Cause.HIT_SELF);
            snake.advance(next);
            if (ate) {
                score++;
                for (GameListener l : listeners) l.onFoodEaten(score, next);
                food = spawner.next(board, snake);   // the spawner never returns a cell under the body
            }
            if (snake.length() == board.cells()) return end(Status.WON, Cause.BOARD_FULL);
            return status;
        }

        // LeetCode 353: move(direction) returns the score, or -1 once the game is over.
        synchronized int move(Direction d) {
            turn(d);
            return tick() == Status.OVER ? -1 : score;
        }

        private Status end(Status s, Cause c) {
            status = s; cause = c;
            for (GameListener l : listeners) l.onGameEnd(s, c, score);
            return s;
        }

        synchronized Status status() { return status; }
        synchronized Cause cause() { return cause; }
        synchronized int score() { return score; }
        synchronized Position head() { return snake.head(); }
        synchronized Position tail() { return snake.tail(); }
        synchronized int length() { return snake.length(); }
        synchronized Position food() { return food; }
        synchronized Direction direction() { return direction; }
        synchronized boolean invariantsHold() {
            return snake.consistent() && (food == null || !snake.occupies(food));
        }

        synchronized String render() {
            char[][] g = new char[board.rows][board.cols];
            for (char[] row : g) Arrays.fill(row, '.');
            if (food != null) g[food.row][food.col] = '*';
            for (Position p : snake.bodyHeadFirst()) g[p.row][p.col] = 'o';
            g[snake.head().row][snake.head().col] = 'H';
            StringBuilder sb = new StringBuilder();
            for (char[] row : g) sb.append("  ").append(row).append('\n');
            return sb.toString();
        }
    }

    // ---------- game loop: one writer thread ticks; any thread may call turn() ----------
    static final class GameLoop {
        private final Game game;
        private final ScheduledExecutorService exec = Executors.newSingleThreadScheduledExecutor();
        GameLoop(Game game) { this.game = game; }
        void start(long periodMs) {
            exec.scheduleAtFixedRate(() -> {
                if (game.tick() != Status.RUNNING) exec.shutdown();
            }, 0, periodMs, TimeUnit.MILLISECONDS);
        }
        void stop() throws InterruptedException {
            exec.shutdown();
            exec.awaitTermination(2, TimeUnit.SECONDS);
        }
    }

    // ---------- demo driver with simple checks ----------
    static int checks = 0;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }

    public static void main(String[] args) throws Exception {
        // 1. LeetCode 353 example: 3 wide, 2 high, food at (1,2) then (0,1).
        Game g1 = Game.classic(2, 3, new int[][]{{1, 2}, {0, 1}});
        Direction[] moves = {Direction.RIGHT, Direction.DOWN, Direction.RIGHT, Direction.UP, Direction.LEFT, Direction.UP};
        int[] expected = {0, 0, 1, 1, 2, -1};
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < moves.length; i++) {
            int got = g1.move(moves[i]);
            out.append(moves[i]).append("=").append(got).append(' ');
            check(got == expected[i], "LC 353 move " + i);
        }
        System.out.println("Game 1 (LeetCode 353): " + out.toString().trim());
        check(g1.cause() == Cause.HIT_WALL, "game 1 ends on the wall");

        // 2. Boundary strategy: the same move kills with walls and wraps on a torus.
        Game wall = new Game(new Board(4, 4), new Position(0, 0), Direction.LEFT, new WallBoundary(), new FixedFoodSpawner(List.of()), new FoodGrowth());
        Game wrap = new Game(new Board(4, 4), new Position(0, 0), Direction.LEFT, new WrapBoundary(), new FixedFoodSpawner(List.of()), new FoodGrowth());
        wall.tick(); wrap.tick();
        System.out.println("Game 2 (boundary): wall -> " + wall.status() + " " + wall.cause() + ", wrap -> head " + wrap.head());
        check(wall.cause() == Cause.HIT_WALL && wrap.head().equals(new Position(0, 3)), "wall vs wrap");

        // 3. Tail chase is legal: grow to 4 on a 3 x 3 board, then circle a 2 x 2 square for 8 ticks.
        //    Each tick the head enters the cell the tail is leaving.
        Game chase = new Game(new Board(3, 3), new Position(0, 0), Direction.RIGHT, new WallBoundary(),
                new FixedFoodSpawner(List.of(new Position(0, 1), new Position(1, 1), new Position(1, 0))), new FoodGrowth());
        chase.move(Direction.RIGHT); chase.move(Direction.DOWN); chase.move(Direction.LEFT);
        Direction[] circle = {Direction.UP, Direction.RIGHT, Direction.DOWN, Direction.LEFT};
        for (int i = 0; i < 8; i++) chase.move(circle[i % 4]);
        System.out.println("Game 3 (tail chase): status " + chase.status() + ", length " + chase.length() + ", head " + chase.head());
        check(chase.status() == Status.RUNNING && chase.length() == 4, "tail chase is not a collision");

        // 3b. Filling the board is a win: 1 x 3, two foods.
        Game fill = new Game(new Board(1, 3), new Position(0, 0), Direction.RIGHT, new WallBoundary(),
                new FixedFoodSpawner(List.of(new Position(0, 1), new Position(0, 2))), new FoodGrowth());
        fill.tick(); fill.tick();
        System.out.println("Game 3b (fill 1x3): " + fill.status() + " " + fill.cause() + ", score " + fill.score());
        check(fill.status() == Status.WON && fill.cause() == Cause.BOARD_FULL, "full board = WON");

        // 4. Self-collision: grow to 5, then turn into the body.
        Game self = new Game(new Board(5, 5), new Position(2, 0), Direction.RIGHT, new WallBoundary(),
                new FixedFoodSpawner(List.of(new Position(2, 1), new Position(2, 2), new Position(2, 3), new Position(2, 4))), new FoodGrowth());
        for (int i = 0; i < 4; i++) self.tick();                    // body (2,4)..(2,0), length 5
        self.turn(Direction.DOWN); self.tick();                     // head (3,4)
        self.turn(Direction.LEFT); self.tick();                     // head (3,3)
        self.turn(Direction.UP); self.tick();                       // (2,3) is body -> dead
        System.out.println("Game 4 (self): " + self.status() + " " + self.cause() + " at length " + self.length());
        check(self.cause() == Cause.HIT_SELF, "bites itself");

        // 5. Reverse turn is ignored when length > 1; a quick RIGHT->UP->LEFT burst is buffered, not a U-turn.
        Game rev = new Game(new Board(5, 5), new Position(2, 1), Direction.RIGHT, new WallBoundary(),
                new FixedFoodSpawner(List.of(new Position(2, 2))), new FoodGrowth());
        rev.tick();                                                  // eats, length 2, heading RIGHT
        boolean reverseAccepted = rev.turn(Direction.LEFT);
        boolean up = rev.turn(Direction.UP), left = rev.turn(Direction.LEFT);
        rev.tick(); rev.tick();
        System.out.println("Game 5 (input): reverse accepted=" + reverseAccepted + ", UP then LEFT buffered=" + (up && left)
                + ", head " + rev.head() + " status " + rev.status());
        check(!reverseAccepted && up && left && rev.head().equals(new Position(1, 1)) && rev.status() == Status.RUNNING, "input buffer");

        // 6. Random food + "grow every 3 ticks" + observer; print board, head and tail.
        Game g6 = new Game(new Board(6, 8), new Position(2, 2), Direction.RIGHT, new WrapBoundary(), new RandomFoodSpawner(7), new EveryNTicksGrowth(3));
        List<String> events = new ArrayList<>();
        g6.addListener(new GameListener() {
            @Override public void onFoodEaten(int score, Position at) { events.add("ate at " + at); }
        });
        Direction[] script = {Direction.DOWN, Direction.LEFT, Direction.UP};
        for (int t = 0; t < 12; t++) {
            if (t % 4 == 3) g6.turn(script[(t / 4) % script.length]);
            g6.tick();
            check(g6.invariantsHold(), "food never under the snake, tick " + t);
        }
        System.out.println("Game 6 (random food, grow every 3 ticks, wrap): length " + g6.length()
                + ", head " + g6.head() + ", tail " + g6.tail() + ", events " + events.size());
        System.out.print(g6.render());

        // 7. Concurrency: a loop thread ticks every 1 ms while 4 input threads hammer turn().
        Game g7 = new Game(new Board(20, 20), new Position(10, 10), Direction.RIGHT, new WrapBoundary(), new RandomFoodSpawner(1), new FoodGrowth());
        GameLoop gl = new GameLoop(g7);
        gl.start(1);
        ExecutorService inputs = Executors.newFixedThreadPool(4);
        AtomicInteger accepted = new AtomicInteger();
        for (int t = 0; t < 4; t++) {
            final int seed = t;
            inputs.submit(() -> {
                Random r = new Random(seed);
                for (int i = 0; i < 2000; i++) if (g7.turn(Direction.values()[r.nextInt(4)])) accepted.incrementAndGet();
            });
        }
        inputs.shutdown();
        inputs.awaitTermination(5, TimeUnit.SECONDS);
        Thread.sleep(30);
        gl.stop();
        check(g7.invariantsHold(), "invariants after concurrent input");
        System.out.println("Game 7 (concurrent input, 4 threads x 2000 turns): invariants hold");

        System.out.println("All " + checks + " checks passed");
    }
}
"""Snake game (Nokia style): deque body + set for O(1) moves, pluggable boundary / food / growth rules.

Run: python main.py        Tests: python -m unittest test_main
"""
from __future__ import annotations

import random
import threading
from collections import deque
from dataclasses import dataclass
from enum import Enum
from typing import Callable, Deque, List, Optional, Protocol, Set


class Direction(Enum):
    UP = (-1, 0)
    DOWN = (1, 0)
    LEFT = (0, -1)
    RIGHT = (0, 1)

    def is_opposite(self, other: "Direction") -> bool:
        return self.value[0] == -other.value[0] and self.value[1] == -other.value[1]


@dataclass(frozen=True)
class Position:          # frozen -> hashable, safe as a set member
    row: int
    col: int

    def __str__(self) -> str:
        return f"({self.row},{self.col})"


class Status(Enum):
    RUNNING = "RUNNING"
    WON = "WON"
    OVER = "OVER"


class Cause(Enum):
    NONE = "NONE"
    HIT_WALL = "HIT_WALL"
    HIT_SELF = "HIT_SELF"
    BOARD_FULL = "BOARD_FULL"


class Board:
    def __init__(self, rows: int, cols: int) -> None:
        if rows < 1 or cols < 1:
            raise ValueError("board must be at least 1 x 1")
        self.rows, self.cols = rows, cols

    def inside(self, r: int, c: int) -> bool:
        return 0 <= r < self.rows and 0 <= c < self.cols

    def cells(self) -> int:
        return self.rows * self.cols


class Snake:
    def __init__(self, start: Position) -> None:
        self._body: Deque[Position] = deque([start])   # head at index 0
        self._cells: Set[Position] = {start}
        self._pending = 0

    def head(self) -> Position:
        return self._body[0]

    def tail(self) -> Position:
        return self._body[-1]

    def length(self) -> int:
        return len(self._body)

    def occupies(self, p: Position) -> bool:
        return p in self._cells

    def grow(self, n: int) -> None:
        self._pending += n

    def body(self) -> List[Position]:
        return list(self._body)

    def would_bite_itself(self, nxt: Position) -> bool:
        if nxt not in self._cells:
            return False
        return not (nxt == self.tail() and self._pending == 0)   # the tail is leaving this tick

    def advance(self, nxt: Position) -> None:
        if self._pending > 0:
            self._pending -= 1
        else:
            self._cells.discard(self._body.pop())
        self._body.appendleft(nxt)
        self._cells.add(nxt)

    def consistent(self) -> bool:
        return len(self._body) == len(self._cells) and all(p in self._cells for p in self._body)


# ---------- strategies (Protocol = an interface by shape) ----------
class BoundaryPolicy(Protocol):
    def next(self, head: Position, d: Direction, b: Board) -> Optional[Position]: ...


class WallBoundary:
    def next(self, head: Position, d: Direction, b: Board) -> Optional[Position]:
        r, c = head.row + d.value[0], head.col + d.value[1]
        return Position(r, c) if b.inside(r, c) else None


class WrapBoundary:
    def next(self, head: Position, d: Direction, b: Board) -> Optional[Position]:
        return Position((head.row + d.value[0]) % b.rows, (head.col + d.value[1]) % b.cols)


class FoodSpawner(Protocol):
    def next(self, b: Board, s: Snake) -> Optional[Position]: ...


class FixedFoodSpawner:
    def __init__(self, foods: List[Position]) -> None:
        self._queue = deque(foods)

    def next(self, b: Board, s: Snake) -> Optional[Position]:
        while self._queue:
            p = self._queue.popleft()
            if not s.occupies(p):
                return p
        return None


class RandomFoodSpawner:
    def __init__(self, seed: int) -> None:
        self._rnd = random.Random(seed)

    def next(self, b: Board, s: Snake) -> Optional[Position]:
        if s.length() >= b.cells():
            return None
        for _ in range(8):
            p = Position(self._rnd.randrange(b.rows), self._rnd.randrange(b.cols))
            if not s.occupies(p):
                return p
        free = [Position(r, c) for r in range(b.rows) for c in range(b.cols) if not s.occupies(Position(r, c))]
        return self._rnd.choice(free)


GrowthRule = Callable[[bool, int], int]          # (ate_food, tick) -> cells to add


def food_growth(ate: bool, tick: int) -> int:
    return 1 if ate else 0


def every_n_ticks(n: int) -> GrowthRule:
    return lambda ate, tick: (1 if ate else 0) + (1 if tick % n == 0 else 0)


class GameListener:
    def on_food_eaten(self, score: int, at: Position) -> None: ...

    def on_game_end(self, status: Status, cause: Cause, score: int) -> None: ...


class Game:
    MAX_BUFFERED_TURNS = 2

    def __init__(self, board: Board, start: Position, direction: Direction,
                 boundary: BoundaryPolicy, spawner: FoodSpawner, growth: GrowthRule = food_growth) -> None:
        if not board.inside(start.row, start.col):
            raise ValueError(f"start {start} is off the board")
        self._board, self._snake, self._dir = board, Snake(start), direction
        self._boundary, self._spawner, self._growth = boundary, spawner, growth
        self._listeners: List[GameListener] = []
        self._turns: Deque[Direction] = deque()
        self._lock = threading.RLock()
        self._status, self._cause = Status.RUNNING, Cause.NONE
        self._score, self._tick = 0, 0
        self._food = spawner.next(board, self._snake)

    @staticmethod
    def classic(height: int, width: int, food: List[List[int]]) -> "Game":
        """LeetCode 353: start (0,0) moving right, walls, fixed food list."""
        return Game(Board(height, width), Position(0, 0), Direction.RIGHT, WallBoundary(),
                    FixedFoodSpawner([Position(r, c) for r, c in food]))

    def add_listener(self, l: GameListener) -> None:
        self._listeners.append(l)

    def turn(self, d: Direction) -> bool:
        with self._lock:
            last = self._turns[-1] if self._turns else self._dir
            if self._status != Status.RUNNING or d == last or len(self._turns) >= self.MAX_BUFFERED_TURNS:
                return False
            if d.is_opposite(last) and self._snake.length() > 1:
                return False
            self._turns.append(d)
            return True

    def tick(self) -> Status:
        with self._lock:
            if self._status != Status.RUNNING:
                return self._status
            self._tick += 1
            if self._turns:
                self._dir = self._turns.popleft()
            nxt = self._boundary.next(self._snake.head(), self._dir, self._board)
            if nxt is None:
                return self._end(Status.OVER, Cause.HIT_WALL)
            ate = nxt == self._food
            self._snake.grow(self._growth(ate, self._tick))
            if self._snake.would_bite_itself(nxt):
                return self._end(Status.OVER, Cause.HIT_SELF)
            self._snake.advance(nxt)
            if ate:
                self._score += 1
                for l in self._listeners:
                    l.on_food_eaten(self._score, nxt)
                self._food = self._spawner.next(self._board, self._snake)
            if self._snake.length() == self._board.cells():
                return self._end(Status.WON, Cause.BOARD_FULL)
            return self._status

    def move(self, d: Direction) -> int:
        """LeetCode 353: returns the score, or -1 once the game is over."""
        with self._lock:
            self.turn(d)
            return -1 if self.tick() == Status.OVER else self._score

    def _end(self, status: Status, cause: Cause) -> Status:
        self._status, self._cause = status, cause
        for l in self._listeners:
            l.on_game_end(status, cause, self._score)
        return status

    # read-only views
    status = property(lambda self: self._status)
    cause = property(lambda self: self._cause)
    score = property(lambda self: self._score)
    head = property(lambda self: self._snake.head())
    tail = property(lambda self: self._snake.tail())
    length = property(lambda self: self._snake.length())
    food = property(lambda self: self._food)

    def invariants_hold(self) -> bool:
        with self._lock:
            return self._snake.consistent() and (self._food is None or not self._snake.occupies(self._food))

    def render(self) -> str:
        with self._lock:
            g = [["."] * self._board.cols for _ in range(self._board.rows)]
            if self._food:
                g[self._food.row][self._food.col] = "*"
            for p in self._snake.body():
                g[p.row][p.col] = "o"
            g[self.head.row][self.head.col] = "H"
            return "\n".join("  " + "".join(r) for r in g)

// Snake game in C++14: deque body + unordered_set for O(1) moves; boundary / food / growth are strategies.
// Build: g++ -std=c++14 -Wall -Wextra main.cpp -o main && ./main
// Single-threaded demo: MinGW g++ 6.3 (win32 thread model) has no std::mutex. In a threaded build add
// `std::mutex mu_;` to Game and lock it in turn() and tick() only (move() calls both: lock once, call unlocked helpers).
#include <deque>
#include <functional>

enum class Direction { UP, DOWN, LEFT, RIGHT };
static const int DR[] = {-1, 1, 0, 0};
static const int DC[] = {0, 0, -1, 1};
static bool opposite(Direction a, Direction b) {
    int i = static_cast<int>(a), j = static_cast<int>(b);
    return DR[i] == -DR[j] && DC[i] == -DC[j];
}

struct Position {
    int row, col;
    bool operator==(const Position& o) const { return row == o.row && col == o.col; }
};
struct PositionHash {
    size_t operator()(const Position& p) const { return std::hash<long long>()((long long)p.row << 32 ^ (unsigned)p.col); }
};
static std::string str(const Position& p) { return "(" + std::to_string(p.row) + "," + std::to_string(p.col) + ")"; }

enum class Status { RUNNING, WON, OVER };
enum class Cause { NONE, HIT_WALL, HIT_SELF, BOARD_FULL };

struct Board {
    int rows, cols;
    Board(int r, int c) : rows(r), cols(c) { if (r < 1 || c < 1) throw std::invalid_argument("board must be at least 1 x 1"); }
    bool inside(int r, int c) const { return r >= 0 && r < rows && c >= 0 && c < cols; }
    int cells() const { return rows * cols; }
};

class Snake {
    std::deque<Position> body_;                       // head at front
    std::unordered_set<Position, PositionHash> cells_;
    int pending_ = 0;
public:
    explicit Snake(Position start) { body_.push_front(start); cells_.insert(start); }
    Position head() const { return body_.front(); }
    Position tail() const { return body_.back(); }
    int length() const { return static_cast<int>(body_.size()); }
    bool occupies(const Position& p) const { return cells_.count(p) != 0; }
    void grow(int n) { pending_ += n; }
    const std::deque<Position>& body() const { return body_; }
    bool wouldBiteItself(const Position& next) const {
        if (!occupies(next)) return false;
        return !(next == tail() && pending_ == 0);    // the tail leaves this tick unless we are growing
    }
    void advance(const Position& next) {
        if (pending_ > 0) --pending_;
        else { cells_.erase(body_.back()); body_.pop_back(); }
        body_.push_front(next);
        cells_.insert(next);
    }
    bool consistent() const { return body_.size() == cells_.size(); }
};

// ---- strategies: abstract classes are the C++ form of a C ops table ----
struct BoundaryPolicy {
    virtual ~BoundaryPolicy() {}
    virtual bool next(const Position& h, Direction d, const Board& b, Position& out) const = 0;   // false = wall
};
struct WallBoundary : BoundaryPolicy {
    bool next(const Position& h, Direction d, const Board& b, Position& out) const override {
        int i = static_cast<int>(d);
        out = Position{h.row + DR[i], h.col + DC[i]};
        return b.inside(out.row, out.col);
    }
};
struct WrapBoundary : BoundaryPolicy {
    bool next(const Position& h, Direction d, const Board& b, Position& out) const override {
        int i = static_cast<int>(d);
        out = Position{((h.row + DR[i]) % b.rows + b.rows) % b.rows, ((h.col + DC[i]) % b.cols + b.cols) % b.cols};
        return true;
    }
};

struct FoodSpawner {
    virtual ~FoodSpawner() {}
    virtual bool next(const Board& b, const Snake& s, Position& out) = 0;   // false = no food
};
struct FixedFoodSpawner : FoodSpawner {
    std::deque<Position> q;
    explicit FixedFoodSpawner(std::vector<Position> foods) : q(foods.begin(), foods.end()) {}
    bool next(const Board&, const Snake& s, Position& out) override {
        while (!q.empty()) {
            Position p = q.front(); q.pop_front();
            if (!s.occupies(p)) { out = p; return true; }
        }
        return false;
    }
};
struct RandomFoodSpawner : FoodSpawner {
    std::mt19937 rng;
    explicit RandomFoodSpawner(unsigned seed) : rng(seed) {}
    bool next(const Board& b, const Snake& s, Position& out) override {
        if (s.length() >= b.cells()) return false;
        std::uniform_int_distribution<int> R(0, b.rows - 1), C(0, b.cols - 1);
        for (int i = 0; i < 8; ++i) {
            Position p{R(rng), C(rng)};
            if (!s.occupies(p)) { out = p; return true; }
        }
        std::vector<Position> freeCells;
        for (int r = 0; r < b.rows; ++r)
            for (int c = 0; c < b.cols; ++c)
                if (!s.occupies(Position{r, c})) freeCells.push_back(Position{r, c});
        std::uniform_int_distribution<size_t> pick(0, freeCells.size() - 1);
        out = freeCells[pick(rng)];
        return true;
    }
};

using GrowthRule = std::function<int(bool ate, long tick)>;
static int foodGrowth(bool ate, long) { return ate ? 1 : 0; }
static GrowthRule everyNTicks(int n) { return [n](bool ate, long t) { return (ate ? 1 : 0) + (t % n == 0 ? 1 : 0); }; }

class Game {
    static const size_t MAX_BUFFERED_TURNS = 2;
    Board board_;
    Snake snake_;
    std::unique_ptr<BoundaryPolicy> boundary_;
    std::unique_ptr<FoodSpawner> spawner_;
    GrowthRule growth_;
    std::deque<Direction> turns_;
    Direction dir_;
    bool hasFood_ = false;
    Position food_{0, 0};
    Status status_ = Status::RUNNING;
    Cause cause_ = Cause::NONE;
    int score_ = 0;
    long tick_ = 0;

    Status end(Status s, Cause c) { status_ = s; cause_ = c; return s; }
public:
    std::function<void(int, Position)> onFoodEaten;   // a one-listener Observer

    Game(Board b, Position start, Direction d, std::unique_ptr<BoundaryPolicy> bp,
         std::unique_ptr<FoodSpawner> fs, GrowthRule g = foodGrowth)
        : board_(b), snake_(start), boundary_(std::move(bp)), spawner_(std::move(fs)), growth_(g), dir_(d) {
        if (!b.inside(start.row, start.col)) throw std::invalid_argument("start is off the board");
        hasFood_ = spawner_->next(board_, snake_, food_);
    }

    bool turn(Direction d) {                    // input side (threaded build: lock mu_)
        Direction last = turns_.empty() ? dir_ : turns_.back();
        if (status_ != Status::RUNNING || d == last || turns_.size() >= MAX_BUFFERED_TURNS) return false;
        if (opposite(d, last) && snake_.length() > 1) return false;
        turns_.push_back(d);
        return true;
    }

    Status tick() {                             // game-loop side (threaded build: lock mu_)
        if (status_ != Status::RUNNING) return status_;
        ++tick_;
        if (!turns_.empty()) { dir_ = turns_.front(); turns_.pop_front(); }
        Position next{0, 0};
        if (!boundary_->next(snake_.head(), dir_, board_, next)) return end(Status::OVER, Cause::HIT_WALL);
        bool ate = hasFood_ && next == food_;
        snake_.grow(growth_(ate, tick_));
        if (snake_.wouldBiteItself(next)) return end(Status::OVER, Cause::HIT_SELF);
        snake_.advance(next);
        if (ate) {
            ++score_;
            if (onFoodEaten) onFoodEaten(score_, next);
            hasFood_ = spawner_->next(board_, snake_, food_);
        }
        if (snake_.length() == board_.cells()) return end(Status::WON, Cause::BOARD_FULL);
        return status_;
    }

    int move(Direction d) {   // LeetCode 353 API
        turn(d);
        return tick() == Status::OVER ? -1 : score_;
    }

    Status status() const { return status_; }
    Cause cause() const { return cause_; }
    int score() const { return score_; }
    int length() const { return snake_.length(); }
    Position head() const { return snake_.head(); }
    Position tail() const { return snake_.tail(); }
    bool invariantsHold() const {
        return snake_.consistent() && !(hasFood_ && snake_.occupies(food_));
    }
};

static int checks = 0;
static void check(bool ok, const char* what) {
    if (!ok) { std::cerr << "FAILED: " << what << "\n"; std::exit(1); }
    ++checks;
}
static std::unique_ptr<FoodSpawner> fixed(std::vector<Position> v) { return std::unique_ptr<FoodSpawner>(new FixedFoodSpawner(v)); }
static std::unique_ptr<BoundaryPolicy> walls() { return std::unique_ptr<BoundaryPolicy>(new WallBoundary()); }
static std::unique_ptr<BoundaryPolicy> wrap() { return std::unique_ptr<BoundaryPolicy>(new WrapBoundary()); }

int main() {
    // 1. LeetCode 353 example
    Game g1(Board(2, 3), Position{0, 0}, Direction::RIGHT, walls(), fixed({{1, 2}, {0, 1}}));
    Direction moves[] = {Direction::RIGHT, Direction::DOWN, Direction::RIGHT, Direction::UP, Direction::LEFT, Direction::UP};
    int expected[] = {0, 0, 1, 1, 2, -1};
    std::ostringstream out;
    for (int i = 0; i < 6; ++i) { int got = g1.move(moves[i]); out << got << ' '; check(got == expected[i], "LC 353"); }
    std::cout << "Game 1 (LeetCode 353): " << out.str() << "\n";

    // 2. walls vs wrap
    Game wl(Board(4, 4), Position{0, 0}, Direction::LEFT, walls(), fixed({}));
    Game wr(Board(4, 4), Position{0, 0}, Direction::LEFT, wrap(), fixed({}));
    wl.tick(); wr.tick();
    check(wl.cause() == Cause::HIT_WALL && wr.head() == Position{0, 3}, "wall vs wrap");
    std::cout << "Game 2 (boundary): wrap head " << str(wr.head()) << "\n";

    // 3. tail chase is legal; growing into the tail is not
    Game chase(Board(3, 3), Position{0, 0}, Direction::RIGHT, walls(), fixed({{0, 1}, {1, 1}, {1, 0}}));
    chase.move(Direction::RIGHT); chase.move(Direction::DOWN); chase.move(Direction::LEFT);
    Direction circle[] = {Direction::UP, Direction::RIGHT, Direction::DOWN, Direction::LEFT};
    for (int i = 0; i < 8; ++i) chase.move(circle[i % 4]);
    check(chase.status() == Status::RUNNING && chase.length() == 4, "tail chase");
    Game grow(Board(3, 3), Position{0, 0}, Direction::RIGHT, walls(), fixed({{0, 1}, {1, 1}, {1, 0}}), everyNTicks(4));
    grow.move(Direction::RIGHT); grow.move(Direction::DOWN); grow.move(Direction::LEFT); grow.move(Direction::UP);
    check(grow.cause() == Cause::HIT_SELF, "growing into the tail");
    std::cout << "Game 3 (tail): chase RUNNING, grow-into-tail HIT_SELF\n";

    // 4. fill the board
    Game fill(Board(1, 3), Position{0, 0}, Direction::RIGHT, walls(), fixed({{0, 1}, {0, 2}}));
    fill.tick(); fill.tick();
    check(fill.status() == Status::WON, "fill = WON");

    // 5. reverse ignored, burst buffered
    Game rev(Board(5, 5), Position{2, 1}, Direction::RIGHT, walls(), fixed({{2, 2}}));
    rev.tick();
    check(!rev.turn(Direction::LEFT) && rev.turn(Direction::UP) && rev.turn(Direction::LEFT), "input buffer");
    rev.tick(); rev.tick();
    check(rev.head() == Position{1, 1}, "buffered turns applied in order");

    // 6. random food with observer
    Game g6(Board(4, 4), Position{0, 0}, Direction::RIGHT, wrap(), std::unique_ptr<FoodSpawner>(new RandomFoodSpawner(3)));
    int eaten = 0;
    g6.onFoodEaten = [&eaten](int, Position) { ++eaten; };
    std::mt19937 r(5);
    for (int i = 0; i < 300 && g6.status() == Status::RUNNING; ++i) {
        g6.turn(static_cast<Direction>(r() % 4));
        g6.tick();
        check(g6.invariantsHold(), "random food invariants");
    }
    check(eaten == g6.score(), "observer saw every food");
    std::cout << "Game 6 (random food 4x4): length " << g6.length() << ", head " << str(g6.head())
              << ", tail " << str(g6.tail()) << "\n";

    std::cout << "All " << checks << " checks passed\n";
    return 0;
}
