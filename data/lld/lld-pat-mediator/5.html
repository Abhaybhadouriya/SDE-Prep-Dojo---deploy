Code: auction house, control tower, chat room and a bus

Each tab is a complete program that was compiled and run. Java is the auction above, with targeted and broadcast messages and rule checks. Python is an air-traffic-control tower that queues planes for one runway. C++14 is a chat room with direct messages and muting, plus a tiny bus that pairs devices with drivers the way the Linux driver core does. Files live in lld-code/lld-pat-mediator/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Java interface: a struct of function pointers with no data. A class that implements it must fill every slot.
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • std::unique_ptr<T>: an owning pointer that frees the object when it goes out of scope (no manual free); std::move hands that ownership to someone else.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

public class MediatorDemo {

    // ---- Mediator interface: the only thing a bidder knows about ----
    interface AuctionMediator {
        void join(Bidder b);
        void placeBid(Bidder from, long amount);
    }

    // ---- Colleague: talks to the mediator, never to other bidders ----
    static class Bidder {
        final String name;
        private final AuctionMediator house;
        Bidder(String name, AuctionMediator house) {
            this.name = name;
            this.house = house;
            house.join(this);
        }
        void bid(long amount) { house.placeBid(this, amount); }     // outgoing: always via mediator

        // incoming: the mediator calls these
        void rejected(String why)          { log("rejected: " + why); }
        void outbid(String by, long price) { log("outbid by " + by + " at " + price); }
        void priceChanged(long price)      { log("new price " + price); }
        void won(long price)               { log("WON at " + price); }
        void lost(String winner)           { log("lost to " + winner); }
        private void log(String m)         { System.out.println("  " + name + ": " + m); }
    }

    // ---- Concrete mediator: owns the rules and the who-talks-to-whom logic ----
    static final class AuctionHouse implements AuctionMediator {
        private final List<Bidder> bidders = new ArrayList<>();
        private final long increment;
        private long price;
        private Bidder leader;
        private boolean open = true;

        AuctionHouse(long startPrice, long increment) { this.price = startPrice; this.increment = increment; }

        @Override public void join(Bidder b) { bidders.add(b); }

        @Override public void placeBid(Bidder from, long amount) {
            System.out.println(from.name + " bids " + amount);
            if (!open)                   { from.rejected("auction closed"); return; }
            if (from == leader)          { from.rejected("you already lead"); return; }
            long min = leader == null ? price : price + increment;
            if (amount < min)            { from.rejected("minimum is " + min); return; }

            Bidder previous = leader;
            leader = from;
            price = amount;
            if (previous != null) previous.outbid(from.name, amount);   // targeted message
            for (Bidder b : bidders)                                     // broadcast to the rest
                if (b != from && b != previous) b.priceChanged(amount);
        }

        void close() {
            open = false;
            System.out.println("-- hammer down --");
            if (leader == null) { System.out.println("  no sale"); return; }
            for (Bidder b : bidders) {
                if (b == leader) b.won(price); else b.lost(leader.name);
            }
        }
    }

    public static void main(String[] args) {
        AuctionHouse house = new AuctionHouse(1_000, 100);
        Bidder asha = new Bidder("Asha", house);
        Bidder ravi = new Bidder("Ravi", house);
        Bidder meera = new Bidder("Meera", house);

        asha.bid(1_000);
        ravi.bid(1_050);       // below price + increment
        ravi.bid(1_100);
        meera.bid(1_300);
        meera.bid(1_400);      // already leading
        house.close();
        asha.bid(2_000);       // too late
    }
}

Bidder has no field of type Bidder. Every rule ("minimum is 1100", "you already lead", "auction closed") is in one class, so a different auction format is a different mediator, not an edit to every bidder. Registering in the constructor (house.join(this)) keeps the demo short; in production prefer registering after construction so this does not escape a half-built object.

"""Mediator in Python: an air-traffic-control tower coordinating planes and one runway."""
from collections import deque


class Tower:
    """Concrete mediator. Planes never talk to each other; the tower owns the runway rule."""

    def __init__(self, name: str):
        self.name = name
        self._runway_busy_by = None
        self._queue = deque()

    def request_landing(self, plane: "Plane") -> None:
        if self._runway_busy_by is None:
            self._runway_busy_by = plane
            plane.cleared_to_land()
        else:
            self._queue.append(plane)
            plane.hold(position=len(self._queue))

    def runway_vacated(self, plane: "Plane") -> None:
        assert plane is self._runway_busy_by, "only the plane on the runway can vacate it"
        print(f"[{self.name}] runway free after {plane.callsign}")
        self._runway_busy_by = None
        if self._queue:                                # hand the runway to the next plane
            nxt = self._queue.popleft()
            self._runway_busy_by = nxt
            nxt.cleared_to_land()
            for i, p in enumerate(self._queue, start=1):
                p.hold(position=i)                     # everyone else moves up


class Plane:
    """Colleague: knows only its tower."""

    def __init__(self, callsign: str, tower: Tower):
        self.callsign, self._tower = callsign, tower

    def approach(self):
        print(f"{self.callsign}: requesting landing")
        self._tower.request_landing(self)

    def touchdown_and_exit(self):
        self._tower.runway_vacated(self)

    # messages from the mediator
    def cleared_to_land(self):
        print(f"  {self.callsign}: cleared to land")

    def hold(self, position: int):
        print(f"  {self.callsign}: holding, number {position}")


if __name__ == "__main__":
    tower = Tower("BLR")
    ai, ix, uk = Plane("AI-501", tower), Plane("6E-212", tower), Plane("UK-818", tower)
    ai.approach()
    ix.approach()
    uk.approach()
    ai.touchdown_and_exit()
    ix.touchdown_and_exit()
    uk.touchdown_and_exit()

The runway is a shared resource and the tower is its only owner. Planes cannot grab it, so two planes can never be cleared at once. That is the same reason a mediator is a natural place to put a lock when colleagues run on different threads (one lock around one invariant; see locks and conditions, and the deadlock page for why colleagues must not be called while that lock is held).

// Mediator in C++14: a chat room, then a tiny "bus" that pairs devices with drivers like the Linux driver core.
// Build: g++ -std=c++14 -Wall mediator.cpp -o mediator
#include <iostream>
#include <map>
#include <set>
#include <string>
#include <vector>

// ---------------- Chat room ----------------
class User;

class ChatMediator {                       // Mediator interface
public:
    virtual ~ChatMediator() = default;
    virtual void join(User* u) = 0;
    virtual void send(User* from, const std::string& to, const std::string& text) = 0;
};

class User {                               // Colleague
public:
    User(std::string name, ChatMediator& room) : name_(std::move(name)), room_(room) { room_.join(this); }
    const std::string& name() const { return name_; }
    void say(const std::string& text) { room_.send(this, "", text); }            // to everyone
    void dm(const std::string& to, const std::string& text) { room_.send(this, to, text); }
    void receive(const std::string& from, const std::string& text) {
        std::cout << "  [" << name_ << "] " << from << ": " << text << "\n";
    }
private:
    std::string name_;
    ChatMediator& room_;
};

class ChatRoom : public ChatMediator {     // Concrete mediator: routing + rules live here only
public:
    void join(User* u) override { users_[u->name()] = u; }
    void mute(const std::string& name) { muted_.insert(name); }
    void send(User* from, const std::string& to, const std::string& text) override {
        if (muted_.count(from->name())) { std::cout << "  (" << from->name() << " is muted)\n"; return; }
        if (!to.empty()) {                                   // direct message
            auto it = users_.find(to);
            if (it == users_.end()) { from->receive("room", "no such user " + to); return; }
            it->second->receive(from->name() + " (dm)", text);
            return;
        }
        for (auto& kv : users_)                              // broadcast
            if (kv.second != from) kv.second->receive(from->name(), text);
    }
private:
    std::map<std::string, User*> users_;
    std::set<std::string> muted_;
};

// ---------------- Driver-core style bus ----------------
struct Device { std::string compatible; std::string name; };
struct Driver {
    std::string compatible;
    std::string name;
    void (*probe)(const Device&);           // C-style callback, like struct device_driver.probe
};

class Bus {                                 // devices and drivers never look each other up
public:
    void add_device(const Device& d) { devices_.push_back(d); for (auto& drv : drivers_) try_bind(d, drv); }
    void add_driver(const Driver& drv) { drivers_.push_back(drv); for (auto& d : devices_) try_bind(d, drv); }
private:
    static bool match(const Device& d, const Driver& drv) { return d.compatible == drv.compatible; }
    void try_bind(const Device& d, const Driver& drv) {
        if (match(d, drv) && !bound_.count(d.name)) { bound_.insert(d.name); drv.probe(d); }
    }
    std::vector<Device> devices_;
    std::vector<Driver> drivers_;
    std::set<std::string> bound_;
};

static void nic_probe(const Device& d) { std::cout << "  e1000 probe -> " << d.name << "\n"; }
static void uart_probe(const Device& d) { std::cout << "  8250 probe  -> " << d.name << "\n"; }

int main() {
    ChatRoom room;
    User asha("asha", room), ravi("ravi", room), meera("meera", room);
    std::cout << "chat:\n";
    asha.say("standup in 5?");
    ravi.dm("asha", "joining late");
    room.mute("meera");
    meera.say("spam spam");
    ravi.dm("bob", "hello?");

    std::cout << "bus:\n";
    Bus pci;
    pci.add_device({"intel,e1000", "eth0"});      // no driver yet: waits
    pci.add_driver({"ns16550", "8250", uart_probe});
    pci.add_driver({"intel,e1000", "e1000", nic_probe});   // bus pairs it with eth0
    pci.add_device({"ns16550", "ttyS0"});          // driver already present: bound at once
    return 0;
}

The Bus keeps both lists and runs match() whichever side arrives second, which is how device_add() and driver_register() both end up probing. The driver is a struct with a function pointer, so the C idiom and the pattern line up exactly.
