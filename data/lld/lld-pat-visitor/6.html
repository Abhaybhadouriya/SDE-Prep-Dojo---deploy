Code: cart (Java), AST (Python), file tree (C++14)

All programs were compiled and run; the output is copied from the real runs. Java shows a cart with three visitors, the overload trap, and the Java 21 sealed-switch alternative. Python visits an expression tree and then uses the standard library's ast.NodeVisitor on real Python source. C++14 computes a du-style size over a directory tree. Files live in lld-code/lld-pat-visitor/.

Syntax you will meet in this code, mapped to C. Skim it now and come back when a line puzzles you:

  • Generics Foo<T>: a type parameter chosen by the user of the class. Think of a macro that generates one typed struct per element type, but checked by the compiler.
  • Void: Java's "returns nothing" type for generics (a generic method must return some object type, so a Void visitor returns null).
  • sealed interface ... permits A, B: the interface lists every allowed implementation, so the compiler knows the set is closed (like a tagged union).
  • record Foo(int a, String b) (Java): an immutable struct with a constructor, getters (a(), b()), equals and toString generated for you.
  • Anonymous class new Foo() { ... }: defines and creates a one-off implementation in place, like a static ops table filled inline.
  • @Override: asks the compiler to check that this method really fills a slot from the parent or interface; a typo becomes a compile error instead of a new, never-called function.
  • Python @dataclass: generates __init__, __eq__ and __repr__ from the field list, like a record.
  • virtual void f() = 0; (C++): a pure virtual function: a NULL slot in the ops table that every subclass must fill. override marks the function that fills it.
  • std::unique_ptr<T>: an owning pointer that frees the object when it goes out of scope (no manual free); std::move hands that ownership to someone else.

The longer walk-through is OOP from C; records, lambdas, generics and the rest also have short entries in the OOP toolkit.

import java.util.*;

public class VisitorDemo {

    // ---- Visitor: one visit method per concrete element type ----
    interface CartVisitor<R> {
        R visitBook(Book b);
        R visitElectronics(Electronics e);
        R visitGrocery(Grocery g);
    }

    // ---- Element: the only thing it knows is how to accept a visitor ----
    interface Item {
        <R> R accept(CartVisitor<R> v);
    }

    // ---- Concrete elements: stable data classes, no tax or shipping code inside ----
    record Book(String title, double price) implements Item {
        public <R> R accept(CartVisitor<R> v) { return v.visitBook(this); }        // dispatch #2
    }
    record Electronics(String name, double price, double weightKg) implements Item {
        public <R> R accept(CartVisitor<R> v) { return v.visitElectronics(this); }
    }
    record Grocery(String name, double price, boolean perishable) implements Item {
        public <R> R accept(CartVisitor<R> v) { return v.visitGrocery(this); }
    }

    // ---- Concrete visitor 1: GST-style tax, rate depends on the item type ----
    static final class TaxVisitor implements CartVisitor<Double> {
        public Double visitBook(Book b)               { return 0.0; }                 // exempt
        public Double visitElectronics(Electronics e) { return e.price() * 0.18; }
        public Double visitGrocery(Grocery g)         { return g.perishable() ? 0.0 : g.price() * 0.05; }
    }

    // ---- Concrete visitor 2: shipping cost, added later WITHOUT editing any Item class ----
    static final class ShippingVisitor implements CartVisitor<Double> {
        public Double visitBook(Book b)               { return 20.0; }
        public Double visitElectronics(Electronics e) { return 50.0 + 10.0 * e.weightKg(); }
        public Double visitGrocery(Grocery g)         { return g.perishable() ? 40.0 : 15.0; }
    }

    // ---- Concrete visitor 3: a stateful visitor that builds a receipt line by line ----
    static final class ReceiptVisitor implements CartVisitor<Void> {
        private final StringBuilder sb = new StringBuilder();
        public Void visitBook(Book b)               { line("Book", b.title(), b.price()); return null; }
        public Void visitElectronics(Electronics e) { line("Elec", e.name(), e.price()); return null; }
        public Void visitGrocery(Grocery g)         { line(g.perishable() ? "Frsh" : "Groc", g.name(), g.price()); return null; }
        private void line(String tag, String name, double p) { sb.append(String.format("  [%s] %-22s %9.2f%n", tag, name, p)); }
        String text() { return sb.toString(); }
    }

    // ---- Object structure: knows how to walk itself, not what is computed ----
    static final class Cart {
        private final List<Item> items = new ArrayList<>();
        Cart add(Item i) { items.add(i); return this; }
        double sum(CartVisitor<Double> v) {
            double total = 0;
            for (Item i : items) total += i.accept(v);                                // dispatch #1
            return total;
        }
        void walk(CartVisitor<?> v) { for (Item i : items) i.accept(v); }
    }

    // ---- The Java 21 alternative: sealed types + pattern-matching switch ----
    sealed interface Shape permits Circle, Rect { }
    record Circle(double r) implements Shape { }
    record Rect(double w, double h) implements Shape { }
    static double area(Shape s) {
        return switch (s) {                       // compiler checks the switch is exhaustive
            case Circle c -> Math.PI * c.r() * c.r();
            case Rect r   -> r.w() * r.h();
        };
    }

    public static void main(String[] args) {
        Cart cart = new Cart()
            .add(new Book("Clean Code", 450))
            .add(new Electronics("Headphones", 2000, 0.4))
            .add(new Grocery("Milk", 60, true))
            .add(new Grocery("Rice 5kg", 400, false));

        ReceiptVisitor receipt = new ReceiptVisitor();
        cart.walk(receipt);
        System.out.print(receipt.text());
        double subtotal = cart.sum(new CartVisitor<Double>() {        // even "price" is just a visitor
            public Double visitBook(Book b) { return b.price(); }
            public Double visitElectronics(Electronics e) { return e.price(); }
            public Double visitGrocery(Grocery g) { return g.price(); }
        });
        double tax = cart.sum(new TaxVisitor());
        double ship = cart.sum(new ShippingVisitor());
        System.out.printf("Subtotal %.2f | Tax %.2f | Shipping %.2f | Total %.2f%n",
                subtotal, tax, ship, subtotal + tax + ship);

        // Why accept() is needed: overloads are chosen by the STATIC type at compile time.
        Item unknown = new Book("SICP", 900);
        System.out.println("static type Item, overload picked: " + describe(unknown));
        System.out.println("via accept (double dispatch): tax = " + unknown.accept(new TaxVisitor()));

        System.out.printf("Sealed switch alternative: circle %.2f, rect %.2f%n", area(new Circle(1)), area(new Rect(2, 3)));
    }

    static String describe(Item i) { return "describe(Item)"; }
    static String describe(Book b) { return "describe(Book)"; }
}

"""Visitor over a tiny expression AST: one tree, two operations (evaluate, print)."""
import ast
from dataclasses import dataclass


@dataclass
class Num:
    value: float

    def accept(self, v):
        return v.visit_num(self)


@dataclass
class Add:
    left: object
    right: object

    def accept(self, v):
        return v.visit_add(self)


@dataclass
class Mul:
    left: object
    right: object

    def accept(self, v):
        return v.visit_mul(self)


class Evaluator:                      # operation 1
    def visit_num(self, n):
        return n.value

    def visit_add(self, n):
        return n.left.accept(self) + n.right.accept(self)

    def visit_mul(self, n):
        return n.left.accept(self) * n.right.accept(self)


class Printer:                        # operation 2, added without touching Num/Add/Mul
    def visit_num(self, n):
        return str(n.value)

    def visit_add(self, n):
        return f"({n.left.accept(self)} + {n.right.accept(self)})"

    def visit_mul(self, n):
        return f"{n.left.accept(self)} * {n.right.accept(self)}"


# The standard library does the same thing for real Python code.
class CallCounter(ast.NodeVisitor):
    """ast.NodeVisitor dispatches on the node class name: visit_Call, visit_Name, ..."""

    def __init__(self):
        self.calls = {}

    def visit_Call(self, node):
        if isinstance(node.func, ast.Name):
            self.calls[node.func.id] = self.calls.get(node.func.id, 0) + 1
        self.generic_visit(node)      # keep walking into the arguments


if __name__ == "__main__":
    tree = Add(Num(2), Mul(Num(3), Num(4)))          # 2 + 3 * 4
    print("print :", tree.accept(Printer()))
    print("eval  :", tree.accept(Evaluator()))

    source = "x = max(len(a), len(b))\nprint(max(x, 3))"
    counter = CallCounter()
    counter.visit(ast.parse(source))
    print("calls :", dict(sorted(counter.calls.items())))
$ python visitor.py

eval  : 14
calls : {'len': 2, 'max': 2, 'print': 1}

Python has no overloading, so the method name itself carries the type (visit_num, visit_add). ast.NodeVisitor.visit() builds that name at runtime as "visit_" + type(node).__name__ and falls back to generic_visit, so you only write the cases you care about.

// Visitor in C++14: a file-system tree with two operations (du-style size, and a listing).
// Build: g++ -std=c++14 -Wall visitor.cpp -o visitor
#include <iostream>
#include <memory>
#include <string>
#include <vector>

class File;
class Directory;
class Symlink;

// Visitor: one overload per concrete node type.
class FsVisitor {
public:
    virtual ~FsVisitor() = default;
    virtual void visit(const File& f) = 0;
    virtual void visit(const Directory& d) = 0;
    virtual void visit(const Symlink& s) = 0;
};

// Element
class Node {
public:
    explicit Node(std::string n) : name_(std::move(n)) {}
    virtual ~Node() = default;
    virtual void accept(FsVisitor& v) const = 0;   // first dispatch: virtual on the node
    const std::string& name() const { return name_; }
private:
    std::string name_;
};

class File : public Node {
public:
    File(std::string n, long bytes) : Node(std::move(n)), bytes_(bytes) {}
    long bytes() const { return bytes_; }
    void accept(FsVisitor& v) const override { v.visit(*this); }   // second dispatch: *this is a File
private:
    long bytes_;
};

class Symlink : public Node {
public:
    Symlink(std::string n, std::string target) : Node(std::move(n)), target_(std::move(target)) {}
    const std::string& target() const { return target_; }
    void accept(FsVisitor& v) const override { v.visit(*this); }
private:
    std::string target_;
};

class Directory : public Node {
public:
    explicit Directory(std::string n) : Node(std::move(n)) {}
    Directory& add(std::unique_ptr<Node> child) { kids_.push_back(std::move(child)); return *this; }
    const std::vector<std::unique_ptr<Node>>& children() const { return kids_; }
    void accept(FsVisitor& v) const override { v.visit(*this); }
private:
    std::vector<std::unique_ptr<Node>> kids_;
};

// Concrete visitor 1: total size, like `du -sb` (a symlink counts as 0, we do not follow it).
class SizeVisitor : public FsVisitor {
public:
    long total = 0;
    void visit(const File& f) override { total += f.bytes(); }
    void visit(const Directory& d) override { for (const auto& c : d.children()) c->accept(*this); }
    void visit(const Symlink&) override {}
};

// Concrete visitor 2: an indented listing, like `tree`. Added without touching File/Directory/Symlink.
class TreePrinter : public FsVisitor {
public:
    void visit(const File& f) override { pad(); std::cout << f.name() << " (" << f.bytes() << " B)\n"; }
    void visit(const Symlink& s) override { pad(); std::cout << s.name() << " -> " << s.target() << "\n"; }
    void visit(const Directory& d) override {
        pad(); std::cout << d.name() << "/\n";
        ++depth_;
        for (const auto& c : d.children()) c->accept(*this);
        --depth_;
    }
private:
    int depth_ = 0;
    void pad() const { std::cout << std::string(depth_ * 2, ' '); }
};

int main() {
    std::unique_ptr<Directory> etc(new Directory("etc"));
    etc->add(std::unique_ptr<Node>(new File("hosts", 220)))
        .add(std::unique_ptr<Node>(new File("fstab", 640)));
    std::unique_ptr<Directory> log(new Directory("log"));
    log->add(std::unique_ptr<Node>(new File("syslog", 40960)))
        .add(std::unique_ptr<Node>(new Symlink("messages", "syslog")));
    Directory root("root");
    root.add(std::move(etc)).add(std::move(log)).add(std::unique_ptr<Node>(new File("vmlinuz", 8192)));

    TreePrinter printer;
    root.accept(printer);
    SizeVisitor du;
    root.accept(du);
    std::cout << "total bytes: " << du.total << "\n";
    return 0;
}

In C++ the visitor can use overloading (visit(const File&), visit(const Directory&)) because inside each accept the type of *this is exact. Note who walks the tree: here the visitor recurses into children. In the Java cart, the structure loops. Both are legal; pick one and be consistent.

The JDK's own Visitor: Files.walkFileTree

java.nio.file.FileVisitor is the visitor you are most likely to use for real. The JDK owns the walk; you supply callbacks for "entering a directory", "a file", "leaving a directory" and "a file I could not read". The return value, a FileVisitResult, lets the visitor steer the walk (CONTINUE, SKIP_SUBTREE, SKIP_SIBLINGS, TERMINATE).

import java.io.IOException;
import java.nio.file.*;
import java.nio.file.attribute.BasicFileAttributes;

// The JDK's own Visitor: Files.walkFileTree calls back into a FileVisitor for every entry.
public class WalkDemo {
    public static void main(String[] args) throws IOException {
        Path root = Files.createTempDirectory("walkdemo");
        Files.createDirectories(root.resolve("src/main"));
        Files.writeString(root.resolve("src/main/App.java"), "class App {}");          // 12 bytes
        Files.writeString(root.resolve("src/README.md"), "# demo\n");                  // 7 bytes
        Files.createDirectories(root.resolve("node_modules/left-pad"));
        Files.writeString(root.resolve("node_modules/left-pad/index.js"), "x".repeat(500));

        long[] total = {0};
        Files.walkFileTree(root, new SimpleFileVisitor<Path>() {
            @Override
            public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) {
                if (dir.getFileName().toString().equals("node_modules")) {
                    System.out.println("skip  " + root.relativize(dir));
                    return FileVisitResult.SKIP_SUBTREE;                              // the visitor steers the walk
                }
                return FileVisitResult.CONTINUE;
            }
            @Override
            public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) {
                total[0] += attrs.size();
                System.out.println("file  " + root.relativize(file).toString().replace('\\', '/') + " " + attrs.size());
                return FileVisitResult.CONTINUE;
            }
        });
        System.out.println("bytes (without node_modules): " + total[0]);

        // clean up: post-order delete is another visitor
        Files.walkFileTree(root, new SimpleFileVisitor<Path>() {
            @Override public FileVisitResult visitFile(Path f, BasicFileAttributes a) throws IOException { Files.delete(f); return FileVisitResult.CONTINUE; }
            @Override public FileVisitResult postVisitDirectory(Path d, IOException e) throws IOException { Files.delete(d); return FileVisitResult.CONTINUE; }
        });
    }
}
$ java -cp out WalkDemo
skip  node_modules
file  src/main/App.java 12
file  src/README.md 7
bytes (without node_modules): 19

Strictly, FileVisitor does single dispatch (the "element types" are fixed events, not classes), but it has the Visitor spirit: one traversal, many operations, each operation in one class. A post-order delete is just another visitor.
