import java.util.*;

// ---------- storage: a gap buffer (what Emacs uses) ----------
final class GapBuffer {
    private char[] buf;
    private int gapStart, gapEnd;               // the gap is buf[gapStart, gapEnd)

    GapBuffer(int capacity) { buf = new char[Math.max(capacity, 8)]; gapEnd = buf.length; }

    int length() { return buf.length - (gapEnd - gapStart); }

    char charAt(int i) {
        if (i < 0 || i >= length()) throw new IndexOutOfBoundsException("index " + i);
        return i < gapStart ? buf[i] : buf[i + (gapEnd - gapStart)];
    }

    private void moveGap(int pos) {             // O(distance): typing at one place is O(1)
        if (pos < gapStart) {
            int n = gapStart - pos;
            System.arraycopy(buf, pos, buf, gapEnd - n, n);
            gapStart -= n; gapEnd -= n;
        } else if (pos > gapStart) {
            int n = pos - gapStart;
            System.arraycopy(buf, gapEnd, buf, gapStart, n);
            gapStart += n; gapEnd += n;
        }
    }

    private void ensureGap(int need) {          // grow by doubling, like a vector
        if (gapEnd - gapStart >= need) return;
        int cap = Math.max(buf.length * 2, length() + need + 8);
        char[] nb = new char[cap];
        int tail = buf.length - gapEnd;
        System.arraycopy(buf, 0, nb, 0, gapStart);
        System.arraycopy(buf, gapEnd, nb, cap - tail, tail);
        gapEnd = cap - tail;
        buf = nb;
    }

    void insert(int pos, String s) {
        checkPos(pos);
        moveGap(pos);
        ensureGap(s.length());
        s.getChars(0, s.length(), buf, gapStart);
        gapStart += s.length();
    }

    String delete(int pos, int len) {           // returns the removed text (commands keep it for undo)
        checkPos(pos);
        if (len < 0 || pos + len > length()) throw new IndexOutOfBoundsException("delete " + pos + "+" + len);
        moveGap(pos);
        String removed = new String(buf, gapEnd, len);
        gapEnd += len;
        return removed;
    }

    private void checkPos(int pos) {
        if (pos < 0 || pos > length()) throw new IndexOutOfBoundsException("position " + pos);
    }

    @Override public String toString() {
        return new String(buf, 0, gapStart) + new String(buf, gapEnd, buf.length - gapEnd);
    }
}

// ---------- the document: text plus line arithmetic ----------
final class Document {
    private final GapBuffer text = new GapBuffer(16);

    int length() { return text.length(); }
    char charAt(int i) { return text.charAt(i); }
    void insert(int pos, String s) { text.insert(pos, s); }
    String delete(int pos, int len) { return text.delete(pos, len); }
    @Override public String toString() { return text.toString(); }

    int lineStart(int pos) { while (pos > 0 && charAt(pos - 1) != '\n') pos--; return pos; }
    int lineEnd(int pos) { while (pos < length() && charAt(pos) != '\n') pos++; return pos; }
    int lineOf(int pos) { int n = 0; for (int i = 0; i < pos; i++) if (charAt(i) == '\n') n++; return n; }
}

// ---------- Command: every edit knows how to undo itself ----------
interface EditCommand {
    int execute(Document d);                    // apply; returns where the cursor goes
    void undo(Document d);                      // apply the exact inverse
}

final class InsertCommand implements EditCommand {
    private final int pos; private String text;
    InsertCommand(int pos, String text) { this.pos = pos; this.text = text; }
    public int execute(Document d) { d.insert(pos, text); return pos + text.length(); }
    public void undo(Document d) { d.delete(pos, text.length()); }

    /** Word-level undo: glue single typed characters onto this insert until a space or newline. */
    boolean absorb(InsertCommand next) {
        char last = text.charAt(text.length() - 1);
        boolean ok = next.pos == pos + text.length() && next.text.length() == 1
                && next.text.charAt(0) != '\n' && last != ' ' && last != '\n';
        if (ok) text += next.text;
        return ok;
    }
}

final class DeleteCommand implements EditCommand {
    private final int pos, len; private String removed = "";
    DeleteCommand(int pos, int len) { this.pos = pos; this.len = len; }
    public int execute(Document d) { removed = d.delete(pos, len); return pos; }
    public void undo(Document d) { d.insert(pos, removed); }
}

final class MacroCommand implements EditCommand {   // Composite: many edits, one undo step
    private final List<EditCommand> parts;
    MacroCommand(List<EditCommand> parts) { this.parts = parts; }
    public int execute(Document d) { int c = 0; for (EditCommand p : parts) c = p.execute(d); return c; }
    public void undo(Document d) { for (int i = parts.size() - 1; i >= 0; i--) parts.get(i).undo(d); }
}

// ---------- Flyweight: one shared Glyph per (character, font) ----------
final class Glyph {
    final char ch; final String font;           // intrinsic state only; position is extrinsic
    Glyph(char ch, String font) { this.ch = ch; this.font = font; }
}

final class GlyphFactory {
    private final Map<String, Glyph> pool = new HashMap<>();
    Glyph get(char ch, String font) { return pool.computeIfAbsent(font + '\u0000' + ch, k -> new Glyph(ch, font)); }
    int size() { return pool.size(); }
}

// ---------- the editor: facade, cursor, history ----------
final class Editor {
    /** Memento: an opaque snapshot. Only Editor can read it. */
    static final class Snapshot {
        private final String text; private final int cursor;
        private Snapshot(String text, int cursor) { this.text = text; this.cursor = cursor; }
    }

    private static final class Entry {          // one undo step: the command + the cursor before it
        final EditCommand cmd; final int cursorBefore;
        Entry(EditCommand cmd, int cursorBefore) { this.cmd = cmd; this.cursorBefore = cursorBefore; }
    }

    private final Document doc = new Document();
    private final Deque<Entry> undo = new ArrayDeque<>(), redo = new ArrayDeque<>();
    private final int historyLimit, pageLines;
    private int cursor, goalCol = -1;           // goalCol: column remembered across up/down moves
    private boolean canMerge;

    Editor(int historyLimit, int pageLines) { this.historyLimit = historyLimit; this.pageLines = pageLines; }

    // ----- edits -----
    synchronized void type(String s) {
        if (s.isEmpty()) return;
        InsertCommand c = new InsertCommand(cursor, s);
        if (canMerge && !undo.isEmpty() && undo.peek().cmd instanceof InsertCommand
                && ((InsertCommand) undo.peek().cmd).absorb(c)) {
            cursor = c.execute(doc);            // applied, but folded into the previous undo step
            redo.clear();
            goalCol = -1;
        } else {
            run(c);
        }
        canMerge = s.length() == 1;
    }

    synchronized void backspace() { if (cursor > 0) run(new DeleteCommand(cursor - 1, 1)); }
    synchronized void deleteForward() { if (cursor < doc.length()) run(new DeleteCommand(cursor, 1)); }

    /** Replace every non-overlapping match, left to right. One undo step. Returns the count. */
    synchronized int replaceAll(String find, String with) {
        if (find.isEmpty()) throw new IllegalArgumentException("empty search string");
        String t = doc.toString();
        List<Integer> hits = new ArrayList<>();
        for (int i = t.indexOf(find); i >= 0; i = t.indexOf(find, i + find.length())) hits.add(i);
        if (hits.isEmpty()) return 0;
        List<EditCommand> parts = new ArrayList<>();
        for (int k = hits.size() - 1; k >= 0; k--) {   // right to left: earlier offsets stay valid
            parts.add(new DeleteCommand(hits.get(k), find.length()));
            parts.add(new InsertCommand(hits.get(k), with));
        }
        run(new MacroCommand(parts));
        return hits.size();
    }

    // ----- memento checkpoint: save and revert (the revert is itself undoable) -----
    synchronized Snapshot save() { return new Snapshot(doc.toString(), cursor); }

    synchronized void restore(Snapshot s) {
        run(new MacroCommand(List.of(new DeleteCommand(0, doc.length()), new InsertCommand(0, s.text))));
        cursor = s.cursor;
    }

    private void run(EditCommand c) {
        int before = cursor;
        cursor = c.execute(doc);
        undo.push(new Entry(c, before));
        if (undo.size() > historyLimit) undo.removeLast();   // bounded history: drop the oldest
        redo.clear();                                       // a new edit forks history
        canMerge = false; goalCol = -1;
    }

    synchronized boolean undo() {
        if (undo.isEmpty()) return false;
        Entry e = undo.pop();
        e.cmd.undo(doc);
        cursor = e.cursorBefore;
        redo.push(e);
        canMerge = false; goalCol = -1;
        return true;
    }

    synchronized boolean redo() {
        if (redo.isEmpty()) return false;
        Entry e = redo.pop();
        cursor = e.cmd.execute(doc);
        undo.push(e);
        canMerge = false; goalCol = -1;
        return true;
    }

    // ----- cursor moves (not edits: never recorded in history) -----
    synchronized void left() { moved(Math.max(0, cursor - 1)); }
    synchronized void right() { moved(Math.min(doc.length(), cursor + 1)); }
    synchronized void home() { moved(doc.lineStart(cursor)); }
    synchronized void end() { moved(doc.lineEnd(cursor)); }
    synchronized void up() { vertical(-1); }
    synchronized void down() { vertical(+1); }
    synchronized void pageUp() { for (int i = 0; i < pageLines; i++) vertical(-1); }
    synchronized void pageDown() { for (int i = 0; i < pageLines; i++) vertical(+1); }

    private void moved(int pos) { cursor = pos; goalCol = -1; canMerge = false; }

    private void vertical(int dir) {
        int start = doc.lineStart(cursor);
        if (goalCol < 0) goalCol = cursor - start;
        int target;
        if (dir < 0) {
            if (start == 0) return;             // already on the first line
            target = doc.lineStart(start - 1);
        } else {
            int end = doc.lineEnd(cursor);
            if (end == doc.length()) return;    // already on the last line
            target = end + 1;
        }
        cursor = Math.min(target + goalCol, doc.lineEnd(target));
        canMerge = false;
    }

    // ----- queries -----
    synchronized String text() { return doc.toString(); }
    synchronized int cursor() { return cursor; }
    synchronized String where() { return (doc.lineOf(cursor) + 1) + ":" + (cursor - doc.lineStart(cursor) + 1); }
    synchronized String view() { String t = doc.toString(); return (t.substring(0, cursor) + "|" + t.substring(cursor)).replace("\n", "\\n"); }
    synchronized int undoDepth() { return undo.size(); }

    synchronized List<Glyph> layout(GlyphFactory f, String font) {
        List<Glyph> out = new ArrayList<>();
        for (int i = 0; i < doc.length(); i++) out.add(f.get(doc.charAt(i), font));
        return out;
    }
}

public class Main {
    private static int checks;
    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError("FAILED: " + what);
        checks++;
    }
    static void show(Editor e, String label) { System.out.println("  " + label + ": \"" + e.view() + "\" at " + e.where()); }

    public static void main(String[] args) {
        Editor ed = new Editor(100, 2);

        System.out.println("1) type 'hello world' one key at a time");
        for (char c : "hello world".toCharArray()) ed.type(String.valueOf(c));
        show(ed, "text");
        check(ed.text().equals("hello world") && ed.undoDepth() == 2, "word-level groups: 'hello ' + 'world'");

        System.out.println("2) undo twice, redo once");
        ed.undo(); show(ed, "undo");
        check(ed.text().equals("hello "), "undo removes the last word");
        ed.undo(); show(ed, "undo");
        check(ed.text().isEmpty() && ed.cursor() == 0, "undo to empty");
        ed.redo(); show(ed, "redo");
        check(ed.text().equals("hello ") && ed.cursor() == 6, "redo restores text and cursor");

        System.out.println("3) a new edit after undo clears redo");
        ed.type("there");
        check(!ed.redo(), "redo stack forked away");
        show(ed, "text");

        System.out.println("4) multi-line text, arrows remember the column");
        ed.type("\nab\nlonger line");
        show(ed, "text");
        ed.up(); show(ed, "up");
        check(ed.where().equals("2:3"), "short line clamps column");
        ed.up(); show(ed, "up");
        check(ed.where().equals("1:12"), "goal column 12 restored on a long line");
        ed.pageDown(); show(ed, "pgdn");
        check(ed.where().equals("3:12"), "page down by 2 lines");
        ed.home(); ed.left(); show(ed, "home, left");
        check(ed.where().equals("2:3"), "left wraps to the end of the previous line");

        System.out.println("5) backspace and forward delete, then undo all three");
        ed.backspace(); ed.backspace(); show(ed, "2 x backspace");
        ed.deleteForward(); show(ed, "delete");
        check(ed.text().equals("hello there\nlonger line"), "deleted 'ab' and a newline");
        ed.undo(); ed.undo(); ed.undo(); show(ed, "3 x undo");
        check(ed.text().equals("hello there\nab\nlonger line") && ed.where().equals("2:3"), "edits reversed exactly");

        System.out.println("6) replace all is one undo step");
        Editor.Snapshot saved = ed.save();
        int n = ed.replaceAll("l", "L");
        show(ed, "replaced " + n);
        check(n == 4 && ed.text().equals("heLLo there\nab\nLonger Line"), "4 matches replaced");
        ed.undo();
        check(ed.text().equals("hello there\nab\nlonger line"), "one undo reverts all 4");
        ed.redo(); ed.type("!");
        show(ed, "redo + type");

        System.out.println("7) revert to the saved checkpoint (Memento), and undo the revert");
        ed.restore(saved); show(ed, "restored");
        check(ed.text().equals("hello there\nab\nlonger line"), "checkpoint restored");
        ed.undo(); show(ed, "undo revert");
        check(ed.text().contains("!"), "revert is itself undoable");

        System.out.println("8) bounded history keeps the newest 3 steps");
        Editor small = new Editor(3, 1);
        for (String w : new String[]{"a ", "b ", "c ", "d ", "e "}) small.type(w);
        int undone = 0; while (small.undo()) undone++;
        System.out.println("  after " + undone + " undos: \"" + small.text() + "\"");
        check(undone == 3 && small.text().equals("a b "), "oldest steps dropped");

        System.out.println("9) Flyweight glyphs");
        Editor big = new Editor(10, 1);
        for (int i = 0; i < 100; i++) big.type("hello world ");
        GlyphFactory f = new GlyphFactory();
        List<Glyph> glyphs = big.layout(f, "Mono-12");
        System.out.println("  " + glyphs.size() + " glyph references, " + f.size() + " glyph objects");
        check(glyphs.size() == 1200 && f.size() == 8 && glyphs.get(0) == glyphs.get(12), "shared glyphs");

        System.out.println("All " + checks + " checks passed");
    }
}
"""Text editor with undo/redo: gap buffer + Command + Memento + Flyweight glyphs."""
from __future__ import annotations

import threading
from abc import ABC, abstractmethod
from collections import deque


# ---------- storage: a gap buffer ----------
class GapBuffer:
    def __init__(self, capacity: int = 16) -> None:
        self._buf: list[str] = [""] * max(capacity, 8)
        self._gs, self._ge = 0, len(self._buf)          # gap is _buf[_gs:_ge]

    def __len__(self) -> int:
        return len(self._buf) - (self._ge - self._gs)

    def char_at(self, i: int) -> str:
        if not 0 <= i < len(self):
            raise IndexError(f"index {i}")
        return self._buf[i] if i < self._gs else self._buf[i + self._ge - self._gs]

    def _move_gap(self, pos: int) -> None:              # O(distance moved)
        if pos < self._gs:
            n = self._gs - pos
            self._buf[self._ge - n:self._ge] = self._buf[pos:self._gs]
            self._gs, self._ge = self._gs - n, self._ge - n
        elif pos > self._gs:
            n = pos - self._gs
            self._buf[self._gs:self._gs + n] = self._buf[self._ge:self._ge + n]
            self._gs, self._ge = self._gs + n, self._ge + n

    def _ensure_gap(self, need: int) -> None:
        if self._ge - self._gs >= need:
            return
        cap = max(len(self._buf) * 2, len(self) + need + 8)
        tail = self._buf[self._ge:]
        self._buf = self._buf[:self._gs] + [""] * (cap - self._gs - len(tail)) + tail
        self._ge = cap - len(tail)

    def _check(self, pos: int) -> None:
        if not 0 <= pos <= len(self):
            raise IndexError(f"position {pos}")

    def insert(self, pos: int, s: str) -> None:
        self._check(pos)
        self._move_gap(pos)
        self._ensure_gap(len(s))
        self._buf[self._gs:self._gs + len(s)] = list(s)
        self._gs += len(s)

    def delete(self, pos: int, n: int) -> str:
        self._check(pos)
        if n < 0 or pos + n > len(self):
            raise IndexError(f"delete {pos}+{n}")
        self._move_gap(pos)
        removed = "".join(self._buf[self._ge:self._ge + n])
        self._ge += n
        return removed

    def __str__(self) -> str:
        return "".join(self._buf[:self._gs]) + "".join(self._buf[self._ge:])


class Document:
    def __init__(self) -> None:
        self._text = GapBuffer()

    def __len__(self) -> int:
        return len(self._text)

    def char_at(self, i: int) -> str:
        return self._text.char_at(i)

    def insert(self, pos: int, s: str) -> None:
        self._text.insert(pos, s)

    def delete(self, pos: int, n: int) -> str:
        return self._text.delete(pos, n)

    def __str__(self) -> str:
        return str(self._text)

    def line_start(self, pos: int) -> int:
        while pos > 0 and self.char_at(pos - 1) != "\n":
            pos -= 1
        return pos

    def line_end(self, pos: int) -> int:
        while pos < len(self) and self.char_at(pos) != "\n":
            pos += 1
        return pos

    def line_of(self, pos: int) -> int:
        return sum(1 for i in range(pos) if self.char_at(i) == "\n")


# ---------- Command ----------
class EditCommand(ABC):
    @abstractmethod
    def execute(self, d: Document) -> int: ...      # returns the new cursor

    @abstractmethod
    def undo(self, d: Document) -> None: ...


class InsertCommand(EditCommand):
    def __init__(self, pos: int, text: str) -> None:
        self.pos, self.text = pos, text

    def execute(self, d: Document) -> int:
        d.insert(self.pos, self.text)
        return self.pos + len(self.text)

    def undo(self, d: Document) -> None:
        d.delete(self.pos, len(self.text))

    def absorb(self, nxt: "InsertCommand") -> bool:
        last = self.text[-1]
        ok = (nxt.pos == self.pos + len(self.text) and len(nxt.text) == 1
              and nxt.text != "\n" and last not in " \n")
        if ok:
            self.text += nxt.text
        return ok


class DeleteCommand(EditCommand):
    def __init__(self, pos: int, n: int) -> None:
        self.pos, self.n, self.removed = pos, n, ""

    def execute(self, d: Document) -> int:
        self.removed = d.delete(self.pos, self.n)
        return self.pos

    def undo(self, d: Document) -> None:
        d.insert(self.pos, self.removed)


class MacroCommand(EditCommand):
    def __init__(self, parts: list[EditCommand]) -> None:
        self.parts = parts

    def execute(self, d: Document) -> int:
        c = 0
        for p in self.parts:
            c = p.execute(d)
        return c

    def undo(self, d: Document) -> None:
        for p in reversed(self.parts):
            p.undo(d)


# ---------- Flyweight ----------
class Glyph:
    __slots__ = ("ch", "font")

    def __init__(self, ch: str, font: str) -> None:
        self.ch, self.font = ch, font


class GlyphFactory:
    def __init__(self) -> None:
        self._pool: dict[tuple[str, str], Glyph] = {}

    def get(self, ch: str, font: str) -> Glyph:
        key = (ch, font)
        if key not in self._pool:
            self._pool[key] = Glyph(ch, font)
        return self._pool[key]

    def __len__(self) -> int:
        return len(self._pool)


# ---------- editor ----------
class Snapshot:
    """Memento: callers hold it but should not look inside."""
    __slots__ = ("_text", "_cursor")

    def __init__(self, text: str, cursor: int) -> None:
        self._text, self._cursor = text, cursor


class Editor:
    def __init__(self, history_limit: int = 100, page_lines: int = 20) -> None:
        self._doc = Document()
        self._undo: deque[tuple[EditCommand, int]] = deque()
        self._redo: list[tuple[EditCommand, int]] = []
        self._limit, self._page = history_limit, page_lines
        self._cursor, self._goal, self._can_merge = 0, -1, False
        self._lock = threading.RLock()

    # ----- edits -----
    def type(self, s: str) -> None:
        with self._lock:
            if not s:
                return
            c = InsertCommand(self._cursor, s)
            top = self._undo[-1][0] if self._undo else None
            if self._can_merge and isinstance(top, InsertCommand) and top.absorb(c):
                self._cursor = c.execute(self._doc)
                self._redo.clear()
                self._goal = -1
            else:
                self._run(c)
            self._can_merge = len(s) == 1

    def backspace(self) -> None:
        with self._lock:
            if self._cursor > 0:
                self._run(DeleteCommand(self._cursor - 1, 1))

    def delete_forward(self) -> None:
        with self._lock:
            if self._cursor < len(self._doc):
                self._run(DeleteCommand(self._cursor, 1))

    def replace_all(self, find: str, with_: str) -> int:
        with self._lock:
            if not find:
                raise ValueError("empty search string")
            t, hits, i = str(self._doc), [], 0
            while (i := t.find(find, i)) >= 0:
                hits.append(i)
                i += len(find)
            if not hits:
                return 0
            parts: list[EditCommand] = []
            for h in reversed(hits):                     # right to left keeps offsets valid
                parts += [DeleteCommand(h, len(find)), InsertCommand(h, with_)]
            self._run(MacroCommand(parts))
            return len(hits)

    def save(self) -> Snapshot:
        with self._lock:
            return Snapshot(str(self._doc), self._cursor)

    def restore(self, s: Snapshot) -> None:
        with self._lock:
            self._run(MacroCommand([DeleteCommand(0, len(self._doc)), InsertCommand(0, s._text)]))
            self._cursor = s._cursor

    def _run(self, c: EditCommand) -> None:
        before = self._cursor
        self._cursor = c.execute(self._doc)
        self._undo.append((c, before))
        if len(self._undo) > self._limit:
            self._undo.popleft()                         # drop the oldest step
        self._redo.clear()
        self._can_merge, self._goal = False, -1

    def undo(self) -> bool:
        with self._lock:
            if not self._undo:
                return False
            c, before = self._undo.pop()
            c.undo(self._doc)
            self._cursor = before
            self._redo.append((c, before))
            self._can_merge, self._goal = False, -1
            return True

    def redo(self) -> bool:
        with self._lock:
            if not self._redo:
                return False
            c, before = self._redo.pop()
            self._cursor = c.execute(self._doc)
            self._undo.append((c, before))
            self._can_merge, self._goal = False, -1
            return True

    # ----- moves -----
    def _moved(self, pos: int) -> None:
        self._cursor, self._goal, self._can_merge = pos, -1, False

    def left(self) -> None:
        with self._lock:
            self._moved(max(0, self._cursor - 1))

    def right(self) -> None:
        with self._lock:
            self._moved(min(len(self._doc), self._cursor + 1))

    def home(self) -> None:
        with self._lock:
            self._moved(self._doc.line_start(self._cursor))

    def end(self) -> None:
        with self._lock:
            self._moved(self._doc.line_end(self._cursor))

    def up(self) -> None:
        with self._lock:
            self._vertical(-1)

    def down(self) -> None:
        with self._lock:
            self._vertical(+1)

    def page_up(self) -> None:
        with self._lock:
            for _ in range(self._page):
                self._vertical(-1)

    def page_down(self) -> None:
        with self._lock:
            for _ in range(self._page):
                self._vertical(+1)

    def _vertical(self, direction: int) -> None:
        d = self._doc
        start = d.line_start(self._cursor)
        if self._goal < 0:
            self._goal = self._cursor - start
        if direction < 0:
            if start == 0:
                return
            target = d.line_start(start - 1)
        else:
            end = d.line_end(self._cursor)
            if end == len(d):
                return
            target = end + 1
        self._cursor = min(target + self._goal, d.line_end(target))
        self._can_merge = False

    # ----- queries -----
    @property
    def text(self) -> str:
        return str(self._doc)

    @property
    def cursor(self) -> int:
        return self._cursor

    def where(self) -> str:
        d = self._doc
        return f"{d.line_of(self._cursor) + 1}:{self._cursor - d.line_start(self._cursor) + 1}"

    def view(self) -> str:
        t = self.text
        return (t[:self._cursor] + "|" + t[self._cursor:]).replace("\n", "\\n")

    def undo_depth(self) -> int:
        return len(self._undo)

    def layout(self, f: GlyphFactory, font: str) -> list[Glyph]:
        with self._lock:
            return [f.get(self._doc.char_at(i), font) for i in range(len(self._doc))]

class UndoRedoTest(unittest.TestCase):
    def test_word_level_grouping(self):
        ed = Editor()
        for ch in "hello world\nbye":
            ed.type(ch)
        self.assertEqual(ed.undo_depth(), 4)             # "hello ", "world", "\n", "bye"

        ed.type("y")
        self.assertEqual(ed.undo_depth(), 4)
        ed.undo()
        self.assertEqual(ed.text, "hello world\n")

    def test_command_undo_equals_snapshot_undo(self):
        """Random edits: Command-based undo must give exactly what full snapshots (Memento) give."""
        rng = random.Random(42)
        for _ in range(30):
            ed = Editor(history_limit=1000, page_lines=3)
            text, cursor = "", 0
            undo, redo = [], []                           # reference: (before, after) whole-state snapshots
            for _ in range(300):
                op = rng.choice(["type", "type", "bs", "del", "move", "repl", "undo", "redo"])
                before = (text, cursor)
                if op == "type":
                    s = "".join(rng.choice("ab \n") for _ in range(rng.randint(2, 4)))
                    ed.type(s)
                    text, cursor = text[:cursor] + s + text[cursor:], cursor + len(s)
                elif op == "bs":
                    ed.backspace()
                    if cursor > 0:
                        text, cursor = text[:cursor - 1] + text[cursor:], cursor - 1
                elif op == "del":
                    ed.delete_forward()
                    if cursor < len(text):
                        text = text[:cursor] + text[cursor + 1:]
                elif op == "move":
                    rng.choice([ed.left, ed.right, ed.up, ed.down, ed.home, ed.end, ed.page_up, ed.page_down])()
                    cursor = ed.cursor                    # moves are tested elsewhere
                    continue
                elif op == "repl":
                    find, rep = rng.choice([("a", "bb"), ("ab", ""), ("b", "a"), ("\n", " ")])
                    n = ed.replace_all(find, rep)
                    self.assertEqual(n, text.count(find))
                    if n:
                        first = text.find(find)
                        text, cursor = text.replace(find, rep), first + len(rep)
                elif op == "undo":
                    ok = ed.undo()
                    self.assertEqual(ok, bool(undo))
                    if undo:
                        step = undo.pop()
                        redo.append(step)
                        text, cursor = step[0]
                    self.assertEqual((ed.text, ed.cursor), (text, cursor))
                    continue
                else:
                    ok = ed.redo()
                    self.assertEqual(ok, bool(redo))
                    if redo:
                        step = redo.pop()
                        undo.append(step)
                        text, cursor = step[1]
                    self.assertEqual((ed.text, ed.cursor), (text, cursor))
                    continue
                if (text, cursor) != before:                  # only real edits enter history
                    undo.append((before, (text, cursor)))
                    redo.clear()
                self.assertEqual((ed.text, ed.cursor), (text, cursor))

// Text editor with undo/redo: gap buffer + Command + Memento + Flyweight. C++14.
// Threaded build: add -DED_THREADS -pthread (a std::mutex around every public call). Our MinGW g++ 6.3 has no
// std::mutex, so the default build is single-threaded, like one keyboard feeding one editor. The threaded
// build was compiled and run on WSL Ubuntu (g++ 13): same output, see run-cpp-linux.txt.
#include <algorithm>
#include <cstdio>
#include <cstdlib>
#include <deque>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>
#ifdef ED_THREADS
#include <mutex>
#define ED_LOCK std::lock_guard<std::mutex> guard_(mu_)
#define ED_MUTEX mutable std::mutex mu_;
#else
#define ED_LOCK (void)0
#define ED_MUTEX
#endif

// ---------- gap buffer ----------
class GapBuffer {
    std::vector<char> buf_;
    size_t gs_ = 0, ge_;                                   // gap is [gs_, ge_)
    void moveGap(size_t pos) {
        if (pos < gs_) {
            size_t n = gs_ - pos;
            std::copy_backward(buf_.begin() + pos, buf_.begin() + gs_, buf_.begin() + ge_);
            gs_ -= n; ge_ -= n;
        } else if (pos > gs_) {
            size_t n = pos - gs_;
            std::copy(buf_.begin() + ge_, buf_.begin() + ge_ + n, buf_.begin() + gs_);
            gs_ += n; ge_ += n;
        }
    }
    void ensureGap(size_t need) {
        if (ge_ - gs_ >= need) return;
        size_t cap = std::max(buf_.size() * 2, length() + need + 8);
        size_t tail = buf_.size() - ge_;
        std::vector<char> nb(cap);
        std::copy(buf_.begin(), buf_.begin() + gs_, nb.begin());
        std::copy(buf_.begin() + ge_, buf_.end(), nb.end() - tail);
        ge_ = cap - tail;
        buf_.swap(nb);
    }
    void check(size_t pos) const { if (pos > length()) throw std::out_of_range("position " + std::to_string(pos)); }
public:
    explicit GapBuffer(size_t cap = 16) : buf_(std::max<size_t>(cap, 8)), ge_(buf_.size()) {}
    size_t length() const { return buf_.size() - (ge_ - gs_); }
    char at(size_t i) const {
        if (i >= length()) throw std::out_of_range("index " + std::to_string(i));
        return i < gs_ ? buf_[i] : buf_[i + (ge_ - gs_)];
    }
    void insert(size_t pos, const std::string& s) {
        check(pos); moveGap(pos); ensureGap(s.size());
        std::copy(s.begin(), s.end(), buf_.begin() + gs_);
        gs_ += s.size();
    }
    std::string erase(size_t pos, size_t n) {
        check(pos);
        if (pos + n > length()) throw std::out_of_range("erase past end");
        moveGap(pos);
        std::string removed(buf_.begin() + ge_, buf_.begin() + ge_ + n);
        ge_ += n;
        return removed;
    }
    std::string str() const {
        return std::string(buf_.begin(), buf_.begin() + gs_) + std::string(buf_.begin() + ge_, buf_.end());
    }
};

class Document {
    GapBuffer text_;
public:
    size_t length() const { return text_.length(); }
    char at(size_t i) const { return text_.at(i); }
    void insert(size_t pos, const std::string& s) { text_.insert(pos, s); }
    std::string erase(size_t pos, size_t n) { return text_.erase(pos, n); }
    std::string str() const { return text_.str(); }
    size_t lineStart(size_t p) const { while (p > 0 && at(p - 1) != '\n') --p; return p; }
    size_t lineEnd(size_t p) const { while (p < length() && at(p) != '\n') ++p; return p; }
    size_t lineOf(size_t p) const { size_t n = 0; for (size_t i = 0; i < p; ++i) n += at(i) == '\n'; return n; }
};

// ---------- commands ----------
struct EditCommand {                                       // like a vtable of two function pointers
    virtual ~EditCommand() = default;
    virtual size_t execute(Document& d) = 0;               // returns the new cursor
    virtual void undo(Document& d) = 0;
};

class InsertCommand : public EditCommand {
    size_t pos_; std::string text_;
public:
    InsertCommand(size_t pos, std::string t) : pos_(pos), text_(std::move(t)) {}
    size_t execute(Document& d) override { d.insert(pos_, text_); return pos_ + text_.size(); }
    void undo(Document& d) override { d.erase(pos_, text_.size()); }
    bool absorb(const InsertCommand& n) {
        char last = text_.back();
        bool ok = n.pos_ == pos_ + text_.size() && n.text_.size() == 1 && n.text_[0] != '\n' && last != ' ' && last != '\n';
        if (ok) text_ += n.text_;
        return ok;
    }
};

class DeleteCommand : public EditCommand {
    size_t pos_, n_; std::string removed_;
public:
    DeleteCommand(size_t pos, size_t n) : pos_(pos), n_(n) {}
    size_t execute(Document& d) override { removed_ = d.erase(pos_, n_); return pos_; }
    void undo(Document& d) override { d.insert(pos_, removed_); }
};

class MacroCommand : public EditCommand {
    std::vector<std::unique_ptr<EditCommand>> parts_;
public:
    explicit MacroCommand(std::vector<std::unique_ptr<EditCommand>> p) : parts_(std::move(p)) {}
    size_t execute(Document& d) override { size_t c = 0; for (auto& p : parts_) c = p->execute(d); return c; }
    void undo(Document& d) override { for (auto it = parts_.rbegin(); it != parts_.rend(); ++it) (*it)->undo(d); }
};

// ---------- flyweight ----------
struct Glyph { char ch; std::string font; };
class GlyphFactory {
    std::map<std::pair<char, std::string>, std::unique_ptr<Glyph>> pool_;
public:
    const Glyph* get(char c, const std::string& font) {
        auto& slot = pool_[std::make_pair(c, font)];
        if (!slot) slot.reset(new Glyph{c, font});
        return slot.get();
    }
    size_t size() const { return pool_.size(); }
};

// ---------- editor ----------
class Editor {
public:
    class Snapshot {                                       // memento: only Editor reads it
        friend class Editor;
        std::string text_; size_t cursor_;
        Snapshot(std::string t, size_t c) : text_(std::move(t)), cursor_(c) {}
    };
private:
    struct Entry { std::unique_ptr<EditCommand> cmd; size_t before; };
    Document doc_;
    std::deque<Entry> undo_; std::vector<Entry> redo_;
    size_t limit_, page_, cursor_ = 0;
    long goal_ = -1;
    bool canMerge_ = false;
    ED_MUTEX

    void run(std::unique_ptr<EditCommand> c) {
        size_t before = cursor_;
        cursor_ = c->execute(doc_);
        undo_.push_back(Entry{std::move(c), before});
        if (undo_.size() > limit_) undo_.pop_front();
        redo_.clear();
        canMerge_ = false; goal_ = -1;
    }
    void moved(size_t p) { cursor_ = p; goal_ = -1; canMerge_ = false; }
    void vertical(int dir) {
        size_t start = doc_.lineStart(cursor_), target;
        if (goal_ < 0) goal_ = static_cast<long>(cursor_ - start);
        if (dir < 0) {
            if (start == 0) return;
            target = doc_.lineStart(start - 1);
        } else {
            size_t end = doc_.lineEnd(cursor_);
            if (end == doc_.length()) return;
            target = end + 1;
        }
        cursor_ = std::min(target + static_cast<size_t>(goal_), doc_.lineEnd(target));
        canMerge_ = false;
    }
public:
    Editor(size_t limit, size_t page) : limit_(limit), page_(page) {}

    void type(const std::string& s) {
        ED_LOCK;
        if (s.empty()) return;
        std::unique_ptr<InsertCommand> c(new InsertCommand(cursor_, s));
        InsertCommand* top = undo_.empty() ? nullptr : dynamic_cast<InsertCommand*>(undo_.back().cmd.get());
        if (canMerge_ && top && top->absorb(*c)) {
            cursor_ = c->execute(doc_);
            redo_.clear(); goal_ = -1;
        } else {
            run(std::move(c));
        }
        canMerge_ = s.size() == 1;
    }
    void backspace() { ED_LOCK; if (cursor_ > 0) run(std::unique_ptr<EditCommand>(new DeleteCommand(cursor_ - 1, 1))); }
    void deleteForward() { ED_LOCK; if (cursor_ < doc_.length()) run(std::unique_ptr<EditCommand>(new DeleteCommand(cursor_, 1))); }

    int replaceAll(const std::string& find, const std::string& with) {
        ED_LOCK;
        if (find.empty()) throw std::invalid_argument("empty search string");
        std::string t = doc_.str();
        std::vector<size_t> hits;
        for (size_t i = t.find(find); i != std::string::npos; i = t.find(find, i + find.size())) hits.push_back(i);
        if (hits.empty()) return 0;
        std::vector<std::unique_ptr<EditCommand>> parts;
        for (auto it = hits.rbegin(); it != hits.rend(); ++it) {
            parts.emplace_back(new DeleteCommand(*it, find.size()));
            parts.emplace_back(new InsertCommand(*it, with));
        }
        run(std::unique_ptr<EditCommand>(new MacroCommand(std::move(parts))));
        return static_cast<int>(hits.size());
    }

    Snapshot save() const { ED_LOCK; return Snapshot(doc_.str(), cursor_); }
    void restore(const Snapshot& s) {
        ED_LOCK;
        std::vector<std::unique_ptr<EditCommand>> parts;
        parts.emplace_back(new DeleteCommand(0, doc_.length()));
        parts.emplace_back(new InsertCommand(0, s.text_));
        run(std::unique_ptr<EditCommand>(new MacroCommand(std::move(parts))));
        cursor_ = s.cursor_;
    }

    bool undo() {
        ED_LOCK;
        if (undo_.empty()) return false;
        Entry e = std::move(undo_.back()); undo_.pop_back();
        e.cmd->undo(doc_);
        cursor_ = e.before;
        redo_.push_back(std::move(e));
        canMerge_ = false; goal_ = -1;
        return true;
    }
    bool redo() {
        ED_LOCK;
        if (redo_.empty()) return false;
        Entry e = std::move(redo_.back()); redo_.pop_back();
        cursor_ = e.cmd->execute(doc_);
        undo_.push_back(std::move(e));
        canMerge_ = false; goal_ = -1;
        return true;
    }

    void left() { ED_LOCK; moved(cursor_ > 0 ? cursor_ - 1 : 0); }
    void right() { ED_LOCK; moved(std::min(doc_.length(), cursor_ + 1)); }
    void home() { ED_LOCK; moved(doc_.lineStart(cursor_)); }
    void end() { ED_LOCK; moved(doc_.lineEnd(cursor_)); }
    void up() { ED_LOCK; vertical(-1); }
    void down() { ED_LOCK; vertical(+1); }
    void pageUp() { ED_LOCK; for (size_t i = 0; i < page_; ++i) vertical(-1); }
    void pageDown() { ED_LOCK; for (size_t i = 0; i < page_; ++i) vertical(+1); }

    std::string text() const { ED_LOCK; return doc_.str(); }
    size_t cursor() const { ED_LOCK; return cursor_; }
    size_t undoDepth() const { ED_LOCK; return undo_.size(); }
    std::string where() const {
        ED_LOCK;
        return std::to_string(doc_.lineOf(cursor_) + 1) + ":" + std::to_string(cursor_ - doc_.lineStart(cursor_) + 1);
    }
    std::string view() const {
        ED_LOCK;
        std::string t = doc_.str(), out;
        for (size_t i = 0; i <= t.size(); ++i) {
            if (i == cursor_) out += '|';
            if (i < t.size()) out += t[i] == '\n' ? std::string("\\n") : std::string(1, t[i]);
        }
        return out;
    }
    std::vector<const Glyph*> layout(GlyphFactory& f, const std::string& font) const {
        ED_LOCK;
        std::vector<const Glyph*> out;
        for (size_t i = 0; i < doc_.length(); ++i) out.push_back(f.get(doc_.at(i), font));
        return out;
    }
};

static int checks = 0;
static void check(bool ok, const char* what) {
    if (!ok) { std::printf("FAILED: %s\n", what); std::exit(1); }
    ++checks;
}
static void show(const Editor& e, const char* label) { std::printf("  %s: \"%s\" at %s\n", label, e.view().c_str(), e.where().c_str()); }

int main() {
    Editor ed(100, 2);
    std::printf("1) type 'hello world' one key at a time\n");
    for (char c : std::string("hello world")) ed.type(std::string(1, c));
    show(ed, "text");
    check(ed.undoDepth() == 2, "two word groups");
    ed.undo(); show(ed, "undo");
    ed.redo(); ed.type("\nab\nlonger line");
    std::printf("2) arrows remember the column\n");
    ed.up(); show(ed, "up");
    ed.up(); show(ed, "up");
    check(ed.where() == "1:12", "goal column restored");
    ed.pageDown(); check(ed.where() == "3:12", "page down");
    std::printf("3) replace all, one undo\n");
    Editor::Snapshot saved = ed.save();
    int n = ed.replaceAll("l", "L");
    show(ed, ("replaced " + std::to_string(n)).c_str());
    check(n == 5, "5 matches");
    ed.undo(); check(ed.text() == "hello world\nab\nlonger line", "one undo reverts all");
    ed.redo(); ed.restore(saved); show(ed, "restored checkpoint");
    ed.undo(); check(ed.text() == "heLLo worLd\nab\nLonger Line", "restore is undoable");
    std::printf("4) Flyweight glyphs\n");
    GlyphFactory f;
    auto glyphs = ed.layout(f, "Mono-12");
    std::printf("  %lu glyph references, %lu glyph objects\n", (unsigned long)glyphs.size(), (unsigned long)f.size());
    check(glyphs[2] == glyphs[3], "same glyph object for both L");

    return 0;
