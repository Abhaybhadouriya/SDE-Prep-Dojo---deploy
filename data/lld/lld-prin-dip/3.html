After: the policy owns the interfaces

We do three things: (1) describe what the order logic needs as two small interfaces, (2) put those interfaces in the order package, (3) hand the real implementations in through the constructor.

// ---- package order (high level): it OWNS these two interfaces ----
interface OrderRepository { void save(Order o); Order byId(String id); }
interface Notifier        { void send(String to, String message); }

static class OrderService {
    private final OrderRepository repo;
    private final Notifier notifier;

    OrderService(OrderRepository repo, Notifier notifier) {   // constructor injection
        this.repo = repo;
        this.notifier = notifier;
    }

    void place(Order o) {
        if (o.amountPaise() <= 0) throw new IllegalArgumentException("amount");
        repo.save(o);
        notifier.send(o.customerEmail(), "Order " + o.id() + " placed");
    }
}

// ---- package infra (low level) - depends UP on order's interfaces ----
static class MySqlOrderRepository implements OrderRepository {
    private final Map<String, Order> fakeTable = new HashMap<>();   // stands in for JDBC
    public void save(Order o) { fakeTable.put(o.id(), o); System.out.println("[mysql] INSERT order " + o.id()); }
    public Order byId(String id) { return fakeTable.get(id); }
}
static class EmailNotifier implements Notifier {                    // stands in for SMTP
    public void send(String to, String m) { System.out.println("[email] to=" + to + " " + m); }
}
static class SmsNotifier implements Notifier {                      // stands in for an SMS gateway
    public void send(String to, String m) { System.out.println("[sms] to=" + to + " " + m); }
}

// ---- composition root: the ONE place that knows concrete classes ----
static OrderService productionService() {
    return new OrderService(new MySqlOrderRepository(), new EmailNotifier());
}
static class InMemoryOrderRepository implements OrderRepository {
    final Map<String, Order> rows = new HashMap<>();
    public void save(Order o) { rows.put(o.id(), o); }
    public Order byId(String id) { return rows.get(id); }
}
static class RecordingNotifier implements Notifier {
    final List<String> sent = new ArrayList<>();
    public void send(String to, String m) { sent.add(to + "|" + m); }
}

// No database, no network, runs in microseconds. Explicit checks, not `assert`
// (Java skips assert statements unless you run with -ea).
static void unitTest() {
    var repo = new InMemoryOrderRepository();
    var notifier = new RecordingNotifier();
    var svc = new OrderService(repo, notifier);
    svc.place(new Order("o4", "c@x.com", 100));
    check(repo.byId("o4") != null, "order saved");
    check(notifier.sent.size() == 1, "one notification sent");
    try { svc.place(new Order("o5", "d@x.com", 0)); check(false, "should reject"); }
    catch (IllegalArgumentException e) { check(repo.byId("o5") == null && notifier.sent.size() == 1, "bad order rejected, nothing saved or sent"); }
}

static void check(boolean ok, String what) {
    if (!ok) throw new AssertionError(what);
    System.out.println("PASS " + what);
}
class OrderRepository(Protocol):          # owned by the order (policy) layer
    def save(self, order: Order) -> None: ...
    def by_id(self, order_id: str) -> Order | None: ...


class Notifier(Protocol):
    def send(self, to: str, message: str) -> None: ...


class OrderService:
    def __init__(self, repo: OrderRepository, notifier: Notifier) -> None:
        self._repo = repo
        self._notifier = notifier

    def place(self, order: Order) -> None:
        if order.amount_paise <= 0:
            raise ValueError("amount")
        self._repo.save(order)
        self._notifier.send(order.email, f"Order {order.id} placed")
