Build it: the VPC as code

Good to know (SDE-2)

Terraform is taught properly in Terraform Basics; for now read each resource block as one AWS object from this page, and its arguments as the settings you just learned (count makes one copy per AZ, cidrsubnet() does the subnet maths from the CIDR section, and a reference such as aws_vpc.main.id tells Terraform what must exist first).

Nobody clicks a production VPC together in the console. Here is the core of a two-AZ VPC in Terraform, written resource by resource so you can see every piece from this page. (Most teams use the community terraform-aws-modules/vpc/aws module, which does the same thing with a few inputs.)

# main.tf (Terraform >= 1.5, AWS provider 5.x or 6.x)
locals {
  azs     = ["ap-south-1a", "ap-south-1b"]
  public  = ["10.10.0.0/20", "10.10.16.0/20"]
  private = ["10.10.64.0/19", "10.10.96.0/19"]
}

resource "aws_vpc" "main" {
  cidr_block           = "10.10.0.0/16"
  enable_dns_support   = true
  enable_dns_hostnames = true
  tags = { Name = "prod-mumbai" }
}

resource "aws_internet_gateway" "igw" { vpc_id = aws_vpc.main.id }

resource "aws_subnet" "public" {
  count                   = 2
  vpc_id                  = aws_vpc.main.id
  cidr_block              = local.public[count.index]
  availability_zone       = local.azs[count.index]
  map_public_ip_on_launch = false            # only the ALB / NAT get public IPs
  tags = { Name = "public-${local.azs[count.index]}", "kubernetes.io/role/elb" = "1" }
}

resource "aws_subnet" "private" {
  count             = 2
  vpc_id            = aws_vpc.main.id
  cidr_block        = local.private[count.index]
  availability_zone = local.azs[count.index]
  tags = { Name = "private-${local.azs[count.index]}", "kubernetes.io/role/internal-elb" = "1" }
}

resource "aws_eip" "nat" {
  count  = 2
  domain = "vpc"
}

resource "aws_nat_gateway" "nat" {            # one per AZ
  count         = 2
  allocation_id = aws_eip.nat[count.index].id
  subnet_id     = aws_subnet.public[count.index].id
}

resource "aws_route_table" "public" {
  vpc_id = aws_vpc.main.id
  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.igw.id   # this line makes it public
  }
}

resource "aws_route_table" "private" {         # one per AZ -> its own NAT
  count  = 2
  vpc_id = aws_vpc.main.id
  route {
    cidr_block     = "0.0.0.0/0"
    nat_gateway_id = aws_nat_gateway.nat[count.index].id
  }
}

resource "aws_route_table_association" "public" {
  count          = 2
  subnet_id      = aws_subnet.public[count.index].id
  route_table_id = aws_route_table.public.id
}

resource "aws_route_table_association" "private" {
  count          = 2
  subnet_id      = aws_subnet.private[count.index].id
  route_table_id = aws_route_table.private[count.index].id
}

resource "aws_vpc_endpoint" "s3" {             # free gateway endpoint
  vpc_id            = aws_vpc.main.id
  service_name      = "com.amazonaws.ap-south-1.s3"
  vpc_endpoint_type = "Gateway"
  route_table_ids   = aws_route_table.private[*].id
}
$ terraform plan
...
Plan: 17 to add, 0 to change, 0 to destroy.

$ terraform apply -auto-approve
...
Apply complete! Resources: 17 added, 0 changed, 0 destroyed.

$ aws ec2 describe-subnets --filters Name=vpc-id,Values=vpc-0f1e2d3c \
    --query 'Subnets[*].[CidrBlock,AvailabilityZone,AvailableIpAddressCount]' --output text
10.10.0.0/20    ap-south-1a   4090
10.10.16.0/20   ap-south-1b   4090
10.10.64.0/19   ap-south-1a   8187
10.10.96.0/19   ap-south-1b   8187

Notice the public /20s show 4,090, not 4,091: each NAT gateway ENI already took one address (load balancer nodes will take more later). The private /19s show 8,192 - 5 = 8,187. Always check AvailableIpAddressCount when debugging "pods stuck in ContainerCreating: failed to assign an IP address".
