Data model (be complete: interviewers check attributes)

The Amazon interviewer in the 2026-08 post was strict about missing attributes, so list them. Relational (PostgreSQL / MySQL / Aurora): the data is relational, the volume is modest, and we need transactions and unique constraints. Store choice reasoning is in Choosing Storage.

CREATE TABLE plan (
  plan_id        BIGINT PRIMARY KEY,
  merchant_id    BIGINT NOT NULL,
  product_id     TEXT,                    -- NULL for pure service plans
  quantity       INT    NOT NULL DEFAULT 1,
  interval_unit  TEXT   NOT NULL,         -- DAY, WEEK, MONTH, YEAR
  interval_count INT    NOT NULL DEFAULT 1,
  anchor         JSONB,                   -- {"weekday":"MON"} or {"monthDay":1}
  price_amount   BIGINT NOT NULL,         -- minor units (paise)
  currency       CHAR(3) NOT NULL,
  trial_cycles   INT    NOT NULL DEFAULT 0,
  max_cycles     INT,                     -- NULL = until cancelled (12 = one year monthly)
  active         BOOLEAN NOT NULL DEFAULT TRUE,
  created_at     TIMESTAMPTZ NOT NULL
);

CREATE TABLE subscription (
  subscription_id BIGINT PRIMARY KEY,
  customer_id     BIGINT NOT NULL,
  plan_id         BIGINT NOT NULL REFERENCES plan,
  status          TEXT   NOT NULL,     -- PENDING_AUTH, ACTIVE, PAUSED, PAST_DUE, CANCELLED, COMPLETED, EXPIRED
  quantity        INT    NOT NULL,     -- may differ from plan default
  price_amount    BIGINT NOT NULL,     -- price locked at subscribe time (grandfathering)
  start_date      DATE   NOT NULL,
  end_date        DATE,
  timezone        TEXT   NOT NULL,     -- 'Asia/Kolkata': "every Monday" is local time
  address_id      BIGINT,
  delivery_slot   TEXT,
  mandate_id      BIGINT,
  cycles_done     INT    NOT NULL DEFAULT 0,
  next_cycle_no   INT    NOT NULL DEFAULT 1,
  next_cycle_at   TIMESTAMPTZ,         -- when the scheduler must act (cut-off or pre-debit time)
  cancel_at_period_end BOOLEAN NOT NULL DEFAULT FALSE,
  version         INT    NOT NULL DEFAULT 0,   -- optimistic locking
  created_at TIMESTAMPTZ NOT NULL, updated_at TIMESTAMPTZ NOT NULL
);
CREATE INDEX sub_due ON subscription (next_cycle_at) WHERE status = 'ACTIVE';

CREATE TABLE cycle (
  subscription_id BIGINT NOT NULL,
  cycle_no        INT    NOT NULL,
  due_date        DATE   NOT NULL,     -- delivery or billing date
  state           TEXT   NOT NULL,     -- SCHEDULED, NOTIFIED, ORDERED, CHARGING, PAID, FAILED, SKIPPED, REFUNDED
  amount          BIGINT NOT NULL,
  order_id        BIGINT,
  payment_ref     TEXT,
  attempts        INT    NOT NULL DEFAULT 0,
  next_retry_at   TIMESTAMPTZ,
  PRIMARY KEY (subscription_id, cycle_no)      -- the idempotency anchor
);

CREATE TABLE mandate (
  mandate_id   BIGINT PRIMARY KEY,
  customer_id  BIGINT NOT NULL,
  method       TEXT   NOT NULL,        -- CARD_EMANDATE, UPI_AUTOPAY, ENACH
  gateway      TEXT   NOT NULL,
  gateway_token TEXT  NOT NULL,        -- token or mandate reference, never a raw card number
  max_amount   BIGINT NOT NULL,
  frequency    TEXT,                   -- as registered with the bank
  status       TEXT   NOT NULL,        -- PENDING, ACTIVE, PAUSED, REVOKED, EXPIRED
  valid_until  DATE
);

CREATE TABLE schedule_change (
  change_id BIGINT PRIMARY KEY, subscription_id BIGINT NOT NULL,
  kind TEXT NOT NULL,                  -- PAUSE, SKIP, QTY, PLAN, ADDRESS
  effective_from DATE NOT NULL, effective_to DATE, payload JSONB,
  created_by TEXT NOT NULL, created_at TIMESTAMPTZ NOT NULL
);

CREATE TABLE payment_attempt (
  subscription_id BIGINT, cycle_no INT, attempt_no INT,
  idempotency_key TEXT UNIQUE,         -- 'sub_881:5:2' sent to the gateway
  status TEXT, decline_code TEXT, gateway_ref TEXT, created_at TIMESTAMPTZ,
  PRIMARY KEY (subscription_id, cycle_no, attempt_no)
);

Attributes that interviewers notice when missing: timezone (Monday where?), price locked at subscribe time (plan price changes should not silently change existing subscribers), money in minor units (paise, never floats), max_cycles / end_date (a one-year plan), cancel_at_period_end, mandate max amount and expiry, version for concurrency, and created_by on changes for audit.
