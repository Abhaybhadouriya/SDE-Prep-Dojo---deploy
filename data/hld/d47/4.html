-- Cassandra-style messages table
CREATE TABLE messages (
  conv_id      text,
  message_seq  bigint,          -- monotonically increasing per conversation
  message_id   text,
  role         text,            -- 'user' | 'assistant' | 'system'
  content      text,
  status       text,            -- 'streaming' | 'complete' | 'cancelled' | 'error'
  input_tokens int,
  output_tokens int,
  model        text,
  created_at   timestamp,
  PRIMARY KEY ((conv_id), message_seq)
) WITH CLUSTERING ORDER BY (message_seq ASC);

Why NoSQL: the access patterns are simple (by user, by conversation), writes are huge and append-only, and we never join. Postgres is fine for users, plans and billing accounts. See Choosing Storage.

Write the assistant message twice: insert it as streaming when generation starts, then update to complete with the full text and token counts at the end (or flush partial text every few seconds). That way a crash mid-answer leaves a visible, honest "interrupted" message instead of nothing.
