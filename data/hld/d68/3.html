Key APIs

# Which store serves me, and how fast?
GET  /v1/serviceability?lat=12.97&lon=77.64
  -> {store_id: "BLR-114", eta_min: 9, open: true, surge_fee: 0}

# Store-scoped search and browse (store_id from the step above, in a signed token)
GET  /v1/stores/BLR-114/search?q=milk&page_token=...
GET  /v1/stores/BLR-114/categories/dairy
  -> [{sku_id, name, price, mrp, in_stock: true, max_qty: 4}]

# Checkout: reserve stock for the whole cart, then pay
POST /v1/orders           Idempotency-Key: c81f...
  {store_id, lines:[{sku_id, qty}], address_id, payment_method:"UPI"}
  -> 201 {order_id, status:"RESERVED", reserved_until, amount, payment_action}
  -> 409 {unavailable:[{sku_id, available_qty}]}       # partial: let the user adjust

POST /v1/payments/webhook {psp_ref, order_id, status, signature}
GET  /v1/orders/{order_id}            # status + rider location (or SSE stream)
POST /v1/orders/{order_id}/cancel

# Store-side
POST /v1/stores/{store_id}/inbound    {grn_id, lines:[{sku_id, qty, bin}]}
POST /v1/picks/{pick_task_id}/scan    {sku_id, bin, qty}      # picker scans
POST /v1/stores/{store_id}/adjust     {sku_id, delta, reason:"DAMAGE"|"COUNT"}
