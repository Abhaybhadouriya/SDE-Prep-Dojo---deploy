CREATE TABLE alert (
  alert_id    BIGINT PRIMARY KEY,
  user_id     BIGINT NOT NULL,
  symbol      TEXT   NOT NULL,
  type        TEXT   NOT NULL,      -- FROM_CLOSE | DELTA | WINDOW
  direction   TEXT   NOT NULL,      -- UP | DOWN | ANY
  percent     NUMERIC(6,3),
  window_code TEXT,                 -- 1m,5m,1h,1d,1w,1M,1y (WINDOW only)
  base_price  NUMERIC(14,4),        -- price at creation (DELTA)
  status      TEXT NOT NULL DEFAULT 'ACTIVE',   -- ACTIVE | FIRED | EXPIRED | CANCELLED
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  expires_at  TIMESTAMPTZ NOT NULL              -- created_at + 1 month
);
CREATE INDEX alert_active_by_symbol ON alert (symbol) WHERE status = 'ACTIVE';  -- rebuild path only

The index alert_active_by_symbol exists for rebuilds, not ticks. Saying "this index is read when a worker restarts, never per tick" shows the interviewer you have separated the durable copy from the hot path.
