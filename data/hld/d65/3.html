Key APIs

# (a) Setup: pairing and transfer
POST /setup/sessions            {new_device_id, user_token}        -> {session_id, pairing_code, expires_in: 300}
GET  /users/me/devices          -> [{device_id, model, name, online, last_seen, settings_version}]
POST /setup/sessions/{id}/source {old_device_id}                   # server pushes "send settings" to old device
WS   /setup/sessions/{id}/relay  # encrypted bytes from old to new; the server relays, does not store
POST /setup/sessions/{id}/complete {applied_version}

# (b) State sync
PUT  /users/me/state/{key}      {value, base_version, client_op_id} -> {version} | 409 {current}
GET  /users/me/state?since=1287 -> {changes:[...], version: 1301} | 410 Gone (too old, take snapshot)
GET  /users/me/state/snapshot   -> {state:{...}, version: 1301}
MQTT topic users/{user_id}/state   # push channel, devices subscribe after auth

# (c) Backup
POST /backups                   {device_id, snapshot_id, manifest_hash}       -> {upload_session}
POST /chunks/missing            {hashes:[...]}            -> {missing:[...], upload_urls:{hash: presigned_url}}
PUT  {presigned_url}            # chunk bytes straight to object storage
POST /backups/{id}/commit       {manifest}                 # atomic: snapshot becomes visible only now
GET  /backups?device_id=...     -> [{snapshot_id, created_at, size, status}]
POST /restores                  {snapshot_id, target_device_id} -> {restore_id}
  • base_version on state writes gives optimistic locking: AWS IoT Device Shadow works the same way, rejecting an update whose version does not match the latest.
  • /chunks/missing is how incremental upload works: the device asks which hashes the server lacks and uploads only those, using presigned URLs so bytes skip our servers.
  • A backup is invisible until commit: a half-uploaded snapshot can never be restored by mistake.
