Data model

-- Immutable versions: never UPDATE a published version
CREATE TABLE config_version (
  config_key   TEXT,            -- "payments/prod/limits"
  version      BIGINT,          -- 1, 2, 3 ... per key, assigned at commit
  content      JSONB,           -- or a blob ref in object storage if large
  checksum     TEXT,            -- clients verify after download
  schema_ref   TEXT,
  author       TEXT, reason TEXT, created_at TIMESTAMPTZ,
  PRIMARY KEY (config_key, version)
);

-- What each target group should run right now (moves during a rollout)
CREATE TABLE config_pointer (
  config_key   TEXT,
  target       TEXT,            -- "prod/us-east/canary", "prod/us-east/all", ...
  version      BIGINT,
  revision     BIGINT,          -- global, monotonically increasing change number
  PRIMARY KEY (config_key, target)
);

CREATE TABLE deployment (
  deployment_id UUID PRIMARY KEY, config_key TEXT, from_version BIGINT, to_version BIGINT,
  strategy TEXT, state TEXT,     -- PENDING | WAVE_1 | BAKING | WAVE_2 | DONE | ROLLED_BACK
  alarms TEXT[], started_by TEXT, started_at TIMESTAMPTZ
);

CREATE TABLE audit_log (id BIGSERIAL, at TIMESTAMPTZ, actor TEXT, action TEXT,
  config_key TEXT, details JSONB);          -- append-only; shipped to a WORM store

Two ideas make this work. (1) Versions are immutable, so rollback never edits data: it moves a pointer back. (2) A single, global, increasing revision number on every pointer change lets clients and distributors ask "what changed since revision R?" cheaply, exactly like etcd's revision or ZooKeeper's zxid.
