Data model

Cart: a key-value document

// DynamoDB-style item. Partition key = cart owner. One read gets the whole cart.
{
  "pk": "USER#42",               // or "GUEST#c8f1..." for a guest cart token
  "items": { "B0-PHONE-128": { "qty": 1, "addedAt": 1732350000, "priceSeen": 1899900 } },
  "version": 17,                 // optimistic concurrency for multi-device edits
  "ttl": 1740000000              // guest carts expire
}

If a phone and a laptop both edit the cart, use a conditional write on version. If you ever accept conflicting versions (for example from two regions), merge them by keeping the union of items, which is the approach the Dynamo paper describes. The worst case is that a deleted item comes back, and the customer can remove it again. Losing an added item would cost a sale, so the union is the kinder failure.

Checkout: relational tables (the truth)

CREATE TABLE inventory (
  sku TEXT, warehouse_id INT,
  on_hand  INT NOT NULL,             -- physically in the building
  reserved INT NOT NULL DEFAULT 0,   -- promised to orders not yet shipped
  PRIMARY KEY (sku, warehouse_id),
  CHECK (reserved >= 0 AND reserved <= on_hand)
);

CREATE TABLE reservations (
  reservation_id BIGINT PRIMARY KEY, order_id BIGINT, sku TEXT, warehouse_id INT, qty INT,
  status TEXT NOT NULL,              -- HELD, CONFIRMED, RELEASED
  expires_at TIMESTAMPTZ NOT NULL,   -- e.g. now() + 15 min
  UNIQUE (order_id, sku)             -- a retried reserve step cannot double-hold
);

CREATE TABLE orders (
  order_id BIGINT PRIMARY KEY, customer_id BIGINT NOT NULL,
  idempotency_key UUID NOT NULL, request_hash TEXT NOT NULL,
  status TEXT NOT NULL,              -- PENDING, CONFIRMED, PAYMENT_FAILED, CANCELLED, SHIPPED
  total_minor BIGINT NOT NULL, currency CHAR(3) NOT NULL,
  quote_id TEXT NOT NULL, response JSONB,     -- saved response, replayed on retry
  created_at TIMESTAMPTZ DEFAULT now(),
  UNIQUE (customer_id, idempotency_key)
);
-- order states: PENDING -> CONFIRMED -> SHIPPED (happy path)
--               PENDING -> PAYMENT_FAILED (declined; stock released)
--               CONFIRMED -> CANCELLED (customer cancels before shipping; refund)

CREATE TABLE payments (
  payment_id BIGINT PRIMARY KEY, order_id BIGINT UNIQUE,
  provider TEXT, provider_ref TEXT, status TEXT,   -- AUTH_PENDING, AUTHORIZED, CAPTURED, FAILED, UNKNOWN
  amount_minor BIGINT NOT NULL
);

CREATE TABLE outbox (id BIGSERIAL PRIMARY KEY, aggregate_id BIGINT, type TEXT, payload JSONB);
