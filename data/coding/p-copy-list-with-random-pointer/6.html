The weave version in four languages. Each file in coding-code/copy-list-with-random-pointer/ checks that the copy has the same shape, shares no node with the input, and that the input is restored, on the examples and 200 random lists.

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if (!head) return nullptr;
        // Pass 1: put each copy right after its original: A -> A' -> B -> B' -> ...
        for (Node* cur = head; cur; cur = cur->next->next) {
            Node* copy = new Node(cur->val);
            copy->next = cur->next;
            cur->next = copy;
        }
        // Pass 2: the copy of X->random is X->random->next (it sits right after it)
        for (Node* cur = head; cur; cur = cur->next->next)
            if (cur->random) cur->next->random = cur->random->next;
        // Pass 3: unweave - restore the original list and link the copies together
        Node* newHead = head->next;
        for (Node* cur = head; cur; cur = cur->next) {
            Node* copy = cur->next;
            cur->next = copy->next;                       // original skips its copy
            copy->next = copy->next ? copy->next->next : nullptr;
        }
        return newHead;
    }
};
func copyRandomList(head *Node) *Node {
	if head == nil {
		return nil
	}
	// Pass 1: put each copy right after its original: A -> A' -> B -> B' -> ...
	for cur := head; cur != nil; cur = cur.Next.Next {
		cur.Next = &Node{Val: cur.Val, Next: cur.Next}
	}
	// Pass 2: the copy of X.Random is X.Random.Next (it sits right after it)
	for cur := head; cur != nil; cur = cur.Next.Next {
		if cur.Random != nil {
			cur.Next.Random = cur.Random.Next
		}
	}
	// Pass 3: unweave - restore the original list and link the copies together
	newHead := head.Next
	for cur := head; cur != nil; cur = cur.Next {
		cp := cur.Next
		cur.Next = cp.Next // original skips its copy
		if cp.Next != nil {
			cp.Next = cp.Next.Next
		}
	}
	return newHead
}

class Solution:
    def copyRandomList(self, head: "Optional[Node]") -> "Optional[Node]":
        if not head:
            return None
        # Pass 1: put each copy right after its original: A -> A' -> B -> B' -> ...
        cur = head
        while cur:
            cur.next = Node(cur.val, cur.next)
            cur = cur.next.next
        # Pass 2: the copy of X.random is X.random.next (it sits right after it)
        cur = head
        while cur:
            if cur.random:
                cur.next.random = cur.random.next
            cur = cur.next.next
        # Pass 3: unweave - restore the original list and link the copies together
        new_head, cur = head.next, head
        while cur:
            copy = cur.next
            cur.next = copy.next                    # original skips its copy
            copy.next = copy.next.next if copy.next else None
            cur = cur.next
        return new_head
