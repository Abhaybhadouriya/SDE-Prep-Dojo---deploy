Template: the scheduler in 40 lines

This is the same logic the Revise page runs, written as a small program you can keep in your practice repo or adapt for a spreadsheet. grade moves one card on the ladder; due_queue builds today's list, most overdue first. Days are plain integers to keep it short. Writing it once is also a good exercise in itself: it is a sorted queue with a key, like many design questions.

package main

import (
	"fmt"
	"sort"
	"strings"
)

var ladder = []int{1, 3, 7, 14, 30, 60, 120} // gaps in days between re-solves

type Card struct {
	Slug     string
	Interval int
	Due      int
}

// rungOf returns the index of the largest ladder gap <= interval (-1 for a brand-new card).
func rungOf(interval int) int {
	r := -1
	for i, gap := range ladder {
		if interval >= gap {
			r = i
		}
	}
	return r
}

// grade records one re-solve and schedules the next one.
func grade(c *Card, g string, today int) {
	cur := rungOf(c.Interval)
	var r int
	switch g {
	case "again":
		r = 0 // failed: back to tomorrow
	case "hard":
		r = cur // slow: same gap again
		if r < 0 {
			r = 0
		}
	case "good":
		r = cur + 1 // recalled: next rung
	default:
		r = cur + 2 // instant: skip a rung
	}
	if r < 0 {
		r = 0
	}
	if r > len(ladder)-1 {
		r = len(ladder) - 1
	}
	c.Interval = ladder[r]
	c.Due = today + c.Interval
}

// dueQueue returns cards due today or earlier, most overdue first (ties by name).
func dueQueue(cards []Card, today int) []Card {
	var q []Card
	for _, c := range cards {
		if c.Due <= today {
			q = append(q, c)
		}
	}
	sort.Slice(q, func(i, j int) bool {
		if q[i].Due != q[j].Due {
			return q[i].Due < q[j].Due
		}
		return q[i].Slug < q[j].Slug
	})
	return q
}

func main() {
	card := Card{"two-sum", 1, 1} // solved on day 0
	days := []int{1, 4, 11, 18, 48}
	grades := []string{"good", "good", "hard", "easy", "again"}
	for i, day := range days {
		grade(&card, grades[i], day)
		fmt.Printf("day %2d: %-5s -> next gap %2d, due day %d\n", day, grades[i], card.Interval, card.Due)
	}
	deck := []Card{{"3sum", 3, 9}, {"lru-cache", 7, 5}, {"coin-change", 1, 12}, {"two-sum", 1, 9}}
	var names []string
	for _, c := range dueQueue(deck, 10) {
		names = append(names, c.Slug)
	}
	fmt.Println("due on day 10:", strings.Join(names, ", "))
}
#include <bits/stdc++.h>
using namespace std;

const vector<int> LADDER = {1, 3, 7, 14, 30, 60, 120};  // gaps in days between re-solves

struct Card { string slug; int interval; int due; };

// Index of the largest ladder gap <= interval (-1 for a brand-new card).
int rungOf(int interval) {
    int r = -1;
    for (int i = 0; i < (int)LADDER.size(); i++) if (interval >= LADDER[i]) r = i;
    return r;
}

// Grade one re-solve and schedule the next one.
void grade(Card& c, const string& g, int today) {
    int cur = rungOf(c.interval), r;
    if (g == "again") r = 0;              // failed: back to tomorrow
    else if (g == "hard") r = max(cur, 0); // slow: same gap again
    else if (g == "good") r = cur + 1;     // recalled: next rung
    else r = cur + 2;                      // instant: skip a rung
    r = min(max(r, 0), (int)LADDER.size() - 1);
    c.interval = LADDER[r];
    c.due = today + c.interval;
}

// Cards due today or earlier, most overdue first (ties by name).
vector<Card> dueQueue(const vector<Card>& cards, int today) {
    vector<Card> q;
    for (const Card& c : cards) if (c.due <= today) q.push_back(c);
    sort(q.begin(), q.end(), [](const Card& a, const Card& b) {
        return a.due != b.due ? a.due < b.due : a.slug < b.slug;
    });
    return q;
}

int main() {
    Card card{"two-sum", 1, 1};  // solved on day 0
    vector<pair<int, string>> plan = {{1, "good"}, {4, "good"}, {11, "hard"}, {18, "easy"}, {48, "again"}};
    for (auto& p : plan) {
        grade(card, p.second, p.first);
        printf("day %2d: %-5s -> next gap %2d, due day %d\n", p.first, p.second.c_str(), card.interval, card.due);
    }
    vector<Card> deck = {{"3sum", 3, 9}, {"lru-cache", 7, 5}, {"coin-change", 1, 12}, {"two-sum", 1, 9}};
    vector<Card> q = dueQueue(deck, 10);
    printf("due on day 10:");
    for (size_t i = 0; i < q.size(); i++) printf("%s %s", i ? "," : "", q[i].slug.c_str());
    printf("\n");
    return 0;
}
import java.util.*;

public class Templates {
    static final int[] LADDER = {1, 3, 7, 14, 30, 60, 120}; // gaps in days between re-solves

    static class Card {
        String slug; int interval; int due;
        Card(String s, int i, int d) { slug = s; interval = i; due = d; }
    }

    // Index of the largest ladder gap <= interval (-1 for a brand-new card).
    static int rungOf(int interval) {
        int r = -1;
        for (int i = 0; i < LADDER.length; i++) if (interval >= LADDER[i]) r = i;
        return r;
    }

    // Grade one re-solve and schedule the next one.
    static void grade(Card c, String g, int today) {
        int cur = rungOf(c.interval), r;
        switch (g) {
            case "again": r = 0; break;               // failed: back to tomorrow
            case "hard":  r = Math.max(cur, 0); break; // slow: same gap again
            case "good":  r = cur + 1; break;          // recalled: next rung
            default:      r = cur + 2;                 // instant: skip a rung
        }
        r = Math.min(Math.max(r, 0), LADDER.length - 1);
        c.interval = LADDER[r];
        c.due = today + c.interval;
    }

    // Cards due today or earlier, most overdue first (ties by name).
    static List<Card> dueQueue(List<Card> cards, int today) {
        List<Card> q = new ArrayList<>();
        for (Card c : cards) if (c.due <= today) q.add(c);
        q.sort(Comparator.comparingInt((Card c) -> c.due).thenComparing(c -> c.slug));
        return q;
    }

    public static void main(String[] args) {
        Card card = new Card("two-sum", 1, 1); // solved on day 0
        int[] days = {1, 4, 11, 18, 48};
        String[] grades = {"good", "good", "hard", "easy", "again"};
        for (int i = 0; i < days.length; i++) {
            grade(card, grades[i], days[i]);
            System.out.printf("day %2d: %-5s -> next gap %2d, due day %d%n", days[i], grades[i], card.interval, card.due);
        }
        List<Card> deck = List.of(new Card("3sum", 3, 9), new Card("lru-cache", 7, 5),
                new Card("coin-change", 1, 12), new Card("two-sum", 1, 9));
        StringJoiner sj = new StringJoiner(", ");
        for (Card c : dueQueue(deck, 10)) sj.add(c.slug);
        System.out.println("due on day 10: " + sj);
    }
}
LADDER = [1, 3, 7, 14, 30, 60, 120]          # gaps in days between re-solves


def rung_of(interval):
    """Index of the largest ladder gap <= interval (-1 for a brand-new card)."""
    r = -1
    for i, gap in enumerate(LADDER):
        if interval >= gap:
            r = i
    return r


def grade(card, g, today):
    """Grade one re-solve and schedule the next. card = {slug, interval, due}."""
    cur = rung_of(card["interval"])
    if g == "again":
        r = 0                  # failed: back to tomorrow
    elif g == "hard":
        r = max(cur, 0)        # slow: same gap again
    elif g == "good":
        r = cur + 1            # recalled: next rung
    else:
        r = cur + 2            # instant: skip a rung
    r = min(max(r, 0), len(LADDER) - 1)
    card["interval"] = LADDER[r]
    card["due"] = today + card["interval"]
    return card


def due_queue(cards, today):
    """Cards due today or earlier, most overdue first (ties by name)."""
    q = [c for c in cards if c["due"] <= today]
    return sorted(q, key=lambda c: (c["due"], c["slug"]))


def main():
    card = {"slug": "two-sum", "interval": 1, "due": 1}   # solved on day 0

        grade(card, g, day)
        print(f"day {day:>2}: {g:<5} -> next gap {card['interval']:>2}, due day {card['due']}")

            {"slug": "lru-cache", "interval": 7, "due": 5},
            {"slug": "coin-change", "interval": 1, "due": 12},
            {"slug": "two-sum", "interval": 1, "due": 9}]
    print("due on day 10:", ", ".join(c["slug"] for c in due_queue(deck, 10)))

day  1: good  -> next gap  3, due day 4
day  4: good  -> next gap  7, due day 11
day 11: hard  -> next gap  7, due day 18
day 18: easy  -> next gap 30, due day 48
day 48: again -> next gap  1, due day 49
due on day 10: lru-cache, 3sum, two-sum

Design note: a card stores only interval and due. The rung is recomputed from the interval, so old progress with odd intervals (say 10 days) still lands on a sensible rung (7). With n cards, building the queue is O(n log n); a min-heap keyed by due would make "next card" O(log n) if you ever had a huge deck.
