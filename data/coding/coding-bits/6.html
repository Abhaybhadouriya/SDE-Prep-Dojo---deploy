Templates in four languages

One file per language with every idiom from this page, each as a small function with a test in main. Read the comments: most of them are the language traps from the section above.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

// Bit manipulation templates (coding-bits lesson).
package main

import (
	"fmt"
	"math/bits"
	"reflect"
)

func singleNumber(nums []int) int { // every value twice except one
	x := 0
	for _, v := range nums {
		x ^= v
	}
	return x
}

func singleNumberII(nums []int) int { // every value three times except one
	ones, twos := 0, 0
	for _, x := range nums {
		ones = (ones ^ x) &^ twos // bits seen 1 (mod 3) times; &^ is AND NOT
		twos = (twos ^ x) &^ ones // bits seen 2 (mod 3) times
	}
	return ones
}

func singleNumberIII(nums []int) []int { // two values once, rest twice
	x := 0
	for _, v := range nums {
		x ^= v
	}
	low := x & -x // lowest differing bit (Go ints wrap, no UB)
	a, b := 0, 0
	for _, v := range nums {
		if v&low != 0 {
			a ^= v
		} else {
			b ^= v
		}
	}
	if a > b {
		a, b = b, a
	}
	return []int{a, b}
}

func popcount(n uint32) int { // Kernighan (or bits.OnesCount32(n))
	c := 0
	for n != 0 {
		n &= n - 1
		c++
	}
	return c
}

func isPowerOfTwo(n int) bool { return n > 0 && n&(n-1) == 0 }

func countingBits(n int) []int {
	res := make([]int, n+1)
	for i := 1; i <= n; i++ {
		res[i] = res[i>>1] + i&1
	}
	return res
}

func reverseBits(n uint32) uint32 {
	var r uint32
	for i := 0; i < 32; i++ {
		r = r<<1 | n&1
		n >>= 1
	}
	return r
}

func getSum(a, b int32) int32 { // add without +
	for b != 0 {
		carry := (a & b) << 1
		a ^= b
		b = carry
	}
	return a
}

func subsets(items []int) [][]int { // mask = which items are in the set
	n := len(items)
	out := [][]int{}
	for mask := 0; mask < 1<<n; mask++ {
		s := []int{}
		for i := 0; i < n; i++ {
			if mask>>i&1 == 1 {
				s = append(s, items[i])
			}
		}
		out = append(out, s)
	}
	return out
}

func submasks(m int) []int { // every non-empty submask of m
	out := []int{}
	for s := m; s != 0; s = (s - 1) & m {
		out = append(out, s)
	}
	return out
}

func must(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

func main() {
	must(singleNumber([]int{7, 3, 7}) == 3, "sn1")
	must(singleNumberII([]int{-2, -2, 1, 1, -3, 1, -2}) == -3, "sn2")
	must(reflect.DeepEqual(singleNumberIII([]int{1, 2, 1, 3, 2, 5}), []int{3, 5}), "sn3")
	must(popcount(11) == 3 && popcount(0xFFFFFFFF) == 32 && bits.OnesCount32(11) == 3, "pop")
	must(isPowerOfTwo(64) && !isPowerOfTwo(0) && !isPowerOfTwo(6), "pow2")
	must(reflect.DeepEqual(countingBits(5), []int{0, 1, 1, 2, 1, 2}), "cb")
	must(reverseBits(43261596) == 964176192, "rev")
	must(getSum(2, 3) == 5 && getSum(-7, 3) == -4 && getSum(-5, -6) == -11, "sum")
	must(len(subsets([]int{1, 2})) == 4, "subsets")
	must(reflect.DeepEqual(submasks(5), []int{5, 4, 1}), "submasks")
	fmt.Println("bits.go OK")
}
// Bit manipulation templates (coding-bits lesson). C++14: g++ -std=c++14 -O2 bits.cpp
#include <bits/stdc++.h>
using namespace std;

int singleNumber(const vector<int>& nums) {          // every value twice except one
    int x = 0;
    for (int v : nums) x ^= v;
    return x;
}

int singleNumberII(const vector<int>& nums) {        // every value three times except one
    int ones = 0, twos = 0;
    for (int x : nums) {
        ones = (ones ^ x) & ~twos;                   // bits seen 1 (mod 3) times
        twos = (twos ^ x) & ~ones;                   // bits seen 2 (mod 3) times
    }
    return ones;
}

vector<int> singleNumberIII(const vector<int>& nums) {   // two values once, rest twice
    unsigned x = 0;
    for (int v : nums) x ^= (unsigned)v;
    unsigned low = x & (0u - x);                     // unsigned: -INT_MIN would be UB on int
    int a = 0, b = 0;
    for (int v : nums) {
        if ((unsigned)v & low) a ^= v;               // group with that bit set
        else b ^= v;                                 // group without it
    }
    if (a > b) swap(a, b);
    return {a, b};
}

int popcount(uint32_t n) {                           // Kernighan: one loop per set bit
    int c = 0;
    while (n) { n &= n - 1; ++c; }                   // clear the lowest set bit
    return c;                                        // (or __builtin_popcount(n))
}

bool isPowerOfTwo(long long n) { return n > 0 && (n & (n - 1)) == 0; }

vector<int> countingBits(int n) {
    vector<int> bits(n + 1, 0);
    for (int i = 1; i <= n; ++i) bits[i] = bits[i >> 1] + (i & 1);
    return bits;
}

uint32_t reverseBits(uint32_t n) {
    uint32_t r = 0;
    for (int i = 0; i < 32; ++i) { r = (r << 1) | (n & 1u); n >>= 1; }
    return r;
}

int getSum(int a, int b) {                           // add without +
    unsigned ua = a, ub = b;                         // unsigned: shifting a negative left is UB
    while (ub) {
        unsigned carry = (ua & ub) << 1;
        ua ^= ub;
        ub = carry;
    }
    return (int)ua;
}

vector<vector<int>> subsets(const vector<int>& items) {  // mask = which items are in the set
    int n = items.size();
    vector<vector<int>> out;
    for (int mask = 0; mask < (1 << n); ++mask) {
        vector<int> s;
        for (int i = 0; i < n; ++i) if (mask >> i & 1) s.push_back(items[i]);
        out.push_back(s);
    }
    return out;
}

vector<int> submasks(int m) {                        // every non-empty submask of m
    vector<int> out;
    for (int s = m; s; s = (s - 1) & m) out.push_back(s);
    return out;
}

int main() {
    assert(singleNumber({7, 3, 7}) == 3);
    assert(singleNumber({4, 1, 2, 1, 2}) == 4);
    assert(singleNumberII({5, 5, 9, 5}) == 9);
    assert(singleNumberII({-2, -2, 1, 1, -3, 1, -2}) == -3);
    assert((singleNumberIII({1, 2, 1, 3, 2, 5}) == vector<int>{3, 5}));
    assert((singleNumberIII({-1, 0}) == vector<int>{-1, 0}));
    assert((singleNumberIII({INT_MIN, 7}) == vector<int>{INT_MIN, 7}));
    assert(popcount(11) == 3 && popcount(0xFFFFFFFFu) == 32 && popcount(0) == 0);
    assert(isPowerOfTwo(64) && !isPowerOfTwo(0) && !isPowerOfTwo(-8) && !isPowerOfTwo(6));
    assert((countingBits(5) == vector<int>{0, 1, 1, 2, 1, 2}));
    assert(reverseBits(43261596u) == 964176192u && reverseBits(1u) == 2147483648u);
    assert(getSum(2, 3) == 5 && getSum(-7, 3) == -4 && getSum(-1, 1) == 0 && getSum(-5, -6) == -11);
    assert(subsets({1, 2}).size() == 4 && subsets({1, 2})[3] == (vector<int>{1, 2}));
    assert((submasks(5) == vector<int>{5, 4, 1}));
    puts("bits.cpp OK");
}
// Bit manipulation templates (coding-bits lesson). Java 21: java Bits.java
import java.util.*;

public class Bits {
    static int singleNumber(int[] nums) {            // every value twice except one
        int x = 0;
        for (int v : nums) x ^= v;
        return x;
    }

    static int singleNumberII(int[] nums) {          // every value three times except one
        int ones = 0, twos = 0;
        for (int x : nums) {
            ones = (ones ^ x) & ~twos;               // bits seen 1 (mod 3) times
            twos = (twos ^ x) & ~ones;               // bits seen 2 (mod 3) times
        }
        return ones;
    }

    static int[] singleNumberIII(int[] nums) {       // two values once, rest twice
        int x = 0;
        for (int v : nums) x ^= v;
        int low = x & -x;                            // safe in Java: ints wrap
        int a = 0, b = 0;
        for (int v : nums) {
            if ((v & low) != 0) a ^= v; else b ^= v;
        }
        return a < b ? new int[]{a, b} : new int[]{b, a};
    }

    static int popcount(int n) {                     // Kernighan (or Integer.bitCount(n))
        int c = 0;
        while (n != 0) { n &= n - 1; c++; }
        return c;
    }

    static boolean isPowerOfTwo(int n) { return n > 0 && (n & (n - 1)) == 0; }

    static int[] countingBits(int n) {
        int[] bits = new int[n + 1];
        for (int i = 1; i <= n; i++) bits[i] = bits[i >> 1] + (i & 1);
        return bits;
    }

    static int reverseBits(int n) {                  // treat n as unsigned 32-bit
        int r = 0;
        for (int i = 0; i < 32; i++) { r = (r << 1) | (n & 1); n >>>= 1; }   // >>> not >>
        return r;
    }

    static int getSum(int a, int b) {                // add without +
        while (b != 0) {
            int carry = (a & b) << 1;
            a ^= b;
            b = carry;
        }
        return a;
    }

    static List<List<Integer>> subsets(int[] items) { // mask = which items are in the set
        int n = items.length;
        List<List<Integer>> out = new ArrayList<>();
        for (int mask = 0; mask < (1 << n); mask++) {
            List<Integer> s = new ArrayList<>();
            for (int i = 0; i < n; i++) if ((mask >> i & 1) == 1) s.add(items[i]);
            out.add(s);
        }
        return out;
    }

    static List<Integer> submasks(int m) {           // every non-empty submask of m
        List<Integer> out = new ArrayList<>();
        for (int s = m; s != 0; s = (s - 1) & m) out.add(s);
        return out;
    }

    static void check(boolean ok, String what) { if (!ok) throw new AssertionError(what); }

    public static void main(String[] args) {
        check(singleNumber(new int[]{7, 3, 7}) == 3, "sn1");
        check(singleNumberII(new int[]{-2, -2, 1, 1, -3, 1, -2}) == -3, "sn2");
        check(Arrays.equals(singleNumberIII(new int[]{1, 2, 1, 3, 2, 5}), new int[]{3, 5}), "sn3");
        check(Arrays.equals(singleNumberIII(new int[]{Integer.MIN_VALUE, 7}), new int[]{Integer.MIN_VALUE, 7}), "sn3 min");
        check(popcount(11) == 3 && popcount(-1) == 32, "pop");
        check(isPowerOfTwo(64) && !isPowerOfTwo(0) && !isPowerOfTwo(Integer.MIN_VALUE), "pow2");
        check(Arrays.equals(countingBits(5), new int[]{0, 1, 1, 2, 1, 2}), "cb");
        check(reverseBits(43261596) == 964176192 && reverseBits(1) == Integer.MIN_VALUE, "rev");
        check(getSum(2, 3) == 5 && getSum(-7, 3) == -4 && getSum(-5, -6) == -11, "sum");
        check(subsets(new int[]{1, 2}).toString().equals("[[], [1], [2], [1, 2]]"), "subsets");
        check(submasks(5).equals(List.of(5, 4, 1)), "submasks");
        System.out.println("Bits.java OK");
    }
}
# Bit manipulation templates (coding-bits lesson). Run: python bits.py
from functools import reduce


def single_number(nums):            # every value twice except one
    return reduce(lambda a, b: a ^ b, nums, 0)


def single_number_ii(nums):         # every value three times except one
    ones = twos = 0
    for x in nums:
        ones = (ones ^ x) & ~twos   # bits seen 1 (mod 3) times
        twos = (twos ^ x) & ~ones   # bits seen 2 (mod 3) times
    return ones


def single_number_iii(nums):        # two values once, rest twice
    x = 0
    for v in nums:
        x ^= v                      # x = a ^ b, non-zero
    low = x & -x                    # lowest bit where a and b differ
    a = 0
    for v in nums:
        if v & low:
            a ^= v
    return sorted([a, x ^ a])


def popcount(n):                    # Kernighan: one loop per set bit
    n &= 0xFFFFFFFF                 # Python ints are unbounded: pick a width
    c = 0
    while n:
        n &= n - 1                  # clear the lowest set bit
        c += 1
    return c


def is_power_of_two(n):
    return n > 0 and n & (n - 1) == 0


def counting_bits(n):               # bits[i] = bits[i >> 1] + (i & 1)
    bits = [0] * (n + 1)
    for i in range(1, n + 1):
        bits[i] = bits[i >> 1] + (i & 1)
    return bits


def reverse_bits(n):                # 32-bit unsigned
    r = 0
    for _ in range(32):
        r = (r << 1) | (n & 1)
        n >>= 1
    return r


def get_sum(a, b):                  # add without + (32-bit two's complement)
    M = 0xFFFFFFFF
    a &= M
    b &= M
    while b:
        carry = ((a & b) << 1) & M
        a = a ^ b
        b = carry
    return a if a <= 0x7FFFFFFF else ~(a ^ M)


def subsets(items):                 # mask = which items are in the set
    n = len(items)
    return [[items[i] for i in range(n) if mask >> i & 1] for mask in range(1 << n)]


def submasks(m):                    # every non-empty submask of m, largest first
    out, s = [], m
    while s:
        out.append(s)
        s = (s - 1) & m
    return out
