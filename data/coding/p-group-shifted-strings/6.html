All four shift each word to start at 'a' and return the groups in first-seen order (any order is accepted).

class Solution {
public:
    vector<vector<string>> groupStrings(vector<string>& strings) {
        unordered_map<string, int> groupOf;          // canonical key -> index into groups
        vector<vector<string>> groups;
        for (const string& w : strings) {
            string key = w;
            int shift = w[0] - 'a';                  // shift every letter back by this much
            for (char& c : key)
                c = 'a' + (c - 'a' - shift + 26) % 26;   // +26 keeps the value non-negative (wrap z -> a)
            auto it = groupOf.find(key);
            if (it == groupOf.end()) {               // first word of this shifting sequence
                groupOf.emplace(key, (int)groups.size());
                groups.push_back({w});
            } else {
                groups[it->second].push_back(w);
            }
        }
        return groups;                               // groups in order of first appearance
    }
};
func groupStrings(words []string) [][]string {
	groupOf := make(map[string]int) // canonical key -> index into groups
	var groups [][]string
	for _, w := range words {
		key := []byte(w)
		shift := w[0] - 'a' // shift every letter back by this much
		for i := range key {
			key[i] = 'a' + (key[i]-'a'+26-shift)%26 // +26 keeps the value non-negative (wrap z -> a)
		}
		if idx, ok := groupOf[string(key)]; ok {
			groups[idx] = append(groups[idx], w)
		} else { // first word of this shifting sequence
			groupOf[string(key)] = len(groups)
			groups = append(groups, []string{w})
		}
	}
	return groups // groups in order of first appearance
}

static class Solution {
    public List<List<String>> groupStrings(String[] strings) {
        Map<String, List<String>> groups = new HashMap<>();   // canonical key -> its words
        List<List<String>> result = new ArrayList<>();         // groups in order of first appearance
        for (String w : strings) {
            char[] key = w.toCharArray();
            int shift = w.charAt(0) - 'a';                     // shift every letter back by this much
            for (int i = 0; i < key.length; i++)
                key[i] = (char) ('a' + (key[i] - 'a' - shift + 26) % 26);   // +26: wrap z -> a
            String k = new String(key);
            List<String> g = groups.get(k);
            if (g == null) {                                   // first word of this shifting sequence
                g = new ArrayList<>();
                groups.put(k, g);
                result.add(g);
            }
            g.add(w);
        }
        return result;
    }
}
class Solution:
    def groupStrings(self, strings: List[str]) -> List[List[str]]:
        groups = defaultdict(list)                 # canonical key -> words (dict keeps first-seen order)
        for w in strings:
            shift = ord(w[0])                      # shift every letter back so that w[0] becomes 'a'
            key = tuple((ord(c) - shift) % 26 for c in w)   # Python's % is never negative: wraps z -> a
            groups[key].append(w)
        return list(groups.values())
