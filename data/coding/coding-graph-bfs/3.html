State BFS: when a node is more than a cell

Sometimes "where am I" is not enough to know what you can do next. If you can break up to k walls, then being at cell (r, c) with 3 breaks left is a different situation from being there with 0 left. The fix is to make the BFS node the full state: everything that affects future moves. BFS does not care what a state looks like, only that you can list its neighbours and remember which states you have seen.

Always multiply out the state count before coding. It is your time and memory bound, and it tells you whether BFS is even feasible (n x 2n with n = 12 is about 50,000: fine).

Open the Lock: a graph you never build

LeetCode 752: four wheels, each showing 0 to 9, start at "0000". One move turns one wheel by one step, either way, wrapping around. Some combinations are dead ends that lock the device; find the fewest moves to reach the target, or -1. Here the state is the whole string, each state has 8 neighbours, and the dead ends are simply put into the visited set before the search starts:

Check "0000" itself against the dead ends first (if it is one, the answer is -1), and remember that the answer for target "0000" is 0.

Obstacles elimination: keep the best version of each cell

LeetCode 1293: walk from the top-left to the bottom-right of a 0/1 grid in the fewest steps, allowed to walk through at most k walls. The state is (r, c, breaks left). A plain 3-D visited array of size R x C x (k + 1) works. A smaller trick: for each cell, remember only the most breaks left you ever arrived with. Arriving later (more steps, because BFS goes level by level) with fewer or equal breaks can never do better, so skip it. Also, if k >= R + C - 2 you can bulldoze the straight Manhattan path, so return R + C - 2 at once.

// shortestPath: state BFS where the node is (row, col, removals left).
// 0 free, 1 wall; you may break at most k walls.
func shortestPath(g [][]int, k int) int {
	R, C := len(g), len(g[0])
	if k >= R+C-2 { // enough to walk straight through
		return R + C - 2
	}
	// seen[r][c] = most removals left we ever arrived here with; fewer is never better.
	seen := make([][]int, R)
	for i := range seen {
		seen[i] = make([]int, C)
		for j := range seen[i] {
			seen[i][j] = -1
		}
	}
	seen[0][0] = k
	q := [][3]int{{0, 0, k}}
	for steps := 0; len(q) > 0; steps++ {
		next := [][3]int{}
		for _, s := range q {
			if s[0] == R-1 && s[1] == C-1 {
				return steps
			}
			for _, d := range dirs {
				nr, nc := s[0]+d[0], s[1]+d[1]
				if nr < 0 || nr >= R || nc < 0 || nc >= C {
					continue
				}
				nl := s[2] - g[nr][nc] // a wall costs one removal
				if nl < 0 || nl <= seen[nr][nc] {
					continue
				}
				seen[nr][nc] = nl
				next = append(next, [3]int{nr, nc, nl})
			}
		}
		q = next
	}
	return -1
}

// Shortest Path in a Grid with Obstacles Elimination (0 free, 1 wall, may break k walls).
int shortestPath(const vector<vector<int>>& g, int k) {
    int R = g.size(), C = g[0].size();
    if (k >= R + C - 2) return R + C - 2;            // enough to walk straight through
    // seen[r][c] = most removals left we ever arrived here with; fewer is never better.
    vector<vector<int>> seen(R, vector<int>(C, -1));
    queue<array<int, 3>> q;
    q.push({0, 0, k});
    seen[0][0] = k;
    const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    for (int steps = 0; !q.empty(); steps++) {
        for (int n = q.size(); n > 0; n--) {
            auto s = q.front(); q.pop();
            int r = s[0], c = s[1], left = s[2];
            if (r == R - 1 && c == C - 1) return steps;
            for (int d = 0; d < 4; d++) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr < 0 || nr >= R || nc < 0 || nc >= C) continue;
                int nl = left - g[nr][nc];            // a wall costs one removal
                if (nl < 0 || nl <= seen[nr][nc]) continue;
                seen[nr][nc] = nl;
                q.push({nr, nc, nl});
            }
        }
    }
    return -1;
}

// Shortest Path in a Grid with Obstacles Elimination (0 free, 1 wall, may break k walls).
static int shortestPath(int[][] g, int k) {
    int R = g.length, C = g[0].length;
    if (k >= R + C - 2) return R + C - 2;               // enough to walk straight through
    // seen[r][c] = most removals left we ever arrived here with; fewer is never better.
    int[][] seen = new int[R][C];
    for (int[] row : seen) Arrays.fill(row, -1);
    ArrayDeque<int[]> q = new ArrayDeque<>();
    q.add(new int[]{0, 0, k});
    seen[0][0] = k;
    for (int steps = 0; !q.isEmpty(); steps++) {
        for (int n = q.size(); n > 0; n--) {
            int[] s = q.poll();
            if (s[0] == R - 1 && s[1] == C - 1) return steps;
            for (int d = 0; d < 4; d++) {
                int nr = s[0] + DR[d], nc = s[1] + DC[d];
                if (nr < 0 || nr >= R || nc < 0 || nc >= C) continue;

                if (nl < 0 || nl <= seen[nr][nc]) continue;
                seen[nr][nc] = nl;
                q.add(new int[]{nr, nc, nl});
            }
        }
    }
    return -1;
}
def shortest_path(g: List[List[int]], k: int) -> int:
    """State BFS: the node is (row, col, removals left), not just the cell.
    0 free, 1 wall; you may break at most k walls."""
    R, C = len(g), len(g[0])
    if k >= R + C - 2:                           # enough to walk straight through
        return R + C - 2
    # seen[r][c] = most removals left we ever arrived here with; fewer is never better.
    seen = [[-1] * C for _ in range(R)]
    seen[0][0] = k
    q, steps = deque([(0, 0, k)]), 0
    while q:
        for _ in range(len(q)):
            r, c, left = q.popleft()
            if (r, c) == (R - 1, C - 1):
                return steps
            for dr, dc in DIRS:
                nr, nc = r + dr, c + dc
                if 0 <= nr < R and 0 <= nc < C:
                    nl = left - g[nr][nc]        # a wall costs one removal
                    if nl > seen[nr][nc]:        # also rejects nl < 0 (seen >= -1)
                        seen[nr][nc] = nl
                        q.append((nr, nc, nl))
        steps += 1
    return -1

Interviewers often ask why plain "visited[r][c]" fails here. Answer with a picture: the cheap route to a cell may have used up your breaks, and a longer route that kept a break is the only one that can get through the wall further on. Two states, one cell.
