class Solution {
public:
    bool isBalanced(TreeNode* root) {
        return height(root) >= 0;
    }

    // Same check with an explicit post-order stack (no recursion depth limit).
    bool isBalancedIterative(TreeNode* root) {
        unordered_map<TreeNode*, int> h;            // node -> height; nullptr counts as 0
        h[nullptr] = 0;
        vector<pair<TreeNode*, bool>> st;
        if (root) st.push_back({root, false});
        while (!st.empty()) {
            TreeNode* node = st.back().first;
            bool childrenDone = st.back().second;
            st.pop_back();
            if (childrenDone) {
                int lh = h[node->left], rh = h[node->right];
                if (abs(lh - rh) > 1) return false;
                h[node] = 1 + max(lh, rh);
            } else {
                st.push_back({node, true});         // come back after both children
                if (node->right) st.push_back({node->right, false});
                if (node->left) st.push_back({node->left, false});
            }
        }
        return true;
    }

private:
    // Height of the subtree, or -1 if some node inside it is unbalanced.
    int height(TreeNode* node) {
        if (node == nullptr) return 0;
        int lh = height(node->left);
        if (lh < 0) return -1;                      // left side already failed: stop early
        int rh = height(node->right);
        if (rh < 0 || abs(lh - rh) > 1) return -1;
        return 1 + max(lh, rh);
    }
};

// height returns the subtree height, or -1 if some node inside it is unbalanced.
func height(node *TreeNode) int {
	if node == nil {
		return 0
	}
	lh := height(node.Left)
	if lh < 0 {
		return -1 // left side already failed: stop early
	}
	rh := height(node.Right)
	if rh < 0 || abs(lh-rh) > 1 {
		return -1
	}
	return 1 + max(lh, rh)
}

func abs(x int) int {
	if x < 0 {
		return -x
	}
	return x
}

// isBalancedIterative: same check with an explicit post-order stack.
func isBalancedIterative(root *TreeNode) bool {
	type frame struct {
		node         *TreeNode
		childrenDone bool
	}
	h := map[*TreeNode]int{nil: 0} // node -> height
	var st []frame
	if root != nil {
		st = append(st, frame{root, false})
	}
	for len(st) > 0 {
		f := st[len(st)-1]
		st = st[:len(st)-1]
		if f.childrenDone {
			lh, rh := h[f.node.Left], h[f.node.Right]
			if abs(lh-rh) > 1 {
				return false
			}
			h[f.node] = 1 + max(lh, rh)
		} else {
			st = append(st, frame{f.node, true}) // come back after both children
			if f.node.Right != nil {
				st = append(st, frame{f.node.Right, false})
			}
			if f.node.Left != nil {
				st = append(st, frame{f.node.Left, false})
			}
		}
	}
	return true
}

static class Solution {
    public boolean isBalanced(TreeNode root) {
        return height(root) >= 0;
    }

    // Height of the subtree, or -1 if some node inside it is unbalanced.
    private int height(TreeNode node) {
        if (node == null) return 0;
        int lh = height(node.left);
        if (lh < 0) return -1;                       // left side already failed: stop early
        int rh = height(node.right);
        if (rh < 0 || Math.abs(lh - rh) > 1) return -1;
        return 1 + Math.max(lh, rh);
    }

    // Same check with an explicit post-order stack (no recursion depth limit).
    public boolean isBalancedIterative(TreeNode root) {
        Map<TreeNode, Integer> h = new HashMap<>();  // node -> height; null counts as 0
        Deque<TreeNode> st = new ArrayDeque<>();
        Deque<Boolean> done = new ArrayDeque<>();    // parallel stack: children finished?
        if (root != null) { st.push(root); done.push(false); }
        while (!st.isEmpty()) {
            TreeNode node = st.pop();
            boolean childrenDone = done.pop();
            if (childrenDone) {
                int lh = node.left == null ? 0 : h.get(node.left);
                int rh = node.right == null ? 0 : h.get(node.right);
                if (Math.abs(lh - rh) > 1) return false;
                h.put(node, 1 + Math.max(lh, rh));
            } else {
                st.push(node); done.push(true);      // come back after both children
                if (node.right != null) { st.push(node.right); done.push(false); }
                if (node.left != null) { st.push(node.left); done.push(false); }
            }
        }
        return true;
    }
}
class Solution:
    def isBalanced(self, root: Optional[TreeNode]) -> bool:
        # Recursion depth = tree height. LeetCode raises Python's limit; elsewhere the default
        # (1000 frames) fails on chains deeper than ~990: raise it, or use isBalancedIterative.
        def height(node: Optional[TreeNode]) -> int:
            """Height of the subtree, or -1 if some node inside it is unbalanced."""
            if node is None:
                return 0
            lh = height(node.left)
            if lh < 0:
                return -1                      # left side already failed: stop early
            rh = height(node.right)
            if rh < 0 or abs(lh - rh) > 1:
                return -1
            return 1 + max(lh, rh)
        return height(root) >= 0

    def isBalancedIterative(self, root: Optional[TreeNode]) -> bool:
        """Same check with an explicit post-order stack (no recursion depth limit)."""
        h = {None: 0}                          # node -> height, filled in post-order
        stack = [(root, False)] if root else []
        while stack:
            node, children_done = stack.pop()
            if children_done:
                lh, rh = h[node.left], h[node.right]
                if abs(lh - rh) > 1:
                    return False
                h[node] = 1 + max(lh, rh)
            else:
                stack.append((node, True))     # come back after both children
                if node.right:
                    stack.append((node.right, False))
                if node.left:
                    stack.append((node.left, False))
        return True
