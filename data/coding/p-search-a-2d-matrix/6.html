func searchMatrix(matrix [][]int, target int) bool {
	m, n := len(matrix), len(matrix[0])
	// Read row by row, the matrix is one sorted array of m*n values.
	lo, hi := 0, m*n-1
	for lo <= hi {
		mid := lo + (hi-lo)/2
		v := matrix[mid/n][mid%n] // flat index -> (row, col)
		if v == target {
			return true
		}
		if v < target {
			lo = mid + 1 // target is to the right
		} else {
			hi = mid - 1 // target is to the left
		}
	}
	return false
}

def searchMatrix(self, matrix, target):
    r, c = 0, len(matrix[0]) - 1          # top-right corner
    while r < len(matrix) and c >= 0:
        v = matrix[r][c]
        if v == target:
            return True
        if v < target:
            r += 1                         # everything left in this row is smaller
        else:
            c -= 1                         # everything below in this column is larger
    return False
