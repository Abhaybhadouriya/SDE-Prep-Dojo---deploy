The tested files are in coding-code/find-the-celebrity/; each driver implements knows over a hidden matrix, counts the calls, and asserts the result and the 3n budget against a brute force. On LeetCode, knows is provided: in Java the class extends Relation, in Go the judge passes knows into solution.

class Solution {
public:
    int findCelebrity(int n) {
        // Pass 1: every call rules one person out, so one candidate survives.
        int cand = 0;
        for (int i = 1; i < n; i++)
            if (knows(cand, i)) cand = i;  // cand knows someone -> not the celebrity
                                           // else i is not known by cand -> i is out
        // Pass 2: check the survivor against everyone.
        for (int i = 0; i < n; i++) {
            if (i == cand) continue;
            // for i > cand, knows(cand, i) was already false in pass 1: skip that call
            if (i < cand && knows(cand, i)) return -1;
            if (!knows(i, cand)) return -1;
        }
        return cand;
    }
};

class Solution:
    def findCelebrity(self, n: int) -> int:
        # Pass 1: every call rules one person out, so one candidate survives.
        cand = 0
        for i in range(1, n):
            if knows(cand, i):       # cand knows someone -> cand is not the celebrity
                cand = i             # (and nobody before i can be: they were ruled out)
            # else: i is not known by cand -> i is not the celebrity
        # Pass 2: check the survivor against everyone.
        for i in range(n):
            if i == cand:
                continue
            # for i > cand, knows(cand, i) was already False in pass 1: skip that call
            if i < cand and knows(cand, i):
                return -1
            if not knows(i, cand):
                return -1
        return cand
