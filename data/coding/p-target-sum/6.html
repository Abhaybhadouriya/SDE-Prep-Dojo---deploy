class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int total = 0;
        for (int x : nums) total += x;
        // plus - minus = target and plus + minus = total  =>  plus = (total + target) / 2
        if (abs(target) > total || (total + target) % 2 != 0) return 0;
        int want = (total + target) / 2;

        vector<int> ways(want + 1, 0);   // ways[s] = subsets of the numbers seen so far with sum s
        ways[0] = 1;                     // the empty subset
        for (int x : nums) {
            // walk s DOWNWARD so each number is used at most once; x = 0 doubles every count
            for (int s = want; s >= x; s--) ways[s] += ways[s - x];
        }
        return ways[want];
    }
};
func findTargetSumWays(nums []int, target int) int {
	total := 0
	for _, x := range nums {
		total += x
	}
	// plus - minus = target and plus + minus = total  =>  plus = (total + target) / 2
	if target > total || -target > total || (total+target)%2 != 0 {
		return 0
	}
	want := (total + target) / 2

	ways := make([]int, want+1) // ways[s] = subsets of the numbers seen so far with sum s
	ways[0] = 1                 // the empty subset
	for _, x := range nums {
		// walk s DOWNWARD so each number is used at most once; x = 0 doubles every count
		for s := want; s >= x; s-- {
			ways[s] += ways[s-x]
		}
	}
	return ways[want]
}

class Solution:
    def findTargetSumWays(self, nums: List[int], target: int) -> int:
        total = sum(nums)
        # plus - minus = target and plus + minus = total  =>  plus = (total + target) / 2
        if abs(target) > total or (total + target) % 2:
            return 0
        want = (total + target) // 2

        ways = [0] * (want + 1)      # ways[s] = number of subsets (of numbers seen so far) with sum s
        ways[0] = 1                  # the empty subset
        for x in nums:
            # walk s DOWNWARD so each number joins a subset at most once; x = 0 doubles every count
            for s in range(want, x - 1, -1):
                ways[s] += ways[s - x]
        return ways[want]
