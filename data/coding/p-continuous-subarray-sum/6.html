class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        // sum(l+1..r) = pre[r] - pre[l] is a multiple of k
        // <=> pre[r] % k == pre[l] % k. The length is r - l, and it must be >= 2,
        // so keep the FIRST index of each remainder (the farthest-left start).
        unordered_map<int, int> first;
        first[0] = -1;                      // remainder 0 "before index 0": runs starting at index 0
        long long rem = 0;                  // rem + x <= prefix sum <= 2^31-1 here; 64-bit stays safe without that bound
        for (int i = 0; i < (int)nums.size(); i++) {
            rem = (rem + nums[i]) % k;      // nums >= 0 and k >= 1: rem is never negative
            auto it = first.find((int)rem);
            if (it != first.end()) {
                if (i - it->second >= 2) return true;
            } else {
                first[(int)rem] = i;        // never overwrite: an older index gives a longer run
            }
        }
        return false;
    }
};
