The explicit-stack version in four languages. Each file in coding-code/flood-fill/ also fills a 50 x 50 single-colour image (the recursion-depth case) and checks 3,000 random small grids against a separate BFS with a visited matrix.

class Solution {
public:
    vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color) {
        int old = image[sr][sc];
        if (old == color) return image;        // nothing to do; without this the loop never ends
        int m = image.size(), n = image[0].size();
        const int dr[4] = {-1, 1, 0, 0}, dc[4] = {0, 0, -1, 1};
        vector<pair<int, int>> stk = {{sr, sc}};
        image[sr][sc] = color;                 // paint when pushed, so no pixel is pushed twice
        while (!stk.empty()) {
            int r = stk.back().first, c = stk.back().second;
            stk.pop_back();
            for (int k = 0; k < 4; ++k) {
                int nr = r + dr[k], nc = c + dc[k];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n || image[nr][nc] != old) continue;
                image[nr][nc] = color;
                stk.push_back({nr, nc});
            }
        }
        return image;
    }
};
func floodFill(image [][]int, sr int, sc int, color int) [][]int {
	old := image[sr][sc]
	if old == color {
		return image // nothing to do; without this the loop never ends
	}
	m, n := len(image), len(image[0])
	dirs := [4][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}}
	stack := [][2]int{{sr, sc}}
	image[sr][sc] = color // paint when pushed, so no pixel is pushed twice
	for len(stack) > 0 {
		cur := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		for _, d := range dirs {
			r, c := cur[0]+d[0], cur[1]+d[1]
			if r < 0 || r >= m || c < 0 || c >= n || image[r][c] != old {
				continue
			}
			image[r][c] = color
			stack = append(stack, [2]int{r, c})
		}
	}
	return image
}

static class Solution {
    private static final int[][] DIRS = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    public int[][] floodFill(int[][] image, int sr, int sc, int color) {
        int old = image[sr][sc];
        if (old == color) return image;        // nothing to do; without this the loop never ends
        int m = image.length, n = image[0].length;
        Deque<int[]> stk = new ArrayDeque<>();
        stk.push(new int[]{sr, sc});
        image[sr][sc] = color;                 // paint when pushed, so no pixel is pushed twice
        while (!stk.isEmpty()) {
            int[] cur = stk.pop();
            for (int[] d : DIRS) {
                int r = cur[0] + d[0], c = cur[1] + d[1];
                if (r < 0 || r >= m || c < 0 || c >= n || image[r][c] != old) continue;
                image[r][c] = color;
                stk.push(new int[]{r, c});
            }
        }
        return image;
    }
}
class Solution:
    def floodFill(self, image: List[List[int]], sr: int, sc: int, color: int) -> List[List[int]]:
        old = image[sr][sc]
        if old == color:            # nothing to do; without this the loop never ends
            return image
        m, n = len(image), len(image[0])
        image[sr][sc] = color       # paint when pushed, so no pixel is pushed twice
        stack = [(sr, sc)]
        while stack:
            r, c = stack.pop()
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if 0 <= nr < m and 0 <= nc < n and image[nr][nc] == old:
                    image[nr][nc] = color
                    stack.append((nr, nc))
        return image
