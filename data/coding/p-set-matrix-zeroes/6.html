The O(1)-space marker version in four languages. The test drivers (the examples, 1 × 1 grids with and without a zero, a zero only in column 0 next to INT_MAX/INT_MIN values, a single row, and 500 random grids checked against the two-array version) are in coding-code/set-matrix-zeroes/.

func setZeroes(matrix [][]int) {
	m, n := len(matrix), len(matrix[0])
	col0Zero := false // column 0's own flag: cell [0][0] already speaks for row 0
	// Pass 1 (top-down): record every zero in its row head [i][0] and column head [0][j].
	for i := 0; i < m; i++ {
		if matrix[i][0] == 0 {
			col0Zero = true
		}
		for j := 1; j < n; j++ {
			if matrix[i][j] == 0 {
				matrix[i][0] = 0
				matrix[0][j] = 0
			}
		}
	}
	// Pass 2 (bottom-up, right-to-left): row 0 holds the column marks, so it is rewritten last.
	for i := m - 1; i >= 0; i-- {
		for j := n - 1; j >= 1; j-- {
			if matrix[i][0] == 0 || matrix[0][j] == 0 {
				matrix[i][j] = 0
			}
		}
		if col0Zero {
			matrix[i][0] = 0
		}
	}
}

    def setZeroes(self, matrix: List[List[int]]) -> None:

        col0_zero = False   # column 0's own flag: cell [0][0] already speaks for row 0

        for i in range(m):

                col0_zero = True
