func zigzagLevelOrder(root *TreeNode) [][]int {
	levels := [][]int{}
	if root == nil {
		return levels // empty tree -> []
	}
	q := []*TreeNode{root}
	leftToRight := true
	for len(q) > 0 {
		width := len(q) // fixed before children are added
		row := make([]int, width)
		for i := 0; i < width; i++ {
			node := q[i]
			// Children are always queued left then right; only the write slot flips.
			slot := i
			if !leftToRight {
				slot = width - 1 - i
			}
			row[slot] = node.Val
			if node.Left != nil {
				q = append(q, node.Left)
			}
			if node.Right != nil {
				q = append(q, node.Right)
			}
		}
		q = q[width:] // drop the level we just finished
		levels = append(levels, row)
		leftToRight = !leftToRight
	}
	return levels
}
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        vector<vector<int>> levels;
        if (!root) return levels;                 // empty tree -> []
        queue<TreeNode*> q;
        q.push(root);
        bool leftToRight = true;
        while (!q.empty()) {
            int width = q.size();                 // nodes on this level, fixed before we push children
            vector<int> row(width);
            for (int i = 0; i < width; ++i) {
                TreeNode* node = q.front();
                q.pop();
                // The queue order never changes (always left child, then right child).
                // Only the slot we write into flips on odd levels.
                int slot = leftToRight ? i : width - 1 - i;
                row[slot] = node->val;
                if (node->left) q.push(node->left);
                if (node->right) q.push(node->right);
            }
            levels.push_back(row);
            leftToRight = !leftToRight;
        }
        return levels;
    }
};

    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        levels = []
        if root is None:
            return levels                          # empty tree -> []
        q = deque([root])
        left_to_right = True
        while q:
            width = len(q)                         # fixed before children are added
            row = [0] * width
            for i in range(width):
                node = q.popleft()
                # Children are always queued left then right; only the write slot flips.
                slot = i if left_to_right else width - 1 - i
                row[slot] = node.val
                if node.left:
                    q.append(node.left)
                if node.right:
                    q.append(node.right)
            levels.append(row)
            left_to_right = not left_to_right
        return levels
