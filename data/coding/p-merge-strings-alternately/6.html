class Solution {
public:
    string mergeAlternately(string word1, string word2) {
        size_t n = word1.size(), m = word2.size();
        string out;
        out.reserve(n + m);                       // one allocation for the whole answer
        for (size_t i = 0; i < n || i < m; i++) {
            if (i < n) out += word1[i];           // word1 always goes first in a round
            if (i < m) out += word2[i];           // a finished word just stops contributing
        }
        return out;
    }
};
func mergeAlternately(word1 string, word2 string) string {
	n, m := len(word1), len(word2)
	var b strings.Builder
	b.Grow(n + m) // one allocation for the whole answer
	for i := 0; i < n || i < m; i++ {
		if i < n {
			b.WriteByte(word1[i]) // word1 always goes first in a round
		}
		if i < m {
			b.WriteByte(word2[i]) // a finished word just stops contributing
		}
	}
	return b.String()
}

static class Solution {
    public String mergeAlternately(String word1, String word2) {
        int n = word1.length(), m = word2.length();
        StringBuilder out = new StringBuilder(n + m);  // one allocation for the whole answer
        for (int i = 0; i < n || i < m; i++) {
            if (i < n) out.append(word1.charAt(i));    // word1 always goes first in a round
            if (i < m) out.append(word2.charAt(i));    // a finished word just stops contributing
        }
        return out.toString();
    }
}
class Solution:
    def mergeAlternately(self, word1: str, word2: str) -> str:
        n, m = len(word1), len(word2)
        parts = []                          # join once at the end: no quadratic += on str
        for i in range(max(n, m)):
            if i < n:
                parts.append(word1[i])      # word1 always goes first in a round
            if i < m:
                parts.append(word2[i])      # a finished word just stops contributing
        return "".join(parts)
