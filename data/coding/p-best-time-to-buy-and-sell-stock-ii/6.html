class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int profit = 0;
        for (size_t d = 1; d < prices.size(); ++d)
            if (prices[d] > prices[d - 1])
                profit += prices[d] - prices[d - 1];   // ride every up-step, skip every down-step
        return profit;
    }
};
