class Solution {
public:
    bool uniqueOccurrences(vector<int>& arr) {
        unordered_map<int, int> freq;         // value -> how many times it occurs
        for (int v : arr) freq[v]++;
        unordered_set<int> seen;              // counts already claimed by some value
        for (auto& kv : freq)
            if (!seen.insert(kv.second).second)
                return false;                 // insert failed: two values share this count
        return true;
    }
};
func uniqueOccurrences(arr []int) bool {
	freq := make(map[int]int) // value -> how many times it occurs
	for _, v := range arr {
		freq[v]++
	}
	seen := make(map[int]bool) // counts already claimed by some value
	for _, c := range freq {
		if seen[c] {
			return false // two different values share this count
		}
		seen[c] = true
	}
	return true
}

static class Solution {
    public boolean uniqueOccurrences(int[] arr) {
        Map<Integer, Integer> freq = new HashMap<>();    // value -> how many times it occurs
        for (int v : arr) freq.merge(v, 1, Integer::sum);
        Set<Integer> seen = new HashSet<>();             // counts already claimed by some value
        for (int c : freq.values())
            if (!seen.add(c)) return false;              // add failed: two values share this count
        return true;
    }
}
class Solution:
    def uniqueOccurrences(self, arr: List[int]) -> bool:
        freq = Counter(arr)                  # value -> how many times it occurs
        seen = set()                         # counts already claimed by some value
        for c in freq.values():
            if c in seen:                    # two different values share this count
                return False
            seen.add(c)
        return True
