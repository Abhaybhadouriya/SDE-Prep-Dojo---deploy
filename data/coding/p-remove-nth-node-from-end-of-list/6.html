func removeNthFromEnd(head *ListNode, n int) *ListNode {
	dummy := &ListNode{Next: head} // lets us delete the real head like any other node
	lead, trail := dummy, dummy
	for i := 0; i < n; i++ { // open a gap of n nodes
		lead = lead.Next
	}
	for lead.Next != nil { // move both until lead is on the LAST node
		lead = lead.Next
		trail = trail.Next
	}
	trail.Next = trail.Next.Next // unlink the n-th node from the end
	return dummy.Next
}
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode dummy(0, head);          // lets us delete the real head like any other node
        ListNode* lead = &dummy;
        ListNode* trail = &dummy;
        for (int i = 0; i < n; i++)       // open a gap of n nodes (n <= length, so lead stays valid)
            lead = lead->next;
        while (lead->next) {              // move both until lead is on the LAST node
            lead = lead->next;
            trail = trail->next;
        }
        // trail is now just before the n-th node from the end
        ListNode* victim = trail->next;
        trail->next = victim->next;
        delete victim;                    // C++ only: free the node (LeetCode does not require it)
        return dummy.next;
    }
};

def removeNthFromEnd(self, head, n):
    length, node = 0, head
    while node:                      # pass 1: count
        length, node = length + 1, node.next
    if n == length:                  # the head is the target
        return head.next
    pred = head
    for _ in range(length - n - 1):  # pass 2: stop on the predecessor
        pred = pred.next
    pred.next = pred.next.next
    return head
