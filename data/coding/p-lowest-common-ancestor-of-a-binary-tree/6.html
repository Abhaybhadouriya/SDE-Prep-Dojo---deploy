Recursive post-order solution in four languages. Each file in coding-code/lowest-common-ancestor-of-a-binary-tree/ also checks every pair of nodes in 60 random trees against the parent-pointer method.

class Solution {
public:
    // Returns: the LCA if both p and q are below root, p or q if only one is,
    // nullptr if neither is. At the top, both exist, so it is the LCA.
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == nullptr || root == p || root == q) return root;  // a node is its own descendant
        TreeNode* fromLeft = lowestCommonAncestor(root->left, p, q);
        TreeNode* fromRight = lowestCommonAncestor(root->right, p, q);
        if (fromLeft && fromRight) return root;    // p and q split here: root is the lowest meeting point
        return fromLeft ? fromLeft : fromRight;    // pass up whatever was found (or nullptr)
    }
};
// Returns: the LCA if both p and q are below root, p or q if only one is,
// nil if neither is. At the top, both exist, so it is the LCA.
func lowestCommonAncestor(root, p, q *TreeNode) *TreeNode {
	if root == nil || root == p || root == q { // a node is its own descendant
		return root
	}
	fromLeft := lowestCommonAncestor(root.Left, p, q)
	fromRight := lowestCommonAncestor(root.Right, p, q)
	if fromLeft != nil && fromRight != nil { // p and q split here
		return root
	}
	if fromLeft != nil { // pass up whatever was found
		return fromLeft
	}
	return fromRight
}

static class Solution {
    // Returns: the LCA if both p and q are below root, p or q if only one is,
    // null if neither is. At the top, both exist, so it is the LCA.
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        if (root == null || root == p || root == q) return root;   // a node is its own descendant
        TreeNode fromLeft = lowestCommonAncestor(root.left, p, q);
        TreeNode fromRight = lowestCommonAncestor(root.right, p, q);
        if (fromLeft != null && fromRight != null) return root;    // p and q split here
        return fromLeft != null ? fromLeft : fromRight;            // pass up whatever was found
    }
}
class Solution:
    def lowestCommonAncestor(self, root: "TreeNode", p: "TreeNode", q: "TreeNode") -> "TreeNode":
        # Returns: the LCA if both p and q are below root, p or q if only one is,
        # None if neither is. At the top, both exist, so it is the LCA.
        if root is None or root is p or root is q:   # a node is its own descendant
            return root
        from_left = self.lowestCommonAncestor(root.left, p, q)
        from_right = self.lowestCommonAncestor(root.right, p, q)
        if from_left and from_right:                 # p and q split here
            return root
        return from_left or from_right               # pass up whatever was found
