// Run: go run main.go   (checks; prints "ok")
package main

import (
	"fmt"
	"math"
	"reflect"
)

// FIXED window: best sum of any k consecutive values. O(n) time, O(1) space.
func maxSumFixed(nums []int, k int) int {
	window := 0
	for i := 0; i < k; i++ {
		window += nums[i]
	}
	best := window
	for right := k; right < len(nums); right++ {
		window += nums[right] - nums[right-k] // one value enters, one leaves
		if window > best {
			best = window
		}
	}
	return best
}

// VARIABLE window, longest valid: longest substring with <= k distinct bytes.
func longestAtMostKDistinct(s string, k int) int {
	var count [256]int
	distinct, left, best := 0, 0, 0
	for right := 0; right < len(s); right++ {
		if count[s[right]] == 0 { // 1. expand
			distinct++
		}
		count[s[right]]++
		for distinct > k { // 2. shrink until valid
			count[s[left]]--
			if count[s[left]] == 0 {
				distinct--
			}
			left++
		}
		if right-left+1 > best { // 3. record
			best = right - left + 1
		}
	}
	return best
}

// VARIABLE window, shortest valid: smallest substring of s containing all of t.
func minWindowCover(s, t string) string {
	var need [256]int
	for i := 0; i < len(t); i++ {
		need[t[i]]++
	}
	missing, left, bestLen, bestStart := len(t), 0, math.MaxInt32, 0
	for right := 0; right < len(s); right++ {
		if need[s[right]] > 0 { // 1. expand
			missing--
		}
		need[s[right]]--
		for missing == 0 { // 2. valid: record, then shrink
			if right-left+1 < bestLen {
				bestLen, bestStart = right-left+1, left
			}
			need[s[left]]++
			if need[s[left]] > 0 {
				missing++
			}
			left++
		}
	}
	if bestLen == math.MaxInt32 {
		return ""
	}
	return s[bestStart : bestStart+bestLen]
}

// Number of subarrays with <= k distinct values.
func countAtMostKDistinct(nums []int, k int) int {
	if k < 0 {
		return 0
	}
	count := map[int]int{}
	left, total := 0, 0
	for right, x := range nums {
		count[x]++
		for len(count) > k {
			count[nums[left]]--
			if count[nums[left]] == 0 {
				delete(count, nums[left])
			}
			left++
		}
		total += right - left + 1 // all windows ending at right
	}
	return total
}

// exactly(k) = atMost(k) - atMost(k - 1)
func countExactlyKDistinct(nums []int, k int) int {
	return countAtMostKDistinct(nums, k) - countAtMostKDistinct(nums, k-1)
}

// Max of every window of size k: monotonic deque of indices (a slice used as a deque).
func windowMax(nums []int, k int) []int {
	dq := []int{}
	out := []int{}
	for right, x := range nums {
		for len(dq) > 0 && nums[dq[len(dq)-1]] <= x {
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, right)
		if dq[0] <= right-k { // front left the window
			dq = dq[1:]
		}
		if right >= k-1 {
			out = append(out, nums[dq[0]])
		}
	}
	return out
}

func check(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

func main() {
	check(maxSumFixed([]int{4, 2, -1, 7, 3, 1}, 3) == 11, "fixed")
	check(maxSumFixed([]int{5}, 1) == 5, "fixed single")
	check(longestAtMostKDistinct("eceba", 2) == 3, "eceba")
	check(longestAtMostKDistinct("abaccc", 2) == 4, "abaccc")
	check(longestAtMostKDistinct("aa", 1) == 2, "aa")
	check(minWindowCover("ADOBECODEBANC", "ABC") == "BANC", "BANC")
	check(minWindowCover("a", "aa") == "", "no cover")
	check(minWindowCover("a", "a") == "a", "a")
	check(countExactlyKDistinct([]int{1, 2, 1, 2, 3}, 2) == 7, "exact 7")
	check(countExactlyKDistinct([]int{1, 2, 1, 3, 4}, 3) == 3, "exact 3")
	check(reflect.DeepEqual(windowMax([]int{1, 3, -1, -3, 5, 3, 6, 7}, 3), []int{3, 3, 5, 5, 6, 7}), "max")
	check(reflect.DeepEqual(windowMax([]int{9, 8, 7}, 1), []int{9, 8, 7}), "max k=1")
	fmt.Println("ok")
}

#include <algorithm>
#include <cassert>
#include <climits>
#include <cstdio>
#include <deque>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

// FIXED window: best sum of any k consecutive values. O(n) time, O(1) space.
long long maxSumFixed(const vector<int>& nums, int k) {
    long long window = 0;
    for (int i = 0; i < k; i++) window += nums[i];
    long long best = window;
    for (int right = k; right < (int)nums.size(); right++) {
        window += nums[right] - nums[right - k];        // one value enters, one leaves
        best = max(best, window);
    }
    return best;
}

// VARIABLE window, longest valid: longest substring with <= k distinct chars.
int longestAtMostKDistinct(const string& s, int k) {
    int count[256] = {0}, distinct = 0, left = 0, best = 0;
    for (int right = 0; right < (int)s.size(); right++) {
        if (count[(unsigned char)s[right]]++ == 0) distinct++;   // 1. expand
        while (distinct > k) {                                   // 2. shrink until valid
            if (--count[(unsigned char)s[left]] == 0) distinct--;
            left++;
        }
        best = max(best, right - left + 1);                      // 3. record
    }
    return best;
}

// VARIABLE window, shortest valid: smallest substring of s containing all of t.
string minWindowCover(const string& s, const string& t) {
    int need[256] = {0};
    for (char c : t) need[(unsigned char)c]++;
    int missing = (int)t.size(), left = 0, bestLen = INT_MAX, bestStart = 0;
    for (int right = 0; right < (int)s.size(); right++) {
        if (need[(unsigned char)s[right]]-- > 0) missing--;     // 1. expand
        while (missing == 0) {                                   // 2. valid: record, then shrink
            if (right - left + 1 < bestLen) { bestLen = right - left + 1; bestStart = left; }
            if (++need[(unsigned char)s[left]] > 0) missing++;
            left++;
        }
    }
    return bestLen == INT_MAX ? "" : s.substr(bestStart, bestLen);
}

// Number of subarrays with <= k distinct values.
long long countAtMostKDistinct(const vector<int>& nums, int k) {
    if (k < 0) return 0;
    unordered_map<int, int> count;
    long long total = 0;
    int left = 0;
    for (int right = 0; right < (int)nums.size(); right++) {
        count[nums[right]]++;
        while ((int)count.size() > k) {
            if (--count[nums[left]] == 0) count.erase(nums[left]);
            left++;
        }
        total += right - left + 1;                               // all windows ending at right
    }
    return total;
}

// exactly(k) = atMost(k) - atMost(k - 1)
long long countExactlyKDistinct(const vector<int>& nums, int k) {
    return countAtMostKDistinct(nums, k) - countAtMostKDistinct(nums, k - 1);
}

// Max of every window of size k: monotonic deque of indices, values strictly decreasing.
vector<int> windowMax(const vector<int>& nums, int k) {
    deque<int> dq;
    vector<int> out;
    for (int right = 0; right < (int)nums.size(); right++) {
        while (!dq.empty() && nums[dq.back()] <= nums[right]) dq.pop_back();
        dq.push_back(right);
        if (dq.front() <= right - k) dq.pop_front();             // front left the window
        if (right >= k - 1) out.push_back(nums[dq.front()]);
    }
    return out;
}

int main() {
    assert(maxSumFixed({4, 2, -1, 7, 3, 1}, 3) == 11);
    assert(maxSumFixed({5}, 1) == 5);
    assert(longestAtMostKDistinct("eceba", 2) == 3);
    assert(longestAtMostKDistinct("abaccc", 2) == 4);
    assert(longestAtMostKDistinct("aa", 1) == 2);
    assert(minWindowCover("ADOBECODEBANC", "ABC") == "BANC");
    assert(minWindowCover("a", "aa") == "");
    assert(minWindowCover("a", "a") == "a");
    assert(countExactlyKDistinct({1, 2, 1, 2, 3}, 2) == 7);
    assert(countExactlyKDistinct({1, 2, 1, 3, 4}, 3) == 3);
    assert((windowMax({1, 3, -1, -3, 5, 3, 6, 7}, 3) == vector<int>{3, 3, 5, 5, 6, 7}));
    assert((windowMax({9, 8, 7}, 1) == vector<int>{9, 8, 7}));
    puts("ok");
    return 0;
}

// Run: java SlidingWindow.java   (checks; prints "ok")
import java.util.*;

public class SlidingWindow {

    // FIXED window: best sum of any k consecutive values. O(n) time, O(1) space.
    static long maxSumFixed(int[] nums, int k) {
        long window = 0;
        for (int i = 0; i < k; i++) window += nums[i];
        long best = window;
        for (int right = k; right < nums.length; right++) {
            window += nums[right] - nums[right - k];          // one value enters, one leaves
            best = Math.max(best, window);
        }
        return best;
    }

    // VARIABLE window, longest valid: longest substring with <= k distinct chars.
    static int longestAtMostKDistinct(String s, int k) {
        int[] count = new int[128];
        int distinct = 0, left = 0, best = 0;
        for (int right = 0; right < s.length(); right++) {
            if (count[s.charAt(right)]++ == 0) distinct++;     // 1. expand
            while (distinct > k) {                             // 2. shrink until valid
                if (--count[s.charAt(left)] == 0) distinct--;
                left++;
            }
            best = Math.max(best, right - left + 1);           // 3. record
        }
        return best;
    }

    // VARIABLE window, shortest valid: smallest substring of s containing all of t.
    static String minWindowCover(String s, String t) {
        int[] need = new int[128];
        for (char c : t.toCharArray()) need[c]++;
        int missing = t.length(), left = 0, bestLen = Integer.MAX_VALUE, bestStart = 0;
        for (int right = 0; right < s.length(); right++) {
            if (need[s.charAt(right)]-- > 0) missing--;        // 1. expand
            while (missing == 0) {                             // 2. valid: record, then shrink
                if (right - left + 1 < bestLen) { bestLen = right - left + 1; bestStart = left; }
                if (++need[s.charAt(left)] > 0) missing++;
                left++;
            }
        }
        return bestLen == Integer.MAX_VALUE ? "" : s.substring(bestStart, bestStart + bestLen);
    }

    // Number of subarrays with <= k distinct values.
    static long countAtMostKDistinct(int[] nums, int k) {
        if (k < 0) return 0;
        Map<Integer, Integer> count = new HashMap<>();
        long total = 0;
        int left = 0;
        for (int right = 0; right < nums.length; right++) {
            count.merge(nums[right], 1, Integer::sum);
            while (count.size() > k) {
                if (count.merge(nums[left], -1, Integer::sum) == 0) count.remove(nums[left]);
                left++;
            }
            total += right - left + 1;                         // all windows ending at right
        }
        return total;
    }

    // exactly(k) = atMost(k) - atMost(k - 1)
    static long countExactlyKDistinct(int[] nums, int k) {
        return countAtMostKDistinct(nums, k) - countAtMostKDistinct(nums, k - 1);
    }

    // Max of every window of size k: monotonic deque of indices, values strictly decreasing.
    static int[] windowMax(int[] nums, int k) {
        Deque<Integer> dq = new ArrayDeque<>();
        int[] out = new int[nums.length - k + 1];
        for (int right = 0; right < nums.length; right++) {
            while (!dq.isEmpty() && nums[dq.peekLast()] <= nums[right]) dq.pollLast();
            dq.offerLast(right);
            if (dq.peekFirst() <= right - k) dq.pollFirst();   // front left the window
            if (right >= k - 1) out[right - k + 1] = nums[dq.peekFirst()];
        }
        return out;
    }

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError(what);
    }

    public static void main(String[] args) {
        check(maxSumFixed(new int[]{4, 2, -1, 7, 3, 1}, 3) == 11, "fixed");
        check(maxSumFixed(new int[]{5}, 1) == 5, "fixed single");
        check(longestAtMostKDistinct("eceba", 2) == 3, "eceba");
        check(longestAtMostKDistinct("abaccc", 2) == 4, "abaccc");
        check(longestAtMostKDistinct("aa", 1) == 2, "aa");
        check(minWindowCover("ADOBECODEBANC", "ABC").equals("BANC"), "BANC");
        check(minWindowCover("a", "aa").equals(""), "no cover");
        check(minWindowCover("a", "a").equals("a"), "a");
        check(countExactlyKDistinct(new int[]{1, 2, 1, 2, 3}, 2) == 7, "exact 7");
        check(countExactlyKDistinct(new int[]{1, 2, 1, 3, 4}, 3) == 3, "exact 3");
        check(Arrays.equals(windowMax(new int[]{1, 3, -1, -3, 5, 3, 6, 7}, 3), new int[]{3, 3, 5, 5, 6, 7}), "max");
        check(Arrays.equals(windowMax(new int[]{9, 8, 7}, 1), new int[]{9, 8, 7}), "max k=1");
        System.out.println("ok");
    }
}
"""Sliding window templates (SDE Prep Dojo, lesson coding-sliding-window).

Run: python sliding_window.py   (asserts; prints "ok")
"""
from collections import Counter, deque


def max_sum_fixed(nums, k):
    """FIXED window: best sum of any k consecutive values. O(n) time, O(1) space."""
    window = sum(nums[:k])
    best = window
    for right in range(k, len(nums)):
        window += nums[right] - nums[right - k]  # one value enters, one leaves
        best = max(best, window)
    return best


def longest_at_most_k_distinct(s, k):
    """VARIABLE window, longest valid: longest substring with <= k distinct chars."""
    count = {}
    left = best = 0
    for right, ch in enumerate(s):
        count[ch] = count.get(ch, 0) + 1          # 1. expand: take s[right]
        while len(count) > k:                     # 2. shrink until valid again
            out = s[left]
            count[out] -= 1
            if count[out] == 0:
                del count[out]
            left += 1
        best = max(best, right - left + 1)        # 3. record: window is valid here
    return best


def min_window_cover(s, t):
    """VARIABLE window, shortest valid: smallest substring of s containing all of t."""
    need = Counter(t)          # need[c] > 0 means "still missing c"
    missing = len(t)
    left = 0
    best_len, best_start = float("inf"), 0
    for right, ch in enumerate(s):
        if need[ch] > 0:
            missing -= 1
        need[ch] -= 1                              # 1. expand
        while missing == 0:                        # 2. valid: record, then shrink
            if right - left + 1 < best_len:
                best_len, best_start = right - left + 1, left
            out = s[left]
            need[out] += 1
            if need[out] > 0:
                missing += 1
            left += 1
    return "" if best_len == float("inf") else s[best_start:best_start + best_len]


def count_at_most_k_distinct(nums, k):
    """Number of subarrays with <= k distinct values: every valid window adds right-left+1."""
    if k < 0:
        return 0
    count = {}
    left = total = 0
    for right, x in enumerate(nums):
        count[x] = count.get(x, 0) + 1
        while len(count) > k:
            out = nums[left]
            count[out] -= 1
            if count[out] == 0:
                del count[out]
            left += 1
        total += right - left + 1                  # all windows ending at right
    return total


def count_exactly_k_distinct(nums, k):
    """exactly(k) = atMost(k) - atMost(k - 1)."""
    return count_at_most_k_distinct(nums, k) - count_at_most_k_distinct(nums, k - 1)


def window_max(nums, k):
    """Max of every window of size k with a monotonic deque of indices. O(n)."""
    dq = deque()               # indices; their values are strictly decreasing
    out = []
    for right, x in enumerate(nums):
        while dq and nums[dq[-1]] <= x:            # smaller values can never be a max again
            dq.pop()
        dq.append(right)
        if dq[0] <= right - k:                     # front index left the window
            dq.popleft()
        if right >= k - 1:
            out.append(nums[dq[0]])
    return out
