Four separate walks, no recursion (so a 104-deep chain is safe in every language). Each file in coding-code/boundary-of-binary-tree/ runs the examples and the turn cases; the C++ file adds a 10,000-node left chain, and the Python file checks 500 random trees against the one-pass flag version.

class Solution {
    static bool isLeaf(TreeNode* n) { return n && !n->left && !n->right; }

public:
    vector<int> boundaryOfBinaryTree(TreeNode* root) {
        if (!root) return {};
        vector<int> out{root->val};
        if (isLeaf(root)) return out;            // a lone root is the whole boundary, not a leaf

        // 1. Left edge, top-down: prefer left child, fall back to right. Leaves wait for step 2.
        for (TreeNode* n = root->left; n && !isLeaf(n); n = n->left ? n->left : n->right)
            out.push_back(n->val);

        // 2. Leaves, left to right: iterative preorder (push right first so left pops first).
        vector<TreeNode*> st{root};
        while (!st.empty()) {
            TreeNode* n = st.back(); st.pop_back();
            if (isLeaf(n)) out.push_back(n->val);
            if (n->right) st.push_back(n->right);
            if (n->left) st.push_back(n->left);
        }

        // 3. Right edge, collected top-down, appended bottom-up.
        vector<int> right;
        for (TreeNode* n = root->right; n && !isLeaf(n); n = n->right ? n->right : n->left)
            right.push_back(n->val);
        out.insert(out.end(), right.rbegin(), right.rend());
        return out;
    }
};
func isLeaf(n *TreeNode) bool { return n != nil && n.Left == nil && n.Right == nil }

func boundaryOfBinaryTree(root *TreeNode) []int {
	if root == nil {
		return []int{}
	}
	out := []int{root.Val}
	if isLeaf(root) {
		return out // a lone root is the whole boundary, not a leaf
	}

	// 1. Left edge, top-down: prefer left child, fall back to right. Leaves wait for step 2.
	for n := root.Left; n != nil && !isLeaf(n); {
		out = append(out, n.Val)
		if n.Left != nil {
			n = n.Left
		} else {
			n = n.Right
		}
	}

	// 2. Leaves, left to right: iterative preorder (push right first so left pops first).
	st := []*TreeNode{root}
	for len(st) > 0 {
		n := st[len(st)-1]
		st = st[:len(st)-1]
		if isLeaf(n) {
			out = append(out, n.Val)
		}
		if n.Right != nil {
			st = append(st, n.Right)
		}
		if n.Left != nil {
			st = append(st, n.Left)
		}
	}

	// 3. Right edge, collected top-down, appended bottom-up.
	var right []int
	for n := root.Right; n != nil && !isLeaf(n); {
		right = append(right, n.Val)
		if n.Right != nil {
			n = n.Right
		} else {
			n = n.Left
		}
	}
	for i := len(right) - 1; i >= 0; i-- {
		out = append(out, right[i])
	}
	return out
}

static class Solution {
    private static boolean isLeaf(TreeNode n) { return n != null && n.left == null && n.right == null; }

    public List<Integer> boundaryOfBinaryTree(TreeNode root) {
        List<Integer> out = new ArrayList<>();
        if (root == null) return out;
        out.add(root.val);
        if (isLeaf(root)) return out;            // a lone root is the whole boundary, not a leaf

        // 1. Left edge, top-down: prefer left child, fall back to right. Leaves wait for step 2.
        for (TreeNode n = root.left; n != null && !isLeaf(n); n = n.left != null ? n.left : n.right)
            out.add(n.val);

        // 2. Leaves, left to right: iterative preorder (push right first so left pops first).
        Deque<TreeNode> st = new ArrayDeque<>();
        st.push(root);
        while (!st.isEmpty()) {
            TreeNode n = st.pop();
            if (isLeaf(n)) out.add(n.val);
            if (n.right != null) st.push(n.right);
            if (n.left != null) st.push(n.left);
        }

        // 3. Right edge, collected top-down, appended bottom-up.
        List<Integer> right = new ArrayList<>();
        for (TreeNode n = root.right; n != null && !isLeaf(n); n = n.right != null ? n.right : n.left)
            right.add(n.val);
        Collections.reverse(right);
        out.addAll(right);
        return out;
    }
}
class Solution:
    def boundaryOfBinaryTree(self, root: Optional[TreeNode]) -> List[int]:
        if root is None:
            return []

        def is_leaf(n):
            return n is not None and n.left is None and n.right is None

        out = [root.val]
        if is_leaf(root):
            return out  # a lone root is the whole boundary, not a leaf

        # 1. Left edge, top-down: prefer left child, fall back to right. Leaves wait for step 2.
        n = root.left
        while n and not is_leaf(n):
            out.append(n.val)
            n = n.left or n.right

        # 2. Leaves, left to right: iterative preorder (push right first so left pops first).
        stack = [root]
        while stack:
            n = stack.pop()
            if is_leaf(n):
                out.append(n.val)
            if n.right:
                stack.append(n.right)
            if n.left:
                stack.append(n.left)

        # 3. Right edge, collected top-down, appended bottom-up.
        right = []
        n = root.right
        while n and not is_leaf(n):
            right.append(n.val)
            n = n.right or n.left
        return out + right[::-1]
