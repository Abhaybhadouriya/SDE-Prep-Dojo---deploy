func largestRectangleArea(heights []int) int {
	n, best := len(heights), 0
	st := make([]int, 0, n+1) // indices; heights increase bottom -> top
	// i == n is a virtual bar of height 0: it flushes every bar still waiting.
	for i := 0; i <= n; i++ {
		h := 0
		if i < n {
			h = heights[i]
		}
		// Bar i is lower (or equal): every taller bar on the stack has found its right wall.
		for len(st) > 0 && heights[st[len(st)-1]] >= h {
			height := heights[st[len(st)-1]]
			st = st[:len(st)-1]
			leftWall := -1 // first lower bar on the left
			if len(st) > 0 {
				leftWall = st[len(st)-1]
			}
			best = max(best, height*(i-leftWall-1))
		}
		st = append(st, i)
	}
	return best
}
class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
        int n = heights.size(), best = 0;
        vector<int> st;   // indices whose heights increase from bottom to top
        // i == n is a virtual bar of height 0: it flushes every bar still waiting.
        for (int i = 0; i <= n; ++i) {
            int h = (i == n) ? 0 : heights[i];
            // Bar i is lower (or equal): every taller bar on the stack has found its right wall.
            while (!st.empty() && heights[st.back()] >= h) {
                int height = heights[st.back()];
                st.pop_back();
                int leftWall = st.empty() ? -1 : st.back();   // first lower bar on the left
                best = max(best, height * (i - leftWall - 1));  // width between the two walls
            }
            st.push_back(i);
        }
        return best;   // max area <= 1e5 * 1e4 = 1e9, fits in int
    }
};

static class Solution {
    public int largestRectangleArea(int[] heights) {
        int n = heights.length, best = 0;
        int[] st = new int[n + 1];   // array stack of indices; heights increase bottom -> top
        int top = -1;

        for (int i = 0; i <= n; i++) {
            int h = (i == n) ? 0 : heights[i];
            // Bar i is lower (or equal): every taller bar on the stack has found its right wall.
            while (top >= 0 && heights[st[top]] >= h) {
                int height = heights[st[top--]];
                int leftWall = (top < 0) ? -1 : st[top];        // first lower bar on the left
                best = Math.max(best, height * (i - leftWall - 1)); // width between the walls
            }
            st[++top] = i;
        }
        return best;   // max area <= 1e5 * 1e4 = 1e9, fits in int
    }
}
class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        n, best = len(heights), 0
        st = []                                   # indices; heights increase bottom -> top
        # i == n is a virtual bar of height 0: it flushes every bar still waiting.
        for i in range(n + 1):
            h = 0 if i == n else heights[i]
            # Bar i is lower (or equal): every taller bar on the stack has found its right wall.
            while st and heights[st[-1]] >= h:
                height = heights[st.pop()]
                left_wall = st[-1] if st else -1  # first lower bar on the left
                best = max(best, height * (i - left_wall - 1))
            st.append(i)
        return best
