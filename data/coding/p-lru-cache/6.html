Hash map + hand-written doubly linked list with sentinels, in four languages; the tested files are in coding-code/lru-cache/. Go uses LeetCode's names Constructor, Get and Put.

class LRUCache {
    struct Node {
        int key, val;
        Node *prev, *next;
        Node(int k, int v) : key(k), val(v), prev(nullptr), next(nullptr) {}
    };
    int cap;
    unordered_map<int, Node*> where;   // key -> its node in the list
    Node head{0, 0}, tail{0, 0};       // sentinels: head.next = most recent, tail.prev = least recent

    void unlink(Node* n) {             // O(1): the node knows both neighbours
        n->prev->next = n->next;
        n->next->prev = n->prev;
    }
    void pushFront(Node* n) {          // insert right after the head sentinel
        n->prev = &head;
        n->next = head.next;
        head.next->prev = n;
        head.next = n;
    }

public:
    explicit LRUCache(int capacity) : cap(capacity) {
        head.next = &tail;
        tail.prev = &head;
    }
    ~LRUCache() {
        for (auto& kv : where) delete kv.second;
    }

    int get(int key) {
        auto it = where.find(key);
        if (it == where.end()) return -1;
        Node* n = it->second;
        unlink(n);                     // a read is a use: move to the front
        pushFront(n);
        return n->val;
    }

    void put(int key, int value) {
        auto it = where.find(key);
        if (it != where.end()) {       // update: new value, now most recent, size unchanged
            Node* n = it->second;
            n->val = value;
            unlink(n);
            pushFront(n);
            return;
        }
        if ((int)where.size() == cap) { // full: evict the node just before the tail
            Node* old = tail.prev;
            unlink(old);
            where.erase(old->key);     // the node stores its key, so the map entry can go too
            delete old;
        }
        Node* n = new Node(key, value);
        pushFront(n);
        where[key] = n;
    }
};
type node struct {
	key, val   int
	prev, next *node
}

type LRUCache struct {
	cap        int
	where      map[int]*node // key -> its node
	head, tail *node         // sentinels: head.next = most recent, tail.prev = least recent
}

func Constructor(capacity int) LRUCache {
	h, t := &node{}, &node{}
	h.next, t.prev = t, h
	return LRUCache{cap: capacity, where: make(map[int]*node), head: h, tail: t}
}

func (c *LRUCache) unlink(n *node) {
	n.prev.next = n.next
	n.next.prev = n.prev
}

func (c *LRUCache) pushFront(n *node) {
	n.prev, n.next = c.head, c.head.next
	c.head.next.prev = n
	c.head.next = n
}

func (c *LRUCache) Get(key int) int {
	n, ok := c.where[key]
	if !ok {
		return -1
	}
	c.unlink(n) // a read counts as a use
	c.pushFront(n)
	return n.val
}

func (c *LRUCache) Put(key, value int) {
	if n, ok := c.where[key]; ok { // update in place, then refresh
		n.val = value
		c.unlink(n)
		c.pushFront(n)
		return
	}
	if len(c.where) == c.cap { // evict the least recent node
		old := c.tail.prev
		c.unlink(old)
		delete(c.where, old.key)
	}
	n := &node{key: key, val: value}
	c.pushFront(n)
	c.where[key] = n
}

static class LRUCache {
    private static class Node {
        int key, val;
        Node prev, next;
        Node(int k, int v) { key = k; val = v; }
    }

    private final int cap;
    private final Map<Integer, Node> where = new HashMap<>(); // key -> its node
    private final Node head = new Node(0, 0), tail = new Node(0, 0); // sentinels

    LRUCache(int capacity) {
        cap = capacity;
        head.next = tail;   // head.next = most recent
        tail.prev = head;   // tail.prev = least recent
    }

    private void unlink(Node n) {
        n.prev.next = n.next;
        n.next.prev = n.prev;
    }

    private void pushFront(Node n) {
        n.prev = head;
        n.next = head.next;
        head.next.prev = n;
        head.next = n;
    }

    public int get(int key) {
        Node n = where.get(key);
        if (n == null) return -1;
        unlink(n);          // a read counts as a use
        pushFront(n);
        return n.val;
    }

    public void put(int key, int value) {
        Node n = where.get(key);
        if (n != null) {    // update in place, then refresh
            n.val = value;
            unlink(n);
            pushFront(n);
            return;
        }
        if (where.size() == cap) {  // evict the least recent node
            Node old = tail.prev;
            unlink(old);
            where.remove(old.key);
        }
        n = new Node(key, value);
        pushFront(n);
        where.put(key, n);
    }
}
class Node:
    __slots__ = ("key", "val", "prev", "next")

    def __init__(self, key=0, val=0):
        self.key, self.val = key, val
        self.prev = self.next = None


class LRUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.where = {}                        # key -> its node
        self.head, self.tail = Node(), Node()  # sentinels
        self.head.next, self.tail.prev = self.tail, self.head

    def _unlink(self, n):
        n.prev.next, n.next.prev = n.next, n.prev

    def _push_front(self, n):                  # head.next is the most recent
        n.prev, n.next = self.head, self.head.next
        self.head.next.prev = n
        self.head.next = n

    def get(self, key: int) -> int:
        n = self.where.get(key)
        if n is None:
            return -1
        self._unlink(n)                        # a read counts as a use
        self._push_front(n)
        return n.val

    def put(self, key: int, value: int) -> None:
        n = self.where.get(key)
        if n is not None:                      # update in place, then refresh
            n.val = value
            self._unlink(n)
            self._push_front(n)
            return
        if len(self.where) == self.cap:        # evict tail.prev, the least recent
            old = self.tail.prev
            self._unlink(old)
            del self.where[old.key]
        n = Node(key, value)
        self._push_front(n)
        self.where[key] = n
