func canPartition(nums []int) bool {
	total := 0
	for _, x := range nums {
		total += x
	}
	if total%2 != 0 {
		return false // odd total can never split evenly
	}
	target := total / 2

	// can[s] = some subset of the numbers seen so far sums to exactly s
	can := make([]bool, target+1)
	can[0] = true // the empty subset
	for _, x := range nums {
		// walk s DOWNWARD so this x is used at most once (0/1, not unbounded)
		for s := target; s >= x; s-- {
			if can[s-x] {
				can[s] = true
			}
		}
		if can[target] {
			return true // early exit
		}
	}
	return can[target]
}

class Solution:
    def canPartition(self, nums: List[int]) -> bool:
        total = sum(nums)
        if total % 2:                       # odd total can never split evenly
            return False
        target = total // 2

        # can[s] = some subset of the numbers seen so far sums to exactly s
        can = [False] * (target + 1)
        can[0] = True                       # the empty subset
        for x in nums:
            # walk s DOWNWARD so this x is used at most once (0/1, not unbounded)
            for s in range(target, x - 1, -1):
                if can[s - x]:
                    can[s] = True
            if can[target]:                 # early exit
                return True
        return can[target]
