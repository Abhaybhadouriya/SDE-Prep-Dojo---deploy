BFS with a parallel column value and a hash map. Each file in coding-code/binary-tree-vertical-order-traversal/ runs the three examples, the DFS-trap tree, an empty tree and a left chain; the Python file also compares 500 random trees with a DFS + sort reference.

class Solution {
public:
    vector<vector<int>> verticalOrder(TreeNode* root) {
        if (!root) return {};
        unordered_map<int, vector<int>> cols;     // col -> values in BFS order
        int lo = 0, hi = 0;                       // columns are a contiguous range [lo, hi]
        queue<pair<TreeNode*, int>> q;
        q.push({root, 0});
        while (!q.empty()) {
            TreeNode* node = q.front().first;
            int col = q.front().second;
            q.pop();
            // BFS pops rows top to bottom, and each row left to right: exactly the order we need.
            cols[col].push_back(node->val);
            lo = min(lo, col);
            hi = max(hi, col);
            if (node->left) q.push({node->left, col - 1});
            if (node->right) q.push({node->right, col + 1});
        }
        vector<vector<int>> out;
        for (int c = lo; c <= hi; c++) out.push_back(cols[c]);  // no sort: walk the range
        return out;
    }
};
func verticalOrder(root *TreeNode) [][]int {
	if root == nil {
		return [][]int{}
	}
	type item struct {
		node *TreeNode
		col  int
	}
	cols := map[int][]int{} // col -> values in BFS order
	lo, hi := 0, 0          // columns are a contiguous range [lo, hi]
	q := []item{{root, 0}}
	for len(q) > 0 {
		it := q[0]
		q = q[1:]
		// BFS pops rows top to bottom, and each row left to right: exactly the order we need.
		cols[it.col] = append(cols[it.col], it.node.Val)
		if it.col < lo {
			lo = it.col
		}
		if it.col > hi {
			hi = it.col
		}
		if it.node.Left != nil {
			q = append(q, item{it.node.Left, it.col - 1})
		}
		if it.node.Right != nil {
			q = append(q, item{it.node.Right, it.col + 1})
		}
	}
	out := make([][]int, 0, hi-lo+1)
	for c := lo; c <= hi; c++ { // no sort: walk the range
		out = append(out, cols[c])
	}
	return out
}

static class Solution {
    public List<List<Integer>> verticalOrder(TreeNode root) {
        List<List<Integer>> out = new ArrayList<>();
        if (root == null) return out;
        Map<Integer, List<Integer>> cols = new HashMap<>();  // col -> values in BFS order
        int lo = 0, hi = 0;                                  // columns are a contiguous range [lo, hi]
        Deque<TreeNode> nodes = new ArrayDeque<>();
        Deque<Integer> colOf = new ArrayDeque<>();           // parallel queue: column of each queued node
        nodes.add(root);
        colOf.add(0);
        while (!nodes.isEmpty()) {
            TreeNode node = nodes.poll();
            int col = colOf.poll();
            // BFS pops rows top to bottom, and each row left to right: exactly the order we need.
            cols.computeIfAbsent(col, k -> new ArrayList<>()).add(node.val);
            lo = Math.min(lo, col);
            hi = Math.max(hi, col);
            if (node.left != null) { nodes.add(node.left); colOf.add(col - 1); }
            if (node.right != null) { nodes.add(node.right); colOf.add(col + 1); }
        }
        for (int c = lo; c <= hi; c++) out.add(cols.get(c));  // no sort: walk the range
        return out;
    }
}
class Solution:
    def verticalOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if root is None:
            return []
        cols = defaultdict(list)  # col -> values in BFS order
        lo = hi = 0               # columns are a contiguous range [lo, hi]
        q = deque([(root, 0)])
        while q:
            node, col = q.popleft()
            # BFS pops rows top to bottom, and each row left to right: exactly the order we need.
            cols[col].append(node.val)
            lo, hi = min(lo, col), max(hi, col)
            if node.left:
                q.append((node.left, col - 1))
            if node.right:
                q.append((node.right, col + 1))
        return [cols[c] for c in range(lo, hi + 1)]  # no sort: walk the range
