class Solution {
    int best;
    // Best sum of a path that starts at node and goes DOWN (one side only).
    int arm(TreeNode* node) {
        if (!node) return 0;
        int left = max(arm(node->left), 0);       // a negative arm is dropped, not added
        int right = max(arm(node->right), 0);
        best = max(best, node->val + left + right);   // path that bends at this node
        return node->val + max(left, right);          // the parent may extend only one side
    }
public:
    int maxPathSum(TreeNode* root) {
        best = root->val;                         // a path has at least one node
        arm(root);
        return best;
    }
};
func maxPathSum(root *TreeNode) int {
	best := root.Val // a path has at least one node
	// arm = best sum of a path that starts at node and goes DOWN (one side only)
	var arm func(node *TreeNode) int
	arm = func(node *TreeNode) int {
		if node == nil {
			return 0
		}
		left := max(arm(node.Left), 0) // a negative arm is dropped, not added
		right := max(arm(node.Right), 0)
		best = max(best, node.Val+left+right) // path that bends at this node
		return node.Val + max(left, right)    // the parent may extend only one side
	}
	arm(root)
	return best
}
