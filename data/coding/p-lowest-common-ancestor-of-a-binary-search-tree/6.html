class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        int lo = min(p->val, q->val), hi = max(p->val, q->val);
        TreeNode* node = root;
        while (node) {
            if (hi < node->val) node = node->left;        // both targets smaller: split point is left
            else if (lo > node->val) node = node->right;  // both targets larger: go right
            else return node;     // lo <= val <= hi: p and q part ways here (or node is one of them)
        }
        return nullptr;           // not reached when p and q are in the tree
    }
};
func lowestCommonAncestor(root, p, q *TreeNode) *TreeNode {
	lo, hi := p.Val, q.Val
	if lo > hi {
		lo, hi = hi, lo
	}
	for node := root; node != nil; {
		switch {
		case hi < node.Val: // both targets smaller: split point is on the left
			node = node.Left
		case lo > node.Val: // both targets larger: go right
			node = node.Right
		default: // lo <= Val <= hi: p and q part ways here (or node is one of them)
			return node
		}
	}
	return nil // not reached when p and q are in the tree
}
