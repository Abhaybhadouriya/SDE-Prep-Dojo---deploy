The same early-return loop in four languages. Each file in coding-code/plus-one/ also runs the edge cases below; the Python and Java files check 3,000 random digit arrays (mostly 9s) against a big-integer reference.

class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        // From the least significant digit: a 9 becomes 0 and carries left,
        // any other digit absorbs the carry and we return at once.
        for (int i = (int)digits.size() - 1; i >= 0; --i) {
            if (digits[i] < 9) {
                ++digits[i];
                return digits;
            }
            digits[i] = 0;
        }
        // All nines: 99..9 + 1 = 100..0, one digit longer.
        digits.insert(digits.begin(), 1);
        return digits;
    }
};
func plusOne(digits []int) []int {
	// From the least significant digit: a 9 becomes 0 and carries left,
	// any other digit absorbs the carry and we return at once.
	for i := len(digits) - 1; i >= 0; i-- {
		if digits[i] < 9 {
			digits[i]++
			return digits
		}
		digits[i] = 0
	}
	// All nines: 1 followed by n zeros (make zero-fills the slice).
	longer := make([]int, len(digits)+1)
	longer[0] = 1
	return longer
}

class Solution:
    def plusOne(self, digits: List[int]) -> List[int]:
        # Walk from the least significant digit. A 9 becomes 0 and passes the carry left;
        # any other digit absorbs the carry and we are done.
        for i in range(len(digits) - 1, -1, -1):
            if digits[i] < 9:
                digits[i] += 1
                return digits
            digits[i] = 0
        # Every digit was 9 (999 -> 000): the answer is 1 followed by n zeros.
        return [1] + digits
