func verticalTraversal(root *TreeNode) [][]int {
	var cells [][3]int // one {col, row, val} per node
	var collect func(node *TreeNode, row, col int)
	collect = func(node *TreeNode, row, col int) {
		if node == nil {
			return
		}
		cells = append(cells, [3]int{col, row, node.Val})
		collect(node.Left, row+1, col-1)  // left child: one row down, one column left
		collect(node.Right, row+1, col+1) // right child: one row down, one column right
	}
	collect(root, 0, 0)
	// Column first, then top-to-bottom, then smaller value: the sort is the whole rule.
	sort.Slice(cells, func(i, j int) bool {
		a, b := cells[i], cells[j]
		if a[0] != b[0] {
			return a[0] < b[0]
		}
		if a[1] != b[1] {
			return a[1] < b[1]
		}
		return a[2] < b[2]
	})
	var out [][]int
	for i, c := range cells {
		if i == 0 || c[0] != cells[i-1][0] {
			out = append(out, []int{}) // a new column starts
		}
		out[len(out)-1] = append(out[len(out)-1], c[2])
	}
	return out
}
class Solution {
    vector<array<int, 3>> cells;  // one (col, row, val) triple per node

    void collect(TreeNode* node, int row, int col) {
        if (!node) return;
        cells.push_back({{col, row, node->val}});
        collect(node->left, row + 1, col - 1);   // left child: one row down, one column left
        collect(node->right, row + 1, col + 1);  // right child: one row down, one column right
    }

public:
    vector<vector<int>> verticalTraversal(TreeNode* root) {
        cells.clear();
        collect(root, 0, 0);
        // Sorting the triples IS the whole rule: column first, then top-to-bottom, then smaller value.
        sort(cells.begin(), cells.end());
        vector<vector<int>> out;
        for (size_t i = 0; i < cells.size(); i++) {
            if (i == 0 || cells[i][0] != cells[i - 1][0]) out.push_back({});  // a new column starts
            out.back().push_back(cells[i][2]);
        }
        return out;
    }
};

static class Solution {
    private final List<int[]> cells = new ArrayList<>();  // one {col, row, val} per node

    private void collect(TreeNode node, int row, int col) {
        if (node == null) return;
        cells.add(new int[]{col, row, node.val});
        collect(node.left, row + 1, col - 1);   // left child: one row down, one column left
        collect(node.right, row + 1, col + 1);  // right child: one row down, one column right
    }

    public List<List<Integer>> verticalTraversal(TreeNode root) {
        cells.clear();
        collect(root, 0, 0);
        // Column first, then top-to-bottom, then smaller value: the sort is the whole rule.
        cells.sort((a, b) -> a[0] != b[0] ? Integer.compare(a[0], b[0])
                           : a[1] != b[1] ? Integer.compare(a[1], b[1])
                           : Integer.compare(a[2], b[2]));
        List<List<Integer>> out = new ArrayList<>();
        for (int i = 0; i < cells.size(); i++) {
            if (i == 0 || cells.get(i)[0] != cells.get(i - 1)[0]) out.add(new ArrayList<>());
            out.get(out.size() - 1).add(cells.get(i)[2]);
        }
        return out;
    }
}
class Solution:
    def verticalTraversal(self, root: Optional[TreeNode]) -> List[List[int]]:
        cells = []  # one (col, row, val) per node
        stack = [(root, 0, 0)]  # iterative DFS: no recursion limit to worry about
        while stack:
            node, row, col = stack.pop()
            if node is None:
                continue
            cells.append((col, row, node.val))
            stack.append((node.left, row + 1, col - 1))   # left child: down one row, left one column
            stack.append((node.right, row + 1, col + 1))  # right child: down one row, right one column
        # Column first, then top-to-bottom, then smaller value: the tuple sort is the whole rule.
        cells.sort()
        out = []
        for i, (col, _, val) in enumerate(cells):
            if i == 0 or col != cells[i - 1][0]:
                out.append([])  # a new column starts
            out[-1].append(val)
        return out
