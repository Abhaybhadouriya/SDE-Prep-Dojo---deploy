package main

import (
	"fmt"
	"reflect"

// buildPrefix: P[i] = sum of the first i values, so sum(l..r) = P[r+1] - P[l]. O(n).
func buildPrefix(a []int) []int64 {
	p := make([]int64, len(a)+1)
	for i, x := range a {
		p[i+1] = p[i] + int64(x)
	}
	return p
}

func rangeSum(p []int64, l, r int) int64 { return p[r+1] - p[l] }

// countSubarraysSumK: how many subarrays sum to exactly k (negatives allowed). O(n).
func countSubarraysSumK(a []int, k int) int {
	seen := map[int64]int{0: 1} // the empty prefix
	var run int64
	ans := 0
	for _, x := range a {
		run += int64(x)
		ans += seen[run-int64(k)] // reading a missing key gives 0 and does not insert
		seen[run]++               // AFTER the lookup
	}
	return ans
}

// longestEqual01: longest subarray with as many 0s as 1s. O(n).
func longestEqual01(a []int) int {
	first := map[int]int{0: -1}
	bal, best := 0, 0
	for i, x := range a {
		if x == 1 {
			bal++
		} else {
			bal--
		}
		if j, ok := first[bal]; ok {
			if i-j > best {
				best = i - j
			}
		} else {
			first[bal] = i // keep the FIRST index only
		}
	}
	return best
}

// hasGoodSubarray: length >= 2 and sum a multiple of k (k >= 1). O(n).
func hasGoodSubarray(a []int, k int) bool {
	first := map[int]int{0: -1}
	run := 0
	for i, x := range a {
		run = (run + x) % k
		if j, ok := first[run]; ok {
			if i-j >= 2 {
				return true
			}
		} else {
			first[run] = i
		}
	}
	return false
}

// countDivisibleByK: subarrays whose sum is divisible by k, negatives allowed. O(n + k).
func countDivisibleByK(a []int, k int) int {
	cnt := make([]int, k)
	cnt[0] = 1
	run, ans := 0, 0
	for _, x := range a {
		run = ((run+x)%k + k) % k // Go % can be negative: normalise
		ans += cnt[run]
		cnt[run]++
	}
	return ans
}

// carPooling: difference array; trips[i] = {people, from, to}; riders leave at 'to'.
func carPooling(trips [][]int, capacity int) bool {
	diff := make([]int, 1002)
	for _, t := range trips {
		diff[t[1]] += t[0]
		diff[t[2]] -= t[0]
	}
	load := 0
	for _, d := range diff {
		load += d
		if load > capacity {
			return false
		}
	}
	return true
}

// rangeAdd: updates[i] = {l, r, v} adds v to every index l..r. O(n + q).
func rangeAdd(n int, updates [][]int) []int {
	diff := make([]int, n+1)
	for _, u := range updates {
		diff[u[0]] += u[2]
		diff[u[1]+1] -= u[2]
	}
	out := make([]int, n)
	run := 0
	for i := 0; i < n; i++ {
		run += diff[i]
		out[i] = run
	}
	return out
}

// countXorK: subarrays whose XOR is k. O(n).
func countXorK(a []int, k int) int {
	seen := map[int]int{0: 1}
	run, ans := 0, 0
	for _, x := range a {
		run ^= x
		ans += seen[run^k]
		seen[run]++
	}
	return ans
}

// buildPrefix2D: S[i][j] = sum of the top-left i x j block.
func buildPrefix2D(g [][]int) [][]int64 {
	R, C := len(g), len(g[0])
	s := make([][]int64, R+1)
	for i := range s {
		s[i] = make([]int64, C+1)
	}
	for i := 0; i < R; i++ {
		for j := 0; j < C; j++ {
			s[i+1][j+1] = int64(g[i][j]) + s[i][j+1] + s[i+1][j] - s[i][j]
		}
	}
	return s
}

func sumRegion(s [][]int64, r1, c1, r2, c2 int) int64 {
	return s[r2+1][c2+1] - s[r1][c2+1] - s[r2+1][c1] + s[r1][c1]
}

// Node is a binary tree node for Path Sum III.
type Node struct {
	Val         int
	Left, Right *Node
}

// pathSumCount: prefix map along the root-to-node path, undo on return.
func pathSumCount(root *Node, target int) int {
	seen := map[int64]int{0: 1}
	var dfs func(n *Node, run int64) int
	dfs = func(n *Node, run int64) int {
		if n == nil {
			return 0
		}
		run += int64(n.Val)
		found := seen[run-int64(target)]
		seen[run]++ // visible to this subtree only
		found += dfs(n.Left, run) + dfs(n.Right, run)
		seen[run]-- // backtrack
		return found
	}
	return dfs(root, 0)
}

func check(c bool, m string) {
	if !c {
		panic(m)
	}
}

func main() {
	p := buildPrefix([]int{3, -1, 4, 1, 5})
	check(rangeSum(p, 1, 3) == 4 && rangeSum(p, 0, 4) == 12 && rangeSum(p, 2, 2) == 4, "range")
	check(countSubarraysSumK([]int{1, 1, 1}, 2) == 2, "k1")
	check(countSubarraysSumK([]int{1, 2, 3}, 3) == 2, "k2")
	check(countSubarraysSumK([]int{3, 4, -7, 3, 1, 3, 1, -4}, 7) == 3, "k3")
	check(countSubarraysSumK([]int{0, 0, 0}, 0) == 6, "k4")
	check(longestEqual01([]int{0, 1}) == 2 && longestEqual01([]int{0, 1, 0}) == 2, "01a")
	check(longestEqual01([]int{0, 0, 1, 0, 0, 0, 1, 1}) == 6, "01b")
	check(hasGoodSubarray([]int{23, 2, 4, 6, 7}, 6), "g1")
	check(!hasGoodSubarray([]int{23, 2, 6, 4, 7}, 13), "g2")
	check(hasGoodSubarray([]int{5, 0, 0, 0}, 3), "g3")
	check(!hasGoodSubarray([]int{0}, 1), "g4")
	check(countDivisibleByK([]int{4, 5, 0, -2, -3, 1}, 5) == 7, "d1")
	check(countDivisibleByK([]int{-1, 2, 9}, 2) == 2, "d2")
	check(!carPooling([][]int{{2, 1, 5}, {3, 3, 7}}, 4), "c1")
	check(carPooling([][]int{{2, 1, 5}, {3, 3, 7}}, 5), "c2")
	check(carPooling([][]int{{2, 1, 5}, {3, 5, 7}}, 3), "c3")
	check(reflect.DeepEqual(rangeAdd(5, [][]int{{1, 3, 2}, {2, 4, 3}, {0, 2, -2}}), []int{-2, 0, 3, 5, 3}), "ra")
	check(countXorK([]int{4, 2, 2, 6, 4}, 6) == 4, "x1")
	check(countXorK([]int{5, 6, 7, 8, 9}, 5) == 2, "x2")
	s := buildPrefix2D([][]int{{3, 0, 1, 4, 2}, {5, 6, 3, 2, 1}, {1, 2, 0, 1, 5}, {4, 1, 0, 1, 7}, {1, 0, 3, 0, 5}})
	check(sumRegion(s, 2, 1, 4, 3) == 8 && sumRegion(s, 1, 1, 2, 2) == 11 && sumRegion(s, 1, 2, 2, 4) == 12, "2d")
	tree := &Node{10, &Node{5, &Node{3, &Node{Val: 3}, &Node{Val: -2}}, &Node{2, nil, &Node{Val: 1}}}, &Node{-3, nil, &Node{Val: 11}}}
	check(pathSumCount(tree, 8) == 3, "p1")
	check(pathSumCount(nil, 0) == 0, "p2")
	check(pathSumCount(&Node{1, &Node{Val: -1}, &Node{Val: -1}}, 0) == 2, "p3")

}
// Prefix sum templates (SDE Prep Dojo, lesson coding-prefix-sum). C++14.
// Build: g++ -std=c++14 -O2 prefix_sum.cpp -o ps && ./ps   (asserts; prints "ok")
#include <cassert>
#include <cstdio>
#include <unordered_map>
#include <vector>
using namespace std;
typedef long long ll;

// P[i] = sum of the first i values, so sum(l..r) = P[r+1] - P[l]. O(n).
vector<ll> buildPrefix(const vector<int>& a) {
    vector<ll> p(a.size() + 1, 0);
    for (size_t i = 0; i < a.size(); i++) p[i + 1] = p[i] + a[i];
    return p;
}
ll rangeSum(const vector<ll>& p, int l, int r) { return p[r + 1] - p[l]; }   // O(1)

// How many subarrays sum to exactly k (negatives allowed). O(n) time, O(n) space.
int countSubarraysSumK(const vector<int>& a, int k) {
    unordered_map<ll, int> seen;
    seen[0] = 1;                                  // the empty prefix
    ll run = 0; int ans = 0;
    for (int x : a) {
        run += x;
        auto it = seen.find(run - k);             // find, not [], so we do not insert keys
        if (it != seen.end()) ans += it->second;
        seen[run]++;                              // AFTER the lookup
    }
    return ans;
}

// Longest subarray with as many 0s as 1s: 0 -> -1, longest sum 0. O(n).
int longestEqual01(const vector<int>& a) {
    unordered_map<int, int> first;                // balance -> first index
    first[0] = -1;
    int bal = 0, best = 0;
    for (int i = 0; i < (int)a.size(); i++) {
        bal += a[i] == 1 ? 1 : -1;
        auto it = first.find(bal);
        if (it != first.end()) best = max(best, i - it->second);
        else first[bal] = i;                      // keep the FIRST index only
    }
    return best;
}

// Subarray of length >= 2 whose sum is a multiple of k (k >= 1). O(n).
bool hasGoodSubarray(const vector<int>& a, int k) {
    unordered_map<int, int> first;
    first[0] = -1;
    int run = 0;
    for (int i = 0; i < (int)a.size(); i++) {
        run = (run + a[i]) % k;                   // values are >= 0 in this problem
        auto it = first.find(run);
        if (it != first.end()) { if (i - it->second >= 2) return true; }
        else first[run] = i;
    }
    return false;
}

// Subarrays whose sum is divisible by k, negatives allowed. O(n + k).
int countDivisibleByK(const vector<int>& a, int k) {
    vector<int> cnt(k, 0);
    cnt[0] = 1;
    int run = 0, ans = 0;
    for (int x : a) {
        run = ((run + x) % k + k) % k;            // C++ % can be negative: normalise
        ans += cnt[run]++;
    }
    return ans;
}

// Difference array: trips[i] = {people, from, to}; riders leave at 'to'. O(n + maxStop).
bool carPooling(const vector<vector<int>>& trips, int capacity) {
    vector<int> diff(1002, 0);
    for (const auto& t : trips) { diff[t[1]] += t[0]; diff[t[2]] -= t[0]; }
    int load = 0;
    for (int d : diff) { load += d; if (load > capacity) return false; }
    return true;
}

// updates[i] = {l, r, v}: add v to every index l..r. O(n + q).
vector<int> rangeAdd(int n, const vector<vector<int>>& updates) {
    vector<int> diff(n + 1, 0), out(n);
    for (const auto& u : updates) { diff[u[0]] += u[2]; diff[u[1] + 1] -= u[2]; }
    int run = 0;
    for (int i = 0; i < n; i++) { run += diff[i]; out[i] = run; }
    return out;
}

// Subarrays whose XOR is k: same map trick, '-' becomes '^'. O(n).
int countXorK(const vector<int>& a, int k) {
    unordered_map<int, int> seen;
    seen[0] = 1;
    int run = 0, ans = 0;
    for (int x : a) {
        run ^= x;
        auto it = seen.find(run ^ k);
        if (it != seen.end()) ans += it->second;
        seen[run]++;
    }
    return ans;
}

// 2D prefix: S[i][j] = sum of the top-left i x j block.
struct Prefix2D {
    vector<vector<ll>> s;
    explicit Prefix2D(const vector<vector<int>>& g) {
        int R = g.size(), C = g[0].size();
        s.assign(R + 1, vector<ll>(C + 1, 0));
        for (int i = 0; i < R; i++)
            for (int j = 0; j < C; j++)
                s[i + 1][j + 1] = g[i][j] + s[i][j + 1] + s[i + 1][j] - s[i][j];
    }
    ll sum(int r1, int c1, int r2, int c2) const {
        return s[r2 + 1][c2 + 1] - s[r1][c2 + 1] - s[r2 + 1][c1] + s[r1][c1];
    }
};

// Path Sum III: prefix map along the root-to-node path, undo on return.
struct Node { int val; Node *left, *right; Node(int v, Node* l = nullptr, Node* r = nullptr) : val(v), left(l), right(r) {} };
static int dfsPath(Node* n, ll run, int target, unordered_map<ll, int>& seen) {
    if (!n) return 0;
    run += n->val;
    auto it = seen.find(run - target);
    int found = it == seen.end() ? 0 : it->second;
    seen[run]++;                                  // visible to this subtree only
    found += dfsPath(n->left, run, target, seen) + dfsPath(n->right, run, target, seen);
    seen[run]--;                                  // backtrack
    return found;
}
int pathSumCount(Node* root, int target) {
    unordered_map<ll, int> seen;
    seen[0] = 1;
    return dfsPath(root, 0, target, seen);
}

// Prefix sum templates (SDE Prep Dojo, lesson coding-prefix-sum). Java 21.
// Run: java PrefixSum.java   (checks; prints "ok")
import java.util.*;

public class PrefixSum {
    // P[i] = sum of the first i values, so sum(l..r) = P[r+1] - P[l]. O(n).
    static long[] buildPrefix(int[] a) {
        long[] p = new long[a.length + 1];
        for (int i = 0; i < a.length; i++) p[i + 1] = p[i] + a[i];
        return p;
    }
    static long rangeSum(long[] p, int l, int r) { return p[r + 1] - p[l]; }

    // How many subarrays sum to exactly k (negatives allowed). O(n) time, O(n) space.
    static int countSubarraysSumK(int[] a, int k) {
        Map<Long, Integer> seen = new HashMap<>();
        seen.put(0L, 1);                                   // the empty prefix
        long run = 0; int ans = 0;
        for (int x : a) {
            run += x;
            ans += seen.getOrDefault(run - k, 0);
            seen.merge(run, 1, Integer::sum);              // AFTER the lookup
        }
        return ans;
    }

    // Longest subarray with as many 0s as 1s: 0 -> -1, longest sum 0. O(n).
    static int longestEqual01(int[] a) {
        Map<Integer, Integer> first = new HashMap<>();
        first.put(0, -1);
        int bal = 0, best = 0;
        for (int i = 0; i < a.length; i++) {
            bal += a[i] == 1 ? 1 : -1;
            Integer j = first.get(bal);
            if (j != null) best = Math.max(best, i - j);
            else first.put(bal, i);                        // keep the FIRST index only
        }
        return best;
    }

    // Subarray of length >= 2 whose sum is a multiple of k (k >= 1). O(n).
    static boolean hasGoodSubarray(int[] a, int k) {
        Map<Integer, Integer> first = new HashMap<>();
        first.put(0, -1);
        int run = 0;
        for (int i = 0; i < a.length; i++) {
            run = (run + a[i]) % k;
            Integer j = first.get(run);
            if (j != null) { if (i - j >= 2) return true; }
            else first.put(run, i);
        }
        return false;
    }

    // Subarrays whose sum is divisible by k, negatives allowed. O(n + k).
    static int countDivisibleByK(int[] a, int k) {
        int[] cnt = new int[k];
        cnt[0] = 1;
        int run = 0, ans = 0;
        for (int x : a) {
            run = Math.floorMod(run + x, k);               // Java % can be negative
            ans += cnt[run]++;
        }
        return ans;
    }

    // Difference array: trips[i] = {people, from, to}; riders leave at 'to'.
    static boolean carPooling(int[][] trips, int capacity) {
        int[] diff = new int[1002];
        for (int[] t : trips) { diff[t[1]] += t[0]; diff[t[2]] -= t[0]; }
        int load = 0;
        for (int d : diff) { load += d; if (load > capacity) return false; }
        return true;
    }

    // updates[i] = {l, r, v}: add v to every index l..r. O(n + q).
    static int[] rangeAdd(int n, int[][] updates) {
        int[] diff = new int[n + 1], out = new int[n];
        for (int[] u : updates) { diff[u[0]] += u[2]; diff[u[1] + 1] -= u[2]; }
        int run = 0;
        for (int i = 0; i < n; i++) { run += diff[i]; out[i] = run; }
        return out;
    }

    // Subarrays whose XOR is k. O(n).
    static int countXorK(int[] a, int k) {
        Map<Integer, Integer> seen = new HashMap<>();
        seen.put(0, 1);
        int run = 0, ans = 0;
        for (int x : a) {
            run ^= x;
            ans += seen.getOrDefault(run ^ k, 0);
            seen.merge(run, 1, Integer::sum);
        }
        return ans;
    }

    // 2D prefix: S[i][j] = sum of the top-left i x j block.
    static long[][] buildPrefix2D(int[][] g) {
        int R = g.length, C = g[0].length;
        long[][] s = new long[R + 1][C + 1];
        for (int i = 0; i < R; i++)
            for (int j = 0; j < C; j++)
                s[i + 1][j + 1] = g[i][j] + s[i][j + 1] + s[i + 1][j] - s[i][j];
        return s;
    }
    static long sumRegion(long[][] s, int r1, int c1, int r2, int c2) {
        return s[r2 + 1][c2 + 1] - s[r1][c2 + 1] - s[r2 + 1][c1] + s[r1][c1];
    }

    // Path Sum III: prefix map along the root-to-node path, undo on return.
    record Node(int val, Node left, Node right) { Node(int v) { this(v, null, null); } }
    static int pathSumCount(Node root, int target) {
        Map<Long, Integer> seen = new HashMap<>();
        seen.put(0L, 1);
        return dfs(root, 0, target, seen);
    }
    private static int dfs(Node n, long run, int target, Map<Long, Integer> seen) {
        if (n == null) return 0;
        run += n.val();
        int found = seen.getOrDefault(run - target, 0);
        seen.merge(run, 1, Integer::sum);                 // visible to this subtree only
        found += dfs(n.left(), run, target, seen) + dfs(n.right(), run, target, seen);
        seen.merge(run, -1, Integer::sum);                // backtrack
        return found;
    }

    static void check(boolean c, String m) { if (!c) throw new AssertionError(m); }

    public static void main(String[] args) {
        long[] p = buildPrefix(new int[]{3, -1, 4, 1, 5});
        check(rangeSum(p, 1, 3) == 4 && rangeSum(p, 0, 4) == 12 && rangeSum(p, 2, 2) == 4, "range");
        check(countSubarraysSumK(new int[]{1, 1, 1}, 2) == 2, "k1");
        check(countSubarraysSumK(new int[]{1, 2, 3}, 3) == 2, "k2");
        check(countSubarraysSumK(new int[]{3, 4, -7, 3, 1, 3, 1, -4}, 7) == 3, "k3");
        check(countSubarraysSumK(new int[]{0, 0, 0}, 0) == 6, "k4");
        check(longestEqual01(new int[]{0, 1}) == 2 && longestEqual01(new int[]{0, 1, 0}) == 2, "01a");
        check(longestEqual01(new int[]{0, 0, 1, 0, 0, 0, 1, 1}) == 6, "01b");
        check(hasGoodSubarray(new int[]{23, 2, 4, 6, 7}, 6), "g1");
        check(!hasGoodSubarray(new int[]{23, 2, 6, 4, 7}, 13), "g2");
        check(hasGoodSubarray(new int[]{5, 0, 0, 0}, 3), "g3");
        check(!hasGoodSubarray(new int[]{0}, 1), "g4");
        check(countDivisibleByK(new int[]{4, 5, 0, -2, -3, 1}, 5) == 7, "d1");
        check(countDivisibleByK(new int[]{-1, 2, 9}, 2) == 2, "d2");
        check(!carPooling(new int[][]{{2, 1, 5}, {3, 3, 7}}, 4), "c1");
        check(carPooling(new int[][]{{2, 1, 5}, {3, 3, 7}}, 5), "c2");
        check(carPooling(new int[][]{{2, 1, 5}, {3, 5, 7}}, 3), "c3");
        check(Arrays.equals(rangeAdd(5, new int[][]{{1, 3, 2}, {2, 4, 3}, {0, 2, -2}}), new int[]{-2, 0, 3, 5, 3}), "ra");

"""Prefix sum templates (SDE Prep Dojo, lesson coding-prefix-sum).

Run: python prefix_sum.py   (asserts; prints "ok")
"""
from collections import defaultdict


def build_prefix(nums):
    """P[i] = sum of the first i values, so sum(l..r) = P[r+1] - P[l]. O(n)."""
    p = [0] * (len(nums) + 1)
    for i, x in enumerate(nums):
        p[i + 1] = p[i] + x
    return p


def range_sum(p, l, r):
    """Sum of nums[l..r], both ends included. O(1)."""
    return p[r + 1] - p[l]


def count_subarrays_sum_k(nums, k):
    """How many subarrays sum to exactly k (negatives allowed). O(n) time, O(n) space."""
    seen = defaultdict(int)
    seen[0] = 1                         # the empty prefix: lets a subarray start at index 0
    run = ans = 0
    for x in nums:
        run += x
        ans += seen[run - k]            # earlier prefixes that leave exactly k
        seen[run] += 1                  # count AFTER the lookup (k = 0 would count itself)
    return ans


def longest_equal_01(nums):
    """Longest subarray with as many 0s as 1s: map 0 -> -1, longest sum 0. O(n)."""
    first = {0: -1}                     # balance -> first index where it appeared
    bal = best = 0
    for i, x in enumerate(nums):
        bal += 1 if x == 1 else -1
        if bal in first:
            best = max(best, i - first[bal])
        else:
            first[bal] = i              # keep the FIRST index only: longest wins
    return best


def has_good_subarray(nums, k):
    """Subarray of length >= 2 whose sum is a multiple of k (k >= 1). O(n)."""
    first = {0: -1}
    run = 0
    for i, x in enumerate(nums):
        run = (run + x) % k
        if run in first:
            if i - first[run] >= 2:
                return True
        else:
            first[run] = i
    return False


def count_divisible_by_k(nums, k):
    """Subarrays whose sum is divisible by k, negatives allowed. O(n + k)."""
    cnt = [0] * k
    cnt[0] = 1
    run = ans = 0
    for x in nums:
        run = (run + x) % k             # Python % is already >= 0; C++/Java/Go need +k
        ans += cnt[run]
        cnt[run] += 1
    return ans


def car_pooling(trips, capacity):
    """Difference array: trips[i] = [people, from, to]; riders leave at 'to'. O(n + maxStop)."""
    diff = [0] * 1002                   # stops are 0..1000
    for people, frm, to in trips:
        diff[frm] += people             # they get on at frm
        diff[to] -= people              # and are gone at to
    load = 0
    for d in diff:
        load += d                       # prefix sum of the differences = load at this stop
        if load > capacity:
            return False
    return True


def range_add(n, updates):
    """updates[i] = [l, r, v]: add v to every index l..r. Returns the final array. O(n + q)."""
    diff = [0] * (n + 1)
    for l, r, v in updates:
        diff[l] += v
        diff[r + 1] -= v
    out, run = [], 0
    for i in range(n):
        run += diff[i]
        out.append(run)
    return out


def count_xor_k(nums, k):
    """Subarrays whose XOR is k: same map trick, '-' becomes '^'. O(n)."""
    seen = defaultdict(int)
    seen[0] = 1
    run = ans = 0
    for x in nums:
        run ^= x
        ans += seen[run ^ k]
        seen[run] += 1
    return ans


def xor_queries(arr, queries):
    """XOR of arr[l..r] for each query: X[r+1] ^ X[l]. O(n + q)."""
    x = [0] * (len(arr) + 1)
    for i, v in enumerate(arr):
        x[i + 1] = x[i] ^ v
    return [x[r + 1] ^ x[l] for l, r in queries]


def build_prefix_2d(grid):
    """S[i][j] = sum of the top-left i x j block. Query by inclusion-exclusion."""
    rows, cols = len(grid), len(grid[0])
    s = [[0] * (cols + 1) for _ in range(rows + 1)]
    for i in range(rows):
        for j in range(cols):
            s[i + 1][j + 1] = grid[i][j] + s[i][j + 1] + s[i + 1][j] - s[i][j]
    return s


def sum_region(s, r1, c1, r2, c2):
    return s[r2 + 1][c2 + 1] - s[r1][c2 + 1] - s[r2 + 1][c1] + s[r1][c1]


class Node:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right


def path_sum_count(root, target):
    """Downward paths summing to target: prefix map along the root-to-node path, undo on return."""
    seen = defaultdict(int)
    seen[0] = 1

    def dfs(node, run):
        if node is None:
            return 0
        run += node.val
        found = seen[run - target]
        seen[run] += 1                  # this node's prefix is visible to its subtree only
        found += dfs(node.left, run) + dfs(node.right, run)
        seen[run] -= 1                  # backtrack: siblings must not see it
        return found

    return dfs(root, 0)
