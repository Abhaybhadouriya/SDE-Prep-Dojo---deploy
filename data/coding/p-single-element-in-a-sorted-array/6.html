func singleNonDuplicate(nums []int) int {
	// Before the single value, every pair starts at an EVEN index.
	// From the single value on, every pair starts at an ODD index.
	// lo and hi stay even, so the answer is always inside [lo, hi].
	lo, hi := 0, len(nums)-1
	for lo < hi {
		mid := lo + (hi-lo)/2
		if mid%2 == 1 {
			mid-- // look at the start of a pair
		}
		if nums[mid] == nums[mid+1] {
			lo = mid + 2 // pair intact: single is to the right
		} else {
			hi = mid // pair broken: single is mid or left of it
		}
	}
	return nums[lo]
}

class Solution:
    def singleNonDuplicate(self, nums: List[int]) -> int:
        # Before the single value, every pair starts at an EVEN index.
        # From the single value on, every pair starts at an ODD index.
        # lo and hi stay even, so the answer is always inside [lo, hi].
        lo, hi = 0, len(nums) - 1
        while lo < hi:
            mid = (lo + hi) // 2
            if mid % 2 == 1:
                mid -= 1                    # look at the start of a pair
            if nums[mid] == nums[mid + 1]:
                lo = mid + 2                # pair intact: single is to the right
            else:
                hi = mid                    # pair broken: single is mid or left of it
        return nums[lo]
