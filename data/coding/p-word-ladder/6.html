class Solution {
public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        unordered_set<string> unused(wordList.begin(), wordList.end());
        if (!unused.count(endWord)) return 0;          // the target must be in the dictionary
        unused.erase(beginWord);
        unused.erase(endWord);
        unordered_set<string> front{beginWord}, back{endWord};
        int words = 1;          // words on the shortest path between the two frontiers so far
        while (!front.empty() && !back.empty()) {
            if (front.size() > back.size()) swap(front, back);   // always grow the smaller side
            unordered_set<string> next;
            for (string w : front) {                   // copy: we edit w in place
                for (size_t i = 0; i < w.size(); i++) {
                    char keep = w[i];
                    for (char c = 'a'; c <= 'z'; c++) {
                        if (c == keep) continue;
                        w[i] = c;
                        if (back.count(w)) return words + 1;      // the frontiers touch
                        if (unused.count(w)) {                    // first visit: claim it now
                            unused.erase(w);
                            next.insert(w);
                        }
                    }
                    w[i] = keep;
                }
            }
            front.swap(next);
            words++;
        }
        return 0;                                      // one side ran dry: no ladder
    }
};
func ladderLength(beginWord string, endWord string, wordList []string) int {
	unused := make(map[string]bool, len(wordList))
	for _, w := range wordList {
		unused[w] = true
	}
	if !unused[endWord] { // the target must be in the dictionary
		return 0
	}
	delete(unused, beginWord)
	delete(unused, endWord)
	front := map[string]bool{beginWord: true}
	back := map[string]bool{endWord: true}
	words := 1 // words on the shortest path between the two frontiers so far
	for len(front) > 0 && len(back) > 0 {
		if len(front) > len(back) { // always grow the smaller side
			front, back = back, front
		}
		next := map[string]bool{}
		for word := range front {
			w := []byte(word)
			for i := range w {
				keep := w[i]
				for c := byte('a'); c <= 'z'; c++ {
					if c == keep {
						continue
					}
					w[i] = c
					cand := string(w)
					if back[cand] { // the frontiers touch
						return words + 1
					}
					if unused[cand] { // first visit: claim it now
						delete(unused, cand)
						next[cand] = true
					}
				}
				w[i] = keep
			}
		}
		front = next
		words++
	}
	return 0 // one side ran dry: no ladder
}

class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        unused = set(wordList)
        if endWord not in unused:                  # the target must be in the dictionary
            return 0
        unused.discard(beginWord)
        unused.discard(endWord)
        front, back = {beginWord}, {endWord}
        words = 1          # words on the shortest path between the two frontiers so far
        while front and back:
            if len(front) > len(back):             # always grow the smaller side
                front, back = back, front
            nxt = set()
            for w in front:
                for i in range(len(w)):
                    for c in string.ascii_lowercase:
                        if c == w[i]:
                            continue
                        cand = w[:i] + c + w[i + 1:]
                        if cand in back:           # the frontiers touch
                            return words + 1
                        if cand in unused:         # first visit: claim it now
                            unused.remove(cand)
                            nxt.add(cand)
            front = nxt
            words += 1
        return 0                                   # one side ran dry: no ladder
