func levelOrder(root *TreeNode) [][]int {
	levels := [][]int{}
	if root == nil {
		return levels // empty tree -> []
	}
	q := []*TreeNode{root}
	for len(q) > 0 {
		width := len(q) // freeze: exactly this many nodes form the level
		row := make([]int, 0, width)
		for _, node := range q[:width] {
			row = append(row, node.Val)
			if node.Left != nil { // children join the NEXT level
				q = append(q, node.Left)
			}
			if node.Right != nil {
				q = append(q, node.Right)
			}
		}
		q = q[width:] // drop the level we just read
		levels = append(levels, row)
	}
	return levels
}
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> levels;
        if (!root) return levels;              // empty tree -> []
        queue<TreeNode*> q;
        q.push(root);
        while (!q.empty()) {
            int width = q.size();              // freeze: exactly this many nodes form the level
            vector<int> row;
            row.reserve(width);
            for (int i = 0; i < width; i++) {
                TreeNode* node = q.front();
                q.pop();
                row.push_back(node->val);
                if (node->left) q.push(node->left);   // children join the NEXT level
                if (node->right) q.push(node->right);
            }
            levels.push_back(row);
        }
        return levels;
    }
};

static class Solution {
    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> levels = new ArrayList<>();
        if (root == null) return levels;            // empty tree -> []
        Deque<TreeNode> q = new ArrayDeque<>();
        q.offer(root);
        while (!q.isEmpty()) {
            int width = q.size();                   // freeze: exactly this many nodes form the level
            List<Integer> row = new ArrayList<>(width);
            for (int i = 0; i < width; i++) {
                TreeNode node = q.poll();
                row.add(node.val);
                if (node.left != null) q.offer(node.left);   // children join the NEXT level
                if (node.right != null) q.offer(node.right);
            }
            levels.add(row);
        }
        return levels;
    }
}
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:                       # empty tree -> []
            return []
        levels, q = [], deque([root])
        while q:
            row = []
            for _ in range(len(q)):        # len(q) is frozen: exactly this level's nodes
                node = q.popleft()
                row.append(node.val)
                if node.left:              # children join the NEXT level
                    q.append(node.left)
                if node.right:
                    q.append(node.right)
            levels.append(row)
        return levels
