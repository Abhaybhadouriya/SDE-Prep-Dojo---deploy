The two-pass, O(1)-extra-space solution in four languages. Each file in coding-code/product-of-array-except-self/ also cross-checks against the brute force on 300 random arrays with zeros and negatives.

class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n = (int)nums.size();
        vector<int> out(n, 1);
        // pass 1: out[i] = product of everything LEFT of i
        int left = 1;
        for (int i = 0; i < n; i++) {
            out[i] = left;
            left *= nums[i];
        }
        // pass 2: multiply in the product of everything RIGHT of i
        int right = 1;
        for (int i = n - 1; i >= 0; i--) {
            out[i] *= right;
            right *= nums[i];
        }
        return out;
    }
};
