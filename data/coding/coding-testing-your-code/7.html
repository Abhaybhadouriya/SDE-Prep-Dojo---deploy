Stress testing at home: brute force as the oracle

In practice sessions, and in interviews on an editor that runs code, the fastest way to find a bug you cannot see is a stress test: generate thousands of tiny random inputs, run your fast solution and an obviously correct brute-force one, and stop at the first input where they disagree. Tiny inputs (n up to 8, values from -5 to 5) are enough to hit almost every boundary, and the failing input is small enough to trace by hand.

The template has three parts you can reuse for any problem: a table of named edge cases with expected answers, a brute-force oracle, and a seeded random loop (a fixed seed means a failure can be replayed). It tests the two functions from this page: lower_bound and Kadane's maximum subarray. All four versions (Go, C++, Java, Python) were compiled and run (all table tests pass, 2,000 random cases agree).

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

package main

import (
	"fmt"
	"math/rand"
	"os"
	"sort"
)

// lowerBound returns the first index i with a[i] >= x, or len(a) if every value is < x.
func lowerBound(a []int, x int) int {
	lo, hi := 0, len(a) // [lo, hi): hi is one past the end
	for lo < hi {
		mid := lo + (hi-lo)/2
		if a[mid] < x {
			lo = mid + 1 // mid too small: answer is right of it
		} else {
			hi = mid // mid may be the answer: keep it
		}
	}
	return lo
}

// maxSubarray is Kadane: largest sum of a non-empty contiguous run.
func maxSubarray(a []int) int {
	best, cur := a[0], a[0] // NOT 0: all-negative input
	for _, v := range a[1:] {
		cur = max(v, cur+v)
		best = max(best, cur)
	}
	return best
}

func bruteMaxSubarray(a []int) int { // O(n^2) oracle
	best := a[0]
	for i := range a {
		s := 0
		for j := i; j < len(a); j++ {
			s += a[j]
			best = max(best, s)
		}
	}
	return best
}

func check(name string, got, want int) {
	if got != want {
		fmt.Printf("FAIL %s: got %d, want %d\n", name, got, want)
		os.Exit(1)
	}
	fmt.Println("ok  ", name)
}

func tableTests() {
	cases := []struct {
		name    string
		a       []int
		x, want int
	}{
		{"empty", []int{}, 5, 0}, {"one, equal", []int{5}, 5, 0},
		{"one, x smaller", []int{5}, 1, 0}, {"one, x bigger", []int{5}, 9, 1},
		{"all duplicates", []int{2, 2, 2, 2}, 2, 0}, {"absent, middle", []int{1, 3, 5, 7}, 4, 2},
		{"negatives", []int{-9, -4, -4, 0}, -4, 1}, {"past the end", []int{1, 3}, 99, 2},
	}
	for _, c := range cases {
		check("lower_bound "+c.name, lowerBound(c.a, c.x), c.want)
	}
	check("kadane all negative", maxSubarray([]int{-3, -1, -2}), -1)
	check("kadane one element", maxSubarray([]int{7}), 7)
	check("kadane mixed", maxSubarray([]int{2, -5, 3, -1, 4}), 6)
}

func stress(rounds int, seed int64) bool {
	rng := rand.New(rand.NewSource(seed)) // fixed seed: replayable
	for r := 0; r < rounds; r++ {
		a := make([]int, 1+rng.Intn(8))
		for i := range a {
			a[i] = rng.Intn(11) - 5
		}
		if maxSubarray(a) != bruteMaxSubarray(a) {
			fmt.Println("FAIL kadane on", a)
			return false
		}
		sort.Ints(a)
		x, want := rng.Intn(13)-6, len(a)
		for i, v := range a {
			if v >= x {
				want = i
				break
			}
		}
		if lowerBound(a, x) != want {
			fmt.Println("FAIL lower_bound on", a, x)
			return false
		}
	}
	fmt.Println("stress ok:", rounds, "random cases")
	return true
}

func main() {
	tableTests()
	if !stress(2000, 42) {
		os.Exit(1)
	}
}
#include <bits/stdc++.h>
using namespace std;

// First index i with a[i] >= x; a.size() if every value is < x.
int lowerBound(const vector<int>& a, int x) {
    int lo = 0, hi = (int)a.size();          // [lo, hi): hi is one past the end
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;        // no (lo + hi) overflow
        if (a[mid] < x) lo = mid + 1;        // mid too small: answer is right of it
        else hi = mid;                       // mid may be the answer: keep it
    }
    return lo;
}

// Largest sum of a non-empty contiguous run (Kadane). 64-bit sums.
long long maxSubarray(const vector<int>& a) {
    long long best = a[0], cur = a[0];       // NOT 0: all-negative input
    for (size_t i = 1; i < a.size(); i++) {
        cur = max((long long)a[i], cur + a[i]);
        best = max(best, cur);
    }
    return best;
}

long long bruteMaxSubarray(const vector<int>& a) {   // O(n^2) oracle
    long long best = LLONG_MIN;
    for (size_t i = 0; i < a.size(); i++) {
        long long s = 0;
        for (size_t j = i; j < a.size(); j++) { s += a[j]; best = max(best, s); }
    }
    return best;
}

template <class T> void check(const string& name, T got, T want) {
    if (got != want) { cerr << "FAIL " << name << ": got " << got << ", want " << want << "\n"; exit(1); }
    cout << "ok   " << name << "\n";
}

void tableTests() {
    struct Case { string name; vector<int> a; int x, want; };
    vector<Case> cases = {
        {"empty", {}, 5, 0}, {"one, equal", {5}, 5, 0}, {"one, x smaller", {5}, 1, 0},
        {"one, x bigger", {5}, 9, 1}, {"all duplicates", {2, 2, 2, 2}, 2, 0},
        {"absent, middle", {1, 3, 5, 7}, 4, 2}, {"negatives", {-9, -4, -4, 0}, -4, 1},
        {"past the end", {1, 3}, 99, 2},
    };
    for (auto& c : cases) check("lower_bound " + c.name, lowerBound(c.a, c.x), c.want);
    check("kadane all negative", maxSubarray({-3, -1, -2}), -1LL);
    check("kadane one element", maxSubarray({7}), 7LL);
    check("kadane mixed", maxSubarray({2, -5, 3, -1, 4}), 6LL);
}

bool stress(int rounds = 2000, unsigned seed = 42) {
    mt19937 rng(seed);                                // fixed seed: replayable
    auto rnd = [&](int lo, int hi) { return (int)(rng() % (unsigned)(hi - lo + 1)) + lo; };
    for (int r = 0; r < rounds; r++) {
        vector<int> a(rnd(1, 8));
        for (int& v : a) v = rnd(-5, 5);
        if (maxSubarray(a) != bruteMaxSubarray(a)) { cout << "FAIL kadane\n"; return false; }
        sort(a.begin(), a.end());
        int x = rnd(-6, 6), want = (int)a.size();
        for (int i = 0; i < (int)a.size(); i++) if (a[i] >= x) { want = i; break; }
        if (lowerBound(a, x) != want) { cout << "FAIL lower_bound\n"; return false; }
    }
    cout << "stress ok: " << rounds << " random cases\n";
    return true;
}

int main() { tableTests(); return stress() ? 0 : 1; }

import java.util.*;

public class Templates {
    // First index i with a[i] >= x; a.length if every value is < x.
    static int lowerBound(int[] a, int x) {
        int lo = 0, hi = a.length;               // [lo, hi): hi is one past the end
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;        // no (lo + hi) overflow
            if (a[mid] < x) lo = mid + 1;        // mid too small: answer is right of it
            else hi = mid;                       // mid may be the answer: keep it
        }
        return lo;
    }

    // Largest sum of a non-empty contiguous run (Kadane). long sums.
    static long maxSubarray(int[] a) {
        long best = a[0], cur = a[0];            // NOT 0: all-negative input
        for (int i = 1; i < a.length; i++) {
            cur = Math.max(a[i], cur + a[i]);
            best = Math.max(best, cur);
        }
        return best;
    }

    static long bruteMaxSubarray(int[] a) {      // O(n^2) oracle
        long best = Long.MIN_VALUE;
        for (int i = 0; i < a.length; i++) {
            long s = 0;
            for (int j = i; j < a.length; j++) { s += a[j]; best = Math.max(best, s); }
        }
        return best;
    }

    static void check(String name, long got, long want) {
        if (got != want) throw new AssertionError(name + ": got " + got + ", want " + want);
        System.out.println("ok   " + name);
    }

    static void tableTests() {
        Object[][] cases = {   // name, sorted input, x, expected index
            {"empty", new int[]{}, 5, 0}, {"one, equal", new int[]{5}, 5, 0},
            {"one, x smaller", new int[]{5}, 1, 0}, {"one, x bigger", new int[]{5}, 9, 1},
            {"all duplicates", new int[]{2, 2, 2, 2}, 2, 0}, {"absent, middle", new int[]{1, 3, 5, 7}, 4, 2},
            {"negatives", new int[]{-9, -4, -4, 0}, -4, 1}, {"past the end", new int[]{1, 3}, 99, 2},
        };
        for (Object[] c : cases)
            check("lower_bound " + c[0], lowerBound((int[]) c[1], (int) c[2]), (int) c[3]);
        check("kadane all negative", maxSubarray(new int[]{-3, -1, -2}), -1);
        check("kadane one element", maxSubarray(new int[]{7}), 7);
        check("kadane mixed", maxSubarray(new int[]{2, -5, 3, -1, 4}), 6);
    }

    static boolean stress(int rounds, long seed) {
        Random rng = new Random(seed);           // fixed seed: replayable
        for (int r = 0; r < rounds; r++) {
            int[] a = new int[1 + rng.nextInt(8)];
            for (int i = 0; i < a.length; i++) a[i] = rng.nextInt(11) - 5;

            int x = rng.nextInt(13) - 6, want = a.length;

            if (lowerBound(a, x) != want) { System.out.println("FAIL lower_bound"); return false; }
        }
        System.out.println("stress ok: " + rounds + " random cases");
        return true;
    }

    public static void main(String[] args) {
        tableTests();

    }

import random


def lower_bound(a, x):
    """First index i with a[i] >= x; len(a) if every value is < x."""
    lo, hi = 0, len(a)              # search space is [lo, hi): hi is one past the end
    while lo < hi:                  # empty range when lo == hi
        mid = lo + (hi - lo) // 2
        if a[mid] < x:
            lo = mid + 1            # mid is too small, so the answer is to its right
        else:
            hi = mid                # mid may be the answer: keep it inside [lo, hi)
    return lo


def max_subarray(a):
    """Largest sum of a non-empty contiguous run (Kadane)."""
    best = cur = a[0]               # NOT 0: [-3, -1, -2] must return -1
    for v in a[1:]:
        cur = max(v, cur + v)       # extend the run or start again at v
        best = max(best, cur)
    return best


def brute_max_subarray(a):          # O(n^2), obviously correct, used as the oracle
    return max(sum(a[i:j + 1]) for i in range(len(a)) for j in range(i, len(a)))


def check(name, got, want):
    if got != want:
        raise AssertionError(f"{name}: got {got}, want {want}")
    print("ok  ", name)


def table_tests():
    cases = [  # name, sorted input, x, expected index
        ("empty", [], 5, 0),
        ("one, equal", [5], 5, 0),
        ("one, x smaller", [5], 1, 0),
        ("one, x bigger", [5], 9, 1),
        ("all duplicates", [2, 2, 2, 2], 2, 0),
        ("absent, middle", [1, 3, 5, 7], 4, 2),
        ("negatives", [-9, -4, -4, 0], -4, 1),
        ("past the end", [1, 3], 99, 2),
    ]
    for name, a, x, want in cases:
        check("lower_bound " + name, lower_bound(a, x), want)
    check("kadane all negative", max_subarray([-3, -1, -2]), -1)
    check("kadane one element", max_subarray([7]), 7)
    check("kadane mixed", max_subarray([2, -5, 3, -1, 4]), 6)


def stress(rounds=2000, seed=42):
    rng = random.Random(seed)       # fixed seed: a failure can be replayed
    for _ in range(rounds):
        a = [rng.randint(-5, 5) for _ in range(rng.randint(1, 8))]   # tiny inputs
        if max_subarray(a) != brute_max_subarray(a):
            print("FAIL kadane on", a)
            return False
        a.sort()
        x = rng.randint(-6, 6)
        want = next((i for i, v in enumerate(a) if v >= x), len(a))
        if lower_bound(a, x) != want:
            print("FAIL lower_bound on", a, x)
            return False
    print("stress ok:", rounds, "random cases")
    return True


if __name__ == "__main__":
    table_tests()
    stress()
