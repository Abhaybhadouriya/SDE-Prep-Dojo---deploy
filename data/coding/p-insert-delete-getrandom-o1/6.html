type RandomizedSet struct {
	vals []int       // every stored value, packed with no holes
	pos  map[int]int // value -> its index in vals
}

func Constructor() RandomizedSet {
	return RandomizedSet{pos: map[int]int{}}
}

func (s *RandomizedSet) Insert(val int) bool {
	if _, ok := s.pos[val]; ok {
		return false
	}
	s.pos[val] = len(s.vals)
	s.vals = append(s.vals, val)
	return true
}

func (s *RandomizedSet) Remove(val int) bool {
	i, ok := s.pos[val]
	if !ok {
		return false
	}
	last := s.vals[len(s.vals)-1]
	s.vals[i] = last                // fill the hole with the last value (no-op if val is the last)
	s.pos[last] = i                 // ... and record where it moved
	s.vals = s.vals[:len(s.vals)-1] // shrink from the end: O(1)
	delete(s.pos, val)              // AFTER the update, in case last == val
	return true
}

func (s *RandomizedSet) GetRandom() int {
	return s.vals[rand.Intn(len(s.vals))] // uniform index into a hole-free slice
}
class RandomizedSet {
    vector<int> vals;                 // every stored value, packed with no holes
    unordered_map<int, int> pos;      // value -> its index in vals
    mt19937 rng{380};                 // seeded so the tests are repeatable

public:
    bool insert(int val) {
        if (pos.count(val)) return false;
        pos[val] = (int)vals.size();
        vals.push_back(val);
        return true;
    }

    bool remove(int val) {
        auto it = pos.find(val);
        if (it == pos.end()) return false;
        int i = it->second, last = vals.back();
        vals[i] = last;               // fill the hole with the last value (no-op if val is the last)
        pos[last] = i;                // ... and record where it moved
        vals.pop_back();              // shrink from the end: O(1)
        pos.erase(val);               // erase AFTER the update, in case last == val
        return true;
    }

    int getRandom() {
        uniform_int_distribution<int> pick(0, (int)vals.size() - 1);
        return vals[pick(rng)];       // uniform index into a hole-free array
    }
};

static class RandomizedSet {
    private final List<Integer> vals = new ArrayList<>();     // packed, no holes
    private final Map<Integer, Integer> pos = new HashMap<>(); // value -> index in vals
    private final Random rnd = new Random(380);                // seeded for repeatable tests

    public boolean insert(int val) {
        if (pos.containsKey(val)) return false;
        pos.put(val, vals.size());
        vals.add(val);
        return true;
    }

    public boolean remove(int val) {
        Integer i = pos.get(val);
        if (i == null) return false;
        int last = vals.get(vals.size() - 1);
        vals.set(i, last);              // fill the hole with the last value (no-op if val is last)
        pos.put(last, i);               // ... and record where it moved
        vals.remove(vals.size() - 1);   // remove(int index) from the END: O(1), no shifting
        pos.remove(val);                // AFTER the update, in case last == val
        return true;
    }

    public int getRandom() {
        return vals.get(rnd.nextInt(vals.size())); // uniform index into a hole-free list
    }
}
class RandomizedSet:
    def __init__(self):
        self.vals = []   # every stored value, packed with no holes
        self.pos = {}    # value -> its index in vals

    def insert(self, val: int) -> bool:
        if val in self.pos:
            return False
        self.pos[val] = len(self.vals)
        self.vals.append(val)
        return True

    def remove(self, val: int) -> bool:
        i = self.pos.get(val)
        if i is None:
            return False
        last = self.vals[-1]
        self.vals[i] = last        # fill the hole with the last value (a no-op if val IS the last)
        self.pos[last] = i         # ... and tell the map where it moved
        self.vals.pop()            # the array shrinks from the end: O(1)
        del self.pos[val]          # delete AFTER the update, in case last == val
        return True

    def getRandom(self) -> int:
        return self.vals[random.randrange(len(self.vals))]  # uniform index into a hole-free array
