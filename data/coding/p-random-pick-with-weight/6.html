Each file in coding-code/random-pick-with-weight/ checks the ticket-to-index mapping exhaustively against the expanded-ticket brute force, then draws 200,000 picks and asserts every index's frequency is within one percentage point of w[i] / total.

class Solution {
    vector<int> prefix;                  // prefix[i] = w[0] + ... + w[i]; total <= 1e4 * 1e5 fits in int
    mt19937 rng{random_device{}()};

public:
    explicit Solution(vector<int>& w) {
        int run = 0;
        for (int x : w) {
            run += x;
            prefix.push_back(run);       // index i owns the tickets (prefix[i-1], prefix[i]]
        }
    }

    int pickIndex() {
        uniform_int_distribution<int> ticket(1, prefix.back());
        return indexFor(ticket(rng));
    }

    // owner of ticket t (1 <= t <= total): the first i with prefix[i] >= t
    int indexFor(int t) const {
        return int(lower_bound(prefix.begin(), prefix.end(), t) - prefix.begin());
    }
};

static class Solution {
    private final int[] prefix;           // prefix[i] = w[0] + ... + w[i]; total <= 1e9 fits in int
    private final Random rng = new Random();

    public Solution(int[] w) {
        prefix = new int[w.length];
        int run = 0;
        for (int i = 0; i < w.length; i++) {
            run += w[i];
            prefix[i] = run;              // index i owns the tickets (prefix[i-1], prefix[i]]
        }
    }

    public int pickIndex() {
        int ticket = rng.nextInt(prefix[prefix.length - 1]) + 1;   // uniform in [1, total]
        return indexFor(ticket);
    }

    // owner of ticket t: the first i with prefix[i] >= t
    int indexFor(int t) {
        int lo = 0, hi = prefix.length - 1;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (prefix[mid] >= t) hi = mid;   // mid may be the owner: keep it
            else lo = mid + 1;                // mid's range ends before t
        }
        return lo;
    }
}
