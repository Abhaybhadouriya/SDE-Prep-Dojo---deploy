Counting sort and bucket sort: O(n + k)

Comparison sorts cannot beat O(n log n) in the worst case. Counting sort does not compare: it uses each value as an array index. When the values fit in a small range 0..k, it sorts in O(n + k). Three passes: count each value, turn counts into "first free slot" with a prefix sum, then place items left to right.

  1. Input [3, 1, 2, 1, 0]. Every value is in 0..3, so we can use the values as array indexes. No comparisons at all.
  2. Pass 1, count: one 0, two 1s, one 2, one 3. That is cnt = [1, 2, 1, 1].
  3. Pass 2, prefix sums turn counts into the first free slot of each value: 0s start at 0, 1s at 1, 2s at 3, 3s at 4.
  4. Pass 3 walks the input left to right. 3 goes to slot 4, and the next free slot for 3 becomes 5.
  5. 1 (the first one) goes to slot 1. Next free slot for 1 is now 2.
  6. 2 goes to slot 3.
  7. The second 1 goes to slot 2, after the first 1. Walking left to right is what makes counting sort stable.
  8. 0 goes to slot 0. Output [0, 1, 1, 2, 3]: O(n + k) time, O(n + k) extra space.

Bucket sort is the same idea with a coarser index. The most common interview form is bucket by frequency: in Top K Frequent Elements and Sort Characters by Frequency, a frequency can only be 1..n, so put value v into bucket number freq(v) and read buckets from the high end. That is O(n) with no heap.

Other counting shapes you will meet: H-Index (counts capped at n, then a suffix sum finds the largest h), Height Checker and How Many Numbers Are Smaller Than the Current Number (values 0..100, so a prefix count answers every query), Valid Anagram (26 counters). Maximum Gap uses buckets of width (max - min) / (n - 1): the biggest gap can never be inside one bucket, so only bucket edges matter (pigeonhole).

// 3a. Stable counting sort for keys in [0, k].
func countingSort(a []int, k int) []int {
	cnt := make([]int, k+2)
	for _, x := range a {
		cnt[x+1]++
	}
	for v := 1; v <= k+1; v++ {
		cnt[v] += cnt[v-1]
	}
	out := make([]int, len(a))
	for _, x := range a {
		out[cnt[x]] = x
		cnt[x]++
	}
	return out
}

// 3b. Bucket by frequency (LC 347).
func topKFrequent(nums []int, k int) []int {
	freq := map[int]int{}
	for _, x := range nums {
		freq[x]++
	}
	bucket := make([][]int, len(nums)+1)
	for v, f := range freq {
		bucket[f] = append(bucket[f], v)
	}
	out := []int{}
	for f := len(nums); f >= 1 && len(out) < k; f-- {
		for _, v := range bucket[f] {
			if len(out) < k {
				out = append(out, v)
			}
		}
	}
	return out
}
// 3a. Stable counting sort for keys in [0, k].
vector<int> countingSort(const vector<int>& a, int k) {
    vector<int> cnt(k + 2, 0), out(a.size());
    for (int x : a) cnt[x + 1]++;
    for (int v = 1; v <= k + 1; v++) cnt[v] += cnt[v - 1];  // cnt[v] = first slot of value v
    for (int x : a) out[cnt[x]++] = x;                      // left to right keeps it stable
    return out;
}

// 3b. Bucket by frequency (LC 347): bucket index = how often a value occurs.
vector<int> topKFrequent(vector<int>& nums, int k) {
    unordered_map<int, int> freq;
    for (int x : nums) freq[x]++;
    vector<vector<int>> bucket(nums.size() + 1);
    for (auto& p : freq) bucket[p.second].push_back(p.first);
    vector<int> out;
    for (int f = (int)nums.size(); f >= 1 && (int)out.size() < k; f--)
        for (int v : bucket[f]) if ((int)out.size() < k) out.push_back(v);
    return out;
}

// 3b. Bucket by frequency (LC 347).
static int[] topKFrequent(int[] nums, int k) {
    Map<Integer, Integer> freq = new HashMap<>();
    for (int x : nums) freq.merge(x, 1, Integer::sum);
    List<List<Integer>> bucket = new ArrayList<>();
    for (int i = 0; i <= nums.length; i++) bucket.add(new ArrayList<>());
    for (Map.Entry<Integer, Integer> e : freq.entrySet()) bucket.get(e.getValue()).add(e.getKey());
    int[] out = new int[k];
    int n = 0;
    for (int f = nums.length; f >= 1 && n < k; f--)
        for (int v : bucket.get(f)) if (n < k) out[n++] = v;
    return out;
}
# 3a. Stable counting sort for keys in [0, k].
def counting_sort(a, k):
    cnt = [0] * (k + 2)
    for x in a:
        cnt[x + 1] += 1
    for v in range(1, k + 2):
        cnt[v] += cnt[v - 1]          # cnt[v] = first slot of value v
    out = [0] * len(a)
    for x in a:                       # left to right keeps it stable
        out[cnt[x]] = x
        cnt[x] += 1
    return out

def top_k_frequent(nums, k):
    bucket = [[] for _ in range(len(nums) + 1)]
    for v, f in Counter(nums).items():
        bucket[f].append(v)
    out = []

This is a histogram. You have written it many times as a latency histogram or as per-queue packet counters: an array indexed by the value, incremented in the hot path. Counting sort is that histogram plus one prefix-sum pass. The cost is memory proportional to the range k, which is why it fits "lowercase letters" or "ages 0..150" and not "any 32-bit integer" (radix sort does 32-bit values in 4 passes of 8 bits).
