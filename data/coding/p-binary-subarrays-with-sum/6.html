class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
        return atMost(nums, goal) - atMost(nums, goal - 1);
    }

private:
    // number of subarrays whose sum is <= limit
    int atMost(const vector<int>& nums, int limit) {
        if (limit < 0) return 0;                // no subarray of 0/1 values has a negative sum
        int left = 0, window = 0, count = 0;    // window = sum of nums[left..right]
        for (int right = 0; right < (int)nums.size(); right++) {
            window += nums[right];
            while (window > limit)              // too many ones: shrink from the left
                window -= nums[left++];
            count += right - left + 1;          // every start in [left..right] works for this end
        }
        return count;
    }
};

// atMost counts the subarrays whose sum is <= limit.
func atMost(nums []int, limit int) int {
	if limit < 0 { // no subarray of 0/1 values has a negative sum
		return 0
	}
	left, window, count := 0, 0, 0 // window = sum of nums[left..right]
	for right, x := range nums {
		window += x
		for window > limit { // too many ones: shrink from the left
			window -= nums[left]
			left++
		}
		count += right - left + 1 // every start in [left..right] works for this end
	}
	return count
}

class Solution:
    def numSubarraysWithSum(self, nums: List[int], goal: int) -> int:
        return self.at_most(nums, goal) - self.at_most(nums, goal - 1)

    def at_most(self, nums: List[int], limit: int) -> int:
        """Number of subarrays whose sum is <= limit."""
        if limit < 0:                   # no subarray of 0/1 values has a negative sum
            return 0
        left = 0
        window = 0                      # sum of nums[left..right]
        count = 0
        for right, x in enumerate(nums):
            window += x
            while window > limit:       # too many ones: shrink from the left
                window -= nums[left]
                left += 1
            count += right - left + 1   # every start in [left..right] works for this end
        return count
