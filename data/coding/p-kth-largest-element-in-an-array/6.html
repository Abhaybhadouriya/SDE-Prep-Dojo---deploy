class Solution {
public:
    int findKthLargest(const vector<int>& nums, int k) {
        priority_queue<int, vector<int>, greater<int>> keep;   // min-heap of the k largest so far
        for (int x : nums) {
            keep.push(x);
            if ((int)keep.size() > k) keep.pop();              // smallest can't be in the top k
        }
        return keep.top();
    }

    int findKthLargestQuickselect(vector<int> a, int k) {                     // by value: caller's data untouched
        static mt19937 rng(20261125);
        int target = (int)a.size() - k;                        // index n-k in ascending order
        int lo = 0, hi = (int)a.size() - 1;
        while (true) {
            int pivot = a[uniform_int_distribution<int>(lo, hi)(rng)];
            int lt = lo, i = lo, gt = hi;     // [lo,lt) < pivot, [lt,i) == pivot, (gt,hi] > pivot
            while (i <= gt) {
                if (a[i] < pivot) swap(a[lt++], a[i++]);
                else if (a[i] > pivot) swap(a[i], a[gt--]);    // i stays: new a[i] unchecked
                else i++;
            }
            if (target < lt) hi = lt - 1;
            else if (target > gt) lo = gt + 1;
            else return pivot;                                 // inside the block of pivot copies
        }
    }
};
type minHeap []int

func (h minHeap) Len() int            { return len(h) }
func (h minHeap) Less(i, j int) bool  { return h[i] < h[j] }
func (h minHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x interface{}) { *h = append(*h, x.(int)) }
func (h *minHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

func findKthLargest(nums []int, k int) int {
	keep := &minHeap{} // min-heap of the k largest so far
	for _, x := range nums {
		heap.Push(keep, x)
		if keep.Len() > k {
			heap.Pop(keep) // smallest can't be in the top k
		}
	}
	return (*keep)[0]
}

	lo, hi := 0, len(a)-1

static class Solution {
    public int findKthLargest(int[] nums, int k) {
        PriorityQueue<Integer> keep = new PriorityQueue<>();   // min-heap of the k largest so far
        for (int x : nums) {
            keep.offer(x);
            if (keep.size() > k) keep.poll();                  // smallest can't be in the top k
        }
        return keep.peek();
    }

    public int findKthLargestQuickselect(int[] nums, int k) {
        int[] a = nums.clone();                                // caller's array untouched
        int target = a.length - k;                             // index n-k in ascending order
        int lo = 0, hi = a.length - 1;
        while (true) {
            int pivot = a[ThreadLocalRandom.current().nextInt(lo, hi + 1)];
            int lt = lo, i = lo, gt = hi;   // [lo,lt) < pivot, [lt,i) == pivot, (gt,hi] > pivot
            while (i <= gt) {
                if (a[i] < pivot) swap(a, lt++, i++);
                else if (a[i] > pivot) swap(a, i, gt--);       // i stays: new a[i] unchecked
                else i++;
            }
            if (target < lt) hi = lt - 1;
            else if (target > gt) lo = gt + 1;
            else return pivot;                                 // inside the pivot block
        }
    }

    private static void swap(int[] a, int i, int j) {
        int t = a[i]; a[i] = a[j]; a[j] = t;
    }
}
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        keep = []                          # min-heap holding the k largest values seen so far
        for x in nums:
            heapq.heappush(keep, x)
            if len(keep) > k:
                heapq.heappop(keep)        # drop the smallest: it can no longer be in the top k
        return keep[0]                     # smallest of the top k = k-th largest

    def findKthLargestQuickselect(self, nums: List[int], k: int) -> int:
        a = list(nums)                     # work on a copy; the caller's list stays intact
        target = len(a) - k                # k-th largest = index n-k in ascending order
        lo, hi = 0, len(a) - 1
        while True:
            pivot = a[random.randint(lo, hi)]   # random pivot: no fixed bad input
            lt, i, gt = lo, lo, hi              # a[lo:lt] < pivot, a[lt:i] == pivot, a[gt+1:hi+1] > pivot
            while i <= gt:
                if a[i] < pivot:
                    a[lt], a[i] = a[i], a[lt]
                    lt += 1
                    i += 1
                elif a[i] > pivot:
                    a[gt], a[i] = a[i], a[gt]
                    gt -= 1                     # do not advance i: the swapped-in value is unchecked
                else:
                    i += 1
            if target < lt:
                hi = lt - 1                     # answer is among the smaller values
            elif target > gt:
                lo = gt + 1                     # answer is among the larger values
            else:
                return pivot                    # target falls inside the block of pivot copies
