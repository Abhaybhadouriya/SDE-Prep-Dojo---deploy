class Solution {
public:
    // Sort, fix nums[i], move lo/hi by the sign of (sum - target); keep the closest sum seen.
    int threeSumClosest(vector<int>& nums, int target) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        int best = nums[0] + nums[1] + nums[2];      // a real sum, not INT_MAX (no overflow in abs)
        for (int i = 0; i < n - 2; i++) {
            if (i > 0 && nums[i] == nums[i - 1]) continue;   // same fixed value: same pairs again
            int lo = i + 1, hi = n - 1;
            while (lo < hi) {
                int sum = nums[i] + nums[lo] + nums[hi];
                if (abs(sum - target) < abs(best - target)) best = sum;
                if (sum < target) lo++;              // need a bigger sum: only lo can raise it
                else if (sum > target) hi--;         // need a smaller sum: only hi can lower it
                else return sum;                     // distance 0: nothing can beat it
            }
        }
        return best;
    }

    // Brute force: every triplet of indices, O(n^3). Used as the test oracle.
    int threeSumClosestBrute(const vector<int>& nums, int target) {
        int n = nums.size(), best = nums[0] + nums[1] + nums[2];
        for (int a = 0; a < n; a++)
            for (int b = a + 1; b < n; b++)
                for (int c = b + 1; c < n; c++) {
                    int sum = nums[a] + nums[b] + nums[c];
                    if (abs(sum - target) < abs(best - target)) best = sum;
                }
        return best;
    }
};

	sort.Ints(nums)
	n := len(nums)
	best := nums[0] + nums[1] + nums[2] // a real sum, not math.MaxInt
	for i := 0; i < n-2; i++ {
		if i > 0 && nums[i] == nums[i-1] {
			continue // same fixed value: same pairs again
		}
		lo, hi := i+1, n-1
		for lo < hi {
			sum := nums[i] + nums[lo] + nums[hi]
			if abs(sum-target) < abs(best-target) {
				best = sum
			}
			switch {
			case sum < target:
				lo++ // need a bigger sum: only lo can raise it
			case sum > target:
				hi-- // need a smaller sum: only hi can lower it
			default:
				return sum // distance 0: nothing can beat it
			}
		}
	}
	return best
}

class Solution:
    def threeSumClosest(self, nums: List[int], target: int) -> int:
        """Sort, fix nums[i], move lo/hi by the sign of (sum - target); keep the closest sum seen."""
        nums.sort()
        n = len(nums)
        best = nums[0] + nums[1] + nums[2]           # a real sum, not infinity
        for i in range(n - 2):
            if i > 0 and nums[i] == nums[i - 1]:
                continue                             # same fixed value: same pairs again
            lo, hi = i + 1, n - 1
            while lo < hi:
                total = nums[i] + nums[lo] + nums[hi]
                if abs(total - target) < abs(best - target):
                    best = total
                if total < target:
                    lo += 1                          # need a bigger sum: only lo can raise it
                elif total > target:
                    hi -= 1                          # need a smaller sum: only hi can lower it
                else:
                    return total                     # distance 0: nothing can beat it
        return best

    def threeSumClosestBrute(self, nums: List[int], target: int) -> int:
        """Brute force: every triplet of indices, O(n^3). Used as the test oracle."""
        n = len(nums)
        best = nums[0] + nums[1] + nums[2]
        for a in range(n):
            for b in range(a + 1, n):
                for c in range(b + 1, n):
                    total = nums[a] + nums[b] + nums[c]
                    if abs(total - target) < abs(best - target):
                        best = total
        return best
