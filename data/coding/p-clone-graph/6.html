The test drivers in coding-code/clone-graph/ build graphs from adjacency lists (empty, single node, one edge, a 4-cycle, the complete graph K4, a path and a 100-node ring) and walk original and copy in lockstep, asserting equal values and neighbour order, exactly one copy per node, and no node shared with the original.

func cloneGraph(node *Node) *Node {
	if node == nil {
		return nil // empty graph
	}
	copyOf := map[*Node]*Node{} // original node -> its clone
	var dfs func(cur *Node) *Node
	dfs = func(cur *Node) *Node {
		if c, ok := copyOf[cur]; ok {
			return c // already cloned (maybe still in progress)
		}
		clone := &Node{Val: cur.Val}
		copyOf[cur] = clone // register BEFORE visiting neighbours: breaks cycles
		for _, nb := range cur.Neighbors {
			clone.Neighbors = append(clone.Neighbors, dfs(nb)) // same order as the original list
		}
		return clone
	}
	return dfs(node)
}
class Solution {
    unordered_map<Node*, Node*> copyOf;  // original node -> its clone

    Node* dfs(Node* node) {
        auto it = copyOf.find(node);
        if (it != copyOf.end()) return it->second;  // already cloned (maybe still in progress)
        Node* clone = new Node(node->val);
        copyOf[node] = clone;                       // register BEFORE visiting neighbours: breaks cycles
        for (Node* nb : node->neighbors)
            clone->neighbors.push_back(dfs(nb));    // same order as the original list
        return clone;
    }

public:
    Node* cloneGraph(Node* node) {
        if (node == nullptr) return nullptr;        // empty graph
        copyOf.clear();
        return dfs(node);
    }
};

static class Solution {
    public Node cloneGraph(Node node) {
        if (node == null) return null;                // empty graph
        return dfs(node, new HashMap<>());            // fresh map per call
    }

    private Node dfs(Node node, Map<Node, Node> copyOf) {  // copyOf: original -> clone
        Node done = copyOf.get(node);
        if (done != null) return done;                // already cloned (maybe still in progress)
        Node clone = new Node(node.val);
        copyOf.put(node, clone);                      // register BEFORE visiting neighbours: breaks cycles
        for (Node nb : node.neighbors)
            clone.neighbors.add(dfs(nb, copyOf));     // same order as the original list
        return clone;
    }
}
class Solution:
    def cloneGraph(self, node: Optional[Node]) -> Optional[Node]:
        copy_of = {}  # original node -> its clone

        def dfs(cur: Node) -> Node:
            if cur in copy_of:            # already cloned (maybe still in progress)
                return copy_of[cur]
            clone = Node(cur.val)
            copy_of[cur] = clone          # register BEFORE visiting neighbours: breaks cycles
            for nb in cur.neighbors:
                clone.neighbors.append(dfs(nb))  # same order as the original list
            return clone

        return dfs(node) if node else None  # empty graph -> None
