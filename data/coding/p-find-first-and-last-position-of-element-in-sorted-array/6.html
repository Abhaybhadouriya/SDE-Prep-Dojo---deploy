class Solution {
    // first index i with nums[i] >= target (strict = false) or nums[i] > target (strict = true);
    // returns nums.size() when no such index exists
    static int boundary(const vector<int>& nums, int target, bool strict) {
        int lo = 0, hi = (int)nums.size();           // answer lives in [lo, hi]
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            bool goLeft = strict ? nums[mid] > target : nums[mid] >= target;
            if (goLeft) hi = mid;                    // mid satisfies the test: answer is mid or left of it
            else lo = mid + 1;                       // mid fails: answer is right of mid
        }
        return lo;
    }

public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int first = boundary(nums, target, false);
        if (first == (int)nums.size() || nums[first] != target) return {-1, -1};
        int last = boundary(nums, target, true) - 1;  // one before the first bigger value
        return {first, last};
    }
};
func searchRange(nums []int, target int) []int {
	// first index i with nums[i] >= target (strict = false) or nums[i] > target (strict = true);
	// returns len(nums) when no such index exists
	boundary := func(strict bool) int {
		lo, hi := 0, len(nums) // answer lives in [lo, hi]
		for lo < hi {
			mid := lo + (hi-lo)/2
			goLeft := nums[mid] >= target
			if strict {
				goLeft = nums[mid] > target
			}
			if goLeft {
				hi = mid // mid satisfies the test: answer is mid or left of it
			} else {
				lo = mid + 1 // mid fails: answer is right of mid
			}
		}
		return lo
	}

	first := boundary(false)
	if first == len(nums) || nums[first] != target {
		return []int{-1, -1}
	}
	return []int{first, boundary(true) - 1} // one before the first bigger value
}
