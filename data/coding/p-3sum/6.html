Sort + two pointers in four languages. Each file in coding-code/3sum/ also cross-checks the result (as a set, and with no duplicates) against the O(n³) brute force on 400 random arrays.

class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        vector<vector<int>> out;
        for (int i = 0; i + 2 < n; i++) {
            if (nums[i] > 0) break;                          // all three would be positive
            if (i > 0 && nums[i] == nums[i - 1]) continue;   // same first value: skip
            int lo = i + 1, hi = n - 1;
            while (lo < hi) {
                int s = nums[i] + nums[lo] + nums[hi];
                if (s < 0) {
                    lo++;                                    // need a bigger sum
                } else if (s > 0) {
                    hi--;                                    // need a smaller sum
                } else {
                    out.push_back({nums[i], nums[lo], nums[hi]});
                    lo++;
                    hi--;
                    while (lo < hi && nums[lo] == nums[lo - 1]) lo++;   // skip middle copies
                }
            }
        }
        return out;
    }
};

	n := len(nums)
	out := [][]int{}
	for i := 0; i+2 < n; i++ {
		if nums[i] > 0 {
			break // all three would be positive
		}
		if i > 0 && nums[i] == nums[i-1] {
			continue // same first value: skip
		}
		lo, hi := i+1, n-1
		for lo < hi {
			s := nums[i] + nums[lo] + nums[hi]
			if s < 0 {
				lo++ // need a bigger sum
			} else if s > 0 {
				hi-- // need a smaller sum
			} else {
				out = append(out, []int{nums[i], nums[lo], nums[hi]})

				hi--

					lo++ // skip middle copies
				}
			}
		}
	}
	return out
}

class Solution:
    def threeSum(self, nums: List[int]) -> List[List[int]]:
        nums.sort()
        n, out = len(nums), []
        for i in range(n - 2):
            if nums[i] > 0:                      # smallest of the three is positive: no more zeros
                break
            if i > 0 and nums[i] == nums[i - 1]:
                continue                         # same first value -> same triplets again
            lo, hi = i + 1, n - 1
            while lo < hi:
                s = nums[i] + nums[lo] + nums[hi]
                if s < 0:
                    lo += 1                      # need a bigger sum: only lo can raise it
                elif s > 0:
                    hi -= 1                      # need a smaller sum: only hi can lower it
                else:
                    out.append([nums[i], nums[lo], nums[hi]])
                    lo += 1
                    hi -= 1
                    while lo < hi and nums[lo] == nums[lo - 1]:
                        lo += 1                  # skip copies of the middle value
        return out
