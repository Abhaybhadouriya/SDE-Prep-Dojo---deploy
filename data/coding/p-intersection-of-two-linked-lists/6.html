func getIntersectionNode(headA, headB *ListNode) *ListNode {
	p, q := headA, headB
	// After its own list, each pointer walks the other one. Both then cover
	// (own part) + (shared tail) + (other part) nodes, so they line up.
	for p != q {
		if p == nil {
			p = headB
		} else {
			p = p.Next
		}
		if q == nil {
			q = headA
		} else {
			q = q.Next
		}
	}
	return p // the first shared node, or nil if the lists never meet
}
class Solution {
public:
    ListNode* getIntersectionNode(ListNode* headA, ListNode* headB) {
        ListNode* p = headA;
        ListNode* q = headB;
        // After its own list, each pointer walks the other one. Both then cover
        // (own part) + (shared tail) + (other part) nodes, so they line up.
        while (p != q) {
            p = (p == nullptr) ? headB : p->next;
            q = (q == nullptr) ? headA : q->next;
        }
        return p;   // the first shared node, or nullptr if the lists never meet
    }
};

        p, q = headA, headB
        # After its own list, each pointer walks the other one. Both then cover

            p = headB if p is None else p.next

def getIntersectionNode(self, headA, headB):
    def length(h):
        n = 0
        while h:
            n, h = n + 1, h.next
        return n
    la, lb = length(headA), length(headB)
    p, q = headA, headB
    for _ in range(la - lb): p = p.next   # skip A's extra nodes
    for _ in range(lb - la): q = q.next   # or B's
    while p is not q:
        p, q = p.next, q.next
    return p
