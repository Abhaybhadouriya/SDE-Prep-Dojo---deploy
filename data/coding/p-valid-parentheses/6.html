All four push the expected closer, so a closer is checked with one comparison.

class Solution {
public:
    bool isValid(const string& s) {
        vector<char> stack;                 // closers we still owe, newest at back
        for (char c : s) {
            if (c == '(') stack.push_back(')');
            else if (c == '[') stack.push_back(']');
            else if (c == '{') stack.push_back('}');
            else {                          // c is a closer
                if (stack.empty() || stack.back() != c) return false;
                stack.pop_back();
            }
        }
        return stack.empty();               // leftovers = unclosed openers
    }
};
func isValid(s string) bool {
	closer := map[byte]byte{'(': ')', '[': ']', '{': '}'}
	stack := make([]byte, 0, len(s)) // closers we still owe, newest at the end
	for i := 0; i < len(s); i++ {
		c := s[i]
		if want, ok := closer[c]; ok {
			stack = append(stack, want)
			continue
		}
		if len(stack) == 0 || stack[len(stack)-1] != c {
			return false // nothing open, or the wrong type closes
		}
		stack = stack[:len(stack)-1]
	}
	return len(stack) == 0 // leftovers = unclosed openers
}
static class Solution {
    public boolean isValid(String s) {
        Deque<Character> stack = new ArrayDeque<>();   // closers we still owe
        for (char c : s.toCharArray()) {
            if (c == '(') stack.push(')');
            else if (c == '[') stack.push(']');
            else if (c == '{') stack.push('}');

        }
        return stack.isEmpty();                        // leftovers = unclosed openers
    }
}
class Solution:
    def isValid(self, s: str) -> bool:
        closer = {"(": ")", "[": "]", "{": "}"}
        stack = []                       # closers we still owe, newest on top
        for ch in s:
            if ch in closer:
                stack.append(closer[ch])  # remember which closer must come
            elif not stack or stack.pop() != ch:
                return False              # nothing open, or the wrong type closes
        return not stack                  # every opener must be closed
