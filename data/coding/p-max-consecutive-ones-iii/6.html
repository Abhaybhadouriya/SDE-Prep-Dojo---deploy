func longestOnes(nums []int, k int) int {
	left, zeros, best := 0, 0, 0 // zeros inside nums[left..right]
	for right, x := range nums {
		if x == 0 {
			zeros++
		}
		for zeros > k { // too many flips needed: shrink from the left
			if nums[left] == 0 {
				zeros--
			}
			left++
		}
		if right-left+1 > best { // window is valid again
			best = right - left + 1
		}
	}
	return best
}
class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        int left = 0, zeros = 0, best = 0;      // zeros inside nums[left..right]
        for (int right = 0; right < (int)nums.size(); right++) {
            if (nums[right] == 0) zeros++;
            while (zeros > k) {                 // too many flips needed: shrink from the left
                if (nums[left] == 0) zeros--;
                left++;
            }
            best = max(best, right - left + 1); // window is valid again
        }
        return best;
    }
};

class Solution:
    def longestOnes(self, nums: List[int], k: int) -> int:
        left = 0
        zeros = 0                       # zeros inside nums[left..right]
        best = 0
        for right, x in enumerate(nums):
            if x == 0:
                zeros += 1
            while zeros > k:            # too many flips needed: shrink from the left
                if nums[left] == 0:
                    zeros -= 1
                left += 1
            best = max(best, right - left + 1)   # window is valid again
        return best
