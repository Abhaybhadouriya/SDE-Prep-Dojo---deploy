// Level order; every missing child is written as "#", so the shape is fully recorded.
func (c *Codec) serialize(root *TreeNode) string {
	var out []string
	q := []*TreeNode{root}
	for head := 0; head < len(q); head++ {
		node := q[head]
		if node == nil {
			out = append(out, "#") // a hole: nothing below it
			continue
		}
		out = append(out, strconv.Itoa(node.Val))
		q = append(q, node.Left, node.Right) // both children, even if nil
	}
	return strings.Join(out, ",") // empty tree -> "#"
}

// Read tokens in the same order: each dequeued parent takes the next two tokens.
func (c *Codec) deserialize(data string) *TreeNode {
	tok := strings.Split(data, ",")
	if tok[0] == "#" {
		return nil
	}
	newNode := func(t string) *TreeNode {
		v, _ := strconv.Atoi(t)
		return &TreeNode{Val: v}
	}
	root := newNode(tok[0])
	q := []*TreeNode{root}
	i := 1
	for head := 0; head < len(q); head++ {
		parent := q[head]
		if tok[i] != "#" {
			parent.Left = newNode(tok[i])
			q = append(q, parent.Left)
		}
		if tok[i+1] != "#" {
			parent.Right = newNode(tok[i+1])
			q = append(q, parent.Right)
		}
		i += 2
	}
	return root
}
class Codec {
public:
    // Level order; every missing child is written as "#", so the shape is fully recorded.
    string serialize(TreeNode* root) {
        string out;
        queue<TreeNode*> q;
        q.push(root);
        while (!q.empty()) {
            TreeNode* node = q.front(); q.pop();
            if (!out.empty()) out += ',';
            if (!node) { out += '#'; continue; }   // a hole: nothing below it
            out += to_string(node->val);
            q.push(node->left);                    // both children, even if null
            q.push(node->right);
        }
        return out;                                // empty tree -> "#"
    }

    // Read tokens in the same order: each dequeued parent takes the next two tokens.
    TreeNode* deserialize(string data) {
        vector<string> tok;
        stringstream ss(data);
        for (string t; getline(ss, t, ',');) tok.push_back(t);
        if (tok.empty() || tok[0] == "#") return nullptr;
        TreeNode* root = new TreeNode(stoi(tok[0]));
        queue<TreeNode*> q;
        q.push(root);
        size_t i = 1;
        while (!q.empty()) {
            TreeNode* parent = q.front(); q.pop();
            if (tok[i] != "#") { parent->left = new TreeNode(stoi(tok[i])); q.push(parent->left); }
            i++;
            if (tok[i] != "#") { parent->right = new TreeNode(stoi(tok[i])); q.push(parent->right); }
            i++;
        }
        return root;
    }
};

static class Codec {
    // Level order; every missing child is written as "#", so the shape is fully recorded.
    public String serialize(TreeNode root) {
        StringBuilder out = new StringBuilder();
        List<TreeNode> order = new ArrayList<>();    // queue = list + head index (ArrayDeque refuses null)
        order.add(root);
        for (int head = 0; head < order.size(); head++) {
            TreeNode node = order.get(head);
            if (out.length() > 0) out.append(',');
            if (node == null) { out.append('#'); continue; }   // a hole: nothing below it
            out.append(node.val);
            order.add(node.left);                    // both children, even if null
            order.add(node.right);
        }
        return out.toString();                       // empty tree -> "#"
    }

    // Read tokens in the same order: each dequeued parent takes the next two tokens.
    public TreeNode deserialize(String data) {
        String[] tok = data.split(",");
        if (tok[0].equals("#")) return null;
        TreeNode root = new TreeNode(Integer.parseInt(tok[0]));
        Deque<TreeNode> q = new ArrayDeque<>();
        q.add(root);
        int i = 1;
        while (!q.isEmpty()) {
            TreeNode parent = q.poll();
            if (!tok[i].equals("#")) { parent.left = new TreeNode(Integer.parseInt(tok[i])); q.add(parent.left); }
            i++;
            if (!tok[i].equals("#")) { parent.right = new TreeNode(Integer.parseInt(tok[i])); q.add(parent.right); }
            i++;
        }
        return root;
    }
}

    def serialize(self, root: Optional[TreeNode]) -> str:
        out = []
        q = deque([root])
        while q:
            node = q.popleft()
            if node is None:
                out.append("#")                  # a hole: nothing below it
                continue
            out.append(str(node.val))
            q.append(node.left)                  # both children, even if None
            q.append(node.right)
        return ",".join(out)                     # empty tree -> "#"
