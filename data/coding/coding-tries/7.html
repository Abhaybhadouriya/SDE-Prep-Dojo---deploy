Bit tries: maximum XOR of two numbers

Given an array of non-negative integers (each < 231), find the maximum a XOR b over all pairs. The brute force is O(n2). Build a trie over the bits of each number, from the most significant bit down. To maximise x XOR y for a fixed x, walk the trie and at every bit take the opposite bit if that child exists: a 1 in a higher bit is worth more than all the lower bits combined, so the greedy choice is always right.

Insert each number, then query it against everything inserted so far (inserting first is fine: x XOR x = 0 never beats the real answer). Each insert and query is 31 steps, so the whole thing is O(31·n) time and O(31·n) nodes.

// Bit trie over 31 bits (values in [0, 2^31)). Nodes live in a slice:
// index 0 = root, child 0 means "no child".
func findMaximumXOR(nums []int) int {
	const B = 30
	t := make([][2]int, 1, len(nums)*(B+1)+1)
	best := 0
	for _, x := range nums {
		cur := 0 // insert x, highest bit first
		for b := B; b >= 0; b-- {
			bit := (x >> b) & 1
			if t[cur][bit] == 0 {
				t = append(t, [2]int{})
				t[cur][bit] = len(t) - 1
			}
			cur = t[cur][bit]
		}
		cur, val := 0, 0 // query: take the opposite bit if it exists
		for b := B; b >= 0; b-- {
			bit := (x >> b) & 1
			if t[cur][bit^1] != 0 {
				val |= 1 << b
				cur = t[cur][bit^1]
			} else {
				cur = t[cur][bit]
			}
		}
		best = max(best, val)
	}
	return best
}
// Bit trie over 31 bits (values in [0, 2^31)). Nodes live in a vector: index 0 = root,
// child 0 means "no child" (the root is never anyone's child).
int findMaximumXOR(const vector<int>& nums) {
    const int B = 30;
    vector<array<int, 2>> t(1, array<int, 2>{{0, 0}});
    int best = 0;
    for (int x : nums) {
        int cur = 0;                                     // insert x, highest bit first
        for (int b = B; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (!t[cur][bit]) { t[cur][bit] = t.size(); t.push_back(array<int, 2>{{0, 0}}); }
            cur = t[cur][bit];
        }
        cur = 0;                                         // query: greedily take the opposite bit
        int val = 0;
        for (int b = B; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (t[cur][bit ^ 1]) { val |= 1 << b; cur = t[cur][bit ^ 1]; }
            else cur = t[cur][bit];
        }
        best = max(best, val);
    }
    return best;
}
// Bit trie over 31 bits (values in [0, 2^31)). Nodes live in two int arrays:
// index 0 = root, child 0 means "no child".
static int findMaximumXOR(int[] nums) {
    final int B = 30;
    int[][] t = new int[nums.length * (B + 1) + 1][2];
    int size = 1, best = 0;
    for (int x : nums) {
        int cur = 0;                                // insert x, highest bit first
        for (int b = B; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (t[cur][bit] == 0) t[cur][bit] = size++;
            cur = t[cur][bit];
        }
        cur = 0;                                    // query: take the opposite bit if it exists
        int val = 0;
        for (int b = B; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (t[cur][bit ^ 1] != 0) { val |= 1 << b; cur = t[cur][bit ^ 1]; }
            else cur = t[cur][bit];
        }
        best = Math.max(best, val);
    }
    return best;
}
def find_maximum_xor(nums: List[int]) -> int:
    B = 30                                      # values in [0, 2^31)
    root = [None, None]                         # node = [child0, child1]
    best = 0
    for x in nums:
        cur = root                              # insert x, highest bit first
        for b in range(B, -1, -1):
            bit = (x >> b) & 1
            if cur[bit] is None:
                cur[bit] = [None, None]
            cur = cur[bit]
        cur, val = root, 0                      # query: take the opposite bit if it exists
        for b in range(B, -1, -1):
            bit = (x >> b) & 1
            if cur[bit ^ 1] is not None:
                val |= 1 << b
                cur = cur[bit ^ 1]
            else:
                cur = cur[bit]
        best = max(best, val)
    return best

Maximum XOR With an Element From Array (LC 1707, Hard) adds a limit: for each query (x, m), use only array values <= m. Sort the array and sort the queries by m, then sweep: insert every value <= m before answering that query (an offline technique), or store the minimum value in each subtree and refuse to step into a child whose minimum exceeds m. Related ideas: "count pairs with XOR in a range" stores subtree counts at each node, and "XOR of a subarray" becomes a pair of prefix XORs, so the bit trie runs over prefix values.
