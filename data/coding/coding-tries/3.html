Insert, search, startsWith

All three are the same walk. Insert creates missing children as it goes and sets end on the last node. Search walks and then checks end. StartsWith walks and only checks that it did not fall off. Step through inserting card into a trie that holds cat and car:

  1. Start at the root with i = 0. The root has a child for 'c', so step into it. Nothing is created.
  2. 'a' exists under c: step down. The shared prefix "ca" is reused, not copied.
  3. 'r' exists too. It is already a word end (for "car"), but insert does not care: it keeps walking.
  4. 'd' is missing under r, so create a new node and step into it. Only one node is allocated for the whole word.
  5. The word is finished: set end = true on the last node. Now the trie holds cat, car and card.
  6. search("ca") walks c, a successfully, but a.end is false: no word is exactly "ca". Result false.
  7. startsWith("ca") makes the same walk and only needs the node to exist. Result true: cat, car and card are below it.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

type TrieNode struct {
	next [26]*TrieNode // array children: one slot per letter
	end  bool          // a whole word stops here
}

type Trie struct{ root TrieNode }

func (t *Trie) Insert(w string) {
	cur := &t.root
	for i := 0; i < len(w); i++ {
		c := w[i] - 'a'
		if cur.next[c] == nil {
			cur.next[c] = &TrieNode{}
		}
		cur = cur.next[c]
	}
	cur.end = true
}

func (t *Trie) walk(s string) *TrieNode { // follow s; nil if it falls off
	cur := &t.root
	for i := 0; i < len(s); i++ {
		cur = cur.next[s[i]-'a']
		if cur == nil {
			return nil
		}
	}
	return cur
}

func (t *Trie) Search(w string) bool     { n := t.walk(w); return n != nil && n.end }
func (t *Trie) StartsWith(p string) bool { return t.walk(p) != nil }

func (t *Trie) SearchWild(w string) bool { // '.' = any letter
	var wild func(n *TrieNode, i int) bool
	wild = func(n *TrieNode, i int) bool {
		if i == len(w) {
			return n.end
		}
		if w[i] != '.' { // normal letter: one path
			c := n.next[w[i]-'a']
			return c != nil && wild(c, i+1)
		}
		for _, c := range n.next { // '.': try every child
			if c != nil && wild(c, i+1) {
				return true
			}
		}
		return false
	}
	return wild(&t.root, 0)
}
struct TrieNode {
    TrieNode* next[26] = {};   // array children: one slot per letter, O(1) step
    bool end = false;          // a whole word stops here
};

class Trie {
    TrieNode root;
    TrieNode* walk(const string& s) {            // follow s; nullptr if it falls off
        TrieNode* cur = &root;
        for (char ch : s) {
            cur = cur->next[ch - 'a'];
            if (!cur) return nullptr;
        }
        return cur;
    }
    bool wild(TrieNode* n, const string& w, size_t i) {
        if (i == w.size()) return n->end;
        if (w[i] != '.') {                       // normal letter: one path
            TrieNode* c = n->next[w[i] - 'a'];
            return c && wild(c, w, i + 1);
        }
        for (TrieNode* c : n->next)              // '.': try every child
            if (c && wild(c, w, i + 1)) return true;
        return false;
    }
public:
    void insert(const string& w) {
        TrieNode* cur = &root;
        for (char ch : w) {
            int c = ch - 'a';
            if (!cur->next[c]) cur->next[c] = new TrieNode();
            cur = cur->next[c];
        }
        cur->end = true;
    }
    bool search(const string& w) { TrieNode* n = walk(w); return n && n->end; }
    bool startsWith(const string& p) { return walk(p) != nullptr; }
    bool searchWild(const string& w) { return wild(&root, w, 0); }   // '.' = any letter
};
static class TrieNode {
    TrieNode[] next = new TrieNode[26];   // array children: one slot per letter
    boolean end;                          // a whole word stops here
}

static class Trie {
    private final TrieNode root = new TrieNode();

    void insert(String w) {
        TrieNode cur = root;
        for (char ch : w.toCharArray()) {
            int c = ch - 'a';
            if (cur.next[c] == null) cur.next[c] = new TrieNode();
            cur = cur.next[c];
        }
        cur.end = true;
    }
    boolean search(String w) { TrieNode n = walk(w); return n != null && n.end; }
    boolean startsWith(String p) { return walk(p) != null; }
    boolean searchWild(String w) { return wild(root, w, 0); }   // '.' = any letter

    private TrieNode walk(String s) {     // follow s; null if it falls off
        TrieNode cur = root;
        for (char ch : s.toCharArray()) {
            cur = cur.next[ch - 'a'];
            if (cur == null) return null;
        }
        return cur;
    }
    private boolean wild(TrieNode n, String w, int i) {
        if (i == w.length()) return n.end;
        char ch = w.charAt(i);
        if (ch != '.') {                  // normal letter: one path
            TrieNode c = n.next[ch - 'a'];
            return c != null && wild(c, w, i + 1);
        }
        for (TrieNode c : n.next)         // '.': try every child
            if (c != null && wild(c, w, i + 1)) return true;
        return false;
    }
}
class TrieNode:
    __slots__ = ("next", "end")

    def __init__(self):
        self.next = {}          # map children: only the letters that exist
        self.end = False        # a whole word stops here


class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, w: str) -> None:
        cur = self.root
        for ch in w:
            cur = cur.next.setdefault(ch, TrieNode())
        cur.end = True

    def _walk(self, s: str):    # follow s; None if it falls off
        cur = self.root
        for ch in s:
            cur = cur.next.get(ch)
            if cur is None:
                return None
        return cur

    def search(self, w: str) -> bool:
        n = self._walk(w)
        return n is not None and n.end

    def starts_with(self, p: str) -> bool:
        return self._walk(p) is not None

    def search_wild(self, w: str) -> bool:      # '.' = any letter
        def wild(n, i):
            if i == len(w):
                return n.end
            if w[i] != ".":                     # normal letter: one path
                c = n.next.get(w[i])
                return c is not None and wild(c, i + 1)
            return any(wild(c, i + 1) for c in n.next.values())   # '.': every child
        return wild(self.root, 0)

The C++, Java and Go panels use the 26-slot array; the Python panel uses a dict, which is the natural Python choice. Insert, search and startsWith are all O(L) time; insert allocates at most L nodes. Total memory is O(total characters inserted) nodes, which for the array design means up to 26 pointers each. The C++ version leaks nodes on purpose to stay short; in a real codebase give the trie a destructor or a node pool.

Counting variants (Implement Trie II, a Premium problem): store two counters per node, pass (how many inserted words go through this node) and end (how many end here). countWordsStartingWith(p) is the pass of p's node, and erase walks the word decrementing both. The same counter answers "how many words share this prefix" in O(L).
