Word Search II: one DFS for all the words

You get an m x n grid of letters and a list of words; return every word that can be traced through horizontally or vertically adjacent cells without reusing a cell in the same word. Searching each word separately (Word Search I in a loop) repeats the same grid DFS for every word. The trie trick: put all words in one trie, start a DFS from every cell, and move a trie pointer one step down with every grid step. If the trie has no child for the next letter, no word continues that way, so the DFS stops at once.

  1. DFS from cell (0,0) 'c'. The trie root has a child 'c', so some word may start here: mark the cell as used and continue.
  2. Move right to 'a'. The trie pointer moves from c to a. The grid path and the trie path always spell the same prefix.
  3. Move right to 't'. The trie node has a word marker: report "cat" and clear the marker so the same word is not reported twice from another path.
  4. Back at 'a', try the cell below, 'o'. The trie node "ca" has no child 'o', so the DFS returns immediately. This pruning is the whole speed-up.
  5. A new start at 'r' follows r, o, d through the trie and finds "rod".
  6. A start at the centre 'o' goes down to 'a', then right to 'r': "oar". All three words were found in one pass over the grid.
  7. Starts at 'a', 'e', 't' or 'd' die at the root in O(1), because no word begins with those letters.
type WNode struct {
	next [26]*WNode
	word string // the word that ends here ("" if none)
}

func findWords(board [][]byte, words []string) []string {
	root := &WNode{}
	for _, w := range words { // 1) all words into one trie
		cur := root
		for i := 0; i < len(w); i++ {
			c := w[i] - 'a'
			if cur.next[c] == nil {
				cur.next[c] = &WNode{}
			}
			cur = cur.next[c]
		}
		cur.word = w
	}
	R, C := len(board), len(board[0])
	found := []string{}
	dirs := [4][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}}
	var dfs func(r, c int, parent *WNode)
	dfs = func(r, c int, parent *WNode) {
		ch := board[r][c]
		node := parent.next[ch-'a']
		if node == nil { // no word continues this way: prune
			return
		}
		if node.word != "" { // report once, then forget it
			found = append(found, node.word)
			node.word = ""
		}
		board[r][c] = '#' // mark the cell as on the path
		for _, d := range dirs {
			nr, nc := r+d[0], c+d[1]
			if nr >= 0 && nr < R && nc >= 0 && nc < C && board[nr][nc] != '#' {
				dfs(nr, nc, node)
			}
		}
		board[r][c] = ch // undo on the way back
	}
	for r := 0; r < R; r++ { // 2) one DFS per start cell
		for c := 0; c < C; c++ {
			dfs(r, c, root)
		}
	}
	return found
}
struct WNode {
    WNode* next[26] = {};
    int word = -1;                               // index of the word that ends here
};

vector<string> findWords(vector<vector<char>>& board, const vector<string>& words) {
    WNode* root = new WNode();
    for (int k = 0; k < (int)words.size(); k++) {          // 1) all words into one trie
        WNode* cur = root;
        for (char ch : words[k]) {
            int c = ch - 'a';
            if (!cur->next[c]) cur->next[c] = new WNode();
            cur = cur->next[c];
        }
        cur->word = k;
    }
    vector<string> found;
    int R = board.size(), C = board[0].size();
    int dr[] = {-1, 1, 0, 0}, dc[] = {0, 0, -1, 1};
    function<void(int, int, WNode*)> dfs = [&](int r, int c, WNode* parent) {
        char ch = board[r][c];
        WNode* node = parent->next[ch - 'a'];
        if (!node) return;                       // no word continues this way: prune
        if (node->word >= 0) {                   // report once, then forget it
            found.push_back(words[node->word]);
            node->word = -1;
        }
        board[r][c] = '#';                       // mark the cell as on the path
        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < R && nc >= 0 && nc < C && board[nr][nc] != '#')
                dfs(nr, nc, node);
        }
        board[r][c] = ch;                        // undo on the way back
    };
    for (int r = 0; r < R; r++)                  // 2) one DFS per start cell
        for (int c = 0; c < C; c++) dfs(r, c, root);
    return found;
}
static class WNode {
    WNode[] next = new WNode[26];
    String word;                          // the word that ends here (null if none)
}

static List<String> findWords(char[][] board, String[] words) {
    WNode root = new WNode();
    for (String w : words) {                        // 1) all words into one trie
        WNode cur = root;
        for (char ch : w.toCharArray()) {
            int c = ch - 'a';
            if (cur.next[c] == null) cur.next[c] = new WNode();
            cur = cur.next[c];
        }
        cur.word = w;
    }
    List<String> found = new ArrayList<>();
    for (int r = 0; r < board.length; r++)          // 2) one DFS per start cell
        for (int c = 0; c < board[0].length; c++) dfs(board, r, c, root, found);
    return found;
}

static final int[] DR = {-1, 1, 0, 0}, DC = {0, 0, -1, 1};

static void dfs(char[][] b, int r, int c, WNode parent, List<String> found) {
    char ch = b[r][c];
    WNode node = parent.next[ch - 'a'];
    if (node == null) return;                       // no word continues: prune
    if (node.word != null) { found.add(node.word); node.word = null; }  // report once
    b[r][c] = '#';                                  // mark the cell as on the path
    for (int d = 0; d < 4; d++) {
        int nr = r + DR[d], nc = c + DC[d];
        if (nr >= 0 && nr < b.length && nc >= 0 && nc < b[0].length && b[nr][nc] != '#')
            dfs(b, nr, nc, node, found);
    }
    b[r][c] = ch;                                   // undo on the way back
}
def find_words(board: List[List[str]], words: List[str]) -> List[str]:
    root = {}                                   # dict-of-dicts trie; key "$" holds the word
    for w in words:                             # 1) all words into one trie
        cur = root
        for ch in w:
            cur = cur.setdefault(ch, {})
        cur["$"] = w
    R, C = len(board), len(board[0])
    found = []

    def dfs(r, c, parent):
        ch = board[r][c]
        node = parent.get(ch)
        if node is None:                        # no word continues this way: prune
            return
        w = node.pop("$", None)                 # report once, then forget it
        if w:
            found.append(w)
        board[r][c] = "#"                       # mark the cell as on the path
        for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= nr < R and 0 <= nc < C and board[nr][nc] != "#":
                dfs(nr, nc, node)
        board[r][c] = ch                        # undo on the way back
        if not node:                            # optional: drop a dead branch
            parent.pop(ch)

    for r in range(R):                          # 2) one DFS per start cell
        for c in range(C):
            dfs(r, c, root)
    return found

Why it is fast, and the pruning details

  • Store the word at its end node (or its index) instead of rebuilding the string from the path. Clearing it after a hit gives de-duplication for free.
  • Mark cells in place ('#') and restore them on the way back, instead of a separate visited set. It is O(1) per step and the board is unchanged at the end.
  • Remove dead branches: after a child's subtree has no words left, delete it from its parent (the Python panel does this). Later DFS calls stop even earlier. On LeetCode's large tests this is what turns a time-limit-exceeded into an accept.
  • Complexity: building the trie is O(total letters). The search is bounded by O(m·n·4·3L-1) for maximum word length L (the first step has 4 directions, later ones 3 because you cannot go back), but the trie pruning makes the real cost far smaller.
