The template you bring to every practice session

In a real interview you rarely get a compiler. In practice you do, so use it to train the habit interviewers grade: edge cases first, then a check against a slow but obviously correct version. This is a stress test. Write it once and copy it for every problem: replace solve with your fast answer and brute with the simplest correct one. The example below uses maximum subarray sum (Kadane's algorithm vs. checking every subarray), LeetCode 53.

package main

import (
	"fmt"
	"math/rand"
)

// solve is the optimal answer you present: Kadane, O(n) time, O(1) space.
func solve(a []int) int {
	best, cur := a[0], a[0]
	for _, x := range a[1:] {
		cur = max(x, cur+x) // extend the run or start fresh
		best = max(best, cur)
	}
	return best
}

// brute is slow but obviously correct: every subarray, O(n^2).
func brute(a []int) int {
	best := a[0]
	for i := range a {
		s := 0
		for j := i; j < len(a); j++ {
			s += a[j]
			best = max(best, s)
		}
	}
	return best
}

func main() {
	fixed := []struct {
		a    []int
		want int
	}{{[]int{5}, 5}, {[]int{-3, -1, -2}, -1}, {[]int{2, -1, 2}, 3},
		{[]int{-2, 1, -3, 4, -1, 2, 1, -5, 4}, 6}}
	for _, t := range fixed {
		if got := solve(t.a); got != t.want {
			panic(fmt.Sprintf("%v: got %d want %d", t.a, got, t.want))
		}
	}
	rng := rand.New(rand.NewSource(7))
	for it := 0; it < 500; it++ {
		a := make([]int, 1+rng.Intn(8))
		for i := range a {
			a[i] = rng.Intn(21) - 10
		}
		if solve(a) != brute(a) {
			fmt.Println("MISMATCH", a)
			return
		}
	}
	fmt.Println("all tests passed")
}

// Optimal answer you present: Kadane, O(n) time, O(1) space.
long long solve(const vector<int>& a) {
    long long best = a[0], cur = a[0];
    for (size_t i = 1; i < a.size(); i++) {
        cur = max<long long>(a[i], cur + a[i]);  // extend the run or start fresh
        best = max(best, cur);
    }
    return best;
}

// Slow but obviously correct: every subarray, O(n^2).
long long brute(const vector<int>& a) {
    long long best = a[0];
    for (size_t i = 0; i < a.size(); i++) {
        long long s = 0;
        for (size_t j = i; j < a.size(); j++) { s += a[j]; best = max(best, s); }
    }
    return best;
}

int main() {
    // 1. hand-picked edge cases
    vector<pair<vector<int>, long long>> fixed = {
        {{5}, 5}, {{-3, -1, -2}, -1}, {{2, -1, 2}, 3},
        {{-2, 1, -3, 4, -1, 2, 1, -5, 4}, 6}};
    for (auto& t : fixed) assert(solve(t.first) == t.second);
    // 2. stress test against the brute force
    mt19937 rng(7);
    for (int it = 0; it < 500; it++) {
        int n = 1 + rng() % 8;
        vector<int> a(n);
        for (int& x : a) x = (int)(rng() % 21) - 10;
        if (solve(a) != brute(a)) {
            printf("MISMATCH n=%d\n", n);
            return 1;
        }
    }
    puts("all tests passed");
    return 0;
}

public class Templates {
    // Optimal answer you present: Kadane, O(n) time, O(1) space.
    static long solve(int[] a) {
        long best = a[0], cur = a[0];
        for (int i = 1; i < a.length; i++) {
            cur = Math.max(a[i], cur + a[i]); // extend the run or start fresh
            best = Math.max(best, cur);
        }
        return best;
    }

    // Slow but obviously correct: every subarray, O(n^2).
    static long brute(int[] a) {
        long best = a[0];
        for (int i = 0; i < a.length; i++) {
            long s = 0;
            for (int j = i; j < a.length; j++) { s += a[j]; best = Math.max(best, s); }
        }
        return best;
    }

    public static void main(String[] args) {
        int[][] in = {{5}, {-3, -1, -2}, {2, -1, 2}, {-2, 1, -3, 4, -1, 2, 1, -5, 4}};
        long[] want = {5, -1, 3, 6};
        for (int t = 0; t < in.length; t++)

import random


def solve(a):
    """Optimal answer you present: Kadane, O(n) time, O(1) space."""
    best = cur = a[0]
    for x in a[1:]:
        cur = max(x, cur + x)  # extend the run or start fresh at x
        best = max(best, cur)
    return best


def brute(a):
    """Slow but obviously correct: every subarray, O(n^2)."""
    best = a[0]
    for i in range(len(a)):
        s = 0
        for j in range(i, len(a)):
            s += a[j]
            best = max(best, s)
    return best


def main():
    # 1. hand-picked edge cases: one element, all negative, mixed, classic
    fixed = [([5], 5), ([-3, -1, -2], -1), ([2, -1, 2], 3),
             ([-2, 1, -3, 4, -1, 2, 1, -5, 4], 6)]
    for a, want in fixed:
        got = solve(a)
        assert got == want, f"{a}: got {got}, want {want}"
    # 2. stress test: small random inputs, optimal must equal brute force
    rng = random.Random(7)
    for _ in range(500):
        a = [rng.randint(-10, 10) for _ in range(rng.randint(1, 8))]
        if solve(a) != brute(a):
            print("MISMATCH", a, solve(a), brute(a))
            return
    print("all tests passed")


if __name__ == "__main__":
    main()
