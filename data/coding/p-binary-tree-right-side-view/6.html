class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        vector<int> view;
        if (!root) return view;
        queue<TreeNode*> q;
        q.push(root);
        while (!q.empty()) {
            int width = (int)q.size();          // nodes of exactly one level are in the queue
            for (int i = 0; i < width; i++) {
                TreeNode* node = q.front(); q.pop();
                if (i == width - 1) view.push_back(node->val);   // rightmost of this level
                if (node->left) q.push(node->left);
                if (node->right) q.push(node->right);
            }
        }
        return view;
    }
};
func rightSideView(root *TreeNode) []int {
	view := []int{}
	if root == nil {
		return view
	}
	q := []*TreeNode{root}
	for len(q) > 0 {
		width := len(q) // nodes of exactly one level are in the queue
		for i := 0; i < width; i++ {
			node := q[i]
			if i == width-1 {
				view = append(view, node.Val) // rightmost of this level
			}
			if node.Left != nil {
				q = append(q, node.Left)
			}
			if node.Right != nil {
				q = append(q, node.Right)
			}
		}
		q = q[width:] // drop the level we just finished
	}
	return view
}

        q.add(root);
        while (!q.isEmpty()) {
            int width = q.size();               // nodes of exactly one level are in the queue
            for (int i = 0; i < width; i++) {
                TreeNode node = q.poll();
                if (i == width - 1) view.add(node.val);   // rightmost of this level
                if (node.left != null) q.add(node.left);
                if (node.right != null) q.add(node.right);
            }
        }
        return view;
    }
}
