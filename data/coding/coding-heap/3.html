Pattern 1: Top-K with a size-k heap

To keep the k largest items, use a min-heap of size k. It sounds backwards, but the heap holds your current top k and its root is the weakest member of that club: exactly the one a new item must beat. Push the new item; if the size goes over k, pop the root.

  1. 5 arrives. The heap has room (size 1 < k = 3), so just push.
  2. 1 arrives. Push. Heap = {1, 5}. The root is the smallest of what we keep.
  3. 9 arrives. Push. Now the heap is full (3 items). Its root, 1, is the 3rd largest of everything seen so far.
  4. 3 arrives. Push makes 4 items, so pop the root (1): 1 can never be in the top 3 again. Root is now 3.
  5. 7 arrives. Push, size 4, pop 3. The heap keeps {5, 7, 9}.
  6. 8 arrives. Push, pop 5. Final heap {7, 8, 9}: those are the top 3, and the root 7 is the answer to "k-th largest".

At the end, the heap holds the top k and the root is the k-th largest. Each element costs one push and at most one pop on a heap of size k + 1: O(n log k) time, O(k) space. That is why this works on a stream too (Kth Largest in a Stream): you never need the old items again.

Template

import "container/heap"         // you write Len/Less/Swap/Push/Pop; heap.Push/heap.Pop do the sifting

// IntMinHeap is a min-heap of ints for container/heap.
type IntMinHeap []int
func (h IntMinHeap) Len() int            { return len(h) }
func (h IntMinHeap) Less(i, j int) bool  { return h[i] < h[j] }
func (h IntMinHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *IntMinHeap) Push(x interface{}) { *h = append(*h, x.(int)) }
func (h *IntMinHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

// Top-K: size-k MIN-heap, root = k-th largest. O(n log k).
func kthLargest(nums []int, k int) int {
	h := &IntMinHeap{}
	for _, x := range nums {
		heap.Push(h, x)
		if h.Len() > k {
			heap.Pop(h) // drop the smallest of k+1
		}
	}
	return (*h)[0]
}

// Item is a generic (key, a, b) triple ordered by key, then a.
type Item struct{ key, a, b int }

type ItemHeap []Item
func (h ItemHeap) Len() int { return len(h) }
func (h ItemHeap) Less(i, j int) bool {
	if h[i].key != h[j].key {
		return h[i].key < h[j].key
	}
	return h[i].a < h[j].a
}
func (h ItemHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *ItemHeap) Push(x interface{}) { *h = append(*h, x.(Item)) }
func (h *ItemHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

func topKFrequent(nums []int, k int) []int {
	cnt := map[int]int{}
	for _, x := range nums {
		cnt[x]++
	}
	h := &ItemHeap{} // key = count, a = value
	for v, c := range cnt {
		heap.Push(h, Item{c, v, 0})
		if h.Len() > k {
			heap.Pop(h)
		}
	}
	out := make([]int, h.Len())
	for i := len(out) - 1; i >= 0; i-- {
		out[i] = heap.Pop(h).(Item).a // most frequent first
	}
	return out
}
#include <bits/stdc++.h>         // C++14. priority_queue is a MAX-heap unless you pass greater<>
using namespace std;

// Top-K: size-k MIN-heap, root = k-th largest. O(n log k).
int kthLargest(const vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> h;     // min-heap
    for (int x : nums) {
        h.push(x);
        if ((int)h.size() > k) h.pop();                     // drop the smallest of k+1
    }
    return h.top();
}

vector<int> topKFrequent(const vector<int>& nums, int k) {
    unordered_map<int, int> cnt;
    for (int x : nums) cnt[x]++;
    typedef pair<int, int> P;                               // (count, value)
    priority_queue<P, vector<P>, greater<P>> h;
    for (auto& kv : cnt) {
        h.push(P(kv.second, kv.first));
        if ((int)h.size() > k) h.pop();
    }
    vector<int> out;
    while (!h.empty()) { out.push_back(h.top().second); h.pop(); }
    reverse(out.begin(), out.end());                        // most frequent first
    return out;
}
import java.util.*;              // PriorityQueue is a MIN-heap unless you pass a comparator

// Top-K: size-k MIN-heap, root = k-th largest. O(n log k).
static int kthLargest(int[] nums, int k) {
    PriorityQueue<Integer> h = new PriorityQueue<>();       // min-heap by default
    for (int x : nums) {
        h.offer(x);
        if (h.size() > k) h.poll();                         // drop the smallest of k+1
    }
    return h.peek();
}

static int[] topKFrequent(int[] nums, int k) {
    Map<Integer, Integer> cnt = new HashMap<>();
    for (int x : nums) cnt.merge(x, 1, Integer::sum);
    PriorityQueue<int[]> h = new PriorityQueue<>((a, b) -> Integer.compare(a[0], b[0]));  // (count, value)
    for (Map.Entry<Integer, Integer> e : cnt.entrySet()) {
        h.offer(new int[]{e.getValue(), e.getKey()});
        if (h.size() > k) h.poll();
    }
    int[] out = new int[h.size()];
    for (int i = out.length - 1; i >= 0; i--) out[i] = h.poll()[1];   // most frequent first
    return out;
}
import heapq                     # heapq is a MIN-heap; push -x (or (-key, ...)) for a max-heap
from collections import Counter

def kth_largest(nums, k):
    """Size-k MIN-heap: the root is the k-th largest seen so far. O(n log k)."""
    h = []
    for x in nums:
        heapq.heappush(h, x)
        if len(h) > k:
            heapq.heappop(h)          # throw away the smallest of the k+1
    return h[0]

def top_k_frequent(nums, k):
    """Count, then keep the k most frequent in a size-k min-heap keyed by count."""
    cnt = Counter(nums)
    h = []
    for val, c in cnt.items():
        heapq.heappush(h, (c, val))
        if len(h) > k:
            heapq.heappop(h)
    return [val for c, val in sorted(h, reverse=True)]

The flip rule. To keep the k smallest things, use a max-heap and evict the biggest: the root must be the worst of the k you keep.
