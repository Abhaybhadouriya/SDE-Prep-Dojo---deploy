All four use the same lo < hi loop with the comparison against nums[hi].

func findMin(nums []int) int {
	lo, hi := 0, len(nums)-1 // invariant: the minimum is in [lo, hi]
	for lo < hi {
		mid := lo + (hi-lo)/2 // mid < hi always (rounds down)
		if nums[mid] > nums[hi] {
			lo = mid + 1 // mid is in the high run: the drop is to its right
		} else {
			hi = mid // mid..hi is sorted: mid may be the minimum
		}
	}
	return nums[lo] // lo == hi: the start of the low run
}
