Template: the mock picker and grader

About 40 lines in each language: pick_mock applies the four-part picking rule to a list of problems, and grade applies the level bar to your four scores. Feed it a company's Tier A rows (slug, difficulty, group, frequency), copied from the company view, and your seen set. Every version prints the same output for the same Amazon sample.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

package main

import (
	"fmt"
	"sort"
	"strings"
)

type Prob struct {
	Slug, Difficulty, Group string // Difficulty: Easy | Medium | Hard
	Freq                    int    // company frequency 1..8 (8 = most asked)
}

// allowed difficulty of problem 1 and problem 2 for each level
var mix = map[string][2]map[string]bool{
	"SDE1": {{"Easy": true, "Medium": true}, {"Medium": true}},
	"SDE2": {{"Medium": true}, {"Medium": true, "Hard": true}},
	"SDE3": {{"Medium": true}, {"Hard": true}},
}

// pickMock returns two problems for one 45-minute mock: unseen, most asked first, different groups.
func pickMock(probs []Prob, seen map[string]bool, level string) []Prob {
	m := mix[level]
	pool := []Prob{}
	for _, p := range probs {
		if !seen[p.Slug] {
			pool = append(pool, p)
		}
	}
	sort.Slice(pool, func(i, j int) bool {
		if pool[i].Freq != pool[j].Freq {
			return pool[i].Freq > pool[j].Freq
		}
		return pool[i].Slug < pool[j].Slug
	})
	out := []Prob{}
	for _, p := range pool {
		if m[0][p.Difficulty] {
			out = append(out, p)
			break
		}
	}
	if len(out) == 0 {
		return out
	}
	for _, p := range pool {
		if m[1][p.Difficulty] && p.Slug != out[0].Slug && p.Group != out[0].Group {
			out = append(out, p)
			break
		}
	}
	return out
}

// bar per level: {problems solved and tested, lowest allowed signal, minimum total of 16}
var bar = map[string][3]int{"SDE1": {1, 2, 11}, "SDE2": {2, 2, 12}, "SDE3": {2, 3, 13}}

// grade: scores = {communication, problem solving, code, testing}, each 1..4
func grade(scores []int, solved int, level string) string {
	b := bar[level]
	total, low := 0, 4
	for _, s := range scores {
		total += s
		if s < low {
			low = s
		}
	}
	misses := 0
	if solved < b[0] {
		misses++
	}
	if low < b[1] {
		misses++
	}
	if total < b[2] {
		misses++
	}
	switch {
	case misses == 0 && total >= 15:
		return "strong hire"
	case misses == 0:
		return "hire"
	case misses == 1:
		return "lean no hire"
	}
	return "no hire"
}

#include <algorithm>
#include <iostream>
#include <map>
#include <set>
#include <string>
#include <vector>
using namespace std;

struct Prob {
    string slug, difficulty, group;  // difficulty: Easy | Medium | Hard
    int freq;                        // company frequency 1..8 (8 = most asked)
};

// allowed difficulty of problem 1 and problem 2 for each level
const map<string, pair<set<string>, set<string>>> MIX = {
    {"SDE1", {{"Easy", "Medium"}, {"Medium"}}},
    {"SDE2", {{"Medium"}, {"Medium", "Hard"}}},
    {"SDE3", {{"Medium"}, {"Hard"}}}};

// Two problems for one 45-minute mock: unseen, most asked first, different groups.
vector<Prob> pickMock(const vector<Prob>& probs, const set<string>& seen, const string& level) {
    const auto& mix = MIX.at(level);
    vector<Prob> pool;
    for (const auto& p : probs) if (!seen.count(p.slug)) pool.push_back(p);
    sort(pool.begin(), pool.end(), [](const Prob& a, const Prob& b) {
        return a.freq != b.freq ? a.freq > b.freq : a.slug < b.slug;
    });
    vector<Prob> out;
    for (const auto& p : pool) if (mix.first.count(p.difficulty)) { out.push_back(p); break; }
    if (out.empty()) return out;
    for (const auto& p : pool)
        if (mix.second.count(p.difficulty) && p.slug != out[0].slug && p.group != out[0].group) {
            out.push_back(p);
            break;
        }
    return out;
}

// bar per level: {problems solved and tested, lowest allowed signal, minimum total of 16}
const map<string, vector<int>> BAR = {{"SDE1", {1, 2, 11}}, {"SDE2", {2, 2, 12}}, {"SDE3", {2, 3, 13}}};

// scores = {communication, problem solving, code, testing}, each 1..4
string grade(const vector<int>& scores, int solved, const string& level) {
    const auto& bar = BAR.at(level);
    int total = 0, low = 4;
    for (int s : scores) { total += s; low = min(low, s); }
    int misses = (solved < bar[0]) + (low < bar[1]) + (total < bar[2]);
    if (misses == 0) return total >= 15 ? "strong hire" : "hire";
    return misses == 1 ? "lean no hire" : "no hire";
}

import java.util.*;

public class Templates {
    record Prob(String slug, String difficulty, String group, int freq) {}  // freq 1..8, 8 = most asked

    // allowed difficulty of problem 1 and problem 2 for each level
    static final Map<String, List<Set<String>>> MIX = Map.of(
        "SDE1", List.of(Set.of("Easy", "Medium"), Set.of("Medium")),
        "SDE2", List.of(Set.of("Medium"), Set.of("Medium", "Hard")),
        "SDE3", List.of(Set.of("Medium"), Set.of("Hard")));

    // Two problems for one 45-minute mock: unseen, most asked first, different groups.
    static List<Prob> pickMock(List<Prob> probs, Set<String> seen, String level) {
        List<Set<String>> mix = MIX.get(level);
        List<Prob> pool = new ArrayList<>();
        for (Prob p : probs) if (!seen.contains(p.slug())) pool.add(p);
        pool.sort(Comparator.comparingInt((Prob p) -> -p.freq()).thenComparing(Prob::slug));
        List<Prob> out = new ArrayList<>();
        for (Prob p : pool) if (mix.get(0).contains(p.difficulty())) { out.add(p); break; }
        if (out.isEmpty()) return out;
        Prob p1 = out.get(0);
        for (Prob p : pool)
            if (mix.get(1).contains(p.difficulty()) && !p.slug().equals(p1.slug()) && !p.group().equals(p1.group())) {
                out.add(p);
                break;
            }
        return out;
    }

    // bar per level: {problems solved and tested, lowest allowed signal, minimum total of 16}
    static final Map<String, int[]> BAR = Map.of(
        "SDE1", new int[]{1, 2, 11}, "SDE2", new int[]{2, 2, 12}, "SDE3", new int[]{2, 3, 13});

    // scores = {communication, problem solving, code, testing}, each 1..4
    static String grade(int[] scores, int solved, String level) {
        int[] bar = BAR.get(level);
        int total = 0, low = 4;
        for (int s : scores) { total += s; low = Math.min(low, s); }
        int misses = (solved < bar[0] ? 1 : 0) + (low < bar[1] ? 1 : 0) + (total < bar[2] ? 1 : 0);
        if (misses == 0) return total >= 15 ? "strong hire" : "hire";
        return misses == 1 ? "lean no hire" : "no hire";
    }

    public static void main(String[] args) {
        List<Prob> amazon = List.of(
            new Prob("two-sum", "Easy", "arrays", 8),
            new Prob("best-time-to-buy-and-sell-stock", "Easy", "arrays", 7),
            new Prob("lru-cache", "Medium", "design", 7),
            new Prob("number-of-islands", "Medium", "graphs", 7),
            new Prob("longest-substring-without-repeating-characters", "Medium", "sliding-window", 7),
            new Prob("trapping-rain-water", "Hard", "arrays", 7),
            new Prob("group-anagrams", "Medium", "arrays", 6),
            new Prob("merge-k-sorted-lists", "Hard", "heap", 6));
        Set<String> seen = Set.of("two-sum", "lru-cache");
        for (String level : List.of("SDE1", "SDE2", "SDE3")) {
            StringJoiner j = new StringJoiner(" + ");
            for (Prob p : pickMock(amazon, seen, level)) j.add(p.slug());
            System.out.println(level + " -> " + j);
        }
        System.out.println("grade SDE2 [3,3,3,2] solved 2: " + grade(new int[]{3, 3, 3, 2}, 2, "SDE2"));
        System.out.println("grade SDE2 [4,3,3,1] solved 2: " + grade(new int[]{4, 3, 3, 1}, 2, "SDE2"));
        System.out.println("grade SDE3 [4,4,4,3] solved 2: " + grade(new int[]{4, 4, 4, 3}, 2, "SDE3"));
        System.out.println("grade SDE1 [2,2,2,2] solved 0: " + grade(new int[]{2, 2, 2, 2}, 0, "SDE1"));
    }
}
from dataclasses import dataclass

@dataclass
class Prob:
    slug: str
    difficulty: str   # "Easy" | "Medium" | "Hard"
    group: str        # pattern group, e.g. "graphs"
    freq: int         # company frequency 1..8 (8 = most asked)

# allowed difficulty of problem 1 and problem 2 for each level
MIX = {"SDE1": (("Easy", "Medium"), ("Medium",)),
       "SDE2": (("Medium",), ("Medium", "Hard")),
       "SDE3": (("Medium",), ("Hard",))}


def pick_mock(probs, seen, level):
    """Two problems for one 45-minute mock: unseen, most asked first, different groups."""
    first_ok, second_ok = MIX[level]
    pool = sorted((p for p in probs if p.slug not in seen), key=lambda p: (-p.freq, p.slug))
    p1 = next((p for p in pool if p.difficulty in first_ok), None)
    if p1 is None:
        return []
    p2 = next((p for p in pool if p.difficulty in second_ok
               and p.slug != p1.slug and p.group != p1.group), None)
    return [p1] if p2 is None else [p1, p2]


# bar per level: (problems solved and tested, lowest allowed signal, minimum total of 16)
BAR = {"SDE1": (1, 2, 11), "SDE2": (2, 2, 12), "SDE3": (2, 3, 13)}


def grade(scores, solved, level):
    """scores = [communication, problem solving, code, testing], each 1..4."""
    need_solved, floor, need_total = BAR[level]
    misses = (solved < need_solved) + (min(scores) < floor) + (sum(scores) < need_total)
    if misses == 0:
        return "strong hire" if sum(scores) >= 15 else "hire"
    return "lean no hire" if misses == 1 else "no hire"

  • Deterministic on purpose. Sorting by frequency, then by slug, means the same seen set always gives the same mock. If you want variety, shuffle within one frequency value, but keep the different-group rule.
  • May return one problem. If no problem 2 qualifies (a small list, a big seen set), you get one problem: take a Medium from another company for problem 2 rather than relaxing the group rule.
  • The grader counts misses, not points. A 4 in code does not buy back a 1 in testing, the same way a real debrief does not average away a red flag.
