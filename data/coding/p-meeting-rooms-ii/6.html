class Solution {
public:
    int minMeetingRooms(vector<vector<int>>& intervals) {
        int n = intervals.size();
        vector<int> starts(n), ends(n);
        for (int i = 0; i < n; i++) {
            starts[i] = intervals[i][0];
            ends[i] = intervals[i][1];
        }
        sort(starts.begin(), starts.end());
        sort(ends.begin(), ends.end());
        int rooms = 0, e = 0;
        for (int s : starts) {
            if (s < ends[e]) rooms++;   // nothing has ended yet: open a new room
            else e++;                   // the earliest-ending meeting is over: reuse its room
        }
        return rooms;
    }
};
func minMeetingRooms(intervals [][]int) int {
	n := len(intervals)
	starts, ends := make([]int, n), make([]int, n)
	for i, m := range intervals {
		starts[i], ends[i] = m[0], m[1]
	}
	sort.Ints(starts)
	sort.Ints(ends)
	rooms, e := 0, 0
	for _, s := range starts {
		if s < ends[e] {
			rooms++ // nothing has ended yet: open a new room
		} else {
			e++ // the earliest-ending meeting is over: reuse its room
		}
	}
	return rooms
}
