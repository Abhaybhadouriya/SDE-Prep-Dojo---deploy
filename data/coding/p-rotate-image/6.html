func rotate(matrix [][]int) {
	n := len(matrix)
	// 1. transpose: mirror across the main diagonal (swap only above it, or you undo the swap)
	for r := 0; r < n; r++ {
		for c := r + 1; c < n; c++ {
			matrix[r][c], matrix[c][r] = matrix[c][r], matrix[r][c]
		}
	}
	// 2. mirror left-right: reverse each row with two pointers
	for _, row := range matrix {
		for lo, hi := 0, n-1; lo < hi; lo, hi = lo+1, hi-1 {
			row[lo], row[hi] = row[hi], row[lo]
		}
	}
}

public:
    void rotate(vector<vector<int>>& matrix) {
        int n = matrix.size();
        // 1. transpose: mirror across the main diagonal (swap only above it, or you undo the swap)
        for (int r = 0; r < n; r++)
            for (int c = r + 1; c < n; c++)
                swap(matrix[r][c], matrix[c][r]);
        // 2. mirror left-right: reverse each row
        for (int r = 0; r < n; r++)
            reverse(matrix[r].begin(), matrix[r].end());
    }
};

        for (int r = 0; r < n; r++)
            for (int c = r + 1; c < n; c++) {
                int t = matrix[r][c];
                matrix[r][c] = matrix[c][r];
                matrix[c][r] = t;
            }
        // 2. mirror left-right: reverse each row with two pointers
        for (int[] row : matrix)
            for (int lo = 0, hi = n - 1; lo < hi; lo++, hi--) {
                int t = row[lo];
                row[lo] = row[hi];
                row[hi] = t;
            }
    }
}
