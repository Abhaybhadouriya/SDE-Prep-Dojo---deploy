Reusable templates in four languages

Six templates in one file per language: subsets, subsets with duplicates, permutations, combination sum (both versions behind a reuse flag), palindrome partitioning, word search, and the bitmask N-Queens count.All four versions (Go, C++, Java, Python) were compiled and run with the asserts in main (empty input, a single element, no solution, board restored after the search, n = 1, 4 and 8 queens).

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

// Backtracking templates: choose -> explore -> unchoose

// 1. Subsets: every node of the tree is an answer
func subsets(nums []int) [][]int {
	out, path := [][]int{}, []int{}
	var dfs func(start int)
	dfs = func(start int) {
		cp := make([]int, len(path)) // record a COPY of the current path
		copy(cp, path)
		out = append(out, cp)
		for i := start; i < len(nums); i++ {
			path = append(path, nums[i]) // choose
			dfs(i + 1)                   // explore (i + 1: each item used once)
			path = path[:len(path)-1]    // unchoose
		}
	}
	dfs(0)
	return out
}

// 1b. Subsets with duplicates: sort, then skip equal values at the same depth
func subsetsWithDup(nums []int) [][]int {
	a := append([]int(nil), nums...)
	sort.Ints(a)
	out, path := [][]int{}, []int{}
	var dfs func(start int)
	dfs = func(start int) {
		cp := make([]int, len(path))
		copy(cp, path)
		out = append(out, cp)
		for i := start; i < len(a); i++ {
			if i > start && a[i] == a[i-1] {
				continue // same value already tried at this depth
			}
			path = append(path, a[i])
			dfs(i + 1)
			path = path[:len(path)-1]
		}
	}
	dfs(0)
	return out
}

// 2. Permutations: order matters, so track what is used instead of a start index
func permute(nums []int) [][]int {
	out, path, used := [][]int{}, []int{}, make([]bool, len(nums))
	var dfs func()
	dfs = func() {
		if len(path) == len(nums) {
			out = append(out, append([]int(nil), path...))
			return
		}
		for i := range nums {
			if used[i] {
				continue
			}
			used[i] = true
			path = append(path, nums[i])
			dfs()
			path = path[:len(path)-1]
			used[i] = false
		}
	}
	dfs()
	return out
}

// 3. Combination sum: reuse allowed -> recurse with i; no reuse + dedup -> i + 1 and skip
func combinationSum(cands []int, target int, reuse bool) [][]int {
	a := append([]int(nil), cands...)
	sort.Ints(a)
	out, path := [][]int{}, []int{}
	var dfs func(start, remain int)
	dfs = func(start, remain int) {
		if remain == 0 {
			out = append(out, append([]int(nil), path...))
			return
		}
		for i := start; i < len(a); i++ {
			if a[i] > remain {
				break // sorted: every later value is too big as well
			}
			if !reuse && i > start && a[i] == a[i-1] {
				continue // dedup at the same depth
			}
			path = append(path, a[i])
			next := i + 1
			if reuse {
				next = i
			}
			dfs(next, remain-a[i])
			path = path[:len(path)-1]
		}
	}
	dfs(0, target)
	return out
}

// 4. String partitioning: the choice is "where does the next piece end?"
func partition(s string) [][]string {
	out, path := [][]string{}, []string{}
	isPal := func(a, b int) bool {
		for a < b {
			if s[a] != s[b] {
				return false
			}
			a++
			b--
		}
		return true
	}
	var dfs func(start int)
	dfs = func(start int) {
		if start == len(s) {
			out = append(out, append([]string(nil), path...))
			return
		}
		for end := start; end < len(s); end++ {
			if !isPal(start, end) {
				continue // prune: only cut where the piece is valid
			}
			path = append(path, s[start:end+1])
			dfs(end + 1)
			path = path[:len(path)-1]
		}
	}
	dfs(0)
	return out
}

// 5. Board search: mark the cell in place, restore it on the way back
func exist(board [][]byte, word string) bool {
	rows, cols := len(board), len(board[0])
	var dfs func(r, c, k int) bool
	dfs = func(r, c, k int) bool {
		if k == len(word) {
			return true
		}
		if r < 0 || c < 0 || r >= rows || c >= cols || board[r][c] != word[k] {
			return false
		}
		saved := board[r][c]
		board[r][c] = '#' // choose (mark visited)
		found := dfs(r+1, c, k+1) || dfs(r-1, c, k+1) || dfs(r, c+1, k+1) || dfs(r, c-1, k+1)
		board[r][c] = saved // unchoose (restore)
		return found
	}
	for r := 0; r < rows; r++ {
		for c := 0; c < cols; c++ {
			if dfs(r, c, 0) {
				return true
			}
		}
	}
	return false
}

// 6. N-Queens count with bitmasks: cols, and the two diagonals shifted per row
func totalNQueens(n int) int {
	full := (1 << n) - 1
	var place func(cols, d1, d2 int) int
	place = func(cols, d1, d2 int) int {
		if cols == full {
			return 1
		}
		count := 0
		free := full &^ (cols | d1 | d2)
		for free != 0 {
			bit := free & -free // lowest free column
			free -= bit
			count += place(cols|bit, ((d1|bit)<<1)&full, (d2|bit)>>1)
		}
		return count
	}
	return place(0, 0, 0)
}

}

func main() {
	check(len(subsets([]int{1, 2, 3})) == 8 && len(subsets([]int{})) == 1, "subsets")
	check(reflect.DeepEqual(subsetsWithDup([]int{2, 1, 2}),
		[][]int{{}, {1}, {1, 2}, {1, 2, 2}, {2}, {2, 2}}), "subsetsWithDup")
	check(len(permute([]int{1, 2, 3})) == 6 && len(permute([]int{7})) == 1, "permute")
	check(reflect.DeepEqual(combinationSum([]int{2, 3, 6, 7}, 7, true), [][]int{{2, 2, 3}, {7}}), "comb 1")
	check(reflect.DeepEqual(combinationSum([]int{10, 1, 2, 7, 6, 1, 5}, 8, false),
		[][]int{{1, 1, 6}, {1, 2, 5}, {1, 7}, {2, 6}}), "comb 2")
	check(len(combinationSum([]int{2}, 1, true)) == 0, "comb 3")
	check(reflect.DeepEqual(partition("aab"), [][]string{{"a", "a", "b"}, {"aa", "b"}}), "partition")
	b := [][]byte{[]byte("ABCE"), []byte("SFCS"), []byte("ADEE")}
	check(exist(b, "ABCCED") && exist(b, "SEE") && !exist(b, "ABCB"), "exist")
	check(string(b[0]) == "ABCE" && string(b[1]) == "SFCS", "board restored")
	check(totalNQueens(1) == 1 && totalNQueens(4) == 2 && totalNQueens(8) == 92, "queens")

// Backtracking templates: choose -> explore -> unchoose

// 1. Subsets: every node of the tree is an answer
void subsetsDfs(const vector<int>& a, int start, vector<int>& path, vector<vector<int>>& out) {
    out.push_back(path);                         // record a COPY of the current path
    for (int i = start; i < (int)a.size(); ++i) {
        path.push_back(a[i]);                    // choose
        subsetsDfs(a, i + 1, path, out);         // explore (i + 1: each item used once)
        path.pop_back();                         // unchoose
    }
}
vector<vector<int>> subsets(const vector<int>& nums) {
    vector<vector<int>> out; vector<int> path;
    subsetsDfs(nums, 0, path, out);
    return out;
}

// 1b. Subsets with duplicates: sort, then skip equal values at the same depth
void dupDfs(const vector<int>& a, int start, vector<int>& path, vector<vector<int>>& out) {
    out.push_back(path);
    for (int i = start; i < (int)a.size(); ++i) {
        if (i > start && a[i] == a[i - 1]) continue;   // same value already tried at this depth
        path.push_back(a[i]);
        dupDfs(a, i + 1, path, out);
        path.pop_back();
    }
}
vector<vector<int>> subsetsWithDup(vector<int> nums) {
    sort(nums.begin(), nums.end());
    vector<vector<int>> out; vector<int> path;
    dupDfs(nums, 0, path, out);
    return out;
}

// 2. Permutations: order matters, so track what is used instead of a start index
void permDfs(const vector<int>& a, vector<bool>& used, vector<int>& path, vector<vector<int>>& out) {
    if (path.size() == a.size()) { out.push_back(path); return; }
    for (int i = 0; i < (int)a.size(); ++i) {
        if (used[i]) continue;
        used[i] = true; path.push_back(a[i]);
        permDfs(a, used, path, out);
        path.pop_back(); used[i] = false;
    }
}
vector<vector<int>> permute(const vector<int>& nums) {
    vector<vector<int>> out; vector<int> path; vector<bool> used(nums.size(), false);
    permDfs(nums, used, path, out);
    return out;
}

// 3. Combination sum: reuse allowed -> recurse with i; no reuse + dedup -> i + 1 and skip
void combDfs(const vector<int>& a, int start, int remain, bool reuse,
             vector<int>& path, vector<vector<int>>& out) {
    if (remain == 0) { out.push_back(path); return; }
    for (int i = start; i < (int)a.size(); ++i) {
        if (a[i] > remain) break;                       // sorted: every later value is too big as well
        if (!reuse && i > start && a[i] == a[i - 1]) continue;   // dedup at the same depth
        path.push_back(a[i]);
        combDfs(a, reuse ? i : i + 1, remain - a[i], reuse, path, out);
        path.pop_back();
    }
}
vector<vector<int>> combinationSum(vector<int> cands, int target, bool reuse) {
    sort(cands.begin(), cands.end());
    vector<vector<int>> out; vector<int> path;
    combDfs(cands, 0, target, reuse, path, out);
    return out;
}

// 4. String partitioning: the choice is "where does the next piece end?"
bool isPal(const string& s, int a, int b) {
    while (a < b) if (s[a++] != s[b--]) return false;
    return true;
}
void partDfs(const string& s, int start, vector<string>& path, vector<vector<string>>& out) {
    if (start == (int)s.size()) { out.push_back(path); return; }
    for (int end = start; end < (int)s.size(); ++end) {
        if (!isPal(s, start, end)) continue;            // prune: only cut where the piece is valid
        path.push_back(s.substr(start, end - start + 1));
        partDfs(s, end + 1, path, out);
        path.pop_back();
    }
}
vector<vector<string>> partition(const string& s) {
    vector<vector<string>> out; vector<string> path;
    partDfs(s, 0, path, out);
    return out;
}

// 5. Board search: mark the cell in place, restore it on the way back
bool wordDfs(vector<vector<char>>& b, const string& w, int r, int c, int k) {
    if (k == (int)w.size()) return true;
    if (r < 0 || c < 0 || r >= (int)b.size() || c >= (int)b[0].size() || b[r][c] != w[k]) return false;
    char saved = b[r][c];
    b[r][c] = '#';                                      // choose (mark visited)
    bool found = wordDfs(b, w, r + 1, c, k + 1) || wordDfs(b, w, r - 1, c, k + 1) ||
                 wordDfs(b, w, r, c + 1, k + 1) || wordDfs(b, w, r, c - 1, k + 1);
    b[r][c] = saved;                                    // unchoose (restore)
    return found;
}
bool exist(vector<vector<char>>& board, const string& word) {
    for (int r = 0; r < (int)board.size(); ++r)
        for (int c = 0; c < (int)board[0].size(); ++c)
            if (wordDfs(board, word, r, c, 0)) return true;
    return false;
}

// 6. N-Queens count with bitmasks: cols, and the two diagonals shifted per row
int place(int full, int cols, int d1, int d2) {
    if (cols == full) return 1;
    int count = 0;
    int freeCols = full & ~(cols | d1 | d2);
    while (freeCols) {
        int bit = freeCols & -freeCols;                 // lowest free column
        freeCols -= bit;
        count += place(full, cols | bit, ((d1 | bit) << 1) & full, (d2 | bit) >> 1);
    }
    return count;
}
int totalNQueens(int n) { return place((1 << n) - 1, 0, 0, 0); }

// Backtracking templates: choose -> explore -> unchoose
public class Backtracking {

    // 1. Subsets: every node of the tree is an answer
    static List<List<Integer>> subsets(int[] nums) {
        List<List<Integer>> out = new ArrayList<>();
        subsetsDfs(nums, 0, new ArrayList<>(), out);
        return out;
    }
    static void subsetsDfs(int[] a, int start, List<Integer> path, List<List<Integer>> out) {
        out.add(new ArrayList<>(path));              // record a COPY of the current path
        for (int i = start; i < a.length; i++) {
            path.add(a[i]);                          // choose
            subsetsDfs(a, i + 1, path, out);         // explore (i + 1: each item used once)
            path.remove(path.size() - 1);            // unchoose
        }
    }

    // 1b. Subsets with duplicates: sort, then skip equal values at the same depth
    static List<List<Integer>> subsetsWithDup(int[] nums) {
        int[] a = nums.clone();
        Arrays.sort(a);
        List<List<Integer>> out = new ArrayList<>();
        dupDfs(a, 0, new ArrayList<>(), out);
        return out;
    }
    static void dupDfs(int[] a, int start, List<Integer> path, List<List<Integer>> out) {
        out.add(new ArrayList<>(path));
        for (int i = start; i < a.length; i++) {
            if (i > start && a[i] == a[i - 1]) continue;  // same value already tried at this depth
            path.add(a[i]);
            dupDfs(a, i + 1, path, out);
            path.remove(path.size() - 1);
        }
    }

    // 2. Permutations: order matters, so track what is used instead of a start index
    static List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> out = new ArrayList<>();
        permDfs(nums, new boolean[nums.length], new ArrayList<>(), out);
        return out;
    }
    static void permDfs(int[] a, boolean[] used, List<Integer> path, List<List<Integer>> out) {
        if (path.size() == a.length) { out.add(new ArrayList<>(path)); return; }
        for (int i = 0; i < a.length; i++) {
            if (used[i]) continue;
            used[i] = true; path.add(a[i]);
            permDfs(a, used, path, out);
            path.remove(path.size() - 1); used[i] = false;
        }
    }

    // 3. Combination sum: reuse allowed -> recurse with i; no reuse + dedup -> i + 1 and skip
    static List<List<Integer>> combinationSum(int[] cands, int target, boolean reuse) {
        int[] a = cands.clone();
        Arrays.sort(a);
        List<List<Integer>> out = new ArrayList<>();
        combDfs(a, 0, target, reuse, new ArrayList<>(), out);
        return out;
    }
    static void combDfs(int[] a, int start, int remain, boolean reuse, List<Integer> path, List<List<Integer>> out) {
        if (remain == 0) { out.add(new ArrayList<>(path)); return; }
        for (int i = start; i < a.length; i++) {
            if (a[i] > remain) break;                          // sorted: every later value is too big as well
            if (!reuse && i > start && a[i] == a[i - 1]) continue;   // dedup at the same depth
            path.add(a[i]);
            combDfs(a, reuse ? i : i + 1, remain - a[i], reuse, path, out);
            path.remove(path.size() - 1);
        }
    }

    // 4. String partitioning: the choice is "where does the next piece end?"
    static List<List<String>> partition(String s) {
        List<List<String>> out = new ArrayList<>();
        partDfs(s, 0, new ArrayList<>(), out);
        return out;
    }
    static boolean isPal(String s, int a, int b) {
        while (a < b) if (s.charAt(a++) != s.charAt(b--)) return false;
        return true;
    }
    static void partDfs(String s, int start, List<String> path, List<List<String>> out) {
        if (start == s.length()) { out.add(new ArrayList<>(path)); return; }
        for (int end = start; end < s.length(); end++) {
            if (!isPal(s, start, end)) continue;               // prune: only cut where the piece is valid
            path.add(s.substring(start, end + 1));
            partDfs(s, end + 1, path, out);
            path.remove(path.size() - 1);
        }
    }

    // 5. Board search: mark the cell in place, restore it on the way back
    static boolean exist(char[][] board, String word) {
        for (int r = 0; r < board.length; r++)
            for (int c = 0; c < board[0].length; c++)
                if (wordDfs(board, word, r, c, 0)) return true;
        return false;
    }
    static boolean wordDfs(char[][] b, String w, int r, int c, int k) {
        if (k == w.length()) return true;
        if (r < 0 || c < 0 || r >= b.length || c >= b[0].length || b[r][c] != w.charAt(k)) return false;
        char saved = b[r][c];
        b[r][c] = '#';                                         // choose (mark visited)
        boolean found = wordDfs(b, w, r + 1, c, k + 1) || wordDfs(b, w, r - 1, c, k + 1)
                     || wordDfs(b, w, r, c + 1, k + 1) || wordDfs(b, w, r, c - 1, k + 1);
        b[r][c] = saved;                                       // unchoose (restore)
        return found;
    }

    // 6. N-Queens count with bitmasks: cols, and the two diagonals shifted per row
    static int totalNQueens(int n) { return place((1 << n) - 1, 0, 0, 0); }
    static int place(int full, int cols, int d1, int d2) {
        if (cols == full) return 1;
        int count = 0;
        int free = full & ~(cols | d1 | d2);
        while (free != 0) {
            int bit = free & -free;                            // lowest free column
            free -= bit;
            count += place(full, cols | bit, ((d1 | bit) << 1) & full, (d2 | bit) >> 1);
        }
        return count;
    }

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError(what);
    }

    public static void main(String[] args) {

        List<List<Integer>> d = subsetsWithDup(new int[]{2, 1, 2});
        check(d.equals(List.of(List.of(), List.of(1), List.of(1, 2), List.of(1, 2, 2), List.of(2), List.of(2, 2))), "subsetsWithDup");

# Backtracking templates: choose -> explore -> unchoose


# 1. Subsets: every node of the tree is an answer
def subsets(nums):
    out, path = [], []

    def dfs(start):
        out.append(path[:])            # record a COPY of the current path
        for i in range(start, len(nums)):
            path.append(nums[i])       # choose
            dfs(i + 1)                 # explore (i + 1: each item used once)
            path.pop()                 # unchoose
    dfs(0)
    return out


# 1b. Subsets with duplicates: sort, then skip equal values at the same depth
def subsets_with_dup(nums):
    nums = sorted(nums)
    out, path = [], []

    def dfs(start):
        out.append(path[:])
        for i in range(start, len(nums)):
            if i > start and nums[i] == nums[i - 1]:
                continue               # same value already tried at this depth
            path.append(nums[i])
            dfs(i + 1)
            path.pop()
    dfs(0)
    return out


# 2. Permutations: order matters, so track what is used instead of a start index
def permute(nums):
    out, path, used = [], [], [False] * len(nums)

    def dfs():
        if len(path) == len(nums):
            out.append(path[:])
            return
        for i in range(len(nums)):
            if used[i]:
                continue
            used[i] = True
            path.append(nums[i])
            dfs()
            path.pop()
            used[i] = False
    dfs()
    return out


# 3. Combination sum: reuse allowed -> recurse with i; no reuse + dedup -> i + 1 and skip
def combination_sum(cands, target, reuse):
    cands = sorted(cands)
    out, path = [], []

    def dfs(start, remain):
        if remain == 0:
            out.append(path[:])
            return
        for i in range(start, len(cands)):
            if cands[i] > remain:
                break                  # sorted: every later value is too big as well
            if not reuse and i > start and cands[i] == cands[i - 1]:
                continue               # dedup at the same depth
            path.append(cands[i])
            dfs(i if reuse else i + 1, remain - cands[i])
            path.pop()
    dfs(0, target)
    return out


# 4. String partitioning: the choice is "where does the next piece end?"
def partition(s):
    out, path = [], []

    def is_pal(a, b):
        while a < b:
            if s[a] != s[b]:
                return False
            a, b = a + 1, b - 1
        return True

    def dfs(start):
        if start == len(s):
            out.append(path[:])
            return
        for end in range(start, len(s)):
            if is_pal(start, end):     # prune: only cut where the piece is valid
                path.append(s[start:end + 1])
                dfs(end + 1)
                path.pop()
    dfs(0)
    return out


# 5. Board search: mark the cell in place, restore it on the way back
def exist(board, word):
    rows, cols = len(board), len(board[0])

    def dfs(r, c, k):
        if k == len(word):
            return True
        if r < 0 or c < 0 or r >= rows or c >= cols or board[r][c] != word[k]:
            return False
        saved, board[r][c] = board[r][c], '#'     # choose (mark visited)
        found = (dfs(r + 1, c, k + 1) or dfs(r - 1, c, k + 1) or
                 dfs(r, c + 1, k + 1) or dfs(r, c - 1, k + 1))
        board[r][c] = saved                       # unchoose (restore)
        return found

    return any(dfs(r, c, 0) for r in range(rows) for c in range(cols))


# 6. N-Queens count with bitmasks: cols, and the two diagonals shifted per row
def total_n_queens(n):
    full = (1 << n) - 1

    def place(cols, d1, d2):
        if cols == full:
            return 1
        count = 0
        free = full & ~(cols | d1 | d2)
        while free:
            bit = free & -free         # lowest free column
            free -= bit
            count += place(cols | bit, ((d1 | bit) << 1) & full, (d2 | bit) >> 1)
        return count
    return place(0, 0, 0)
