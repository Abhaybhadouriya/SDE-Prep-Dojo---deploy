Templates that avoid four mistakes by construction

Three small functions: a binary search with a stated invariant and a safe midpoint (mistakes 2 and 3), a 32-bit reverse that checks before it overflows (3), and an island count that neither recurses nor writes to its input (4 and 5). The tests in main include the overflow cases and a 1000 x 1000 grid that would break a recursive version.

Problems behind the templates: LeetCode 35 - Search Insert Position (lower bound) · LeetCode 7 - Reverse Integer · LeetCode 200 - Number of Islands · our write-ups: Search Insert Position, Reverse Integer, Number of Islands.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

import (
	"fmt"
	"math"

// First index i with a[i] >= target (len(a) if none). Invariant: the answer is in [lo, hi].
func lowerBound(a []int, target int) int {
	lo, hi := 0, len(a) // half-open; hi == len(a) means "not found"
	for lo < hi {
		mid := lo + (hi-lo)/2 // (lo + hi) / 2 can overflow for int32 indices
		if a[mid] < target {
			lo = mid + 1 // mid and everything left of it is too small
		} else {
			hi = mid // mid could be the answer: keep it
		}
	}
	return lo
}

// Reverse the decimal digits of a 32-bit int; 0 if the result does not fit.
func reverseInt32(x int32) int32 {
	var r int32
	for x != 0 {
		d := x % 10 // Go keeps the sign: -123 % 10 == -3
		x /= 10
		if r > math.MaxInt32/10 || (r == math.MaxInt32/10 && d > 7) { // check BEFORE r*10+d
			return 0
		}
		if r < math.MinInt32/10 || (r == math.MinInt32/10 && d < -8) {
			return 0
		}
		r = r*10 + d
	}
	return r
}

// Number of 4-connected '1' regions. Iterative DFS, input left unchanged.
func countIslands(grid [][]byte) int {
	if len(grid) == 0 {
		return 0
	}
	R, C, islands := len(grid), len(grid[0]), 0
	seen := make([][]bool, R) // our own marks, not the caller's grid
	for i := range seen {
		seen[i] = make([]bool, C)
	}
	dirs := [4][2]int{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
	stack := [][2]int{} // explicit stack: depth lives on the heap
	for sr := 0; sr < R; sr++ {
		for sc := 0; sc < C; sc++ {
			if grid[sr][sc] != '1' || seen[sr][sc] {
				continue
			}
			islands++
			seen[sr][sc] = true
			stack = append(stack, [2]int{sr, sc})
			for len(stack) > 0 {
				cur := stack[len(stack)-1]
				stack = stack[:len(stack)-1]
				for _, d := range dirs {
					nr, nc := cur[0]+d[0], cur[1]+d[1]
					if nr >= 0 && nr < R && nc >= 0 && nc < C && grid[nr][nc] == '1' && !seen[nr][nc] {
						seen[nr][nc] = true // mark on push, so a cell is pushed once
						stack = append(stack, [2]int{nr, nc})
					}
				}
			}
		}
	}
	return islands
}

func main() {
	a := []int{1, 3, 3, 5, 8}
	fmt.Println(lowerBound(a, 4) == 3, lowerBound(a, 9) == 5, lowerBound(nil, 7) == 0)

	g := [][]byte{[]byte("11000"), []byte("11010"), []byte("00101")}
	fmt.Println(countIslands(g) == 4, g[0][0] == '1')
}

// First index i with a[i] >= target (a.size() if none). Invariant: the answer is in [lo, hi].
int lowerBound(const vector<int>& a, int target) {  // const&: we promise not to touch the input
    int lo = 0, hi = (int)a.size();                 // half-open; hi == size means "not found"
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;               // (lo + hi) / 2 overflows when both are near INT_MAX
        if (a[mid] < target) lo = mid + 1;          // mid and everything left of it is too small
        else hi = mid;                              // mid could be the answer: keep it
    }
    return lo;
}

// Reverse the decimal digits of a 32-bit int; 0 if the result does not fit.
int reverseInt32(int x) {
    int r = 0;
    while (x != 0) {
        int d = x % 10;                              // C++ keeps the sign: -123 % 10 == -3
        x /= 10;
        if (r > INT_MAX / 10 || (r == INT_MAX / 10 && d > 7)) return 0;   // check BEFORE r*10+d
        if (r < INT_MIN / 10 || (r == INT_MIN / 10 && d < -8)) return 0;
        r = r * 10 + d;
    }
    return r;
}

// Number of 4-connected '1' regions. Iterative DFS, input left unchanged.
int countIslands(const vector<string>& grid) {
    if (grid.empty()) return 0;
    int R = grid.size(), C = grid[0].size(), islands = 0;
    vector<vector<char>> seen(R, vector<char>(C, 0)); // our own marks, not the caller's grid
    vector<pair<int, int>> st;                        // explicit stack: depth lives on the heap
    const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    for (int sr = 0; sr < R; sr++)
        for (int sc = 0; sc < C; sc++) {
            if (grid[sr][sc] != '1' || seen[sr][sc]) continue;
            islands++;
            seen[sr][sc] = 1;
            st.push_back({sr, sc});
            while (!st.empty()) {
                pair<int, int> cur = st.back(); st.pop_back();
                for (int d = 0; d < 4; d++) {
                    int nr = cur.first + dr[d], nc = cur.second + dc[d];
                    if (nr >= 0 && nr < R && nc >= 0 && nc < C && grid[nr][nc] == '1' && !seen[nr][nc]) {
                        seen[nr][nc] = 1;             // mark on push, so a cell is pushed once
                        st.push_back({nr, nc});
                    }
                }
            }
        }
    return islands;
}

}

INT_MAX, INT_MIN = 2**31 - 1, -2**31


def lower_bound(a, target):
    """First index i with a[i] >= target (len(a) if none). Invariant: answer in [lo, hi]."""
    lo, hi = 0, len(a)            # half-open search space; hi = len(a) means "not found"
    while lo < hi:
        mid = lo + (hi - lo) // 2  # same habit as C/Java, where lo + hi can overflow
        if a[mid] < target:
            lo = mid + 1           # a[mid] and everything left of it is too small
        else:
            hi = mid               # mid could be the answer: keep it
    return lo


def reverse_int32(x):
    """Reverse decimal digits of a 32-bit int; return 0 if the result overflows."""
    sign = -1 if x < 0 else 1
    x = abs(x)
    r = 0
    while x:
        d = x % 10
        x //= 10
        if r > (INT_MAX - d) // 10:   # check BEFORE r * 10 + d leaves the range
            return 0
        r = r * 10 + d
    r *= sign
    return r if INT_MIN <= r <= INT_MAX else 0


def count_islands(grid):
    """Number of 4-connected '1' regions. Iterative DFS, input left unchanged."""
    if not grid:
        return 0
    rows, cols = len(grid), len(grid[0])
    seen = [[False] * cols for _ in range(rows)]   # our own marks, not the caller's grid
    islands = 0
    for sr in range(rows):
        for sc in range(cols):
            if grid[sr][sc] != '1' or seen[sr][sc]:
                continue
            islands += 1
            seen[sr][sc] = True
            stack = [(sr, sc)]                     # explicit stack: depth is heap memory
            while stack:
                r, c = stack.pop()
                for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == '1' and not seen[nr][nc]:
                        seen[nr][nc] = True        # mark on push, so a cell is pushed once
                        stack.append((nr, nc))
    return islands
