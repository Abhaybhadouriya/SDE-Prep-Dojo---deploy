class Solution {
public:
    bool checkInclusion(string s1, string s2) {
        int n = s1.size(), m = s2.size();
        if (n > m) return false;                 // the window cannot be longer than s2
        int diff[26] = {0};                      // count in s1 - count in the window
        for (int i = 0; i < n; i++) { diff[s1[i] - 'a']++; diff[s2[i] - 'a']--; }
        int matched = 0;                         // letters whose diff is 0 (out of 26)
        for (int d : diff) matched += (d == 0);
        // change diff[c] by delta and keep matched in sync (only a move to or from 0 matters)
        auto update = [&](int c, int delta) {
            if (diff[c] == 0) matched--;
            diff[c] += delta;
            if (diff[c] == 0) matched++;
        };
        for (int right = n; ; right++) {
            if (matched == 26) return true;      // every letter count agrees
            if (right == m) return false;
            update(s2[right] - 'a', -1);         // letter enters on the right
            update(s2[right - n] - 'a', +1);     // letter leaves on the left
        }
    }
};
func checkInclusion(s1 string, s2 string) bool {
	n, m := len(s1), len(s2)
	if n > m {
		return false // the window cannot be longer than s2
	}
	var diff [26]int // diff[c] = count of c in s1 - count of c in the window
	for i := 0; i < n; i++ {
		diff[s1[i]-'a']++
		diff[s2[i]-'a']--
	}
	matched := 0 // letters whose diff is 0 (out of 26)
	for _, d := range diff {
		if d == 0 {
			matched++
		}
	}
	// change diff[c] by delta and keep matched in sync (only a move to or from 0 matters)
	update := func(c byte, delta int) {
		if diff[c] == 0 {
			matched--
		}
		diff[c] += delta
		if diff[c] == 0 {
			matched++
		}
	}
	for right := n; ; right++ {
		if matched == 26 { // every letter count agrees: the window is a permutation of s1
			return true
		}
		if right == m {
			return false
		}
		update(s2[right]-'a', -1)   // letter enters on the right
		update(s2[right-n]-'a', +1) // letter leaves on the left
	}
}

class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        n, m = len(s1), len(s2)
        if n > m:
            return False                      # the window cannot be longer than s2
        diff = [0] * 26                       # count in s1 - count in the window
        for a, b in zip(s1, s2):
            diff[ord(a) - 97] += 1
            diff[ord(b) - 97] -= 1
        matched = diff.count(0)               # letters whose diff is 0 (out of 26)

        def update(c: int, delta: int) -> None:
            nonlocal matched                  # only a move to or from 0 changes matched
            if diff[c] == 0:
                matched -= 1
            diff[c] += delta
            if diff[c] == 0:
                matched += 1

        for right in range(n, m + 1):
            if matched == 26:                 # every letter count agrees
                return True
            if right == m:
                return False
            update(ord(s2[right]) - 97, -1)      # letter enters on the right
            update(ord(s2[right - n]) - 97, +1)  # letter leaves on the left
        return False
