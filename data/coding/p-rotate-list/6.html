All four count the length, reduce k, walk to the new tail and re-link. The k == 0 check after the modulo avoids building a ring that is never cut.

class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        if (!head || !head->next || k == 0) return head;
        // 1. Walk to the tail and count the nodes.
        int len = 1;
        ListNode* tail = head;
        while (tail->next) { tail = tail->next; ++len; }
        // 2. Rotating by len is a no-op, so only k % len matters.
        k %= len;
        if (k == 0) return head;
        // 3. The new tail is node (len - k), counting from 1 at head.
        ListNode* newTail = head;
        for (int i = 1; i < len - k; ++i) newTail = newTail->next;
        // 4. Re-link: old tail -> old head, cut after the new tail.
        ListNode* newHead = newTail->next;
        newTail->next = nullptr;
        tail->next = head;
        return newHead;
    }
};
func rotateRight(head *ListNode, k int) *ListNode {
	if head == nil || head.Next == nil || k == 0 {
		return head
	}
	// 1. Walk to the tail and count the nodes.
	length, tail := 1, head
	for tail.Next != nil {
		tail = tail.Next
		length++
	}
	// 2. Rotating by length is a no-op, so only k % length matters.
	k %= length
	if k == 0 {
		return head
	}
	// 3. The new tail is node (length - k), counting from 1 at head.
	newTail := head
	for i := 1; i < length-k; i++ {
		newTail = newTail.Next
	}
	// 4. Re-link: old tail -> old head, cut after the new tail.
	newHead := newTail.Next
	newTail.Next = nil
	tail.Next = head
	return newHead
}
