// Fenwick: point add, prefix sum. 0-based API, 1-based inside. O(log n) each.
type Fenwick struct{ t []int64 }

func NewFenwick(n int) *Fenwick { return &Fenwick{make([]int64, n+1)} }

func (f *Fenwick) Add(i int, delta int64) {
	for i++; i < len(f.t); i += i & -i { // climb
		f.t[i] += delta
	}
}

// Prefix returns a[0] + ... + a[i-1].
func (f *Fenwick) Prefix(i int) int64 {
	var s int64
	for ; i > 0; i -= i & -i { // drop the lowest set bit
		s += f.t[i]
	}
	return s
}

func (f *Fenwick) RangeSum(l, r int) int64 { return f.Prefix(r+1) - f.Prefix(l) }

// NumArray: Range Sum Query - Mutable; update = add the difference.
type NumArray struct {
	a   []int
	bit *Fenwick
}

func NewNumArray(nums []int) *NumArray {
	na := &NumArray{append([]int(nil), nums...), NewFenwick(len(nums))}
	for i, x := range nums {
		na.bit.Add(i, int64(x))
	}
	return na
}

func (na *NumArray) Update(i, val int) {
	na.bit.Add(i, int64(val-na.a[i]))
	na.a[i] = val
}

func (na *NumArray) SumRange(l, r int) int64 { return na.bit.RangeSum(l, r) }

// countSmallerAfter: counts[i] = #{j > i : nums[j] < nums[i]}. Compress, BIT of counts. O(n log n).
func countSmallerAfter(nums []int) []int {
	vals := append([]int(nil), nums...)
	sort.Ints(vals)
	u := vals[:0] // dedupe in place
	for _, v := range vals {
		if len(u) == 0 || v != u[len(u)-1] {
			u = append(u, v)
		}
	}
	bit := NewFenwick(len(u))
	out := make([]int, len(nums))
	for i := len(nums) - 1; i >= 0; i-- {
		r := sort.SearchInts(u, nums[i])
		out[i] = int(bit.Prefix(r)) // ranks strictly below r
		bit.Add(r, 1)
	}
	return out
}

// SegTree: iterative segment tree, leaves at t[n..2n-1]. Point set, query [l, r] inclusive.
type SegTree struct {
	n  int
	t  []int64
	op func(a, b int64) int64
	id int64
}

func NewSegTree(a []int64, op func(a, b int64) int64, id int64) *SegTree {
	n := len(a)
	s := &SegTree{n, make([]int64, 2*n), op, id}
	copy(s.t[n:], a)
	for i := n - 1; i > 0; i-- {
		s.t[i] = op(s.t[2*i], s.t[2*i+1])
	}
	return s
}

func (s *SegTree) Set(i int, v int64) {
	i += s.n
	s.t[i] = v
	for i >>= 1; i >= 1; i >>= 1 {
		s.t[i] = s.op(s.t[2*i], s.t[2*i+1])
	}
}

func (s *SegTree) Query(l, r int) int64 {
	resL, resR := s.id, s.id
	for l, r = l+s.n, r+s.n+1; l < r; l, r = l>>1, r>>1 { // half-open [l, r)
		if l&1 == 1 {
			resL = s.op(resL, s.t[l])
			l++
		}
		if r&1 == 1 {
			r--
			resR = s.op(s.t[r], resR)
		}
	}
	return s.op(resL, resR)
}

// MyCalendar: book [start, end) if it overlaps nothing. Go has no TreeMap: a sorted slice,
// binary search for floor/ceiling (O(log n)), insert O(n). Fine for 1000 bookings.
// LazySeg is a lazy segment tree: range add, range sum on [l, r] inclusive. Recursive, 4n slots.
type LazySeg struct {
	n        int
	sum, tag []int64 // tag[x]: "my children still owe +tag[x]"
}

func NewLazySeg(n int) *LazySeg { return &LazySeg{n, make([]int64, 4*n), make([]int64, 4*n)} }

func (s *LazySeg) apply(x, lo, hi int, v int64) { s.sum[x] += v * int64(hi-lo+1); s.tag[x] += v }

func (s *LazySeg) push(x, lo, hi int) { // hand the debt to the children
	if s.tag[x] == 0 {
		return
	}
	mid := (lo + hi) / 2
	s.apply(2*x, lo, mid, s.tag[x])
	s.apply(2*x+1, mid+1, hi, s.tag[x])
	s.tag[x] = 0
}

func (s *LazySeg) Add(l, r int, v int64) { s.add(l, r, v, 1, 0, s.n-1) }

func (s *LazySeg) add(l, r int, v int64, x, lo, hi int) {
	if r < lo || hi < l { // outside
		return
	}
	if l <= lo && hi <= r { // inside: tag and stop
		s.apply(x, lo, hi, v)
		return
	}
	s.push(x, lo, hi) // partial: split
	mid := (lo + hi) / 2
	s.add(l, r, v, 2*x, lo, mid)
	s.add(l, r, v, 2*x+1, mid+1, hi)
	s.sum[x] = s.sum[2*x] + s.sum[2*x+1]
}

func (s *LazySeg) Query(l, r int) int64 { return s.query(l, r, 1, 0, s.n-1) }

func (s *LazySeg) query(l, r, x, lo, hi int) int64 {
	if r < lo || hi < l {
		return 0
	}
	if l <= lo && hi <= r {
		return s.sum[x]
	}
	s.push(x, lo, hi)
	mid := (lo + hi) / 2
	return s.query(l, r, 2*x, lo, mid) + s.query(l, r, 2*x+1, mid+1, hi)
}

type MyCalendar struct{ starts, ends []int }

func (c *MyCalendar) Book(start, end int) bool {
	i := sort.SearchInts(c.starts, start+1) // first start > start (ceiling side)
	if i > 0 && c.ends[i-1] > start {       // floor event ends after we start
		return false
	}
	if i < len(c.starts) && c.starts[i] < end { // ceiling event starts before we end
		return false
	}
	c.starts = append(c.starts, 0)
	copy(c.starts[i+1:], c.starts[i:])
	c.starts[i] = start
	c.ends = append(c.ends, 0)
	copy(c.ends[i+1:], c.ends[i:])
	c.ends[i] = end
	return true
}

// MyCalendarThree: max overlap after each booking. Sparse difference array, sorted keys. O(n) per call.
type MyCalendarThree struct {
	keys  []int
	delta map[int]int
}

func (c *MyCalendarThree) Book(start, end int) int {
	if c.delta == nil {
		c.delta = map[int]int{}
	}
	for _, e := range [][2]int{{start, 1}, {end, -1}} {
		if _, ok := c.delta[e[0]]; !ok {
			i := sort.SearchInts(c.keys, e[0])
			c.keys = append(c.keys, 0)
			copy(c.keys[i+1:], c.keys[i:])
			c.keys[i] = e[0]
		}
		c.delta[e[0]] += e[1]
	}
	run, best := 0, 0
	for _, k := range c.keys {
		run += c.delta[k]
		if run > best {
			best = run
		}
	}
	return best
}

func check(c bool, m string) {
	if !c {
		panic(m)
	}
}

func main() {
	f := NewFenwick(6)
	for i, x := range []int64{5, 2, 9, -3, 5, 20} {
		f.Add(i, x)
	}
	check(f.Prefix(3) == 16 && f.RangeSum(1, 4) == 13 && f.RangeSum(5, 5) == 20, "bit1")
	f.Add(3, 10)
	check(f.RangeSum(1, 4) == 23 && f.Prefix(6) == 48, "bit2")

	na := NewNumArray([]int{1, 3, 5})
	check(na.SumRange(0, 2) == 9, "na1")
	na.Update(1, 2)
	check(na.SumRange(0, 2) == 8 && na.SumRange(1, 1) == 2, "na2")

	check(reflect.DeepEqual(countSmallerAfter([]int{5, 2, 6, 1}), []int{2, 1, 1, 0}), "cs1")
	check(reflect.DeepEqual(countSmallerAfter([]int{-1, -1}), []int{0, 0}), "cs2")
	check(reflect.DeepEqual(countSmallerAfter([]int{2, 0, 1}), []int{2, 0, 0}), "cs3")

	add := func(a, b int64) int64 { return a + b }
	mn := func(a, b int64) int64 {
		if a < b {
			return a
		}
		return b
	}
	st := NewSegTree([]int64{5, 2, 9, -3, 5, 20}, add, 0)
	check(st.Query(1, 4) == 13 && st.Query(0, 5) == 38 && st.Query(2, 2) == 9, "st1")
	st.Set(3, 7)
	check(st.Query(1, 4) == 23, "st2")
	m := NewSegTree([]int64{5, 2, 9, -3, 5, 20}, mn, math.MaxInt64)
	check(m.Query(0, 2) == 2 && m.Query(2, 5) == -3 && m.Query(4, 5) == 5, "mn1")
	m.Set(3, 100)
	check(m.Query(2, 5) == 5, "mn2")
	odd := NewSegTree([]int64{4, 1, 3}, mn, math.MaxInt64)
	check(odd.Query(0, 2) == 1 && odd.Query(2, 2) == 3 && odd.Query(0, 0) == 4, "odd")

	lz := NewLazySeg(6)
	lz.Add(0, 5, 2)
	lz.Add(1, 3, 5)
	check(lz.Query(0, 5) == 27 && lz.Query(2, 2) == 7 && lz.Query(4, 5) == 4, "lazy1")
	rng := rand.New(rand.NewSource(11))
	for it := 0; it < 3000; it++ { // stress: random range add / range sum vs a plain slice

				for i := l; i <= r; i++ {
					ref[i] += v
				}
			} else {
				var want int64
				for i := l; i <= r; i++ {
					want += ref[i]
				}
				check(z.Query(l, r) == want, "lazy stress")
			}
		}
	}

	cal := &MyCalendar{}

// Fenwick tree, iterative segment tree, ordered-map floor/ceiling (My Calendar).
// Build: g++ -std=c++14 -O2 range_query.cpp -o rq && ./rq   (asserts; prints "ok")
#include <algorithm>
#include <cassert>
#include <climits>
#include <cstdio>

#include <map>
#include <random>
#include <vector>
using namespace std;
typedef long long ll;

// Point add, prefix sum. 0-based API, 1-based inside. O(log n) each.
struct Fenwick {
    int n; vector<ll> t;
    explicit Fenwick(int n) : n(n), t(n + 1, 0) {}
    void add(int i, ll delta) { for (++i; i <= n; i += i & -i) t[i] += delta; }   // climb
    ll prefix(int i) const { ll s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }  // a[0..i-1]
    ll rangeSum(int l, int r) const { return prefix(r + 1) - prefix(l); }
};

// Range Sum Query - Mutable: update = add the difference.
struct NumArray {
    vector<int> a; Fenwick bit;
    explicit NumArray(const vector<int>& nums) : a(nums), bit(nums.size()) {
        for (int i = 0; i < (int)nums.size(); i++) bit.add(i, nums[i]);
    }
    void update(int i, int val) { bit.add(i, val - a[i]); a[i] = val; }
    ll sumRange(int l, int r) const { return bit.rangeSum(l, r); }
};

// counts[i] = how many nums[j] < nums[i] with j > i. Compress, BIT of counts. O(n log n).
vector<int> countSmallerAfter(const vector<int>& nums) {
    vector<int> vals(nums);
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());
    Fenwick bit(vals.size());
    vector<int> out(nums.size());
    for (int i = (int)nums.size() - 1; i >= 0; i--) {
        int r = lower_bound(vals.begin(), vals.end(), nums[i]) - vals.begin();
        out[i] = (int)bit.prefix(r);              // ranks strictly below r
        bit.add(r, 1);
    }
    return out;
}

// Iterative segment tree. Leaves at t[n..2n-1]; t[i] = op(t[2i], t[2i+1]).
// Point set, query [l, r] inclusive. O(log n). op associative (sum, min, max, gcd).
struct SegTree {
    int n; vector<ll> t; function<ll(ll, ll)> op; ll id;
    SegTree(const vector<ll>& a, function<ll(ll, ll)> op, ll id) : n(a.size()), t(2 * a.size(), id), op(op), id(id) {
        for (int i = 0; i < n; i++) t[n + i] = a[i];
        for (int i = n - 1; i > 0; i--) t[i] = op(t[2 * i], t[2 * i + 1]);
    }
    void set(int i, ll v) {
        for (t[i += n] = v; i > 1; i >>= 1) t[i >> 1] = op(t[i & ~1], t[i | 1]);
    }
    ll query(int l, int r) const {
        ll resL = id, resR = id;
        for (l += n, r += n + 1; l < r; l >>= 1, r >>= 1) {   // half-open [l, r)
            if (l & 1) resL = op(resL, t[l++]);
            if (r & 1) resR = op(t[--r], resR);
        }
        return op(resL, resR);
    }
};

// Lazy segment tree: range add, range sum on [l, r] inclusive. Recursive, 4n slots. O(log n) each.
struct LazySeg {
    int n; vector<ll> sum, tag;                   // tag[x]: "my children still owe +tag[x]"
    LazySeg(int n) : n(n), sum(4 * n), tag(4 * n) {}
    void apply(int x, int lo, int hi, ll v) { sum[x] += v * (hi - lo + 1); tag[x] += v; }
    void push(int x, int lo, int hi) {            // hand the debt to the children
        if (!tag[x]) return;
        int mid = (lo + hi) / 2;
        apply(2 * x, lo, mid, tag[x]); apply(2 * x + 1, mid + 1, hi, tag[x]);
        tag[x] = 0;
    }
    void add(int l, int r, ll v) { add(l, r, v, 1, 0, n - 1); }
    ll query(int l, int r) { return query(l, r, 1, 0, n - 1); }
    void add(int l, int r, ll v, int x, int lo, int hi) {
        if (r < lo || hi < l) return;                              // outside
        if (l <= lo && hi <= r) { apply(x, lo, hi, v); return; }   // inside: tag and stop
        push(x, lo, hi);                                           // partial: split
        int mid = (lo + hi) / 2;
        add(l, r, v, 2 * x, lo, mid); add(l, r, v, 2 * x + 1, mid + 1, hi);
        sum[x] = sum[2 * x] + sum[2 * x + 1];
    }
    ll query(int l, int r, int x, int lo, int hi) {
        if (r < lo || hi < l) return 0;
        if (l <= lo && hi <= r) return sum[x];
        push(x, lo, hi);
        int mid = (lo + hi) / 2;
        return query(l, r, 2 * x, lo, mid) + query(l, r, 2 * x + 1, mid + 1, hi);
    }
};

// My Calendar I: book [start, end) if it overlaps nothing. std::map = red-black tree.
struct MyCalendar {
    map<int, int> ev;                             // start -> end
    bool book(int start, int end) {
        auto next = ev.lower_bound(start);        // ceiling: first start >= start
        if (next != ev.end() && next->first < end) return false;
        if (next != ev.begin() && prev(next)->second > start) return false;   // floor
        ev[start] = end;
        return true;
    }
};

// My Calendar III: max overlap after each booking. Sparse difference array in a map. O(n) per call.
struct MyCalendarThree {
    map<int, int> delta;
    int book(int start, int end) {
        delta[start]++; delta[end]--;
        int run = 0, best = 0;
        for (auto& kv : delta) { run += kv.second; best = max(best, run); }   // keys in order
        return best;
    }
};

// Range query templates (SDE Prep Dojo, lesson coding-range-query). Java 21.
// Fenwick tree, iterative segment tree, TreeMap floor/ceiling (My Calendar).
// Run: java RangeQuery.java   (checks; prints "ok")
import java.util.*;
import java.util.function.LongBinaryOperator;

public class RangeQuery {
    // Point add, prefix sum. 0-based API, 1-based inside. O(log n) each.
    static class Fenwick {
        final int n; final long[] t;
        Fenwick(int n) { this.n = n; t = new long[n + 1]; }
        void add(int i, long delta) { for (i++; i <= n; i += i & -i) t[i] += delta; }        // climb
        long prefix(int i) { long s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }  // a[0..i-1]
        long rangeSum(int l, int r) { return prefix(r + 1) - prefix(l); }
    }

    // Range Sum Query - Mutable: update = add the difference.
    static class NumArray {
        final int[] a; final Fenwick bit;
        NumArray(int[] nums) {
            a = nums.clone(); bit = new Fenwick(nums.length);
            for (int i = 0; i < nums.length; i++) bit.add(i, nums[i]);
        }
        void update(int i, int val) { bit.add(i, val - a[i]); a[i] = val; }
        long sumRange(int l, int r) { return bit.rangeSum(l, r); }
    }

    // counts[i] = how many nums[j] < nums[i] with j > i. Compress, BIT of counts. O(n log n).
    static List<Integer> countSmallerAfter(int[] nums) {
        int[] vals = Arrays.stream(nums).distinct().sorted().toArray();
        Fenwick bit = new Fenwick(vals.length);
        Integer[] out = new Integer[nums.length];
        for (int i = nums.length - 1; i >= 0; i--) {
            int r = Arrays.binarySearch(vals, nums[i]);
            out[i] = (int) bit.prefix(r);              // ranks strictly below r
            bit.add(r, 1);
        }
        return Arrays.asList(out);
    }

    // Iterative segment tree. Leaves at t[n..2n-1]. Point set, query [l, r] inclusive. O(log n).
    static class SegTree {
        final int n; final long[] t; final LongBinaryOperator op; final long id;
        SegTree(long[] a, LongBinaryOperator op, long id) {
            n = a.length; t = new long[2 * n]; this.op = op; this.id = id;
            Arrays.fill(t, id);
            System.arraycopy(a, 0, t, n, n);
            for (int i = n - 1; i > 0; i--) t[i] = op.applyAsLong(t[2 * i], t[2 * i + 1]);
        }
        void set(int i, long v) {
            i += n; t[i] = v;
            for (i >>= 1; i >= 1; i >>= 1) t[i] = op.applyAsLong(t[2 * i], t[2 * i + 1]);
        }
        long query(int l, int r) {
            long resL = id, resR = id;
            for (l += n, r += n + 1; l < r; l >>= 1, r >>= 1) {          // half-open [l, r)
                if ((l & 1) == 1) resL = op.applyAsLong(resL, t[l++]);
                if ((r & 1) == 1) resR = op.applyAsLong(t[--r], resR);
            }
            return op.applyAsLong(resL, resR);
        }
    }

    // Lazy segment tree: range add, range sum on [l, r] inclusive. Recursive, 4n slots. O(log n) each.
    static class LazySeg {
        final int n; final long[] sum, tag;            // tag[x]: "my children still owe +tag[x]"
        LazySeg(int n) { this.n = n; sum = new long[4 * n]; tag = new long[4 * n]; }
        void apply(int x, int lo, int hi, long v) { sum[x] += v * (hi - lo + 1); tag[x] += v; }
        void push(int x, int lo, int hi) {             // hand the debt to the children
            if (tag[x] == 0) return;
            int mid = (lo + hi) / 2;
            apply(2 * x, lo, mid, tag[x]); apply(2 * x + 1, mid + 1, hi, tag[x]);
            tag[x] = 0;
        }
        void add(int l, int r, long v) { add(l, r, v, 1, 0, n - 1); }
        long query(int l, int r) { return query(l, r, 1, 0, n - 1); }
        void add(int l, int r, long v, int x, int lo, int hi) {
            if (r < lo || hi < l) return;                              // outside
            if (l <= lo && hi <= r) { apply(x, lo, hi, v); return; }   // inside: tag and stop
            push(x, lo, hi);                                           // partial: split
            int mid = (lo + hi) / 2;
            add(l, r, v, 2 * x, lo, mid); add(l, r, v, 2 * x + 1, mid + 1, hi);
            sum[x] = sum[2 * x] + sum[2 * x + 1];
        }
        long query(int l, int r, int x, int lo, int hi) {
            if (r < lo || hi < l) return 0;
            if (l <= lo && hi <= r) return sum[x];
            push(x, lo, hi);
            int mid = (lo + hi) / 2;
            return query(l, r, 2 * x, lo, mid) + query(l, r, 2 * x + 1, mid + 1, hi);
        }
    }

    // My Calendar I: book [start, end) if it overlaps nothing. TreeMap = red-black tree.
    static class MyCalendar {
        final TreeMap<Integer, Integer> ev = new TreeMap<>();   // start -> end
        boolean book(int start, int end) {
            Map.Entry<Integer, Integer> lo = ev.floorEntry(start);    // last start <= start
            if (lo != null && lo.getValue() > start) return false;
            Integer hi = ev.ceilingKey(start);                        // first start >= start
            if (hi != null && hi < end) return false;
            ev.put(start, end);
            return true;
        }
    }

    // My Calendar III: max overlap after each booking. Sparse difference array. O(n) per call.
    static class MyCalendarThree {
        final TreeMap<Integer, Integer> delta = new TreeMap<>();
        int book(int start, int end) {
            delta.merge(start, 1, Integer::sum);
            delta.merge(end, -1, Integer::sum);
            int run = 0, best = 0;
            for (int d : delta.values()) { run += d; best = Math.max(best, run); }   // key order
            return best;
        }
    }

    static void check(boolean c, String m) { if (!c) throw new AssertionError(m); }

    public static void main(String[] args) {
        Fenwick f = new Fenwick(6);
        int[] init = {5, 2, 9, -3, 5, 20};
        for (int i = 0; i < 6; i++) f.add(i, init[i]);
        check(f.prefix(3) == 16 && f.rangeSum(1, 4) == 13 && f.rangeSum(5, 5) == 20, "bit1");
        f.add(3, 10);
        check(f.rangeSum(1, 4) == 23 && f.prefix(6) == 48, "bit2");

        na.update(1, 2);
        check(na.sumRange(0, 2) == 8 && na.sumRange(1, 1) == 2, "na2");

        check(countSmallerAfter(new int[]{5, 2, 6, 1}).equals(List.of(2, 1, 1, 0)), "cs1");
        check(countSmallerAfter(new int[]{-1, -1}).equals(List.of(0, 0)), "cs2");
        check(countSmallerAfter(new int[]{2, 0, 1}).equals(List.of(2, 0, 0)), "cs3");

        SegTree st = new SegTree(new long[]{5, 2, 9, -3, 5, 20}, Long::sum, 0);
        check(st.query(1, 4) == 13 && st.query(0, 5) == 38 && st.query(2, 2) == 9, "st1");
        st.set(3, 7);
        check(st.query(1, 4) == 23, "st2");
        SegTree m = new SegTree(new long[]{5, 2, 9, -3, 5, 20}, Math::min, Long.MAX_VALUE);
        check(m.query(0, 2) == 2 && m.query(2, 5) == -3 && m.query(4, 5) == 5, "mn1");
        m.set(3, 100);

"""Range query templates (SDE Prep Dojo, lesson coding-range-query).

Fenwick tree, iterative segment tree, ordered-map floor/ceiling (My Calendar).
Run: python range_query.py   (asserts; prints "ok")
"""
import bisect


class Fenwick:
    """Point add, prefix sum. 0-based API, 1-based inside. O(log n) each."""

    def __init__(self, n):
        self.n = n
        self.t = [0] * (n + 1)

    def add(self, i, delta):            # a[i] += delta
        i += 1
        while i <= self.n:
            self.t[i] += delta
            i += i & -i                 # climb to the next node that covers i

    def prefix(self, i):                # a[0] + ... + a[i-1]
        s = 0
        while i > 0:
            s += self.t[i]
            i -= i & -i                 # drop the lowest set bit
        return s

    def range_sum(self, l, r):          # a[l..r], both included
        return self.prefix(r + 1) - self.prefix(l)


class NumArray:
    """Range Sum Query - Mutable on top of Fenwick: update = add the difference."""

    def __init__(self, nums):
        self.a = list(nums)
        self.bit = Fenwick(len(nums))
        for i, x in enumerate(nums):
            self.bit.add(i, x)

    def update(self, i, val):
        self.bit.add(i, val - self.a[i])
        self.a[i] = val

    def sum_range(self, l, r):
        return self.bit.range_sum(l, r)


def count_smaller_after(nums):
    """counts[i] = how many nums[j] < nums[i] with j > i. Compress values, BIT of counts. O(n log n)."""
    rank = {v: i for i, v in enumerate(sorted(set(nums)))}
    bit = Fenwick(len(rank))
    out = [0] * len(nums)
    for i in range(len(nums) - 1, -1, -1):  # walk from the right: the BIT holds "after i"
        r = rank[nums[i]]
        out[i] = bit.prefix(r)              # ranks strictly below r
        bit.add(r, 1)
    return out


class SegTree:
    """Iterative (bottom-up) segment tree. Leaves at t[n..2n-1]; t[i] = op(t[2i], t[2i+1]).
    Point set, range query on [l, r] inclusive. O(log n) each. op must be associative."""

    def __init__(self, nums, op=lambda a, b: a + b, identity=0):
        self.n, self.op, self.id = len(nums), op, identity
        self.t = [identity] * (2 * self.n)
        self.t[self.n:] = nums
        for i in range(self.n - 1, 0, -1):
            self.t[i] = op(self.t[2 * i], self.t[2 * i + 1])

    def set(self, i, val):
        i += self.n
        self.t[i] = val
        while i > 1:
            i //= 2
            self.t[i] = self.op(self.t[2 * i], self.t[2 * i + 1])

    def query(self, l, r):
        res_l, res_r = self.id, self.id
        l += self.n
        r += self.n + 1                 # half-open [l, r)
        while l < r:
            if l & 1:                   # l is a right child: take it, step right
                res_l = self.op(res_l, self.t[l]); l += 1
            if r & 1:                   # r - 1 is a left child's sibling: take it
                r -= 1; res_r = self.op(self.t[r], res_r)
            l //= 2; r //= 2
        return self.op(res_l, res_r)


class LazySeg:                       # range add, range sum (recursive, 4n slots)
    def __init__(self, n):
        self.n, self.sum, self.tag = n, [0] * (4 * n), [0] * (4 * n)
    def _apply(self, x, lo, hi, v):   # add v to every cell of node x
        self.sum[x] += v * (hi - lo + 1)
        self.tag[x] += v              # "my children still owe +v"
    def _push(self, x, lo, hi):       # hand the debt to the children
        if self.tag[x]:
            mid = (lo + hi) // 2
            self._apply(2 * x, lo, mid, self.tag[x])
            self._apply(2 * x + 1, mid + 1, hi, self.tag[x])
            self.tag[x] = 0
    def add(self, l, r, v, x=1, lo=0, hi=None):
        hi = self.n - 1 if hi is None else hi
        if r < lo or hi < l: return                 # outside
        if l <= lo and hi <= r:                     # inside: tag and stop
            self._apply(x, lo, hi, v); return
        self._push(x, lo, hi)                       # partial: split
        mid = (lo + hi) // 2
        self.add(l, r, v, 2 * x, lo, mid)
        self.add(l, r, v, 2 * x + 1, mid + 1, hi)
        self.sum[x] = self.sum[2 * x] + self.sum[2 * x + 1]
    def query(self, l, r, x=1, lo=0, hi=None):
        hi = self.n - 1 if hi is None else hi
        if r < lo or hi < l: return 0
        if l <= lo and hi <= r: return self.sum[x]
        self._push(x, lo, hi)
        mid = (lo + hi) // 2
        return self.query(l, r, 2 * x, lo, mid) + self.query(l, r, 2 * x + 1, mid + 1, hi)


class MyCalendar:
    """My Calendar I: book [start, end) if it overlaps nothing. Sorted starts + floor/ceiling.
    Python has no TreeMap: bisect gives O(log n) search, insort is O(n) (fine for 1000 calls;
    sortedcontainers.SortedList, available on LeetCode, makes it O(log n))."""

    def __init__(self):
        self.starts, self.ends = [], []

    def book(self, start, end):
        i = bisect.bisect_right(self.starts, start)      # ceiling: first start > start is at i
        if i > 0 and self.ends[i - 1] > start:          # floor event ends after we start
            return False
        if i < len(self.starts) and self.starts[i] < end:  # ceiling event starts before we end
            return False
        self.starts.insert(i, start)
        self.ends.insert(i, end)
        return True


class MyCalendarThree:
    """My Calendar III: max overlap after each booking. Sweep over a sorted map of +1/-1. O(n) per call."""

    def __init__(self):
        self.keys, self.delta = [], {}

    def book(self, start, end):
        for x, d in ((start, 1), (end, -1)):
            if x not in self.delta:
                bisect.insort(self.keys, x)
                self.delta[x] = 0
            self.delta[x] += d
        best = run = 0
        for x in self.keys:             # in key order: the difference array, sparse
            run += self.delta[x]
            best = max(best, run)
        return best
