All four use the 26-count key and return the groups in first-seen order (any order is accepted).

class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, int> groupOf;      // count signature -> index into groups
        vector<vector<string>> groups;
        for (const string& w : strs) {
            string key(26, '\0');                // key[c] = how many times letter c occurs
            for (char c : w) key[c - 'a']++;     // length <= 100, so a count fits in one char
            auto it = groupOf.find(key);
            if (it == groupOf.end()) {           // first word with this signature: new group
                groupOf.emplace(key, (int)groups.size());
                groups.push_back({w});
            } else {
                groups[it->second].push_back(w);
            }
        }
        return groups;                           // groups in order of first appearance
    }
};
func groupAnagrams(strs []string) [][]string {
	groupOf := make(map[[26]byte]int) // count signature -> index into groups (arrays are valid map keys)
	var groups [][]string
	for _, w := range strs {
		var key [26]byte // key[c] = occurrences of letter c; length <= 100 fits in a byte
		for i := 0; i < len(w); i++ {
			key[w[i]-'a']++
		}
		if idx, ok := groupOf[key]; ok {
			groups[idx] = append(groups[idx], w)
		} else { // first word with this signature: new group
			groupOf[key] = len(groups)
			groups = append(groups, []string{w})
		}
	}
	return groups // groups in order of first appearance
}

static class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        Map<String, List<String>> groups = new HashMap<>();   // count signature -> its words
        List<List<String>> result = new ArrayList<>();         // groups in order of first appearance
        for (String w : strs) {
            char[] count = new char[26];                       // count[c] = occurrences of letter c
            for (int i = 0; i < w.length(); i++) count[w.charAt(i) - 'a']++;
            String key = new String(count);                    // 26 chars, equal iff same counts
            List<String> g = groups.get(key);
            if (g == null) {                                   // first word with this signature
                g = new ArrayList<>();
                groups.put(key, g);
                result.add(g);
            }
            g.add(w);
        }
        return result;
    }
}
class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        groups = defaultdict(list)            # count signature -> words (dict keeps first-seen order)
        for w in strs:
            count = [0] * 26                  # count[c] = occurrences of letter c
            for ch in w:
                count[ord(ch) - ord("a")] += 1
            groups[tuple(count)].append(w)    # a list is not hashable; a tuple is
        return list(groups.values())
