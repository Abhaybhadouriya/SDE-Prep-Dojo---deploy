Centre expansion in four languages. Each file in coding-code/palindromic-substrings/ runs the examples and edge cases (one letter, "racecar", 1000 equal letters = 500,500) and, in C++, Java and Python, cross-checks 2000 random strings against the O(n³) brute force.

func countSubstrings(s string) int {
	n, total := len(s), 0
	// 2n - 1 centres: even c = a letter (odd-length palindromes),
	// odd c = the gap between two letters (even-length palindromes).
	for c := 0; c < 2*n-1; c++ {
		lo, hi := c/2, c/2+c%2
		// Each successful step outward is one more palindrome with this centre.
		for lo >= 0 && hi < n && s[lo] == s[hi] {
			total++
			lo--
			hi++
		}
	}
	return total
}
class Solution {
public:
    int countSubstrings(string s) {
        int n = s.size(), total = 0;
        // 2n - 1 centres: even c = a letter (odd-length palindromes),
        // odd c = the gap between two letters (even-length palindromes).
        for (int c = 0; c < 2 * n - 1; ++c) {
            int lo = c / 2, hi = lo + c % 2;
            // Each successful step outward is one more palindrome with this centre.
            while (lo >= 0 && hi < n && s[lo] == s[hi]) {
                ++total;
                --lo;
                ++hi;
            }
        }
        return total;   // at most n(n+1)/2 = 500500 for n = 1000
    }
};

class Solution:
    def countSubstrings(self, s: str) -> int:
        n, total = len(s), 0
        # 2n - 1 centres: even c = a letter (odd-length palindromes),
        # odd c = the gap between two letters (even-length palindromes).
        for c in range(2 * n - 1):
            lo, hi = c // 2, c // 2 + c % 2
            # Each successful step outward is one more palindrome with this centre.
            while lo >= 0 and hi < n and s[lo] == s[hi]:
                total += 1
                lo -= 1
                hi += 1
        return total
