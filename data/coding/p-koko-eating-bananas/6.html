All four use the same lo < hi template and integer ceiling. C++ and Java add the hours in 64 bits.

class Solution {
public:
    int minEatingSpeed(const vector<int>& piles, int h) {
        int lo = 1, hi = *max_element(piles.begin(), piles.end());  // speed max always works
        while (lo < hi) {                             // invariant: the answer is in [lo, hi]
            int mid = lo + (hi - lo) / 2;
            if (hoursAt(piles, mid) <= h) hi = mid;   // mid is fast enough: answer <= mid
            else lo = mid + 1;                        // too slow: answer > mid
        }
        return lo;
    }

private:
    static long long hoursAt(const vector<int>& piles, int k) {
        long long hours = 0;                          // up to 1e4 * 1e9: needs 64 bits
        for (int p : piles) hours += (p + (long long)k - 1) / k;   // ceil(p / k) in integers
        return hours;
    }
};

func hoursAt(piles []int, k int) int {
	hours := 0 // int is 64-bit on the judge; the sum can reach 1e13
	for _, p := range piles {
		hours += (p + k - 1) / k // ceil(p / k)
	}
	return hours
}
static class Solution {
    public int minEatingSpeed(int[] piles, int h) {
        int lo = 1, hi = 0;
        for (int p : piles) hi = Math.max(hi, p);    // speed max(piles) always works
        while (lo < hi) {                            // invariant: the answer is in [lo, hi]
            int mid = lo + (hi - lo) / 2;
            if (hoursAt(piles, mid) <= h) hi = mid;  // fast enough: answer <= mid
            else lo = mid + 1;                       // too slow: answer > mid
        }
        return lo;
    }

    private static long hoursAt(int[] piles, int k) {
        long hours = 0;                              // up to 1e4 * 1e9: needs a long
        for (int p : piles) hours += (p + (long) k - 1) / k;   // ceil(p / k)
        return hours;
    }
}
class Solution:
    def minEatingSpeed(self, piles: List[int], h: int) -> int:
        def hours_at(k: int) -> int:
            return sum((p + k - 1) // k for p in piles)   # ceil(p / k) without floats

        lo, hi = 1, max(piles)          # speed max(piles) always works: one pile per hour
        while lo < hi:                  # invariant: the answer is in [lo, hi]
            mid = (lo + hi) // 2
            if hours_at(mid) <= h:
                hi = mid                # fast enough: answer is mid or smaller
            else:
                lo = mid + 1            # too slow: answer is bigger than mid
        return lo
