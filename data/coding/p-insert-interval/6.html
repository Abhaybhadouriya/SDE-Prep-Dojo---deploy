func insert(intervals [][]int, newInterval []int) [][]int {
	out := make([][]int, 0, len(intervals)+1)
	i, n := 0, len(intervals)
	lo, hi := newInterval[0], newInterval[1]
	for i < n && intervals[i][1] < lo { // phase 1: ends before the new one starts
		out = append(out, intervals[i])
		i++
	}
	for i < n && intervals[i][0] <= hi { // phase 2: touches or overlaps -> absorb
		lo = min(lo, intervals[i][0])
		hi = max(hi, intervals[i][1])
		i++
	}
	out = append(out, []int{lo, hi})
	out = append(out, intervals[i:]...) // phase 3: after the merged block
	return out
}

            lo = Math.min(lo, intervals[i][0]);

        out = []

            out.append(intervals[i])
            i += 1

            lo = min(lo, intervals[i][0])
            hi = max(hi, intervals[i][1])
            i += 1
        out.append([lo, hi])

            out.append(intervals[i])
            i += 1
        return out
