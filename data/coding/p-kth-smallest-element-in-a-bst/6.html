All four use the iterative in-order walk with an explicit stack and return as soon as the k-th node is popped. TreeNode is the usual judge definition (val, left, right).

// Iterative in-order: a BST's in-order is sorted, so the k-th node popped is the answer.
func kthSmallest(root *TreeNode, k int) int {
	stack := []*TreeNode{} // the path of "left turns" still to visit
	cur := root
	for cur != nil || len(stack) > 0 {
		for cur != nil { // go as far left as possible
			stack = append(stack, cur)
			cur = cur.Left
		}
		cur = stack[len(stack)-1] // the next smallest value
		stack = stack[:len(stack)-1]
		k--
		if k == 0 {
			return cur.Val // stop early
		}
		cur = cur.Right // then its right subtree
	}
	return -1 // unreachable when 1 <= k <= n
}
class Solution {
public:
    // Iterative in-order: a BST's in-order is sorted, so the k-th node popped is the answer.
    int kthSmallest(TreeNode* root, int k) {
        stack<TreeNode*> st;               // the path of "left turns" still to visit
        TreeNode* cur = root;
        while (cur || !st.empty()) {
            while (cur) { st.push(cur); cur = cur->left; }   // go as far left as possible
            cur = st.top(); st.pop();      // the next smallest value
            if (--k == 0) return cur->val; // stop early: no need to walk the rest
            cur = cur->right;              // then everything between it and its parent
        }
        return -1;                         // unreachable when 1 <= k <= n
    }
};

class Solution:
    def kthSmallest(self, root: Optional[TreeNode], k: int) -> int:
        # Iterative in-order: a BST's in-order is sorted, so the k-th node popped is the answer.
        stack: List[TreeNode] = []      # the path of "left turns" still to visit
        cur = root
        while cur or stack:
            while cur:                  # go as far left as possible
                stack.append(cur)
                cur = cur.left
            cur = stack.pop()           # the next smallest value
            k -= 1
            if k == 0:
                return cur.val          # stop early
            cur = cur.right             # then its right subtree
        return -1                       # unreachable when 1 <= k <= n
