The bottom-up DP with the length cap in four languages. The test drivers (the examples, one-letter inputs, the 300-letter "aaa...ab" trap, and 500 random cases checked against the plain recursion) are in coding-code/word-break/.

class Solution {
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string> words(wordDict.begin(), wordDict.end());
        int longest = 0;  // no piece can be longer than the longest word
        for (const string& w : wordDict) longest = max(longest, (int)w.size());
        int n = s.size();
        vector<bool> ok(n + 1, false);  // ok[i]: prefix s[0..i) splits into words
        ok[0] = true;                   // the empty prefix needs no words
        for (int i = 1; i <= n; i++) {
            // try every possible last word s[j..i), shortest first
            for (int j = i - 1; j >= max(0, i - longest); j--) {
                if (ok[j] && words.count(s.substr(j, i - j))) {
                    ok[i] = true;
                    break;
                }
            }
        }
        return ok[n];
    }
};
func wordBreak(s string, wordDict []string) bool {
	words := make(map[string]bool, len(wordDict))
	longest := 0 // no piece can be longer than the longest word
	for _, w := range wordDict {
		words[w] = true
		if len(w) > longest {
			longest = len(w)
		}
	}
	n := len(s)
	ok := make([]bool, n+1) // ok[i]: prefix s[0:i] splits into words
	ok[0] = true            // the empty prefix needs no words
	for i := 1; i <= n; i++ {
		lo := i - longest
		if lo < 0 {
			lo = 0
		}
		// try every possible last word s[j:i], shortest first
		for j := i - 1; j >= lo; j-- {
			if ok[j] && words[s[j:i]] {
				ok[i] = true
				break
			}
		}
	}
	return ok[n]
}

static class Solution {
    public boolean wordBreak(String s, List<String> wordDict) {
        Set<String> words = new HashSet<>(wordDict);
        int longest = 0;  // no piece can be longer than the longest word
        for (String w : wordDict) longest = Math.max(longest, w.length());
        int n = s.length();
        boolean[] ok = new boolean[n + 1];  // ok[i]: prefix s[0..i) splits into words
        ok[0] = true;                       // the empty prefix needs no words
        for (int i = 1; i <= n; i++) {
            // try every possible last word s[j..i), shortest first
            for (int j = i - 1; j >= Math.max(0, i - longest); j--) {
                if (ok[j] && words.contains(s.substring(j, i))) {
                    ok[i] = true;
                    break;
                }
            }
        }
        return ok[n];
    }
}
class Solution:
    def wordBreak(self, s: str, wordDict: List[str]) -> bool:
        words = set(wordDict)
        longest = max(len(w) for w in words)  # no piece can be longer than this
        n = len(s)
        ok = [False] * (n + 1)  # ok[i]: the prefix s[:i] splits into dictionary words
        ok[0] = True            # the empty prefix needs no words
        for i in range(1, n + 1):
            # try every possible last word s[j:i], shortest first, never longer than `longest`
            for j in range(i - 1, max(i - longest, 0) - 1, -1):
                if ok[j] and s[j:i] in words:
                    ok[i] = True
                    break
        return ok[n]
