All four use the left-to-right "compare with next" rule.

var romanValue = map[byte]int{'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000}

func romanToInt(s string) int {
	total := 0
	for i := 0; i < len(s); i++ {
		v := romanValue[s[i]]
		// A symbol smaller than its right neighbour is a subtractive prefix (IV, XC, CM).
		if i+1 < len(s) && v < romanValue[s[i+1]] {
			total -= v
		} else {
			total += v
		}
	}
	return total
}
class Solution {
    static int value(char c) {
        switch (c) {
            case 'I': return 1;    case 'V': return 5;    case 'X': return 10;
            case 'L': return 50;   case 'C': return 100;  case 'D': return 500;
            default:  return 1000; // M (input is guaranteed valid)
        }
    }
public:
    int romanToInt(const string& s) {
        int total = 0;
        for (size_t i = 0; i < s.size(); ++i) {
            int v = value(s[i]);

            else total += v;
        }
        return total;
    }
};

    public int romanToInt(String s) {

    VALUE = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}

    def romanToInt(self, s: str) -> int:
