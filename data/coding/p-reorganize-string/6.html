The even-slot fill in four languages. Each file in coding-code/reorganize-string/ also checks 5,000 random strings: when the bound allows an answer, the output must be a permutation of the input with no equal neighbours; otherwise it must be "".

class Solution {
public:
    string reorganizeString(string s) {
        int n = s.size();
        int cnt[26] = {0};
        for (char c : s) cnt[c - 'a']++;

        int top = 0;                                   // most frequent letter
        for (int c = 1; c < 26; ++c)
            if (cnt[c] > cnt[top]) top = c;
        if (cnt[top] > (n + 1) / 2) return "";        // not enough gaps to separate it

        string out(n, ' ');
        int pos = 0;                                   // next free even slot
        auto place = [&](int c) {
            while (cnt[c] > 0) {
                if (pos >= n) pos = 1;                 // evens used up: switch to odds
                out[pos] = char('a' + c);
                pos += 2;
                cnt[c]--;
            }
        };
        place(top);                                    // the top letter MUST go first
        for (int c = 0; c < 26; ++c) place(c);
        return out;
    }
};
func reorganizeString(s string) string {
	n := len(s)
	var cnt [26]int
	for i := 0; i < n; i++ {
		cnt[s[i]-'a']++
	}

	top := 0 // most frequent letter
	for c := 1; c < 26; c++ {
		if cnt[c] > cnt[top] {
			top = c
		}
	}
	if cnt[top] > (n+1)/2 {
		return "" // not enough gaps to separate it
	}

	out := make([]byte, n)
	pos := 0 // next free even slot
	place := func(c int) {
		for ; cnt[c] > 0; cnt[c]-- {
			if pos >= n {
				pos = 1 // evens used up: switch to odds
			}
			out[pos] = byte('a' + c)
			pos += 2
		}
	}
	place(top) // the top letter MUST go first
	for c := 0; c < 26; c++ {
		place(c)
	}
	return string(out)
}

static class Solution {
    private int[] cnt;
    private char[] out;
    private int pos;

    public String reorganizeString(String s) {
        int n = s.length();
        cnt = new int[26];
        for (char c : s.toCharArray()) cnt[c - 'a']++;

        int top = 0;                               // most frequent letter
        for (int c = 1; c < 26; c++)
            if (cnt[c] > cnt[top]) top = c;
        if (cnt[top] > (n + 1) / 2) return "";    // not enough gaps to separate it

        out = new char[n];
        pos = 0;                                   // next free even slot
        place(top);                                // the top letter MUST go first
        for (int c = 0; c < 26; c++) place(c);
        return new String(out);
    }

    private void place(int c) {
        while (cnt[c] > 0) {
            if (pos >= out.length) pos = 1;        // evens used up: switch to odds
            out[pos] = (char) ('a' + c);
            pos += 2;
            cnt[c]--;
        }
    }
}
class Solution:
    def reorganizeString(self, s: str) -> str:
        n = len(s)
        cnt = Counter(s)
        top, top_cnt = cnt.most_common(1)[0]
        if top_cnt > (n + 1) // 2:          # not enough gaps to separate it
            return ""

        out = [""] * n
        pos = 0                             # next free even slot
        # the top letter MUST go first; the order of the rest does not matter
        for ch in [top] + [c for c in cnt if c != top]:
            for _ in range(cnt[ch]):
                if pos >= n:
                    pos = 1                 # evens used up: switch to odds
                out[pos] = ch
                pos += 2
        return "".join(out)
