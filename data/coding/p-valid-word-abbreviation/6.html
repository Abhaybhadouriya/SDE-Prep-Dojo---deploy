class Solution {
public:
    bool validWordAbbreviation(const string& word, const string& abbr) {
        int n = word.size(), m = abbr.size();
        int i = 0, j = 0;                          // i: next char of word, j: next char of abbr
        while (i < n && j < m) {
            if (isdigit((unsigned char)abbr[j])) {
                if (abbr[j] == '0') return false;  // "01" or a lone "0": leading zero / empty skip
                long long skip = 0;
                while (j < m && isdigit((unsigned char)abbr[j])) {
                    skip = skip * 10 + (abbr[j++] - '0');
                    if (skip > n) return false;    // already past the word: stop before overflow
                }
                i += (int)skip;
            } else {
                if (word[i] != abbr[j]) return false;
                i++;
                j++;
            }
        }
        return i == n && j == m;                   // both must be used up exactly
    }
};
