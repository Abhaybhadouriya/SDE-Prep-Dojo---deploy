func removeDuplicates(nums []int) int {
	// nums[0..k-1] is the finished output: sorted and duplicate-free.
	// nums[0] is always kept, so start with k = 1 (n >= 1 by the constraints).
	k := 1
	for i := 1; i < len(nums); i++ {
		// Sorted input: a value is new exactly when it differs from the last kept one.
		if nums[i] != nums[k-1] {
			nums[k] = nums[i]
			k++
		}
	}
	return k
}
