func characterReplacement(s string, k int) int {
	var count [26]int // letter counts inside the window s[left..right]
	maxFreq := 0      // highest count ever seen in a window (never lowered)
	left := 0
	for right := 0; right < len(s); right++ {
		count[s[right]-'A']++
		if count[s[right]-'A'] > maxFreq {
			maxFreq = count[s[right]-'A']
		}
		// letters to repaint = window size - most common letter; too many -> slide by one
		if right-left+1-maxFreq > k {
			count[s[left]-'A']--
			left++
		}
	}
	return len(s) - left // the window only grows or slides, so its final size is the best
}
