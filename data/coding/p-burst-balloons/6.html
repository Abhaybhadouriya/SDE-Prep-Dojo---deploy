class Solution {
public:
    int maxCoins(vector<int>& nums) {
        vector<int> vals;                             // virtual 1-balloons at both ends
        vals.push_back(1);
        for (int x : nums) vals.push_back(x);
        vals.push_back(1);
        int m = vals.size();
        // best[l][r] = most coins from bursting every balloon strictly between l and r,
        // while l and r themselves are still standing.
        vector<vector<int>> best(m, vector<int>(m, 0));
        for (int gap = 2; gap < m; gap++)             // r - l = gap; gap 1 has nothing inside
            for (int l = 0; l + gap < m; l++) {
                int r = l + gap, edge = vals[l] * vals[r], top = 0;
                for (int k = l + 1; k < r; k++)       // k = the LAST balloon burst in (l, r)
                    top = max(top, best[l][k] + best[k][r] + vals[k] * edge);
                best[l][r] = top;
            }
        return best[0][m - 1];
    }
};
func maxCoins(nums []int) int {
	vals := append(append([]int{1}, nums...), 1) // virtual 1-balloons at both ends
	m := len(vals)
	// best[l][r] = most coins from bursting every balloon strictly between l and r,
	// while l and r themselves are still standing.
	best := make([][]int, m)
	for i := range best {
		best[i] = make([]int, m)
	}
	for gap := 2; gap < m; gap++ { // r - l = gap; gap 1 has nothing inside
		for l := 0; l+gap < m; l++ {
			r := l + gap
			edge, top := vals[l]*vals[r], 0
			for k := l + 1; k < r; k++ { // k = the LAST balloon burst in (l, r)
				if got := best[l][k] + best[k][r] + vals[k]*edge; got > top {
					top = got
				}
			}
			best[l][r] = top
		}
	}
	return best[0][m-1]
}

class Solution:
    def maxCoins(self, nums: List[int]) -> int:
        vals = [1] + nums + [1]                 # virtual 1-balloons at both ends
        m = len(vals)
        # best[l][r] = most coins from bursting every balloon strictly between l and r,
        # while l and r themselves are still standing.
        best = [[0] * m for _ in range(m)]
        for gap in range(2, m):                 # r - l = gap; gap 1 has nothing inside
            for l in range(0, m - gap):
                r = l + gap
                edge = vals[l] * vals[r]
                top = 0
                for k in range(l + 1, r):       # k = the LAST balloon burst in (l, r)
                    got = best[l][k] + best[k][r] + vals[k] * edge
                    if got > top:
                        top = got
                best[l][r] = top
        return best[0][m - 1]
