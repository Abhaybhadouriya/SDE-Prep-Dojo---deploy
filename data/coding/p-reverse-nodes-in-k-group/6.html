class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode dummy(0, head);
        ListNode* prevTail = &dummy;            // last node of the part already done
        while (true) {
            // 1. probe: is there a full group of k nodes after prevTail?
            ListNode* kth = prevTail;
            for (int i = 0; i < k && kth; i++) kth = kth->next;
            if (!kth) break;                    // fewer than k left: keep them as they are
            ListNode* first = prevTail->next;   // will become the group's tail
            ListNode* after = kth->next;        // first node of the next group
            // 2. reverse the k nodes; prev starts at `after`, so the new tail
            //    already points to the rest of the list
            ListNode* prev = after;
            ListNode* cur = first;
            while (cur != after) {
                ListNode* nxt = cur->next;
                cur->next = prev;
                prev = cur;
                cur = nxt;
            }
            // 3. stitch: the old k-th node is the new group head
            prevTail->next = kth;
            prevTail = first;
        }
        return dummy.next;
    }
};
func reverseKGroup(head *ListNode, k int) *ListNode {
	dummy := &ListNode{Next: head}
	prevTail := dummy // last node of the part already done
	for {
		// 1. probe: is there a full group of k nodes after prevTail?
		kth := prevTail
		for i := 0; i < k && kth != nil; i++ {
			kth = kth.Next
		}
		if kth == nil {
			break // fewer than k left: keep them as they are
		}
		first, after := prevTail.Next, kth.Next
		// 2. reverse the k nodes; prev starts at `after`
		prev, cur := after, first
		for cur != after {
			nxt := cur.Next
			cur.Next = prev
			prev, cur = cur, nxt
		}
		// 3. stitch: the old k-th node is the new group head
		prevTail.Next = kth
		prevTail = first
	}
	return dummy.Next
}

    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        dummy = ListNode(0, head)
        prev_tail = dummy                # last node of the part already done
        while True:
            # 1. probe: is there a full group of k nodes after prev_tail?
