func buildTree(preorder []int, inorder []int) *TreeNode {
	where := make(map[int]int, len(inorder)) // value -> its index in inorder (values are unique)
	for i, v := range inorder {
		where[v] = i
	}
	next := 0 // next unused position in preorder

	// build the subtree whose nodes are exactly inorder[lo..hi]
	var build func(lo, hi int) *TreeNode
	build = func(lo, hi int) *TreeNode {
		if lo > hi { // empty range: no subtree
			return nil
		}
		root := &TreeNode{Val: preorder[next]} // preorder lists a subtree's root first
		next++
		mid := where[root.Val]          // left of mid = left subtree, right of mid = right subtree
		root.Left = build(lo, mid-1)    // left first: preorder is root, LEFT, right
		root.Right = build(mid+1, hi)
		return root
	}
	return build(0, len(inorder)-1)
}
class Solution {
    unordered_map<int, int> where;   // value -> its index in inorder (values are unique)
    int next = 0;                    // next unused position in preorder

    // build the subtree whose nodes are exactly inorder[lo..hi]
    TreeNode* build(const vector<int>& preorder, int lo, int hi) {
        if (lo > hi) return nullptr;                 // empty range: no subtree
        int rootVal = preorder[next++];              // preorder lists a subtree's root first
        TreeNode* root = new TreeNode(rootVal);
        int mid = where[rootVal];                    // left of mid = left subtree, right of mid = right subtree
        root->left = build(preorder, lo, mid - 1);   // left first: preorder is root, LEFT, right
        root->right = build(preorder, mid + 1, hi);
        return root;
    }

public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        where.clear();
        next = 0;
        for (int i = 0; i < (int)inorder.size(); i++) where[inorder[i]] = i;
        return build(preorder, 0, (int)inorder.size() - 1);
    }
};

static class Solution {
    private Map<Integer, Integer> where;     // value -> its index in inorder (values are unique)
    private int next;                        // next unused position in preorder
    private int[] preorder;

    public TreeNode buildTree(int[] preorder, int[] inorder) {
        this.preorder = preorder;
        this.next = 0;
        this.where = new HashMap<>();
        for (int i = 0; i < inorder.length; i++) where.put(inorder[i], i);
        return build(0, inorder.length - 1);
    }

    // build the subtree whose nodes are exactly inorder[lo..hi]
    private TreeNode build(int lo, int hi) {
        if (lo > hi) return null;                    // empty range: no subtree
        int rootVal = preorder[next++];              // preorder lists a subtree's root first
        TreeNode root = new TreeNode(rootVal);
        int mid = where.get(rootVal);                // left of mid = left subtree, right of mid = right subtree
        root.left = build(lo, mid - 1);              // left first: preorder is root, LEFT, right
        root.right = build(mid + 1, hi);
        return root;
    }
}
class Solution:
    def buildTree(self, preorder: List[int], inorder: List[int]) -> Optional[TreeNode]:
        sys.setrecursionlimit(10000)                  # a 3000-node chain recurses 3000 deep
        where = {v: i for i, v in enumerate(inorder)}  # value -> its index in inorder (values are unique)
        nxt = 0                                        # next unused position in preorder

        def build(lo: int, hi: int) -> Optional[TreeNode]:
            # build the subtree whose nodes are exactly inorder[lo..hi]
            nonlocal nxt
            if lo > hi:                                # empty range: no subtree
                return None
            root = TreeNode(preorder[nxt])             # preorder lists a subtree's root first
            nxt += 1
            mid = where[root.val]                      # left of mid = left subtree, right of mid = right subtree
            root.left = build(lo, mid - 1)             # left first: preorder is root, LEFT, right
            root.right = build(mid + 1, hi)
            return root

        return build(0, len(inorder) - 1)
