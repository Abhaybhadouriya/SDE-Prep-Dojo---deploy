class Solution {
public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        int rows = heights.size(), cols = heights[0].size();
        vector<vector<char>> pac(rows, vector<char>(cols, 0)), atl(rows, vector<char>(cols, 0));
        vector<pair<int, int>> pStart, aStart;
        for (int r = 0; r < rows; r++) {
            pStart.push_back({r, 0});           // left edge touches the Pacific
            aStart.push_back({r, cols - 1});    // right edge touches the Atlantic
        }
        for (int c = 0; c < cols; c++) {
            pStart.push_back({0, c});           // top edge
            aStart.push_back({rows - 1, c});    // bottom edge
        }
        climb(heights, pStart, pac);
        climb(heights, aStart, atl);
        vector<vector<int>> out;
        for (int r = 0; r < rows; r++)
            for (int c = 0; c < cols; c++)
                if (pac[r][c] && atl[r][c]) out.push_back({r, c});
        return out;
    }

private:
    // Reverse the rain: walk UPHILL (to a neighbour at least as high) from the border cells.
    // Every cell reached can drain down to that ocean.
    void climb(const vector<vector<int>>& h, const vector<pair<int, int>>& starts,
               vector<vector<char>>& seen) {
        int rows = h.size(), cols = h[0].size();
        const int dr[4] = {1, -1, 0, 0}, dc[4] = {0, 0, 1, -1};
        queue<pair<int, int>> q;
        for (const auto& s : starts) {
            if (seen[s.first][s.second]) continue;   // corners appear in two edge lists
            seen[s.first][s.second] = 1;             // mark on push
            q.push(s);
        }
        while (!q.empty()) {
            int r = q.front().first, c = q.front().second;
            q.pop();
            for (int d = 0; d < 4; d++) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
                if (seen[nr][nc] || h[nr][nc] < h[r][c]) continue;   // water cannot run uphill to us
                seen[nr][nc] = 1;
                q.push({nr, nc});
            }
        }
    }
};
func pacificAtlantic(heights [][]int) [][]int {
	rows, cols := len(heights), len(heights[0])
	dirs := [4][2]int{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}

	// Reverse the rain: from the border cells, walk UPHILL (neighbour at least as high).
	// Every cell reached can drain down to that ocean.
	climb := func(starts [][2]int) [][]bool {
		seen := make([][]bool, rows)
		for r := range seen {
			seen[r] = make([]bool, cols)
		}
		q := make([][2]int, 0, rows*cols)
		for _, s := range starts {
			if !seen[s[0]][s[1]] { // corners are on two edges
				seen[s[0]][s[1]] = true // mark on push
				q = append(q, s)
			}
		}
		for head := 0; head < len(q); head++ {
			r, c := q[head][0], q[head][1]
			for _, d := range dirs {
				nr, nc := r+d[0], c+d[1]
				if nr < 0 || nc < 0 || nr >= rows || nc >= cols {
					continue
				}
				if seen[nr][nc] || heights[nr][nc] < heights[r][c] {
					continue
				}
				seen[nr][nc] = true
				q = append(q, [2]int{nr, nc})
			}
		}
		return seen
	}

	var pStart, aStart [][2]int
	for r := 0; r < rows; r++ {
		pStart = append(pStart, [2]int{r, 0})        // left edge: Pacific
		aStart = append(aStart, [2]int{r, cols - 1}) // right edge: Atlantic
	}
	for c := 0; c < cols; c++ {
		pStart = append(pStart, [2]int{0, c})        // top edge: Pacific
		aStart = append(aStart, [2]int{rows - 1, c}) // bottom edge: Atlantic
	}
	pac, atl := climb(pStart), climb(aStart)
	out := [][]int{}
	for r := 0; r < rows; r++ {
		for c := 0; c < cols; c++ {
			if pac[r][c] && atl[r][c] {
				out = append(out, []int{r, c})
			}
		}
	}
	return out
}

static class Solution {
    private static final int[] DR = {1, -1, 0, 0}, DC = {0, 0, 1, -1};

    public List<List<Integer>> pacificAtlantic(int[][] heights) {
        int rows = heights.length, cols = heights[0].length;
        boolean[][] pac = new boolean[rows][cols], atl = new boolean[rows][cols];
        ArrayDeque<int[]> pq = new ArrayDeque<>(), aq = new ArrayDeque<>();
        for (int r = 0; r < rows; r++) {
            seed(pq, pac, r, 0);              // left edge: Pacific
            seed(aq, atl, r, cols - 1);       // right edge: Atlantic
        }
        for (int c = 0; c < cols; c++) {
            seed(pq, pac, 0, c);              // top edge: Pacific
            seed(aq, atl, rows - 1, c);       // bottom edge: Atlantic
        }
        climb(heights, pq, pac);
        climb(heights, aq, atl);
        List<List<Integer>> out = new ArrayList<>();
        for (int r = 0; r < rows; r++)
            for (int c = 0; c < cols; c++)
                if (pac[r][c] && atl[r][c]) out.add(List.of(r, c));
        return out;
    }

    private void seed(ArrayDeque<int[]> q, boolean[][] seen, int r, int c) {
        if (seen[r][c]) return;               // corners are on two edges
        seen[r][c] = true;                    // mark on push
        q.add(new int[]{r, c});
    }

    // Reverse the rain: from the border, walk UPHILL (neighbour at least as high).
    private void climb(int[][] h, ArrayDeque<int[]> q, boolean[][] seen) {
        int rows = h.length, cols = h[0].length;
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            for (int d = 0; d < 4; d++) {
                int nr = cur[0] + DR[d], nc = cur[1] + DC[d];
                if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
                if (seen[nr][nc] || h[nr][nc] < h[cur[0]][cur[1]]) continue;
                seen[nr][nc] = true;
                q.add(new int[]{nr, nc});
            }
        }
    }
}
class Solution:
    def pacificAtlantic(self, heights: List[List[int]]) -> List[List[int]]:
        rows, cols = len(heights), len(heights[0])

        def climb(starts):
            # Reverse the rain: from the ocean's border cells, walk UPHILL (to a neighbour that is
            # at least as high). Every cell reached can drain down to this ocean.
            seen = [[False] * cols for _ in range(rows)]
            q = deque()
            for r, c in starts:
                if not seen[r][c]:
                    seen[r][c] = True        # mark on push: each cell enters the queue once
                    q.append((r, c))
            while q:
                r, c = q.popleft()
                for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                    if (0 <= nr < rows and 0 <= nc < cols and not seen[nr][nc]
                            and heights[nr][nc] >= heights[r][c]):
                        seen[nr][nc] = True
                        q.append((nr, nc))
            return seen

        pacific = climb([(0, c) for c in range(cols)] + [(r, 0) for r in range(rows)])
        atlantic = climb([(rows - 1, c) for c in range(cols)] + [(r, cols - 1) for r in range(rows)])
        return [[r, c] for r in range(rows) for c in range(cols)
