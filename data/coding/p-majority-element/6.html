All four use the Boyer-Moore vote (one candidate, one counter).

func majorityElement(nums []int) int {
	// Boyer-Moore vote: every "other" value cancels one copy of the candidate.
	// The majority has more copies than all others together, so it survives.
	candidate, count := 0, 0
	for _, x := range nums {
		if count == 0 { // the prefix so far cancelled out: start fresh
			candidate = x
		}
		if x == candidate {
			count++
		} else {
			count--
		}
	}
	return candidate // valid because a majority is guaranteed
}
class Solution {
public:
    int majorityElement(const vector<int>& nums) {
        // Boyer-Moore vote: every "other" value cancels one copy of the candidate.
        // The majority has more copies than all others together, so it survives.
        int candidate = 0, count = 0;
        for (int x : nums) {
            if (count == 0) candidate = x;      // the prefix so far cancelled out: start fresh
            count += (x == candidate) ? 1 : -1;
        }
        return candidate;                       // valid because a majority is guaranteed
    }
};
