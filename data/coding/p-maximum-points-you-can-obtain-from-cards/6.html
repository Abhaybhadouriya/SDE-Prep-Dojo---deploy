class Solution {
public:
    int maxScore(vector<int>& cardPoints, int k) {
        int n = cardPoints.size();
        int cur = 0;
        for (int i = 0; i < k; i++) cur += cardPoints[i];   // split: k left, 0 right
        int best = cur;
        for (int i = 1; i <= k; i++) {                      // split: k-i left, i right
            cur -= cardPoints[k - i];                       // give back the innermost left card
            cur += cardPoints[n - i];                       // take the next card from the right end
            best = max(best, cur);
        }
        return best;
    }
};
func maxScore(cardPoints []int, k int) int {
	n := len(cardPoints)
	cur := 0
	for i := 0; i < k; i++ { // split: k left, 0 right
		cur += cardPoints[i]
	}
	best := cur
	for i := 1; i <= k; i++ { // split: k-i left, i right
		cur -= cardPoints[k-i] // give back the innermost left card
		cur += cardPoints[n-i] // take the next card from the right end
		if cur > best {
			best = cur
		}
	}
	return best
}

    def maxScore(self, cardPoints: List[int], k: int) -> int:
        n = len(cardPoints)
        cur = sum(cardPoints[:k])          # split: k left, 0 right
        best = cur
        for i in range(1, k + 1):          # split: k-i left, i right
            cur -= cardPoints[k - i]       # give back the innermost left card
            cur += cardPoints[n - i]       # take the next card from the right end
            best = max(best, cur)
