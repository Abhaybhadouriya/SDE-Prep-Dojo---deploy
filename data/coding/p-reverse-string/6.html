The two-pointer loop (the answer) and the recursive version, in four languages; the tested files are in coding-code/reverse-string/. On LeetCode, submit only reverseString.

class Solution {
public:
    // Swap the two ends and walk both pointers toward the middle.
    void reverseString(vector<char>& s) {
        int lo = 0, hi = (int)s.size() - 1;
        while (lo < hi) {                 // lo == hi is the middle of an odd length: leave it
            swap(s[lo], s[hi]);
            lo++;
            hi--;
        }
    }

    // Same swaps, written as recursion (O(n) stack: only to show the idea).
    void reverseStringRec(vector<char>& s) { rec(s, 0, (int)s.size() - 1); }

private:
    void rec(vector<char>& s, int lo, int hi) {
        if (lo >= hi) return;
        swap(s[lo], s[hi]);
        rec(s, lo + 1, hi - 1);
    }
};

// reverseString: swap the two ends and walk both pointers toward the middle.
func reverseString(s []byte) {
	for lo, hi := 0, len(s)-1; lo < hi; lo, hi = lo+1, hi-1 { // lo == hi is the middle: leave it
		s[lo], s[hi] = s[hi], s[lo]
	}
}

// reverseStringRec: same swaps, written as recursion (O(n) stack: only to show the idea).
func reverseStringRec(s []byte) {
	var rec func(lo, hi int)
	rec = func(lo, hi int) {
		if lo >= hi {
			return
		}
		s[lo], s[hi] = s[hi], s[lo]
		rec(lo+1, hi-1)
	}
	rec(0, len(s)-1)
}

static class Solution {
    // Swap the two ends and walk both pointers toward the middle.
    public void reverseString(char[] s) {
        int lo = 0, hi = s.length - 1;
        while (lo < hi) {                 // lo == hi is the middle of an odd length: leave it
            char t = s[lo];
            s[lo++] = s[hi];
            s[hi--] = t;
        }
    }

    // Same swaps, written as recursion (O(n) stack: only to show the idea).
    public void reverseStringRec(char[] s) { rec(s, 0, s.length - 1); }

    private void rec(char[] s, int lo, int hi) {
        if (lo >= hi) return;
        char t = s[lo]; s[lo] = s[hi]; s[hi] = t;
        rec(s, lo + 1, hi - 1);
    }
}
class Solution:
    def reverseString(self, s: List[str]) -> None:
        """Swap the two ends and walk both pointers toward the middle."""
        lo, hi = 0, len(s) - 1
        while lo < hi:                     # lo == hi is the middle of an odd length: leave it
            s[lo], s[hi] = s[hi], s[lo]
            lo += 1
            hi -= 1

    def reverseStringRec(self, s: List[str]) -> None:
        """Same swaps, written as recursion (O(n) stack: only to show the idea)."""
        def rec(lo: int, hi: int) -> None:
            if lo >= hi:
                return
            s[lo], s[hi] = s[hi], s[lo]
            rec(lo + 1, hi - 1)
        rec(0, len(s) - 1)
