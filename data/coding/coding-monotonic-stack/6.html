Reusable templates (Go, C++, Java, Python)

One file per language with everything on this page: next greater (pop time), previous smaller (push time), circular next greater (LC 503), daily temperatures (LC 739), largest rectangle (LC 84), maximal rectangle (LC 85), sum of subarray minimums with the tie rule (LC 907), sum of subarray ranges in one pass per side (LC 2104), remove k digits (LC 402) and remove duplicate letters (LC 316). Each file runs its own checks and prints ok.

What is in each file (same names in every language; Go and Java use camelCase):

  • next_greater - res[i] = index of the first j > i with nums[j] > nums[i], else -1.
  • previous_smaller - res[i] = index of the last j < i with nums[j] < nums[i], else -1.
  • next_greater_circular - LC 503: values of the next greater element in a circular array, else -1.
  • daily_temperatures - LC 739: days to wait for a warmer day (0 if never).
  • largest_rectangle - LC 84: largest rectangle in a histogram with an increasing stack.
  • maximal_rectangle - LC 85: rows of '0'/'1'. Each row turns into a histogram of column heights.
  • sum_subarray_mins - LC 907: sum of the minimum of every subarray, by contribution (strict on one side, non-strict on the other).
  • sub_array_ranges - LC 2104: sum of (max - min) over all subarrays = sum of maxes - sum of mins.
  • remove_k_digits - LC 402: smallest number after deleting k digits.
  • remove_duplicate_letters - LC 316 / 1081: smallest subsequence with every distinct letter exactly once.

package main

import (
	"fmt"
	"reflect"
	"strings"
)

const MOD = 1_000_000_007

// NEXT GREATER: stack of indexes with decreasing values; the answer is written at pop time.
func nextGreater(a []int) []int {
	res := make([]int, len(a))
	st := []int{} // a slice is the stack: append = push, st[:len(st)-1] = pop
	for i, x := range a {
		res[i] = -1
		for len(st) > 0 && a[st[len(st)-1]] < x {
			res[st[len(st)-1]] = i
			st = st[:len(st)-1]
		}
		st = append(st, i)
	}
	return res
}

// PREVIOUS SMALLER: answer read at push time (the top left after popping).
func previousSmaller(a []int) []int {
	res := make([]int, len(a))
	st := []int{}
	for i, x := range a {
		for len(st) > 0 && a[st[len(st)-1]] >= x {
			st = st[:len(st)-1]
		}
		res[i] = -1
		if len(st) > 0 {
			res[i] = st[len(st)-1]
		}
		st = append(st, i)
	}
	return res
}

// LC 503: circular array -> walk it twice, push each index only on the first lap.
func nextGreaterCircular(a []int) []int {
	n := len(a)
	res := make([]int, n)
	for i := range res {
		res[i] = -1
	}
	st := []int{}
	for i := 0; i < 2*n; i++ {
		x := a[i%n]
		for len(st) > 0 && a[st[len(st)-1]] < x {
			res[st[len(st)-1]] = x
			st = st[:len(st)-1]
		}
		if i < n {
			st = append(st, i)
		}
	}
	return res
}

// LC 739: days until a warmer day.
func dailyTemperatures(t []int) []int {
	ng := nextGreater(t)
	res := make([]int, len(t))
	for i, j := range ng {
		if j != -1 {
			res[i] = j - i
		}
	}
	return res
}

// LC 84: increasing stack; when bar top is popped by i, its widest rectangle is (newTop, i).
func largestRectangle(h []int) int64 {
	n := len(h)
	st := []int{}
	var best int64
	for i := 0; i <= n; i++ {
		cur := 0 // sentinel flushes the stack
		if i < n {
			cur = h[i]
		}
		for len(st) > 0 && h[st[len(st)-1]] >= cur {
			top := st[len(st)-1]
			st = st[:len(st)-1]
			left := -1
			if len(st) > 0 {
				left = st[len(st)-1]
			}
			if area := int64(h[top]) * int64(i-left-1); area > best {
				best = area
			}
		}
		st = append(st, i)
	}
	return best
}

// LC 85: each row becomes a histogram of consecutive '1' heights.
func maximalRectangle(m []string) int64 {
	if len(m) == 0 {
		return 0
	}
	heights := make([]int, len(m[0]))
	var best int64
	for _, row := range m {
		for j := 0; j < len(row); j++ {
			if row[j] == '1' {
				heights[j]++
			} else {
				heights[j] = 0
			}
		}
		if v := largestRectangle(heights); v > best {
			best = v
		}
	}
	return best
}

// LC 907: contribution = a[i] * (i - prevStrictlySmaller) * (nextSmallerOrEqual - i).
func sumSubarrayMins(a []int) int64 {
	n := len(a)
	left, right := make([]int64, n), make([]int64, n)
	st := []int{}
	for i := 0; i < n; i++ {
		for len(st) > 0 && a[st[len(st)-1]] > a[i] {
			st = st[:len(st)-1]
		}
		p := -1
		if len(st) > 0 {
			p = st[len(st)-1]
		}
		left[i] = int64(i - p)
		st = append(st, i)
	}
	st = st[:0]
	for i := n - 1; i >= 0; i-- {
		for len(st) > 0 && a[st[len(st)-1]] >= a[i] {
			st = st[:len(st)-1]
		}
		q := n
		if len(st) > 0 {
			q = st[len(st)-1]
		}
		right[i] = int64(q - i)
		st = append(st, i)
	}
	var total int64
	for i := 0; i < n; i++ {
		total = (total + int64(a[i])*left[i]%MOD*right[i]) % MOD
	}
	return total
}

// One pass: sum over all subarrays of the max (wantMax) or min. Span settled at pop time.
func sumOfBest(a []int, wantMax bool) int64 {
	n := len(a)
	st := []int{}
	var total int64
	for i := 0; i <= n; i++ {
		for len(st) > 0 {
			t := a[st[len(st)-1]]
			pop := i == n
			if !pop {
				if wantMax {
					pop = t <= a[i]
				} else {
					pop = t >= a[i]
				}
			}
			if !pop {
				break
			}
			mid := st[len(st)-1]
			st = st[:len(st)-1]
			left := -1
			if len(st) > 0 {
				left = st[len(st)-1]
			}
			total += int64(a[mid]) * int64(mid-left) * int64(i-mid)
		}
		st = append(st, i)
	}
	return total
}

// LC 2104: sum of ranges = sum of maxes - sum of mins.
func subArrayRanges(a []int) int64 { return sumOfBest(a, true) - sumOfBest(a, false) }

// LC 402: greedy stack keeps the kept digits non-decreasing.
func removeKdigits(num string, k int) string {
	st := make([]byte, 0, len(num))
	for i := 0; i < len(num); i++ {
		d := num[i]
		for k > 0 && len(st) > 0 && st[len(st)-1] > d {
			st = st[:len(st)-1]
			k--
		}
		st = append(st, d)
	}
	if k > len(st) {
		k = len(st)
	}
	st = st[:len(st)-k] // leftover deletions cut the tail
	out := strings.TrimLeft(string(st), "0")
	if out == "" {
		return "0"
	}
	return out
}

// LC 316 / 1081: pop a bigger letter only if it appears again later.
func removeDuplicateLetters(s string) string {
	var last [26]int
	var used [26]bool
	for i := 0; i < len(s); i++ {
		last[s[i]-'a'] = i
	}
	st := []byte{}
	for i := 0; i < len(s); i++ {
		c := s[i]
		if used[c-'a'] {
			continue
		}
		for len(st) > 0 && st[len(st)-1] > c && last[st[len(st)-1]-'a'] > i {
			used[st[len(st)-1]-'a'] = false
			st = st[:len(st)-1]
		}
		st = append(st, c)
		used[c-'a'] = true
	}
	return string(st)
}

	}
}

func main() {
	check(reflect.DeepEqual(nextGreater([]int{2, 1, 2, 4, 3}), []int{3, 2, 3, -1, -1}), "ng")
	check(reflect.DeepEqual(previousSmaller([]int{3, 1, 4, 1, 5}), []int{-1, -1, 1, -1, 3}), "ps")
	check(reflect.DeepEqual(nextGreaterCircular([]int{1, 2, 1}), []int{2, -1, 2}), "503")
	check(reflect.DeepEqual(dailyTemperatures([]int{73, 74, 75, 71, 69, 72, 76, 73}), []int{1, 1, 4, 2, 1, 1, 0, 0}), "739")
	check(largestRectangle([]int{2, 1, 5, 6, 2, 3}) == 10, "84")
	check(maximalRectangle([]string{"10100", "10111", "11111", "10010"}) == 6, "85")
	check(sumSubarrayMins([]int{3, 1, 2, 4}) == 17, "907a")
	check(sumSubarrayMins([]int{11, 81, 94, 43, 3}) == 444, "907b")

	check(removeKdigits("10200", 1) == "200", "402b")

#include <algorithm>
#include <cassert>
#include <cstdio>
#include <string>
#include <vector>
using namespace std;
const long long MOD = 1000000007LL;

// NEXT GREATER: stack of indexes with decreasing values; the answer is written at pop time.
vector<int> nextGreater(const vector<int>& a) {
    vector<int> res(a.size(), -1), st;
    for (int i = 0; i < (int)a.size(); i++) {
        while (!st.empty() && a[st.back()] < a[i]) { res[st.back()] = i; st.pop_back(); }
        st.push_back(i);
    }
    return res;
}

// PREVIOUS SMALLER: answer read at push time (the top left after popping).
vector<int> previousSmaller(const vector<int>& a) {
    vector<int> res(a.size(), -1), st;
    for (int i = 0; i < (int)a.size(); i++) {
        while (!st.empty() && a[st.back()] >= a[i]) st.pop_back();
        res[i] = st.empty() ? -1 : st.back();
        st.push_back(i);
    }
    return res;
}

// LC 503: circular array -> walk it twice, push each index only on the first lap.
vector<int> nextGreaterCircular(const vector<int>& a) {
    int n = a.size();
    vector<int> res(n, -1), st;
    for (int i = 0; i < 2 * n; i++) {
        int x = a[i % n];
        while (!st.empty() && a[st.back()] < x) { res[st.back()] = x; st.pop_back(); }
        if (i < n) st.push_back(i);
    }
    return res;
}

// LC 739: days until a warmer day.
vector<int> dailyTemperatures(const vector<int>& t) {
    vector<int> ng = nextGreater(t), res(t.size(), 0);
    for (int i = 0; i < (int)t.size(); i++) if (ng[i] != -1) res[i] = ng[i] - i;
    return res;
}

// LC 84: increasing stack; when bar top is popped by i, its widest rectangle is (newTop, i).
long long largestRectangle(const vector<int>& h) {
    vector<int> st;
    long long best = 0;
    int n = h.size();
    for (int i = 0; i <= n; i++) {
        int cur = i < n ? h[i] : 0;                         // sentinel flushes the stack
        while (!st.empty() && h[st.back()] >= cur) {
            int top = st.back(); st.pop_back();
            int left = st.empty() ? -1 : st.back();
            best = max(best, (long long)h[top] * (i - left - 1));
        }
        st.push_back(i);
    }
    return best;
}

// LC 85: each row becomes a histogram of consecutive '1' heights.
long long maximalRectangle(const vector<string>& m) {
    if (m.empty()) return 0;
    vector<int> heights(m[0].size(), 0);
    long long best = 0;
    for (const string& row : m) {
        for (size_t j = 0; j < row.size(); j++) heights[j] = row[j] == '1' ? heights[j] + 1 : 0;
        best = max(best, largestRectangle(heights));
    }
    return best;
}

// LC 907: contribution = a[i] * (i - prevStrictlySmaller) * (nextSmallerOrEqual - i).
long long sumSubarrayMins(const vector<int>& a) {
    int n = a.size();
    vector<long long> left(n), right(n);
    vector<int> st;
    for (int i = 0; i < n; i++) {
        while (!st.empty() && a[st.back()] > a[i]) st.pop_back();
        left[i] = i - (st.empty() ? -1 : st.back());
        st.push_back(i);
    }
    st.clear();
    for (int i = n - 1; i >= 0; i--) {
        while (!st.empty() && a[st.back()] >= a[i]) st.pop_back();
        right[i] = (st.empty() ? n : st.back()) - i;
        st.push_back(i);
    }
    long long total = 0;
    for (int i = 0; i < n; i++) total = (total + a[i] * left[i] % MOD * right[i]) % MOD;
    return total;
}

// One pass: sum over all subarrays of the max (wantMax) or min. Span settled at pop time.
long long sumOfBest(const vector<int>& a, bool wantMax) {
    vector<int> st;
    long long total = 0;
    int n = a.size();
    for (int i = 0; i <= n; i++) {
        while (!st.empty() && (i == n || (wantMax ? a[st.back()] <= a[i] : a[st.back()] >= a[i]))) {
            int mid = st.back(); st.pop_back();
            int left = st.empty() ? -1 : st.back();
            total += (long long)a[mid] * (mid - left) * (i - mid);
        }
        st.push_back(i);
    }
    return total;
}

// LC 2104: sum of ranges = sum of maxes - sum of mins.
long long subArrayRanges(const vector<int>& a) { return sumOfBest(a, true) - sumOfBest(a, false); }

// LC 402: greedy stack keeps the kept digits non-decreasing.
string removeKdigits(const string& num, int k) {
    string st;
    for (char d : num) {
        while (k > 0 && !st.empty() && st.back() > d) { st.pop_back(); k--; }
        st.push_back(d);
    }
    st.resize(st.size() - min<size_t>(k, st.size()));       // leftover deletions cut the tail
    size_t nz = st.find_first_not_of('0');
    return nz == string::npos ? "0" : st.substr(nz);
}

// LC 316 / 1081: pop a bigger letter only if it appears again later.
string removeDuplicateLetters(const string& s) {
    int last[26];
    bool used[26] = {false};
    for (int i = 0; i < (int)s.size(); i++) last[s[i] - 'a'] = i;
    string st;
    for (int i = 0; i < (int)s.size(); i++) {
        int c = s[i] - 'a';
        if (used[c]) continue;
        while (!st.empty() && st.back() > s[i] && last[st.back() - 'a'] > i) {
            used[st.back() - 'a'] = false;
            st.pop_back();
        }
        st.push_back(s[i]);
        used[c] = true;
    }
    return st;
}

// Run: java MonoStack.java   (checks; prints "ok")
import java.util.*;

public class MonoStack {
    static final long MOD = 1_000_000_007L;

    // NEXT GREATER: stack of indexes with decreasing values; the answer is written at pop time.
    static int[] nextGreater(int[] a) {
        int[] res = new int[a.length];
        Arrays.fill(res, -1);
        Deque<Integer> st = new ArrayDeque<>();                 // ArrayDeque, not java.util.Stack
        for (int i = 0; i < a.length; i++) {
            while (!st.isEmpty() && a[st.peek()] < a[i]) res[st.pop()] = i;
            st.push(i);
        }
        return res;
    }

    // PREVIOUS SMALLER: answer read at push time (the top left after popping).
    static int[] previousSmaller(int[] a) {
        int[] res = new int[a.length];
        Deque<Integer> st = new ArrayDeque<>();
        for (int i = 0; i < a.length; i++) {
            while (!st.isEmpty() && a[st.peek()] >= a[i]) st.pop();
            res[i] = st.isEmpty() ? -1 : st.peek();
            st.push(i);
        }
        return res;
    }

    // LC 503: circular array -> walk it twice, push each index only on the first lap.
    static int[] nextGreaterCircular(int[] a) {
        int n = a.length;
        int[] res = new int[n];
        Arrays.fill(res, -1);
        Deque<Integer> st = new ArrayDeque<>();
        for (int i = 0; i < 2 * n; i++) {
            int x = a[i % n];
            while (!st.isEmpty() && a[st.peek()] < x) res[st.pop()] = x;
            if (i < n) st.push(i);
        }
        return res;
    }

    // LC 739: days until a warmer day.
    static int[] dailyTemperatures(int[] t) {
        int[] ng = nextGreater(t), res = new int[t.length];
        for (int i = 0; i < t.length; i++) if (ng[i] != -1) res[i] = ng[i] - i;
        return res;
    }

    // LC 84: increasing stack; when bar top is popped by i, its widest rectangle is (newTop, i).
    static long largestRectangle(int[] h) {
        int n = h.length;
        int[] st = new int[n + 1];                              // array stack: no boxing
        int sp = 0;
        long best = 0;
        for (int i = 0; i <= n; i++) {
            int cur = i < n ? h[i] : 0;                         // sentinel flushes the stack
            while (sp > 0 && h[st[sp - 1]] >= cur) {
                int top = st[--sp];
                int left = sp == 0 ? -1 : st[sp - 1];
                best = Math.max(best, (long) h[top] * (i - left - 1));
            }
            st[sp++] = i;
        }
        return best;
    }

    // LC 85: each row becomes a histogram of consecutive '1' heights.
    static long maximalRectangle(String[] m) {
        if (m.length == 0) return 0;
        int[] heights = new int[m[0].length()];
        long best = 0;
        for (String row : m) {
            for (int j = 0; j < row.length(); j++) heights[j] = row.charAt(j) == '1' ? heights[j] + 1 : 0;
            best = Math.max(best, largestRectangle(heights));
        }
        return best;
    }

    // LC 907: contribution = a[i] * (i - prevStrictlySmaller) * (nextSmallerOrEqual - i).
    static long sumSubarrayMins(int[] a) {
        int n = a.length;
        long[] left = new long[n], right = new long[n];
        Deque<Integer> st = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            while (!st.isEmpty() && a[st.peek()] > a[i]) st.pop();
            left[i] = i - (st.isEmpty() ? -1 : st.peek());
            st.push(i);
        }
        st.clear();
        for (int i = n - 1; i >= 0; i--) {
            while (!st.isEmpty() && a[st.peek()] >= a[i]) st.pop();
            right[i] = (st.isEmpty() ? n : st.peek()) - i;
            st.push(i);
        }
        long total = 0;
        for (int i = 0; i < n; i++) total = (total + a[i] * left[i] % MOD * right[i]) % MOD;
        return total;
    }

    // One pass: sum over all subarrays of the max (wantMax) or min. Span settled at pop time.
    static long sumOfBest(int[] a, boolean wantMax) {
        int n = a.length;
        int[] st = new int[n + 1];
        int sp = 0;
        long total = 0;
        for (int i = 0; i <= n; i++) {
            while (sp > 0 && (i == n || (wantMax ? a[st[sp - 1]] <= a[i] : a[st[sp - 1]] >= a[i]))) {
                int mid = st[--sp];
                int left = sp == 0 ? -1 : st[sp - 1];
                total += (long) a[mid] * (mid - left) * (i - mid);
            }
            st[sp++] = i;
        }
        return total;
    }

    // LC 2104: sum of ranges = sum of maxes - sum of mins.
    static long subArrayRanges(int[] a) { return sumOfBest(a, true) - sumOfBest(a, false); }

    // LC 402: greedy stack keeps the kept digits non-decreasing.
    static String removeKdigits(String num, int k) {
        StringBuilder st = new StringBuilder();
        for (char d : num.toCharArray()) {
            while (k > 0 && st.length() > 0 && st.charAt(st.length() - 1) > d) {
                st.deleteCharAt(st.length() - 1);
                k--;
            }
            st.append(d);
        }
        st.setLength(st.length() - Math.min(k, st.length()));   // leftover deletions cut the tail
        int nz = 0;
        while (nz < st.length() && st.charAt(nz) == '0') nz++;
        return nz == st.length() ? "0" : st.substring(nz);
    }

    // LC 316 / 1081: pop a bigger letter only if it appears again later.
    static String removeDuplicateLetters(String s) {
        int[] last = new int[26];
        boolean[] used = new boolean[26];
        for (int i = 0; i < s.length(); i++) last[s.charAt(i) - 'a'] = i;
        StringBuilder st = new StringBuilder();
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (used[c - 'a']) continue;
            while (st.length() > 0 && st.charAt(st.length() - 1) > c && last[st.charAt(st.length() - 1) - 'a'] > i) {
                used[st.charAt(st.length() - 1) - 'a'] = false;
                st.deleteCharAt(st.length() - 1);
            }
            st.append(c);
            used[c - 'a'] = true;
        }
        return st.toString();
    }

    static void check(boolean ok, String what) { if (!ok) throw new AssertionError(what); }

    public static void main(String[] args) {
        check(Arrays.equals(nextGreater(new int[]{2, 1, 2, 4, 3}), new int[]{3, 2, 3, -1, -1}), "ng");
        check(Arrays.equals(previousSmaller(new int[]{3, 1, 4, 1, 5}), new int[]{-1, -1, 1, -1, 3}), "ps");
        check(Arrays.equals(nextGreaterCircular(new int[]{1, 2, 1}), new int[]{2, -1, 2}), "503");
        check(Arrays.equals(dailyTemperatures(new int[]{73, 74, 75, 71, 69, 72, 76, 73}), new int[]{1, 1, 4, 2, 1, 1, 0, 0}), "739");
        check(largestRectangle(new int[]{2, 1, 5, 6, 2, 3}) == 10, "84");
        check(maximalRectangle(new String[]{"10100", "10111", "11111", "10010"}) == 6, "85");
        check(sumSubarrayMins(new int[]{3, 1, 2, 4}) == 17, "907a");
        check(sumSubarrayMins(new int[]{11, 81, 94, 43, 3}) == 444, "907b");
        check(sumSubarrayMins(new int[]{2, 2, 2}) == 12, "907c");
        check(subArrayRanges(new int[]{4, -2, -3, 4, 1}) == 59, "2104");
        check(removeKdigits("1432219", 3).equals("1219"), "402a");
        check(removeKdigits("10200", 1).equals("200"), "402b");

"""Monotonic stack templates (SDE Prep Dojo, lesson coding-monotonic-stack).

Run: python mono_stack.py   (random checks against brute force; prints "ok")
"""
import random

MOD = 10**9 + 7


# ---- 1. NEXT GREATER: decreasing stack of indexes, answer decided at pop time ----
def next_greater(nums):
    """res[i] = index of the first j > i with nums[j] > nums[i], else -1."""
    res = [-1] * len(nums)
    stack = []                                    # indexes, values strictly decreasing
    for i, x in enumerate(nums):
        while stack and nums[stack[-1]] < x:      # x is the answer for everyone smaller
            res[stack.pop()] = i
        stack.append(i)
    return res


def previous_smaller(nums):
    """res[i] = index of the last j < i with nums[j] < nums[i], else -1.
    Answer decided at push time: after popping, the top is the answer."""
    res = [-1] * len(nums)
    stack = []                                    # values strictly increasing
    for i, x in enumerate(nums):
        while stack and nums[stack[-1]] >= x:
            stack.pop()
        res[i] = stack[-1] if stack else -1
        stack.append(i)
    return res


def next_greater_circular(nums):
    """LC 503: values of the next greater element in a circular array, else -1."""
    n = len(nums)
    res = [-1] * n
    stack = []
    for i in range(2 * n):                        # walk the array twice
        x = nums[i % n]
        while stack and nums[stack[-1]] < x:
            res[stack.pop()] = x
        if i < n:                                 # push each index only once
            stack.append(i)
    return res


def daily_temperatures(temps):
    """LC 739: days to wait for a warmer day (0 if never)."""
    ng = next_greater(temps)
    return [j - i if j != -1 else 0 for i, j in enumerate(ng)]


# ---- 2. HISTOGRAM: increasing stack, a bar's rectangle is settled when it is popped ----
def largest_rectangle(heights):
    """LC 84. When bar `top` is popped by i, its rectangle spans (new top, i)."""
    stack, best = [], 0                           # indexes, heights increasing
    for i in range(len(heights) + 1):
        h = heights[i] if i < len(heights) else 0  # sentinel 0 flushes the stack
        while stack and heights[stack[-1]] >= h:
            top = stack.pop()
            left = stack[-1] if stack else -1      # first smaller on the left
            best = max(best, heights[top] * (i - left - 1))
        stack.append(i)
    return best


def maximal_rectangle(matrix):
    """LC 85: rows of '0'/'1'. Each row turns into a histogram of column heights."""
    if not matrix:
        return 0
    heights = [0] * len(matrix[0])
    best = 0
    for row in matrix:
        for j, c in enumerate(row):
            heights[j] = heights[j] + 1 if c == "1" else 0
        best = max(best, largest_rectangle(heights))
    return best


# ---- 3. CONTRIBUTION: each element is the min of (i - left) * (right - i) subarrays ----
def sum_subarray_mins(arr):
    """LC 907 (mod 1e9+7). Ties: strict on one side, non-strict on the other,
    so a subarray with two equal minimums is counted exactly once."""
    n = len(arr)
    left = [0] * n                                # distance to previous strictly smaller
    right = [0] * n                               # distance to next smaller-or-equal
    stack = []
    for i in range(n):
        while stack and arr[stack[-1]] > arr[i]:
            stack.pop()
        left[i] = i - (stack[-1] if stack else -1)
        stack.append(i)
    stack = []
    for i in range(n - 1, -1, -1):
        while stack and arr[stack[-1]] >= arr[i]:
            stack.pop()
        right[i] = (stack[-1] if stack else n) - i
        stack.append(i)
    return sum(arr[i] * left[i] * right[i] for i in range(n)) % MOD


def _sum_of(arr, better):
    """Sum over all subarrays of their best element (better(a, b): a beats b)."""
    total, stack = 0, []
    for i in range(len(arr) + 1):
        while stack and (i == len(arr) or not better(arr[stack[-1]], arr[i])):
            mid = stack.pop()                     # mid's span is settled now
            left = stack[-1] if stack else -1
            total += arr[mid] * (mid - left) * (i - mid)
        stack.append(i)
    return total


def sub_array_ranges(nums):
    """LC 2104: sum of (max - min) over all subarrays = sum of maxes - sum of mins."""
    return _sum_of(nums, lambda a, b: a > b) - _sum_of(nums, lambda a, b: a < b)


# ---- 4. GREEDY STACK: keep the answer string itself monotone ----
def remove_k_digits(num, k):
    """LC 402: smallest number after deleting k digits."""
    stack = []
    for d in num:
        while k and stack and stack[-1] > d:      # a bigger digit before a smaller one: drop it
            stack.pop()
            k -= 1
        stack.append(d)
    if k:
        stack = stack[:-k]                        # still have deletions: cut the tail
    return "".join(stack).lstrip("0") or "0"


def remove_duplicate_letters(s):
    """LC 316 / 1081: smallest subsequence with every distinct letter exactly once."""
    last = {c: i for i, c in enumerate(s)}
    stack, used = [], set()
    for i, c in enumerate(s):
        if c in used:
            continue
        while stack and stack[-1] > c and last[stack[-1]] > i:  # can re-add it later
            used.discard(stack.pop())
        stack.append(c)
        used.add(c)
    return "".join(stack)


# ---------------- brute-force checks ----------------
def _check():
    from itertools import combinations
    rnd = random.Random(11)
    for _ in range(400):
        n = rnd.randint(1, 10)
        a = [rnd.randint(1, 5) for _ in range(n)]
        assert next_greater(a) == [next((j for j in range(i + 1, n) if a[j] > a[i]), -1) for i in range(n)]
        assert previous_smaller(a) == [next((j for j in range(i - 1, -1, -1) if a[j] < a[i]), -1) for i in range(n)]
        circ = []
        for i in range(n):
            circ.append(next((a[(i + d) % n] for d in range(1, n) if a[(i + d) % n] > a[i]), -1))
        assert next_greater_circular(a) == circ
        subs = [a[i:j] for i in range(n) for j in range(i + 1, n + 1)]
        assert largest_rectangle(a) == max(min(w) * len(w) for w in subs)
        assert sum_subarray_mins(a) == sum(min(w) for w in subs) % MOD
        assert sub_array_ranges(a) == sum(max(w) - min(w) for w in subs)
        num = "".join(rnd.choice("0123") for _ in range(n))
        k = rnd.randint(0, n)
        cands = ["".join(c) for c in combinations(num, n - k)]
        exp = min((int(c) if c else 0) for c in cands) if n - k else 0
        assert int(remove_k_digits(num, k)) == exp
        s = "".join(rnd.choice("abcd") for _ in range(n))
        letters = set(s)
        best = min("".join(c) for r in [len(letters)] for c in combinations(s, r) if set(c) == letters)
        assert remove_duplicate_letters(s) == best
        rows, cols = rnd.randint(1, 4), rnd.randint(1, 4)
        m = ["".join(rnd.choice("01") for _ in range(cols)) for _ in range(rows)]
        bf = 0
        for r1 in range(rows):
            for r2 in range(r1, rows):
                for c1 in range(cols):
                    for c2 in range(c1, cols):
                        if all(m[r][c] == "1" for r in range(r1, r2 + 1) for c in range(c1, c2 + 1)):
                            bf = max(bf, (r2 - r1 + 1) * (c2 - c1 + 1))
        assert maximal_rectangle(m) == bf
    assert daily_temperatures([73, 74, 75, 71, 69, 72, 76, 73]) == [1, 1, 4, 2, 1, 1, 0, 0]
    assert largest_rectangle([2, 1, 5, 6, 2, 3]) == 10
    assert remove_k_digits("1432219", 3) == "1219"
    assert remove_duplicate_letters("cbacdcbc") == "acdb"
    print("ok")


if __name__ == "__main__":
    _check()

Language notes. In Go a slice is the stack (append to push, st[:len(st)-1] to pop). In C++ use std::vector<int> rather than std::stack when you need to look below the top. In Java use ArrayDeque<Integer> (never Stack, which is synchronised and slower) or a plain int[] with a top index, which avoids boxing. In Python a list with append / pop is the stack.
