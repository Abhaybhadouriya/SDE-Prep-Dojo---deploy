class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& grid) {
        int cols = grid[0].size();
        // long long: a cell that cannot reach the goal may still collect a huge count
        vector<long long> ways(cols, 0);   // ways[c] = paths into (current row, c)
        ways[0] = 1;                       // the start (cleared below if it is a rock)
        for (const auto& row : grid) {
            for (int c = 0; c < cols; ++c) {
                if (row[c] == 1) ways[c] = 0;           // rock: no path goes through it
                else if (c > 0) ways[c] += ways[c - 1]; // from above + from the left
            }
        }
        return (int)ways[cols - 1];
    }
};
func uniquePathsWithObstacles(grid [][]int) int {

class Solution:
    def uniquePathsWithObstacles(self, grid: List[List[int]]) -> int:
        cols = len(grid[0])
        ways = [0] * cols          # ways[c] = paths into cell (current row, c)
        ways[0] = 1                # one way to stand on the start (zeroed below if blocked)
        for row in grid:
            for c in range(cols):
                if row[c] == 1:
                    ways[c] = 0                # a rock: nobody passes through here
                elif c > 0:
                    ways[c] += ways[c - 1]     # from above (old ways[c]) + from the left
        return ways[-1]
