The 26-slot stock in four languages. Each file in coding-code/ransom-note/ also checks the result against the cross-out brute force on 500 random string pairs.

class Solution {
public:
    bool canConstruct(string ransomNote, string magazine) {
        if (ransomNote.size() > magazine.size()) return false;  // not enough letters in total
        int stock[26] = {0};                  // stock[c] = copies of letter c still unused
        for (char ch : magazine) stock[ch - 'a']++;
        for (char ch : ransomNote)
            if (--stock[ch - 'a'] < 0) return false;   // the note needs one copy too many
        return true;
    }
};
func canConstruct(ransomNote string, magazine string) bool {
	if len(ransomNote) > len(magazine) {
		return false // not enough letters in total
	}
	var stock [26]int // stock[c] = copies of letter c still unused in the magazine
	for i := 0; i < len(magazine); i++ {
		stock[magazine[i]-'a']++
	}
	for i := 0; i < len(ransomNote); i++ {
		c := ransomNote[i] - 'a'
		stock[c]--
		if stock[c] < 0 {
			return false // the note needs one more copy than the magazine has
		}
	}
	return true
}

class Solution:
    def canConstruct(self, ransomNote: str, magazine: str) -> bool:
        if len(ransomNote) > len(magazine):       # not enough letters in total
            return False
        stock = [0] * 26                          # stock[c] = copies of letter c still unused

            stock[ord(ch) - ord('a')] += 1
        for ch in ransomNote:
            i = ord(ch) - ord('a')
            stock[i] -= 1
