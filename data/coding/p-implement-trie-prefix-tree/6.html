type trieNode struct {
	child [26]*trieNode // child[c-'a'], nil = no edge
	isEnd bool          // a whole inserted word stops here
}

type Trie struct {
	root *trieNode
}

func Constructor() Trie {
	return Trie{root: &trieNode{}}
}

func (t *Trie) Insert(word string) {
	cur := t.root
	for i := 0; i < len(word); i++ {
		k := word[i] - 'a'
		if cur.child[k] == nil { // create only missing edges
			cur.child[k] = &trieNode{}
		}
		cur = cur.child[k]
	}
	cur.isEnd = true // mark the last node, not a new node
}

func (t *Trie) Search(word string) bool {
	n := t.walk(word)
	return n != nil && n.isEnd // the path exists AND a word ends here
}

func (t *Trie) StartsWith(prefix string) bool {
	return t.walk(prefix) != nil // the path alone is enough
}

// walk follows s letter by letter; nil as soon as an edge is missing.
func (t *Trie) walk(s string) *trieNode {
	cur := t.root
	for i := 0; i < len(s) && cur != nil; i++ {
		cur = cur.child[s[i]-'a']
	}
	return cur
}
class Trie {
    struct Node {
        Node* child[26] = {};   // child[c - 'a'], nullptr = no edge
        bool isEnd = false;     // a whole inserted word stops here
    };
    Node* root;

    // Follow s letter by letter; nullptr as soon as an edge is missing.
    Node* walk(const string& s) const {
        Node* cur = root;
        for (char ch : s) {
            cur = cur->child[ch - 'a'];
            if (!cur) return nullptr;
        }
        return cur;
    }
    static void freeAll(Node* n) {
        if (!n) return;
        for (Node* c : n->child) freeAll(c);
        delete n;
    }

public:
    Trie() : root(new Node()) {}
    ~Trie() { freeAll(root); }
    Trie(const Trie&) = delete;             // owns raw nodes: no copies
    Trie& operator=(const Trie&) = delete;

    void insert(const string& word) {
        Node* cur = root;
        for (char ch : word) {
            int k = ch - 'a';
            if (!cur->child[k]) cur->child[k] = new Node();   // create only missing edges
            cur = cur->child[k];
        }
        cur->isEnd = true;      // mark the last node, not a new node
    }

    bool search(const string& word) const {
        Node* n = walk(word);
        return n && n->isEnd;   // the path exists AND a word ends here
    }

    bool startsWith(const string& prefix) const {
        return walk(prefix) != nullptr;   // the path alone is enough
    }
};

class TrieNode:
    __slots__ = ("child", "is_end")

    def __init__(self):
        self.child = {}        # letter -> TrieNode (a dict keeps only real edges)
        self.is_end = False    # a whole inserted word stops here


class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word: str) -> None:
        cur = self.root
        for ch in word:
            if ch not in cur.child:            # create only missing edges
                cur.child[ch] = TrieNode()
            cur = cur.child[ch]
        cur.is_end = True                      # mark the last node, not a new node

    def search(self, word: str) -> bool:
        node = self._walk(word)
        return node is not None and node.is_end   # the path exists AND a word ends here

    def startsWith(self, prefix: str) -> bool:
        return self._walk(prefix) is not None     # the path alone is enough

    def _walk(self, s: str):
        """Follow s letter by letter; None as soon as an edge is missing."""
        cur = self.root
        for ch in s:
            cur = cur.child.get(ch)
            if cur is None:
                return None
        return cur
