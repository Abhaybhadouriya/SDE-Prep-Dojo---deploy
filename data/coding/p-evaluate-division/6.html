Weighted union-find in four languages; the tests in coding-code/evaluate-division/ cover unknown variables, x / x, separate groups and a 20-link chain. Recursion in find is safe here: there are at most 40 variables.

class Solution {
public:
    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values,
                                vector<vector<string>>& queries) {
        for (size_t i = 0; i < equations.size(); i++) {
            const string &a = equations[i][0], &b = equations[i][1];
            add(a); add(b);
            auto pa = find(a), pb = find(b);          // a = wa * ra, b = wb * rb
            if (pa.first != pb.first) {               // hang ra under rb: ra / rb = v * wb / wa
                parent[pa.first] = pb.first;
                ratio[pa.first] = values[i] * pb.second / pa.second;
            }
        }
        vector<double> out;
        for (auto& q : queries) {
            if (!parent.count(q[0]) || !parent.count(q[1])) { out.push_back(-1.0); continue; }
            auto pc = find(q[0]), pd = find(q[1]);
            out.push_back(pc.first == pd.first ? pc.second / pd.second : -1.0);
        }
        return out;
    }

private:
    unordered_map<string, string> parent;   // variable -> parent variable
    unordered_map<string, double> ratio;    // variable -> variable / parent

    void add(const string& x) {
        if (!parent.count(x)) { parent[x] = x; ratio[x] = 1.0; }
    }
    // returns {root, x / root} and compresses the path
    pair<string, double> find(const string& x) {
        if (parent[x] != x) {
            auto up = find(parent[x]);         // up.second = parent / root
            parent[x] = up.first;
            ratio[x] *= up.second;
        }
        return {parent[x], ratio[x]};
    }
};
func calcEquation(equations [][]string, values []float64, queries [][]string) []float64 {
	parent := map[string]string{}  // variable -> parent variable
	ratio := map[string]float64{}  // variable -> variable / parent
	var find func(x string) string // returns the root; then ratio[x] = x / root
	find = func(x string) string {
		p := parent[x]
		if p == x {
			return x
		}
		root := find(p) // now ratio[p] = p / root
		parent[x] = root
		ratio[x] *= ratio[p]
		return root
	}
	for i, e := range equations {
		a, b := e[0], e[1]
		for _, x := range []string{a, b} {
			if _, ok := parent[x]; !ok {
				parent[x], ratio[x] = x, 1.0
			}
		}
		ra, rb := find(a), find(b)
		if ra != rb { // hang ra under rb: ra / rb = v * wb / wa
			parent[ra] = rb
			ratio[ra] = values[i] * ratio[b] / ratio[a]
		}
	}
	out := make([]float64, len(queries))
	for i, q := range queries {
		_, okC := parent[q[0]]
		_, okD := parent[q[1]]
		if !okC || !okD || find(q[0]) != find(q[1]) {
			out[i] = -1.0 // unknown variable or different groups
			continue
		}
		out[i] = ratio[q[0]] / ratio[q[1]]
	}
	return out
}

static class Solution {
    private final Map<String, String> parent = new HashMap<>();  // variable -> parent variable
    private final Map<String, Double> ratio = new HashMap<>();   // variable -> variable / parent

    public double[] calcEquation(List<List<String>> equations, double[] values, List<List<String>> queries) {
        for (int i = 0; i < equations.size(); i++) {
            String a = equations.get(i).get(0), b = equations.get(i).get(1);
            add(a);
            add(b);
            String ra = find(a), rb = find(b);       // after find, ratio holds x / root
            if (!ra.equals(rb)) {                    // hang ra under rb: ra / rb = v * wb / wa
                parent.put(ra, rb);
                ratio.put(ra, values[i] * ratio.get(b) / ratio.get(a));
            }
        }
        double[] out = new double[queries.size()];
        for (int i = 0; i < queries.size(); i++) {
            String c = queries.get(i).get(0), d = queries.get(i).get(1);
            if (!parent.containsKey(c) || !parent.containsKey(d)) { out[i] = -1.0; continue; }
            out[i] = find(c).equals(find(d)) ? ratio.get(c) / ratio.get(d) : -1.0;
        }
        return out;
    }

    private void add(String x) {
        if (!parent.containsKey(x)) { parent.put(x, x); ratio.put(x, 1.0); }
    }

    // returns the root; afterwards parent[x] = root and ratio[x] = x / root
    private String find(String x) {
        String p = parent.get(x);
        if (p.equals(x)) return x;
        String root = find(p);                       // now ratio[p] = p / root
        parent.put(x, root);
        ratio.put(x, ratio.get(x) * ratio.get(p));
        return root;
    }
}
class Solution:
    def calcEquation(self, equations: List[List[str]], values: List[float],
                     queries: List[List[str]]) -> List[float]:
        parent = {}   # variable -> parent variable
        ratio = {}    # variable -> variable / parent  (so a root has ratio 1)

        def find(x):
            """Return (root, x / root); compresses the path on the way back."""
            if parent[x] != x:
                root, up = find(parent[x])    # up = parent / root
                parent[x] = root
                ratio[x] *= up                # x/root = x/parent * parent/root
            return parent[x], ratio[x]

        for (a, b), v in zip(equations, values):
            for x in (a, b):
                if x not in parent:
                    parent[x], ratio[x] = x, 1.0
            ra, wa = find(a)                  # a = wa * ra
            rb, wb = find(b)                  # b = wb * rb
            if ra != rb:                      # hang ra under rb: ra / rb = v * wb / wa
                parent[ra] = rb
                ratio[ra] = v * wb / wa

        out = []
        for c, d in queries:
            if c not in parent or d not in parent:
                out.append(-1.0)              # unknown variable, even for x / x
                continue
            rc, wc = find(c)
            rd, wd = find(d)
            out.append(wc / wd if rc == rd else -1.0)   # different groups: no relation
        return out
