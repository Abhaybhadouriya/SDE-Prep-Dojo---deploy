The XOR fold in four languages. Each file in coding-code/single-number/ also builds 3,000 random arrays (values in the full ±3·104 range, shuffled) and checks the answer against a counting solution.

func singleNumber(nums []int) int {
	acc := 0
	for _, x := range nums {
		acc ^= x // x ^ x = 0 and x ^ 0 = x, so every pair cancels out
	}
	return acc // only the value without a partner is left
}
