Monotonic stack in four languages. Each file in coding-code/daily-temperatures/ also checks 3,000 random arrays (with many equal temperatures) against the O(n²) scan. The Java version uses a plain int[] as the stack to avoid Integer boxing on 105 elements.

func dailyTemperatures(temperatures []int) []int {
	n := len(temperatures)
	wait := make([]int, n)         // 0 stays for days that never see a warmer day
	pending := make([]int, 0, n)   // indices still waiting; temps strictly decrease bottom -> top
	for day, t := range temperatures {
		// Today is the first warmer day for every colder pending day on top.
		for len(pending) > 0 && temperatures[pending[len(pending)-1]] < t {
			prev := pending[len(pending)-1]
			pending = pending[:len(pending)-1]
			wait[prev] = day - prev
		}
		pending = append(pending, day) // today now waits for its own warmer day
	}
	return wait
}
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = temperatures.size();
        vector<int> wait(n, 0);        // 0 stays for days that never see a warmer day
        vector<int> pending;           // indices still waiting; temps strictly decrease bottom -> top
        pending.reserve(n);
        for (int day = 0; day < n; ++day) {
            // Today is the first warmer day for every colder pending day on top.
            while (!pending.empty() && temperatures[pending.back()] < temperatures[day]) {
                int prev = pending.back();
                pending.pop_back();
                wait[prev] = day - prev;
            }
            pending.push_back(day);    // today now waits for its own warmer day
        }
        return wait;
    }
};

static class Solution {
    public int[] dailyTemperatures(int[] temperatures) {
        int n = temperatures.length;
        int[] wait = new int[n];       // 0 stays for days that never see a warmer day
        int[] pending = new int[n];    // array used as a stack of indices: pending[0..top-1]
        int top = 0;                   // temps strictly decrease bottom -> top
        for (int day = 0; day < n; day++) {
            // Today is the first warmer day for every colder pending day on top.
            while (top > 0 && temperatures[pending[top - 1]] < temperatures[day]) {
                int prev = pending[--top];
                wait[prev] = day - prev;
            }
            pending[top++] = day;      // today now waits for its own warmer day
        }
        return wait;
    }
}
class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        n = len(temperatures)
        wait = [0] * n               # 0 stays for days that never see a warmer day
        pending: List[int] = []      # indices still waiting; their temps strictly decrease bottom -> top
        for day, t in enumerate(temperatures):
            # Today is the first warmer day for every colder pending day on top.
            while pending and temperatures[pending[-1]] < t:
                prev = pending.pop()
                wait[prev] = day - prev
            pending.append(day)      # today now waits for its own warmer day
        return wait
