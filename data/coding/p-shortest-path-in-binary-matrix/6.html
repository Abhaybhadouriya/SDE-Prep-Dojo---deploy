class Solution {
public:
    int shortestPathBinaryMatrix(vector<vector<int>>& grid) {
        int n = grid.size();
        if (grid[0][0] == 1 || grid[n - 1][n - 1] == 1) return -1;  // blocked start or exit
        static const int dr[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
        static const int dc[8] = {-1, 0, 1, -1, 1, -1, 0, 1};
        queue<pair<int, int>> q;
        q.push({0, 0});
        grid[0][0] = 1;                          // mark visited when PUSHED, so no cell is queued twice
        for (int len = 1; !q.empty(); len++) {   // len = cells on a path to this BFS level
            for (int k = q.size(); k > 0; k--) {
                int r = q.front().first, c = q.front().second;
                q.pop();
                if (r == n - 1 && c == n - 1) return len;
                for (int d = 0; d < 8; d++) {
                    int nr = r + dr[d], nc = c + dc[d];
                    if (nr < 0 || nc < 0 || nr >= n || nc >= n || grid[nr][nc] != 0) continue;
                    grid[nr][nc] = 1;
                    q.push({nr, nc});
                }
            }
        }
        return -1;                               // the queue ran dry: the exit is cut off
    }
};
func shortestPathBinaryMatrix(grid [][]int) int {
	n := len(grid)
	if grid[0][0] == 1 || grid[n-1][n-1] == 1 {
		return -1 // blocked start or exit
	}
	q := [][2]int{{0, 0}}
	grid[0][0] = 1                          // mark visited when PUSHED, so no cell is queued twice
	for length := 1; len(q) > 0; length++ { // length = cells on a path to this BFS level
		var next [][2]int
		for _, cell := range q {
			r, c := cell[0], cell[1]
			if r == n-1 && c == n-1 {
				return length
			}
			for dr := -1; dr <= 1; dr++ {
				for dc := -1; dc <= 1; dc++ {
					nr, nc := r+dr, c+dc
					if nr >= 0 && nc >= 0 && nr < n && nc < n && grid[nr][nc] == 0 {
						grid[nr][nc] = 1
						next = append(next, [2]int{nr, nc})
					}
				}
			}
		}
		q = next
	}
	return -1 // the queue ran dry: the exit is cut off
}
