class MinStack {
    // st[i] = {value, minimum of st[0..i]}: every entry remembers the answer
    // to getMin() for the stack exactly as it was when that entry was on top.
    vector<pair<int, int>> st;
public:
    MinStack() {}

    void push(int val) {
        int lowest = st.empty() ? val : min(val, st.back().second);
        st.push_back({val, lowest});
    }

    void pop() { st.pop_back(); }          // the entry below already knows its own min

    int top() { return st.back().first; }

    int getMin() { return st.back().second; }
};
type frame struct {
	val, low int // low = minimum of this frame and every frame below it
}

type MinStack struct {
	st []frame
}

func Constructor() MinStack {
	return MinStack{}
}

func (s *MinStack) Push(val int) {
	low := val
	if n := len(s.st); n > 0 && s.st[n-1].low < low {
		low = s.st[n-1].low
	}
	s.st = append(s.st, frame{val, low})
}

func (s *MinStack) Pop() {
	s.st = s.st[:len(s.st)-1] // the frame below already knows its own min
}

func (s *MinStack) Top() int {
	return s.st[len(s.st)-1].val
}

func (s *MinStack) GetMin() int {
	return s.st[len(s.st)-1].low
}

static class MinStack {
    // each frame is {value, minimum of this frame and everything below it}
    private final Deque<int[]> st = new ArrayDeque<>();

    public MinStack() {}

    public void push(int val) {
        int lowest = st.isEmpty() ? val : Math.min(val, st.peek()[1]);
        st.push(new int[]{val, lowest});
    }

    public void pop() { st.pop(); }      // the frame below already knows its own min

    public int top() { return st.peek()[0]; }

    public int getMin() { return st.peek()[1]; }
}
class MinStack:
    def __init__(self):
        # each entry is (value, minimum of this entry and everything below it)
        self.st = []

    def push(self, val: int) -> None:
        lowest = val if not self.st else min(val, self.st[-1][1])
        self.st.append((val, lowest))

    def pop(self) -> None:
        self.st.pop()                       # the entry below already knows its own min

    def top(self) -> int:
        return self.st[-1][0]

    def getMin(self) -> int:
        return self.st[-1][1]
