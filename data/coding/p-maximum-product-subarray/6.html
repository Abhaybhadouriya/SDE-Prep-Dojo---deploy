Max/min ending here in four languages. Each file in coding-code/maximum-product-subarray/ checks the examples and edge cases, 2,000 random short arrays against the O(n²) brute force, and a 2·104-element array. C++ and Java keep the running products in 64-bit to stay clear of signed overflow.

func maxProduct(nums []int) int {
	// hi / lo = largest / smallest product of a subarray that ENDS at index i.
	// We need lo because a negative number turns the smallest product into the largest.
	// Go's int is 64-bit on LeetCode, so the running products cannot overflow here.
	hi, lo, best := nums[0], nums[0], nums[0]
	for _, x := range nums[1:] {
		if x < 0 {
			hi, lo = lo, hi // multiplying by a negative flips the order
		}
		hi = max(x, hi*x) // extend the best run, or start fresh at x
		lo = min(x, lo*x) // same for the most negative run
		best = max(best, hi)
	}
	return best
}

    int maxProduct(vector<int>& nums) {
        // hi / lo = largest / smallest product of a subarray that ENDS at index i.
        // We need lo because a negative number turns the smallest product into the largest.
        // 64-bit for the running products: cheap insurance against signed overflow (UB in C++).
        long long hi = nums[0], lo = nums[0], best = nums[0];
        for (size_t i = 1; i < nums.size(); i++) {
            long long x = nums[i];
            if (x < 0) swap(hi, lo);          // multiplying by a negative flips the order
            hi = max(x, hi * x);              // extend the best run, or start fresh at x
            lo = min(x, lo * x);              // same for the most negative run
            best = max(best, hi);
        }
        return (int)best;
    }
