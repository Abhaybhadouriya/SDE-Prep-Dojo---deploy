func addTwoNumbers(a *ListNode, b *ListNode) *ListNode {
	dummy := &ListNode{}
	tail := dummy
	carry := 0
	// keep going while either list has digits OR a carry is still pending
	for a != nil || b != nil || carry != 0 {
		sum := carry
		if a != nil {
			sum += a.Val
			a = a.Next
		}
		if b != nil {
			sum += b.Val
			b = b.Next
		}
		carry = sum / 10 // at most 19, so carry is 0 or 1
		tail.Next = &ListNode{Val: sum % 10}
		tail = tail.Next
	}
	return dummy.Next
}

def addTwoNumbers(self, a, b, carry=0):
    if not a and not b and not carry:
        return None
    total = carry + (a.val if a else 0) + (b.val if b else 0)
    node = ListNode(total % 10)
    node.next = self.addTwoNumbers(a.next if a else None, b.next if b else None, total // 10)
    return node
