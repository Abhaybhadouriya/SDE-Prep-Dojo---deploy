Two templates the companies love

One signature problem each from the two companies with the most reports. Both are short, have traps, and come back as variants.

Meta: minimum removals to balance brackets

The problem, in our words. A string s holds lowercase letters and the characters ( and ). Delete as few brackets as possible (at any positions) so that the result is a valid bracket string: every ( has a later matching ) and vice versa; letters do not matter. Return any valid result.

  • 1 <= s.length <= 105; s[i] is (, ) or a lowercase English letter.
  • Example: s = "a)b(c(d)" -> "abc(d)". Example: s = "))((" -> "" (empty). Example: s = "x(y(z)" -> "xy(z)" or "x(yz)".

Original: LeetCode 1249 - Minimum Remove to Make Valid Parentheses · community solutions · our write-up: Minimum Remove.

Keep a stack of indices of unmatched (. A ) either pops a partner or is unmatched and must go. At the end, every index still on the stack is an unmatched ( and must go too.

  1. 'a' is kept. ')' at index 1 finds an empty stack: nothing to its left can match it, so it is removed (red).
  2. 'b' is kept. '(' at index 3 is pushed: it may find a partner later.
  3. 'c' is kept. '(' at index 5 is pushed on top of 3.
  4. 'd' is kept. ')' at 7 pops the most recent open bracket, 5. The pair (5, 7) is matched (green).
  5. The scan ends with 3 on the stack: no ')' ever came for it, so it is removed. Two removals, which is the minimum.

Why it is minimal: every removed ')' had no '(' left of it, and every removed '(' had no ')' right of it, so any valid result must delete them too. O(n) time, O(n) space. The Meta follow-up "do it with O(1) extra space" uses two passes with a counter: left to right drop unmatched ')', then right to left drop surplus '('.

Amazon: multi-source BFS

The problem, in our words. A grid of m x n cells holds 0 (empty), 1 (fresh fruit) or 2 (rotten fruit). Every minute, each fresh fruit next to a rotten one (up, down, left, right) rots. Return the minutes until no fresh fruit is left, or -1 if some fruit can never rot.

  • 1 <= m, n <= 10; grid[i][j] is 0, 1 or 2.
  • Example: [[2,1,1],[1,1,0],[0,1,1]] -> 4. Example: [[2,1,1],[0,1,1],[1,0,1]] -> -1 (the bottom-left fruit is cut off). Example: [[0,2]] -> 0.

Original: LeetCode 994 - Rotting Oranges · community solutions · our write-up: Rotting Oranges.

Put all rotten cells in the queue at time 0, then run BFS layer by layer; each layer is one minute. Count fresh cells up front so you can answer -1 without a second scan. The same skeleton solves Walls and Gates, 01 Matrix and As Far from Land as Possible.

The planner: top-k across company lists

The third function sums each problem's frequency across your target companies and keeps the top k in a size-k min-heap: O(N log k), the plan below by hand. It is Top K Frequent Elements with weights.

What is in each file (same names in every language; Go and Java use camelCase):

  • min_remove_to_make_valid - Meta favourite: delete the fewest brackets so s is balanced.
  • oranges_rotting - LC 994: multi-source BFS, minutes until no fresh orange is left, or -1.
  • plan_from_companies - Merge company lists [(slug, freq)] into the top-k slugs by total frequency.

package main

import (
	"container/heap"
	"fmt"
	"strings"
)

// Meta favourite: delete the fewest brackets so s is balanced.
func minRemoveToMakeValid(s string) string {
	b := []byte(s) // a copy: the caller's string is untouched
	open := []int{} // stack of indices of unmatched '('
	for i, c := range b {
		if c == '(' {
			open = append(open, i)
		} else if c == ')' {
			if len(open) > 0 {
				open = open[:len(open)-1] // matched
			} else {
				b[i] = '#' // unmatched ')': mark for deletion
			}
		}
	}
	for _, i := range open { // leftover '(' have no partner
		b[i] = '#'
	}
	return strings.ReplaceAll(string(b), "#", "")
}

// Amazon favourite: multi-source BFS; minutes until all fresh rot, -1 if never.
func orangesRotting(grid [][]int) int {
	R, C, fresh := len(grid), len(grid[0]), 0
	g := make([][]int, R) // copy: do not mutate the input
	q := [][2]int{}
	for r := range grid {
		g[r] = append([]int(nil), grid[r]...)
		for c, v := range g[r] {
			if v == 2 {
				q = append(q, [2]int{r, c})
			} else if v == 1 {
				fresh++
			}
		}
	}
	dirs := [4][2]int{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
	minutes := 0
	for len(q) > 0 && fresh > 0 {
		next := [][2]int{} // one BFS layer = one minute
		for _, cur := range q {
			for _, d := range dirs {
				nr, nc := cur[0]+d[0], cur[1]+d[1]
				if nr >= 0 && nr < R && nc >= 0 && nc < C && g[nr][nc] == 1 {
					g[nr][nc] = 2
					fresh--
					next = append(next, [2]int{nr, nc})
				}
			}
		}
		q = next
		minutes++
	}
	if fresh > 0 {
		return -1
	}
	return minutes
}

type entry struct {
	slug  string
	score int
}

// minHeap keeps the weakest kept entry (lowest score, then largest name) on top.
type minHeap []entry

func (h minHeap) Len() int { return len(h) }
func (h minHeap) Less(i, j int) bool {
	if h[i].score != h[j].score {
		return h[i].score < h[j].score
	}
	return h[i].slug > h[j].slug
}
func (h minHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x interface{}) { *h = append(*h, x.(entry)) }
func (h *minHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

type row struct {
	slug string
	freq int
}

// Merge company lists of (slug, freq) into the top-k slugs by total frequency.
func planFromCompanies(lists map[string][]row, k int) []string {
	score := map[string]int{}
	for _, rows := range lists {
		for _, r := range rows {
			score[r.slug] += r.freq
		}
	}
	h := &minHeap{}
	for s, v := range score {
		heap.Push(h, entry{s, v})
		if h.Len() > k {
			heap.Pop(h)
		}
	}
	out := make([]string, h.Len())
	for i := len(out) - 1; i >= 0; i-- {
		out[i] = heap.Pop(h).(entry).slug
	}
	return out
}

// Meta favourite: delete the fewest brackets so s is balanced.
string minRemoveToMakeValid(string s) {            // by value: the caller's string is untouched
    vector<int> open;                               // stack of indices of unmatched '('
    for (int i = 0; i < (int)s.size(); i++) {
        if (s[i] == '(') open.push_back(i);
        else if (s[i] == ')') {
            if (!open.empty()) open.pop_back();     // matched
            else s[i] = '#';                        // unmatched ')': mark for deletion
        }
    }
    for (int i : open) s[i] = '#';                  // leftover '(' have no partner
    s.erase(remove(s.begin(), s.end(), '#'), s.end());
    return s;
}

// Amazon favourite: multi-source BFS; minutes until all fresh rot, -1 if never.
int orangesRotting(vector<vector<int>> g) {         // by value: we may mark cells
    int R = g.size(), C = g[0].size(), fresh = 0;
    queue<pair<int, int>> q;
    for (int r = 0; r < R; r++)
        for (int c = 0; c < C; c++) {
            if (g[r][c] == 2) q.push({r, c});
            else if (g[r][c] == 1) fresh++;
        }
    const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    int minutes = 0;
    while (!q.empty() && fresh > 0) {
        for (int n = q.size(); n > 0; n--) {        // one BFS layer = one minute
            pair<int, int> cur = q.front(); q.pop();
            for (int d = 0; d < 4; d++) {
                int nr = cur.first + dr[d], nc = cur.second + dc[d];
                if (nr >= 0 && nr < R && nc >= 0 && nc < C && g[nr][nc] == 1) {
                    g[nr][nc] = 2; fresh--; q.push({nr, nc});
                }
            }
        }
        minutes++;
    }
    return fresh == 0 ? minutes : -1;
}

// Merge company lists of (slug, freq) into the top-k slugs by total frequency.
vector<string> planFromCompanies(const map<string, vector<pair<string, int>>>& lists, int k) {
    unordered_map<string, int> score;
    for (const auto& co : lists)
        for (const auto& row : co.second) score[row.first] += row.second;
    // min-heap of size k on (score, -name): the weakest kept entry sits on top
    auto worse = [](const pair<int, string>& a, const pair<int, string>& b) {
        return a.first != b.first ? a.first > b.first : a.second < b.second;
    };
    priority_queue<pair<int, string>, vector<pair<int, string>>, decltype(worse)> heap(worse);
    for (const auto& e : score) {
        heap.push({e.second, e.first});
        if ((int)heap.size() > k) heap.pop();
    }
    vector<string> out;
    while (!heap.empty()) { out.push_back(heap.top().second); heap.pop(); }
    reverse(out.begin(), out.end());
    return out;
}

    puts("ok");
}
import java.util.*;

public class Main {
    // Meta favourite: delete the fewest brackets so s is balanced.
    static String minRemoveToMakeValid(String s) {
        char[] a = s.toCharArray();                 // a copy: Java strings are immutable anyway
        Deque<Integer> open = new ArrayDeque<>();   // stack of indices of unmatched '('
        for (int i = 0; i < a.length; i++) {
            if (a[i] == '(') open.push(i);
            else if (a[i] == ')') {
                if (!open.isEmpty()) open.pop();    // matched
                else a[i] = '#';                    // unmatched ')': mark for deletion
            }
        }
        for (int i : open) a[i] = '#';              // leftover '(' have no partner
        StringBuilder sb = new StringBuilder();
        for (char c : a) if (c != '#') sb.append(c);
        return sb.toString();
    }

    // Amazon favourite: multi-source BFS; minutes until all fresh rot, -1 if never.
    static int orangesRotting(int[][] grid) {
        int R = grid.length, C = grid[0].length, fresh = 0;
        int[][] g = new int[R][];
        for (int r = 0; r < R; r++) g[r] = grid[r].clone();   // copy: do not mutate the input
        ArrayDeque<int[]> q = new ArrayDeque<>();
        for (int r = 0; r < R; r++)
            for (int c = 0; c < C; c++) {
                if (g[r][c] == 2) q.add(new int[]{r, c});
                else if (g[r][c] == 1) fresh++;
            }
        int[] dr = {1, -1, 0, 0}, dc = {0, 0, 1, -1};
        int minutes = 0;
        while (!q.isEmpty() && fresh > 0) {
            for (int n = q.size(); n > 0; n--) {    // one BFS layer = one minute
                int[] cur = q.poll();
                for (int d = 0; d < 4; d++) {
                    int nr = cur[0] + dr[d], nc = cur[1] + dc[d];
                    if (nr >= 0 && nr < R && nc >= 0 && nc < C && g[nr][nc] == 1) {
                        g[nr][nc] = 2; fresh--; q.add(new int[]{nr, nc});
                    }
                }
            }
            minutes++;
        }
        return fresh == 0 ? minutes : -1;
    }

    // Merge company lists of (slug, freq) into the top-k slugs by total frequency.
    static List<String> planFromCompanies(Map<String, List<Map.Entry<String, Integer>>> lists, int k) {
        Map<String, Integer> score = new HashMap<>();
        for (List<Map.Entry<String, Integer>> rows : lists.values())
            for (Map.Entry<String, Integer> e : rows) score.merge(e.getKey(), e.getValue(), Integer::sum);
        // min-heap of size k: the weakest kept entry (lowest score, then largest name) on top
        PriorityQueue<String> heap = new PriorityQueue<>((x, y) -> {
            int c = Integer.compare(score.get(x), score.get(y));
            return c != 0 ? c : y.compareTo(x);
        });
        for (String s : score.keySet()) {
            heap.add(s);
            if (heap.size() > k) heap.poll();
        }
        LinkedList<String> out = new LinkedList<>();
        while (!heap.isEmpty()) out.addFirst(heap.poll());
        return out;
    }

    static void check(boolean ok, String what) { if (!ok) throw new AssertionError(what); }

    public static void main(String[] args) {
        check(minRemoveToMakeValid("ab)c(d)").equals("abc(d)"), "mr1");
        check(minRemoveToMakeValid("))((").equals(""), "mr2");
        check(minRemoveToMakeValid("x(y(z)").equals("xy(z)"), "mr3");
        check(minRemoveToMakeValid("(a)(b)").equals("(a)(b)"), "mr4");
        int[][] g1 = {{2, 1, 1}, {1, 1, 0}, {0, 1, 1}};
        check(orangesRotting(g1) == 4, "or1");
        check(g1[0][1] == 1, "or-copy");
        check(orangesRotting(new int[][]{{2, 1, 1}, {0, 1, 1}, {1, 0, 1}}) == -1, "or2");
        check(orangesRotting(new int[][]{{0, 2}}) == 0, "or3");
        Map<String, List<Map.Entry<String, Integer>>> lists = new HashMap<>();
        lists.put("meta", List.of(Map.entry("lru", 6), Map.entry("islands", 5), Map.entry("minrem", 8)));
        lists.put("amazon", List.of(Map.entry("lru", 7), Map.entry("islands", 7), Map.entry("oranges", 6)));
        lists.put("google", List.of(Map.entry("islands", 5), Map.entry("lru", 5)));
        check(planFromCompanies(lists, 3).equals(List.of("lru", "islands", "minrem")), "plan");
        System.out.println("ok");
    }
}
from collections import deque, defaultdict
import heapq


def min_remove_to_make_valid(s: str) -> str:
    """Meta favourite: delete the fewest brackets so s is balanced."""
    chars = list(s)          # work on a copy, never on the caller's data
    open_idx = []            # stack of indices of unmatched '('
    for i, c in enumerate(chars):
        if c == '(':
            open_idx.append(i)
        elif c == ')':
            if open_idx:
                open_idx.pop()       # matched
            else:
                chars[i] = ''        # unmatched ')': drop it
    for i in open_idx:               # leftover '(' have no partner
        chars[i] = ''
    return ''.join(chars)


def oranges_rotting(grid):
    """Amazon favourite: multi-source BFS, minutes until all fresh rot (-1 if never)."""
    rows, cols = len(grid), len(grid[0])
    g = [row[:] for row in grid]     # copy: do not mutate the input
    q = deque()
    fresh = 0
    for r in range(rows):
        for c in range(cols):
            if g[r][c] == 2:
                q.append((r, c))
            elif g[r][c] == 1:
                fresh += 1
    minutes = 0
    while q and fresh:
        for _ in range(len(q)):      # one BFS layer = one minute
            r, c = q.popleft()
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if 0 <= nr < rows and 0 <= nc < cols and g[nr][nc] == 1:
                    g[nr][nc] = 2
                    fresh -= 1
                    q.append((nr, nc))
        minutes += 1
    return minutes if fresh == 0 else -1


def plan_from_companies(lists, k):
    """Merge company lists [(slug, freq)] into the top-k slugs by total frequency."""
    score = defaultdict(int)
    for rows in lists.values():
        for slug, freq in rows:
            score[slug] += freq
    # nsmallest on (-score, slug): highest score first, ties by name
    return [s for _, s in heapq.nsmallest(k, ((-v, s) for s, v in score.items()))]
