Iterators: do the work lazily

An iterator design (next(), hasNext()) is judged on memory. Flattening everything into a list in the constructor works but costs O(n) memory and does all the work up front. The better answer keeps only the frontier: a stack of "where I am" and expands it one step per call.

  • BST Iterator (173): a stack of the left spine. next() pops a node, then pushes the left spine of its right child. O(h) memory, amortized O(1) per call (each node is pushed and popped once).
  • Flatten Nested List Iterator (341): a stack of (list, position) frames. hasNext() does the expanding so that an empty nested list never produces a false "yes".
  • Peeking Iterator (284) and Zigzag Iterator (281): cache one element ahead, or keep a queue of iterators and rotate.
// 6. Lazy BST iterator (LC 173).
type TreeNode struct {
	Val         int
	Left, Right *TreeNode
}

type BSTIterator struct{ st []*TreeNode }

func NewBSTIterator(root *TreeNode) *BSTIterator {
	it := &BSTIterator{}
	it.pushLeft(root)
	return it
}
func (it *BSTIterator) pushLeft(n *TreeNode) {
	for ; n != nil; n = n.Left {
		it.st = append(it.st, n)
	}
}
func (it *BSTIterator) HasNext() bool { return len(it.st) > 0 }
func (it *BSTIterator) Next() int {
	n := it.st[len(it.st)-1]
	it.st = it.st[:len(it.st)-1]
	it.pushLeft(n.Right)
	return n.Val
}
// 6. Lazy BST iterator (LC 173): stack holds the left spine, O(h) memory.
struct TreeNode { int val; TreeNode *left, *right; TreeNode(int v, TreeNode* l = nullptr, TreeNode* r = nullptr) : val(v), left(l), right(r) {} };
class BSTIterator {
    vector<TreeNode*> st;
    void pushLeft(TreeNode* n) { for (; n; n = n->left) st.push_back(n); }
public:
    explicit BSTIterator(TreeNode* root) { pushLeft(root); }
    bool hasNext() { return !st.empty(); }
    int next() { TreeNode* n = st.back(); st.pop_back(); pushLeft(n->right); return n->val; }
};
// 6. Lazy BST iterator (LC 173).
static class TreeNode { int val; TreeNode left, right; TreeNode(int v, TreeNode l, TreeNode r) { val = v; left = l; right = r; } }
static class BSTIterator {
    final Deque<TreeNode> st = new ArrayDeque<>();
    BSTIterator(TreeNode root) { pushLeft(root); }
    void pushLeft(TreeNode n) { for (; n != null; n = n.left) st.push(n); }
    boolean hasNext() { return !st.isEmpty(); }
    int next() { TreeNode n = st.pop(); pushLeft(n.right); return n.val; }
}
# 6. Lazy BST iterator (LC 173).
class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right


class BSTIterator:
    def __init__(self, root):
        self.st = []
        self._push_left(root)

    def _push_left(self, n):
        while n:
            self.st.append(n)
            n = n.left

    def hasNext(self):
        return bool(self.st)

    def next(self):
        n = self.st.pop()
        self._push_left(n.right)
        return n.val

Interviewers love this: "Put the work in hasNext(), make it idempotent, and let next() call it." Then calling hasNext() twice, or next() without hasNext(), both stay correct.
