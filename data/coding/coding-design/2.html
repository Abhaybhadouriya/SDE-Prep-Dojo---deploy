LRU cache: hash map + doubly linked list

The task (LeetCode 146): a cache with a fixed capacity, get(key) returns the value or -1, put(key, value) inserts or updates, and when the cache is full the least recently used key is evicted. Both calls must be O(1) on average. Using a key, by get or put, makes it the most recent.

  1. put(1, A). The map stores key 1 pointing to its list node; the node goes right after the head sentinel.
  2. put(2, B). New items always enter at the head, so the list reads newest to oldest.
  3. put(3, C). Three items, capacity reached. The oldest (key 1) sits just before tail.
  4. get(1). The map finds the node in O(1); unlink it (two pointer writes) and insert it after head. Key 2 is now the oldest.
  5. put(4, D) on a full cache. Evict tail.prev: unlink it, and use the key stored inside the node to delete it from the map.
  6. get(2). Miss, return -1. A miss changes nothing.
  7. put(3, Z). The key exists: overwrite the value and move it to the head. No eviction, the size did not change.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

// 1. LRU cache (LC 146): map key -> *list.Element, front = most recent.
type entry struct{ key, val int }

type LRUCache struct {
	cap   int
	order *list.List
	pos   map[int]*list.Element
}

func NewLRUCache(capacity int) *LRUCache {
	return &LRUCache{cap: capacity, order: list.New(), pos: map[int]*list.Element{}}
}

func (c *LRUCache) Get(key int) int {
	e, ok := c.pos[key]
	if !ok {
		return -1
	}
	c.order.MoveToFront(e)
	return e.Value.(*entry).val
}

func (c *LRUCache) Put(key, value int) {
	if e, ok := c.pos[key]; ok {
		e.Value.(*entry).val = value
		c.order.MoveToFront(e)
		return
	}
	if c.order.Len() == c.cap {
		old := c.order.Back()
		c.order.Remove(old)
		delete(c.pos, old.Value.(*entry).key)
	}
	c.pos[key] = c.order.PushFront(&entry{key, value})
}
// 1. LRU cache (LC 146): hash map key -> list node, list front = most recent.
class LRUCache {
    int cap;
    list<pair<int, int>> order;                                   // (key, value), front = newest
    unordered_map<int, list<pair<int, int>>::iterator> pos;       // key -> node in the list
public:
    explicit LRUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        auto it = pos.find(key);
        if (it == pos.end()) return -1;
        order.splice(order.begin(), order, it->second);            // O(1) move to front
        return it->second->second;
    }
    void put(int key, int value) {
        auto it = pos.find(key);
        if (it != pos.end()) {
            it->second->second = value;
            order.splice(order.begin(), order, it->second);
            return;
        }
        if ((int)order.size() == cap) {                            // evict the tail
            pos.erase(order.back().first);
            order.pop_back();
        }
        order.emplace_front(key, value);
        pos[key] = order.begin();
    }
};
// 1. LRU cache (LC 146): HashMap + doubly linked list with sentinels.
static class LRUCache {
    static class Node { int key, val; Node prev, next; Node(int k, int v) { key = k; val = v; } }
    final int cap;
    final Map<Integer, Node> pos = new HashMap<>();
    final Node head = new Node(0, 0), tail = new Node(0, 0);   // head.next = newest
    LRUCache(int capacity) { cap = capacity; head.next = tail; tail.prev = head; }
    void unlink(Node n) { n.prev.next = n.next; n.next.prev = n.prev; }
    void pushFront(Node n) { n.prev = head; n.next = head.next; head.next.prev = n; head.next = n; }
    int get(int key) {
        Node n = pos.get(key);
        if (n == null) return -1;
        unlink(n); pushFront(n);
        return n.val;
    }
    void put(int key, int value) {
        Node n = pos.get(key);
        if (n != null) { n.val = value; unlink(n); pushFront(n); return; }
        if (pos.size() == cap) { Node old = tail.prev; unlink(old); pos.remove(old.key); }
        n = new Node(key, value);
        pos.put(key, n);
        pushFront(n);
    }
}
# 1. LRU cache (LC 146): explicit hash map + doubly linked list with sentinels.
class Node:
    __slots__ = ("key", "val", "prev", "next")

    def __init__(self, key=0, val=0):
        self.key, self.val, self.prev, self.next = key, val, None, None


class LRUCache:
    def __init__(self, capacity):
        self.cap, self.pos = capacity, {}
        self.head, self.tail = Node(), Node()      # sentinels: head.next = newest
        self.head.next, self.tail.prev = self.tail, self.head

    def _unlink(self, n):
        n.prev.next, n.next.prev = n.next, n.prev

    def _push_front(self, n):
        n.prev, n.next = self.head, self.head.next
        self.head.next.prev = n
        self.head.next = n

    def get(self, key):
        n = self.pos.get(key)
        if not n:
            return -1
        self._unlink(n)
        self._push_front(n)
        return n.val

    def put(self, key, value):
        n = self.pos.get(key)
        if n:
            n.val = value
            self._unlink(n)
            self._push_front(n)
            return
        if len(self.pos) == self.cap:
            old = self.tail.prev                    # least recently used
            self._unlink(old)
            del self.pos[old.key]                   # this is why the node stores its key
        n = Node(key, value)
        self.pos[key] = n
        self._push_front(n)

Go and C++ use the standard list (container/list, std::list::splice); Java and Python show the hand-written list, because interviewers often say "no LinkedHashMap, no OrderedDict". Know the shortcuts anyway: LinkedHashMap in access order with removeEldestEntry, or OrderedDict.move_to_end with popitem(last=False).
