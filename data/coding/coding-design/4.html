Augmented stacks and queues

Min stack: remember the answer at every depth

LeetCode 155 wants push, pop, top and getMin in O(1). The insight: a stack only changes at the top, so the minimum of everything below each item never changes while that item is there. Store it with the item: push (x, min(x, previous min)). Max Stack (716, Hard) adds popMax; that needs a second structure (a sorted map or heap with lazy deletion, or a doubly linked list plus a TreeMap from value to nodes), O(log n) per call.

Queue from two stacks: amortized O(1)

Push onto in. Pop from out. Only when out is empty, pour all of in into out, which reverses the order. Each item is pushed and popped at most twice in its life, so any n operations cost O(n).

  1. Three pushes land on stack in. The oldest item (1) is at the bottom, where a stack cannot reach it.
  2. First pop(): out is empty, so pop everything from in and push it onto out. The order flips: 1 is now on top.
  3. Pop from out: returns 1. Queue order is correct.
  4. push(4) goes to in. We must not pour now: 4 would land above 2 and 3, which are older.
  5. pop() returns 2 straight from out. Pour only when out is empty.
  6. pop() returns 3. out is empty again.
  7. Next pop() pours 4 across and returns it. Each item is moved at most once, so n operations cost O(n) total: amortized O(1).
// 3a. Min stack (LC 155).
type MinStack struct{ st [][2]int } // {value, min so far}

func (s *MinStack) Push(x int) {
	m := x
	if len(s.st) > 0 && s.st[len(s.st)-1][1] < m {
		m = s.st[len(s.st)-1][1]
	}
	s.st = append(s.st, [2]int{x, m})
}
func (s *MinStack) Pop()        { s.st = s.st[:len(s.st)-1] }
func (s *MinStack) Top() int    { return s.st[len(s.st)-1][0] }
func (s *MinStack) GetMin() int { return s.st[len(s.st)-1][1] }

// 3b. Queue from two stacks (LC 232).
type MyQueue struct{ in, out []int }

func (q *MyQueue) shift() {
	if len(q.out) == 0 {
		for len(q.in) > 0 {
			q.out = append(q.out, q.in[len(q.in)-1])
			q.in = q.in[:len(q.in)-1]
		}
	}
}
func (q *MyQueue) Push(x int) { q.in = append(q.in, x) }
func (q *MyQueue) Pop() int {
	q.shift()
	v := q.out[len(q.out)-1]
	q.out = q.out[:len(q.out)-1]
	return v
}
func (q *MyQueue) Peek() int   { q.shift(); return q.out[len(q.out)-1] }
func (q *MyQueue) Empty() bool { return len(q.in) == 0 && len(q.out) == 0 }
// 3a. Min stack (LC 155): each entry remembers the min below it.
class MinStack {
    vector<pair<int, int>> st;                                     // (value, min so far)
public:
    void push(int x) { st.push_back({x, st.empty() ? x : min(x, st.back().second)}); }
    void pop() { st.pop_back(); }
    int top() { return st.back().first; }
    int getMin() { return st.back().second; }
};

// 3b. Queue from two stacks (LC 232): amortized O(1).
class MyQueue {
    vector<int> in, out;
    void shift() { if (out.empty()) while (!in.empty()) { out.push_back(in.back()); in.pop_back(); } }
public:
    void push(int x) { in.push_back(x); }
    int pop() { shift(); int v = out.back(); out.pop_back(); return v; }
    int peek() { shift(); return out.back(); }
    bool empty() { return in.empty() && out.empty(); }
};
// 3a. Min stack (LC 155).
static class MinStack {
    final Deque<int[]> st = new ArrayDeque<>();                // {value, min so far}
    void push(int x) { st.push(new int[]{x, st.isEmpty() ? x : Math.min(x, st.peek()[1])}); }
    void pop() { st.pop(); }
    int top() { return st.peek()[0]; }
    int getMin() { return st.peek()[1]; }
}

// 3b. Queue from two stacks (LC 232).
static class MyQueue {
    final Deque<Integer> in = new ArrayDeque<>(), out = new ArrayDeque<>();
    void shift() { if (out.isEmpty()) while (!in.isEmpty()) out.push(in.pop()); }
    void push(int x) { in.push(x); }
    int pop() { shift(); return out.pop(); }
    int peek() { shift(); return out.peek(); }
    boolean empty() { return in.isEmpty() && out.isEmpty(); }
}
# 3a. Min stack (LC 155).
class MinStack:
    def __init__(self):
        self.st = []                                # (value, min so far)

    def push(self, x):
        self.st.append((x, min(x, self.st[-1][1]) if self.st else x))

    def pop(self):
        self.st.pop()

    def top(self):
        return self.st[-1][0]

    def getMin(self):
        return self.st[-1][1]

# 3b. Queue from two stacks (LC 232).
class MyQueue:
    def __init__(self):
        self.inb, self.out = [], []

    def _shift(self):
        if not self.out:                            # only when out is empty
            while self.inb:
                self.out.append(self.inb.pop())

    def push(self, x):
        self.inb.append(x)

    def pop(self):
        self._shift()
        return self.out.pop()

    def peek(self):
        self._shift()
        return self.out[-1]

    def empty(self):
        return not self.inb and not self.out

Circular queue: a ring buffer

Design Circular Queue (622) is a fixed array plus a head index and a count: the tail is (head + count) % k. Keeping a count avoids the classic "is full the same as empty?" ambiguity of two indexes.

From your world

This is a NIC descriptor ring or kfifo. kfifo makes the size a power of two so % k becomes & (k - 1), and lets in and out run freely as unsigned counters; in - out is the fill level and wraps correctly. Mention that trick when the interviewer asks how to make it faster.
