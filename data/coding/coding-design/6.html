Your own hash map

Design HashMap (706) and Design HashSet (705) are Easy on LeetCode because keys are small, but interviewers use them to test the real thing: an array of buckets, a hash function, collision handling and resizing.

// 5. Own hash map (LC 706): chaining, resize at load factor 0.75.
type kv struct{ k, v int }

type MyHashMap struct {
	b [][]kv
	n int
}

func NewMyHashMap() *MyHashMap { return &MyHashMap{b: make([][]kv, 16)} }

// slot keeps the HIGH bits of the 32-bit product (like the kernel's hash_32); m is a power of two.
// The low bits of k*odd depend only on the low bits of k, so a plain % m would send every
// multiple of m to bucket 0.
func slot(k, m int) int {
	sh := uint(32)
	for x := m; x > 1; x >>= 1 {
		sh--
	}
	return int((uint32(k) * 2654435761) >> sh)
}

func (h *MyHashMap) Put(k, v int) {
	s := slot(k, len(h.b))
	for i := range h.b[s] {
		if h.b[s][i].k == k {
			h.b[s][i].v = v
			return
		}
	}
	h.b[s] = append(h.b[s], kv{k, v})
	h.n++
	if h.n*4 > len(h.b)*3 {
		nb := make([][]kv, len(h.b)*2)
		for _, chain := range h.b {
			for _, e := range chain {
				t := slot(e.k, len(nb))
				nb[t] = append(nb[t], e)
			}
		}
		h.b = nb
	}
}

func (h *MyHashMap) Get(k int) int {
	for _, e := range h.b[slot(k, len(h.b))] {
		if e.k == k {
			return e.v
		}
	}
	return -1
}

func (h *MyHashMap) Remove(k int) {
	s := slot(k, len(h.b))
	for i, e := range h.b[s] {
		if e.k == k {
			h.b[s] = append(h.b[s][:i], h.b[s][i+1:]...)
			h.n--
			return
		}
	}
}
// 5. Own hash map (LC 706): chaining + resize at load factor 0.75.
class MyHashMap {
    vector<list<pair<int, int>>> b;
    size_t n = 0;
    // Keep the HIGH bits of the 32-bit product (like the kernel's hash_32); m is a power of two.
    // The low bits of k*odd depend only on the low bits of k, so % m would send every multiple of m to bucket 0.
    size_t slot(int k, size_t m) const {
        unsigned sh = 32;
        for (size_t x = m; x > 1; x >>= 1) --sh;
        return (size_t)(((unsigned)k * 2654435761u) >> sh);
    }
    void grow() {
        vector<list<pair<int, int>>> nb(b.size() * 2);
        for (auto& l : b) for (auto& kv : l) nb[slot(kv.first, nb.size())].push_back(kv);
        b.swap(nb);
    }
public:
    MyHashMap() : b(16) {}
    size_t longestChain() const { size_t m = 0; for (auto& l : b) m = max(m, l.size()); return m; }
    void put(int k, int v) {
        for (auto& kv : b[slot(k, b.size())]) if (kv.first == k) { kv.second = v; return; }
        b[slot(k, b.size())].push_back({k, v});
        if (++n * 4 > b.size() * 3) grow();
    }
    int get(int k) {
        for (auto& kv : b[slot(k, b.size())]) if (kv.first == k) return kv.second;
        return -1;
    }
    void remove(int k) {
        auto& l = b[slot(k, b.size())];
        for (auto it = l.begin(); it != l.end(); ++it) if (it->first == k) { l.erase(it); n--; return; }
    }
};
// 5. Own hash map (LC 706): chaining, resize at load factor 0.75.
static class MyHashMap {
    static class E { final int k; int v; E next; E(int k, int v, E next) { this.k = k; this.v = v; this.next = next; } }
    E[] b = new E[16];
    int n = 0;
    // Keep the HIGH bits of the 32-bit product (like the kernel's hash_32); m is a power of two.
    // The low bits of k*odd depend only on the low bits of k, so a plain mod m would send every multiple of m to bucket 0.
    static int slot(int k, int m) { return (k * 0x9E3779B1) >>> (32 - Integer.numberOfTrailingZeros(m)); }
    void put(int k, int v) {
        for (E e = b[slot(k, b.length)]; e != null; e = e.next) if (e.k == k) { e.v = v; return; }
        int s = slot(k, b.length);
        b[s] = new E(k, v, b[s]);                                // push at chain head
        if (++n * 4 > b.length * 3) grow();
    }
    void grow() {
        E[] nb = new E[b.length * 2];
        for (E head : b) for (E e = head; e != null; e = e.next) { int s = slot(e.k, nb.length); nb[s] = new E(e.k, e.v, nb[s]); }
        b = nb;
    }
    int get(int k) {
        for (E e = b[slot(k, b.length)]; e != null; e = e.next) if (e.k == k) return e.v;
        return -1;
    }
    void remove(int k) {
        int s = slot(k, b.length);
        for (E prev = null, e = b[s]; e != null; prev = e, e = e.next)
            if (e.k == k) { if (prev == null) b[s] = e.next; else prev.next = e.next; n--; return; }
    }
}
# 5. Own hash map (LC 706): chaining, resize at load factor 0.75.
class MyHashMap:
    def __init__(self):
        self.b, self.n = [[] for _ in range(16)], 0

    def _slot(self, k):
        # Multiplicative hash: keep the HIGH bits of the 32-bit product (like the kernel's hash_32).
        # The low bits of k * odd depend only on the low bits of k, so % 2^b would put every
        # multiple of 2^b in bucket 0.
        bits = len(self.b).bit_length() - 1          # len(self.b) is a power of two
        return ((k * 2654435761) & 0xFFFFFFFF) >> (32 - bits)

    def put(self, k, v):
        chain = self.b[self._slot(k)]
        for pair in chain:
            if pair[0] == k:
                pair[1] = v
                return
        chain.append([k, v])
        self.n += 1
        if self.n * 4 > len(self.b) * 3:
            old, self.b = self.b, [[] for _ in range(len(self.b) * 2)]
            for ch in old:
                for kk, vv in ch:
                    self.b[self._slot(kk)].append([kk, vv])

    def get(self, k):
        for kk, vv in self.b[self._slot(k)]:
            if kk == k:
                return vv
        return -1

    def remove(self, k):
        chain = self.b[self._slot(k)]
        for i, pair in enumerate(chain):
            if pair[0] == k:
                chain.pop(i)
                self.n -= 1
                return

Talking points: multiplying by a large odd constant spreads nearby integer keys only if you keep the high bits of the product (the kernel's hash_32() does the same): the low bits of k × odd depend only on the low bits of k, so % 16 would send every multiple of 16 to bucket 0; open addressing is more cache friendly but needs tombstones on delete; Java's HashMap turns a chain longer than 8 into a red-black tree so crafted input cannot make it O(n).

From your world

include/linux/hashtable.h is this exact design: an array of hlist_head buckets, each a singly linked chain with a back-pointer for O(1) delete, and hash_min() choosing the bucket. The kernel's rhashtable adds the resize step, done incrementally so lookups never stop.
