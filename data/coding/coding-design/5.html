Randomized set: array + index map

LeetCode 380: insert, remove and getRandom (each element equally likely), all average O(1). A hash set cannot pick a random element in O(1); an array can (random index), but cannot delete from the middle in O(1). So keep both, and delete by moving the last element into the hole.

// 4. Randomized set (LC 380).
type RandomizedSet struct {
	vals []int
	idx  map[int]int
}

func NewRandomizedSet() *RandomizedSet { return &RandomizedSet{idx: map[int]int{}} }

func (s *RandomizedSet) Insert(x int) bool {
	if _, ok := s.idx[x]; ok {
		return false
	}
	s.idx[x] = len(s.vals)
	s.vals = append(s.vals, x)
	return true
}

func (s *RandomizedSet) Remove(x int) bool {
	i, ok := s.idx[x]
	if !ok {
		return false
	}
	last := s.vals[len(s.vals)-1]
	s.vals[i], s.idx[last] = last, i
	s.vals = s.vals[:len(s.vals)-1]
	delete(s.idx, x) // after fixing last: x may be last
	return true
}

func (s *RandomizedSet) GetRandom() int { return s.vals[rand.Intn(len(s.vals))] }
// 4. Randomized set (LC 380): array + value -> index map, delete = swap with last.
class RandomizedSet {
    vector<int> vals;
    unordered_map<int, int> idx;
    mt19937 rng{12345};
public:
    bool insert(int x) {
        if (idx.count(x)) return false;
        idx[x] = (int)vals.size();
        vals.push_back(x);
        return true;
    }
    bool remove(int x) {
        auto it = idx.find(x);
        if (it == idx.end()) return false;
        int i = it->second, last = vals.back();
        vals[i] = last;                                            // move last into the hole
        idx[last] = i;
        vals.pop_back();
        idx.erase(x);                                              // erase AFTER fixing last (x may be last)
        return true;
    }
    int getRandom() { return vals[rng() % vals.size()]; }
};
// 4. Randomized set (LC 380).
static class RandomizedSet {
    final List<Integer> vals = new ArrayList<>();
    final Map<Integer, Integer> idx = new HashMap<>();
    final Random rng = new Random(12345);
    boolean insert(int x) {
        if (idx.containsKey(x)) return false;
        idx.put(x, vals.size());
        vals.add(x);
        return true;
    }
    boolean remove(int x) {
        Integer i = idx.get(x);
        if (i == null) return false;
        int last = vals.get(vals.size() - 1);
        vals.set(i, last);
        idx.put(last, i);
        vals.remove(vals.size() - 1);                            // remove by INDEX, not by value
        idx.remove(x);
        return true;
    }
    int getRandom() { return vals.get(rng.nextInt(vals.size())); }
}
# 4. Randomized set (LC 380).
class RandomizedSet:
    def __init__(self):
        self.vals, self.idx = [], {}

    def insert(self, x):
        if x in self.idx:
            return False
        self.idx[x] = len(self.vals)
        self.vals.append(x)
        return True

    def remove(self, x):
        if x not in self.idx:
            return False
        i, last = self.idx[x], self.vals[-1]
        self.vals[i], self.idx[last] = last, i      # move last into the hole
        self.vals.pop()
        del self.idx[x]                             # after fixing last: x may be last
        return True

    def getRandom(self):
        return random.choice(self.vals)

With duplicates allowed (381, Hard) the map goes from value to a set of indexes; the swap step then removes one index from the deleted value's set and fixes one index in the moved value's set. Order matters: when the element being removed is itself the last one, update the map before you delete, or you re-add what you removed.
