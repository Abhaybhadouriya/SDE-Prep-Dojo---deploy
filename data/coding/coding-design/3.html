LFU cache: one list per frequency

LeetCode 460 evicts the least frequently used key, and among keys with the same count, the least recently used. Both calls O(1). A heap would be O(log n), so the trick is to group keys by count:

  • byFreq[f]: an LRU list of the keys used exactly f times.
  • pos[key]: the node (value, count) of each key.
  • minFreq: the smallest count that has any key. It is the only list we ever evict from.

Why can minFreq be kept in O(1)? It only changes in two ways. A touch can raise it by exactly one (the key left the last list at minFreq and is now at minFreq+1). A brand-new key always resets it to 1. It never needs a search.

// 2. LFU cache (LC 460): freq -> list (front = newest) + minFreq.
type lfuNode struct{ key, val, freq int }

type LFUCache struct {
	cap, minFreq int
	byFreq       map[int]*list.List
	pos          map[int]*list.Element
}

func NewLFUCache(capacity int) *LFUCache {
	return &LFUCache{cap: capacity, byFreq: map[int]*list.List{}, pos: map[int]*list.Element{}}
}

func (c *LFUCache) bucket(f int) *list.List {
	if c.byFreq[f] == nil {
		c.byFreq[f] = list.New()
	}
	return c.byFreq[f]
}

func (c *LFUCache) touch(e *list.Element) {
	n := e.Value.(*lfuNode)
	l := c.byFreq[n.freq]
	l.Remove(e)
	if l.Len() == 0 {
		delete(c.byFreq, n.freq)
		if c.minFreq == n.freq {
			c.minFreq++
		}
	}
	n.freq++
	c.pos[n.key] = c.bucket(n.freq).PushFront(n)
}

func (c *LFUCache) Get(key int) int {
	e, ok := c.pos[key]
	if !ok {
		return -1
	}
	v := e.Value.(*lfuNode).val
	c.touch(e)
	return v
}

func (c *LFUCache) Put(key, value int) {
	if c.cap == 0 {
		return
	}
	if e, ok := c.pos[key]; ok {
		e.Value.(*lfuNode).val = value
		c.touch(e)
		return
	}
	if len(c.pos) == c.cap {
		l := c.byFreq[c.minFreq]
		old := l.Back()
		l.Remove(old)
		if l.Len() == 0 {
			delete(c.byFreq, c.minFreq)
		}
		delete(c.pos, old.Value.(*lfuNode).key)
	}
	c.minFreq = 1
	c.pos[key] = c.bucket(1).PushFront(&lfuNode{key, value, 1})
}
// 2. LFU cache (LC 460): one LRU list per frequency + minFreq.
class LFUCache {
    struct Node { int key, val, freq; };
    int cap, minFreq = 0;
    unordered_map<int, list<Node>> byFreq;                         // freq -> nodes, front = newest
    unordered_map<int, list<Node>::iterator> pos;
    void touch(list<Node>::iterator it) {                          // freq f -> f+1
        Node n = *it;
        byFreq[n.freq].erase(it);
        if (byFreq[n.freq].empty()) {
            byFreq.erase(n.freq);
            if (minFreq == n.freq) minFreq++;
        }
        n.freq++;
        byFreq[n.freq].push_front(n);
        pos[n.key] = byFreq[n.freq].begin();
    }
public:
    explicit LFUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        auto it = pos.find(key);
        if (it == pos.end()) return -1;
        int v = it->second->val;
        touch(it->second);
        return v;
    }
    void put(int key, int value) {
        if (cap == 0) return;
        auto it = pos.find(key);
        if (it != pos.end()) { it->second->val = value; touch(it->second); return; }
        if ((int)pos.size() == cap) {                              // evict LRU among minFreq
            auto& lst = byFreq[minFreq];
            pos.erase(lst.back().key);
            lst.pop_back();
            if (lst.empty()) byFreq.erase(minFreq);
        }
        minFreq = 1;                                               // a new key always has freq 1
        byFreq[1].push_front({key, value, 1});
        pos[key] = byFreq[1].begin();
    }
};
// 2. LFU cache (LC 460): freq -> LinkedHashSet (insertion order = LRU order) + minFreq.
static class LFUCache {
    final int cap;
    int minFreq = 0;
    final Map<Integer, Integer> val = new HashMap<>(), freq = new HashMap<>();
    final Map<Integer, LinkedHashSet<Integer>> byFreq = new HashMap<>();
    LFUCache(int capacity) { cap = capacity; }
    void touch(int key) {
        int f = freq.get(key);
        LinkedHashSet<Integer> s = byFreq.get(f);
        s.remove(key);
        if (s.isEmpty()) { byFreq.remove(f); if (minFreq == f) minFreq++; }
        freq.put(key, f + 1);
        byFreq.computeIfAbsent(f + 1, x -> new LinkedHashSet<>()).add(key);
    }
    int get(int key) {
        if (!val.containsKey(key)) return -1;
        touch(key);
        return val.get(key);
    }
    void put(int key, int value) {
        if (cap == 0) return;
        if (val.containsKey(key)) { val.put(key, value); touch(key); return; }
        if (val.size() == cap) {
            LinkedHashSet<Integer> s = byFreq.get(minFreq);
            int old = s.iterator().next();                       // oldest at this freq
            s.remove(old);
            if (s.isEmpty()) byFreq.remove(minFreq);
            val.remove(old); freq.remove(old);
        }
        val.put(key, value); freq.put(key, 1); minFreq = 1;
        byFreq.computeIfAbsent(1, x -> new LinkedHashSet<>()).add(key);
    }
}
# 2. LFU cache (LC 460): freq -> OrderedDict (an LRU list per frequency) + min_freq.
class LFUCache:
    def __init__(self, capacity):
        self.cap, self.min_freq = capacity, 0
        self.val, self.freq = {}, {}
        self.by_freq = defaultdict(OrderedDict)    # freq -> keys, oldest first

    def _touch(self, key):
        f = self.freq[key]
        del self.by_freq[f][key]
        if not self.by_freq[f]:
            del self.by_freq[f]
            if self.min_freq == f:
                self.min_freq += 1
        self.freq[key] = f + 1
        self.by_freq[f + 1][key] = None

    def get(self, key):
        if key not in self.val:
            return -1
        self._touch(key)
        return self.val[key]

    def put(self, key, value):
        if self.cap == 0:
            return
        if key in self.val:
            self.val[key] = value
            self._touch(key)
            return
        if len(self.val) == self.cap:
            old, _ = self.by_freq[self.min_freq].popitem(last=False)
            if not self.by_freq[self.min_freq]:
                del self.by_freq[self.min_freq]
            del self.val[old], self.freq[old]
        self.val[key], self.freq[key], self.min_freq = value, 1, 1
        self.by_freq[1][key] = None

The two LFU traps: capacity 0 (every put must do nothing), and evicting before inserting the new key. If you insert first, the new key (count 1) is the one you evict.
