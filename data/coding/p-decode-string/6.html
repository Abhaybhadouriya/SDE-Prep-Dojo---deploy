class Solution {
public:
    string decodeString(string s) {
        vector<pair<string, int>> stk;   // (text before this '[', repeat count of this '[')
        string cur;                      // text of the innermost open bracket so far
        int k = 0;                       // the number being read (may have several digits)
        for (char c : s) {
            if (isdigit(c)) {
                k = k * 10 + (c - '0');
            } else if (c == '[') {       // open a frame: save the outer text and the count
                stk.push_back({move(cur), k});
                cur.clear();
                k = 0;
            } else if (c == ']') {       // close it: outer + cur repeated count times
                string outer = move(stk.back().first);
                int times = stk.back().second;
                stk.pop_back();
                outer.reserve(outer.size() + cur.size() * times);
                for (int t = 0; t < times; t++) outer += cur;
                cur = move(outer);
            } else {
                cur += c;                // a plain letter belongs to the current frame
            }
        }
        return cur;
    }
};
func decodeString(s string) string {
	type frame struct {
		outer string // text before this '['
		times int    // repeat count of this '['
	}
	stack := []frame{}
	var cur strings.Builder // text of the innermost frame so far
	k := 0                  // the number being read (may have several digits)
	for _, c := range s {
		switch {
		case c >= '0' && c <= '9':
			k = k*10 + int(c-'0')
		case c == '[': // open a frame: save the outer text and the count
			stack = append(stack, frame{cur.String(), k})
			cur.Reset()
			k = 0
		case c == ']': // close it: outer + cur repeated count times
			top := stack[len(stack)-1]
			stack = stack[:len(stack)-1]
			inner := cur.String()
			cur.Reset()
			cur.WriteString(top.outer)
			cur.WriteString(strings.Repeat(inner, top.times))
		default:
			cur.WriteRune(c) // a plain letter belongs to the current frame
		}
	}
	return cur.String()
}

static class Solution {
    public String decodeString(String s) {
        Deque<Integer> counts = new ArrayDeque<>();          // repeat count of each open '['
        Deque<StringBuilder> outers = new ArrayDeque<>();    // text before each open '['
        StringBuilder cur = new StringBuilder();             // text of the innermost frame
        int k = 0;                                           // number being read
        for (char c : s.toCharArray()) {
            if (Character.isDigit(c)) {
                k = k * 10 + (c - '0');
            } else if (c == '[') {          // open a frame: save the outer text and the count
                counts.push(k);
                outers.push(cur);
                cur = new StringBuilder();
                k = 0;
            } else if (c == ']') {          // close it: outer + cur repeated count times
                StringBuilder outer = outers.pop();
                int times = counts.pop();
                for (int t = 0; t < times; t++) outer.append(cur);
                cur = outer;
            } else {
                cur.append(c);              // a plain letter belongs to the current frame
            }
        }
        return cur.toString();
    }
}
class Solution:
    def decodeString(self, s: str) -> str:
        stack = []          # frames: (parts before this '[', repeat count of this '[')
        cur = []            # pieces of the innermost frame (joined once: no O(n^2) +=)
        k = 0               # the number being read (may have several digits)
        for c in s:
            if c.isdigit():
                k = k * 10 + int(c)
            elif c == '[':                     # open a frame: save outer text and count
                stack.append((cur, k))
                cur, k = [], 0
            elif c == ']':                     # close it: outer + cur repeated count times
                outer, times = stack.pop()
                outer.append(''.join(cur) * times)
                cur = outer
            else:
                cur.append(c)                  # a plain letter belongs to the current frame
        return ''.join(cur)
