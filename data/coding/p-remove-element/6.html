class Solution {
public:
    // Reader r scans; every kept value is copied down to the write slot w. Keeps the order.
    int removeElement(vector<int>& nums, int val) {
        int w = 0;                                   // nums[0..w) = kept values, in input order
        for (int r = 0; r < (int)nums.size(); r++)
            if (nums[r] != val) nums[w++] = nums[r];
        return w;
    }

    // Removals are rare: overwrite a hit with the last live value and shrink. Order not kept.
    int removeElementSwap(vector<int>& nums, int val) {
        int i = 0, n = nums.size();                  // live part is nums[0..n)
        while (i < n) {
            if (nums[i] == val) nums[i] = nums[--n]; // re-check slot i: the moved value may be val too
            else i++;
        }
        return n;
    }
};

	w := 0 // nums[:w] = kept values, in input order
	for _, x := range nums {

			w++
		}
	}
	return w
}

class Solution:
    def removeElement(self, nums: List[int], val: int) -> int:
        """Reader r scans; every kept value is copied down to the write slot w. Keeps the order."""
        w = 0                                    # nums[0:w] = kept values, in input order
        for r in range(len(nums)):
            if nums[r] != val:
                nums[w] = nums[r]
                w += 1
        return w

    def removeElementSwap(self, nums: List[int], val: int) -> int:
        """Removals are rare: overwrite a hit with the last live value and shrink. Order not kept."""
        i, n = 0, len(nums)                      # live part is nums[0:n]
        while i < n:
            if nums[i] == val:
                n -= 1
                nums[i] = nums[n]                # re-check slot i: the moved value may be val too
            else:
                i += 1
        return n
