All four use the backward merge with the single while j >= 0 loop, so m = 0 and n = 0 need no special case.

class Solution {
public:
    void merge(vector<int>& nums1, int m, const vector<int>& nums2, int n) {
        int i = m - 1;          // last real value of nums1
        int j = n - 1;          // last value of nums2
        int w = m + n - 1;      // write slot, filled from the back (the free space is there)
        while (j >= 0) {        // when nums2 is used up, nums1[0..i] is already in place
            if (i >= 0 && nums1[i] > nums2[j]) nums1[w--] = nums1[i--];
            else nums1[w--] = nums2[j--];       // ties take nums2: either is fine
        }
    }
};
func merge(nums1 []int, m int, nums2 []int, n int) {
	i, j, w := m-1, n-1, m+n-1 // two read heads and one write head, all at the back
	for j >= 0 {               // when nums2 is used up, nums1[0..i] is already in place
		if i >= 0 && nums1[i] > nums2[j] {
			nums1[w] = nums1[i]
			i--
		} else { // ties take nums2: either is fine
			nums1[w] = nums2[j]
			j--
		}
		w--
	}
}
