func isValidBST(root *TreeNode) bool {
	// Every node must lie strictly inside the open interval (lo, hi) its ancestors allow.
	// int64 bounds: values use the whole int32 range, so they cannot be sentinels.
	var inRange func(node *TreeNode, lo, hi int64) bool
	inRange = func(node *TreeNode, lo, hi int64) bool {
		if node == nil {
			return true
		}
		v := int64(node.Val)
		if v <= lo || v >= hi { // duplicates are invalid too
			return false
		}
		return inRange(node.Left, lo, v) && // left side: tighten the upper bound
			inRange(node.Right, v, hi) // right side: tighten the lower bound
	}
	return inRange(root, math.MinInt64, math.MaxInt64)
}
class Solution {
    // Every node must lie strictly inside the open interval (lo, hi) that its ancestors allow.
    // 64-bit bounds: node values use the whole int range, so INT_MIN/INT_MAX cannot be sentinels.
    bool inRange(TreeNode* node, long long lo, long long hi) {
        if (!node) return true;
        if (node->val <= lo || node->val >= hi) return false;       // duplicates are invalid too
        return inRange(node->left, lo, node->val)                    // left side: tighten the upper bound
            && inRange(node->right, node->val, hi);                  // right side: tighten the lower bound
    }
public:
    bool isValidBST(TreeNode* root) {
        return inRange(root, LLONG_MIN, LLONG_MAX);
    }
};

class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        def in_range(node: Optional[TreeNode], lo: float, hi: float) -> bool:
            """Every node must lie strictly inside the open interval (lo, hi) its ancestors allow."""
            if node is None:
                return True
            if not (lo < node.val < hi):               # duplicates are invalid too
                return False
            return (in_range(node.left, lo, node.val)   # left side: tighten the upper bound
                    and in_range(node.right, node.val, hi))  # right side: tighten the lower bound

        return in_range(root, float("-inf"), float("inf"))
