Cyclic sort in four languages. Each file in coding-code/first-missing-positive/ checks the examples, duplicates, extreme 32-bit values, an all-present array, and 3,000 random arrays against the hash-set version.

class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        // The answer is in 1..n+1, so only values 1..n matter. Park every such
        // value v in its home slot v-1. Each swap puts one value home for good,
        // so the while loop runs at most n times in total: O(n).
        for (int i = 0; i < n; i++) {
            while (nums[i] >= 1 && nums[i] <= n && nums[nums[i] - 1] != nums[i])
                swap(nums[i], nums[nums[i] - 1]);   // the guard also stops on duplicates
        }
        // The first slot that does not hold its own value names the gap.
        for (int i = 0; i < n; i++)
            if (nums[i] != i + 1) return i + 1;
        return n + 1;                               // 1..n are all present
    }
};

                int tmp = nums[home];
                nums[home] = nums[i];

            }

        for (int i = 0; i < n; i++)
            if (nums[i] != i + 1) return i + 1;

class Solution:
    def firstMissingPositive(self, nums: List[int]) -> int:
        n = len(nums)
        # The answer is in 1..n+1, so only values 1..n matter. Park every such
        # value v in its home slot v-1. Each swap puts one value home for good,
        # so the while loop runs at most n times in total: O(n).
        for i in range(n):
            while 1 <= nums[i] <= n and nums[nums[i] - 1] != nums[i]:
                home = nums[i] - 1             # compute the index first: a, b = b, a
                nums[i], nums[home] = nums[home], nums[i]  # with nums[nums[i]-1] breaks
        # The first slot that does not hold its own value names the gap.
        for i in range(n):
            if nums[i] != i + 1:
                return i + 1
        return n + 1                           # 1..n are all present
