class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        int n = nums.size();
        vector<int> out(n);
        int lo = 0, hi = n - 1;               // the biggest |value| left is at lo or at hi
        for (int w = n - 1; w >= 0; w--) {    // fill the largest slot first
            int a = nums[lo] * nums[lo];
            int b = nums[hi] * nums[hi];
            if (a > b) { out[w] = a; lo++; }  // the negative end wins
            else       { out[w] = b; hi--; }  // ties: take hi, either is fine
        }
        return out;
    }
};

class Solution:
    def sortedSquares(self, nums: List[int]) -> List[int]:
        n = len(nums)
        out = [0] * n
        lo, hi = 0, n - 1                  # the biggest |value| left is at lo or at hi
        for w in range(n - 1, -1, -1):     # fill the largest slot first
            a, b = nums[lo] * nums[lo], nums[hi] * nums[hi]
            if a > b:                      # the negative end wins
                out[w] = a
                lo += 1
            else:                          # ties: take hi, either is fine
                out[w] = b
                hi -= 1
        return out
