The sliding window in four languages. Each file in coding-code/subarray-product-less-than-k/ runs the examples and edge cases and cross-checks 500 random arrays against the O(n²) brute force.

class Solution {
public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        if (k <= 1) return 0;                    // every product is >= 1, nothing is < 1
        long long prod = 1;                      // product of nums[left..right]
        int left = 0, count = 0;
        for (int right = 0; right < (int)nums.size(); right++) {
            prod *= nums[right];
            while (prod >= k) prod /= nums[left++]; // shrink until the window is valid
            count += right - left + 1;           // windows that end at right, start in [left, right]
        }
        return count;
    }
};
func numSubarrayProductLessThanK(nums []int, k int) int {
	if k <= 1 { // every product is >= 1, nothing is < 1
		return 0
	}
	prod, left, count := 1, 0, 0 // prod = product of nums[left..right]

		prod *= x

			prod /= nums[left]
			left++
		}
		count += right - left + 1 // windows that end at right, start in [left, right]
	}
	return count
}

class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:                      # every product is >= 1, nothing is < 1
            return 0
        prod = 1                        # product of nums[left..right]
        left = 0
        count = 0
        for right, x in enumerate(nums):
            prod *= x
            while prod >= k:            # shrink until the window is valid
                prod //= nums[left]     # exact: nums[left] divides prod
                left += 1
            count += right - left + 1   # windows that end at right, start in [left, right]
        return count
