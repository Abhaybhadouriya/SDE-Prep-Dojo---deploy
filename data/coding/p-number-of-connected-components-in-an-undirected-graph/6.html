Union-find with path halving and union by size; the tested files are in coding-code/number-of-connected-components-in-an-undirected-graph/ (the Python test also cross-checks 500 random graphs against an iterative DFS).

class Solution {
    vector<int> parent, sz;

    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];   // path halving: point to the grandparent
            x = parent[x];
        }
        return x;
    }

public:
    int countComponents(int n, vector<vector<int>>& edges) {
        parent.resize(n);
        iota(parent.begin(), parent.end(), 0);   // every node starts as its own component
        sz.assign(n, 1);
        int components = n;
        for (auto& e : edges) {
            int ra = find(e[0]), rb = find(e[1]);
            if (ra == rb) continue;              // already connected: this edge closes a cycle
            if (sz[ra] < sz[rb]) swap(ra, rb);   // hang the smaller tree under the bigger one
            parent[rb] = ra;
            sz[ra] += sz[rb];
            components--;                        // two components became one
        }
        return components;
    }
};
func countComponents(n int, edges [][]int) int {
	parent := make([]int, n)
	size := make([]int, n)
	for i := range parent {
		parent[i], size[i] = i, 1 // every node starts as its own component
	}
	find := func(x int) int {
		for parent[x] != x {
			parent[x] = parent[parent[x]] // path halving: point to the grandparent
			x = parent[x]
		}
		return x
	}
	components := n

    private int find(int x) {
        while (parent[x] != x) {

        }
        return x;
    }

    public int countComponents(int n, int[][] edges) {
        parent = new int[n];
        size = new int[n];
        for (int i = 0; i < n; i++) { parent[i] = i; size[i] = 1; }   // n singletons
        int components = n;
        for (int[] e : edges) {
            int ra = find(e[0]), rb = find(e[1]);
            if (ra == rb) continue;                  // already connected: this edge closes a cycle
            if (size[ra] < size[rb]) { int t = ra; ra = rb; rb = t; }   // smaller under bigger
            parent[rb] = ra;
            size[ra] += size[rb];
            components--;                            // two components became one
        }
        return components;
    }
}
