Iterative solution plus the recursive form in four languages. Each file in coding-code/reverse-linked-list/ checks both on the examples, a duplicate-value list and a 5000-node list; the Python file shows that the recursive form hits RecursionError at 5000 nodes with the default limit.

class Solution {
public:
    // Iterative: walk once, turning each next pointer around. O(n) time, O(1) space.
    ListNode* reverseList(ListNode* head) {
        ListNode* prev = nullptr;          // head of the part that is already reversed
        ListNode* cur = head;              // first node of the part that is not
        while (cur) {
            ListNode* nxt = cur->next;     // save the rest of the list before we cut the link
            cur->next = prev;              // turn the arrow around
            prev = cur;                    // the reversed part grows by one node
            cur = nxt;
        }
        return prev;                       // the old tail is the new head
    }

    // Recursive: reverse the rest, then hang this node behind its old successor. O(n) stack.
    ListNode* reverseRecursive(ListNode* head) {
        if (!head || !head->next) return head;
        ListNode* newHead = reverseRecursive(head->next);
        head->next->next = head;           // the old successor is now the tail: point it back at us
        head->next = nullptr;              // we are the new tail
        return newHead;
    }
};
// reverseList walks once, turning each Next pointer around. O(n) time, O(1) space.
func reverseList(head *ListNode) *ListNode {
	var prev *ListNode // head of the part that is already reversed
	cur := head        // first node of the part that is not
	for cur != nil {
		nxt := cur.Next // save the rest of the list before we cut the link
		cur.Next = prev // turn the arrow around
		prev = cur      // the reversed part grows by one node
		cur = nxt
	}
	return prev // the old tail is the new head
}

// reverseRecursive reverses the rest, then hangs this node behind its old successor. O(n) stack.
func reverseRecursive(head *ListNode) *ListNode {
	if head == nil || head.Next == nil {
		return head
	}
	newHead := reverseRecursive(head.Next)
	head.Next.Next = head // the old successor is now the tail: point it back at us
	head.Next = nil       // we are the new tail
	return newHead
}

class Solution:
    def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        """Iterative: walk once, turning each next pointer around. O(n) time, O(1) space."""
        prev, cur = None, head          # prev = reversed part, cur = the rest
        while cur:
            nxt = cur.next              # save the rest of the list before we cut the link
            cur.next = prev             # turn the arrow around
            prev, cur = cur, nxt        # the reversed part grows by one node
        return prev                     # the old tail is the new head

    def reverseRecursive(self, head: Optional[ListNode]) -> Optional[ListNode]:
        """Recursive: reverse the rest, then hang this node behind its old successor.
        O(n) stack: Python's default limit (1000 frames) fails on 5000 nodes."""
        if head is None or head.next is None:
            return head
        new_head = self.reverseRecursive(head.next)
        head.next.next = head           # the old successor is now the tail: point it back at us
        head.next = None                # we are the new tail
        return new_head
