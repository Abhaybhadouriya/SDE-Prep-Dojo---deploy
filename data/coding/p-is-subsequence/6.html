class Solution {
public:
    bool isSubsequence(string s, string t) {
        size_t i = 0;                                   // s[0..i-1] is already matched
        for (size_t j = 0; j < t.size() && i < s.size(); j++) {
            if (s[i] == t[j]) i++;                      // earliest match never hurts
        }
        return i == s.size();
    }
};

// Follow-up: one t, a huge stream of queries s. Build once, then O(|s| log |t|) per query.
class SubseqIndex {
    vector<int> pos[26];                                // pos[c] = indices of letter c in t, ascending
public:
    explicit SubseqIndex(const string& t) {
        for (int j = 0; j < (int)t.size(); j++) pos[t[j] - 'a'].push_back(j);
    }
    bool query(const string& s) const {
        int next = 0;                                   // first index of t still free to use
        for (char c : s) {
            const vector<int>& p = pos[c - 'a'];
            auto it = lower_bound(p.begin(), p.end(), next);  // earliest c at or after next
            if (it == p.end()) return false;
            next = *it + 1;
        }
        return true;
    }
};

		c := t[j] - 'a'
		x.pos[c] = append(x.pos[c], j)

func (x *SubseqIndex) Query(s string) bool {
	next := 0 // first index of t still free to use
	for i := 0; i < len(s); i++ {
		p := x.pos[s[i]-'a']
		k := sort.SearchInts(p, next) // earliest occurrence at or after next
		if k == len(p) {
			return false
		}
		next = p[k] + 1
	}
	return true
}

// Follow-up: one t, a huge stream of queries s. Build once, then O(|s| log |t|) per query.
static class SubseqIndex {
    private final int[][] pos = new int[26][];         // pos[c] = indices of letter c in t, ascending

    SubseqIndex(String t) {
        int[] cnt = new int[26];
        for (int j = 0; j < t.length(); j++) cnt[t.charAt(j) - 'a']++;
        for (int c = 0; c < 26; c++) pos[c] = new int[cnt[c]];
        int[] fill = new int[26];
        for (int j = 0; j < t.length(); j++) {
            int c = t.charAt(j) - 'a';
            pos[c][fill[c]++] = j;
        }
    }

    boolean query(String s) {
        int next = 0;                                  // first index of t still free to use
        for (int i = 0; i < s.length(); i++) {
            int[] p = pos[s.charAt(i) - 'a'];
            int lo = 0, hi = p.length;                 // lower_bound(p, next)
            while (lo < hi) {
                int mid = (lo + hi) >>> 1;
                if (p[mid] < next) lo = mid + 1; else hi = mid;
            }
            if (lo == p.length) return false;
            next = p[lo] + 1;
        }
        return true;
    }
}

class SubseqIndex:
    """Follow-up: one t, a huge stream of queries s. Build once, then O(|s| log |t|) per query."""

    def __init__(self, t: str):
        self.pos: defaultdict = defaultdict(list)   # pos[c] = indices of letter c in t, ascending
        for j, ch in enumerate(t):
            self.pos[ch].append(j)

    def query(self, s: str) -> bool:
        nxt = 0                             # first index of t still free to use
        for ch in s:
            p: List[int] = self.pos.get(ch, [])
            k = bisect_left(p, nxt)         # earliest occurrence at or after nxt
            if k == len(p):
                return False
            nxt = p[k] + 1
        return True
