The same design in four languages; the tested files are in coding-code/lfu-cache/. C++ uses std::list per count with stored iterators (erase by iterator is O(1)); Java and Go hand-write a sentinel doubly linked list; Python uses an OrderedDict per count (an O(1) linked hash map). The Python driver also checks 200 random operation sequences against a slow reference. Go uses LeetCode's names Constructor, Get and Put.

class LFUCache {
    struct Entry {
        int val, freq;
        list<int>::iterator pos;            // where the key sits inside byFreq[freq]
    };
    int cap, minFreq = 0;
    unordered_map<int, Entry> entries;      // key -> value, use count, handle
    unordered_map<int, list<int>> byFreq;   // use count -> keys, most recent at the front

    // One more use of `key`: move it from list f to the front of list f+1.
    void touch(int key) {
        Entry& e = entries[key];
        int f = e.freq;
        byFreq[f].erase(e.pos);
        if (byFreq[f].empty()) {
            byFreq.erase(f);
            if (minFreq == f) minFreq = f + 1;   // the key we just moved now has f+1 uses
        }
        e.freq = f + 1;
        byFreq[f + 1].push_front(key);
        e.pos = byFreq[f + 1].begin();
    }

public:
    LFUCache(int capacity) : cap(capacity) {}

    int get(int key) {
        auto it = entries.find(key);
        if (it == entries.end()) return -1;
        touch(key);                              // a successful get is a use
        return it->second.val;
    }

    void put(int key, int value) {
        if (cap <= 0) return;
        auto it = entries.find(key);
        if (it != entries.end()) {               // update: one more use, never an eviction
            it->second.val = value;
            touch(key);
            return;
        }
        if ((int)entries.size() == cap) {        // evict: fewest uses, then least recent
            list<int>& lst = byFreq[minFreq];
            int victim = lst.back();
            lst.pop_back();
            if (lst.empty()) byFreq.erase(minFreq);
            entries.erase(victim);
        }
        byFreq[1].push_front(key);               // a new key has exactly one use
        entries[key] = Entry{value, 1, byFreq[1].begin()};
        minFreq = 1;                             // nothing can have fewer uses than the newcomer
    }
};
type lfuNode struct {
	key, val, freq int
	prev, next     *lfuNode
}

// dlist: sentinels; head.next = most recent, tail.prev = least recent.
type dlist struct {
	head, tail *lfuNode
	size       int
}

func newDlist() *dlist {
	h, t := &lfuNode{}, &lfuNode{}
	h.next, t.prev = t, h
	return &dlist{head: h, tail: t}
}

func (l *dlist) pushFront(n *lfuNode) {
	n.prev, n.next = l.head, l.head.next
	l.head.next.prev = n
	l.head.next = n
	l.size++
}

func (l *dlist) unlink(n *lfuNode) {
	n.prev.next, n.next.prev = n.next, n.prev
	l.size--
}

type LFUCache struct {
	cap, minFreq int
	nodes        map[int]*lfuNode // key -> node
	byFreq       map[int]*dlist   // use count -> list
}

func Constructor(capacity int) LFUCache {
	return LFUCache{cap: capacity, nodes: map[int]*lfuNode{}, byFreq: map[int]*dlist{}}
}

// touch: one more use, move the node from list f to the front of list f+1.
func (c *LFUCache) touch(n *lfuNode) {
	from := c.byFreq[n.freq]
	from.unlink(n)
	if from.size == 0 {
		delete(c.byFreq, n.freq)
		if c.minFreq == n.freq {
			c.minFreq++ // the moved node now has freq+1 uses
		}
	}
	n.freq++
	to, ok := c.byFreq[n.freq]
	if !ok {
		to = newDlist()
		c.byFreq[n.freq] = to
	}
	to.pushFront(n)
}

func (c *LFUCache) Get(key int) int {
	n, ok := c.nodes[key]
	if !ok {
		return -1
	}
	c.touch(n) // a successful get is a use
	return n.val
}

func (c *LFUCache) Put(key int, value int) {
	if c.cap <= 0 {
		return
	}
	if n, ok := c.nodes[key]; ok { // update: one more use, never an eviction
		n.val = value
		c.touch(n)
		return
	}
	if len(c.nodes) == c.cap { // evict: fewest uses, then least recent
		lst := c.byFreq[c.minFreq]
		victim := lst.tail.prev
		lst.unlink(victim)
		if lst.size == 0 {
			delete(c.byFreq, c.minFreq)
		}
		delete(c.nodes, victim.key)
	}
	n := &lfuNode{key: key, val: value, freq: 1} // a new key has exactly one use
	one, ok := c.byFreq[1]
	if !ok {
		one = newDlist()
		c.byFreq[1] = one
	}
	one.pushFront(n)
	c.nodes[key] = n
	c.minFreq = 1 // nothing has fewer uses than the newcomer
}

static class LFUCache {
    private static class Node {
        int key, val, freq = 1;
        Node prev, next;
        Node(int k, int v) { key = k; val = v; }
    }

    // Doubly linked list with sentinels; head.next = most recent, tail.prev = least recent.
    private static class DList {
        final Node head = new Node(0, 0), tail = new Node(0, 0);
        int size;
        DList() { head.next = tail; tail.prev = head; }
        void pushFront(Node n) {
            n.prev = head; n.next = head.next;
            head.next.prev = n; head.next = n;
            size++;
        }
        void unlink(Node n) {
            n.prev.next = n.next; n.next.prev = n.prev;
            size--;
        }
    }

    private final int cap;
    private int minFreq = 0;
    private final Map<Integer, Node> nodes = new HashMap<>();    // key -> node
    private final Map<Integer, DList> byFreq = new HashMap<>();  // use count -> list

    LFUCache(int capacity) { cap = capacity; }

    // One more use: move the node from list f to the front of list f+1.
    private void touch(Node n) {
        DList from = byFreq.get(n.freq);
        from.unlink(n);
        if (from.size == 0) {
            byFreq.remove(n.freq);
            if (minFreq == n.freq) minFreq++;      // the moved node now has freq+1 uses
        }
        n.freq++;
        byFreq.computeIfAbsent(n.freq, f -> new DList()).pushFront(n);
    }

    public int get(int key) {
        Node n = nodes.get(key);
        if (n == null) return -1;
        touch(n);                                  // a successful get is a use
        return n.val;
    }

    public void put(int key, int value) {
        if (cap <= 0) return;
        Node n = nodes.get(key);
        if (n != null) {                           // update: one more use, never an eviction
            n.val = value;
            touch(n);
            return;
        }
        if (nodes.size() == cap) {                 // evict: fewest uses, then least recent
            DList lst = byFreq.get(minFreq);
            Node victim = lst.tail.prev;
            lst.unlink(victim);
            if (lst.size == 0) byFreq.remove(minFreq);
            nodes.remove(victim.key);              // the node knows its key
        }
        n = new Node(key, value);                  // a new key has exactly one use
        byFreq.computeIfAbsent(1, f -> new DList()).pushFront(n);
        nodes.put(key, n);
        minFreq = 1;                               // nothing has fewer uses than the newcomer
    }
}
class LFUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.min_freq = 0
        self.val = {}                             # key -> value
        self.freq = {}                            # key -> use count
        self.by_freq = defaultdict(OrderedDict)   # use count -> keys, oldest first

    def _touch(self, key):
        """One more use: move key from bucket f to the newest end of bucket f+1."""
        f = self.freq[key]
        del self.by_freq[f][key]
        if not self.by_freq[f]:
            del self.by_freq[f]
            if self.min_freq == f:
                self.min_freq = f + 1             # the moved key now has f+1 uses
        self.freq[key] = f + 1
        self.by_freq[f + 1][key] = None

    def get(self, key: int) -> int:
        if key not in self.val:
            return -1
        self._touch(key)                          # a successful get is a use
        return self.val[key]

    def put(self, key: int, value: int) -> None:
        if self.cap <= 0:
            return
        if key in self.val:                       # update: one more use, never an eviction
            self.val[key] = value
            self._touch(key)
            return
        if len(self.val) == self.cap:             # evict: fewest uses, then least recent
            bucket = self.by_freq[self.min_freq]
            victim, _ = bucket.popitem(last=False)
            if not bucket:
                del self.by_freq[self.min_freq]
            del self.val[victim], self.freq[victim]
        self.val[key], self.freq[key] = value, 1  # a new key has exactly one use
        self.by_freq[1][key] = None
        self.min_freq = 1                         # nothing has fewer uses than the newcomer
