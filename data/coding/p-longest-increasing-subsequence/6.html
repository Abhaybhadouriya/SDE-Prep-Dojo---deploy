All four use the tails array with a lower-bound search (C++ lower_bound, Java a hand-written search, Python bisect_left, Go sort.SearchInts).

class Solution {
public:
    int lengthOfLIS(const vector<int>& nums) {
        // tails[k] = smallest last value of ANY strictly increasing subsequence of length k+1.
        // tails is strictly increasing, so we can binary search it.
        vector<int> tails;
        for (int x : nums) {
            auto it = lower_bound(tails.begin(), tails.end(), x);   // first tail >= x
            if (it == tails.end()) tails.push_back(x);   // x extends the longest one
            else *it = x;                                 // same length, smaller tail
        }
        return (int)tails.size();   // the length is right; tails itself need not be an LIS
    }
};
func lengthOfLIS(nums []int) int {
	// tails[k] = smallest last value of ANY strictly increasing subsequence of length k+1
	tails := make([]int, 0, len(nums))
	for _, x := range nums {
		k := sort.SearchInts(tails, x) // first tail >= x (keeps the increase strict)
		if k == len(tails) {
			tails = append(tails, x) // x extends the longest subsequence so far
		} else {
			tails[k] = x // same length, smaller tail: better for the future
		}
	}
	return len(tails) // the length is right; tails itself need not be an LIS
}
static class Solution {
    public int lengthOfLIS(int[] nums) {
        // tails[k] = smallest last value of ANY strictly increasing subsequence of length k+1.
        int[] tails = new int[nums.length];
        int size = 0;
        for (int x : nums) {
            int lo = 0, hi = size;                  // find the first tail >= x in tails[0..size)
            while (lo < hi) {
                int mid = (lo + hi) >>> 1;
                if (tails[mid] < x) lo = mid + 1;
                else hi = mid;
            }
            tails[lo] = x;                          // replace it, or append when lo == size
            if (lo == size) size++;
        }
        return size;            // the length is right; tails itself need not be an LIS
    }
}
class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        # tails[k] = smallest last value of ANY strictly increasing subsequence of length k+1
        tails: List[int] = []
        for x in nums:
            k = bisect_left(tails, x)   # first tail >= x (bisect_left keeps the increase strict)
            if k == len(tails):
                tails.append(x)         # x extends the longest subsequence so far
            else:
                tails[k] = x            # same length, smaller tail: better for the future
        return len(tails)               # the length is right; tails itself need not be an LIS
