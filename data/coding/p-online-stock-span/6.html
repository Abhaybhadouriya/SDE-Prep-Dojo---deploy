The (price, span) stack in four languages; the tested files are in coding-code/online-stock-span/ (the Python test also checks 300 random series against the walk-back version). The Go version follows LeetCode's template: Constructor() and a Next method.

class StockSpanner {
    stack<pair<int, int>> st;   // (price, span), prices strictly decreasing bottom -> top

public:
    StockSpanner() {}

    int next(int price) {
        int span = 1;                                // today counts itself
        while (!st.empty() && st.top().first <= price) {
            span += st.top().second;                 // that day and everything it absorbed
            st.pop();
        }
        st.push({price, span});
        return span;
    }
};
type pair struct{ price, span int }

type StockSpanner struct {
	stack []pair // prices strictly decreasing bottom -> top
}

func Constructor() StockSpanner {
	return StockSpanner{}
}

func (this *StockSpanner) Next(price int) int {
	span := 1 // today counts itself
	for len(this.stack) > 0 && this.stack[len(this.stack)-1].price <= price {
		span += this.stack[len(this.stack)-1].span // that day and everything it absorbed
		this.stack = this.stack[:len(this.stack)-1]
	}
	this.stack = append(this.stack, pair{price, span})
	return span
}

static class StockSpanner {
    private final Deque<int[]> stack = new ArrayDeque<>();   // {price, span}, prices decreasing

    public StockSpanner() {}

    public int next(int price) {
        int span = 1;                                      // today counts itself
        while (!stack.isEmpty() && stack.peek()[0] <= price) {
            span += stack.pop()[1];                        // that day and everything it absorbed
        }
        stack.push(new int[]{price, span});
        return span;
    }
}
class StockSpanner:
    def __init__(self):
        self.stack = []              # (price, span) pairs, prices strictly decreasing bottom -> top

    def next(self, price: int) -> int:
        span = 1                     # today counts itself
        while self.stack and self.stack[-1][0] <= price:
            span += self.stack.pop()[1]   # that day and everything it had already absorbed
        self.stack.append((price, span))
        return span
