class Solution {
public:
    // Three reversals: (A B) -> reverse all -> (B' A') -> reverse each part -> (B A).
    void rotate(vector<int>& nums, int k) {
        int n = nums.size();
        k %= n;                                   // rotating by n changes nothing
        reverse(nums.begin(), nums.end());        // the last k values are now at the front, backwards
        reverse(nums.begin(), nums.begin() + k);  // fix their order
        reverse(nums.begin() + k, nums.end());    // fix the order of the other n-k
    }

    // Cyclic replacements: carry each value to (i + k) % n until the cycle closes.
    void rotateCycles(vector<int>& nums, int k) {
        int n = nums.size();
        k %= n;
        if (k == 0) return;
        int moved = 0;                            // every value moves exactly once
        for (int start = 0; moved < n; start++) { // gcd(n, k) cycles, starting at 0, 1, 2, ...
            int cur = start, carry = nums[start];
            do {
                cur = (cur + k) % n;
                swap(carry, nums[cur]);           // drop the carried value, pick up the one it displaced
                moved++;
            } while (cur != start);
        }
    }
};

func rotate(nums []int, k int) {
	n := len(nums)
	k %= n            // rotating by n changes nothing
	reverse(nums)     // the last k values are now at the front, backwards
	reverse(nums[:k]) // fix their order
	reverse(nums[k:]) // fix the order of the other n-k
}

func reverse(a []int) { // reverse the slice in place
	for lo, hi := 0, len(a)-1; lo < hi; lo, hi = lo+1, hi-1 {
		a[lo], a[hi] = a[hi], a[lo]
	}
}

static class Solution {
    // Three reversals: (A B) -> reverse all -> (B' A') -> reverse each part -> (B A).
    public void rotate(int[] nums, int k) {
        int n = nums.length;
        k %= n;                          // rotating by n changes nothing
        reverse(nums, 0, n - 1);         // the last k values are now at the front, backwards
        reverse(nums, 0, k - 1);         // fix their order
        reverse(nums, k, n - 1);         // fix the order of the other n-k
    }

    private void reverse(int[] a, int lo, int hi) {   // reverse a[lo..hi] in place
        while (lo < hi) {
            int t = a[lo]; a[lo] = a[hi]; a[hi] = t;
            lo++; hi--;
        }
    }

    // Cyclic replacements: carry each value to (i + k) % n until the cycle closes.
    public void rotateCycles(int[] nums, int k) {
        int n = nums.length;
        k %= n;
        if (k == 0) return;

                cur = (cur + k) % n;
                int t = nums[cur]; nums[cur] = carry; carry = t;   // drop carried, pick up displaced
                moved++;
            } while (cur != start);
        }
    }
}
