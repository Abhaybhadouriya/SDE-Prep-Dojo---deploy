class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        return keepAtMost(nums, 2);
    }

    // General form: keep every value at most m times (m >= 1).
    int keepAtMost(vector<int>& nums, int m) {
        // nums[0..k-1] is the finished output: sorted, each value at most m times.
        int k = 0;
        for (int i = 0; i < (int)nums.size(); i++) {
            // The kept prefix is sorted, so nums[k-m] == nums[i] means
            // m copies of this value are already kept.
            if (k < m || nums[i] != nums[k - m]) nums[k++] = nums[i];
        }
        return k;
    }
};
