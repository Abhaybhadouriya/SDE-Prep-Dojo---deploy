class Solution {
public:
    int maxProfit(const vector<int>& prices) {
        int cheapest = prices[0];                 // best buy day for any later sell day
        int best = 0;                             // "no trade" gives 0, so never go below 0
        for (size_t i = 1; i < prices.size(); i++) {
            best = max(best, prices[i] - cheapest);   // sell today
            cheapest = min(cheapest, prices[i]);      // then let today become a buy candidate
        }
        return best;
    }
};
func maxProfit(prices []int) int {
	cheapest := prices[0] // best buy day for any later sell day
	best := 0             // "no trade" gives 0, so never go below 0
	for _, p := range prices[1:] {
		if p-cheapest > best { // sell today
			best = p - cheapest
		}
		if p < cheapest { // then today becomes a buy candidate
			cheapest = p
		}
	}
	return best
}

class Solution:
    def maxProfit(self, prices: List[int]) -> int:
        cheapest = prices[0]          # lowest price seen so far = best buy day for any later sell
        best = 0                      # no trade at all is allowed, so the answer is never below 0
        for p in prices[1:]:
            best = max(best, p - cheapest)   # sell today, bought at the cheapest earlier day
            cheapest = min(cheapest, p)      # update AFTER: buy and sell on the same day gains 0 anyway
        return best
