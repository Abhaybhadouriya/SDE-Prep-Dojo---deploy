In-place DFS with both prunings in four languages. Each file in coding-code/word-search/ also checks that the board is restored after every call, and cross-checks 3,000 random boards against a reference DFS without pruning.

class Solution {
    int m = 0, n = 0;

    // Does word[k..] start at (i, j)? Cells on the current path are marked '#'.
    bool dfs(vector<vector<char>>& b, const string& w, int i, int j, int k) {
        if (i < 0 || j < 0 || i >= m || j >= n || b[i][j] != w[k]) return false;
        if (k + 1 == (int)w.size()) return true;          // matched the last letter
        char saved = b[i][j];
        b[i][j] = '#';                                     // in use: no letter equals '#'
        bool found = dfs(b, w, i + 1, j, k + 1) || dfs(b, w, i - 1, j, k + 1) ||
                     dfs(b, w, i, j + 1, k + 1) || dfs(b, w, i, j - 1, k + 1);
        b[i][j] = saved;                                   // undo: free the cell for other paths
        return found;
    }

public:
    bool exist(vector<vector<char>>& board, string word) {
        m = board.size();
        n = board[0].size();
        if ((int)word.size() > m * n) return false;
        // Prune 1: the board must hold enough copies of every letter.
        int cnt[128] = {0};
        for (auto& row : board)
            for (char c : row) cnt[(int)c]++;
        int need[128] = {0};
        for (char c : word)
            if (++need[(int)c] > cnt[(int)c]) return false;
        // Prune 2: start from the rarer end, so fewer starts survive the first letter.
        if (cnt[(int)word.front()] > cnt[(int)word.back()]) reverse(word.begin(), word.end());
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (dfs(board, word, i, j, 0)) return true;
        return false;
    }
};
func exist(board [][]byte, word string) bool {
	m, n := len(board), len(board[0])
	if len(word) > m*n {
		return false
	}
	// Prune 1: the board must hold enough copies of every letter.
	var cnt, need [128]int
	for _, row := range board {
		for _, c := range row {
			cnt[c]++
		}
	}
	for i := 0; i < len(word); i++ {
		need[word[i]]++
		if need[word[i]] > cnt[word[i]] {
			return false
		}
	}
	// Prune 2: start from the rarer end, so fewer starts survive the first letter.
	w := []byte(word)
	if cnt[w[0]] > cnt[w[len(w)-1]] {
		for l, r := 0, len(w)-1; l < r; l, r = l+1, r-1 {
			w[l], w[r] = w[r], w[l]
		}
	}
	// Does w[k:] start at (i, j)? Cells on the current path hold '#'.
	var dfs func(i, j, k int) bool
	dfs = func(i, j, k int) bool {
		if i < 0 || j < 0 || i >= m || j >= n || board[i][j] != w[k] {
			return false
		}
		if k+1 == len(w) {
			return true // matched the last letter
		}
		saved := board[i][j]
		board[i][j] = '#' // in use: no letter equals '#'
		found := dfs(i+1, j, k+1) || dfs(i-1, j, k+1) || dfs(i, j+1, k+1) || dfs(i, j-1, k+1)
		board[i][j] = saved // undo: free the cell for other paths
		return found
	}
	for i := 0; i < m; i++ {
		for j := 0; j < n; j++ {
			if dfs(i, j, 0) {
				return true
			}
		}
	}
	return false
}

static class Solution {
    private char[][] b;
    private char[] w;
    private int m, n;

    public boolean exist(char[][] board, String word) {
        b = board;
        m = board.length;
        n = board[0].length;
        if (word.length() > m * n) return false;
        // Prune 1: the board must hold enough copies of every letter.
        int[] cnt = new int[128];
        for (char[] row : board)
            for (char c : row) cnt[c]++;
        int[] need = new int[128];
        for (char c : word.toCharArray())
            if (++need[c] > cnt[c]) return false;
        // Prune 2: start from the rarer end, so fewer starts survive the first letter.
        if (cnt[word.charAt(0)] > cnt[word.charAt(word.length() - 1)])
            word = new StringBuilder(word).reverse().toString();
        w = word.toCharArray();
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (dfs(i, j, 0)) return true;
        return false;
    }

    // Does w[k..] start at (i, j)? Cells on the current path are marked '#'.
    private boolean dfs(int i, int j, int k) {
        if (i < 0 || j < 0 || i >= m || j >= n || b[i][j] != w[k]) return false;
        if (k + 1 == w.length) return true;           // matched the last letter
        char saved = b[i][j];
        b[i][j] = '#';                                // in use: no letter equals '#'
        boolean found = dfs(i + 1, j, k + 1) || dfs(i - 1, j, k + 1)
                     || dfs(i, j + 1, k + 1) || dfs(i, j - 1, k + 1);
        b[i][j] = saved;                              // undo: free the cell for other paths
        return found;
    }
}
class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        m, n = len(board), len(board[0])
        if len(word) > m * n:
            return False
        # Prune 1: the board must hold enough copies of every letter.
        have = Counter(c for row in board for c in row)
        if any(have[c] < k for c, k in Counter(word).items()):
            return False
        # Prune 2: start from the rarer end, so fewer starts survive the first letter.
        if have[word[0]] > have[word[-1]]:
            word = word[::-1]

        def dfs(i: int, j: int, k: int) -> bool:
            # Does word[k:] start at (i, j)? Cells on the current path hold '#'.
            if i < 0 or j < 0 or i >= m or j >= n or board[i][j] != word[k]:
                return False
            if k + 1 == len(word):
                return True                      # matched the last letter
            saved, board[i][j] = board[i][j], "#"   # in use: no letter equals '#'
            found = (dfs(i + 1, j, k + 1) or dfs(i - 1, j, k + 1) or
                     dfs(i, j + 1, k + 1) or dfs(i, j - 1, k + 1))
            board[i][j] = saved                  # undo: free the cell for other paths
            return found

        return any(dfs(i, j, 0) for i in range(m) for j in range(n))
