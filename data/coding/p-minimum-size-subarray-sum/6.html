class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int n = nums.size(), best = n + 1;      // n + 1 = "no window found yet"
        int left = 0, window = 0;               // window = sum of nums[left..right]
        for (int right = 0; right < n; right++) {
            window += nums[right];
            while (window >= target) {          // valid: record it, then try a shorter one
                best = min(best, right - left + 1);
                window -= nums[left++];
            }
        }
        return best == n + 1 ? 0 : best;
    }
};
func minSubArrayLen(target int, nums []int) int {
	n := len(nums)
	best := n + 1        // n + 1 = "no window found yet"
	left, window := 0, 0 // window = sum of nums[left..right]
	for right, x := range nums {
		window += x
		for window >= target { // valid: record it, then try a shorter one
			if right-left+1 < best {
				best = right - left + 1
			}
			window -= nums[left]
			left++
		}
	}
	if best == n+1 {
		return 0
	}
	return best
}

    def minSubArrayLen(self, target: int, nums: List[int]) -> int:
        n = len(nums)
        best = n + 1                    # n + 1 = "no window found yet"
        left = 0
        window = 0                      # sum of nums[left..right]

                best = min(best, right - left + 1)
                window -= nums[left]
                left += 1
        return 0 if best == n + 1 else best
