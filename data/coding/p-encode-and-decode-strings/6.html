class Codec {
public:
    // Encodes a list of strings to a single string.
    string encode(vector<string>& strs) {
        string out;
        for (const string& s : strs) {
            out += to_string(s.size());   // header: decimal length ...
            out += '#';                   // ... ended by '#'
            out += s;                     // payload: copied raw, never inspected
        }
        return out;
    }

    // Decodes a single string to a list of strings.
    vector<string> decode(string s) {
        vector<string> out;
        size_t i = 0;
        while (i < s.size()) {
            size_t hash = s.find('#', i);           // the header is all digits, so the first '#' ends it
            size_t len = stoul(s.substr(i, hash - i));
            out.push_back(s.substr(hash + 1, len)); // take exactly len bytes, whatever they are
            i = hash + 1 + len;                     // jump to the next header
        }
        return out;
    }
};

func (codec *Codec) Encode(strs []string) string {
	var b strings.Builder
	for _, s := range strs {
		b.WriteString(strconv.Itoa(len(s))) // header: decimal byte length ...
		b.WriteByte('#')                    // ... ended by '#'
		b.WriteString(s)                    // payload: copied raw, never inspected
	}
	return b.String()
}

// Decode decodes a single string to a list of strings.
func (codec *Codec) Decode(s string) []string {
	out := []string{}
	for i := 0; i < len(s); {
		hash := i + strings.IndexByte(s[i:], '#') // the header is all digits, so the first '#' ends it
		n, _ := strconv.Atoi(s[i:hash])
		out = append(out, s[hash+1:hash+1+n]) // exactly n bytes, whatever they are
		i = hash + 1 + n                      // jump to the next header
	}
	return out
}

class Codec:
    def encode(self, strs: List[str]) -> str:
        """Encodes a list of strings to a single string."""
        return "".join(f"{len(s)}#{s}" for s in strs)    # header "<len>#", then the raw payload

    def decode(self, s: str) -> List[str]:
        """Decodes a single string to a list of strings."""
        out, i = [], 0
        while i < len(s):
            hash_at = s.index("#", i)        # the header is all digits, so the first '#' ends it
            length = int(s[i:hash_at])
            start = hash_at + 1
            out.append(s[start:start + length])   # exactly `length` chars, whatever they are
            i = start + length               # jump to the next header
        return out
