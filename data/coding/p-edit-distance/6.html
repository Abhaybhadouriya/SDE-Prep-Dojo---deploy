class Solution {
public:
    int minDistance(string word1, string word2) {
        int m = word1.size(), n = word2.size();
        vector<int> prev(n + 1), cur(n + 1);
        for (int j = 0; j <= n; j++) prev[j] = j;  // "" -> word2[0..j) needs j inserts
        for (int i = 1; i <= m; i++) {
            cur[0] = i;                             // word1[0..i) -> "" needs i deletes
            for (int j = 1; j <= n; j++) {
                if (word1[i - 1] == word2[j - 1])
                    cur[j] = prev[j - 1];           // last chars match: free
                else
                    cur[j] = 1 + min({prev[j - 1],  // replace word1[i-1] by word2[j-1]
                                      prev[j],      // delete word1[i-1]
                                      cur[j - 1]}); // insert word2[j-1]
            }
            swap(prev, cur);
        }
        return prev[n];
    }
};
func minDistance(word1 string, word2 string) int {
	m, n := len(word1), len(word2)
	prev, cur := make([]int, n+1), make([]int, n+1)
	for j := 0; j <= n; j++ {
		prev[j] = j // "" -> word2[0:j] needs j inserts
	}
	for i := 1; i <= m; i++ {
		cur[0] = i // word1[0:i] -> "" needs i deletes
		for j := 1; j <= n; j++ {
			if word1[i-1] == word2[j-1] {
				cur[j] = prev[j-1] // last chars match: free
			} else {
				// replace, delete, insert
				cur[j] = 1 + min3(prev[j-1], prev[j], cur[j-1])
			}
		}
		prev, cur = cur, prev
	}
	return prev[n]
}

func min3(a, b, c int) int {
	if b < a {
		a = b
	}
	if c < a {
		a = c
	}
	return a
}

        for (int i = 1; i <= m; i++) {
            cur[0] = i;                             // word1[0..i) -> "" needs i deletes
            for (int j = 1; j <= n; j++) {
                if (word1.charAt(i - 1) == word2.charAt(j - 1))
                    cur[j] = prev[j - 1];           // last chars match: free
                else
                    cur[j] = 1 + Math.min(prev[j - 1],             // replace
                                 Math.min(prev[j], cur[j - 1]));   // delete, insert
            }
            int[] t = prev; prev = cur; cur = t;
        }
        return prev[n];
    }
}
class Solution:
    def minDistance(self, word1: str, word2: str) -> int:
        m, n = len(word1), len(word2)
        prev = list(range(n + 1))      # row i = 0: "" -> word2[:j] needs j inserts
        for i in range(1, m + 1):
            cur = [i] + [0] * n        # word1[:i] -> "" needs i deletes
            for j in range(1, n + 1):
                if word1[i - 1] == word2[j - 1]:
                    cur[j] = prev[j - 1]               # last chars match: free
                else:
                    cur[j] = 1 + min(prev[j - 1],      # replace word1[i-1] by word2[j-1]
                                     prev[j],          # delete word1[i-1]
                                     cur[j - 1])       # insert word2[j-1]
            prev = cur
        return prev[n]
