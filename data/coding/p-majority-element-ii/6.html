All four use the two-slot vote and a verification pass. The two candidates start with different placeholder values so the verification can never report the same value twice.

class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        // At most two values can appear more than n/3 times.
        // Pass 1: Boyer-Moore with two slots. A value matching neither slot
        // cancels one copy of each candidate (a triple of distinct values).
        int cand1 = 0, cand2 = 1, cnt1 = 0, cnt2 = 0;   // different start values
        for (int x : nums) {
            if (x == cand1)      ++cnt1;
            else if (x == cand2) ++cnt2;
            else if (cnt1 == 0)  { cand1 = x; cnt1 = 1; }
            else if (cnt2 == 0)  { cand2 = x; cnt2 = 1; }
            else                 { --cnt1; --cnt2; }    // throw away a triple
        }
        // Pass 2: the survivors are only candidates; count them for real.
        cnt1 = cnt2 = 0;
        for (int x : nums) {
            if (x == cand1)      ++cnt1;
            else if (x == cand2) ++cnt2;
        }
        vector<int> result;
        int limit = (int)nums.size() / 3;
        if (cnt1 > limit) result.push_back(cand1);
        if (cnt2 > limit) result.push_back(cand2);
        return result;
    }
};
func majorityElement(nums []int) []int {
	// At most two values can appear more than n/3 times.
	// Pass 1: Boyer-Moore with two slots. A value matching neither slot
	// cancels one copy of each candidate (a triple of distinct values).
	cand1, cand2, cnt1, cnt2 := 0, 1, 0, 0 // different start values
	for _, x := range nums {
		switch {
		case x == cand1:
			cnt1++
		case x == cand2:
			cnt2++
		case cnt1 == 0:
			cand1, cnt1 = x, 1
		case cnt2 == 0:
			cand2, cnt2 = x, 1
		default:
			cnt1-- // throw away a triple
			cnt2--
		}
	}
	// Pass 2: the survivors are only candidates; count them for real.
	cnt1, cnt2 = 0, 0
	for _, x := range nums {
		if x == cand1 {
			cnt1++
		} else if x == cand2 {
			cnt2++
		}
	}
	result := []int{}
	limit := len(nums) / 3
	if cnt1 > limit {
		result = append(result, cand1)
	}
	if cnt2 > limit {
		result = append(result, cand2)
	}
	return result
}

class Solution:
    def majorityElement(self, nums: List[int]) -> List[int]:
        # At most two values can appear more than n/3 times.
        # Pass 1: Boyer-Moore with two slots. A value matching neither slot
        # cancels one copy of each candidate (a triple of distinct values).
        cand1, cand2, cnt1, cnt2 = 0, 1, 0, 0       # different start values
        for x in nums:
            if x == cand1:
                cnt1 += 1
            elif x == cand2:
                cnt2 += 1
            elif cnt1 == 0:
                cand1, cnt1 = x, 1
            elif cnt2 == 0:
                cand2, cnt2 = x, 1
            else:
                cnt1 -= 1                            # throw away a triple
                cnt2 -= 1
        # Pass 2: the survivors are only candidates; count them for real.
        limit = len(nums) // 3
        return [c for c in (cand1, cand2) if nums.count(c) > limit]
