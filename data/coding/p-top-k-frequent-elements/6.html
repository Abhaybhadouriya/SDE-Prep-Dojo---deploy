Bucket-sort solution in four languages. Each file in coding-code/top-k-frequent-elements/ also checks 300 random arrays: the counts of the returned values must equal the k largest counts.

class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        int n = nums.size();
        unordered_map<int, int> freq;               // value -> how many times it appears
        for (int x : nums) freq[x]++;

        // bucket[f] holds every value seen exactly f times; f can never exceed n
        vector<vector<int>> bucket(n + 1);
        for (auto& kv : freq) bucket[kv.second].push_back(kv.first);

        vector<int> top;                            // walk from the highest count down
        for (int f = n; f >= 1 && (int)top.size() < k; f--) {
            for (int x : bucket[f]) {
                top.push_back(x);
                if ((int)top.size() == k) break;    // the answer is unique, so no tie is split here
            }
        }
        return top;
    }
};
func topKFrequent(nums []int, k int) []int {
	n := len(nums)
	freq := make(map[int]int) // value -> how many times it appears
	for _, x := range nums {
		freq[x]++
	}

	// bucket[f] holds every value seen exactly f times; f can never exceed n
	bucket := make([][]int, n+1)
	for value, f := range freq {
		bucket[f] = append(bucket[f], value)
	}

	top := make([]int, 0, k) // walk from the highest count down
	for f := n; f >= 1 && len(top) < k; f-- {
		for _, value := range bucket[f] {
			top = append(top, value)
			if len(top) == k { // the answer is unique, so no tie is split here
				return top
			}
		}
	}
	return top
}

static class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        int n = nums.length;
        Map<Integer, Integer> freq = new HashMap<>();   // value -> how many times it appears
        for (int x : nums) freq.merge(x, 1, Integer::sum);

        // bucket[f] holds every value seen exactly f times; f can never exceed n
        List<List<Integer>> bucket = new ArrayList<>(n + 1);
        for (int f = 0; f <= n; f++) bucket.add(new ArrayList<>());
        for (Map.Entry<Integer, Integer> e : freq.entrySet()) bucket.get(e.getValue()).add(e.getKey());

        int[] top = new int[k];
        int filled = 0;                                  // walk from the highest count down
        for (int f = n; f >= 1 && filled < k; f--) {
            for (int x : bucket.get(f)) {
                top[filled++] = x;
                if (filled == k) break;                  // the answer is unique, so no tie is split here
            }
        }
        return top;
    }
}
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        freq = Counter(nums)                     # value -> how many times it appears
        # bucket[f] holds every value seen exactly f times; f can never exceed len(nums)
        bucket = [[] for _ in range(len(nums) + 1)]
        for value, f in freq.items():
            bucket[f].append(value)

        top = []                                 # walk from the highest count down
        for f in range(len(nums), 0, -1):
            for value in bucket[f]:
                top.append(value)
                if len(top) == k:                # the answer is unique, so no tie is split here
                    return top
        return top
