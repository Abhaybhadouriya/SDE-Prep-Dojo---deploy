The loop + skip + break version in four languages. The tested files in coding-code/combination-sum-ii/ check that no combination appears twice and compare 300 random inputs with an all-subsets brute force; one case has a hundred equal cards.

func combinationSum2(candidates []int, target int) [][]int {
	sort.Ints(candidates) // equal values side by side, small first
	var found [][]int
	var pick []int

	// choose the NEXT element of the combination from candidates[start:]; left = target still to reach
	var dfs func(start, left int)
	dfs = func(start, left int) {
		if left == 0 {
			found = append(found, append([]int(nil), pick...)) // copy: pick keeps changing
			return
		}
		for i := start; i < len(candidates); i++ {
			if i > start && candidates[i] == candidates[i-1] { // this value was already tried in this slot
				continue
			}
			if candidates[i] > left { // sorted: every later value is too big as well
				break
			}
			pick = append(pick, candidates[i])
			dfs(i+1, left-candidates[i]) // i + 1: each card is used at most once
			pick = pick[:len(pick)-1]
		}
	}
	dfs(0, target)
	return found
}
class Solution {
public:
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());   // equal values side by side, small first
        vector<vector<int>> found;
        vector<int> pick;
        dfs(candidates, 0, target, pick, found);
        return found;
    }

private:
    // choose the NEXT element of the combination from a[start..]; `left` = target still to reach
    void dfs(const vector<int>& a, int start, int left, vector<int>& pick, vector<vector<int>>& found) {
        if (left == 0) { found.push_back(pick); return; }
        for (int i = start; i < (int)a.size(); i++) {
            if (i > start && a[i] == a[i - 1]) continue;   // this value was already tried in this slot
            if (a[i] > left) break;                         // sorted: every later value is too big as well
            pick.push_back(a[i]);
            dfs(a, i + 1, left - a[i], pick, found);        // i + 1: each card is used at most once
            pick.pop_back();
        }
    }
};

static class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);                         // equal values side by side, small first
        List<List<Integer>> found = new ArrayList<>();
        dfs(candidates, 0, target, new ArrayList<>(), found);
        return found;
    }

    // choose the NEXT element of the combination from a[start..]; `left` = target still to reach
    private void dfs(int[] a, int start, int left, List<Integer> pick, List<List<Integer>> found) {
        if (left == 0) { found.add(new ArrayList<>(pick)); return; }
        for (int i = start; i < a.length; i++) {
            if (i > start && a[i] == a[i - 1]) continue;  // this value was already tried in this slot
            if (a[i] > left) break;                        // sorted: every later value is too big as well
            pick.add(a[i]);
            dfs(a, i + 1, left - a[i], pick, found);       // i + 1: each card is used at most once
            pick.remove(pick.size() - 1);
        }
    }
}
class Solution:
    def combinationSum2(self, candidates: List[int], target: int) -> List[List[int]]:
        a = sorted(candidates)                       # equal values side by side, small first
        found, pick = [], []

        def dfs(start: int, left: int) -> None:
            # choose the NEXT element of the combination from a[start:]; left = target still to reach
            if left == 0:
                found.append(pick[:])
                return
            for i in range(start, len(a)):
                if i > start and a[i] == a[i - 1]:   # this value was already tried in this slot
                    continue
                if a[i] > left:                      # sorted: every later value is too big as well
                    break
                pick.append(a[i])
                dfs(i + 1, left - a[i])              # i + 1: each card is used at most once
                pick.pop()

        dfs(0, target)
        return found
