Iterative square and multiply in four languages. Each file in coding-code/powx-n/ checks the examples, the four -231 / 231-1 traps and 500 random (x, n) pairs against the library pow.

class Solution {
public:
    double myPow(double x, int n) {
        long long e = n;              // widen first: -INT_MIN does not fit in an int
        if (e < 0) {                  // x^-k = (1/x)^k
            x = 1 / x;
            e = -e;
        }
        double result = 1.0;
        while (e > 0) {
            if (e & 1) result *= x;   // this bit is set: fold in the current square
            x *= x;                   // x, x^2, x^4, x^8 ... one square per bit
            e >>= 1;
        }
        return result;
    }
};

class Solution:
    def myPow(self, x: float, n: int) -> float:
        e = n
        if e < 0:                 # x^-k = (1/x)^k; Python ints never overflow, but keep the C shape
            x, e = 1 / x, -e
        result = 1.0
        while e:
            if e & 1:             # this bit of the exponent is set: fold in the current square
                result *= x
            x *= x                # x, x^2, x^4, x^8 ... one square per bit
            e >>= 1
        return result
