Pivot-successor-reverse in four languages. The C++ file checks 2,000 random arrays with repeats against std::next_permutation; the Java and Python files check against a brute force that lists all distinct orderings.

class Solution {
public:
    void nextPermutation(vector<int>& nums) {
        int n = nums.size();
        // 1. pivot = rightmost i with nums[i] < nums[i+1]; everything after it is non-increasing
        int i = n - 2;
        while (i >= 0 && nums[i] >= nums[i + 1]) i--;
        if (i >= 0) {
            // 2. successor = rightmost element of the suffix that is strictly bigger than the pivot
            int j = n - 1;
            while (nums[j] <= nums[i]) j--;
            swap(nums[i], nums[j]);   // the suffix stays non-increasing after this swap
        }
        // 3. make the suffix as small as possible: non-increasing -> non-decreasing
        //    (when there was no pivot, i = -1 and this reverses the whole array)
        int l = i + 1, r = n - 1;
        while (l < r) swap(nums[l++], nums[r--]);
    }
};
func nextPermutation(nums []int) {
	n := len(nums)
	// 1. pivot = rightmost i with nums[i] < nums[i+1]; everything after it is non-increasing
	i := n - 2
	for i >= 0 && nums[i] >= nums[i+1] {
		i--
	}
	if i >= 0 {
		// 2. successor = rightmost element of the suffix that is strictly bigger than the pivot
		j := n - 1
		for nums[j] <= nums[i] {
			j--
		}
		nums[i], nums[j] = nums[j], nums[i] // the suffix stays non-increasing after this swap
	}
	// 3. make the suffix as small as possible: non-increasing -> non-decreasing
	//    (when there was no pivot, i = -1 and this reverses the whole slice)
	for l, r := i+1, n-1; l < r; l, r = l+1, r-1 {
		nums[l], nums[r] = nums[r], nums[l]
	}
}

static class Solution {
    public void nextPermutation(int[] nums) {
        int n = nums.length;

        int i = n - 2;

        if (i >= 0) {
            // 2. successor = rightmost element of the suffix that is strictly bigger than the pivot
            int j = n - 1;
            while (nums[j] <= nums[i]) j--;
            swap(nums, i, j);   // the suffix stays non-increasing after this swap
        }
        // 3. make the suffix as small as possible: non-increasing -> non-decreasing
        //    (when there was no pivot, i = -1 and this reverses the whole array)
        for (int l = i + 1, r = n - 1; l < r; l++, r--) swap(nums, l, r);
    }

    private static void swap(int[] a, int x, int y) {
        int t = a[x];
        a[x] = a[y];
        a[y] = t;
    }
}
