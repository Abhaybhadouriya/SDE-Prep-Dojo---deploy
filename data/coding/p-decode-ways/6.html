All four use the two-variable bottom-up DP.

func numDecodings(s string) int {
	// ways(i) = number of decodings of the first i digits.
	prev2 := 1 // ways(i-2); ways(0) = 1 (empty prefix)
	prev1 := 0 // ways(i-1); ways(1)
	if s[0] != '0' {
		prev1 = 1
	}
	for i := 2; i <= len(s); i++ {
		cur := 0
		if s[i-1] != '0' { // last digit alone: 1..9
			cur += prev1
		}
		two := int(s[i-2]-'0')*10 + int(s[i-1]-'0')
		if two >= 10 && two <= 26 { // last two digits: 10..26
			cur += prev2
		}
		prev2, prev1 = prev1, cur
	}
	return prev1
}
class Solution {
public:
    int numDecodings(const string& s) {
        // ways(i) = number of decodings of the first i digits.
        int prev2 = 1;                          // ways(i-2); ways(0) = 1 (empty prefix)
        int prev1 = (s[0] != '0') ? 1 : 0;      // ways(i-1); ways(1)
        for (size_t i = 2; i <= s.size(); ++i) {
            int cur = 0;
            if (s[i - 1] != '0') cur += prev1;              // last digit alone: 1..9
            int two = (s[i - 2] - '0') * 10 + (s[i - 1] - '0');
            if (two >= 10 && two <= 26) cur += prev2;       // last two digits: 10..26
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
};
static class Solution {
    public int numDecodings(String s) {
        // ways(i) = number of decodings of the first i digits.
        int prev2 = 1;                              // ways(i-2); ways(0) = 1 (empty prefix)
        int prev1 = s.charAt(0) != '0' ? 1 : 0;     // ways(i-1); ways(1)
        for (int i = 2; i <= s.length(); i++) {

            if (two >= 10 && two <= 26) cur += prev2;                       // last two digits: 10..26
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
}
class Solution:
    def numDecodings(self, s: str) -> int:
        # ways(i) = number of decodings of the first i digits.
        prev2 = 1                              # ways(i-2); ways(0) = 1 (empty prefix)
        prev1 = 1 if s[0] != "0" else 0        # ways(i-1); ways(1)
        for i in range(2, len(s) + 1):
            cur = 0
            if s[i - 1] != "0":                # last digit alone: 1..9
                cur += prev1
            if 10 <= int(s[i - 2:i]) <= 26:    # last two digits: 10..26
                cur += prev2
            prev2, prev1 = prev1, cur
        return prev1
