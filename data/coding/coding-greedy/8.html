Reusable templates in four languages

One file per language with the six shapes from this page, each 5 to 15 lines. All four files (Go, C++, Java, Python) were compiled and run with the asserts in main (examples and edge cases: a single element, an empty string, all zeros, touching intervals).

What is in each file (same names in every language; Go and Java use camelCase):

  • can_jump - Reachability: can we stand on the last index?
  • min_jumps - Fewest jumps: BFS by "levels" of reach, O(n)
  • can_complete_circuit - Gas station: one pass, reset the start when the tank goes negative
  • candy - Two-pass: candy (each pass fixes one neighbour rule)
  • check_valid_string - Balance counter as a range [lo, hi] of possible open counts
  • largest_number - Comparator greedy: order by "which concatenation is bigger"
  • erase_overlap_intervals - Exchange argument classic: keep intervals that end earliest

package main

import (
	"fmt"
	"math"
	"sort"
	"strconv"
	"strings"
)

// 1. Reachability: can we stand on the last index?
func canJump(nums []int) bool {
	reach := 0 // furthest index reachable so far
	for i, step := range nums {
		if i > reach { // i is past everything we can reach: stuck
			return false
		}
		if i+step > reach {
			reach = i + step
		}
	}
	return true
}

// 1b. Fewest jumps: BFS by "levels" of reach, O(n)
func minJumps(nums []int) int {
	jumps, curEnd, far := 0, 0, 0
	for i := 0; i+1 < len(nums); i++ { // never jump FROM the last index
		if i+nums[i] > far {
			far = i + nums[i]
		}
		if i == curEnd { // finished this level: one more jump
			jumps++
			curEnd = far
		}
	}
	return jumps
}

// 2. Gas station: one pass, reset the start when the tank goes negative
func canCompleteCircuit(gas, cost []int) int {
	total, tank, start := 0, 0, 0
	for i := range gas {
		d := gas[i] - cost[i]
		total += d
		tank += d
		if tank < 0 { // no station in [start..i] can be the start
			start, tank = i+1, 0
		}
	}
	if total < 0 {
		return -1
	}
	return start
}

// 3. Two-pass: candy (each pass fixes one neighbour rule)
func candy(r []int) int {
	n := len(r)
	c := make([]int, n)
	for i := range c {
		c[i] = 1
	}
	for i := 1; i < n; i++ { // left neighbour rule
		if r[i] > r[i-1] {
			c[i] = c[i-1] + 1
		}
	}
	for i := n - 2; i >= 0; i-- { // right neighbour rule, keep the left result
		if r[i] > r[i+1] && c[i+1]+1 > c[i] {
			c[i] = c[i+1] + 1
		}
	}
	sum := 0
	for _, x := range c {
		sum += x
	}
	return sum
}

// 4. Balance counter as a range [lo, hi] of possible open counts
func checkValidString(s string) bool {
	lo, hi := 0, 0
	for _, ch := range s {
		switch ch {
		case '(':
			lo, hi = lo+1, hi+1
		case ')':
			lo, hi = lo-1, hi-1
		default: // '*' may be ')', '' or '('
			lo, hi = lo-1, hi+1
		}
		if hi < 0 { // even the most generous reading has too many ')'
			return false
		}
		if lo < 0 { // open count can never be negative
			lo = 0
		}
	}
	return lo == 0
}

// 5. Comparator greedy: order by "which concatenation is bigger"
func largestNumber(nums []int) string {
	s := make([]string, len(nums))
	for i, x := range nums {
		s[i] = strconv.Itoa(x)
	}
	sort.Slice(s, func(i, j int) bool { return s[i]+s[j] > s[j]+s[i] })
	if s[0] == "0" {
		return "0"
	}
	return strings.Join(s, "")
}

// 6. Exchange argument classic: keep intervals that end earliest
func eraseOverlapIntervals(iv [][]int) int {
	a := append([][]int(nil), iv...)
	sort.Slice(a, func(i, j int) bool { return a[i][1] < a[j][1] })
	kept, end := 0, math.MinInt64
	for _, x := range a {
		if x[0] >= end { // touching ends do not overlap
			kept++
			end = x[1]
		}
	}
	return len(a) - kept
}

func check(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

func main() {
	check(canJump([]int{2, 3, 1, 1, 4}) && !canJump([]int{3, 2, 1, 0, 4}) && canJump([]int{0}), "canJump")
	check(minJumps([]int{2, 3, 1, 1, 4}) == 2 && minJumps([]int{0}) == 0 && minJumps([]int{1, 1, 1, 1}) == 3, "minJumps")
	check(canCompleteCircuit([]int{1, 2, 3, 4, 5}, []int{3, 4, 5, 1, 2}) == 3, "gas 1")
	check(canCompleteCircuit([]int{2, 3, 4}, []int{3, 4, 3}) == -1, "gas 2")
	check(candy([]int{1, 0, 2}) == 5 && candy([]int{1, 2, 2}) == 4 && candy([]int{1, 3, 4, 5, 2}) == 11 && candy([]int{2, 4, 3, 2, 1}) == 11, "candy")
	check(checkValidString("(*))") && checkValidString("(*)") && checkValidString(""), "valid 1")
	check(!checkValidString(")(") && !checkValidString("((*"), "valid 2")
	check(largestNumber([]int{3, 30, 34, 5, 9}) == "9534330", "largest 1")
	check(largestNumber([]int{0, 0}) == "0" && largestNumber([]int{10, 2}) == "210", "largest 2")
	check(eraseOverlapIntervals([][]int{{1, 2}, {2, 3}, {3, 4}, {1, 3}}) == 1, "erase 1")
	check(eraseOverlapIntervals([][]int{{1, 2}, {1, 2}, {1, 2}}) == 2, "erase 2")
	check(eraseOverlapIntervals([][]int{{1, 2}, {2, 3}}) == 0, "erase 3")

using namespace std;

// 1. Reachability: can we stand on the last index?
bool canJump(const vector<int>& nums) {
    int reach = 0;                              // furthest index reachable so far
    for (int i = 0; i < (int)nums.size(); ++i) {
        if (i > reach) return false;            // i is past everything we can reach: stuck
        reach = max(reach, i + nums[i]);
    }
    return true;
}

// 1b. Fewest jumps: BFS by "levels" of reach, O(n)
int minJumps(const vector<int>& nums) {
    int jumps = 0, curEnd = 0, far = 0;
    for (int i = 0; i + 1 < (int)nums.size(); ++i) {   // never jump FROM the last index
        far = max(far, i + nums[i]);
        if (i == curEnd) { ++jumps; curEnd = far; }     // finished this level: one more jump
    }
    return jumps;
}

// 2. Gas station: one pass, reset the start when the tank goes negative
int canCompleteCircuit(const vector<int>& gas, const vector<int>& cost) {
    int total = 0, tank = 0, start = 0;
    for (int i = 0; i < (int)gas.size(); ++i) {
        int d = gas[i] - cost[i];
        total += d;
        tank += d;
        if (tank < 0) { start = i + 1; tank = 0; }      // no station in [start..i] can be the start
    }
    return total >= 0 ? start : -1;
}

// 3. Two-pass: candy (each pass fixes one neighbour rule)
int candy(const vector<int>& r) {
    int n = r.size();
    vector<int> c(n, 1);
    for (int i = 1; i < n; ++i)                 // left neighbour rule
        if (r[i] > r[i - 1]) c[i] = c[i - 1] + 1;
    for (int i = n - 2; i >= 0; --i)            // right neighbour rule, keep the left result
        if (r[i] > r[i + 1]) c[i] = max(c[i], c[i + 1] + 1);
    return accumulate(c.begin(), c.end(), 0);
}

// 4. Balance counter as a range [lo, hi] of possible open counts
bool checkValidString(const string& s) {
    int lo = 0, hi = 0;
    for (char ch : s) {
        if (ch == '(')      { ++lo; ++hi; }
        else if (ch == ')') { --lo; --hi; }
        else                { --lo; ++hi; }     // '*' may be ')', '' or '('
        if (hi < 0) return false;               // even the most generous reading has too many ')'
        lo = max(lo, 0);                        // open count can never be negative
    }
    return lo == 0;
}

// 5. Comparator greedy: order by "which concatenation is bigger"
string largestNumber(const vector<int>& nums) {
    vector<string> s;
    for (int x : nums) s.push_back(to_string(x));
    sort(s.begin(), s.end(), [](const string& a, const string& b) { return a + b > b + a; });
    if (s[0] == "0") return "0";
    string out;
    for (auto& t : s) out += t;
    return out;
}

// 6. Exchange argument classic: keep intervals that end earliest
int eraseOverlapIntervals(vector<vector<int>> iv) {
    sort(iv.begin(), iv.end(), [](const vector<int>& a, const vector<int>& b) { return a[1] < b[1]; });
    int kept = 0;
    long long end = LLONG_MIN;
    for (auto& x : iv)
        if (x[0] >= end) { ++kept; end = x[1]; }   // touching ends do not overlap
    return (int)iv.size() - kept;
}

import java.util.*;

public class Greedy {

    // 1. Reachability: can we stand on the last index?
    static boolean canJump(int[] nums) {
        int reach = 0;                              // furthest index reachable so far
        for (int i = 0; i < nums.length; i++) {
            if (i > reach) return false;            // i is past everything we can reach: stuck
            reach = Math.max(reach, i + nums[i]);
        }
        return true;
    }

    // 1b. Fewest jumps: BFS by "levels" of reach, O(n)
    static int minJumps(int[] nums) {
        int jumps = 0, curEnd = 0, far = 0;
        for (int i = 0; i + 1 < nums.length; i++) {     // never jump FROM the last index
            far = Math.max(far, i + nums[i]);
            if (i == curEnd) { jumps++; curEnd = far; } // finished this level: one more jump
        }
        return jumps;
    }

    // 2. Gas station: one pass, reset the start when the tank goes negative
    static int canCompleteCircuit(int[] gas, int[] cost) {
        int total = 0, tank = 0, start = 0;
        for (int i = 0; i < gas.length; i++) {
            int d = gas[i] - cost[i];
            total += d;
            tank += d;
            if (tank < 0) { start = i + 1; tank = 0; }  // no station in [start..i] can be the start
        }
        return total >= 0 ? start : -1;
    }

    // 3. Two-pass: candy (each pass fixes one neighbour rule)
    static int candy(int[] r) {
        int n = r.length;
        int[] c = new int[n];
        Arrays.fill(c, 1);
        for (int i = 1; i < n; i++)                 // left neighbour rule
            if (r[i] > r[i - 1]) c[i] = c[i - 1] + 1;
        for (int i = n - 2; i >= 0; i--)            // right neighbour rule, keep the left result
            if (r[i] > r[i + 1]) c[i] = Math.max(c[i], c[i + 1] + 1);
        int sum = 0;
        for (int x : c) sum += x;
        return sum;
    }

    // 4. Balance counter as a range [lo, hi] of possible open counts
    static boolean checkValidString(String s) {
        int lo = 0, hi = 0;
        for (char ch : s.toCharArray()) {
            if (ch == '(')      { lo++; hi++; }
            else if (ch == ')') { lo--; hi--; }
            else                { lo--; hi++; }     // '*' may be ')', '' or '('
            if (hi < 0) return false;               // even the most generous reading has too many ')'
            lo = Math.max(lo, 0);                   // open count can never be negative
        }
        return lo == 0;
    }

    // 5. Comparator greedy: order by "which concatenation is bigger"
    static String largestNumber(int[] nums) {
        String[] s = new String[nums.length];
        for (int i = 0; i < nums.length; i++) s[i] = String.valueOf(nums[i]);
        Arrays.sort(s, (a, b) -> (b + a).compareTo(a + b));
        if (s[0].equals("0")) return "0";
        return String.join("", s);
    }

    // 6. Exchange argument classic: keep intervals that end earliest
    static int eraseOverlapIntervals(int[][] iv) {
        int[][] a = iv.clone();
        Arrays.sort(a, (x, y) -> Integer.compare(x[1], y[1]));
        int kept = 0;
        long end = Long.MIN_VALUE;
        for (int[] x : a)
            if (x[0] >= end) { kept++; end = x[1]; }   // touching ends do not overlap
        return a.length - kept;
    }

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError(what);
    }

    public static void main(String[] args) {
        check(canJump(new int[]{2, 3, 1, 1, 4}) && !canJump(new int[]{3, 2, 1, 0, 4}) && canJump(new int[]{0}), "canJump");
        check(minJumps(new int[]{2, 3, 1, 1, 4}) == 2 && minJumps(new int[]{0}) == 0 && minJumps(new int[]{1, 1, 1, 1}) == 3, "minJumps");
        check(canCompleteCircuit(new int[]{1, 2, 3, 4, 5}, new int[]{3, 4, 5, 1, 2}) == 3, "gas 1");
        check(canCompleteCircuit(new int[]{2, 3, 4}, new int[]{3, 4, 3}) == -1, "gas 2");
        check(candy(new int[]{1, 0, 2}) == 5 && candy(new int[]{1, 2, 2}) == 4 && candy(new int[]{1, 3, 4, 5, 2}) == 11 && candy(new int[]{2, 4, 3, 2, 1}) == 11, "candy");
        check(checkValidString("(*))") && checkValidString("(*)") && checkValidString(""), "valid 1");
        check(!checkValidString(")(") && !checkValidString("((*"), "valid 2");
        check(largestNumber(new int[]{3, 30, 34, 5, 9}).equals("9534330"), "largest 1");
        check(largestNumber(new int[]{0, 0}).equals("0") && largestNumber(new int[]{10, 2}).equals("210"), "largest 2");
        check(eraseOverlapIntervals(new int[][]{{1, 2}, {2, 3}, {3, 4}, {1, 3}}) == 1, "erase 1");
        check(eraseOverlapIntervals(new int[][]{{1, 2}, {1, 2}, {1, 2}}) == 2, "erase 2");
        check(eraseOverlapIntervals(new int[][]{{1, 2}, {2, 3}}) == 0, "erase 3");
        System.out.println("Greedy.java: all tests passed");
    }
}
from functools import cmp_to_key


# 1. Reachability: can we stand on the last index?
def can_jump(nums):
    reach = 0                          # furthest index reachable so far
    for i, step in enumerate(nums):
        if i > reach:                  # i is past everything we can reach: stuck
            return False
        reach = max(reach, i + step)
    return True


# 1b. Fewest jumps: BFS by "levels" of reach, O(n)
def min_jumps(nums):
    jumps, cur_end, far = 0, 0, 0
    for i in range(len(nums) - 1):     # never jump FROM the last index
        far = max(far, i + nums[i])
        if i == cur_end:               # finished this level: one more jump
            jumps += 1
            cur_end = far
    return jumps


# 2. Gas station: one pass, reset the start when the tank goes negative
def can_complete_circuit(gas, cost):
    total = tank = start = 0
    for i in range(len(gas)):
        d = gas[i] - cost[i]
        total += d
        tank += d
        if tank < 0:                   # no station in [start..i] can be the start
            start, tank = i + 1, 0
    return start if total >= 0 else -1


# 3. Two-pass: candy (each pass fixes one neighbour rule)
def candy(ratings):
    n = len(ratings)
    c = [1] * n
    for i in range(1, n):              # left neighbour rule
        if ratings[i] > ratings[i - 1]:
            c[i] = c[i - 1] + 1
    for i in range(n - 2, -1, -1):     # right neighbour rule, keep the left result
        if ratings[i] > ratings[i + 1]:
            c[i] = max(c[i], c[i + 1] + 1)
    return sum(c)


# 4. Balance counter as a range [lo, hi] of possible open counts
def check_valid_string(s):
    lo = hi = 0
    for ch in s:
        if ch == '(':
            lo, hi = lo + 1, hi + 1
        elif ch == ')':
            lo, hi = lo - 1, hi - 1
        else:                          # '*' may be ')', '' or '('
            lo, hi = lo - 1, hi + 1
        if hi < 0:                     # even the most generous reading has too many ')'
            return False
        lo = max(lo, 0)                # open count can never be negative
    return lo == 0


# 5. Comparator greedy: order by "which concatenation is bigger"
def largest_number(nums):
    s = [str(x) for x in nums]
    s.sort(key=cmp_to_key(lambda a, b: -1 if a + b > b + a else (1 if a + b < b + a else 0)))
    return "0" if s[0] == "0" else "".join(s)


# 6. Exchange argument classic: keep intervals that end earliest
def erase_overlap_intervals(iv):
    iv = sorted(iv, key=lambda x: x[1])  # sort by END, not start
    kept, end = 0, float("-inf")
    for a, b in iv:
        if a >= end:                   # touching ends do not overlap
            kept += 1
            end = b
    return len(iv) - kept
