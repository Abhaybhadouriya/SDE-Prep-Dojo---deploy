class Solution {
    // Returns the sum of all numbers on root-to-leaf paths below node, given the prefix number above it.
    int dfs(TreeNode* node, int prefix) {
        if (!node) return 0;                          // an empty side contributes nothing
        int cur = prefix * 10 + node->val;            // append this digit
        if (!node->left && !node->right) return cur;  // a leaf ends exactly one number
        return dfs(node->left, cur) + dfs(node->right, cur);
    }

public:
    int sumNumbers(TreeNode* root) { return dfs(root, 0); }
};

static class Solution {

    // Returns the sum of all numbers on root-to-leaf paths below node, given the prefix number above it.
    private int dfs(TreeNode node, int prefix) {
        if (node == null) return 0;                            // an empty side contributes nothing
        int cur = prefix * 10 + node.val;                      // append this digit
        if (node.left == null && node.right == null) return cur; // a leaf ends exactly one number
        return dfs(node.left, cur) + dfs(node.right, cur);
    }
}
class Solution:
    def sumNumbers(self, root: Optional[TreeNode]) -> int:
        total = 0
        stack = [(root, 0)]  # (node, number formed by the digits above it)
        while stack:
            node, prefix = stack.pop()
            if node is None:
                continue
            cur = prefix * 10 + node.val  # append this digit
            if node.left is None and node.right is None:
                total += cur  # a leaf ends exactly one number
            else:
                stack.append((node.right, cur))
                stack.append((node.left, cur))
        return total
