Sort by end, take-or-skip with an upper-bound binary search; the tested files are in coding-code/maximum-profit-in-job-scheduling/ (the Python test also cross-checks 300 random cases against the brute force).

func jobScheduling(startTime []int, endTime []int, profit []int) int {
	n := len(startTime)
	order := make([]int, n)
	for i := range order {
		order[i] = i
	}
	sort.Slice(order, func(a, b int) bool { return endTime[order[a]] < endTime[order[b]] })

	ends := make([]int, n) // end times in sorted order
	for k, j := range order {
		ends[k] = endTime[j]
	}

	// best[k] = max profit using only the first k jobs (by end time)
	best := make([]int, n+1)
	for k, j := range order {
		// how many earlier jobs end at or before this job starts
		fit := sort.Search(k, func(i int) bool { return ends[i] > startTime[j] })
		take := best[fit] + profit[j] // do job j after the best compatible prefix
		skip := best[k]               // ignore job j
		if take > skip {
			best[k+1] = take
		} else {
			best[k+1] = skip
		}
	}
	return best[n]
}

    int jobScheduling(vector<int>& startTime, vector<int>& endTime, vector<int>& profit) {
        int n = startTime.size();
        vector<int> order(n);
        iota(order.begin(), order.end(), 0);
        sort(order.begin(), order.end(),
             [&](int a, int b) { return endTime[a] < endTime[b]; });

        vector<int> ends(n);                       // end times in sorted order
        for (int k = 0; k < n; k++) ends[k] = endTime[order[k]];

        // best[k] = max profit using only the first k jobs (by end time)
        vector<int> best(n + 1, 0);
        for (int k = 0; k < n; k++) {
            int j = order[k];
            // how many earlier jobs end at or before this job starts
            int fit = upper_bound(ends.begin(), ends.begin() + k, startTime[j]) - ends.begin();

            int skip = best[k];                    // ignore job j
            best[k + 1] = max(take, skip);
        }
        return best[n];
    }

static class Solution {
    public int jobScheduling(int[] startTime, int[] endTime, int[] profit) {
        int n = startTime.length;
        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> Integer.compare(endTime[a], endTime[b]));

        int[] ends = new int[n];                   // end times in sorted order
        for (int k = 0; k < n; k++) ends[k] = endTime[order[k]];

        // best[k] = max profit using only the first k jobs (by end time)
        int[] best = new int[n + 1];
        for (int k = 0; k < n; k++) {
            int j = order[k];
            int fit = countEndsAtMost(ends, k, startTime[j]);
            int take = best[fit] + profit[j];      // do job j after the best compatible prefix
            int skip = best[k];                    // ignore job j
            best[k + 1] = Math.max(take, skip);
        }
        return best[n];
    }

    // number of entries in ends[0..k) that are <= t (ends is sorted): an upper bound
    private int countEndsAtMost(int[] ends, int k, int t) {
        int lo = 0, hi = k;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (ends[mid] <= t) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }
}
class Solution:
    def jobScheduling(self, startTime: List[int], endTime: List[int], profit: List[int]) -> int:
        jobs = sorted(zip(endTime, startTime, profit))   # sort by end time
        ends = [e for e, _, _ in jobs]

        # best[k] = max profit using only the first k jobs (by end time)
        best = [0] * (len(jobs) + 1)
        for k, (e, s, p) in enumerate(jobs):
            # how many earlier jobs end at or before this job starts
            fit = bisect_right(ends, s, 0, k)
            take = best[fit] + p            # do this job after the best compatible prefix
            skip = best[k]                  # ignore this job
            best[k + 1] = max(take, skip)
        return best[-1]
