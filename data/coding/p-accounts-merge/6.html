Union-find on account indices with path halving. The tests in coding-code/accounts-merge/ compare results after sorting the outer list, because any account order is accepted.

class Solution {
public:
    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        int n = accounts.size();
        parent.resize(n);
        iota(parent.begin(), parent.end(), 0);

        unordered_map<string, int> firstOwner;          // email -> first account that listed it
        for (int i = 0; i < n; i++)
            for (size_t j = 1; j < accounts[i].size(); j++) {
                auto it = firstOwner.find(accounts[i][j]);
                if (it == firstOwner.end()) firstOwner[accounts[i][j]] = i;
                else parent[find(i)] = find(it->second);  // shared email: same person
            }

        unordered_map<int, vector<string>> groups;      // root account -> its emails (each once)
        for (auto& kv : firstOwner) groups[find(kv.second)].push_back(kv.first);

        vector<vector<string>> out;
        for (auto& kv : groups) {
            sort(kv.second.begin(), kv.second.end());
            vector<string> row{accounts[kv.first][0]};
            row.insert(row.end(), kv.second.begin(), kv.second.end());
            out.push_back(move(row));
        }
        return out;
    }

private:
    vector<int> parent;
    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];               // path halving
            x = parent[x];
        }
        return x;
    }
};
func accountsMerge(accounts [][]string) [][]string {
	parent := make([]int, len(accounts))
	for i := range parent {
		parent[i] = i
	}
	find := func(x int) int {
		for parent[x] != x {
			parent[x] = parent[parent[x]] // path halving
			x = parent[x]
		}
		return x
	}

	firstOwner := map[string]int{} // email -> first account that listed it
	for i, acc := range accounts {
		for _, email := range acc[1:] {
			if owner, seen := firstOwner[email]; seen {
				parent[find(i)] = find(owner) // shared email: same person
			} else {
				firstOwner[email] = i
			}
		}
	}

	groups := map[int][]string{} // root account -> its emails (each once)
	for email, i := range firstOwner {
		root := find(i)
		groups[root] = append(groups[root], email)
	}
	out := [][]string{}
	for root, emails := range groups {
		sort.Strings(emails)
		out = append(out, append([]string{accounts[root][0]}, emails...))
	}
	return out
}

static class Solution {
    private int[] parent;

    public List<List<String>> accountsMerge(List<List<String>> accounts) {
        int n = accounts.size();
        parent = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;

        Map<String, Integer> firstOwner = new HashMap<>();   // email -> first account that listed it
        for (int i = 0; i < n; i++) {
            List<String> acc = accounts.get(i);
            for (int j = 1; j < acc.size(); j++) {
                Integer owner = firstOwner.putIfAbsent(acc.get(j), i);
                if (owner != null) parent[find(i)] = find(owner);   // shared email: same person
            }
        }

        Map<Integer, List<String>> groups = new HashMap<>();  // root account -> its emails
        for (Map.Entry<String, Integer> e : firstOwner.entrySet())
            groups.computeIfAbsent(find(e.getValue()), k -> new ArrayList<>()).add(e.getKey());

        List<List<String>> out = new ArrayList<>();
        for (Map.Entry<Integer, List<String>> g : groups.entrySet()) {
            List<String> emails = g.getValue();
            Collections.sort(emails);
            emails.add(0, accounts.get(g.getKey()).get(0));   // the name goes first
            out.add(emails);
        }
        return out;
    }

    private int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];                    // path halving
            x = parent[x];
        }
        return x;
    }
}
class Solution:
    def accountsMerge(self, accounts: List[List[str]]) -> List[List[str]]:
        parent = list(range(len(accounts)))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]    # path halving
                x = parent[x]
            return x

        first_owner = {}                         # email -> first account index that listed it
        for i, acc in enumerate(accounts):
            for email in acc[1:]:
                if email in first_owner:
                    parent[find(i)] = find(first_owner[email])   # shared email: same person
                else:
                    first_owner[email] = i

        groups = defaultdict(list)               # root account -> its emails (each once)
        for email, i in first_owner.items():
            groups[find(i)].append(email)
        return [[accounts[root][0]] + sorted(emails) for root, emails in groups.items()]
