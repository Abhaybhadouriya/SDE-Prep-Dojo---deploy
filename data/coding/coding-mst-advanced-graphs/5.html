Bridges: Tarjan's low-link DFS

Critical Connections in a Network: n servers, undirected connections, return every connection whose removal disconnects some server from another. Those connections are the bridges. The brute force removes each edge and runs a BFS, O(E x (V + E)), far too slow for 105 servers. Tarjan finds all of them in one DFS.

During the DFS give each vertex two numbers:

  • tin[u]: the time u was first visited (0, 1, 2, ...).
  • low[u]: the smallest tin reachable from u's DFS subtree using tree edges downwards and then one back edge. It is computed on the way back up: low[u] = min(tin[u], tin[w] for each back edge u-w, low[c] for each child c).

In an undirected DFS every non-tree edge goes to an ancestor (a back edge); there are no cross edges. So the tree edge u-v (v a child of u) is a bridge exactly when nothing in v's subtree has a back edge to u or above it, which is low[v] > tin[u].

Template 3: bridges

// 3. Tarjan bridges (Critical Connections): edge u-v is a bridge when low[v] > tin[u].
func criticalConnections(n int, conns [][]int) [][]int {
	type arc struct{ v, id int }
	adj := make([][]arc, n)
	for i, c := range conns {
		adj[c[0]] = append(adj[c[0]], arc{c[1], i})
		adj[c[1]] = append(adj[c[1]], arc{c[0], i})
	}
	tin, low := make([]int, n), make([]int, n)
	for i := range tin {
		tin[i] = -1
	}
	timer := 0
	bridges := [][]int{}
	var dfs func(u, parentEdge int)
	dfs = func(u, parentEdge int) {
		tin[u], low[u] = timer, timer
		timer++
		for _, a := range adj[u] {
			if a.id == parentEdge {
				continue // skip the edge we came in on (by id)
			}
			if tin[a.v] != -1 {
				if tin[a.v] < low[u] {
					low[u] = tin[a.v] // back edge to an ancestor
				}
			} else {
				dfs(a.v, a.id)
				if low[a.v] < low[u] {
					low[u] = low[a.v]
				}
				if low[a.v] > tin[u] {
					bridges = append(bridges, []int{u, a.v})
				}
			}
		}
	}
	for u := 0; u < n; u++ {
		if tin[u] == -1 {
			dfs(u, -1)
		}
	}
	return bridges
}
// 3. Tarjan bridges (Critical Connections): edge u-v is a bridge when low[v] > tin[u].
vector<vector<int>> criticalConnections(int n, const vector<vector<int>>& conns) {
    vector<vector<pair<int,int>>> adj(n);         // {neighbour, edge id}
    for (int i = 0; i < (int)conns.size(); i++) {
        adj[conns[i][0]].push_back({conns[i][1], i});
        adj[conns[i][1]].push_back({conns[i][0], i});
    }
    vector<int> tin(n, -1), low(n, 0);
    vector<vector<int>> bridges;
    int timer = 0;
    function<void(int, int)> dfs = [&](int u, int parentEdge) {
        tin[u] = low[u] = timer++;
        for (const auto& a : adj[u]) {
            int v = a.first, id = a.second;
            if (id == parentEdge) continue;       // skip the edge we came in on (by id, not by vertex)
            if (tin[v] != -1) {
                low[u] = min(low[u], tin[v]);     // back edge to an ancestor
            } else {
                dfs(v, id);
                low[u] = min(low[u], low[v]);
                if (low[v] > tin[u]) bridges.push_back({u, v});
            }
        }
    };
    for (int u = 0; u < n; u++)
        if (tin[u] == -1) dfs(u, -1);
    return bridges;
}

static List<int[]>[] adj;
static int[] tin, low;
static int timer;
static List<List<Integer>> bridges;

@SuppressWarnings("unchecked")
static List<List<Integer>> criticalConnections(int n, int[][] conns) {
    adj = new List[n];
    for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
    for (int i = 0; i < conns.length; i++) {   // {neighbour, edge id}
        adj[conns[i][0]].add(new int[]{conns[i][1], i});
        adj[conns[i][1]].add(new int[]{conns[i][0], i});
    }
    tin = new int[n]; low = new int[n];
    Arrays.fill(tin, -1);
    timer = 0;
    bridges = new ArrayList<>();
    for (int u = 0; u < n; u++) if (tin[u] == -1) dfs(u, -1);
    return bridges;
}

static void dfs(int u, int parentEdge) {
    tin[u] = low[u] = timer++;
    for (int[] a : adj[u]) {
        int v = a[0], id = a[1];
        if (id == parentEdge) continue;        // skip the edge we came in on (by id)
        if (tin[v] != -1) {
            low[u] = Math.min(low[u], tin[v]); // back edge to an ancestor
        } else {
            dfs(v, id);
            low[u] = Math.min(low[u], low[v]);
            if (low[v] > tin[u]) bridges.add(Arrays.asList(u, v));
        }
    }
}
# 3. Tarjan bridges (Critical Connections), iterative: edge u-v is a bridge when low[v] > tin[u].
def critical_connections(n, connections):
    adj = [[] for _ in range(n)]              # (neighbour, edge id)
    for i, (u, v) in enumerate(connections):
        adj[u].append((v, i))
        adj[v].append((u, i))
    tin, low = [-1] * n, [0] * n
    timer, bridges = 0, []
    for root in range(n):
        if tin[root] != -1:
            continue
        tin[root] = low[root] = timer
        timer += 1
        stack = [(root, -1, 0)]               # (vertex, edge id we came in on, next adj index)
        while stack:
            u, pe, i = stack[-1]
            if i < len(adj[u]):
                stack[-1] = (u, pe, i + 1)
                v, eid = adj[u][i]
                if eid == pe:
                    continue                  # skip the edge we came in on (by id)
                if tin[v] == -1:              # tree edge: "recurse"
                    tin[v] = low[v] = timer
                    timer += 1
                    stack.append((v, eid, 0))
                else:
                    low[u] = min(low[u], tin[v])     # back edge to an ancestor
            else:                             # u is finished: "return" to its parent
                stack.pop()
                if stack:
                    p = stack[-1][0]
                    low[p] = min(low[p], low[u])
                    if low[u] > tin[p]:
                        bridges.append([p, u])
    return bridges

Two details in the code are what separate a correct answer from a nearly-correct one:

  • Skip the parent edge by its id, not by the parent vertex. With two parallel cables between the same servers, neither is a bridge. Skipping "the vertex I came from" would ignore the second cable and wrongly report a bridge. The test suite above includes that case.
  • Recursion depth. A path of 105 vertices recurses 105 deep. Python's default limit is 1,000, so the Python template is iterative: an explicit stack of (vertex, incoming edge id, next neighbour index), the same frame a compiler would keep. C++ handles this depth with the usual 8 MB stack. Java's default thread stack may not, so either run the DFS in a new Thread(null, task, "dfs", 1 << 26) or convert it the same way.

Articulation points

A vertex whose removal disconnects the graph is an articulation point or cut vertex. Same DFS, two rule changes. A non-root u is a cut vertex if some child v has low[v] ≥ tin[u] (v's subtree can reach u itself, but not above it, so it still hangs from u alone). The root is a cut vertex only if it has two or more DFS children. In the figure, 3-4 is not a bridge (the triangle 3-4-5 survives losing it) but vertex 3 is still a cut vertex: > versus ≥ is the whole difference.

# 6. Articulation points (cut vertices): low[v] >= tin[u], and the root rule.
def articulation_points(n, adj):         # adj[u] = [v], undirected, recursive for clarity
    tin, low, cut = [-1] * n, [0] * n, set()
    timer = 0

    def dfs(u, parent):
        nonlocal timer
        tin[u] = low[u] = timer
        timer += 1
        children = 0
        for v in adj[u]:
            if v == parent:
                continue
            if tin[v] != -1:
                low[u] = min(low[u], tin[v])
            else:
                dfs(v, u)
                children += 1
                low[u] = min(low[u], low[v])
                if parent != -1 and low[v] >= tin[u]:
                    cut.add(u)           # v's subtree cannot climb above u
        if parent == -1 and children > 1:
            cut.add(u)                   # root: cut vertex only with 2+ DFS children

    for u in range(n):
        if tin[u] == -1:
            dfs(u, -1)
    return sorted(cut)

Minimum Number of Days to Disconnect Island is the grid version: the answer is always 0, 1 or 2 (you can always cut a corner cell off with two removals). It is 0 if the land is already split, 1 if some land cell is an articulation point (or there is only one cell), else 2. Many accepted solutions simply try removing each cell and count islands, which fits a 30 x 30 grid; the articulation-point version is the one to mention for scale.

Same name, different algorithm. Tarjan also has a strongly connected components algorithm for directed graphs that uses the same tin/low idea with a stack. If an interviewer says "Tarjan", ask which one. For SCCs in an interview, Kosaraju (two DFS passes, the second on the reversed graph) is often easier to get right.
