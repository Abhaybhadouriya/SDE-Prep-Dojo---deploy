Kruskal: sort the edges, skip the ones that close a cycle

Kruskal applies the cut property greedily. Sort all edges by weight. Walk them from lightest to heaviest. Keep an edge if its two ends are in different components; otherwise it would close a cycle, so skip it. A union-find (disjoint set union, DSU) answers "same component?" and merges components in almost constant time. Stop once you have V - 1 edges.

  1. Sort the edges by weight. Every vertex starts as its own component in a union-find. The total is 0.
  2. A-B (1): find(A) and find(B) differ, so the edge joins two components. Take it and union them.
  3. B-C (2): C is alone, A and B are together; different roots. Take it. Component {A B C}.
  4. A-C (3): find(A) == find(C). A and C are already connected, so this edge would close the cycle A-B-C. Skip it.
  5. C-D (4): different roots. Take it. Total 7.
  6. B-D (5): B and D are both in {A B C D}. Skip.
  7. D-E (6): take it. Now 4 edges = V - 1 are in the tree, so stop. C-E (7) is never read. Total 13, the same tree as the figure above.

Why is each kept edge safe? When Kruskal takes edge (u, v), put u's component on one side of a cut and everything else on the other. Every lighter edge was already processed and either joined two components or was inside one, so none crosses this cut. Edge (u, v) is therefore the lightest crossing edge, and the cut property says it belongs to an MST.

// 1. Kruskal: sort edges, keep each edge that joins two different components (union-find).
type DSU struct{ parent, size []int }

func newDSU(n int) *DSU {
	d := &DSU{make([]int, n), make([]int, n)}
	for i := range d.parent {
		d.parent[i], d.size[i] = i, 1
	}
	return d
}

func (d *DSU) find(x int) int {
	for d.parent[x] != x {
		d.parent[x] = d.parent[d.parent[x]] // path halving
		x = d.parent[x]
	}
	return x
}

func (d *DSU) unite(a, b int) bool {
	a, b = d.find(a), d.find(b)
	if a == b {
		return false // already connected: would close a cycle
	}
	if d.size[a] < d.size[b] {
		a, b = b, a // union by size
	}
	d.parent[b] = a
	d.size[a] += d.size[b]
	return true
}

// edges {u, v, w} undirected. Returns the MST weight, or -1 if the graph is not connected.
func kruskal(n int, edges [][3]int) int {
	es := append([][3]int(nil), edges...)
	sort.Slice(es, func(i, j int) bool { return es[i][2] < es[j][2] })
	dsu := newDSU(n)
	total, used := 0, 0
	for _, e := range es {
		if used == n-1 {
			break // a tree on n vertices has n - 1 edges
		}
		if dsu.unite(e[0], e[1]) {
			total += e[2]
			used++
		}
	}
	if used != n-1 {
		return -1
	}
	return total
}
// 1. Kruskal: sort edges, keep each edge that joins two different components (union-find).
struct DSU {
    vector<int> parent, size;
    DSU(int n) : parent(n), size(n, 1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }   // path halving
        return x;
    }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;                 // already connected: this edge would close a cycle
        if (size[a] < size[b]) swap(a, b);        // union by size
        parent[b] = a; size[a] += size[b];
        return true;
    }
};

    sort(edges.begin(), edges.end(), [](const vector<int>& a, const vector<int>& b) { return a[2] < b[2]; });

    if (size[a] < size[b]) { int t = a; a = b; b = t; }   // union by size

# 1. Kruskal: sort edges, keep each edge that joins two different components (union-find).
def kruskal(n, edges):
    """edges: (u, v, w) undirected. Returns the MST weight, or -1 if not connected."""
    parent, size = list(range(n)), [1] * n

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]     # path halving
            x = parent[x]
        return x

    total = used = 0
    for u, v, w in sorted(edges, key=lambda e: e[2]):
        if used == n - 1:
            break                             # a tree on n vertices has n - 1 edges
        a, b = find(u), find(v)
        if a == b:
            continue                          # already connected: would close a cycle
        if size[a] < size[b]:
            a, b = b, a                       # union by size
        parent[b] = a
        size[a] += size[b]
        total += w
        used += 1
    return total if used == n - 1 else -1

Cost. Sorting is O(E log E), which is O(E log V) because E < V2. The union-find work is O(E α(V)), where α is the inverse Ackermann function and is at most 4 for any input you will ever see. Space O(V) beyond the edge list.

Kruskal's skip test is a cycle detector. Redundant Connection (a tree plus one extra edge, find the extra one) is just Kruskal's loop without the sort: the first edge whose ends already share a root is the answer.
