Prim: grow one tree from a start vertex

Prim applies the cut property to the cut "tree so far" versus "everything else". Start from any vertex. Repeatedly add the cheapest edge that leaves the tree, and the vertex at its other end. It looks like Dijkstra, with one difference that is a classic interview check: Prim's key is the weight of the single edge into the tree, Dijkstra's key is the total distance from the source.

Min Cost to Connect All Points is the standard question. You get up to 1,000 points on a plane; connecting two points costs their Manhattan distance |x1 - x2| + |y1 - y2|; return the minimum cost to connect them all. Every pair is a possible edge, so the graph is complete: about 500,000 edges for 1,000 points. On a complete graph, the array version of Prim is the cleanest: keep best[v] = cheapest edge from the tree to v, pick the smallest unvisited one by a linear scan, then update all others. O(V2) time, O(V) memory, and no edge list at all.

Template 2: Prim on a dense graph

// 2. Prim, O(V^2) array version for a dense (complete) graph: Min Cost to Connect All Points.
func minCostConnectPoints(pts [][]int) int {
	abs := func(x int) int {
		if x < 0 {
			return -x
		}
		return x
	}
	n, total := len(pts), 0
	best := make([]int, n) // cheapest edge from the tree to each vertex
	for i := range best {
		best[i] = math.MaxInt64
	}
	inTree := make([]bool, n)
	best[0] = 0
	for it := 0; it < n; it++ {
		u := -1
		for v := 0; v < n; v++ {
			if !inTree[v] && (u == -1 || best[v] < best[u]) {
				u = v
			}
		}
		inTree[u] = true
		total += best[u]
		for v := 0; v < n; v++ {
			if inTree[v] {
				continue
			}
			if d := abs(pts[u][0]-pts[v][0]) + abs(pts[u][1]-pts[v][1]); d < best[v] {
				best[v] = d
			}
		}
	}
	return total
}

def min_cost_connect_points(pts):
    n = len(pts)
    best = [float('inf')] * n                 # cheapest edge from the tree to each vertex
    in_tree = [False] * n
    best[0] = 0
    total = 0
    for _ in range(n):
        u = min((v for v in range(n) if not in_tree[v]), key=best.__getitem__)
        in_tree[u] = True
        total += best[u]
        ux, uy = pts[u]
        for v in range(n):
            if not in_tree[v]:
                d = abs(ux - pts[v][0]) + abs(uy - pts[v][1])
                if d < best[v]:
                    best[v] = d
    return total

On a sparse graph given as an adjacency list, use a heap instead. It is lazy in the same way as Dijkstra: push every edge out of a newly added vertex, and skip a popped entry whose vertex is already in the tree.

import heapq

# 5. Prim with a heap, for a sparse graph given as an adjacency list.
def prim_heap(n, adj):                   # adj[u] = [(v, w)], undirected
    seen = [False] * n
    heap = [(0, 0)]                      # (edge weight into the tree, vertex)
    total = taken = 0
    while heap and taken < n:
        w, u = heapq.heappop(heap)
        if seen[u]:
            continue                     # stale: u joined the tree through a cheaper edge
        seen[u] = True
        total += w
        taken += 1
        for v, wv in adj[u]:
            if not seen[v]:
                heapq.heappush(heap, (wv, v))
    return total if taken == n else -1
