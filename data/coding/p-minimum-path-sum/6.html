func minPathSum(grid [][]int) int {
	rows, cols := len(grid), len(grid[0])
	// best[c] = cheapest cost from (0,0) to (r,c) for the row r being filled.
	// Before cell (r,c) is updated, best[c] still holds the answer for (r-1,c): the cell above.
	best := make([]int, cols)
	for r := 0; r < rows; r++ {
		for c := 0; c < cols; c++ {
			switch {
			case r == 0 && c == 0:
				best[c] = grid[0][0] // start cell
			case r == 0:
				best[c] = best[c-1] + grid[r][c] // top row: only from the left
			case c == 0:
				best[c] = best[c] + grid[r][c] // left column: only from above
			default:
				best[c] = min(best[c], best[c-1]) + grid[r][c] // above vs left (min is built in since Go 1.21)
			}
		}
	}
	return best[cols-1]
}
