const MOD = 1_000_000_007

func modPow(b, e, m int64) int64 { // fast power: O(log e) multiplications
	b %= m
	if b < 0 {
		b += m
	}
	r := int64(1) % m
	for e > 0 {
		if e&1 == 1 { // this bit of e is set: multiply it in
			r = r * b % m
		}
		b = b * b % m
		e >>= 1
	}
	return r
}

func myPow(x float64, n int) float64 { // LeetCode 50 (Go int is 64-bit, so -n is safe)
	e := int64(n)
	if e < 0 {
		x, e = 1/x, -e
	}
	r := 1.0
	for e > 0 {
		if e&1 == 1 {
			r *= x
		}
		x *= x
		e >>= 1
	}
	return r
}

func gcd(a, b int) int {
	for b != 0 {
		a, b = b, a%b
	}
	if a < 0 {
		return -a
	}
	return a
}

func lcm(a, b int) int { return a / gcd(a, b) * b } // divide first

func countPrimes(n int) int { // primes strictly below n
	if n < 3 {
		return 0
	}
	composite := make([]bool, n)
	count := 0
	for i := 2; i < n; i++ {
		if composite[i] {
			continue
		}
		count++
		for j := i * i; j < n; j += i { // start at i*i
			composite[j] = true
		}
	}
	return count
}

func reverseInt(x int) int { // LeetCode 7: 0 when the result leaves int32
	r := 0
	for x != 0 {
		d := x % 10 // Go keeps the sign like C
		x /= 10
		if r > math.MaxInt32/10 || (r == math.MaxInt32/10 && d > 7) {
			return 0
		}
		if r < math.MinInt32/10 || (r == math.MinInt32/10 && d < -8) {
			return 0
		}
		r = r*10 + d
	}
	return r
}

func addStrings(a, b string) string {
	i, j, carry := len(a)-1, len(b)-1, 0
	out := []byte{}
	for i >= 0 || j >= 0 || carry > 0 {
		s := carry
		if i >= 0 {
			s += int(a[i] - '0')
			i--
		}
		if j >= 0 {
			s += int(b[j] - '0')
			j--
		}
		out = append(out, byte('0'+s%10))
		carry = s / 10
	}
	for l, r := 0, len(out)-1; l < r; l, r = l+1, r-1 {
		out[l], out[r] = out[r], out[l]
	}
	return string(out)
}

func multiplyStrings(a, b string) string { // digit i*j lands in i+j and i+j+1
	if a == "0" || b == "0" {
		return "0"
	}
	res := make([]int, len(a)+len(b))
	for i := len(a) - 1; i >= 0; i-- {
		for j := len(b) - 1; j >= 0; j-- {
			p := int(a[i]-'0')*int(b[j]-'0') + res[i+j+1]
			res[i+j+1] = p % 10
			res[i+j] += p / 10
		}
	}
	var sb strings.Builder
	for _, d := range res {
		if sb.Len() == 0 && d == 0 {
			continue
		}
		sb.WriteByte(byte('0' + d))
	}
	return sb.String()
}

func pascal(rows int) [][]int { // LeetCode 118
	tri := [][]int{}
	for r := 0; r < rows; r++ {
		row := make([]int, r+1)
		row[0], row[r] = 1, 1
		for k := 1; k < r; k++ {
			row[k] = tri[r-1][k-1] + tri[r-1][k]
		}
		tri = append(tri, row)
	}
	return tri
}

type NCr struct{ f, inv []int64 } // nCr mod p (p prime): factorials + Fermat

func newNCr(n int) *NCr {
	c := &NCr{make([]int64, n+1), make([]int64, n+1)}
	c.f[0] = 1
	for i := 1; i <= n; i++ {
		c.f[i] = c.f[i-1] * int64(i) % MOD
	}
	c.inv[n] = modPow(c.f[n], MOD-2, MOD) // a^(p-2) = a^-1 mod p
	for i := n; i > 0; i-- {
		c.inv[i-1] = c.inv[i] * int64(i) % MOD
	}
	return c
}

func (c *NCr) C(n, r int) int64 {
	if r < 0 || r > n {
		return 0
	}
	return c.f[n] * c.inv[r] % MOD * c.inv[n-r] % MOD
}

type WeightedPicker struct { // LeetCode 528: prefix sums + binary search
	pre []int
	rng *rand.Rand
}

func newWeightedPicker(w []int) *WeightedPicker {
	p := &WeightedPicker{rng: rand.New(rand.NewSource(12345))}
	total := 0
	for _, x := range w {
		total += x
		p.pre = append(p.pre, total)
	}
	return p
}

func (p *WeightedPicker) pickWith(t int) int { // t in [1, total]: first prefix >= t
	return sort.SearchInts(p.pre, t)
}

func (p *WeightedPicker) pick() int { return p.pickWith(p.rng.Intn(p.pre[len(p.pre)-1]) + 1) }

func reservoirPick(stream []int, rng *rand.Rand) int { // one uniform item, unknown length
	chosen, seen := -1, 0
	for _, x := range stream {
		seen++
		if rng.Intn(seen) == 0 { // keep the new item with probability 1/seen
			chosen = x
		}
	}
	return chosen
}

func toBase(n int64, b int) string { return strconv.FormatInt(n, b) } // "0" for 0

var romanV = []int{1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1}
var romanS = []string{"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"}

func intToRoman(n int) string { // greedy: largest symbol that fits
	var sb strings.Builder
	for i, v := range romanV {
		for n >= v {
			sb.WriteString(romanS[i])
			n -= v
		}
	}
	return sb.String()
}

func romanToInt(s string) int { // subtract when smaller comes before larger
	val := map[byte]int{'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000}
	total := 0
	for i := 0; i < len(s); i++ {
		x := val[s[i]]
		if i+1 < len(s) && x < val[s[i+1]] {
			total -= x
		} else {
			total += x
		}
	}
	return total
}

func cross(o, a, b [2]int64) int64 { // >0 left turn, <0 right, 0 collinear
	return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
}

func maxPoints(pts [][]int) int { // LeetCode 149: O(n^2), exact slopes
	best := 0
	for i := range pts {
		cnt := map[[2]int]int{}
		top := 0
		for j := i + 1; j < len(pts); j++ {
			dy, dx := pts[j][1]-pts[i][1], pts[j][0]-pts[i][0]
			g := gcd(dy, dx)
			dy, dx = dy/g, dx/g
			if dx < 0 || (dx == 0 && dy < 0) { // one sign convention
				dy, dx = -dy, -dx
			}
			k := [2]int{dy, dx}
			cnt[k]++
			if cnt[k] > top {
				top = cnt[k]
			}
		}
		if top+1 > best {
			best = top + 1
		}
	}
	return best
}

func must(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

// Math and geometry templates (coding-math lesson). C++14: g++ -std=c++14 -O2 mathx.cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 1000000007LL;

ll modPow(ll b, ll e, ll m = MOD) {                  // fast power: O(log e) multiplications
    b %= m; if (b < 0) b += m;
    ll r = 1 % m;
    while (e > 0) {
        if (e & 1) r = r * b % m;                    // this bit of e is set: multiply it in
        b = b * b % m;                               // b, b^2, b^4 ... (m < 3e9 so no overflow)
        e >>= 1;
    }
    return r;
}

double myPow(double x, int n) {                      // LeetCode 50
    ll e = n;                                        // 64-bit: -(-2^31) does not fit in int
    if (e < 0) { x = 1 / x; e = -e; }
    double r = 1;
    while (e) { if (e & 1) r *= x; x *= x; e >>= 1; }
    return r;
}

ll gcdl(ll a, ll b) { while (b) { ll t = a % b; a = b; b = t; } return a < 0 ? -a : a; }
ll lcml(ll a, ll b) { return a / gcdl(a, b) * b; } // divide first

int countPrimes(int n) {                             // primes strictly below n
    if (n < 3) return 0;
    vector<char> isP(n, 1);
    isP[0] = isP[1] = 0;
    for (ll i = 2; i * i < n; ++i)
        if (isP[i]) for (ll j = i * i; j < n; j += i) isP[j] = 0;   // start at i*i
    return (int)count(isP.begin(), isP.end(), 1);
}

int reverseInt(int x) {                              // LeetCode 7: 0 on 32-bit overflow
    int r = 0;
    while (x != 0) {
        int d = x % 10;                              // C++ keeps the sign: -4310 % 10 == 0, -431 % 10 == -1
        x /= 10;
        if (r > INT_MAX / 10 || (r == INT_MAX / 10 && d > 7)) return 0;
        if (r < INT_MIN / 10 || (r == INT_MIN / 10 && d < -8)) return 0;
        r = r * 10 + d;                              // checked before it can overflow
    }
    return r;
}

string addStrings(const string& a, const string& b) {
    int i = a.size() - 1, j = b.size() - 1, carry = 0;
    string out;
    while (i >= 0 || j >= 0 || carry) {
        int s = carry;
        if (i >= 0) s += a[i--] - '0';
        if (j >= 0) s += b[j--] - '0';
        out.push_back('0' + s % 10);
        carry = s / 10;
    }
    reverse(out.begin(), out.end());
    return out;
}

string multiplyStrings(const string& a, const string& b) {   // digit i*j lands in i+j and i+j+1
    if (a == "0" || b == "0") return "0";
    vector<int> res(a.size() + b.size(), 0);
    for (int i = a.size() - 1; i >= 0; --i)
        for (int j = b.size() - 1; j >= 0; --j) {
            int p = (a[i] - '0') * (b[j] - '0') + res[i + j + 1];
            res[i + j + 1] = p % 10;
            res[i + j] += p / 10;
        }
    string s;
    for (int d : res) if (!(s.empty() && d == 0)) s.push_back('0' + d);
    return s;
}

vector<vector<int>> pascal(int rows) {               // LeetCode 118
    vector<vector<int>> tri;
    for (int r = 0; r < rows; ++r) {
        vector<int> row(r + 1, 1);
        for (int k = 1; k < r; ++k) row[k] = tri[r - 1][k - 1] + tri[r - 1][k];
        tri.push_back(row);
    }
    return tri;
}

struct NCr {                                         // nCr mod p (p prime): factorials + Fermat
    vector<ll> f, inv;
    NCr(int n) : f(n + 1, 1), inv(n + 1, 1) {
        for (int i = 1; i <= n; ++i) f[i] = f[i - 1] * i % MOD;
        inv[n] = modPow(f[n], MOD - 2);              // a^(p-2) = a^-1 mod p
        for (int i = n; i > 0; --i) inv[i - 1] = inv[i] * i % MOD;
    }
    ll c(int n, int r) const {
        if (r < 0 || r > n) return 0;
        return f[n] * inv[r] % MOD * inv[n - r] % MOD;
    }
};

struct WeightedPicker {                              // LeetCode 528: prefix sums + binary search
    vector<int> pre;
    mt19937 rng{12345};
    explicit WeightedPicker(const vector<int>& w) {
        int total = 0;
        for (int x : w) { total += x; pre.push_back(total); }
    }
    int pickWith(int t) const {                      // t in [1, total]
        return lower_bound(pre.begin(), pre.end(), t) - pre.begin();   // first prefix >= t
    }
    int pick() { return pickWith(uniform_int_distribution<int>(1, pre.back())(rng)); }
};

template <class It>
int reservoirPick(It first, It last, mt19937& rng) { // one uniform item, stream of unknown length
    int chosen = -1; long long seen = 0;
    for (; first != last; ++first) {
        ++seen;
        if (uniform_int_distribution<long long>(0, seen - 1)(rng) == 0) chosen = *first;   // keep with 1/seen
    }
    return chosen;
}

string toBase(ll n, int b) {                         // n >= 0, 2 <= b <= 36
    if (n == 0) return "0";
    const char* digits = "0123456789abcdefghijklmnopqrstuvwxyz";
    string out;
    while (n) { out.push_back(digits[n % b]); n /= b; }
    reverse(out.begin(), out.end());
    return out;
}

string intToRoman(int n) {                           // greedy: largest symbol that fits
    static const int v[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    static const char* s[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    string out;
    for (int i = 0; i < 13; ++i) while (n >= v[i]) { out += s[i]; n -= v[i]; }
    return out;
}

int romanToInt(const string& s) {                    // subtract when smaller comes before larger
    auto val = [](char c) {
        switch (c) { case 'I': return 1; case 'V': return 5; case 'X': return 10; case 'L': return 50;
                     case 'C': return 100; case 'D': return 500; default: return 1000; }
    };
    int total = 0;
    for (size_t i = 0; i < s.size(); ++i) {
        int x = val(s[i]);
        if (i + 1 < s.size() && x < val(s[i + 1])) total -= x; else total += x;
    }
    return total;
}

ll cross(pair<ll, ll> o, pair<ll, ll> a, pair<ll, ll> b) {   // >0 left turn, <0 right, 0 collinear
    return (a.first - o.first) * (b.second - o.second) - (a.second - o.second) * (b.first - o.first);
}

pair<int, int> slopeKey(const vector<int>& p, const vector<int>& q) {   // reduced (dy, dx), no floats
    int dy = q[1] - p[1], dx = q[0] - p[0];
    int g = (int)gcdl(dy, dx);
    dy /= g; dx /= g;
    if (dx < 0 || (dx == 0 && dy < 0)) { dy = -dy; dx = -dx; }  // one sign convention
    return {dy, dx};
}

int maxPoints(const vector<vector<int>>& pts) {      // LeetCode 149: O(n^2)
    int best = 0;
    for (size_t i = 0; i < pts.size(); ++i) {
        map<pair<int, int>, int> cnt;
        int top = 0;
        for (size_t j = i + 1; j < pts.size(); ++j) top = max(top, ++cnt[slopeKey(pts[i], pts[j])]);
        best = max(best, top + 1);
    }
    return best;
}

// Math and geometry templates (coding-math lesson). Java 21: java MathX.java
import java.util.*;

public class MathX {
    static final long MOD = 1_000_000_007L;

    static long modPow(long b, long e, long m) {     // fast power: O(log e) multiplications
        b %= m; if (b < 0) b += m;
        long r = 1 % m;
        while (e > 0) {
            if ((e & 1) == 1) r = r * b % m;         // this bit of e is set: multiply it in
            b = b * b % m;
            e >>= 1;
        }
        return r;
    }

    static double myPow(double x, int n) {           // LeetCode 50
        long e = n;                                  // long: -(-2^31) does not fit in int
        if (e < 0) { x = 1 / x; e = -e; }
        double r = 1;
        while (e > 0) { if ((e & 1) == 1) r *= x; x *= x; e >>= 1; }
        return r;
    }

    static long gcd(long a, long b) { while (b != 0) { long t = a % b; a = b; b = t; } return Math.abs(a); }
    static long lcm(long a, long b) { return a / gcd(a, b) * b; }   // divide first

    static int countPrimes(int n) {                  // primes strictly below n
        if (n < 3) return 0;
        boolean[] composite = new boolean[n];
        int count = 0;
        for (long i = 2; i < n; i++) {
            if (composite[(int) i]) continue;
            count++;
            for (long j = i * i; j < n; j += i) composite[(int) j] = true;   // long: i*i can pass 2^31
        }
        return count;
    }

    static int reverseInt(int x) {                   // LeetCode 7: 0 on 32-bit overflow
        int r = 0;
        while (x != 0) {
            int d = x % 10;                          // keeps the sign in Java
            x /= 10;
            if (r > Integer.MAX_VALUE / 10 || (r == Integer.MAX_VALUE / 10 && d > 7)) return 0;
            if (r < Integer.MIN_VALUE / 10 || (r == Integer.MIN_VALUE / 10 && d < -8)) return 0;
            r = r * 10 + d;
        }
        return r;
    }

    static String addStrings(String a, String b) {
        int i = a.length() - 1, j = b.length() - 1, carry = 0;
        StringBuilder out = new StringBuilder();
        while (i >= 0 || j >= 0 || carry > 0) {
            int s = carry;
            if (i >= 0) s += a.charAt(i--) - '0';
            if (j >= 0) s += b.charAt(j--) - '0';
            out.append((char) ('0' + s % 10));
            carry = s / 10;
        }
        return out.reverse().toString();
    }

    static String multiplyStrings(String a, String b) {   // digit i*j lands in i+j and i+j+1
        if (a.equals("0") || b.equals("0")) return "0";
        int[] res = new int[a.length() + b.length()];
        for (int i = a.length() - 1; i >= 0; i--)
            for (int j = b.length() - 1; j >= 0; j--) {
                int p = (a.charAt(i) - '0') * (b.charAt(j) - '0') + res[i + j + 1];
                res[i + j + 1] = p % 10;
                res[i + j] += p / 10;
            }
        StringBuilder s = new StringBuilder();
        for (int d : res) if (!(s.length() == 0 && d == 0)) s.append(d);
        return s.toString();
    }

    static List<List<Integer>> pascal(int rows) {    // LeetCode 118
        List<List<Integer>> tri = new ArrayList<>();
        for (int r = 0; r < rows; r++) {
            List<Integer> row = new ArrayList<>();
            for (int k = 0; k <= r; k++)
                row.add(k == 0 || k == r ? 1 : tri.get(r - 1).get(k - 1) + tri.get(r - 1).get(k));
            tri.add(row);
        }
        return tri;
    }

    static final class NCr {                         // nCr mod p (p prime): factorials + Fermat
        final long[] f, inv;
        NCr(int n) {
            f = new long[n + 1]; inv = new long[n + 1];
            f[0] = 1;
            for (int i = 1; i <= n; i++) f[i] = f[i - 1] * i % MOD;
            inv[n] = modPow(f[n], MOD - 2, MOD);     // a^(p-2) = a^-1 mod p
            for (int i = n; i > 0; i--) inv[i - 1] = inv[i] * i % MOD;
        }
        long c(int n, int r) {
            if (r < 0 || r > n) return 0;
            return f[n] * inv[r] % MOD * inv[n - r] % MOD;
        }
    }

    static final class WeightedPicker {              // LeetCode 528: prefix sums + binary search
        final int[] pre;
        final Random rng = new Random(12345);
        WeightedPicker(int[] w) {
            pre = new int[w.length];
            int total = 0;
            for (int i = 0; i < w.length; i++) { total += w[i]; pre[i] = total; }
        }
        int pickWith(int t) {                        // t in [1, total]: first prefix >= t
            int lo = 0, hi = pre.length - 1;
            while (lo < hi) {
                int mid = (lo + hi) >>> 1;
                if (pre[mid] >= t) hi = mid; else lo = mid + 1;
            }
            return lo;
        }
        int pick() { return pickWith(rng.nextInt(pre[pre.length - 1]) + 1); }
    }

    static int reservoirPick(Iterator<Integer> stream, Random rng) {   // one uniform item
        int chosen = -1; long seen = 0;
        while (stream.hasNext()) {
            int x = stream.next();
            seen++;
            if (rng.nextLong(seen) == 0) chosen = x;   // keep the new item with probability 1/seen
        }
        return chosen;
    }

    static String toBase(long n, int b) { return n == 0 ? "0" : Long.toString(n, b); }   // n >= 0

    static final int[] RV = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    static final String[] RS = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};

    static String intToRoman(int n) {                // greedy: largest symbol that fits
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < RV.length; i++) while (n >= RV[i]) { out.append(RS[i]); n -= RV[i]; }
        return out.toString();
    }

    static int val(char c) {
        return switch (c) { case 'I' -> 1; case 'V' -> 5; case 'X' -> 10; case 'L' -> 50;
                            case 'C' -> 100; case 'D' -> 500; default -> 1000; };
    }

    static int romanToInt(String s) {                // subtract when smaller comes before larger
        int total = 0;
        for (int i = 0; i < s.length(); i++) {
            int x = val(s.charAt(i));
            if (i + 1 < s.length() && x < val(s.charAt(i + 1))) total -= x; else total += x;
        }
        return total;
    }

    static long cross(long[] o, long[] a, long[] b) { // >0 left turn, <0 right, 0 collinear
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0]);
    }

    static int maxPoints(int[][] pts) {              // LeetCode 149: O(n^2), exact slopes
        int best = 0;
        for (int i = 0; i < pts.length; i++) {
            Map<Long, Integer> cnt = new HashMap<>();
            int top = 0;
            for (int j = i + 1; j < pts.length; j++) {
                int dy = pts[j][1] - pts[i][1], dx = pts[j][0] - pts[i][0];
                int g = (int) gcd(dy, dx);
                dy /= g; dx /= g;
                if (dx < 0 || (dx == 0 && dy < 0)) { dy = -dy; dx = -dx; }   // one sign convention
                long key = ((long) dy << 32) ^ (dx & 0xFFFFFFFFL);
                top = Math.max(top, cnt.merge(key, 1, Integer::sum));
            }
            best = Math.max(best, top + 1);
        }
        return best;
    }

    static void check(boolean ok, String what) { if (!ok) throw new AssertionError(what); }

    public static void main(String[] args) {
        check(modPow(3, 13, 1000) == 323, "modpow");
        check(Math.abs(myPow(2.0, -2) - 0.25) < 1e-12 && myPow(1.0, Integer.MIN_VALUE) == 1.0, "mypow");
        check(gcd(84, 36) == 12 && lcm(4, 6) == 12, "gcd");
        check(countPrimes(10) == 4 && countPrimes(2) == 0 && countPrimes(100) == 25, "sieve");
        check(reverseInt(-4310) == -134 && reverseInt(1534236469) == 0 && reverseInt(Integer.MIN_VALUE) == 0, "rev");
        check(addStrings("987", "45").equals("1032"), "add");
        check(multiplyStrings("123", "45").equals("5535") && multiplyStrings("999", "999").equals("998001"), "mul");
        check(pascal(5).get(4).equals(List.of(1, 4, 6, 4, 1)), "pascal");
        NCr nc = new NCr(100);
        check(nc.c(5, 2) == 10 && nc.c(3, 5) == 0, "ncr");
        WeightedPicker wp = new WeightedPicker(new int[]{2, 5, 3});
        int[] ts = {1, 2, 3, 7, 8, 10}, want = {0, 0, 1, 1, 2, 2};
        for (int k = 0; k < 6; k++) check(wp.pickWith(ts[k]) == want[k], "pick " + k);
        Random rng = new Random(7);
        int[] cnt = new int[3];
        for (int t = 0; t < 30000; t++) cnt[reservoirPick(List.of(0, 1, 2).iterator(), rng)]++;
        for (int c : cnt) check(c > 9000 && c < 11000, "reservoir");
        check(toBase(255, 16).equals("ff") && toBase(0, 7).equals("0"), "base");
        check(intToRoman(1994).equals("MCMXCIV") && romanToInt("MCMXCIV") == 1994, "roman");
        for (int n = 1; n < 4000; n++) check(romanToInt(intToRoman(n)) == n, "roman " + n);
        check(cross(new long[]{0, 0}, new long[]{1, 0}, new long[]{0, 1}) > 0, "cross");
        check(maxPoints(new int[][]{{1, 1}, {3, 2}, {5, 3}, {4, 1}, {2, 3}, {1, 4}}) == 4, "maxpts");

        System.out.println("MathX.java OK");
    }

# Math and geometry templates (coding-math lesson). Run: python mathx.py
import bisect
import random
from math import gcd

MOD = 1_000_000_007


def mod_pow(b, e, m=MOD):           # fast power: O(log e) multiplications
    b %= m
    r = 1
    while e:
        if e & 1:                   # this bit of e is set: multiply it in
            r = r * b % m
        b = b * b % m               # b, b^2, b^4, b^8 ...
        e >>= 1
    return r


def my_pow(x, n):                   # LeetCode 50: double x, 32-bit n (may be -2^31)
    if n < 0:
        x, n = 1 / x, -n            # Python ints do not overflow; in C/Java use a 64-bit n
    r = 1.0
    while n:
        if n & 1:
            r *= x
        x *= x
        n >>= 1
    return r


def lcm(a, b):
    return a // gcd(a, b) * b       # divide first: smaller intermediate


def count_primes(n):                # primes strictly below n (sieve of Eratosthenes)
    if n < 3:
        return 0
    is_p = bytearray([1]) * n
    is_p[0] = is_p[1] = 0
    i = 2
    while i * i < n:
        if is_p[i]:
            is_p[i * i::i] = bytearray(len(range(i * i, n, i)))   # start at i*i
        i += 1
    return sum(is_p)


INT_MAX, INT_MIN = 2**31 - 1, -2**31


def reverse_int(x):                 # LeetCode 7: 0 on 32-bit overflow
    sign = -1 if x < 0 else 1
    x, r = abs(x), 0
    limit = INT_MAX if sign > 0 else -INT_MIN       # 2^31 - 1 or 2^31
    while x:
        d = x % 10
        x //= 10
        if r > (limit - d) // 10:                   # r*10 + d would pass the limit
            return 0
        r = r * 10 + d
    return sign * r


def add_strings(a, b):              # school addition from the right
    i, j, carry, out = len(a) - 1, len(b) - 1, 0, []
    while i >= 0 or j >= 0 or carry:
        s = carry
        if i >= 0:
            s += ord(a[i]) - 48
            i -= 1
        if j >= 0:
            s += ord(b[j]) - 48
            j -= 1
        out.append(chr(48 + s % 10))
        carry = s // 10
    return "".join(reversed(out))


def multiply_strings(a, b):         # digit i * digit j lands in slots i+j and i+j+1
    if a == "0" or b == "0":
        return "0"
    res = [0] * (len(a) + len(b))
    for i in range(len(a) - 1, -1, -1):
        for j in range(len(b) - 1, -1, -1):
            p = (ord(a[i]) - 48) * (ord(b[j]) - 48) + res[i + j + 1]
            res[i + j + 1] = p % 10
            res[i + j] += p // 10
    s = "".join(map(str, res)).lstrip("0")
    return s


def pascal(rows):                   # LeetCode 118
    tri = [[1]]
    for _ in range(rows - 1):
        prev = tri[-1]
        tri.append([1] + [prev[k - 1] + prev[k] for k in range(1, len(prev))] + [1])
    return tri[:rows]


class NCr:                          # nCr mod p (p prime) with factorials + Fermat inverse
    def __init__(self, n, p=MOD):
        self.p = p
        self.f = [1] * (n + 1)
        for i in range(1, n + 1):
            self.f[i] = self.f[i - 1] * i % p
        self.inv = [1] * (n + 1)
        self.inv[n] = mod_pow(self.f[n], p - 2, p)     # a^(p-2) = a^-1 mod p
        for i in range(n, 0, -1):
            self.inv[i - 1] = self.inv[i] * i % p

    def c(self, n, r):
        if r < 0 or r > n:
            return 0
        return self.f[n] * self.inv[r] % self.p * self.inv[n - r] % self.p


class WeightedPicker:               # LeetCode 528: prefix sums + binary search
    def __init__(self, w, rng=random):
        self.pre = []
        total = 0
        for x in w:
            total += x
            self.pre.append(total)
        self.rng = rng

    def pick_with(self, t):         # t in [1, total]
        return bisect.bisect_left(self.pre, t)          # first prefix >= t

    def pick(self):
        return self.pick_with(self.rng.randint(1, self.pre[-1]))


def reservoir_pick(stream, rng=random):   # one uniform item from a stream of unknown length
    chosen, seen = None, 0
    for x in stream:
        seen += 1
        if rng.randrange(seen) == 0:       # keep the new item with probability 1/seen
            chosen = x
    return chosen


def to_base(n, b):                  # n >= 0, 2 <= b <= 36
    if n == 0:
        return "0"
    digits = "0123456789abcdefghijklmnopqrstuvwxyz"
    out = []
    while n:
        out.append(digits[n % b])
        n //= b
    return "".join(reversed(out))


ROMAN = [(1000, "M"), (900, "CM"), (500, "D"), (400, "CD"), (100, "C"), (90, "XC"),
         (50, "L"), (40, "XL"), (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")]


def int_to_roman(n):                # greedy: largest symbol that fits
    out = []
    for v, s in ROMAN:
        while n >= v:
            out.append(s)
            n -= v
    return "".join(out)


def roman_to_int(s):                # subtract when a smaller symbol comes before a larger one
    val = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}
    total = 0
    for i, ch in enumerate(s):
        v = val[ch]
        if i + 1 < len(s) and v < val[s[i + 1]]:
            total -= v
        else:
            total += v
    return total


def cross(o, a, b):                 # > 0: left turn, < 0: right turn, 0: collinear
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def slope_key(p, q):                # exact slope as a reduced (dy, dx) pair, no floats
    dy, dx = q[1] - p[1], q[0] - p[0]
    g = gcd(dy, dx)                 # never 0 for distinct points
    dy, dx = dy // g, dx // g
    if dx < 0 or (dx == 0 and dy < 0):
        dy, dx = -dy, -dx           # one sign convention
    return dy, dx


def max_points(points):             # LeetCode 149: O(n^2)
    best = 0
    for i, p in enumerate(points):
        count = {}
        for q in points[i + 1:]:
            k = slope_key(p, q)
            count[k] = count.get(k, 0) + 1
        best = max(best, 1 + max(count.values(), default=0))
    return best
