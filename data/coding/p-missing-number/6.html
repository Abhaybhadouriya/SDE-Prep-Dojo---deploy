The XOR version in four languages. The test drivers (the examples, n = 1 both ways, the missing value at 0 and at n, a 10,000-element case, and 500 random shuffles checked against the sum formula) are in coding-code/missing-number/.

func missingNumber(nums []int) int {
	acc := len(nums) // index n has no slot in the array, so start with it
	for i, x := range nums {
		acc ^= i ^ x // every present number meets its twin index and cancels
	}
	return acc // only the number that never appeared as a value survives
}
class Solution {
public:
    int missingNumber(vector<int>& nums) {
        int n = nums.size();
        int acc = n;  // index n has no slot in the array, so start with it
        for (int i = 0; i < n; i++) {
            acc ^= i ^ nums[i];  // every present number meets its twin index and cancels
        }
        return acc;  // only the number that never appeared as a value survives
    }
};
