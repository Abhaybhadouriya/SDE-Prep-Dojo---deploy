class Solution {
public:
    int maxArea(vector<int>& height) {
        int l = 0, r = (int)height.size() - 1;
        int best = 0;                       // max area is 1e4 * 1e5 = 1e9, fits in int
        while (l < r) {
            int level = min(height[l], height[r]);
            best = max(best, level * (r - l));
            // The shorter line is the limit; every narrower container that keeps it
            // has a level <= it and a smaller width, so it can be dropped for good.
            if (height[l] < height[r]) l++;
            else r--;
        }
        return best;
    }
};
func maxArea(height []int) int {
	l, r := 0, len(height)-1
	best := 0
	for l < r {
		level := height[l]
		if height[r] < level {
			level = height[r]
		}
		if area := level * (r - l); area > best {
			best = area
		}
		// The shorter line is the limit; every narrower container that keeps it
		// has a level <= it and a smaller width, so it can be dropped for good.
		if height[l] < height[r] {
			l++
		} else {
			r--
		}
	}
	return best
}
