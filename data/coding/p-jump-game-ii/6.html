func jump(nums []int) int {
	n := len(nums)
	jumps := 0
	levelEnd := 0 // last index reachable with `jumps` jumps
	farthest := 0 // farthest index reachable with `jumps + 1` jumps
	for i := 0; i+1 < n; i++ { // never "jump from" the last index
		if i+nums[i] > farthest {
			farthest = i + nums[i]
		}
		if i == levelEnd { // this level is fully scanned
			jumps++             // one more jump is needed...
			levelEnd = farthest // ...and it can land anywhere up to farthest
			if levelEnd >= n-1 { // the last index is inside the new level
				break
			}
		}
	}
	return jumps
}
