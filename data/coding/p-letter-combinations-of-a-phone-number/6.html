class Solution {
    const vector<string> keys = {"", "", "abc", "def", "ghi", "jkl",
                                 "mno", "pqrs", "tuv", "wxyz"};

    // path holds one letter per digit already decided (positions 0..i-1)
    void build(const string& digits, int i, string& path, vector<string>& out) {
        if (i == (int)digits.size()) {   // every digit has a letter: one full word
            out.push_back(path);
            return;
        }
        for (char c : keys[digits[i] - '0']) {
            path.push_back(c);           // choose
            build(digits, i + 1, path, out);
            path.pop_back();             // undo, try the next letter of this key
        }
    }

public:
    vector<string> letterCombinations(string digits) {
        vector<string> out;
        if (digits.empty()) return out;  // no digits -> no words (not [""])
        string path;
        build(digits, 0, path, out);
        return out;
    }
};

func letterCombinations(digits string) []string {
	out := []string{}
	if len(digits) == 0 { // no digits -> no words (not [""])
		return out
	}
	path := make([]byte, 0, len(digits)) // one letter per digit already decided
	var build func(i int)
	build = func(i int) {
		if i == len(digits) { // every digit has a letter: one full word
			out = append(out, string(path))
			return
		}
		letters := keys[digits[i]-'0']
		for j := 0; j < len(letters); j++ {
			path = append(path, letters[j]) // choose
			build(i + 1)
			path = path[:len(path)-1] // undo, try the next letter of this key
		}
	}
	build(0)
	return out
}
static class Solution {
    private static final String[] KEYS = {"", "", "abc", "def", "ghi", "jkl",
                                          "mno", "pqrs", "tuv", "wxyz"};

    public List<String> letterCombinations(String digits) {
        List<String> out = new ArrayList<>();
        if (digits.isEmpty()) return out;   // no digits -> no words (not [""])
        build(digits, 0, new StringBuilder(), out);
        return out;
    }

    // path holds one letter per digit already decided (positions 0..i-1)
    private void build(String digits, int i, StringBuilder path, List<String> out) {
        if (i == digits.length()) {         // every digit has a letter: one full word
            out.add(path.toString());
            return;
        }
        for (char c : KEYS[digits.charAt(i) - '0'].toCharArray()) {
            path.append(c);                 // choose
            build(digits, i + 1, path, out);
            path.deleteCharAt(path.length() - 1);  // undo
        }
    }
}
class Solution:
    KEYS = ["", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"]

    def letterCombinations(self, digits: str) -> List[str]:
        if not digits:                 # no digits -> no words (not [""])
            return []
        out, path = [], []

        def build(i: int) -> None:     # path holds one letter per digit 0..i-1
            if i == len(digits):       # every digit has a letter: one full word
                out.append("".join(path))
                return
            for c in self.KEYS[int(digits[i])]:
                path.append(c)         # choose
                build(i + 1)
                path.pop()             # undo, try the next letter of this key

        build(0)
        return out
