func isAnagram(s string, t string) bool {
	if len(s) != len(t) {
		return false // different lengths can never match
	}
	var count [26]int // count[c] = (#c in s) - (#c in t)
	for i := 0; i < len(s); i++ {
		count[s[i]-'a']++
		count[t[i]-'a']--
	}
	for _, c := range count {
		if c != 0 {
			return false // some letter is unbalanced
		}
	}
	return true
}

class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        if len(s) != len(t):                 # different lengths can never match
            return False
        count = [0] * 26                     # count[c] = (#c in s) - (#c in t)
        for a, b in zip(s, t):
            count[ord(a) - ord('a')] += 1
            count[ord(b) - ord('a')] -= 1
        return all(c == 0 for c in count)    # every letter must balance
