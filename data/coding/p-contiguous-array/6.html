One pass with a balance counter and a first-index array (offset by n, so no hash map); the tested files are in coding-code/contiguous-array/ (the Python driver also cross-checks 400 random arrays against the O(n2) version). A HashMap<Integer,Integer> seeded with {0: -1} is equally correct.

class Solution {
public:
    int findMaxLength(vector<int>& nums) {
        // bal = (#1s - #0s) in nums[0..i]. Equal 0s and 1s in nums[l+1..r]
        // <=> bal after r == bal after l. For the longest run ending at r we
        // want the EARLIEST index with the same balance, so store first seen only.
        int n = nums.size();
        vector<int> first(2 * n + 1, -2);   // bal is in [-n, n]: shift by n; -2 = "not seen"
        first[n] = -1;                      // bal 0 "before index 0": runs starting at index 0
        int bal = 0, best = 0;
        for (int i = 0; i < n; i++) {
            bal += nums[i] == 1 ? 1 : -1;
            int& f = first[bal + n];
            if (f == -2) f = i;             // first time: remember, never overwrite
            else best = max(best, i - f);
        }
        return best;
    }
};
func findMaxLength(nums []int) int {
	// bal = (#1s - #0s) in nums[0..i]. Equal 0s and 1s in nums[l+1..r]
	// <=> bal after r == bal after l. For the longest run ending at r we
	// want the EARLIEST index with the same balance, so store first seen only.
	n := len(nums)
	first := make([]int, 2*n+1) // bal is in [-n, n]: shift by n
	for i := range first {
		first[i] = -2 // -2 = "not seen"
	}
	first[n] = -1 // bal 0 "before index 0": runs starting at index 0
	bal, best := 0, 0
	for i, x := range nums {
		if x == 1 {
			bal++
		} else {
			bal--
		}
		if first[bal+n] == -2 {
			first[bal+n] = i // first time: remember, never overwrite
		} else if i-first[bal+n] > best {
			best = i - first[bal+n]
		}
	}
	return best
}

class Solution:
    def findMaxLength(self, nums: List[int]) -> int:
        # bal = (#1s - #0s) in nums[0..i]. Equal 0s and 1s in nums[l+1..r]
        # <=> bal after r == bal after l. For the longest run ending at r we
        # want the EARLIEST index with the same balance, so store first seen only.
        n = len(nums)
        first = [-2] * (2 * n + 1)   # bal is in [-n, n]: shift by n; -2 = "not seen"
        first[n] = -1                # bal 0 "before index 0": runs starting at index 0
        bal = best = 0
        for i, x in enumerate(nums):
            bal += 1 if x == 1 else -1
            if first[bal + n] == -2:
                first[bal + n] = i   # first time: remember, never overwrite
            else:
                best = max(best, i - first[bal + n])
        return best
