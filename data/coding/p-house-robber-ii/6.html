class Solution {
public:
    int rob(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return nums[0];  // both ranges below would be empty
        // either house n-1 is skipped, or house 0 is skipped (they are neighbours)
        return max(line(nums, 0, n - 2), line(nums, 1, n - 1));
    }

private:
    // House Robber I on nums[lo..hi]
    int line(const vector<int>& nums, int lo, int hi) {
        int prev2 = 0, prev1 = 0;  // best up to i-2, best up to i-1
        for (int i = lo; i <= hi; i++) {
            int cur = max(prev1, prev2 + nums[i]);  // skip house i, or take it
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
};
func rob(nums []int) int {
	n := len(nums)
	if n == 1 {
		return nums[0] // both ranges below would be empty
	}
	// either house n-1 is skipped, or house 0 is skipped (they are neighbours)
	a, b := robLine(nums[:n-1]), robLine(nums[1:])
	if a > b {
		return a
	}
	return b
}

// House Robber I on a line of houses
func robLine(houses []int) int {
	prev2, prev1 := 0, 0 // best up to i-2, best up to i-1
	for _, x := range houses {
		cur := prev1 // skip house i
		if prev2+x > cur {
			cur = prev2 + x // or take it
		}
		prev2, prev1 = prev1, cur
	}
	return prev1
}

class Solution:
    def rob(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1:
            return nums[0]          # both ranges below would be empty

        def line(lo: int, hi: int) -> int:
            # House Robber I on nums[lo..hi]: prev2/prev1 = best up to i-2 / i-1
            prev2, prev1 = 0, 0
            for i in range(lo, hi + 1):
                prev2, prev1 = prev1, max(prev1, prev2 + nums[i])
            return prev1

        # either house n-1 is skipped, or house 0 is skipped (they are neighbours)
        return max(line(0, n - 2), line(1, n - 1))
