func combinationSum(candidates []int, target int) [][]int {
	sort.Ints(candidates) // lets us stop a loop at the first value that is too big
	out := [][]int{}
	pick := []int{}
	var dfs func(start, left int)
	// choose the next value from candidates[start:]; never go back to a smaller index,
	// so each multiset is built in exactly one (non-decreasing) order
	dfs = func(start, left int) {
		if left == 0 {
			out = append(out, append([]int(nil), pick...)) // exact hit: record a copy
			return
		}
		for i := start; i < len(candidates) && candidates[i] <= left; i++ {
			pick = append(pick, candidates[i])
			dfs(i, left-candidates[i]) // i, not i + 1: the same value may be used again
			pick = pick[:len(pick)-1]  // undo the choice before trying the next value
		}
	}
	dfs(0, target)
	return out
}
class Solution {
public:
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());  // lets us stop a loop at the first value that is too big
        vector<vector<int>> out;
        vector<int> pick;
        dfs(candidates, 0, target, pick, out);
        return out;
    }

private:
    // Choose the next value from candidates[start..]; never go back to a smaller index,
    // so each multiset is built in exactly one (non-decreasing) order.
    void dfs(const vector<int>& c, int start, int left, vector<int>& pick, vector<vector<int>>& out) {
        if (left == 0) {
            out.push_back(pick);  // exact hit: record a copy
            return;
        }
        for (int i = start; i < (int)c.size() && c[i] <= left; i++) {
            pick.push_back(c[i]);
            dfs(c, i, left - c[i], pick, out);  // i, not i + 1: the same value may be used again
            pick.pop_back();                    // undo the choice before trying the next value
        }
    }
};

static class Solution {
    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        Arrays.sort(candidates);  // lets us stop a loop at the first value that is too big
        List<List<Integer>> out = new ArrayList<>();
        dfs(candidates, 0, target, new ArrayList<>(), out);
        return out;
    }

    // Choose the next value from candidates[start..]; never go back to a smaller index,
    // so each multiset is built in exactly one (non-decreasing) order.
    private void dfs(int[] c, int start, int left, List<Integer> pick, List<List<Integer>> out) {
        if (left == 0) {
            out.add(new ArrayList<>(pick));  // exact hit: record a copy
            return;
        }
        for (int i = start; i < c.length && c[i] <= left; i++) {
            pick.add(c[i]);
            dfs(c, i, left - c[i], pick, out);  // i, not i + 1: the same value may be used again
            pick.remove(pick.size() - 1);       // undo the choice before trying the next value
        }
    }
}
class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        cand = sorted(candidates)        # lets us stop a loop at the first value that is too big
        out, pick = [], []

        def dfs(start: int, left: int) -> None:
            # choose the next value from cand[start:]; never go back to a smaller index,
            # so each multiset is built in exactly one (non-decreasing) order
            if left == 0:
                out.append(pick[:])      # exact hit: record a copy
                return
            for i in range(start, len(cand)):
                if cand[i] > left:       # sorted: every later value is too big as well
                    break
                pick.append(cand[i])
                dfs(i, left - cand[i])   # i, not i + 1: the same value may be used again
                pick.pop()               # undo the choice before trying the next value

        dfs(0, target)
        return out
