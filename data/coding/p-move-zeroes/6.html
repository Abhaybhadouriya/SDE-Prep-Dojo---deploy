class Solution {
public:
    // One pass: swap every non-zero into the next write slot.
    void moveZeroes(vector<int>& nums) {
        int w = 0;                                   // nums[0..w) = non-zeros so far, in order
        for (int r = 0; r < (int)nums.size(); r++) {
            if (nums[r] != 0) {
                if (r != w) swap(nums[w], nums[r]);  // nums[w..r] was all zeros
                w++;
            }
        }
    }

    // Copy non-zeros forward, then fill the tail with zeros.
    void moveZeroesTwoPass(vector<int>& nums) {
        int w = 0;
        for (int x : nums)
            if (x != 0) nums[w++] = x;
        while (w < (int)nums.size()) nums[w++] = 0;
    }
};
// moveZeroes: one pass, swap every non-zero into the next write slot.
func moveZeroes(nums []int) {
	w := 0 // nums[:w] = non-zeros so far, in order
	for r := range nums {
		if nums[r] != 0 {
			if r != w { // nums[w..r] was all zeros
				nums[w], nums[r] = nums[r], nums[w]
			}
			w++
		}
	}
}

// moveZeroesTwoPass: copy non-zeros forward, then fill the tail with zeros.
func moveZeroesTwoPass(nums []int) {
	w := 0
	for _, x := range nums {
		if x != 0 {
			nums[w] = x
			w++
		}
	}
	for ; w < len(nums); w++ {
		nums[w] = 0
	}
}

class Solution:
    def moveZeroes(self, nums: List[int]) -> None:
        """One pass: swap every non-zero into the next write slot."""
        w = 0                                    # nums[0:w] = non-zeros so far, in order
        for r in range(len(nums)):
            if nums[r] != 0:
                if r != w:                       # skip the self-swap while no zero has been seen
                    nums[w], nums[r] = nums[r], nums[w]
                w += 1                           # nums[w:r+1] is now all zeros

    def moveZeroesTwoPass(self, nums: List[int]) -> None:
        """Copy non-zeros forward, then fill the tail with zeros."""
        w = 0
        for x in nums:
            if x != 0:
                nums[w] = x
                w += 1
        for i in range(w, len(nums)):
            nums[i] = 0
