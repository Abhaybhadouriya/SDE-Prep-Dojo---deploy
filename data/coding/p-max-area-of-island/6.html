C++, Java and Go use the recursive sink (depth ≤ 2,500 here). Python uses an explicit stack, because its default recursion limit is 1,000.

class Solution {
public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int best = 0;
        for (int r = 0; r < (int)grid.size(); ++r)
            for (int c = 0; c < (int)grid[0].size(); ++c)
                if (grid[r][c] == 1)                   // unvisited land: a new island
                    best = max(best, sink(grid, r, c));
        return best;
    }

private:
    // Flood-fill one island, turning 1 -> 0 so no cell is counted twice.
    // Returns the number of cells sunk (the island's area).
    int sink(vector<vector<int>>& grid, int r, int c) {
        if (r < 0 || c < 0 || r >= (int)grid.size() || c >= (int)grid[0].size()) return 0;
        if (grid[r][c] != 1) return 0;                 // water or already counted
        grid[r][c] = 0;                                // mark BEFORE recursing
        return 1 + sink(grid, r + 1, c) + sink(grid, r - 1, c)
                 + sink(grid, r, c + 1) + sink(grid, r, c - 1);
    }
};
func maxAreaOfIsland(grid [][]int) int {
	rows, cols := len(grid), len(grid[0])
	// Flood-fill one island, turning 1 -> 0 so no cell is counted twice.
	// Returns the number of cells sunk (the island's area).
	var sink func(r, c int) int
	sink = func(r, c int) int {
		if r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] != 1 {
			return 0 // off the map, water, or already counted
		}
		grid[r][c] = 0 // mark BEFORE recursing
		return 1 + sink(r+1, c) + sink(r-1, c) + sink(r, c+1) + sink(r, c-1)
	}
	best := 0
	for r := 0; r < rows; r++ {
		for c := 0; c < cols; c++ {
			if grid[r][c] == 1 { // unvisited land: a new island
				if a := sink(r, c); a > best {
					best = a
				}
			}
		}
	}
	return best
}

static class Solution {
    public int maxAreaOfIsland(int[][] grid) {
        int best = 0;
        for (int r = 0; r < grid.length; r++)
            for (int c = 0; c < grid[0].length; c++)
                if (grid[r][c] == 1)                   // unvisited land: a new island
                    best = Math.max(best, sink(grid, r, c));
        return best;
    }

    // Flood-fill one island, turning 1 -> 0 so no cell is counted twice.
    // Returns the number of cells sunk (the island's area).
    private int sink(int[][] grid, int r, int c) {
        if (r < 0 || c < 0 || r >= grid.length || c >= grid[0].length) return 0;
        if (grid[r][c] != 1) return 0;                 // water or already counted
        grid[r][c] = 0;                                // mark BEFORE recursing
        return 1 + sink(grid, r + 1, c) + sink(grid, r - 1, c)
                 + sink(grid, r, c + 1) + sink(grid, r, c - 1);
    }
}
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        rows, cols = len(grid), len(grid[0])

        def sink(r: int, c: int) -> int:
            # Iterative flood fill (no recursion limit worries): turn 1 -> 0
            # when a cell is pushed, so no cell is counted twice.
            grid[r][c] = 0
            stack, area = [(r, c)], 0
            while stack:
                cr, cc = stack.pop()
                area += 1
                for nr, nc in ((cr + 1, cc), (cr - 1, cc), (cr, cc + 1), (cr, cc - 1)):
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 1:
                        grid[nr][nc] = 0          # mark when pushed, not when popped
                        stack.append((nr, nc))
            return area

        best = 0
        for r in range(rows):
            for c in range(cols):
                if grid[r][c] == 1:               # unvisited land: a new island
                    best = max(best, sink(r, c))
        return best
