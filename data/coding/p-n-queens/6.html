func solveNQueens(n int) [][]string {
	boards := [][]string{}
	queenCol := make([]int, n) // queenCol[r] = column of the queen in row r
	colUsed := make([]bool, n)
	diagUsed := make([]bool, 2*n-1) // r - c + n - 1 is the same along a "\" diagonal
	antiUsed := make([]bool, 2*n-1) // r + c is the same along a "/" diagonal

	var place func(r int)
	place = func(r int) {
		if r == n { // every row has a queen: record the board
			b := make([]string, n)
			for row, c := range queenCol {
				b[row] = strings.Repeat(".", c) + "Q" + strings.Repeat(".", n-c-1)
			}
			boards = append(boards, b)
			return
		}
		for c := 0; c < n; c++ {
			d, a := r-c+n-1, r+c
			if colUsed[c] || diagUsed[d] || antiUsed[a] {
				continue // attacked by a queen in an earlier row
			}
			queenCol[r] = c
			colUsed[c], diagUsed[d], antiUsed[a] = true, true, true
			place(r + 1)
			colUsed[c], diagUsed[d], antiUsed[a] = false, false, false // undo: try the next column
		}
	}

	place(0)
	return boards
}
class Solution {
    int n;
    vector<vector<string>> boards;
    vector<string> board;
    vector<char> colUsed, diagUsed, antiUsed;   // char, not vector<bool> (bit-packed and slower)

    void place(int r) {
        if (r == n) {                    // every row has a queen: record the board
            boards.push_back(board);
            return;
        }
        for (int c = 0; c < n; c++) {
            int d = r - c + n - 1, a = r + c;   // "\" diagonal id, "/" diagonal id
            if (colUsed[c] || diagUsed[d] || antiUsed[a]) continue;   // attacked from an earlier row
            board[r][c] = 'Q';
            colUsed[c] = diagUsed[d] = antiUsed[a] = 1;
            place(r + 1);
            colUsed[c] = diagUsed[d] = antiUsed[a] = 0;              // undo: try the next column
            board[r][c] = '.';
        }
    }

public:
    vector<vector<string>> solveNQueens(int n) {
        this->n = n;
        boards.clear();
        board.assign(n, string(n, '.'));
        colUsed.assign(n, 0);
        diagUsed.assign(2 * n - 1, 0);
        antiUsed.assign(2 * n - 1, 0);
        place(0);
        return boards;
    }
};
static class Solution {
    private int n;
    private List<List<String>> boards;
    private char[][] board;
    private boolean[] colUsed, diagUsed, antiUsed;

    public List<List<String>> solveNQueens(int n) {
        this.n = n;
        boards = new ArrayList<>();
        board = new char[n][n];
        for (char[] row : board) Arrays.fill(row, '.');
        colUsed = new boolean[n];
        diagUsed = new boolean[2 * n - 1];   // r - c + n - 1 is the same along a "\" diagonal
        antiUsed = new boolean[2 * n - 1];   // r + c is the same along a "/" diagonal
        place(0);
        return boards;
    }

    private void place(int r) {
        if (r == n) {                        // every row has a queen: record the board
            List<String> b = new ArrayList<>();
            for (char[] row : board) b.add(new String(row));
            boards.add(b);
            return;
        }
        for (int c = 0; c < n; c++) {
            int d = r - c + n - 1, a = r + c;
            if (colUsed[c] || diagUsed[d] || antiUsed[a]) continue;   // attacked from an earlier row
            board[r][c] = 'Q';
            colUsed[c] = diagUsed[d] = antiUsed[a] = true;
            place(r + 1);
            colUsed[c] = diagUsed[d] = antiUsed[a] = false;          // undo: try the next column
            board[r][c] = '.';
        }
    }
}
class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        boards = []
        queen_col = [0] * n                 # queen_col[r] = column of the queen in row r
        col_used = [False] * n
        diag_used = [False] * (2 * n - 1)   # r - c + n - 1 is the same along a "\" diagonal
        anti_used = [False] * (2 * n - 1)   # r + c is the same along a "/" diagonal

        def place(r: int) -> None:
            if r == n:                      # every row has a queen: record the board
                boards.append(["." * c + "Q" + "." * (n - c - 1) for c in queen_col])
                return
            for c in range(n):
                d, a = r - c + n - 1, r + c
                if col_used[c] or diag_used[d] or anti_used[a]:
                    continue                # attacked by a queen in an earlier row
                queen_col[r] = c
                col_used[c] = diag_used[d] = anti_used[a] = True
                place(r + 1)
                col_used[c] = diag_used[d] = anti_used[a] = False   # undo: try the next column

        place(0)
        return boards
