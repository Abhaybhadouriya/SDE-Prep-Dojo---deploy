All four use the one-pass map. C++ and Java compute target - x in 64 bits, because it can reach ±2 × 10^9.

class Solution {
public:
    vector<int> twoSum(const vector<int>& nums, int target) {
        unordered_map<long long, int> seen;        // value -> index, elements left of i
        seen.reserve(nums.size() * 2);
        for (int i = 0; i < (int)nums.size(); ++i) {
            long long need = (long long)target - nums[i];   // 64-bit: |need| can reach 2e9
            auto it = seen.find(need);
            if (it != seen.end()) return {it->second, i};    // check before insert
            seen[nums[i]] = i;
        }
        return {};                                  // unreachable for valid input
    }
};
func twoSum(nums []int, target int) []int {
	seen := make(map[int]int, len(nums)) // value -> index, elements left of i
	for i, x := range nums {
		if j, ok := seen[target-x]; ok { // check before insert: no self-pairing
			return []int{j, i}
		}
		seen[x] = i
	}
	return nil // unreachable: exactly one answer is guaranteed
}
static class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Long, Integer> seen = new HashMap<>();      // value -> index, elements left of i
        for (int i = 0; i < nums.length; i++) {
            long need = (long) target - nums[i];        // can leave int range (|need| up to 2e9)
            Integer j = seen.get(need);
            if (j != null) return new int[]{j, i};      // check before insert
            seen.put((long) nums[i], i);
        }
        return new int[0];                              // unreachable for valid input
    }
}
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        seen = {}                      # value -> index, only for elements LEFT of i
        for i, x in enumerate(nums):
            need = target - x
            if need in seen:           # look up BEFORE inserting x: no self-pairing
                return [seen[need], i]
            seen[x] = i
        return []                      # unreachable: exactly one answer is guaranteed
