Reusable templates (Go, C++, Java, Python)

One file per language with four functions: window max, longest subarray with max - min <= limit (two deques), shortest subarray with sum at least K (deque over prefix sums), and Jump Game VI (deque over dp). Each file checks itself against brute force on thousands of random arrays and prints ok.

What is in each file (same names in every language; Go and Java use camelCase):

  • max_sliding_window - LC 239: sliding window maximum with a monotonic deque.
  • longest_subarray - Longest subarray with max - min <= limit. max_q decreasing, min_q increasing.
  • shortest_subarray - Shortest subarray with sum >= k (negatives allowed), -1 if none.
  • max_result - Jump Game VI: dp[i] = nums[i] + max(dp[i-k .. i-1]).

// A slice used as a deque: push back = append, pop back = dq[:len-1], pop front = dq[1:].
package main

import (
	"fmt"
	"math/rand"
)

// Max of every window of size k. dq holds INDEXES, values strictly decreasing.
func maxSlidingWindow(nums []int, k int) []int {
	dq := []int{}
	out := []int{}
	for i, x := range nums {
		for len(dq) > 0 && nums[dq[len(dq)-1]] <= x { // never the max again
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, i)
		if dq[0] <= i-k { // front slid out of the window
			dq = dq[1:]
		}
		if i >= k-1 {
			out = append(out, nums[dq[0]])
		}
	}
	return out
}

// Longest subarray with max - min <= limit. maxQ decreasing, minQ increasing.
func longestSubarray(nums []int, limit int) int {
	maxQ, minQ := []int{}, []int{}
	lo, best := 0, 0
	for hi, x := range nums {
		for len(maxQ) > 0 && nums[maxQ[len(maxQ)-1]] <= x {
			maxQ = maxQ[:len(maxQ)-1]
		}
		for len(minQ) > 0 && nums[minQ[len(minQ)-1]] >= x {
			minQ = minQ[:len(minQ)-1]
		}
		maxQ = append(maxQ, hi)
		minQ = append(minQ, hi)
		for nums[maxQ[0]]-nums[minQ[0]] > limit { // shrink until valid again
			lo++
			if maxQ[0] < lo {
				maxQ = maxQ[1:]
			}
			if minQ[0] < lo {
				minQ = minQ[1:]
			}
		}
		if hi-lo+1 > best {
			best = hi - lo + 1
		}
	}
	return best
}

// Shortest subarray with sum >= k (negatives allowed), -1 if none. dq: prefix indexes, increasing P.
func shortestSubarray(nums []int, k int64) int {
	n := len(nums)
	P := make([]int64, n+1)
	for i, x := range nums {
		P[i+1] = P[i] + int64(x)
	}
	dq := []int{}
	best := n + 1
	for j := 0; j <= n; j++ {
		for len(dq) > 0 && P[j]-P[dq[0]] >= k { // best start for j; no later j can beat it
			if j-dq[0] < best {
				best = j - dq[0]
			}
			dq = dq[1:]
		}
		for len(dq) > 0 && P[dq[len(dq)-1]] >= P[j] { // later AND lower start wins
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, j)
	}
	if best <= n {
		return best
	}
	return -1
}

// Jump Game VI: dp[i] = nums[i] + max(dp[i-k .. i-1]).
func maxResult(nums []int, k int) int {
	dp := make([]int, len(nums))
	dq := []int{}
	for i, x := range nums {
		if len(dq) > 0 && dq[0] < i-k {
			dq = dq[1:]
		}
		dp[i] = x
		if len(dq) > 0 {
			dp[i] += dp[dq[0]]
		}
		for len(dq) > 0 && dp[dq[len(dq)-1]] <= dp[i] {
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, i)
	}
	return dp[len(nums)-1]
}

func check(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

func main() {
	got := maxSlidingWindow([]int{1, 3, -1, -3, 5, 3, 6, 7}, 3)
	check(fmt.Sprint(got) == "[3 3 5 5 6 7]", "max")
	check(longestSubarray([]int{10, 1, 2, 4, 7, 2}, 5) == 4, "longest")
	check(shortestSubarray([]int{2, -1, 2}, 3) == 3 && shortestSubarray([]int{1, 2}, 4) == -1, "shortest")
	check(maxResult([]int{1, -1, -2, 4, -7, 3}, 2) == 7, "jump")
	r := rand.New(rand.NewSource(7))
	for t := 0; t < 3000; t++ {
		n := 1 + r.Intn(12)
		k, lim, kk := 1+r.Intn(n), r.Intn(12), int64(1+r.Intn(25))
		a := make([]int, n)
		for i := range a {
			a[i] = r.Intn(21) - 10
		}
		m := maxSlidingWindow(a, k)
		longest, shortest := 0, -1
		for i := 0; i < n; i++ {
			mx, mn, s := a[i], a[i], int64(0)
			for j := i; j < n; j++ {
				if a[j] > mx {
					mx = a[j]
				}
				if a[j] < mn {
					mn = a[j]
				}
				s += int64(a[j])
				if mx-mn <= lim && j-i+1 > longest {
					longest = j - i + 1
				}
				if s >= kk && (shortest == -1 || j-i+1 < shortest) {
					shortest = j - i + 1
				}
				if j == i+k-1 {
					check(m[i] == mx, "max rnd")
				}
			}
		}
		check(longestSubarray(a, lim) == longest, "longest rnd")
		check(shortestSubarray(a, kk) == shortest, "shortest rnd")
		dp := make([]int, n)
		dp[0] = a[0]
		for i := 1; i < n; i++ {
			dp[i] = -1 << 60
			for j := i - k; j < i; j++ {
				if j >= 0 && dp[j]+a[i] > dp[i] {
					dp[i] = dp[j] + a[i]
				}
			}
		}
		check(maxResult(a, k) == dp[n-1], "jump rnd")
	}
	fmt.Println("ok")
}

#include <algorithm>
#include <cassert>
#include <cstdio>
#include <cstdlib>
#include <deque>
#include <vector>
using namespace std;

// Max of every window of size k. dq holds INDEXES, values strictly decreasing front -> back.
vector<int> maxSlidingWindow(const vector<int>& nums, int k) {
    deque<int> dq;
    vector<int> out;
    for (int i = 0; i < (int)nums.size(); i++) {
        while (!dq.empty() && nums[dq.back()] <= nums[i]) dq.pop_back(); // they can never be the max again
        dq.push_back(i);
        if (dq.front() <= i - k) dq.pop_front();                       // front slid out of the window
        if (i >= k - 1) out.push_back(nums[dq.front()]);
    }
    return out;
}

// Longest subarray with max - min <= limit. Two deques: maxQ decreasing, minQ increasing.
int longestSubarray(const vector<int>& nums, int limit) {
    deque<int> maxQ, minQ;
    int lo = 0, best = 0;
    for (int hi = 0; hi < (int)nums.size(); hi++) {
        while (!maxQ.empty() && nums[maxQ.back()] <= nums[hi]) maxQ.pop_back();
        while (!minQ.empty() && nums[minQ.back()] >= nums[hi]) minQ.pop_back();
        maxQ.push_back(hi);
        minQ.push_back(hi);
        while (nums[maxQ.front()] - nums[minQ.front()] > limit) { // shrink until valid again
            lo++;
            if (maxQ.front() < lo) maxQ.pop_front();
            if (minQ.front() < lo) minQ.pop_front();
        }
        best = max(best, hi - lo + 1);
    }
    return best;
}

// Shortest non-empty subarray with sum >= k (values may be negative). -1 if none.
// P[j] - P[i] >= k with i < j. dq holds prefix indexes with INCREASING P.
int shortestSubarray(const vector<int>& nums, long long k) {
    int n = nums.size();
    vector<long long> P(n + 1, 0);
    for (int i = 0; i < n; i++) P[i + 1] = P[i] + nums[i];
    deque<int> dq;
    int best = n + 1;
    for (int j = 0; j <= n; j++) {
        while (!dq.empty() && P[j] - P[dq.front()] >= k) { // front is the best start for j; no later j can beat it
            best = min(best, j - dq.front());
            dq.pop_front();
        }
        while (!dq.empty() && P[dq.back()] >= P[j]) dq.pop_back(); // j is a later AND lower start: strictly better
        dq.push_back(j);
    }
    return best <= n ? best : -1;
}

// Jump Game VI: dp[i] = nums[i] + max(dp[i-k .. i-1]). Same window-max deque, over dp values.
int maxResult(const vector<int>& nums, int k) {
    int n = nums.size();
    vector<long long> dp(n);
    deque<int> dq;
    for (int i = 0; i < n; i++) {
        if (!dq.empty() && dq.front() < i - k) dq.pop_front();
        dp[i] = nums[i] + (dq.empty() ? 0 : dp[dq.front()]);
        while (!dq.empty() && dp[dq.back()] <= dp[i]) dq.pop_back();
        dq.push_back(i);
    }
    return (int)dp[n - 1];
}

// ---- brute force checks ----
static vector<int> bfMax(const vector<int>& a, int k) {
    vector<int> o;
    for (int i = 0; i + k <= (int)a.size(); i++) o.push_back(*max_element(a.begin() + i, a.begin() + i + k));
    return o;
}
static int bfLongest(const vector<int>& a, int limit) {
    int best = 0;
    for (int i = 0; i < (int)a.size(); i++) {
        int mx = a[i], mn = a[i];
        for (int j = i; j < (int)a.size(); j++) {
            mx = max(mx, a[j]); mn = min(mn, a[j]);
            if (mx - mn <= limit) best = max(best, j - i + 1);
        }
    }
    return best;
}
static int bfShortest(const vector<int>& a, long long k) {
    int best = -1;
    for (int i = 0; i < (int)a.size(); i++) {
        long long s = 0;
        for (int j = i; j < (int)a.size(); j++) {
            s += a[j];
            if (s >= k && (best == -1 || j - i + 1 < best)) best = j - i + 1;
        }
    }
    return best;
}
static int bfJump(const vector<int>& a, int k) {
    int n = a.size();
    vector<long long> dp(n, -(1LL << 60));
    dp[0] = a[0];
    for (int i = 1; i < n; i++)
        for (int j = max(0, i - k); j < i; j++) dp[i] = max(dp[i], dp[j] + a[i]);
    return (int)dp[n - 1];
}

}

import java.util.*;

public class MonoDeque {
    // Max of every window of size k. dq holds INDEXES, values strictly decreasing.
    static int[] maxSlidingWindow(int[] nums, int k) {
        Deque<Integer> dq = new ArrayDeque<>();
        int[] out = new int[nums.length - k + 1];
        for (int i = 0; i < nums.length; i++) {
            while (!dq.isEmpty() && nums[dq.peekLast()] <= nums[i]) dq.pollLast(); // never the max again
            dq.offerLast(i);
            if (dq.peekFirst() <= i - k) dq.pollFirst();                          // slid out
            if (i >= k - 1) out[i - k + 1] = nums[dq.peekFirst()];
        }
        return out;
    }

    // Longest subarray with max - min <= limit. maxQ decreasing, minQ increasing.
    static int longestSubarray(int[] nums, int limit) {
        Deque<Integer> maxQ = new ArrayDeque<>(), minQ = new ArrayDeque<>();
        int lo = 0, best = 0;
        for (int hi = 0; hi < nums.length; hi++) {
            while (!maxQ.isEmpty() && nums[maxQ.peekLast()] <= nums[hi]) maxQ.pollLast();
            while (!minQ.isEmpty() && nums[minQ.peekLast()] >= nums[hi]) minQ.pollLast();
            maxQ.offerLast(hi);
            minQ.offerLast(hi);
            while (nums[maxQ.peekFirst()] - nums[minQ.peekFirst()] > limit) { // shrink until valid
                lo++;
                if (maxQ.peekFirst() < lo) maxQ.pollFirst();
                if (minQ.peekFirst() < lo) minQ.pollFirst();
            }
            best = Math.max(best, hi - lo + 1);
        }
        return best;
    }

    // Shortest subarray with sum >= k (negatives allowed), -1 if none. dq: prefix indexes, increasing P.
    static int shortestSubarray(int[] nums, long k) {
        int n = nums.length;
        long[] P = new long[n + 1];
        for (int i = 0; i < n; i++) P[i + 1] = P[i] + nums[i];
        Deque<Integer> dq = new ArrayDeque<>();
        int best = n + 1;
        for (int j = 0; j <= n; j++) {
            while (!dq.isEmpty() && P[j] - P[dq.peekFirst()] >= k) best = Math.min(best, j - dq.pollFirst());
            while (!dq.isEmpty() && P[dq.peekLast()] >= P[j]) dq.pollLast(); // later AND lower start wins
            dq.offerLast(j);
        }
        return best <= n ? best : -1;
    }

    // Jump Game VI: dp[i] = nums[i] + max(dp[i-k .. i-1]).
    static int maxResult(int[] nums, int k) {
        int n = nums.length;
        long[] dp = new long[n];
        Deque<Integer> dq = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            if (!dq.isEmpty() && dq.peekFirst() < i - k) dq.pollFirst();
            dp[i] = nums[i] + (dq.isEmpty() ? 0 : dp[dq.peekFirst()]);
            while (!dq.isEmpty() && dp[dq.peekLast()] <= dp[i]) dq.pollLast();
            dq.offerLast(i);
        }
        return (int) dp[n - 1];
    }

    static void check(boolean ok, String what) { if (!ok) throw new AssertionError(what); }

    public static void main(String[] args) {
        check(Arrays.equals(maxSlidingWindow(new int[]{1, 3, -1, -3, 5, 3, 6, 7}, 3), new int[]{3, 3, 5, 5, 6, 7}), "max");
        check(longestSubarray(new int[]{10, 1, 2, 4, 7, 2}, 5) == 4, "longest");
        check(shortestSubarray(new int[]{2, -1, 2}, 3) == 3 && shortestSubarray(new int[]{1, 2}, 4) == -1, "shortest");
        check(maxResult(new int[]{1, -1, -2, 4, -7, 3}, 2) == 7, "jump");
        Random r = new Random(7);
        for (int t = 0; t < 3000; t++) {
            int n = 1 + r.nextInt(12), k = 1 + r.nextInt(n), lim = r.nextInt(12);
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = r.nextInt(21) - 10;
            int[] m = maxSlidingWindow(a, k);
            int longest = 0, shortest = -1;
            long kk = 1 + r.nextInt(25);
            for (int i = 0; i < n; i++) {
                int mx = a[i], mn = a[i];
                long s = 0;
                for (int j = i; j < n; j++) {
                    mx = Math.max(mx, a[j]); mn = Math.min(mn, a[j]); s += a[j];
                    if (mx - mn <= lim) longest = Math.max(longest, j - i + 1);
                    if (s >= kk && (shortest == -1 || j - i + 1 < shortest)) shortest = j - i + 1;
                    if (j == i + k - 1) check(m[i] == mx, "max rnd");
                }
            }
            check(longestSubarray(a, lim) == longest, "longest rnd");
            check(shortestSubarray(a, kk) == shortest, "shortest rnd");
            long[] dp = new long[n];
            dp[0] = a[0];
            for (int i = 1; i < n; i++) {
                dp[i] = Long.MIN_VALUE;
                for (int j = Math.max(0, i - k); j < i; j++) dp[i] = Math.max(dp[i], dp[j] + a[i]);
            }
            check(maxResult(a, k) == dp[n - 1], "jump rnd");
        }
        System.out.println("ok");
    }
}
# Monotonic deque templates (SDE Prep Dojo, lesson coding-monotonic-deque). Python 3.11.
# Run: python deque.py   (random tests against brute force; prints "ok")
import random
from collections import deque


def max_sliding_window(nums, k):
    """Max of every window of size k. dq holds indexes, values strictly decreasing."""
    dq, out = deque(), []
    for i, x in enumerate(nums):
        while dq and nums[dq[-1]] <= x:   # they can never be the max again
            dq.pop()
        dq.append(i)
        if dq[0] <= i - k:                # front slid out of the window
            dq.popleft()
        if i >= k - 1:
            out.append(nums[dq[0]])
    return out


def longest_subarray(nums, limit):
    """Longest subarray with max - min <= limit. max_q decreasing, min_q increasing."""
    max_q, min_q = deque(), deque()
    lo = best = 0
    for hi, x in enumerate(nums):
        while max_q and nums[max_q[-1]] <= x:
            max_q.pop()
        while min_q and nums[min_q[-1]] >= x:
            min_q.pop()
        max_q.append(hi)
        min_q.append(hi)
        while nums[max_q[0]] - nums[min_q[0]] > limit:   # shrink until valid again
            lo += 1
            if max_q[0] < lo:
                max_q.popleft()
            if min_q[0] < lo:
                min_q.popleft()
        best = max(best, hi - lo + 1)
    return best


def shortest_subarray(nums, k):
    """Shortest subarray with sum >= k (negatives allowed), -1 if none.
    dq holds prefix indexes with increasing P."""
    P = [0]
    for x in nums:
        P.append(P[-1] + x)
    dq, best = deque(), len(nums) + 1
    for j, pj in enumerate(P):
        while dq and pj - P[dq[0]] >= k:   # best start for j; no later j can beat it
            best = min(best, j - dq.popleft())
        while dq and P[dq[-1]] >= pj:      # j is a later AND lower start: strictly better
            dq.pop()
        dq.append(j)
    return best if best <= len(nums) else -1


def max_result(nums, k):
    """Jump Game VI: dp[i] = nums[i] + max(dp[i-k .. i-1])."""
    dp = [0] * len(nums)
    dq = deque()
    for i, x in enumerate(nums):
        if dq and dq[0] < i - k:
            dq.popleft()
        dp[i] = x + (dp[dq[0]] if dq else 0)
        while dq and dp[dq[-1]] <= dp[i]:
            dq.pop()
        dq.append(i)
    return dp[-1]


def _bf_shortest(a, k):
    best = -1
    for i in range(len(a)):
        s = 0
        for j in range(i, len(a)):
            s += a[j]
            if s >= k and (best == -1 or j - i + 1 < best):
                best = j - i + 1
    return best


def _bf_longest(a, lim):
    return max(j - i + 1 for i in range(len(a)) for j in range(i, len(a))
               if max(a[i:j + 1]) - min(a[i:j + 1]) <= lim)


def _bf_jump(a, k):
    dp = [a[0]] + [None] * (len(a) - 1)
    for i in range(1, len(a)):
        dp[i] = a[i] + max(dp[max(0, i - k):i])
    return dp[-1]
