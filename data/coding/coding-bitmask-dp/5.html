Assignment problems: popcount tells you who is next

Minimum XOR Sum of Two Arrays (LeetCode 1879), told our way: two integer arrays a and b of the same length n. You may reorder b any way you like; then the cost is the sum of a[i] XOR b[i]. Return the smallest possible cost. Constraints: 1 <= n <= 14, values between 0 and 107. Example: a = [2, 7, 4], b = [6, 3, 5] gives 3 (pair 2 with 3, 7 with 6, 4 with 5: 1 + 1 + 1).

Reordering b is a perfect matching between the two arrays: n! options. The key observation: if we always give out the a's in order, then after popcount(mask) assignments the next a to serve is a[popcount(mask)]. So mask (which b's are taken) is the whole state, and dp[mask] = cheapest way to match a[0 .. popcount(mask) - 1] with exactly the b's in mask.

The same template solves Maximum Compatibility Score Sum (students and mentors), Campus Bikes II (workers and bikes, Manhattan distance), and Maximum AND Sum of Array (each slot has capacity 2, so use a base-3 "mask" or two bits per slot).

// minimumXORSum (LC 1879): assign b[j] to a[i], each used once.
// dp[mask] = best cost when the b's in mask are given to a[0 .. popcount(mask)-1].
func minimumXORSum(a, b []int) int {
	n := len(a)
	dp := make([]int, 1<<n)
	for i := range dp {
		dp[i] = math.MaxInt
	}
	dp[0] = 0
	for mask := 0; mask < 1<<n; mask++ {
		if dp[mask] == math.MaxInt {
			continue
		}
		i := bits.OnesCount(uint(mask)) // next a to assign
		if i == n {
			continue
		}
		for j := 0; j < n; j++ {
			if mask&(1<<j) == 0 {
				dp[mask|1<<j] = min(dp[mask|1<<j], dp[mask]+(a[i]^b[j]))
			}
		}
	}
	return dp[1<<n-1]
}
// Minimum XOR Sum of Two Arrays (LC 1879): assign b[j] to a[i], each used once.
// dp[mask] = best cost when the b's in mask are given to a[0 .. popcount(mask)-1].
int minimumXORSum(const vector<int>& a, const vector<int>& b) {
    int n = a.size();
    vector<int> dp(1 << n, INT_MAX);
    dp[0] = 0;
    for (int mask = 0; mask < (1 << n); mask++) {
        if (dp[mask] == INT_MAX) continue;
        int i = __builtin_popcount(mask);            // next a to assign
        if (i == n) continue;
        for (int j = 0; j < n; j++)
            if (!(mask & (1 << j)))
                dp[mask | (1 << j)] = min(dp[mask | (1 << j)], dp[mask] + (a[i] ^ b[j]));
    }
    return dp[(1 << n) - 1];
}
// Minimum XOR Sum of Two Arrays (LC 1879): assign b[j] to a[i], each used once.
// dp[mask] = best cost when the b's in mask are given to a[0 .. bitCount(mask)-1].
static int minimumXORSum(int[] a, int[] b) {
    int n = a.length;
    int[] dp = new int[1 << n];
    Arrays.fill(dp, Integer.MAX_VALUE);
    dp[0] = 0;
    for (int mask = 0; mask < (1 << n); mask++) {
        if (dp[mask] == Integer.MAX_VALUE) continue;
        int i = Integer.bitCount(mask);              // next a to assign
        if (i == n) continue;
        for (int j = 0; j < n; j++)
            if ((mask & (1 << j)) == 0)
                dp[mask | (1 << j)] = Math.min(dp[mask | (1 << j)], dp[mask] + (a[i] ^ b[j]));
    }
    return dp[(1 << n) - 1];
}
def minimum_xor_sum(a: List[int], b: List[int]) -> int:
    """LC 1879. dp[mask] = best cost when the b's in mask go to a[0 .. popcount(mask)-1]."""
    n = len(a)
    INF = float("inf")
    dp = [INF] * (1 << n)
    dp[0] = 0
    for mask in range(1 << n):
        if dp[mask] == INF:
            continue
        i = bin(mask).count("1")  # next a to assign
        if i == n:
            continue
        for j in range(n):
            if not mask & (1 << j):
                nm = mask | (1 << j)
                dp[nm] = min(dp[nm], dp[mask] + (a[i] ^ b[j]))
    return dp[-1]

For assignment with n in the hundreds, the real algorithm is the Hungarian method, O(n3), or min-cost max-flow. Mention it if asked "what if n = 500?", but interviewers almost never ask you to code it. The constraint n <= 14 is the signal they want the mask DP.
