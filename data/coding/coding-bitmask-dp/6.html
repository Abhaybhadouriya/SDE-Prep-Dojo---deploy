Splitting items among k people: DP over submasks

Fair Distribution of Cookies (LeetCode 2305), told our way: you have bags of cookies (cookies[i] cookies in bag i) and k children. Every bag goes, whole, to exactly one child. The unfairness is the largest total any one child gets. Return the smallest possible unfairness. Constraints: 2 <= n <= 8, bag sizes between 1 and 105, 2 <= k <= n. Example: bags [5, 9, 4, 7, 3] and 2 children give 14 ({9, 5} and {4, 7, 3}).

Now each step hands a whole group of items to one person, so the transition enumerates submasks. Let best_c[mask] = smallest possible max share when the bags in mask are split among c children. Then

best_1[mask] = sum[mask]
best_c[mask] = min over sub of mask of  max(best_(c-1)[mask ^ sub], sum[sub])   # child c takes sub

With the precomputed subset sums and the submask walk, that is O(k 3n): tiny for n = 8. With n = 8 a plain backtracking over "which child gets bag i" (kn = 88 = 1.7 x 107 with pruning) also passes, which is why most top solutions are backtracking; the submask DP is the one that still works when n grows to about 16 and k is small.

// distributeCookies (LC 2305): k children, minimise the largest share.
// best[mask] = smallest possible max share giving the bags in mask to c children. O(k * 3^n).
func distributeCookies(cookies []int, k int) int {
	n := len(cookies)
	full := 1<<n - 1
	sum := subsetSums(cookies)
	best := append([]int(nil), sum...) // one child takes everything in mask
	for c := 2; c <= k; c++ {
		next := make([]int, 1<<n)
		for mask := 0; mask <= full; mask++ {
			res := best[mask] // the new child gets nothing
			for sub := mask; sub > 0; sub = (sub - 1) & mask { // the new child takes sub
				res = min(res, max(best[mask^sub], sum[sub]))
			}
			next[mask] = res
		}
		best = next
	}
	return best[full]
}
// Fair Distribution of Cookies (LC 2305): k children, minimise the largest share.
// best[c][mask] = smallest possible max share giving the bags in mask to c children. O(k * 3^n).
int distributeCookies(const vector<int>& cookies, int k) {
    int n = cookies.size(), full = (1 << n) - 1;
    vector<int> sum = subsetSums(cookies);
    vector<int> best(sum);                           // one child takes everything in mask
    for (int c = 2; c <= k; c++) {
        vector<int> next(1 << n);
        for (int mask = 0; mask <= full; mask++) {
            int res = best[mask];                    // the new child gets nothing
            for (int sub = mask; sub > 0; sub = (sub - 1) & mask)  // the new child takes `sub`
                res = min(res, max(best[mask ^ sub], sum[sub]));
            next[mask] = res;
        }
        best = next;
    }
    return best[full];
}

static int distributeCookies(int[] cookies, int k) {
    int n = cookies.length, full = (1 << n) - 1;
    int[] sum = subsetSums(cookies);
    int[] best = sum.clone();                        // one child takes everything in mask
    for (int c = 2; c <= k; c++) {
        int[] next = new int[1 << n];
        for (int mask = 0; mask <= full; mask++) {
            int res = best[mask];                    // the new child gets nothing
            for (int sub = mask; sub > 0; sub = (sub - 1) & mask)  // the new child takes `sub`
                res = Math.min(res, Math.max(best[mask ^ sub], sum[sub]));
            next[mask] = res;
        }
        best = next;
    }
    return best[full];
}
def distribute_cookies(cookies: List[int], k: int) -> int:
    """LC 2305. best[mask] = smallest max share giving the bags in mask to c children. O(k * 3^n)."""
    n = len(cookies)
    full = (1 << n) - 1
    s = subset_sums(cookies)
    best = s[:]  # one child takes everything in mask
    for _ in range(2, k + 1):
        nxt = [0] * (1 << n)
        for mask in range(full + 1):
            res, sub = best[mask], mask  # sub = 0 (empty share) gives best[mask]
            while sub:  # the new child takes `sub`
                res = min(res, max(best[mask ^ sub], s[sub]))
                sub = (sub - 1) & mask
            nxt[mask] = res
        best = nxt
    return best[full]
