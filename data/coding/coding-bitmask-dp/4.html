Shortest path visiting all nodes: BFS over (node, mask)

Shortest Path Visiting All Nodes (LeetCode 847), told our way: an undirected, connected graph with n nodes is given as adjacency lists. Return the length (number of edges) of the shortest walk that visits every node. You may start and end anywhere, revisit nodes and reuse edges. Constraints: 1 <= n <= 12. Example: a star with centre 0 and leaves 1, 2, 3 gives 4 (1, 0, 2, 0, 3).

Where you can go next depends on where you are and which nodes you have seen, so the state is the pair (node, visited mask). Every edge costs 1, so plain BFS over these n x 2n states finds the shortest walk; start from all n states (i, 1 << i) at distance 0 (multi-source BFS) and stop at the first state whose mask is full.

Cost: n x 2n states, each scanning its neighbours, so O(2n x (n + E)). For n = 12 that is about 49,000 states.

// shortestPathLength (LC 847): BFS over states (node, visited mask).
func shortestPathLength(graph [][]int) int {
	n := len(graph)
	full := 1<<n - 1
	seen := make([][]bool, n)
	type state struct{ node, mask int }
	var q []state
	for i := range seen { // multi-source: start anywhere
		seen[i] = make([]bool, 1<<n)
		seen[i][1<<i] = true
		q = append(q, state{i, 1 << i})
	}
	for steps := 0; len(q) > 0; steps++ {
		var next []state // one BFS layer = one more edge
		for _, s := range q {
			if s.mask == full {
				return steps
			}
			for _, nb := range graph[s.node] {
				nm := s.mask | 1<<nb
				if !seen[nb][nm] {
					seen[nb][nm] = true
					next = append(next, state{nb, nm})
				}
			}
		}
		q = next
	}
	return -1 // unreachable for a connected graph
}
// Shortest Path Visiting All Nodes (LC 847): BFS over states (node, visited mask).
int shortestPathLength(const vector<vector<int>>& graph) {
    int n = graph.size(), full = (1 << n) - 1;
    vector<vector<char>> seen(n, vector<char>(1 << n, 0));
    queue<pair<int, int>> q;
    for (int i = 0; i < n; i++) {                    // multi-source: start anywhere
        q.push({i, 1 << i});
        seen[i][1 << i] = 1;
    }
    for (int steps = 0; !q.empty(); steps++) {
        for (int sz = q.size(); sz > 0; sz--) {      // one BFS layer = one more edge
            int node = q.front().first, mask = q.front().second;
            q.pop();
            if (mask == full) return steps;
            for (int nb : graph[node]) {
                int nm = mask | (1 << nb);
                if (!seen[nb][nm]) {
                    seen[nb][nm] = 1;
                    q.push({nb, nm});
                }
            }
        }
    }
    return -1;                                       // unreachable for a connected graph
}

static int shortestPathLength(int[][] graph) {
    int n = graph.length, full = (1 << n) - 1;
    boolean[][] seen = new boolean[n][1 << n];
    ArrayDeque<int[]> q = new ArrayDeque<>();
    for (int i = 0; i < n; i++) {                    // multi-source: start anywhere
        q.add(new int[]{i, 1 << i});
        seen[i][1 << i] = true;
    }
    for (int steps = 0; !q.isEmpty(); steps++) {
        for (int sz = q.size(); sz > 0; sz--) {      // one BFS layer = one more edge
            int[] cur = q.poll();
            if (cur[1] == full) return steps;
            for (int nb : graph[cur[0]]) {
                int nm = cur[1] | (1 << nb);
                if (!seen[nb][nm]) {
                    seen[nb][nm] = true;
                    q.add(new int[]{nb, nm});
                }
            }
        }
    }
    return -1;                                       // unreachable for a connected graph
}
def shortest_path_length(graph: List[List[int]]) -> int:
    """LC 847: BFS over states (node, visited mask)."""
    n = len(graph)
    full = (1 << n) - 1
    q = deque((i, 1 << i) for i in range(n))  # multi-source: start anywhere
    seen = {(i, 1 << i) for i in range(n)}
    steps = 0
    while q:
        for _ in range(len(q)):  # one BFS layer = one more edge
            node, mask = q.popleft()
            if mask == full:
                return steps
            for nb in graph[node]:
                state = (nb, mask | (1 << nb))
                if state not in seen:
                    seen.add(state)
                    q.append(state)
        steps += 1
    return -1  # unreachable for a connected graph

The weighted version: Held-Karp

If edges have weights (travelling salesman, Find the Shortest Superstring where the "weight" is the overlap between two words), BFS no longer works. Use dp[mask][last] = cheapest path that visits exactly mask and ends at last; transition to every next not in mask: dp[mask | 1 << next][next] = min(..., dp[mask][last] + w[last][next]). That is O(n2 2n), the classic Held-Karp algorithm. To rebuild the path (needed for the superstring), keep a parent[mask][last] table.

From your world

A BFS over (node, mask) is a BFS over a product graph, the same idea as running a protocol FSM in lockstep with a topology: the "real" position is the node, the extra mask is the protocol's memory. Shortest Path to Get All Keys (LeetCode 864) is the grid version, with the mask being the set of keys picked up.
