All four use the optimal cut search on the shorter array, with "minus / plus infinity" sentinels for empty sides.

class Solution {
public:
    double findMedianSortedArrays(vector<int>& a, vector<int>& b) {
        if (a.size() > b.size()) return findMedianSortedArrays(b, a);  // search the shorter one
        int m = a.size(), n = b.size();
        int half = (m + n + 1) / 2;          // size of the left part (gets the extra one when odd)
        int lo = 0, hi = m;                  // i = how many of a go to the left part
        while (lo <= hi) {
            int i = lo + (hi - lo) / 2;
            int j = half - i;                // the rest of the left part comes from b
            int aLeft  = i > 0 ? a[i - 1] : INT_MIN;   // empty side = -inf / +inf
            int aRight = i < m ? a[i]     : INT_MAX;
            int bLeft  = j > 0 ? b[j - 1] : INT_MIN;
            int bRight = j < n ? b[j]     : INT_MAX;
            if (aLeft <= bRight && bLeft <= aRight) {   // every left value <= every right value
                int leftMax = max(aLeft, bLeft);
                if ((m + n) % 2 == 1) return leftMax;
                return (leftMax + (double)min(aRight, bRight)) / 2.0;
            }
            if (aLeft > bRight) hi = i - 1;  // took too many from a
            else lo = i + 1;                 // took too few from a
        }
        return 0.0;                          // unreachable for sorted input
    }
};
func findMedianSortedArrays(a []int, b []int) float64 {
	if len(a) > len(b) {
		a, b = b, a // binary-search the shorter array
	}
	m, n := len(a), len(b)
	half := (m + n + 1) / 2 // size of the left part (gets the extra one when odd)
	lo, hi := 0, m          // i = how many of a go to the left part
	for lo <= hi {
		i := lo + (hi-lo)/2
		j := half - i // the rest of the left part comes from b
		aLeft, aRight := math.MinInt, math.MaxInt // empty side = -inf / +inf
		bLeft, bRight := math.MinInt, math.MaxInt
		if i > 0 {
			aLeft = a[i-1]
		}
		if i < m {
			aRight = a[i]
		}
		if j > 0 {
			bLeft = b[j-1]
		}
		if j < n {
			bRight = b[j]
		}
		if aLeft <= bRight && bLeft <= aRight { // every left value <= every right value
			leftMax := aLeft
			if bLeft > leftMax {
				leftMax = bLeft
			}
			if (m+n)%2 == 1 {
				return float64(leftMax)
			}
			rightMin := aRight
			if bRight < rightMin {
				rightMin = bRight
			}
			return (float64(leftMax) + float64(rightMin)) / 2.0
		}
		if aLeft > bRight {
			hi = i - 1 // took too many from a
		} else {
			lo = i + 1 // took too few from a
		}
	}
	return 0.0 // unreachable for sorted input
}

class Solution:
    def findMedianSortedArrays(self, a: List[int], b: List[int]) -> float:
        if len(a) > len(b):
            a, b = b, a                       # binary-search the shorter array
        m, n = len(a), len(b)
        half = (m + n + 1) // 2               # size of the left part (gets the extra one when odd)
        lo, hi = 0, m                         # i = how many of a go to the left part
        INF = float("inf")
        while lo <= hi:
            i = (lo + hi) // 2
            j = half - i                      # the rest of the left part comes from b
            a_left = a[i - 1] if i > 0 else -INF     # empty side = -inf / +inf
            a_right = a[i] if i < m else INF
            b_left = b[j - 1] if j > 0 else -INF
            b_right = b[j] if j < n else INF
            if a_left <= b_right and b_left <= a_right:   # every left value <= every right value
                left_max = max(a_left, b_left)
                if (m + n) % 2:
                    return float(left_max)
                return (left_max + min(a_right, b_right)) / 2
            if a_left > b_right:
                hi = i - 1                    # took too many from a
            else:
                lo = i + 1                    # took too few from a
        return 0.0                            # unreachable for sorted input
