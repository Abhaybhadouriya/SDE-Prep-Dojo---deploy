All four run k + 1 Bellman-Ford rounds over the edge list, reading the previous round and writing a copy.

// Bellman-Ford limited to k + 1 rounds: after round r, cost[v] is the cheapest
// price to reach v using at most r flights.
func findCheapestPrice(n int, flights [][]int, src int, dst int, k int) int {
	const inf = math.MaxInt32
	cost := make([]int, n)
	for i := range cost {
		cost[i] = inf
	}
	cost[src] = 0
	for round := 0; round <= k; round++ { // k stops = at most k + 1 flights
		next := append([]int(nil), cost...) // relax from the OLD round only
		for _, f := range flights {
			u, v, price := f[0], f[1], f[2]
			if cost[u] != inf && cost[u]+price < next[v] {
				next[v] = cost[u] + price
			}
		}
		cost = next
	}
	if cost[dst] == inf {
		return -1
	}
	return cost[dst]
}
class Solution {
public:
    // Bellman-Ford limited to k + 1 rounds: after round r, cost[v] is the cheapest
    // price to reach v using at most r flights.
    int findCheapestPrice(int n, const vector<vector<int>>& flights, int src, int dst, int k) {
        const int INF = 1e9;
        vector<int> cost(n, INF);
        cost[src] = 0;
        for (int round = 0; round <= k; round++) {     // k stops = at most k + 1 flights
            vector<int> next = cost;                   // relax from the OLD round only
            for (const auto& f : flights) {
                int u = f[0], v = f[1], price = f[2];
                if (cost[u] != INF && cost[u] + price < next[v])
                    next[v] = cost[u] + price;
            }
            cost.swap(next);
        }
        return cost[dst] == INF ? -1 : cost[dst];
    }
};

class Solution:
    def findCheapestPrice(self, n: int, flights: List[List[int]], src: int, dst: int, k: int) -> int:
        # Bellman-Ford limited to k + 1 rounds: after round r, cost[v] is the cheapest
        # price to reach v using at most r flights.
        INF = float("inf")
        cost = [INF] * n
        cost[src] = 0
        for _ in range(k + 1):          # k stops = at most k + 1 flights
            nxt = cost[:]               # relax from the OLD round only
            for u, v, price in flights:
                if cost[u] + price < nxt[v]:
                    nxt[v] = cost[u] + price
            cost = nxt
        return -1 if cost[dst] == INF else cost[dst]
