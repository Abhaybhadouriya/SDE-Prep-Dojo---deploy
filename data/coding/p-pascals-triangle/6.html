class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> tri;
        for (int i = 0; i < numRows; ++i) {
            vector<int> row(i + 1, 1);                     // both ends stay 1
            for (int j = 1; j < i; ++j)
                row[j] = tri[i - 1][j - 1] + tri[i - 1][j];  // up-left + up
            tri.push_back(row);
        }
        return tri;
    }
};
func generate(numRows int) [][]int {
	tri := make([][]int, numRows)
	for i := range tri {
		row := make([]int, i+1)
		row[0], row[i] = 1, 1 // both ends are 1
		for j := 1; j < i; j++ {
			row[j] = tri[i-1][j-1] + tri[i-1][j] // up-left + up
		}
		tri[i] = row
	}
	return tri
}

static class Solution {
    public List<List<Integer>> generate(int numRows) {
        List<List<Integer>> tri = new ArrayList<>();
        for (int i = 0; i < numRows; i++) {
            List<Integer> row = new ArrayList<>(Collections.nCopies(i + 1, 1));   // ends stay 1
            List<Integer> up = i > 0 ? tri.get(i - 1) : null;
            for (int j = 1; j < i; j++)
                row.set(j, up.get(j - 1) + up.get(j));                          // up-left + up
            tri.add(row);
        }
        return tri;
    }
}
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        tri = []
        for i in range(numRows):
