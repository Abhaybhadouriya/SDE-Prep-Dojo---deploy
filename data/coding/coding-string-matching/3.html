KMP search: the same loop against the text

Searching is the prefix-function loop run over the text instead of the pattern. k now means "how many pattern characters are matched right now". On a full match (k == m), record i - m + 1 and fall back to k = pi[m-1], so overlapping matches are found too. The text index i only moves forward.

  1. Find every abab in abaababab. pi of the pattern is [0, 0, 1, 2]. Start with k = 0 at i = 0.
  2. i = 0, 1, 2: a, b, a match, so k = 3.
  3. i = 3: the text has a, the pattern wants b. Do not move i. The matched part aba has border length pi[2] = 1, so continue as if 1 character were matched.
  4. The pattern is now aligned at 2 (its a sits on text[2], kept for free). pat[1] = 'b' still differs from text[3] = 'a', so fall back again: k = pi[0] = 0.
  5. With k = 0, pat[0] = 'a' matches text[3]. Then i = 4, 5, 6 match b a b: k = 4 = m, a match starts at 6 - 4 + 1 = 3. Set k = pi[3] = 2 to look for overlaps.
  6. The border ab is already in place at 5-6. i = 7, 8 match a b, k = 4 again: a second match at 5. Total: 9 text reads and 3 fall-backs, O(n + m).

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

// prefixFunction: pi[i] = longest proper prefix of s[0..i] that is also a suffix of it.
func prefixFunction(s string) []int {
	pi := make([]int, len(s))
	for i := 1; i < len(s); i++ {
		k := pi[i-1]
		for k > 0 && s[i] != s[k] {
			k = pi[k-1]
		}
		if s[i] == s[k] {
			k++
		}
		pi[i] = k
	}
	return pi
}

// kmpSearch returns all start indices of pat in text. O(n + m).
func kmpSearch(text, pat string) []int {
	res := []int{}
	m := len(pat)
	if m == 0 {
		return res
	}
	pi := prefixFunction(pat)
	k := 0
	for i := 0; i < len(text); i++ {
		for k > 0 && text[i] != pat[k] {
			k = pi[k-1]
		}
		if text[i] == pat[k] {
			k++
		}
		if k == m {
			res = append(res, i-m+1)
			k = pi[k-1]
		}
	}
	return res
}
// pi[i] = length of the longest proper prefix of s[0..i] that is also a suffix of it.
vector<int> prefixFunction(const string& s) {
    int n = s.size();
    vector<int> pi(n, 0);
    for (int i = 1; i < n; i++) {
        int k = pi[i - 1];                        // best border of s[0..i-1]
        while (k > 0 && s[i] != s[k]) k = pi[k - 1]; // fall back to a shorter border
        if (s[i] == s[k]) k++;                    // extend it by s[i]
        pi[i] = k;
    }
    return pi;
}

// All start indices of pat in text. O(n + m). The text index never moves back.
vector<int> kmpSearch(const string& text, const string& pat) {
    vector<int> res;
    int m = pat.size();
    if (m == 0) return res;
    vector<int> pi = prefixFunction(pat);
    int k = 0;                                    // pattern chars matched so far
    for (int i = 0; i < (int)text.size(); i++) {
        while (k > 0 && text[i] != pat[k]) k = pi[k - 1];
        if (text[i] == pat[k]) k++;
        if (k == m) {                             // full match ends at i
            res.push_back(i - m + 1);
            k = pi[k - 1];                        // keep going for overlaps
        }
    }
    return res;
}
// pi[i] = length of the longest proper prefix of s[0..i] that is also a suffix of it.
static int[] prefixFunction(String s) {
    int n = s.length();
    int[] pi = new int[n];
    for (int i = 1; i < n; i++) {
        int k = pi[i - 1];
        while (k > 0 && s.charAt(i) != s.charAt(k)) k = pi[k - 1];
        if (s.charAt(i) == s.charAt(k)) k++;
        pi[i] = k;
    }
    return pi;
}

def prefix_function(s: str) -> list[int]:
    """pi[i] = length of the longest proper prefix of s[:i+1] that is also its suffix."""
    pi = [0] * len(s)
    for i in range(1, len(s)):
        k = pi[i - 1]
        while k > 0 and s[i] != s[k]:
            k = pi[k - 1]
        if s[i] == s[k]:
            k += 1
        pi[i] = k
    return pi


def kmp_search(text: str, pat: str) -> list[int]:
    """All start indices of pat in text. O(n + m)."""
    m = len(pat)
    if m == 0:
        return []
    pi, res, k = prefix_function(pat), [], 0
    for i, ch in enumerate(text):
        while k > 0 and ch != pat[k]:
            k = pi[k - 1]
        if ch == pat[k]:
            k += 1
        if k == m:
            res.append(i - m + 1)
            k = pi[k - 1]
    return res

The builder and the search share the same three lines: fall back while mismatch, extend on match, act. Learn them once. Some people compute pi of pat + "#" + text and look for values equal to m; that works if # cannot occur in either string, but the separate loop above needs no separator and only O(m) memory, which matters for a long text or a stream.
