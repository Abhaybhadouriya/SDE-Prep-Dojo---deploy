The Z-function: the same power, a different shape

z[i] = the length of the longest common prefix of s and s[i..]: "how far does the string, read from i, agree with its own start?". By convention z[0] = n. To find a pattern, compute z of pat + text; text position j matches when z[m + j] >= m. Because the test is "at least m", no separator is needed.

The algorithm keeps the Z-box [l, r): the match window that reaches furthest right. For i inside the box, s[i..r) is a copy of s[i-l .. r-l), whose answer z[i-l] is already known, so z[i] starts at min(r - i, z[i - l]). Only characters beyond r are compared, and each comparison pushes r right, so the total is O(n).

// zFunction: z[i] = longest common prefix of s and s[i:]; z[0] = n.
func zFunction(s string) []int {
	n := len(s)
	z := make([]int, n)
	if n > 0 {
		z[0] = n
	}
	for i, l, r := 1, 0, 0; i < n; i++ {
		if i < r {
			z[i] = min(r-i, z[i-l])
		}
		for i+z[i] < n && s[z[i]] == s[i+z[i]] {
			z[i]++
		}
		if i+z[i] > r {
			l, r = i, i+z[i]
		}
	}
	return z
}

func zSearch(text, pat string) []int {
	res := []int{}
	m := len(pat)
	if m == 0 {
		return res
	}
	z := zFunction(pat + text)
	for j := 0; j+m <= len(text); j++ {
		if z[m+j] >= m {
			res = append(res, j)
		}
	}
	return res
}
// z[i] = length of the longest common prefix of s and s[i..]; z[0] = n by convention.
vector<int> zFunction(const string& s) {
    int n = s.size();
    vector<int> z(n, 0);
    if (n) z[0] = n;
    for (int i = 1, l = 0, r = 0; i < n; i++) {   // [l, r) = rightmost match window
        if (i < r) z[i] = min(r - i, z[i - l]);   // reuse what the window already knows
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
        if (i + z[i] > r) { l = i; r = i + z[i]; }
    }
    return z;
}

// Z search: in pat + text, position m + j matches when z >= m (no separator needed).
vector<int> zSearch(const string& text, const string& pat) {
    vector<int> res;
    int m = pat.size();
    if (m == 0) return res;
    vector<int> z = zFunction(pat + text);
    for (int j = 0; j + m <= (int)text.size(); j++)
        if (z[m + j] >= m) res.push_back(j);
    return res;
}
// z[i] = longest common prefix of s and s[i..]; z[0] = n.
static int[] zFunction(String s) {
    int n = s.length();
    int[] z = new int[n];
    if (n > 0) z[0] = n;
    for (int i = 1, l = 0, r = 0; i < n; i++) {
        if (i < r) z[i] = Math.min(r - i, z[i - l]);
        while (i + z[i] < n && s.charAt(z[i]) == s.charAt(i + z[i])) z[i]++;
        if (i + z[i] > r) { l = i; r = i + z[i]; }
    }
    return z;
}

static List<Integer> zSearch(String text, String pat) {
    List<Integer> res = new ArrayList<>();
    int m = pat.length();
    if (m == 0) return res;
    int[] z = zFunction(pat + text);
    for (int j = 0; j + m <= text.length(); j++) if (z[m + j] >= m) res.add(j);
    return res;
}
def z_function(s: str) -> list[int]:
    """z[i] = longest common prefix of s and s[i:]; z[0] = n."""
    n = len(s)
    z = [0] * n
    if n:
        z[0] = n
    l = r = 0
    for i in range(1, n):
        if i < r:
            z[i] = min(r - i, z[i - l])
        while i + z[i] < n and s[z[i]] == s[i + z[i]]:
            z[i] += 1
        if i + z[i] > r:
            l, r = i, i + z[i]
    return z


def z_search(text: str, pat: str) -> list[int]:
    m = len(pat)
    if m == 0:
        return []
    z = z_function(pat + text)
    return [j for j in range(len(text) - m + 1) if z[m + j] >= m]

When to choose Z over KMP: the question is literally about "prefix of s at position i". Sum of scores of built strings (LC 2223) is the sum of the Z-array. Minimum time to revert word to initial state II (LC 3031) asks for the first multiple t of k with z[t] = n - t. Both are one Z pass. For plain search, use whichever one you can write without notes; their power is the same.
