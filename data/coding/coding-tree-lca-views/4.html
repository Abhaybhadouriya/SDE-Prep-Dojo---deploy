Views: right, left, top, vertical and boundary

Problem (restated from LeetCode 199, Binary Tree Right Side View). You stand to the right of a binary tree and look at it. Return the values you can see, from top to bottom: the rightmost node of each level. For the tree below the answer is [50, 70, 80]; for an empty tree it is []. Limits: 0 to 100 nodes, values in [-100, 100].

A view is a question about coordinates. Put the root at (row 0, col 0); a left child is (row + 1, col - 1) and a right child (row + 1, col + 1). Then:

  • Right view = the last node in each row. BFS by levels and take the last one popped. DFS alternative: visit right before left and record the first node you meet at each new depth.
  • Left view = the first node in each row (same code, mirrored).
  • Top view = the shallowest node in each column; bottom view = the deepest one. BFS meets shallow nodes first, so keep the first value per column for the top view and overwrite for the bottom view. (These two are GfG classics rather than LeetCode problems.)
  • Vertical order = all nodes grouped by column, left to right.

Problem (restated from LeetCode 987, Vertical Order Traversal of a Binary Tree). Using the coordinates above, return one list per column, from the leftmost column to the rightmost. Inside a column, list nodes from top to bottom; when two nodes share both row and column, put the smaller value first. For the tree above: [[20], [30], [50, 52, 58], [70], [80]]. Limits: 1 to 1,000 nodes, values in [0, 1000].

Its twin, LeetCode 314, Binary Tree Vertical Order Traversal (Premium, a Meta favourite): same columns, but nodes that share a row and column stay in left-to-right order, as BFS meets them. Answer above: [[20], [30], [50, 58, 52], [70], [80]]. Limits: 0 to 100 nodes, values in [-100, 100].

For 987 the simplest correct code collects (col, row, value) for every node, sorts the triples and groups by column. O(n log n). For 314, BFS with a (node, col) queue appends each value to byCol[col]; BFS order already is "top to bottom, then left to right", and tracking the min and max column avoids sorting the keys: O(n). The classic bug is solving 314 with DFS: DFS meets a deep left node before a shallow right one and the order inside a column comes out wrong.

Template: right view, left view (DFS), vertical order and top view

// BFS by levels: the last node of each level is what you see from the right.
func rightSideView(root *TreeNode) []int {
	out := []int{}
	level := []*TreeNode{}
	if root != nil {
		level = append(level, root)
	}
	for len(level) > 0 {
		out = append(out, level[len(level)-1].Val) // rightmost node of this level
		var next []*TreeNode
		for _, n := range level {
			if n.Left != nil {
				next = append(next, n.Left)
			}
			if n.Right != nil {
				next = append(next, n.Right)
			}
		}
		level = next
	}
	return out
}

// DFS version: left child first; the first node met at each depth is visible from the left.
func leftViewDfs(node *TreeNode, d int, out *[]int) {
	if node == nil {
		return
	}
	if d == len(*out) {
		*out = append(*out, node.Val)
	}
	leftViewDfs(node.Left, d+1, out)
	leftViewDfs(node.Right, d+1, out)
}

// LeetCode 987: column = x, row = depth. Sort by (col, row, value); group by col.
type cell struct{ col, row, val int }

func collect(n *TreeNode, row, col int, cells *[]cell) {
	if n == nil {
		return
	}
	*cells = append(*cells, cell{col, row, n.Val})
	collect(n.Left, row+1, col-1, cells)
	collect(n.Right, row+1, col+1, cells)
}

func verticalTraversal(root *TreeNode) [][]int {
	var cells []cell
	collect(root, 0, 0, &cells)
	sort.Slice(cells, func(i, j int) bool {
		a, b := cells[i], cells[j]
		if a.col != b.col {
			return a.col < b.col
		}
		if a.row != b.row {
			return a.row < b.row
		}
		return a.val < b.val
	})
	out := [][]int{}
	for i, c := range cells {
		if i == 0 || c.col != cells[i-1].col {
			out = append(out, []int{})
		}
		out[len(out)-1] = append(out[len(out)-1], c.val)
	}
	return out
}

// Top view: BFS meets the shallowest node of each column first; keep the first.
// (Bottom view: overwrite instead of keeping the first.)
func topView(root *TreeNode) []int {
	type item struct {
		n   *TreeNode
		col int
	}
	first := map[int]int{}
	minC, maxC := 0, -1
	q := []item{}
	if root != nil {
		q = append(q, item{root, 0})
		maxC = 0
	}
	for len(q) > 0 {
		it := q[0]
		q = q[1:]
		if _, ok := first[it.col]; !ok {
			first[it.col] = it.n.Val
			if it.col < minC {
				minC = it.col
			}
			if it.col > maxC {
				maxC = it.col
			}
		}
		if it.n.Left != nil {
			q = append(q, item{it.n.Left, it.col - 1})
		}
		if it.n.Right != nil {
			q = append(q, item{it.n.Right, it.col + 1})
		}
	}
	out := []int{}
	for c := minC; c <= maxC; c++ { // columns are contiguous
		out = append(out, first[c])
	}
	return out
}
// BFS by levels: the last node of each level is what you see from the right.
vector<int> rightSideView(TreeNode* root) {
    vector<int> out;
    queue<TreeNode*> q;
    if (root) q.push(root);
    while (!q.empty()) {
        int n = q.size();                        // freeze the level size
        for (int i = 0; i < n; i++) {
            TreeNode* node = q.front(); q.pop();
            if (i == n - 1) out.push_back(node->val);
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
    }
    return out;
}

// DFS version: left child first; the first node met at each depth is visible from the left.
void leftViewDfs(TreeNode* node, int d, vector<int>& out) {
    if (!node) return;
    if (d == (int)out.size()) out.push_back(node->val);
    leftViewDfs(node->left, d + 1, out);
    leftViewDfs(node->right, d + 1, out);
}

// LeetCode 987: column = x, row = depth. Sort by (col, row, value); group by col.
void collect(TreeNode* n, int row, int col, vector<tuple<int, int, int>>& cells) {
    if (!n) return;
    cells.emplace_back(col, row, n->val);
    collect(n->left, row + 1, col - 1, cells);
    collect(n->right, row + 1, col + 1, cells);
}
vector<vector<int>> verticalTraversal(TreeNode* root) {
    vector<tuple<int, int, int>> cells;
    collect(root, 0, 0, cells);
    sort(cells.begin(), cells.end());
    vector<vector<int>> out;
    int prev = INT_MIN;
    for (auto& c : cells) {
        if (get<0>(c) != prev) { out.emplace_back(); prev = get<0>(c); }
        out.back().push_back(get<2>(c));
    }
    return out;
}

// Top view: BFS meets the shallowest node of each column first; keep the first.
// (Bottom view: overwrite instead of keeping the first.)
vector<int> topView(TreeNode* root) {
    map<int, int> first;                         // ordered by column
    queue<pair<TreeNode*, int>> q;
    if (root) q.push({root, 0});
    while (!q.empty()) {
        TreeNode* n = q.front().first; int col = q.front().second; q.pop();
        first.insert({col, n->val});             // insert does nothing if col is already there
        if (n->left) q.push({n->left, col - 1});
        if (n->right) q.push({n->right, col + 1});
    }
    vector<int> out;
    for (auto& kv : first) out.push_back(kv.second);
    return out;
}
// BFS by levels: the last node of each level is what you see from the right.
static List<Integer> rightSideView(TreeNode root) {
    List<Integer> out = new ArrayList<>();
    ArrayDeque<TreeNode> q = new ArrayDeque<>();
    if (root != null) q.add(root);
    while (!q.isEmpty()) {
        int n = q.size();                    // freeze the level size
        for (int i = 0; i < n; i++) {
            TreeNode node = q.poll();
            if (i == n - 1) out.add(node.val);
            if (node.left != null) q.add(node.left);
            if (node.right != null) q.add(node.right);
        }
    }
    return out;
}

// DFS version: left child first; the first node met at each depth is visible from the left.
static void leftViewDfs(TreeNode node, int d, List<Integer> out) {
    if (node == null) return;
    if (d == out.size()) out.add(node.val);
    leftViewDfs(node.left, d + 1, out);
    leftViewDfs(node.right, d + 1, out);
}

// LeetCode 987: column = x, row = depth. Sort by (col, row, value); group by col.
static void collect(TreeNode n, int row, int col, List<int[]> cells) {
    if (n == null) return;
    cells.add(new int[]{col, row, n.val});
    collect(n.left, row + 1, col - 1, cells);
    collect(n.right, row + 1, col + 1, cells);
}
static List<List<Integer>> verticalTraversal(TreeNode root) {
    List<int[]> cells = new ArrayList<>();
    collect(root, 0, 0, cells);
    cells.sort((a, b) -> a[0] != b[0] ? Integer.compare(a[0], b[0])
                       : a[1] != b[1] ? Integer.compare(a[1], b[1])
                       : Integer.compare(a[2], b[2]));
    List<List<Integer>> out = new ArrayList<>();
    for (int i = 0; i < cells.size(); i++) {
        if (i == 0 || cells.get(i)[0] != cells.get(i - 1)[0]) out.add(new ArrayList<>());
        out.get(out.size() - 1).add(cells.get(i)[2]);
    }
    return out;
}

// Top view: BFS meets the shallowest node of each column first; keep the first.
// (Bottom view: overwrite instead of keeping the first.)
static List<Integer> topView(TreeNode root) {
    TreeMap<Integer, Integer> first = new TreeMap<>();   // ordered by column
    ArrayDeque<TreeNode> q = new ArrayDeque<>();
    ArrayDeque<Integer> cols = new ArrayDeque<>();
    if (root != null) { q.add(root); cols.add(0); }
    while (!q.isEmpty()) {
        TreeNode n = q.poll();
        int col = cols.poll();
        first.putIfAbsent(col, n.val);
        if (n.left != null) { q.add(n.left); cols.add(col - 1); }
        if (n.right != null) { q.add(n.right); cols.add(col + 1); }
    }
    return new ArrayList<>(first.values());
}
def right_side_view(root):
    """BFS by levels: the last node of each level is what you see from the right."""
    out = []
    q = deque([root] if root else [])
    while q:
        for _ in range(len(q)):          # freeze the level size
            node = q.popleft()
            last = node                  # ends as the rightmost node of this level
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        out.append(last.val)
    return out


def left_view_dfs(root):
    """DFS version: visit left child first; the first node met at each depth is visible from the left."""
    out = []

    def dfs(node, d):
        if not node:
            return
        if d == len(out):
            out.append(node.val)         # first time at this depth
        dfs(node.left, d + 1)
        dfs(node.right, d + 1)
    dfs(root, 0)
    return out

def vertical_traversal(root):
    """LeetCode 987: column = x, row = depth. Sort by (col, row, value); group by col."""
    cells = []

    def dfs(node, row, col):
        if node:
            cells.append((col, row, node.val))
            dfs(node.left, row + 1, col - 1)
            dfs(node.right, row + 1, col + 1)
    dfs(root, 0, 0)
    cells.sort()
    out, prev = [], None
    for col, _, val in cells:
        if col != prev:
            out.append([])
            prev = col
        out[-1].append(val)
    return out


def top_view(root):
    """Top view: BFS (so the shallowest node per column is met first), keep the first per column.
    Bottom view: same loop, but overwrite instead of keeping the first."""
    first = {}
    q = deque([(root, 0)] if root else [])
    while q:
        node, col = q.popleft()
        if col not in first:
            first[col] = node.val
        if node.left:
            q.append((node.left, col - 1))
        if node.right:
            q.append((node.right, col + 1))
    return [first[c] for c in sorted(first)]

Boundary of a binary tree

Problem (restated from LeetCode 545, Boundary of Binary Tree, Premium). Return the boundary of a binary tree, anticlockwise from the root: the root, then the left boundary, then all leaves from left to right, then the right boundary in reverse (bottom-up) order. The left boundary starts at the root's left child and keeps going to the left child, or to the right child if there is no left one, until it reaches a leaf; that leaf is not part of the left boundary. If the root has no left child the left boundary is empty. The right boundary is the mirror image. Leaves are the nodes with no children, not counting the root. For the tree below: [40, 20, 10, 25, 35, 45, 55, 50, 60]. Limits: 1 to 104 nodes, values in [-1000, 1000].

Three passes, each with one simple rule, are far easier to get right than one clever pass: walk the left edge adding non-leaves, run a preorder that adds only leaves, walk the right edge collecting non-leaves and reverse them. The traps are exactly the ones in the figure: a leaf on the edge path (count it once, as a leaf), an edge that has to turn (no left child, so take the right one), and a root with only one child (the root is not a leaf, and the missing side's edge is empty).

// Anticlockwise boundary: root, left edge (no leaves), leaves left to right, right edge reversed.
func isLeaf(n *TreeNode) bool { return n.Left == nil && n.Right == nil }

func addLeaves(n *TreeNode, out *[]int) {
	if n == nil {
		return
	}
	if isLeaf(n) {
		*out = append(*out, n.Val)
		return
	}
	addLeaves(n.Left, out)
	addLeaves(n.Right, out)
}

func boundaryOfBinaryTree(root *TreeNode) []int {
	if root == nil {
		return []int{}
	}
	out := []int{root.Val}
	if isLeaf(root) {
		return out
	}
	for n := root.Left; n != nil; { // left edge: prefer left, else right
		if !isLeaf(n) {
			out = append(out, n.Val)
		}
		if n.Left != nil {
			n = n.Left
		} else {
			n = n.Right
		}
	}
	addLeaves(root.Left, &out)
	addLeaves(root.Right, &out)
	var right []int
	for n := root.Right; n != nil; { // right edge: prefer right, else left
		if !isLeaf(n) {
			right = append(right, n.Val)
		}
		if n.Right != nil {
			n = n.Right
		} else {
			n = n.Left
		}
	}
	for i := len(right) - 1; i >= 0; i-- {
		out = append(out, right[i])
	}
	return out
}
// Anticlockwise boundary: root, left edge (no leaves), leaves left to right, right edge reversed.
bool isLeaf(TreeNode* n) { return !n->left && !n->right; }
void addLeaves(TreeNode* n, vector<int>& out) {
    if (!n) return;
    if (isLeaf(n)) { out.push_back(n->val); return; }
    addLeaves(n->left, out);
    addLeaves(n->right, out);
}
vector<int> boundaryOfBinaryTree(TreeNode* root) {
    if (!root) return {};
    vector<int> out{root->val};
    if (isLeaf(root)) return out;
    for (TreeNode* n = root->left; n; n = n->left ? n->left : n->right)    // left edge
        if (!isLeaf(n)) out.push_back(n->val);
    addLeaves(root->left, out);
    addLeaves(root->right, out);
    vector<int> right;
    for (TreeNode* n = root->right; n; n = n->right ? n->right : n->left)  // right edge
        if (!isLeaf(n)) right.push_back(n->val);
    out.insert(out.end(), right.rbegin(), right.rend());
    return out;
}
// Anticlockwise boundary: root, left edge (no leaves), leaves left to right, right edge reversed.
static boolean isLeaf(TreeNode n) { return n.left == null && n.right == null; }
static void addLeaves(TreeNode n, List<Integer> out) {
    if (n == null) return;
    if (isLeaf(n)) { out.add(n.val); return; }
    addLeaves(n.left, out);
    addLeaves(n.right, out);
}
static List<Integer> boundaryOfBinaryTree(TreeNode root) {
    List<Integer> out = new ArrayList<>();
    if (root == null) return out;
    out.add(root.val);
    if (isLeaf(root)) return out;
    for (TreeNode n = root.left; n != null; n = n.left != null ? n.left : n.right)    // left edge
        if (!isLeaf(n)) out.add(n.val);
    addLeaves(root.left, out);
    addLeaves(root.right, out);
    List<Integer> right = new ArrayList<>();
    for (TreeNode n = root.right; n != null; n = n.right != null ? n.right : n.left)  // right edge
        if (!isLeaf(n)) right.add(n.val);
    Collections.reverse(right);
    out.addAll(right);
    return out;
}
def boundary(root):
    """Anticlockwise boundary: root, left edge (no leaves) top-down, all leaves left to right,
    right edge (no leaves) bottom-up."""
    if not root:
        return []
    is_leaf = lambda n: not n.left and not n.right
    out = [root.val]
    if is_leaf(root):
        return out
    node = root.left                       # left edge: prefer left, else right
    while node:
        if not is_leaf(node):
            out.append(node.val)
        node = node.left if node.left else node.right

    def leaves(n):
        if n:
            if is_leaf(n):
                out.append(n.val)
            leaves(n.left)
            leaves(n.right)
    leaves(root.left)
    leaves(root.right)
    stack, node = [], root.right           # right edge: prefer right, else left; print reversed
    while node:
        if not is_leaf(node):
            stack.append(node.val)
        node = node.right if node.right else node.left
    out.extend(reversed(stack))
    return out
