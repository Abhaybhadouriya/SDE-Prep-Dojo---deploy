func findDuplicate(nums []int) int {
	// Read the array as a linked list: index i points to index nums[i].
	// Index 0 is never a target (values are 1..n), so the walk from 0 has a tail and then a cycle.
	// The cycle's entry has two arrows into it, i.e. two indices hold the same value: the duplicate.
	slow, fast := 0, 0
	for {
		slow = nums[slow]       // 1 step
		fast = nums[nums[fast]] // 2 steps
		if slow == fast {       // they meet somewhere inside the cycle
			break
		}
	}

	finder := 0 // tail length == distance from the meeting point to the entry (mod cycle)
	for finder != slow {
		finder = nums[finder]
		slow = nums[slow]
	}
	return finder // the entry of the cycle
}
