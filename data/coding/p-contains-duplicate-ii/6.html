class Solution {
public:
    // window holds the values of the k indices before i (all distinct).
    bool containsNearbyDuplicate(vector<int>& nums, int k) {
        unordered_set<int> window;
        for (int i = 0; i < (int)nums.size(); i++) {
            if (!window.insert(nums[i]).second) return true;  // equal value at most k back
            if (i >= k) window.erase(nums[i - k]);            // too far from index i+1
        }
        return false;
    }

    // value -> latest index; the newest copy is always the best partner.
    bool containsNearbyDuplicateLastSeen(vector<int>& nums, int k) {
        unordered_map<int, int> last;
        for (int i = 0; i < (int)nums.size(); i++) {
            auto it = last.find(nums[i]);
            if (it != last.end() && i - it->second <= k) return true;
            last[nums[i]] = i;                                // an older index is only farther
        }
        return false;
    }
};
// containsNearbyDuplicate: window holds the values of the k indices before i.
func containsNearbyDuplicate(nums []int, k int) bool {
	window := make(map[int]struct{}) // values of nums[i-k .. i-1], all distinct
	for i, x := range nums {
		if _, seen := window[x]; seen {
			return true // equal value at most k indices back
		}
		window[x] = struct{}{}
		if i >= k {
			delete(window, nums[i-k]) // nums[i-k] is too far from index i+1
		}
	}
	return false
}

// containsNearbyDuplicateLastSeen: value -> latest index; the newest copy is the best partner.
func containsNearbyDuplicateLastSeen(nums []int, k int) bool {
	last := make(map[int]int)
	for i, x := range nums {
		if j, ok := last[x]; ok && i-j <= k {
			return true
		}
		last[x] = i // overwrite: an older index can only be farther away
	}
	return false
}

static class Solution {
    // window holds the values of the k indices before i (all distinct).
    public boolean containsNearbyDuplicate(int[] nums, int k) {
        Set<Integer> window = new HashSet<>();
        for (int i = 0; i < nums.length; i++) {
            if (!window.add(nums[i])) return true;   // equal value at most k back
            if (i >= k) window.remove(nums[i - k]);  // too far from index i+1
        }
        return false;
    }

    // value -> latest index; the newest copy is always the best partner.
    public boolean containsNearbyDuplicateLastSeen(int[] nums, int k) {
        Map<Integer, Integer> last = new HashMap<>();
        for (int i = 0; i < nums.length; i++) {
            Integer j = last.put(nums[i], i);        // returns the previous index, if any
            if (j != null && i - j <= k) return true;
        }
        return false;
    }
}
class Solution:
    def containsNearbyDuplicate(self, nums: List[int], k: int) -> bool:
        window = set()                        # values of the k indices before i, all distinct
        for i, x in enumerate(nums):
            if x in window:
                return True                   # equal value at most k indices back
            window.add(x)
            if i >= k:
                window.remove(nums[i - k])    # too far from index i+1
        return False

    def containsNearbyDuplicateLastSeen(self, nums: List[int], k: int) -> bool:
        last = {}                             # value -> latest index
        for i, x in enumerate(nums):
            if x in last and i - last[x] <= k:
                return True
            last[x] = i                       # an older index can only be farther away
        return False
