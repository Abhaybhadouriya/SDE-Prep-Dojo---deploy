func eraseOverlapIntervals(intervals [][]int) int {
	// Keep the interval that finishes first: it leaves the most room for the rest.
	sort.Slice(intervals, func(a, b int) bool { return intervals[a][1] < intervals[b][1] })
	kept := 0
	lastEnd := math.MinInt32 // end of the last interval we kept
	for _, iv := range intervals {
		if iv[0] >= lastEnd { // fits after the last kept one (touching is fine)
			kept++
			lastEnd = iv[1]
		} // else: overlaps, drop it (it ends later)
	}
	return len(intervals) - kept // removed = total - max kept
}

class Solution:
    def eraseOverlapIntervals(self, intervals: List[List[int]]) -> int:
        # Keep the interval that finishes first: it leaves the most room for the rest.
        intervals.sort(key=lambda iv: iv[1])
        kept = 0
        last_end = float("-inf")              # end of the last interval we kept
        for start, end in intervals:
            if start >= last_end:             # fits after the last kept one (touching is fine)
                kept += 1
                last_end = end
            # else: overlaps, drop it (it ends later)
        return len(intervals) - kept          # removed = total - max kept
