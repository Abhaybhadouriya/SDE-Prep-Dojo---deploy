class Solution {
public:
    vector<int> findSubstring(string s, vector<string>& words) {
        int n = s.size(), k = words.size(), w = words[0].size();
        vector<int> res;
        if ((long long)k * w > n) return res;
        unordered_map<string, int> need;         // word -> copies the concatenation must contain
        for (auto& wd : words) need[wd]++;
        for (int offset = 0; offset < w; offset++) {  // windows starting at offset, offset+w, ...
            unordered_map<string, int> have;     // word counts inside the window s[left .. right)
            int left = offset, count = 0;        // count = words in the window
            for (int right = offset; right + w <= n; right += w) {
                string word = s.substr(right, w);
                auto it = need.find(word);
                if (it == need.end()) {          // a stranger: no window can cross it
                    have.clear();
                    count = 0;
                    left = right + w;
                    continue;
                }
                have[word]++;
                count++;
                while (have[word] > it->second) {  // one copy too many: drop words from the left
                    have[s.substr(left, w)]--;
                    count--;
                    left += w;
                }
                if (count == k) {                // exactly the multiset of words
                    res.push_back(left);
                    have[s.substr(left, w)]--;   // slide on by one word
                    count--;
                    left += w;
                }
            }
        }
        sort(res.begin(), res.end());            // any order is accepted; sorted is stable
        return res;
    }
};
func findSubstring(s string, words []string) []int {
	n, k, w := len(s), len(words), len(words[0])
	res := []int{}
	if k*w > n {
		return res
	}
	need := map[string]int{} // word -> how many times the concatenation must contain it
	for _, wd := range words {
		need[wd]++
	}
	for offset := 0; offset < w; offset++ { // windows that start at offset, offset+w, offset+2w, ...
		have := map[string]int{} // word counts inside the window s[left : right]
		left, count := offset, 0 // count = words in the window
		for right := offset; right+w <= n; right += w {
			word := s[right : right+w]
			if _, ok := need[word]; !ok { // a stranger: no window can cross it
				have = map[string]int{}
				count, left = 0, right+w
				continue
			}
			have[word]++
			count++
			for have[word] > need[word] { // one copy too many: drop words from the left
				have[s[left:left+w]]--
				count--
				left += w
			}
			if count == k { // exactly the multiset of words
				res = append(res, left)
				have[s[left:left+w]]-- // slide on by one word
				count--
				left += w
			}
		}
	}
	sort.Ints(res) // any order is accepted; sorted makes the output stable
	return res
}

static class Solution {
    public List<Integer> findSubstring(String s, String[] words) {

        List<Integer> res = new ArrayList<>();
        if ((long) k * w > n) return res;
        Map<String, Integer> need = new HashMap<>();  // word -> copies the concatenation must contain
        for (String wd : words) need.merge(wd, 1, Integer::sum);
        for (int offset = 0; offset < w; offset++) {  // windows starting at offset, offset+w, ...
            Map<String, Integer> have = new HashMap<>();  // word counts inside s[left .. right)
            int left = offset, count = 0;             // count = words in the window
            for (int right = offset; right + w <= n; right += w) {
                String word = s.substring(right, right + w);
                Integer want = need.get(word);
                if (want == null) {                   // a stranger: no window can cross it
                    have.clear();
                    count = 0;
                    left = right + w;
                    continue;
                }
                have.merge(word, 1, Integer::sum);
                count++;
                while (have.get(word) > want) {       // one copy too many: drop words from the left
                    have.merge(s.substring(left, left + w), -1, Integer::sum);
                    count--;
                    left += w;
                }
                if (count == k) {                     // exactly the multiset of words
                    res.add(left);
                    have.merge(s.substring(left, left + w), -1, Integer::sum);  // slide on by one word
                    count--;
                    left += w;
                }
            }
        }
        Collections.sort(res);                        // any order is accepted; sorted is stable
        return res;
    }
}
class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        n, k, w = len(s), len(words), len(words[0])
        res: List[int] = []
        if k * w > n:
            return res
        need = Counter(words)                 # word -> copies the concatenation must contain
        for offset in range(w):               # windows starting at offset, offset + w, ...
            have = defaultdict(int)           # word counts inside the window s[left : right]
            left, count = offset, 0           # count = words in the window
            for right in range(offset, n - w + 1, w):
                word = s[right:right + w]
                if word not in need:          # a stranger: no window can cross it
                    have.clear()
                    count, left = 0, right + w
                    continue
                have[word] += 1
                count += 1
                while have[word] > need[word]:   # one copy too many: drop words from the left
                    have[s[left:left + w]] -= 1
                    count -= 1
                    left += w
                if count == k:                # exactly the multiset of words
                    res.append(left)
                    have[s[left:left + w]] -= 1  # slide on by one word
                    count -= 1
                    left += w
        res.sort()                            # any order is accepted; sorted is stable
        return res
