class Solution {
public:
    bool validPalindrome(const string& s) {
        int lo = 0, hi = (int)s.size() - 1;
        while (lo < hi) {
            if (s[lo] != s[hi])
                // the outer part matched; one of these two chars must be the one we delete
                return isPal(s, lo + 1, hi) || isPal(s, lo, hi - 1);
            lo++;
            hi--;
        }
        return true;                              // already a palindrome: delete nothing
    }

private:
    // plain check of s[lo..hi], no more deletions allowed
    static bool isPal(const string& s, int lo, int hi) {
        while (lo < hi)
            if (s[lo++] != s[hi--]) return false;
        return true;
    }
};

class Solution:
    def validPalindrome(self, s: str) -> bool:
        def is_pal(lo, hi):                  # plain check of s[lo..hi], no more deletions
            while lo < hi:
                if s[lo] != s[hi]:
                    return False
                lo += 1
                hi -= 1
            return True

        lo, hi = 0, len(s) - 1
        while lo < hi:
            if s[lo] != s[hi]:
                # the outer part matched; one of these two chars must be the one we delete
                return is_pal(lo + 1, hi) or is_pal(lo, hi - 1)
            lo += 1
            hi -= 1
        return True                          # already a palindrome: delete nothing
