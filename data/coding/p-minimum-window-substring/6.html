class Solution {
public:
    string minWindow(string s, string t) {
        vector<int> need(128, 0);   // need[c] > 0: window still lacks that many copies of c
        for (char c : t) need[c]++;
        int missing = t.size();     // characters of t (with duplicates) not yet covered
        int bestLen = s.size() + 1, bestStart = 0;
        for (int left = 0, right = 0; right < (int)s.size(); right++) {
            if (need[s[right]] > 0) missing--;   // this copy was still needed
            need[s[right]]--;                    // extra copies go below zero (surplus)
            while (missing == 0) {               // window covers t: record, then shrink
                if (right - left + 1 < bestLen) { bestLen = right - left + 1; bestStart = left; }
                if (++need[s[left]] > 0) missing++;  // dropped a copy we needed
                left++;
            }
        }
        return bestLen > (int)s.size() ? "" : s.substr(bestStart, bestLen);
    }
};
func minWindow(s string, t string) string {
	var need [128]int // need[c] > 0: window still lacks that many copies of c
	for i := 0; i < len(t); i++ {
		need[t[i]]++
	}
	missing := len(t) // characters of t (with duplicates) not yet covered
	bestLen, bestStart := len(s)+1, 0
	left := 0
	for right := 0; right < len(s); right++ {
		c := s[right]
		if need[c] > 0 { // this copy was still needed
			missing--
		}
		need[c]-- // extra copies go below zero (surplus)
		for missing == 0 { // window covers t: record it, then shrink from the left
			if right-left+1 < bestLen {
				bestLen, bestStart = right-left+1, left
			}
			need[s[left]]++
			if need[s[left]] > 0 { // dropped a copy we needed
				missing++
			}
			left++
		}
	}
	if bestLen > len(s) {
		return ""
	}
	return s[bestStart : bestStart+bestLen]
}

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        need = [0] * 128               # need[c] > 0: window still lacks that many copies of c
        for c in t:
            need[ord(c)] += 1
        missing = len(t)               # characters of t (with duplicates) not yet covered
        best_len, best_start = len(s) + 1, 0
        left = 0
        for right, c in enumerate(s):
            if need[ord(c)] > 0:       # this copy was still needed
                missing -= 1
            need[ord(c)] -= 1          # extra copies drive need below zero (surplus)
            while missing == 0:        # window covers t: record it, then shrink from the left
                if right - left + 1 < best_len:
                    best_len, best_start = right - left + 1, left
                d = ord(s[left])
                need[d] += 1
                if need[d] > 0:        # we just dropped a copy we needed
                    missing += 1
                left += 1
        return "" if best_len > len(s) else s[best_start:best_start + best_len]
