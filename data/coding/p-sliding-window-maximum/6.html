class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int> dq;  // indices; their values strictly decrease from front to back
        vector<int> out;
        out.reserve(nums.size() - k + 1);
        for (int i = 0; i < (int)nums.size(); i++) {
            if (!dq.empty() && dq.front() <= i - k) dq.pop_front();       // slid out of the window
            while (!dq.empty() && nums[dq.back()] <= nums[i]) dq.pop_back(); // can never be a max again
            dq.push_back(i);
            if (i >= k - 1) out.push_back(nums[dq.front()]);  // first full window ends at k-1
        }
        return out;
    }
};
func maxSlidingWindow(nums []int, k int) []int {
	// dq holds indices whose values strictly decrease front to back.
	// A slice with a moving head works as a deque: each index is appended and removed once.
	dq := make([]int, 0, k)
	out := make([]int, 0, len(nums)-k+1)
	for i, x := range nums {
		if len(dq) > 0 && dq[0] <= i-k { // front index slid out of the window
			dq = dq[1:]
		}
		for len(dq) > 0 && nums[dq[len(dq)-1]] <= x { // can never be a max again
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, i)
		if i >= k-1 { // first full window ends at index k-1
			out = append(out, nums[dq[0]])
		}
	}
	return out
}

static class Solution {
    public int[] maxSlidingWindow(int[] nums, int k) {
        Deque<Integer> dq = new ArrayDeque<>();  // indices; values strictly decrease front to back
        int[] out = new int[nums.length - k + 1];
        for (int i = 0; i < nums.length; i++) {
            if (!dq.isEmpty() && dq.peekFirst() <= i - k) dq.pollFirst();          // slid out
            while (!dq.isEmpty() && nums[dq.peekLast()] <= nums[i]) dq.pollLast(); // never a max again
            dq.offerLast(i);
            if (i >= k - 1) out[i - k + 1] = nums[dq.peekFirst()];  // first full window ends at k-1
        }
        return out;
    }
}
class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        dq = deque()   # indices; their values strictly decrease from front to back
        out = []
        for i, x in enumerate(nums):
            if dq and dq[0] <= i - k:        # front index slid out of the window
                dq.popleft()
            while dq and nums[dq[-1]] <= x:  # smaller-or-equal values can never be a max again
                dq.pop()
            dq.append(i)
            if i >= k - 1:                   # first full window ends at index k-1
                out.append(nums[dq[0]])
        return out
