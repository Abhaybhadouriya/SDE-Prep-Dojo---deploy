All four are Kadane with cur and best both started at nums[0].

class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int cur = nums[0];    // best sum of a subarray that ends at the current index
        int best = nums[0];   // best sum seen anywhere so far (non-empty, so start at nums[0])
        for (size_t i = 1; i < nums.size(); ++i) {
            // extend the run ending at i-1, or start fresh at i if that run only drags us down
            cur = max(nums[i], cur + nums[i]);
            best = max(best, cur);
        }
        return best;
    }
};
func maxSubArray(nums []int) int {
	cur, best := nums[0], nums[0] // cur = best sum ending here; best = best anywhere (non-empty)
	for _, x := range nums[1:] {
		// extend the run ending just before x, or start fresh at x if that run drags us down
		if cur+x > x {
			cur += x
		} else {
			cur = x
		}
		if cur > best {
			best = cur
		}
	}
	return best
}

static class Solution {

            // extend the run ending at i-1, or start fresh at i if that run only drags us down

        }
        return best;
    }
}
class Solution:
    def maxSubArray(self, nums: List[int]) -> int:
        cur = best = nums[0]      # cur = best sum ending here; best = best anywhere (non-empty)

            # extend the run ending just before x, or start fresh at x if that run drags us down
