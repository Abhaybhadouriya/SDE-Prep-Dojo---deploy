Interval DP with a split point: burst balloons

Problem (restated from LeetCode 312, Burst Balloons). A row of n balloons carries numbers nums. Bursting balloon i earns left * nums[i] * right coins. Here left and right are the numbers on its current neighbours; a missing neighbour counts as 1. After a burst, its two neighbours become adjacent. Burst all of them and return the most coins you can collect. Example: [2, 4, 3, 5] gives 115. Limits: 1 <= n <= 300, 0 <= nums[i] <= 100.

The obvious state, "which balloon do I burst first?", does not work. After you burst k first, the balloons on its left and right become neighbours, so the two sides are no longer separate problems. Ask the opposite question. Which balloon in the open range (l, r) is burst last? If it is k, then every balloon between l and k goes before k, and so does every balloon between k and r. While those go, l, k and r are all still standing, so the left part only ever sees l and k as its walls, and the right part only sees k and r. The two halves are independent.

Pad the row with a 1 at each end. Define dp[l][r] = the most coins from bursting every balloon strictly between l and r (l and r themselves stay). Then dp[l][r] = max over l < k < r of dp[l][k] + a[l]·a[k]·a[r] + dp[k][r], with dp[l][l+1] = 0. Fill by gap r - l:

  1. r - l = 1: nothing lies strictly between l and r, so 0 coins. These are the base cases.
  2. r - l = 2: exactly one balloon inside, so it is the last: a[l] * a[k] * a[r]. dp[0][2] = 8 (k = 1), dp[1][3] = 24 (k = 2), dp[2][4] = 60 (k = 3), dp[3][5] = 15 (k = 4).
  3. r - l = 3: two balloons inside; try each as the last one. dp[0][3] = 30 (k = 1), dp[1][4] = 100 (k = 2), dp[2][5] = 80 (k = 4).
  4. r - l = 4: dp[0][4] = 110 (k = 1), dp[1][5] = 110 (k = 4).
  5. r - l = 5, the whole row: dp[0][5] = 115 with balloon 5 (index 4) burst last. That is the answer for [2, 4, 3, 5].
// Burst balloons. Pad with 1s. dp[l][r] = best coins from bursting everything
// strictly between l and r; k = the LAST balloon burst in (l, r).
func maxCoins(nums []int) int {
	a := append(append([]int{1}, nums...), 1)
	n := len(a)
	dp := make([][]int, n)
	for i := range dp {
		dp[i] = make([]int, n)
	}
	for length := 2; length < n; length++ {
		for l := 0; l+length < n; l++ {
			r := l + length
			for k := l + 1; k < r; k++ {
				dp[l][r] = max(dp[l][r], dp[l][k]+a[l]*a[k]*a[r]+dp[k][r])
			}
		}
	}
	return dp[0][n-1]
}
// Burst balloons. Pad with 1 at both ends. dp[l][r] = best coins from bursting
// every balloon strictly between l and r. Choose k = the LAST balloon burst in
// (l, r): its neighbours at that moment are exactly l and r.
int maxCoins(vector<int> nums) {
    nums.insert(nums.begin(), 1);
    nums.push_back(1);
    int n = nums.size();
    vector<vector<int>> dp(n, vector<int>(n, 0));
    for (int len = 2; len < n; len++)                           // r - l
        for (int l = 0; l + len < n; l++) {
            int r = l + len;
            for (int k = l + 1; k < r; k++)
                dp[l][r] = max(dp[l][r], dp[l][k] + nums[l] * nums[k] * nums[r] + dp[k][r]);
        }
    return dp[0][n - 1];
}
// Burst balloons. Pad with 1s. dp[l][r] = best coins from bursting everything
// strictly between l and r; k = the LAST balloon burst in (l, r).
static int maxCoins(int[] nums) {
    int n = nums.length + 2;
    int[] a = new int[n];
    a[0] = a[n - 1] = 1;
    System.arraycopy(nums, 0, a, 1, nums.length);
    int[][] dp = new int[n][n];
    for (int len = 2; len < n; len++)
        for (int l = 0; l + len < n; l++) {
            int r = l + len;
            for (int k = l + 1; k < r; k++)
                dp[l][r] = Math.max(dp[l][r], dp[l][k] + a[l] * a[k] * a[r] + dp[k][r]);
        }
    return dp[0][n - 1];
}
def max_coins(nums: List[int]) -> int:
    """dp[l][r] = best coins from bursting everything strictly between l and r;
    k = the LAST balloon burst in (l, r), so its neighbours are l and r."""
    a = [1] + nums + [1]
    n = len(a)
    dp = [[0] * n for _ in range(n)]
    for length in range(2, n):
        for l in range(n - length):
            r = l + length
            dp[l][r] = max(dp[l][k] + a[l] * a[k] * a[r] + dp[k][r] for k in range(l + 1, r))
    return dp[0][n - 1]

There are O(n2) cells and each one tries O(n) split points, so the cost is O(n3). With n = 300 that is about 4.5 million inner steps (the open-interval bounds cut it by about 6), which is fine. The open-interval form also makes the padding natural: the walls are real elements of the padded array.

Same shape: cutting a stick

Problem (restated from LeetCode 1547, Minimum Cost to Cut a Stick). A wooden stick runs from position 0 to n. You must cut it at every position in cuts, in any order you choose. One cut costs the current length of the piece being cut. Return the smallest total cost. Example: n = 10, cuts = [2, 4, 7] gives 20. Limits: 2 <= n <= 106, up to 100 distinct cuts strictly inside the stick.

Sort the cuts and add the ends 0 and n as walls. The positions are now the elements of the "row", and n does not matter any more: only about 100 positions exist. dp[l][r] = the cheapest way to make every cut strictly between positions l and r. Whichever cut k you make first inside that piece costs p[r] - p[l] and splits it into two independent pieces. Here "first" works because a cut really does split the stick. For balloons it had to be "last". Ask yourself which event makes the two sides independent.

// Minimum cost to cut a stick: positions 0, sorted cuts, length; same shape as burst.
func minCost(length int, cuts []int) int {
	p := append(append([]int{0}, cuts...), length)
	sort.Ints(p)
	m := len(p)
	dp := make([][]int, m)
	for i := range dp {
		dp[i] = make([]int, m)
	}
	for gap := 2; gap < m; gap++ {
		for l := 0; l+gap < m; l++ {
			r, best := l+gap, math.MaxInt
			for k := l + 1; k < r; k++ {
				best = min(best, dp[l][k]+dp[k][r])
			}
			dp[l][r] = best + p[r] - p[l]
		}
	}
	return dp[0][m-1]
}
// Minimum cost to cut a stick: same shape. Positions 0, sorted cuts, len.
// dp[l][r] = cheapest way to make every cut strictly between positions l and r.
int minCost(int len, vector<int> cuts) {
    cuts.push_back(0);
    cuts.push_back(len);
    sort(cuts.begin(), cuts.end());
    int m = cuts.size();
    vector<vector<int>> dp(m, vector<int>(m, 0));
    for (int gap = 2; gap < m; gap++)
        for (int l = 0; l + gap < m; l++) {
            int r = l + gap, best = INT_MAX;
            for (int k = l + 1; k < r; k++) best = min(best, dp[l][k] + dp[k][r]);
            dp[l][r] = best + cuts[r] - cuts[l];                  // this cut costs the piece length
        }
    return dp[0][m - 1];
}
// Minimum cost to cut a stick: positions 0, sorted cuts, len; same shape as burst.
static int minCost(int len, int[] cuts) {
    int m = cuts.length + 2;
    int[] p = new int[m];
    System.arraycopy(cuts, 0, p, 1, cuts.length);
    p[m - 1] = len;
    Arrays.sort(p);
    int[][] dp = new int[m][m];
    for (int gap = 2; gap < m; gap++)
        for (int l = 0; l + gap < m; l++) {
            int r = l + gap, best = Integer.MAX_VALUE;
            for (int k = l + 1; k < r; k++) best = Math.min(best, dp[l][k] + dp[k][r]);
            dp[l][r] = best + p[r] - p[l];
        }
    return dp[0][m - 1];
}
def min_cost(length: int, cuts: List[int]) -> int:
    """Positions 0, sorted cuts, length; dp[l][r] = cheapest way to make every cut in (l, r)."""
    p = [0] + sorted(cuts) + [length]
    m = len(p)
    dp = [[0] * m for _ in range(m)]
    for gap in range(2, m):
        for l in range(m - gap):
            r = l + gap
            dp[l][r] = min(dp[l][k] + dp[k][r] for k in range(l + 1, r)) + p[r] - p[l]
    return dp[0][m - 1]

Other problems with the same "try every split" shape: Minimum Score Triangulation of Polygon (1039) and the classic matrix-chain multiplication (choose the last multiplication); Minimum Cost Tree From Leaf Values (1130; it also has an O(n) monotonic-stack answer); Strange Printer (664); Guess Number Higher or Lower II (375, a minimax over the split); and Minimum Cost to Merge Stones (1000, which needs an extra "number of piles" dimension). Games where two players take from either end, such as Predict the Winner (486) and Stone Game (877), peel the ends instead: dp[i][j] = max(a[i] - dp[i+1][j], a[j] - dp[i][j-1]).
