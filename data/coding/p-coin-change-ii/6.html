class Solution {
public:
    int change(int amount, vector<int>& coins) {
        // ways[a] = number of coin multisets (order ignored) that sum to a, using the coins seen so far.
        // Unsigned on purpose: cells below `amount` can exceed even 64 bits, and signed overflow is UB.
        // Unsigned arithmetic wraps mod 2^64, and since the true answer fits in 32 bits, ways[amount] is exact.
        vector<unsigned long long> ways(amount + 1, 0);
        ways[0] = 1;                                  // one way to make 0: take nothing
        for (int c : coins)                           // coins outer: each multiset counted once
            for (int a = c; a <= amount; a++)         // upwards: coin c may be reused
                ways[a] += ways[a - c];               // without c (old value) + using c at least once
        return (int)ways[amount];
    }
};
func change(amount int, coins []int) int {
	// ways[a] = number of coin multisets (order ignored) that sum to a, using the coins seen so far.
	// Go integer addition wraps (defined behaviour), so even if a cell below amount overflows,
	// ways[amount] is exact because the true answer fits in 32 bits.
	ways := make([]int, amount+1)
	ways[0] = 1               // one way to make 0: take nothing
	for _, c := range coins { // coins outer: each multiset counted once
		for a := c; a <= amount; a++ { // upwards: coin c may be reused
			ways[a] += ways[a-c] // without c (old value) + using c at least once
		}
	}
	return ways[amount]
}

class Solution:
    def change(self, amount: int, coins: List[int]) -> int:
        # ways[a] = number of coin multisets (order ignored) that sum to a,
        # using only the coins processed so far.
        ways = [1] + [0] * amount               # one way to make 0: take nothing
        for c in coins:                         # coins outer: each multiset is built in coin order, once
            for a in range(c, amount + 1):      # upwards: coin c may be reused
                ways[a] += ways[a - c]          # ways without c (old value) + ways using c at least once
        return ways[amount]
