type trieNode struct {
	kid  [26]*trieNode
	word string // non-empty on the node where a word ends
	live int    // number of non-nil children (for pruning)
}

func findWords(board [][]byte, words []string) []string {
	root := &trieNode{}
	for _, w := range words { // 1. all words share one prefix tree
		cur := root
		for i := 0; i < len(w); i++ {
			k := w[i] - 'a'
			if cur.kid[k] == nil {
				cur.kid[k] = &trieNode{}
				cur.live++
			}
			cur = cur.kid[k]
		}
		cur.word = w
	}
	rows, cols := len(board), len(board[0])
	found := []string{}
	var dfs func(r, c int, parent *trieNode)
	dfs = func(r, c int, parent *trieNode) {
		ch := board[r][c]
		if ch == '#' {
			return // already on the current path
		}
		node := parent.kid[ch-'a']
		if node == nil {
			return // no word continues with ch: prune
		}
		if node.word != "" {
			found = append(found, node.word)
			node.word = "" // report each word once
		}
		board[r][c] = '#'
		if r+1 < rows {
			dfs(r+1, c, node)
		}
		if r > 0 {
			dfs(r-1, c, node)
		}
		if c+1 < cols {
			dfs(r, c+1, node)
		}
		if c > 0 {
			dfs(r, c-1, node)
		}
		board[r][c] = ch // backtrack
		if node.live == 0 && node.word == "" { // branch exhausted: cut it off
			parent.kid[ch-'a'] = nil
			parent.live--
		}
	}
	for r := 0; r < rows; r++ { // 2. every cell can start a word
		for c := 0; c < cols; c++ {
			dfs(r, c, root)
		}
	}
	return found
}
class Solution {
    struct Node {
        int kid[26];
        int word = -1;      // index into words if a word ends here, else -1
        int live = 0;       // number of non-null children (for pruning)
        Node() { fill(kid, kid + 26, -1); }
    };
    vector<Node> trie;      // node pool; trie[0] is the root
    vector<string> found;

    void dfs(vector<vector<char>>& board, const vector<string>& words, int r, int c, int parent) {
        char ch = board[r][c];
        if (ch == '#') return;                         // already on the current path
        int node = trie[parent].kid[ch - 'a'];
        if (node < 0) return;                          // no word continues with ch: prune
        if (trie[node].word >= 0) {
            found.push_back(words[trie[node].word]);
            trie[node].word = -1;                      // report each word once
        }
        board[r][c] = '#';
        static const int dr[4] = {1, -1, 0, 0}, dc[4] = {0, 0, 1, -1};
        for (int d = 0; d < 4; ++d) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < (int)board.size() && nc >= 0 && nc < (int)board[0].size())
                dfs(board, words, nr, nc, node);
        }
        board[r][c] = ch;                              // backtrack
        if (trie[node].live == 0 && trie[node].word < 0) {
            trie[parent].kid[ch - 'a'] = -1;           // branch exhausted: cut it off
            --trie[parent].live;
        }
    }

public:
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        trie.assign(1, Node());
        found.clear();
        for (int i = 0; i < (int)words.size(); ++i) {   // 1. all words share one prefix tree
            int cur = 0;
            for (char ch : words[i]) {
                int k = ch - 'a';
                if (trie[cur].kid[k] < 0) {
                    trie[cur].kid[k] = trie.size();     // index of the node pushed next
                    ++trie[cur].live;
                    trie.push_back(Node());
                }
                cur = trie[cur].kid[k];
            }
            trie[cur].word = i;
        }
        for (int r = 0; r < (int)board.size(); ++r)     // 2. every cell can start a word
            for (int c = 0; c < (int)board[0].size(); ++c)
                dfs(board, words, r, c, 0);
        return found;
    }
};

static class Solution {
    static class TrieNode {
        TrieNode[] kid = new TrieNode[26];
        String word;            // set on the node where a word ends
        int live;               // number of non-null children (for pruning)
    }

    private final List<String> found = new ArrayList<>();

    public List<String> findWords(char[][] board, String[] words) {
        TrieNode root = new TrieNode();
        for (String w : words) {                     // 1. all words share one prefix tree
            TrieNode cur = root;
            for (char ch : w.toCharArray()) {
                int k = ch - 'a';
                if (cur.kid[k] == null) { cur.kid[k] = new TrieNode(); cur.live++; }
                cur = cur.kid[k];
            }
            cur.word = w;
        }
        for (int r = 0; r < board.length; r++)       // 2. every cell can start a word
            for (int c = 0; c < board[0].length; c++)
                dfs(board, r, c, root);
        return found;
    }

    private void dfs(char[][] board, int r, int c, TrieNode parent) {
        char ch = board[r][c];
        if (ch == '#') return;                       // already on the current path
        TrieNode node = parent.kid[ch - 'a'];
        if (node == null) return;                    // no word continues with ch: prune
        if (node.word != null) {
            found.add(node.word);
            node.word = null;                        // report each word once
        }
        board[r][c] = '#';
        if (r + 1 < board.length) dfs(board, r + 1, c, node);
        if (r > 0) dfs(board, r - 1, c, node);
        if (c + 1 < board[0].length) dfs(board, r, c + 1, node);
        if (c > 0) dfs(board, r, c - 1, node);
        board[r][c] = ch;                            // backtrack
        if (node.live == 0 && node.word == null) {   // branch exhausted: cut it off
            parent.kid[ch - 'a'] = null;
            parent.live--;
        }
    }
}
class TrieNode:
    __slots__ = ("kids", "word")

    def __init__(self) -> None:
        self.kids = {}                  # letter -> TrieNode
        self.word: Optional[str] = None # set on the node where a word ends


class Solution:
    def findWords(self, board: List[List[str]], words: List[str]) -> List[str]:
        root = TrieNode()
        for w in words:                 # 1. all words share one prefix tree
            node = root
            for ch in w:
                node = node.kids.setdefault(ch, TrieNode())
            node.word = w

        rows, cols = len(board), len(board[0])
        found: List[str] = []

        def dfs(r: int, c: int, parent: TrieNode) -> None:
            ch = board[r][c]
            node = parent.kids.get(ch)  # '#' (visited) is never a key
            if node is None:
                return                  # no word continues with this letter: prune
            if node.word is not None:
                found.append(node.word)
                node.word = None        # report each word once
            board[r][c] = "#"           # mark the cell as used on this path
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if 0 <= nr < rows and 0 <= nc < cols:
                    dfs(nr, nc, node)
            board[r][c] = ch            # backtrack: restore the cell
            if not node.kids and node.word is None:
                del parent.kids[ch]     # branch exhausted: never walk it again

        for r in range(rows):           # 2. every cell can start a word
            for c in range(cols):
                dfs(r, c, root)
        return found
