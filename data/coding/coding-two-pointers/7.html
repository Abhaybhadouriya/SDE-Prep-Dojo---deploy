Same direction 2: the Dutch national flag (three-way partition)

The problem, in our words. An array c holds only the values 0, 1 and 2 (three paint colours). Rearrange it in place so that all 0s come first, then all 1s, then all 2s. Do not use the library sort. 1 <= c.length <= 300; c[i] is 0, 1 or 2. Follow-up: one pass and O(1) extra space. Example: [2, 0, 1, 2, 0, 1, 0] -> [0, 0, 0, 1, 1, 2, 2]. Practice it: LeetCode 75 - Sort Colors (takeuforward), our write-up Sort Colors.

Counting (two passes: count 0s, 1s, 2s, then overwrite) is a correct first answer. The one-pass answer, Dijkstra's Dutch national flag partition, keeps four regions with three pointers:

# invariant:  [0, lo) = 0s   [lo, mid) = 1s   [mid, hi] = not looked at yet   (hi, end] = 2s
while mid <= hi:   # the unknown region shrinks by one cell every step
  1. lo = 0, mid = 0, hi = 6. a[mid] = 2: swap it with a[hi] and hi--. mid does NOT move: the value that came from hi has never been looked at.
  2. lo = 0, mid = 0, hi = 5. a[mid] = 0: swap it with a[lo] (which is a known 1, or the same cell), then lo++ and mid++.
  3. lo = 1, mid = 1, hi = 5. a[mid] = 0: swap it with a[lo] (which is a known 1, or the same cell), then lo++ and mid++.
  4. lo = 2, mid = 2, hi = 5. a[mid] = 1: it is already in the right zone. Just mid++.
  5. lo = 2, mid = 3, hi = 5. a[mid] = 2: swap it with a[hi] and hi--. mid does NOT move: the value that came from hi has never been looked at.
  6. lo = 2, mid = 3, hi = 4. a[mid] = 1: it is already in the right zone. Just mid++.
  7. lo = 2, mid = 4, hi = 4. a[mid] = 0: swap it with a[lo] (which is a known 1, or the same cell), then lo++ and mid++.
  8. mid = 5 is past hi = 4: no unknown cells are left. The array is [0, 0, 0, 1, 1, 2, 2] after one pass.
func removeDuplicates(nums []int) int {
	if len(nums) == 0 {
		return 0
	}
	w := 1 // nums[:w] = kept, distinct
	for r := 1; r < len(nums); r++ {
		if nums[r] != nums[w-1] {
			nums[w] = nums[r]
			w++
		}
	}
	return w
}

func sortColors(a []int) {
	lo, mid, hi := 0, 0, len(a)-1 // [0,lo)=0  [lo,mid)=1  [mid,hi]=?  (hi,end]=2
	for mid <= hi {
		switch a[mid] {
		case 0:
			a[lo], a[mid] = a[mid], a[lo]
			lo++
			mid++
		case 1:
			mid++
		default:
			a[mid], a[hi] = a[hi], a[mid]
			hi-- // mid stays: the swapped-in value is unchecked
		}
	}
}

func mergeInto(a []int, m int, b []int, n int) {
	i, j, w := m-1, n-1, m+n-1 // fill from the back
	for j >= 0 {
		if i >= 0 && a[i] > b[j] {
			a[w] = a[i]
			i--
		} else {
			a[w] = b[j]
			j--
		}
		w--
	}
}
int removeDuplicates(vector<int>& nums) {
    if (nums.empty()) return 0;
    int w = 1;                                   // nums[0..w) is the kept, distinct part
    for (size_t r = 1; r < nums.size(); r++)
        if (nums[r] != nums[w - 1]) nums[w++] = nums[r];
    return w;
}

void sortColors(vector<int>& a) {
    int lo = 0, mid = 0, hi = (int)a.size() - 1; // [0,lo)=0  [lo,mid)=1  [mid,hi]=?  (hi,end]=2
    while (mid <= hi) {
        if (a[mid] == 0) swap(a[lo++], a[mid++]);
        else if (a[mid] == 1) mid++;
        else swap(a[mid], a[hi--]);              // mid stays: the swapped-in value is unchecked
    }
}

void mergeInto(vector<int>& a, int m, const vector<int>& b, int n) {
    int i = m - 1, j = n - 1, w = m + n - 1;     // fill from the back, where the free space is
    while (j >= 0) {
        if (i >= 0 && a[i] > b[j]) a[w--] = a[i--];
        else a[w--] = b[j--];
    }
}

static void sortColors(int[] a) {
    int lo = 0, mid = 0, hi = a.length - 1;      // [0,lo)=0  [lo,mid)=1  [mid,hi]=?  (hi,end]=2
    while (mid <= hi) {
        if (a[mid] == 0) { int t = a[lo]; a[lo++] = a[mid]; a[mid++] = t; }
        else if (a[mid] == 1) mid++;
        else { int t = a[hi]; a[hi--] = a[mid]; a[mid] = t; }   // mid stays: unchecked value
    }
}

def remove_duplicates(nums):
    if not nums:
        return 0
    w = 1                              # nums[0:w] = kept part, all distinct
    for r in range(1, len(nums)):
        if nums[r] != nums[w - 1]:     # new value: keep it
            nums[w] = nums[r]
            w += 1
    return w


def sort_colors(nums):
    lo, mid, hi = 0, 0, len(nums) - 1  # [0,lo)=0s  [lo,mid)=1s  [mid,hi]=unknown  (hi,end]=2s
    while mid <= hi:
        if nums[mid] == 0:
            nums[lo], nums[mid] = nums[mid], nums[lo]
            lo += 1
            mid += 1
        elif nums[mid] == 1:
            mid += 1
        else:
            nums[mid], nums[hi] = nums[hi], nums[mid]
            hi -= 1                    # do NOT advance mid: the swapped-in value is unchecked


def merge_into(a, m, b, n):
    i, j, w = m - 1, n - 1, m + n - 1  # fill from the back: free space is at the end of a
    while j >= 0:                      # once b is used up, the rest of a is already in place
        if i >= 0 and a[i] > b[j]:
            a[w] = a[i]
            i -= 1
        else:
            a[w] = b[j]
            j -= 1
        w -= 1

The one-line bug. After swapping a[mid] with a[hi], do not advance mid: the value that just arrived from hi is unknown. After swapping with a[lo] you may advance mid, because everything in [lo, mid) is a known 1 (or lo == mid). And the loop condition is mid <= hi, not <: the last unknown cell still needs a look.

The same partition step is the heart of three-way quicksort (all keys equal to the pivot go in the middle region), and Sort Array by Parity is its two-region version.
