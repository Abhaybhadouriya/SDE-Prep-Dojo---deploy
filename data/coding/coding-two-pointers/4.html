Opposite ends 3: palindromes

A palindrome reads the same both ways, so compare s[i] with s[j] from the two ends and move both inwards while they match. The interview version adds a twist.

The problem, in our words. Given a lowercase string s, return true if it can be made into a palindrome by deleting at most one character. 1 <= s.length <= 105; lowercase English letters. Example: "racecxar" -> true (delete x). Example: "abca" -> true (delete b or c). Example: "abcd" -> false. Practice it: LeetCode 680 - Valid Palindrome II, our write-up Valid Palindrome II (a Meta favourite: frequency 8 of 8 in the company data).

The insight: walk inwards while the ends match. At the first mismatch, one of the two characters has to be the deleted one, so check whether s[i+1..j] or s[i..j-1] is a palindrome. That is two more linear scans at most, so the whole thing is O(n) and O(1) space. No need to try deleting every character (that would be O(n2)).

Sibling: Valid Palindrome skips non-alphanumeric characters and ignores case: inside the loop, while i < j and not s[i].isalnum(): i += 1 (and the same for j), then compare lowercase letters. Reverse String is the same loop with a swap instead of a compare.

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

func twoSumSorted(nums []int, target int) []int {
	lo, hi := 0, len(nums)-1
	for lo < hi {
		s := nums[lo] + nums[hi]
		switch {
		case s == target:
			return []int{lo + 1, hi + 1} // 1-based positions
		case s < target:
			lo++ // nums[lo] too small even with the biggest partner
		default:
			hi-- // nums[hi] too big even with the smallest partner
		}
	}
	return nil
}

func maxArea(h []int) int {
	lo, hi, best := 0, len(h)-1, 0
	for lo < hi {
		best = max(best, (hi-lo)*min(h[lo], h[hi]))
		if h[lo] < h[hi] { // shorter wall can only get worse partners
			lo++
		} else {
			hi--
		}
	}
	return best
}

func isPal(s string, i, j int) bool {
	for ; i < j; i, j = i+1, j-1 {
		if s[i] != s[j] {
			return false
		}
	}
	return true
}

func validPalindromeOneDelete(s string) bool {
	for i, j := 0, len(s)-1; i < j; i, j = i+1, j-1 {
		if s[i] != s[j] { // first mismatch: one of the two must go
			return isPal(s, i+1, j) || isPal(s, i, j-1)
		}
	}
	return true
}
vector<int> twoSumSorted(const vector<int>& nums, int target) {
    int lo = 0, hi = (int)nums.size() - 1;
    while (lo < hi) {
        int s = nums[lo] + nums[hi];
        if (s == target) return {lo + 1, hi + 1};   // 1-based positions
        if (s < target) lo++;                        // nums[lo] too small even with the biggest partner
        else hi--;                                   // nums[hi] too big even with the smallest partner
    }
    return {};
}

def max_area(height):
    lo, hi, best = 0, len(height) - 1, 0
    while lo < hi:
        best = max(best, (hi - lo) * min(height[lo], height[hi]))
        if height[lo] < height[hi]:    # the shorter wall cannot do better with any closer partner
            lo += 1
        else:
            hi -= 1
    return best


def valid_palindrome_one_delete(s):
    def is_pal(i, j):
        while i < j:
            if s[i] != s[j]:
                return False
            i, j = i + 1, j - 1
        return True

    i, j = 0, len(s) - 1
    while i < j:
        if s[i] != s[j]:               # first mismatch: one of the two must go
            return is_pal(i + 1, j) or is_pal(i, j - 1)
        i, j = i + 1, j - 1
    return True
