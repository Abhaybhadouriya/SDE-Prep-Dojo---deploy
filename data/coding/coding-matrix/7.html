Templates you can write from memory

Go, C++, Java and Python versions were compiled and tested (spiral and set-zeroes against brute force on hundreds of random matrices, rotation against a reference).

1. Spiral order with four walls

// spiralOrder walks the outer ring clockwise, shrinks the four walls, repeats. O(R*C).
func spiralOrder(m [][]int) []int {
	out := []int{}
	if len(m) == 0 || len(m[0]) == 0 {
		return out
	}
	top, bottom, left, right := 0, len(m)-1, 0, len(m[0])-1
	for top <= bottom && left <= right {
		for c := left; c <= right; c++ { // 1. top row
			out = append(out, m[top][c])
		}
		top++
		for r := top; r <= bottom; r++ { // 2. right column
			out = append(out, m[r][right])
		}
		right--
		if top <= bottom { // 3. bottom row, if any left
			for c := right; c >= left; c-- {
				out = append(out, m[bottom][c])
			}
			bottom--
		}
		if left <= right { // 4. left column, if any left
			for r := bottom; r >= top; r-- {
				out = append(out, m[r][left])
			}
			left++
		}
	}
	return out
}
// Walk the outer ring clockwise, shrink the four walls, repeat. O(R*C).
vector<int> spiralOrder(const vector<vector<int>>& m) {
    vector<int> out;
    if (m.empty() || m[0].empty()) return out;
    int top = 0, bottom = m.size() - 1, left = 0, right = m[0].size() - 1;
    while (top <= bottom && left <= right) {
        for (int c = left; c <= right; ++c) out.push_back(m[top][c]);      // 1. top row
        ++top;
        for (int r = top; r <= bottom; ++r) out.push_back(m[r][right]);    // 2. right column
        --right;
        if (top <= bottom) {                                               // 3. bottom row, if any left
            for (int c = right; c >= left; --c) out.push_back(m[bottom][c]);
            --bottom;
        }
        if (left <= right) {                                               // 4. left column, if any left
            for (int r = bottom; r >= top; --r) out.push_back(m[r][left]);
            ++left;
        }
    }
    return out;
}
// Walk the outer ring clockwise, shrink the four walls, repeat. O(R*C).
static List<Integer> spiralOrder(int[][] m) {
    List<Integer> out = new ArrayList<>();
    if (m.length == 0 || m[0].length == 0) return out;
    int top = 0, bottom = m.length - 1, left = 0, right = m[0].length - 1;
    while (top <= bottom && left <= right) {
        for (int c = left; c <= right; c++) out.add(m[top][c]);       // 1. top row
        top++;
        for (int r = top; r <= bottom; r++) out.add(m[r][right]);     // 2. right column
        right--;
        if (top <= bottom) {                                          // 3. bottom row, if any left
            for (int c = right; c >= left; c--) out.add(m[bottom][c]);
            bottom--;
        }
        if (left <= right) {                                          // 4. left column, if any left
            for (int r = bottom; r >= top; r--) out.add(m[r][left]);
            left++;
        }
    }
    return out;
}
def spiral_order(m: list) -> list:
    """Walk the outer ring clockwise, shrink the four walls, repeat. O(R*C)."""
    out = []
    if not m or not m[0]:
        return out
    top, bottom, left, right = 0, len(m) - 1, 0, len(m[0]) - 1
    while top <= bottom and left <= right:
        for c in range(left, right + 1):          # 1. along the top row
            out.append(m[top][c])
        top += 1
        for r in range(top, bottom + 1):          # 2. down the right column
            out.append(m[r][right])
        right -= 1
        if top <= bottom:                         # 3. back along the bottom row (if one is left)
            for c in range(right, left - 1, -1):
                out.append(m[bottom][c])
            bottom -= 1
        if left <= right:                         # 4. up the left column (if one is left)
            for r in range(bottom, top - 1, -1):
                out.append(m[r][left])
            left += 1
    return out

2. Rotate 90 degrees clockwise in place

// rotate turns an n x n matrix 90 degrees clockwise in place: transpose, then reverse every row.
func rotate(m [][]int) {
	n := len(m)
	for i := 0; i < n; i++ {
		for j := i + 1; j < n; j++ { // upper triangle only
			m[i][j], m[j][i] = m[j][i], m[i][j]
		}
	}
	for _, row := range m {
		for l, r := 0, n-1; l < r; l, r = l+1, r-1 {
			row[l], row[r] = row[r], row[l]
		}
	}
}
// Rotate n x n 90 degrees clockwise in place: transpose, then reverse every row.
void rotate(vector<vector<int>>& m) {
    int n = m.size();
    for (int i = 0; i < n; ++i)
        for (int j = i + 1; j < n; ++j)          // upper triangle only
            swap(m[i][j], m[j][i]);
    for (auto& row : m) reverse(row.begin(), row.end());
}
// Rotate n x n 90 degrees clockwise in place: transpose, then reverse every row.
static void rotate(int[][] m) {
    int n = m.length;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) {           // upper triangle only
            int t = m[i][j]; m[i][j] = m[j][i]; m[j][i] = t;
        }
    for (int[] row : m)
        for (int l = 0, r = n - 1; l < r; l++, r--) {
            int t = row[l]; row[l] = row[r]; row[r] = t;
        }
}
def rotate(m: list) -> None:
    """Rotate an n x n matrix 90 degrees clockwise in place: transpose, then reverse every row."""
    n = len(m)
    for i in range(n):
        for j in range(i + 1, n):                 # upper triangle only, or you swap twice
            m[i][j], m[j][i] = m[j][i], m[i][j]
    for row in m:
        row.reverse()

3. Set matrix zeroes in O(1) extra space

// setZeroes: any 0 wipes its row and column. O(1) extra space: row 0 / column 0 hold the marks.
func setZeroes(m [][]int) {
	R, C := len(m), len(m[0])
	col0 := false // column 0's own fate
	for r := 0; r < R; r++ {
		if m[r][0] == 0 {
			col0 = true
		}
		for c := 1; c < C; c++ {
			if m[r][c] == 0 {
				m[r][0], m[0][c] = 0, 0
			}
		}
	}
	for r := R - 1; r >= 0; r-- { // bottom-up: row 0 is wiped last
		for c := C - 1; c >= 1; c-- {
			if m[r][0] == 0 || m[0][c] == 0 {
				m[r][c] = 0
			}
		}
		if col0 {
			m[r][0] = 0
		}
	}
}
// Any 0 wipes its row and column. O(1) extra space: row 0 / column 0 hold the marks.
void setZeroes(vector<vector<int>>& m) {
    int R = m.size(), C = m[0].size();
    bool col0 = false;                           // column 0's own fate
    for (int r = 0; r < R; ++r) {
        if (m[r][0] == 0) col0 = true;
        for (int c = 1; c < C; ++c)
            if (m[r][c] == 0) m[r][0] = m[0][c] = 0;
    }
    for (int r = R - 1; r >= 0; --r) {           // bottom-up: row 0 is wiped last
        for (int c = C - 1; c >= 1; --c)
            if (m[r][0] == 0 || m[0][c] == 0) m[r][c] = 0;
        if (col0) m[r][0] = 0;
    }
}
// Any 0 wipes its row and column. O(1) extra space: row 0 / column 0 hold the marks.
static void setZeroes(int[][] m) {
    int R = m.length, C = m[0].length;
    boolean col0 = false;                           // column 0's own fate
    for (int r = 0; r < R; r++) {
        if (m[r][0] == 0) col0 = true;
        for (int c = 1; c < C; c++)
            if (m[r][c] == 0) { m[r][0] = 0; m[0][c] = 0; }
    }
    for (int r = R - 1; r >= 0; r--) {              // bottom-up: row 0 is wiped last
        for (int c = C - 1; c >= 1; c--)
            if (m[r][0] == 0 || m[0][c] == 0) m[r][c] = 0;
        if (col0) m[r][0] = 0;
    }
}
def set_zeroes(m: list) -> None:
    """Any 0 wipes its row and column. O(1) extra space: row 0 / column 0 hold the marks."""
    R, C = len(m), len(m[0])
    col0 = any(m[r][0] == 0 for r in range(R))    # column 0's own fate, kept aside
    for r in range(R):
        for c in range(1, C):
            if m[r][c] == 0:
                m[r][0] = 0                       # mark the row
                m[0][c] = 0                       # mark the column
    for r in range(R - 1, -1, -1):                # bottom-up so row 0's marks are read before row 0 is wiped
        for c in range(C - 1, 0, -1):
            if m[r][0] == 0 or m[0][c] == 0:
                m[r][c] = 0
        if col0:
            m[r][0] = 0

4. Row / column / box validation with bitmasks

// isValidSudoku: each digit at most once per row, column and 3x3 box. One 9-bit mask per unit.
func isValidSudoku(board [][]byte) bool {
	var rows, cols, boxes [9]int
	for r := 0; r < 9; r++ {
		for c := 0; c < 9; c++ {
			if board[r][c] == '.' {
				continue
			}
			bit := 1 << (board[r][c] - '1')
			b := (r/3)*3 + c/3 // box id 0..8, row-major
			if (rows[r]|cols[c]|boxes[b])&bit != 0 {
				return false
			}
			rows[r] |= bit
			cols[c] |= bit
			boxes[b] |= bit
		}
	}
	return true
}
// Each digit at most once per row, column and 3x3 box. One 9-bit mask per unit.
bool isValidSudoku(const vector<vector<char>>& board) {
    int rows[9] = {0}, cols[9] = {0}, boxes[9] = {0};
    for (int r = 0; r < 9; ++r)
        for (int c = 0; c < 9; ++c) {
            if (board[r][c] == '.') continue;
            int bit = 1 << (board[r][c] - '1');
            int b = (r / 3) * 3 + c / 3;         // box id 0..8, row-major
            if ((rows[r] | cols[c] | boxes[b]) & bit) return false;
            rows[r] |= bit; cols[c] |= bit; boxes[b] |= bit;
        }
    return true;
}
// Each digit at most once per row, column and 3x3 box. One 9-bit mask per unit.
static boolean isValidSudoku(char[][] board) {
    int[] rows = new int[9], cols = new int[9], boxes = new int[9];
    for (int r = 0; r < 9; r++)
        for (int c = 0; c < 9; c++) {
            if (board[r][c] == '.') continue;
            int bit = 1 << (board[r][c] - '1');
            int b = (r / 3) * 3 + c / 3;            // box id 0..8, row-major
            if (((rows[r] | cols[c] | boxes[b]) & bit) != 0) return false;
            rows[r] |= bit; cols[c] |= bit; boxes[b] |= bit;
        }
    return true;
}
def is_valid_sudoku(board: list) -> bool:
    """Each digit at most once per row, column and 3x3 box. One 9-bit mask per unit."""
    rows, cols, boxes = [0] * 9, [0] * 9, [0] * 9
    for r in range(9):
        for c in range(9):
            ch = board[r][c]
            if ch == '.':
                continue
            bit = 1 << (ord(ch) - ord('1'))
            b = (r // 3) * 3 + c // 3             # box id 0..8, row-major
            if (rows[r] | cols[c] | boxes[b]) & bit:
                return False
            rows[r] |= bit
            cols[c] |= bit
            boxes[b] |= bit
    return True
