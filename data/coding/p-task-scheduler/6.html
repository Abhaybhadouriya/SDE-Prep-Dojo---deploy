func leastInterval(tasks []byte, n int) int {
	var freq [26]int
	for _, t := range tasks {
		freq[t-'A']++
	}
	top, width := 0, 0
	for _, f := range freq {
		if f > top { // new maximum: rows in the frame
			top, width = f, 1
		} else if f == top { // another label reaching the last row
			width++
		}
	}
	frame := (top-1)*(n+1) + width // full rows of n+1 slots + short last row
	if len(tasks) > frame {        // frame overflows -> no idle at all
		return len(tasks)
	}
	return frame
}
class Solution {
public:
    int leastInterval(vector<char>& tasks, int n) {
        int freq[26] = {0};
        for (char t : tasks) freq[t - 'A']++;
        int top = *max_element(freq, freq + 26);           // rows in the frame
        int width = count(freq, freq + 26, top);           // labels that reach the last row
        int frame = (top - 1) * (n + 1) + width;           // full rows of n+1 slots + short last row
        return max((int)tasks.size(), frame);              // frame overflows -> no idle at all
    }
};

static class Solution {
    public int leastInterval(char[] tasks, int n) {
        int[] freq = new int[26];
        for (char t : tasks) freq[t - 'A']++;
        int top = 0, width = 0;
        for (int f : freq) {
            if (f > top) { top = f; width = 1; }      // new maximum: rows in the frame
            else if (f == top) width++;               // another label reaching the last row
        }
        int frame = (top - 1) * (n + 1) + width;      // full rows of n+1 slots + short last row
        return Math.max(tasks.length, frame);         // frame overflows -> no idle at all
    }
}
class Solution:
    def leastInterval(self, tasks: List[str], n: int) -> int:
        freq = [0] * 26
        for t in tasks:
            freq[ord(t) - ord('A')] += 1
        top = max(freq)                       # rows in the frame
        width = freq.count(top)               # labels that reach the last row
        frame = (top - 1) * (n + 1) + width   # (top-1) full rows of n+1 slots + a short last row
        return max(len(tasks), frame)         # frame overflows -> no idle needed at all
