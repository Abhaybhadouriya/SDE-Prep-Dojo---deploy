All four keep a forward and a reverse table and stop at the first contradiction. C++, Go and Java use 256-slot arrays; Python uses two dicts.

class Solution {
public:
    bool isIsomorphic(string s, string t) {
        int sToT[256], tToS[256];                 // -1 = character not mapped yet
        for (int c = 0; c < 256; c++) sToT[c] = tToS[c] = -1;
        for (size_t i = 0; i < s.size(); i++) {
            unsigned char a = s[i], b = t[i];     // unsigned: a char >= 128 must not go negative
            if (sToT[a] == -1 && tToS[b] == -1) { // both unseen: create the pair a <-> b
                sToT[a] = b;
                tToS[b] = a;
            } else if (sToT[a] != b || tToS[b] != a) {
                return false;                     // a already maps elsewhere, or b is taken by another char
            }
        }
        return true;
    }
};
func isIsomorphic(s string, t string) bool {
	var sToT, tToS [256]int // 0 = not mapped yet; otherwise the partner byte + 1
	for i := 0; i < len(s); i++ {
		a, b := s[i], t[i]
		if sToT[a] == 0 && tToS[b] == 0 { // both unseen: create the pair a <-> b
			sToT[a] = int(b) + 1
			tToS[b] = int(a) + 1
		} else if sToT[a] != int(b)+1 || tToS[b] != int(a)+1 {
			return false // a already maps elsewhere, or b is taken by another char
		}
	}
	return true
}

class Solution:
    def isIsomorphic(self, s: str, t: str) -> bool:
        s_to_t, t_to_s = {}, {}
        for a, b in zip(s, t):
            if a not in s_to_t and b not in t_to_s:   # both unseen: create the pair a <-> b
                s_to_t[a] = b
                t_to_s[b] = a
            elif s_to_t.get(a) != b or t_to_s.get(b) != a:
                return False                          # a already maps elsewhere, or b is taken
        return True
