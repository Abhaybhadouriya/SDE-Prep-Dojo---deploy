The O(1)-space rewiring in four languages. The test drivers in coding-code/flatten-binary-tree-to-linked-list/ build trees from level order, flatten them, and check that the chain is exactly the original preorder (same node objects) with every left pointer null: the examples, the empty tree, one node, an existing right chain, a left chain, and 2,000 random trees. The Python driver also checks the reverse-preorder version.

class Solution {
public:
    void flatten(TreeNode* root) {
        TreeNode* cur = root;
        while (cur != nullptr) {
            if (cur->left != nullptr) {
                // Hang the right subtree on the rightmost node of the left subtree:
                // it then follows the whole left subtree, so the preorder is unchanged.
                TreeNode* pre = cur->left;
                while (pre->right != nullptr) pre = pre->right;
                pre->right = cur->right;   // old right subtree now hangs off pre
                cur->right = cur->left;    // left subtree comes right after cur
                cur->left = nullptr;
            }
            cur = cur->right;              // next node in preorder
        }
    }
};
func flatten(root *TreeNode) {
	cur := root
	for cur != nil {
		if cur.Left != nil {
			// Hang the right subtree on the rightmost node of the left subtree:
			// it then follows the whole left subtree, so the preorder is unchanged.
			pre := cur.Left
			for pre.Right != nil {
				pre = pre.Right
			}
			pre.Right = cur.Right // old right subtree now hangs off pre
			cur.Right = cur.Left  // left subtree comes right after cur
			cur.Left = nil
		}
		cur = cur.Right // next node in preorder
	}
}
