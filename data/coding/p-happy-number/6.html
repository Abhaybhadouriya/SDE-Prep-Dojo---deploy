Floyd with a digit-square helper in four languages. Each file in coding-code/happy-number/ checks every n below 3,000 and 3,000 random values up to 231 - 1 against the hash-set version.

class Solution {
    static int step(int x) {               // sum of the squares of the digits of x
        int total = 0;                     // at most 10 digits * 81 = 810: no overflow
        while (x > 0) {
            int d = x % 10;
            total += d * d;
            x /= 10;
        }
        return total;
    }
public:
    bool isHappy(int n) {
        int slow = n, fast = step(n);
        while (fast != 1 && slow != fast) { // 1 is a fixed point (step(1) == 1)
            slow = step(slow);              // one hop
            fast = step(step(fast));        // two hops
        }
        return fast == 1;                   // met at 1 = happy, met elsewhere = loop
    }
};
func step(x int) int { // sum of the squares of the digits of x
	total := 0
	for x > 0 {
		d := x % 10
		total += d * d
		x /= 10
	}
	return total
}

class Solution:
    def isHappy(self, n: int) -> bool:
        def step(x: int) -> int:            # sum of the squares of the digits of x
            total = 0
            while x:
                x, d = divmod(x, 10)
                total += d * d
            return total

        slow, fast = n, step(n)
        while fast != 1 and slow != fast:   # 1 is a fixed point (step(1) == 1)
            slow = step(slow)               # one hop
            fast = step(step(fast))         # two hops
        return fast == 1                    # met at 1 = happy, met elsewhere = loop
