Template 1: sort, then merge

Merge Intervals: given a list of intervals, merge all overlapping ones and return the disjoint result. Sort by start. Keep the output list. For each interval, if it starts inside the last output block, stretch that block's end; otherwise start a new block.

  1. Sort by start. Now any interval that overlaps the current merged block must come right after it: no need to look back further.
  2. Take [1,3]. The output is empty, so it becomes the first merged block.
  3. [2,6] starts at 2, which is <= the last block's end (3). They overlap: extend the end to max(3, 6) = 6. Use max, because the next interval may be shorter.
  4. [8,10] starts after 6. Gap: close the block [1,6] and open [8,10].
  5. [15,18] also starts after the last end. Result: [1,6], [8,10], [15,18]. O(n log n) for the sort, O(n) for the scan.
import (
	"container/heap"
	"math"
	"sort"
)

func merge(iv [][]int) [][]int {
	sort.Slice(iv, func(i, j int) bool { return iv[i][0] < iv[j][0] })
	var out [][]int
	for _, x := range iv {
		if len(out) > 0 && x[0] <= out[len(out)-1][1] {
			if x[1] > out[len(out)-1][1] {
				out[len(out)-1][1] = x[1]
			}
		} else {
			out = append(out, []int{x[0], x[1]})
		}
	}
	return out
}
#include <bits/stdc++.h>         // C++14
using namespace std;
typedef vector<vector<int>> VV;

VV merge(VV iv) {
    sort(iv.begin(), iv.end());                         // by start (then end)
    VV out;
    for (auto& x : iv) {
        if (!out.empty() && x[0] <= out.back()[1]) out.back()[1] = max(out.back()[1], x[1]);
        else out.push_back(x);
    }
    return out;
}

static int[][] merge(int[][] iv) {
    Arrays.sort(iv, (a, b) -> Integer.compare(a[0], b[0]));
    List<int[]> out = new ArrayList<>();
    for (int[] x : iv) {
        if (!out.isEmpty() && x[0] <= out.get(out.size() - 1)[1])
            out.get(out.size() - 1)[1] = Math.max(out.get(out.size() - 1)[1], x[1]);
        else out.add(new int[]{x[0], x[1]});
    }
    return out.toArray(new int[0][]);
}
import heapq

def merge(iv):
    """Sort by start; extend the last merged interval while the next one starts inside it."""
    iv.sort(key=lambda x: x[0])
    out = []
    for s, e in iv:
        if out and s <= out[-1][1]:          # touching [1,4],[4,5] counts as overlap here
            out[-1][1] = max(out[-1][1], e)
        else:
            out.append([s, e])
    return out

Why it is enough to compare with the last block only: after the sort, every earlier block ends before the last block starts (or they would have been merged). The current interval starts at or after the last block's start, so it cannot reach back past it.

Summary Ranges is the same walk on a sorted array of integers ("consecutive numbers" means touching). Remove Covered Intervals sorts by start ascending and end descending, so a covering interval comes before what it covers.
