Template 2 and 3: insert, and intersect two lists

Insert Interval: the list is already sorted and has no overlaps. Add one new interval and merge where needed. You could append it and call merge (O(n log n)), but the order is already there, so one O(n) pass in three phases is the expected answer:

Interval List Intersections: two lists, each sorted and disjoint. Walk them with two pointers. The common part of the two current intervals is [max(starts), min(ends)] if that is not empty. Then advance the pointer whose interval ends first: it cannot meet anything else in the other list.

func insert(iv [][]int, nw []int) [][]int {
	out := [][]int{}
	i, n, s, e := 0, len(iv), nw[0], nw[1]
	for i < n && iv[i][1] < s { // left of new
		out = append(out, iv[i])
		i++
	}
	for i < n && iv[i][0] <= e { // overlaps new
		if iv[i][0] < s {
			s = iv[i][0]
		}
		if iv[i][1] > e {
			e = iv[i][1]
		}
		i++
	}
	out = append(out, []int{s, e})
	return append(out, iv[i:]...) // right of new
}

func intersect(a, b [][]int) [][]int {
	out := [][]int{}
	i, j := 0, 0
	for i < len(a) && j < len(b) {
		lo, hi := a[i][0], a[i][1]
		if b[j][0] > lo {
			lo = b[j][0]
		}
		if b[j][1] < hi {
			hi = b[j][1]
		}
		if lo <= hi {
			out = append(out, []int{lo, hi})
		}
		if a[i][1] < b[j][1] {
			i++
		} else {
			j++
		}
	}
	return out
}
VV insertInterval(const VV& iv, vector<int> nw) {
    VV out;
    size_t i = 0, n = iv.size();
    while (i < n && iv[i][1] < nw[0]) out.push_back(iv[i++]);            // left of new
    while (i < n && iv[i][0] <= nw[1]) {                                   // overlaps new
        nw[0] = min(nw[0], iv[i][0]); nw[1] = max(nw[1], iv[i][1]); i++;
    }
    out.push_back(nw);
    while (i < n) out.push_back(iv[i++]);                                  // right of new
    return out;
}

VV intersect(const VV& a, const VV& b) {
    VV out;
    size_t i = 0, j = 0;
    while (i < a.size() && j < b.size()) {
        int lo = max(a[i][0], b[j][0]), hi = min(a[i][1], b[j][1]);
        if (lo <= hi) out.push_back({lo, hi});
        if (a[i][1] < b[j][1]) i++; else j++;                              // drop the one that ends first
    }
    return out;
}
static int[][] insert(int[][] iv, int[] nw) {
    List<int[]> out = new ArrayList<>();
    int i = 0, n = iv.length, s = nw[0], e = nw[1];
    while (i < n && iv[i][1] < s) out.add(iv[i++]);                  // left of new
    while (i < n && iv[i][0] <= e) {                                 // overlaps new
        s = Math.min(s, iv[i][0]); e = Math.max(e, iv[i][1]); i++;
    }
    out.add(new int[]{s, e});
    while (i < n) out.add(iv[i++]);                                  // right of new
    return out.toArray(new int[0][]);
}

static int[][] intersect(int[][] a, int[][] b) {
    List<int[]> out = new ArrayList<>();
    int i = 0, j = 0;
    while (i < a.length && j < b.length) {
        int lo = Math.max(a[i][0], b[j][0]), hi = Math.min(a[i][1], b[j][1]);
        if (lo <= hi) out.add(new int[]{lo, hi});
        if (a[i][1] < b[j][1]) i++; else j++;
    }
    return out.toArray(new int[0][]);
}
def insert(iv, new):
    """Input is sorted and disjoint: copy the left part, absorb the overlaps, copy the right part."""
    out, i, n = [], 0, len(iv)
    s, e = new
    while i < n and iv[i][1] < s:            # ends before new starts
        out.append(iv[i]); i += 1
    while i < n and iv[i][0] <= e:           # overlaps new
        s, e = min(s, iv[i][0]), max(e, iv[i][1]); i += 1
    out.append([s, e])
    out.extend(iv[i:])
    return out

def intersect(a, b):
    """Two sorted disjoint lists: overlap = [max starts, min ends]; advance the one that ends first."""
    i = j = 0
    out = []
    while i < len(a) and j < len(b):
        lo, hi = max(a[i][0], b[j][0]), min(a[i][1], b[j][1])
        if lo <= hi:
            out.append([lo, hi])
        if a[i][1] < b[j][1]:
            i += 1
        else:
            j += 1
    return out

Meeting Scheduler (two people's free slots, find the first common slot of length d) is the intersect walk with a length check. Employee Free Time is the reverse: merge everyone's busy time (a k-way merge or a sort), then report the gaps.

From your world

Intersect is the merge step of two sorted lists, the same walk you would write to find common VLAN ranges allowed on both ends of a trunk, or the overlap of two ACL port ranges. Advancing the one that ends first is the rule that keeps the walk linear.
