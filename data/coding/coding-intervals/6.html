Template 6: minimum rooms (heap or sweep)

Meeting Rooms II: the fewest rooms so that no two overlapping meetings share one. That number equals the largest number of meetings running at the same moment. Two ways to get it, both O(n log n):

  1. Min-heap of end times. Sort by start. The heap holds the end time of every room in use. For each meeting, if the room that frees first (the heap top) is free by this start, reuse it (pop). Push this meeting's end. The heap size at the end is the answer.
  2. Sweep line. Sort all starts and all ends separately. Walk the starts; before counting a start, retire every end that is <= it. Track the largest count.
  1. Turn each meeting into two events: +1 at its start and -1 at its end. Sort events by time. Sweep left to right with a counter. t = 0: [0,30] starts, 1 room in use.
  2. t = 5: [5,10] starts while [0,30] is running. 2 rooms. New maximum.
  3. t = 10: two events share this time: [5,10] ends and [10,20] starts. Events are sorted by (time, delta) with -1 before +1, so the end goes first: back to 1 room, and that room is now free.
  4. t = 10, second event: [10,20] starts and reuses the room [5,10] just freed. Count 2, maximum stays 2. Had the start been processed first, the count would have touched 3 and the answer would be wrong: meetings are half-open, so [5,10] and [10,20] do not overlap.
  5. t = 20: it ends. Count 1.
  6. t = 30: the long meeting ends. The answer is the largest count ever seen: 2 rooms. The tie rule (end before start at the same time) is what kept it at 2 at t = 10.
type MinHeap []int
func (h MinHeap) Len() int            { return len(h) }
func (h MinHeap) Less(i, j int) bool  { return h[i] < h[j] }
func (h MinHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *MinHeap) Push(x interface{}) { *h = append(*h, x.(int)) }
func (h *MinHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

func minRoomsHeap(iv [][]int) int {
	sort.Slice(iv, func(i, j int) bool { return iv[i][0] < iv[j][0] })
	ends := &MinHeap{}
	for _, x := range iv {
		if ends.Len() > 0 && (*ends)[0] <= x[0] {
			heap.Pop(ends) // reuse the room that frees first
		}
		heap.Push(ends, x[1])
	}
	return ends.Len()
}

func minRoomsSweep(iv [][]int) int {
	n := len(iv)
	s, e := make([]int, n), make([]int, n)
	for i, x := range iv {
		s[i], e[i] = x[0], x[1]
	}
	sort.Ints(s)
	sort.Ints(e)
	rooms, best, j := 0, 0, 0
	for _, st := range s {
		for j < n && e[j] <= st { // ends first on ties
			rooms--
			j++
		}
		rooms++
		if rooms > best {
			best = rooms
		}
	}
	return best
}
int minRoomsHeap(VV iv) {
    sort(iv.begin(), iv.end());
    priority_queue<int, vector<int>, greater<int>> ends;                   // end times of rooms in use
    for (auto& x : iv) {
        if (!ends.empty() && ends.top() <= x[0]) ends.pop();               // reuse the room that frees first
        ends.push(x[1]);
    }
    return (int)ends.size();
}

int minRoomsSweep(const VV& iv) {
    vector<int> s, e;
    for (auto& x : iv) { s.push_back(x[0]); e.push_back(x[1]); }
    sort(s.begin(), s.end()); sort(e.begin(), e.end());
    int rooms = 0, best = 0;
    size_t j = 0;
    for (int st : s) {
        while (j < e.size() && e[j] <= st) { rooms--; j++; }               // ends first on ties
        best = max(best, ++rooms);
    }
    return best;
}
static int minRoomsHeap(int[][] iv) {
    Arrays.sort(iv, (a, b) -> Integer.compare(a[0], b[0]));
    PriorityQueue<Integer> ends = new PriorityQueue<>();
    for (int[] x : iv) {
        if (!ends.isEmpty() && ends.peek() <= x[0]) ends.poll();
        ends.offer(x[1]);
    }
    return ends.size();
}

static int minRoomsSweep(int[][] iv) {
    int n = iv.length;
    int[] s = new int[n], e = new int[n];
    for (int i = 0; i < n; i++) { s[i] = iv[i][0]; e[i] = iv[i][1]; }
    Arrays.sort(s); Arrays.sort(e);
    int rooms = 0, best = 0, j = 0;
    for (int st : s) {
        while (j < n && e[j] <= st) { rooms--; j++; }
        best = Math.max(best, ++rooms);
    }
    return best;
}
def min_rooms_heap(iv):
    """Min rooms: sort by start, min-heap of end times = rooms in use; reuse the one that frees first."""
    iv = sorted(iv)
    ends = []
    for s, e in iv:
        if ends and ends[0] <= s:
            heapq.heapreplace(ends, e)
        else:
            heapq.heappush(ends, e)
    return len(ends)

def min_rooms_sweep(iv):
    """Same answer with two sorted arrays: +1 at each start, -1 at each end, ends first on ties."""
    starts = sorted(s for s, _ in iv)
    ends = sorted(e for _, e in iv)
    rooms = best = j = 0
    for s in starts:
        while j < len(ends) and ends[j] <= s:
            rooms -= 1; j += 1
        rooms += 1
        best = max(best, rooms)
    return best

Why the heap version is correct: popping when top <= start reuses the room that frees first, which never hurts, and the heap only grows when every room is busy at that start, which means that many meetings really do overlap there. The sweep version counts those overlaps directly. If you also need which room each meeting gets, use the heap (and see Meeting Rooms III on Heaps, which adds a second heap of free room ids).
