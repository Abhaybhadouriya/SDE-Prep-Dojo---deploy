Paint + score in four languages, with an explicit stack so a 250,000-cell island cannot overflow the call stack. Each file in coding-code/making-a-large-island/ runs the examples and a 500 x 500 grid; C++ and Python also compare with the O(n4) brute force on random small grids.

func largestIsland(grid [][]int) int {
	n := len(grid)
	dr := [4]int{1, -1, 0, 0}
	dc := [4]int{0, 0, 1, -1}
	area := []int{0, 0} // area[id]; ids 0 (water) and 1 (unpainted land) are reserved
	best := 0

	// Pass 1: paint every island with its own id 2, 3, ... and record its area.
	// An explicit stack, because one island can hold 250,000 cells.
	for r := 0; r < n; r++ {
		for c := 0; c < n; c++ {
			if grid[r][c] != 1 {
				continue
			}
			id, size := len(area), 0
			grid[r][c] = id
			stack := [][2]int{{r, c}}
			for len(stack) > 0 {
				cur := stack[len(stack)-1]
				stack = stack[:len(stack)-1]
				size++
				for d := 0; d < 4; d++ {
					nr, nc := cur[0]+dr[d], cur[1]+dc[d]
					if nr >= 0 && nr < n && nc >= 0 && nc < n && grid[nr][nc] == 1 {
						grid[nr][nc] = id // paint on push, so no cell is pushed twice
						stack = append(stack, [2]int{nr, nc})
					}
				}
			}
			area = append(area, size)
			if size > best {
				best = size // the answer if there is no 0 to flip
			}
		}
	}

	// Pass 2: flip each 0 in turn; it joins the distinct islands among its 4 neighbours.
	for r := 0; r < n; r++ {
		for c := 0; c < n; c++ {
			if grid[r][c] != 0 {
				continue
			}
			var seen [4]int
			k, total := 0, 1 // 1 = the flipped cell itself
			for d := 0; d < 4; d++ {
				nr, nc := r+dr[d], c+dc[d]
				if nr < 0 || nr >= n || nc < 0 || nc >= n || grid[nr][nc] == 0 {
					continue
				}
				id := grid[nr][nc]
				dup := false
				for t := 0; t < k; t++ {
					if seen[t] == id { // same island from two sides
						dup = true
					}
				}
				if dup {
					continue
				}
				seen[k] = id
				k++
				total += area[id]
			}
			if total > best {
				best = total
			}
		}
	}
	return best
}
class Solution {
public:
    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        const int dr[4] = {1, -1, 0, 0}, dc[4] = {0, 0, 1, -1};
        vector<int> area = {0, 0};  // area[id]; ids 0 (water) and 1 (unpainted land) are reserved
        int best = 0;

        // Pass 1: paint every island with its own id 2, 3, ... and record its area.
        // An explicit stack, because one island can hold 250,000 cells (too deep for recursion).
        for (int r = 0; r < n; r++)
            for (int c = 0; c < n; c++) {
                if (grid[r][c] != 1) continue;
                int id = area.size(), size = 0;
                vector<pair<int, int>> stack(1, make_pair(r, c));
                grid[r][c] = id;
                while (!stack.empty()) {
                    pair<int, int> cur = stack.back();
                    stack.pop_back();
                    size++;
                    for (int d = 0; d < 4; d++) {
                        int nr = cur.first + dr[d], nc = cur.second + dc[d];
                        if (nr >= 0 && nr < n && nc >= 0 && nc < n && grid[nr][nc] == 1) {
                            grid[nr][nc] = id;  // paint on push, so no cell is pushed twice
                            stack.push_back(make_pair(nr, nc));
                        }
                    }
                }
                area.push_back(size);
                best = max(best, size);  // the answer if there is no 0 to flip
            }

        // Pass 2: flip each 0 in turn; it joins the distinct islands among its 4 neighbours.
        for (int r = 0; r < n; r++)
            for (int c = 0; c < n; c++) {
                if (grid[r][c] != 0) continue;
                int seen[4], k = 0, total = 1;  // 1 = the flipped cell itself
                for (int d = 0; d < 4; d++) {
                    int nr = r + dr[d], nc = c + dc[d];
                    if (nr < 0 || nr >= n || nc < 0 || nc >= n || grid[nr][nc] == 0) continue;
                    int id = grid[nr][nc];
                    if (find(seen, seen + k, id) != seen + k) continue;  // same island from two sides
                    seen[k++] = id;
                    total += area[id];
                }
                best = max(best, total);
            }
        return best;
    }
};

static class Solution {
    private static final int[] DR = {1, -1, 0, 0}, DC = {0, 0, 1, -1};

    public int largestIsland(int[][] grid) {
        int n = grid.length;
        List<Integer> area = new ArrayList<>(List.of(0, 0));  // area[id]; ids 0 and 1 are reserved
        int best = 0;

        // Pass 1: paint every island with its own id 2, 3, ... and record its area.
        // An explicit stack, because one island can hold 250,000 cells (too deep for recursion).
        int[] stack = new int[n * n];
        for (int r = 0; r < n; r++)
            for (int c = 0; c < n; c++) {
                if (grid[r][c] != 1) continue;
                int id = area.size(), size = 0, top = 0;
                stack[top++] = r * n + c;
                grid[r][c] = id;
                while (top > 0) {
                    int cell = stack[--top], cr = cell / n, cc = cell % n;
                    size++;
                    for (int d = 0; d < 4; d++) {
                        int nr = cr + DR[d], nc = cc + DC[d];
                        if (nr >= 0 && nr < n && nc >= 0 && nc < n && grid[nr][nc] == 1) {
                            grid[nr][nc] = id;  // paint on push, so no cell is pushed twice
                            stack[top++] = nr * n + nc;
                        }
                    }
                }
                area.add(size);
                best = Math.max(best, size);  // the answer if there is no 0 to flip
            }

        // Pass 2: flip each 0 in turn; it joins the distinct islands among its 4 neighbours.
        for (int r = 0; r < n; r++)
            for (int c = 0; c < n; c++) {
                if (grid[r][c] != 0) continue;
                int[] seen = new int[4];
                int k = 0, total = 1;  // 1 = the flipped cell itself
                for (int d = 0; d < 4; d++) {
                    int nr = r + DR[d], nc = c + DC[d];
                    if (nr < 0 || nr >= n || nc < 0 || nc >= n || grid[nr][nc] == 0) continue;
                    int id = grid[nr][nc];
                    boolean dup = false;
                    for (int t = 0; t < k; t++) if (seen[t] == id) dup = true;  // same island twice
                    if (dup) continue;
                    seen[k++] = id;
                    total += area.get(id);
                }
                best = Math.max(best, total);
            }
        return best;
    }
}
class Solution:
    def largestIsland(self, grid: List[List[int]]) -> int:
        n = len(grid)
        dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))
        area = [0, 0]  # area[id]; ids 0 (water) and 1 (unpainted land) are reserved
        best = 0
        # Pass 1: paint every island with its own id 2, 3, ... and record its area.
        # An explicit stack, because one island can hold 250,000 cells (too deep for recursion).
        for r in range(n):
            for c in range(n):
                if grid[r][c] != 1:
                    continue
                pid, size = len(area), 0
                grid[r][c] = pid
                stack = [(r, c)]
                while stack:
                    cr, cc = stack.pop()
                    size += 1
                    for dr, dc in dirs:
                        nr, nc = cr + dr, cc + dc
                        if 0 <= nr < n and 0 <= nc < n and grid[nr][nc] == 1:
                            grid[nr][nc] = pid  # paint on push, so no cell is pushed twice
                            stack.append((nr, nc))
                area.append(size)
                best = max(best, size)  # the answer if there is no 0 to flip
        # Pass 2: flip each 0 in turn; it joins the distinct islands among its 4 neighbours.
        for r in range(n):
            for c in range(n):
                if grid[r][c] == 0:
                    ids = {grid[r + dr][c + dc] for dr, dc in dirs
                           if 0 <= r + dr < n and 0 <= c + dc < n}  # a set: same island counted once
                    best = max(best, 1 + sum(area[i] for i in ids))  # area[0] == 0, so water adds nothing
        return best
