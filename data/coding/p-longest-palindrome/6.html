Count table, all pairs, plus one middle letter. Each file in coding-code/longest-palindrome/ also checks the result against the every-subset brute force on random short strings over a A b B c (mixed case on purpose).

class Solution {
public:
    int longestPalindrome(string s) {
        int count[128] = {0};                 // ASCII table: 'A' and 'a' are different letters
        for (char ch : s) count[(unsigned char)ch]++;
        int length = 0;
        for (int c : count) length += c / 2 * 2;   // every pair goes on both sides
        if (length < (int)s.size()) length++;      // a leftover letter can sit in the middle
        return length;
    }
};
func longestPalindrome(s string) int {
	var count [128]int // ASCII table: 'A'..'Z' and 'a'..'z' are different letters
	for i := 0; i < len(s); i++ {
		count[s[i]]++
	}
	length := 0
	for _, c := range count {
		length += c / 2 * 2 // every pair goes on both sides of the middle
	}
	if length < len(s) {
		length++ // some letter was left over: one of them can sit in the middle
	}
	return length
}

class Solution:
    def longestPalindrome(self, s: str) -> int:
        count = [0] * 128                    # ASCII table: 'A' and 'a' are different letters
        for ch in s:
            count[ord(ch)] += 1
        length = sum(c // 2 * 2 for c in count)   # every pair goes on both sides
        if length < len(s):                  # a leftover letter can sit in the middle
            length += 1
        return length
