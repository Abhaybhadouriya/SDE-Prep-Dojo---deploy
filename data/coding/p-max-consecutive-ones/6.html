The running-counter version in four languages. The test drivers in coding-code/max-consecutive-ones/ run the examples, an all-0 array, an all-1 array, a single element and 3,000 random arrays against the brute force.

class Solution {
public:
    int findMaxConsecutiveOnes(vector<int>& nums) {
        int best = 0;   // longest run of 1s seen so far
        int run = 0;    // length of the run that ends at the current index
        for (int x : nums) {
            if (x == 1) {
                run++;
                best = max(best, run);   // update inside the loop: the array may end on a run
            } else {
                run = 0;                 // a 0 breaks the run
            }
        }
        return best;
    }
};
func findMaxConsecutiveOnes(nums []int) int {
	best := 0 // longest run of 1s seen so far
	run := 0  // length of the run that ends at the current index
	for _, x := range nums {
		if x == 1 {
			run++
			if run > best { // update inside the loop: the array may end on a run
				best = run
			}
		} else {
			run = 0 // a 0 breaks the run
		}
	}
	return best
}
