Sort + sliding window in four languages. The test drivers (the examples, a single value, all equal, a 10^5-element overflow case, and 300 random lists checked against the brute force) are in coding-code/frequency-of-the-most-frequent-element/.

class Solution {
public:
    int maxFrequency(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end());   // the cheapest values to raise to nums[right] sit just left of it
        long long windowSum = 0;          // sum of nums[left..right]; can reach 1e10, so 64-bit
        int left = 0, best = 0;
        for (int right = 0; right < (int)nums.size(); right++) {
            windowSum += nums[right];
            // raising every value in the window to nums[right] costs target * size - sum
            while ((long long)nums[right] * (right - left + 1) - windowSum > k) {
                windowSum -= nums[left];      // too expensive: drop the smallest (farthest) value
                left++;
            }
            best = max(best, right - left + 1);  // the window is affordable here
        }
        return best;
    }
};
