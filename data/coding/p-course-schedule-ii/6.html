All four use Kahn's algorithm. The Go version uses the queue slice itself as the order, since every course is appended to it exactly once.

class Solution {
public:
    vector<int> findOrder(int numCourses, const vector<vector<int>>& prerequisites) {
        vector<vector<int>> nxt(numCourses);   // edge b -> a: finishing b unlocks a
        vector<int> indeg(numCourses, 0);      // unmet prerequisites per course
        for (const auto& p : prerequisites) {
            nxt[p[1]].push_back(p[0]);
            indeg[p[0]]++;
        }
        queue<int> ready;
        for (int c = 0; c < numCourses; ++c)
            if (indeg[c] == 0) ready.push(c);
        vector<int> order;
        order.reserve(numCourses);
        while (!ready.empty()) {
            int c = ready.front(); ready.pop();  // c has no unmet prerequisite left
            order.push_back(c);
            for (int d : nxt[c])
                if (--indeg[d] == 0) ready.push(d);  // d's last prerequisite finished
        }
        // courses on a cycle never reach indegree 0, so the order stays short
        if ((int)order.size() != numCourses) return {};
        return order;
    }
};
func findOrder(numCourses int, prerequisites [][]int) []int {
	nxt := make([][]int, numCourses)  // edge b -> a: finishing b unlocks a
	indeg := make([]int, numCourses) // unmet prerequisites per course
	for _, p := range prerequisites {
		a, b := p[0], p[1]
		nxt[b] = append(nxt[b], a)
		indeg[a]++
	}
	ready := make([]int, 0, numCourses) // FIFO queue: slice + head index
	for c := 0; c < numCourses; c++ {
		if indeg[c] == 0 {
			ready = append(ready, c)
		}
	}
	for head := 0; head < len(ready); head++ {
		c := ready[head] // c has no unmet prerequisite left
		for _, d := range nxt[c] {
			indeg[d]--
			if indeg[d] == 0 { // d's last prerequisite just finished
				ready = append(ready, d)
			}
		}
	}
	// the queue itself is the order; courses on a cycle never enter it
	if len(ready) != numCourses {
		return []int{}
	}
	return ready
}
static class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        List<List<Integer>> nxt = new ArrayList<>();      // edge b -> a: b unlocks a
        for (int i = 0; i < numCourses; i++) nxt.add(new ArrayList<>());
        int[] indeg = new int[numCourses];                // unmet prerequisites
        for (int[] p : prerequisites) {
            nxt.get(p[1]).add(p[0]);
            indeg[p[0]]++;
        }
        ArrayDeque<Integer> ready = new ArrayDeque<>();
        for (int c = 0; c < numCourses; c++) if (indeg[c] == 0) ready.add(c);
        int[] order = new int[numCourses];
        int taken = 0;
        while (!ready.isEmpty()) {
            int c = ready.poll();                         // no unmet prerequisite left
            order[taken++] = c;
            for (int d : nxt.get(c))
                if (--indeg[d] == 0) ready.add(d);        // d's last prerequisite done
        }
        // courses on a cycle never reach indegree 0
        return taken == numCourses ? order : new int[0];
    }
}
class Solution:
    def findOrder(self, num_courses: int, prerequisites: List[List[int]]) -> List[int]:
        nxt = [[] for _ in range(num_courses)]   # edge b -> a: finishing b unlocks a
        indeg = [0] * num_courses                # unmet prerequisites per course
        for a, b in prerequisites:
            nxt[b].append(a)
            indeg[a] += 1
        ready = deque(c for c in range(num_courses) if indeg[c] == 0)
        order = []
        while ready:
            c = ready.popleft()                  # c has no unmet prerequisite left
            order.append(c)
            for d in nxt[c]:
                indeg[d] -= 1
                if indeg[d] == 0:                # last prerequisite of d just finished
                    ready.append(d)
        # courses on a cycle never reach indegree 0, so they never enter the order
        return order if len(order) == num_courses else []
