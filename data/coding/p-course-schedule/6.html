class Solution {
public:
    bool canFinish(int numCourses, const vector<vector<int>>& prerequisites) {
        vector<vector<int>> after(numCourses);    // after[b] = courses unlocked by finishing b
        vector<int> waiting(numCourses, 0);       // in-degree: prerequisites still missing
        for (const auto& p : prerequisites) {     // [a, b]: take b before a, edge b -> a
            after[p[1]].push_back(p[0]);
            waiting[p[0]]++;
        }
        queue<int> ready;
        for (int c = 0; c < numCourses; c++)
            if (waiting[c] == 0) ready.push(c);
        int taken = 0;
        while (!ready.empty()) {
            int c = ready.front(); ready.pop();
            taken++;
            for (int nxt : after[c])
                if (--waiting[nxt] == 0) ready.push(nxt);   // last prerequisite done
        }
        return taken == numCourses;               // leftovers are on or behind a cycle
    }
};
func canFinish(numCourses int, prerequisites [][]int) bool {
	after := make([][]int, numCourses) // after[b] = courses unlocked by finishing b
	waiting := make([]int, numCourses) // in-degree: prerequisites still missing
	for _, p := range prerequisites {  // [a, b]: take b before a, edge b -> a
		after[p[1]] = append(after[p[1]], p[0])
		waiting[p[0]]++
	}
	ready := []int{}
	for c := 0; c < numCourses; c++ {
		if waiting[c] == 0 {
			ready = append(ready, c)
		}
	}
	taken := 0
	for len(ready) > 0 {
		c := ready[0]
		ready = ready[1:]
		taken++
		for _, nxt := range after[c] {
			waiting[nxt]--
			if waiting[nxt] == 0 { // last prerequisite done
				ready = append(ready, nxt)
			}
		}
	}
	return taken == numCourses // leftovers sit on or behind a cycle
}
static class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        List<List<Integer>> after = new ArrayList<>();   // after[b] = courses unlocked by b
        for (int i = 0; i < numCourses; i++) after.add(new ArrayList<>());
        int[] waiting = new int[numCourses];              // in-degree
        for (int[] p : prerequisites) {                   // [a, b]: edge b -> a
            after.get(p[1]).add(p[0]);
            waiting[p[0]]++;
        }
        ArrayDeque<Integer> ready = new ArrayDeque<>();
        for (int c = 0; c < numCourses; c++) if (waiting[c] == 0) ready.add(c);
        int taken = 0;
        while (!ready.isEmpty()) {
            int c = ready.poll();
            taken++;
            for (int nxt : after.get(c))
                if (--waiting[nxt] == 0) ready.add(nxt);  // last prerequisite done
        }
        return taken == numCourses;                       // leftovers sit on/behind a cycle
    }
}
class Solution:
    def canFinish(self, num_courses: int, prerequisites: List[List[int]]) -> bool:
        after = [[] for _ in range(num_courses)]   # after[b] = courses unlocked by finishing b
        waiting = [0] * num_courses                # waiting[a] = prerequisites a still needs (in-degree)
        for a, b in prerequisites:                 # pair [a, b]: take b before a, edge b -> a
            after[b].append(a)
            waiting[a] += 1

        ready = deque(c for c in range(num_courses) if waiting[c] == 0)
        taken = 0
        while ready:
            c = ready.popleft()
            taken += 1
            for nxt in after[c]:
                waiting[nxt] -= 1
                if waiting[nxt] == 0:              # its last prerequisite just finished
                    ready.append(nxt)
        return taken == num_courses                # courses never freed sit on (or behind) a cycle
