Sort + two loops + two pointers, with both pruning checks. Each file in coding-code/4sum/ runs the examples, the overflow traps and a short array; the C++ and Python files also compare against brute force on hundreds of random arrays full of duplicates.

class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> out;
        int n = nums.size();
        sort(nums.begin(), nums.end());
        for (int i = 0; i + 3 < n; i++) {
            if (i > 0 && nums[i] == nums[i - 1]) continue;              // same first value: same quadruplets
            // pruning: the 4 smallest from i are too big -> stop; i with the 3 largest too small -> next i
            if ((long long)nums[i] + nums[i + 1] + nums[i + 2] + nums[i + 3] > target) break;
            if ((long long)nums[i] + nums[n - 3] + nums[n - 2] + nums[n - 1] < target) continue;
            for (int j = i + 1; j + 2 < n; j++) {
                if (j > i + 1 && nums[j] == nums[j - 1]) continue;      // same second value in this slot
                if ((long long)nums[i] + nums[j] + nums[j + 1] + nums[j + 2] > target) break;
                if ((long long)nums[i] + nums[j] + nums[n - 2] + nums[n - 1] < target) continue;
                long long need = (long long)target - nums[i] - nums[j];  // up to 3e9: needs 64 bits
                int lo = j + 1, hi = n - 1;
                while (lo < hi) {
                    long long s = (long long)nums[lo] + nums[hi];
                    if (s < need) lo++;
                    else if (s > need) hi--;
                    else {
                        out.push_back({nums[i], nums[j], nums[lo], nums[hi]});
                        lo++, hi--;
                        while (lo < hi && nums[lo] == nums[lo - 1]) lo++;  // skip copies of the pair
                        while (lo < hi && nums[hi] == nums[hi + 1]) hi--;
                    }
                }
            }
        }
        return out;
    }
};
func fourSum(nums []int, target int) [][]int {
	// Go's int is 64-bit on every judge platform, so sums up to 4e9 do not overflow.
	sort.Ints(nums)
	n := len(nums)
	out := [][]int{}
	for i := 0; i+3 < n; i++ {
		if i > 0 && nums[i] == nums[i-1] {
			continue // same first value: same quadruplets
		}
		// pruning: the 4 smallest from i are too big -> stop; i with the 3 largest too small -> next i
		if nums[i]+nums[i+1]+nums[i+2]+nums[i+3] > target {
			break
		}
		if nums[i]+nums[n-3]+nums[n-2]+nums[n-1] < target {
			continue
		}
		for j := i + 1; j+2 < n; j++ {
			if j > i+1 && nums[j] == nums[j-1] {
				continue // same second value in this slot
			}
			if nums[i]+nums[j]+nums[j+1]+nums[j+2] > target {
				break
			}
			if nums[i]+nums[j]+nums[n-2]+nums[n-1] < target {
				continue
			}
			need := target - nums[i] - nums[j]
			lo, hi := j+1, n-1
			for lo < hi {
				s := nums[lo] + nums[hi]
				switch {
				case s < need:
					lo++
				case s > need:
					hi--
				default:
					out = append(out, []int{nums[i], nums[j], nums[lo], nums[hi]})
					lo, hi = lo+1, hi-1
					for lo < hi && nums[lo] == nums[lo-1] {
						lo++ // skip copies of the pair
					}
					for lo < hi && nums[hi] == nums[hi+1] {
						hi--
					}
				}
			}
		}
	}
	return out
}

class Solution:
    def fourSum(self, nums: List[int], target: int) -> List[List[int]]:
        nums.sort()
        n, out = len(nums), []
        for i in range(n - 3):
            if i > 0 and nums[i] == nums[i - 1]:
                continue  # same first value: same quadruplets
            # pruning: the 4 smallest from i are too big -> stop; i with the 3 largest too small -> next i
            if nums[i] + nums[i + 1] + nums[i + 2] + nums[i + 3] > target:
                break
            if nums[i] + nums[-3] + nums[-2] + nums[-1] < target:
                continue
            for j in range(i + 1, n - 2):
                if j > i + 1 and nums[j] == nums[j - 1]:
                    continue  # same second value in this slot
                if nums[i] + nums[j] + nums[j + 1] + nums[j + 2] > target:
                    break
                if nums[i] + nums[j] + nums[-2] + nums[-1] < target:
                    continue
                need = target - nums[i] - nums[j]  # Python ints never overflow; C++/Java use 64 bits here
                lo, hi = j + 1, n - 1
                while lo < hi:
                    s = nums[lo] + nums[hi]
                    if s < need:
                        lo += 1
                    elif s > need:
                        hi -= 1
                    else:
                        out.append([nums[i], nums[j], nums[lo], nums[hi]])
                        lo, hi = lo + 1, hi - 1
                        while lo < hi and nums[lo] == nums[lo - 1]:
                            lo += 1  # skip copies of the pair
                        while lo < hi and nums[hi] == nums[hi + 1]:
                            hi -= 1
        return out
