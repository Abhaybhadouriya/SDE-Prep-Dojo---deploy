The size-k min-heap in four languages. Go needs a small container/heap adapter; it replaces the root and calls heap.Fix instead of pop + push. Each file in coding-code/kth-largest-element-in-a-stream/ runs the three examples and 500 random streams checked against sorting.

class KthLargest {
    int k;
    priority_queue<int, vector<int>, greater<int>> top;   // min-heap of the k largest so far
public:
    KthLargest(int k, vector<int>& nums) : k(k) {
        for (int x : nums) add(x);                         // reuse add(): O(n log k)
    }
    int add(int val) {
        if ((int)top.size() < k) top.push(val);            // not full yet (nums may be short)
        else if (val > top.top()) {                        // beats the weakest of the k
            top.pop();
            top.push(val);
        }
        return top.top();                                  // smallest of the k largest
    }
};
// minHeap implements heap.Interface over ints (smallest on top).
type minHeap []int

func (h minHeap) Len() int           { return len(h) }
func (h minHeap) Less(i, j int) bool { return h[i] < h[j] }
func (h minHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *minHeap) Pop() any {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

type KthLargest struct {
	k   int
	top *minHeap // the k largest values seen so far
}

func Constructor(k int, nums []int) KthLargest {
	kl := KthLargest{k: k, top: &minHeap{}}
	for _, x := range nums {
		kl.Add(x) // reuse Add: O(n log k)
	}
	return kl
}

func (kl *KthLargest) Add(val int) int {
	if kl.top.Len() < kl.k { // not full yet (nums may be short)
		heap.Push(kl.top, val)
	} else if val > (*kl.top)[0] { // beats the weakest of the k
		(*kl.top)[0] = val
		heap.Fix(kl.top, 0) // sift the new value down: O(log k)
	}
	return (*kl.top)[0] // smallest of the k largest
}

static class KthLargest {
    private final int k;
    private final PriorityQueue<Integer> top;   // min-heap of the k largest so far

    public KthLargest(int k, int[] nums) {
        this.k = k;
        this.top = new PriorityQueue<>(k + 1);
        for (int x : nums) add(x);               // reuse add(): O(n log k)
    }

    public int add(int val) {
        if (top.size() < k) top.offer(val);      // not full yet (nums may be short)
        else if (val > top.peek()) {             // beats the weakest of the k
            top.poll();
            top.offer(val);
        }
        return top.peek();                       // smallest of the k largest
    }
}
class KthLargest:
    def __init__(self, k: int, nums: List[int]):
        self.k = k
        self.top = list(nums)          # min-heap of the k largest values seen so far
        heapq.heapify(self.top)        # O(n)
        while len(self.top) > k:       # drop the smallest until only k remain
            heapq.heappop(self.top)

    def add(self, val: int) -> int:
        if len(self.top) < self.k:     # the start can hold k-1 values (nums may be short)
            heapq.heappush(self.top, val)
        elif val > self.top[0]:        # val beats the weakest of the k: swap it in
            heapq.heapreplace(self.top, val)
        return self.top[0]             # smallest of the k largest = kth largest
