func subsetsWithDup(nums []int) [][]int {
	sort.Ints(nums) // equal values become neighbours
	out := [][]int{}
	cur := []int{}
	var build func(start int)
	build = func(start int) {
		// every node of the tree is one subset; copy, because cur keeps changing
		out = append(out, append([]int{}, cur...))
		for i := start; i < len(nums); i++ {
			// same value already tried at THIS position -> same subtree, skip it
			if i > start && nums[i] == nums[i-1] {
				continue
			}
			cur = append(cur, nums[i]) // choose
			build(i + 1)               // explore with later elements only
			cur = cur[:len(cur)-1]     // un-choose
		}
	}
	build(0)
	return out
}
class Solution {
public:
    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        sort(nums.begin(), nums.end());       // equal values become neighbours
        vector<vector<int>> out;
        vector<int> cur;
        build(nums, 0, cur, out);
        return out;
    }

private:
    void build(const vector<int>& nums, int start, vector<int>& cur, vector<vector<int>>& out) {
        out.push_back(cur);                   // every node of the tree is one subset
        for (int i = start; i < (int)nums.size(); i++) {
            // same value already tried at THIS position -> same subtree, skip it
            if (i > start && nums[i] == nums[i - 1]) continue;
            cur.push_back(nums[i]);           // choose
            build(nums, i + 1, cur, out);     // explore with later elements only
            cur.pop_back();                   // un-choose
        }
    }
};

static class Solution {
    public List<List<Integer>> subsetsWithDup(int[] nums) {
        Arrays.sort(nums);                    // equal values become neighbours
        List<List<Integer>> out = new ArrayList<>();
        build(nums, 0, new ArrayList<>(), out);
        return out;
    }

    private void build(int[] nums, int start, List<Integer> cur, List<List<Integer>> out) {
        out.add(new ArrayList<>(cur));        // every node of the tree is one subset
        for (int i = start; i < nums.length; i++) {
            // same value already tried at THIS position -> same subtree, skip it
            if (i > start && nums[i] == nums[i - 1]) continue;
            cur.add(nums[i]);                 // choose
            build(nums, i + 1, cur, out);     // explore with later elements only
            cur.remove(cur.size() - 1);       // un-choose
        }
    }
}
class Solution:
    def subsetsWithDup(self, nums: List[int]) -> List[List[int]]:
        nums = sorted(nums)                 # equal values become neighbours
        out, cur = [], []

        def build(start: int) -> None:
            out.append(cur[:])              # every node of the tree is one subset
            for i in range(start, len(nums)):
                # same value already tried at THIS position -> same subtree, skip it
                if i > start and nums[i] == nums[i - 1]:
                    continue
                cur.append(nums[i])         # choose
                build(i + 1)                # explore with later elements only
                cur.pop()                   # un-choose

        build(0)
        return out
