class Solution {
    // Grow s[lo..hi] while both ends match; return the length of the palindrome found.
    int expand(const string& s, int lo, int hi) {
        while (lo >= 0 && hi < (int)s.size() && s[lo] == s[hi]) { lo--; hi++; }
        return hi - lo - 1;               // the loop stops one step past each end
    }
public:
    string longestPalindrome(string s) {
        int start = 0, best = 1;          // s is non-empty, so one char is a palindrome
        for (int c = 0; c < (int)s.size(); c++) {
            int odd = expand(s, c, c);        // centre on a char: "aba"
            int even = expand(s, c, c + 1);   // centre between chars: "abba"
            int len = odd > even ? odd : even;
            if (len > best) {
                best = len;
                start = c - (len - 1) / 2;    // works for both odd and even lengths
            }
        }
        return s.substr(start, best);
    }
};

func expand(s string, lo, hi int) int {
	for lo >= 0 && hi < len(s) && s[lo] == s[hi] {
		lo--
		hi++
	}
	return hi - lo - 1 // the loop stops one step past each end
}

		}

class Solution:
    def longestPalindrome(self, s: str) -> str:
        def expand(lo: int, hi: int) -> int:
            # Grow s[lo..hi] while both ends match; return the palindrome's length.
            while lo >= 0 and hi < len(s) and s[lo] == s[hi]:
                lo -= 1
                hi += 1
            return hi - lo - 1              # the loop stops one step past each end

        start, best = 0, 1                  # s is non-empty, so one char is a palindrome
        for c in range(len(s)):
            odd = expand(c, c)              # centre on a char: "aba"
            even = expand(c, c + 1)         # centre between chars: "abba"
            length = max(odd, even)
            if length > best:
                best = length
                start = c - (length - 1) // 2   # works for both odd and even lengths
        return s[start:start + best]
