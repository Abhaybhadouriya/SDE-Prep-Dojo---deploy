class Solution {
public:
    int myAtoi(string s) {
        int i = 0, n = s.size();
        while (i < n && s[i] == ' ') i++;          // 1. skip leading spaces (only ' ')
        int sign = 1;
        if (i < n && (s[i] == '+' || s[i] == '-')) // 2. at most one sign
            sign = (s[i++] == '-') ? -1 : 1;
        int value = 0;                             // stays in [0, INT_MAX]: never overflows
        while (i < n && s[i] >= '0' && s[i] <= '9') {   // 3. digits until a non-digit
            int d = s[i] - '0';
            if (value > (INT_MAX - d) / 10)        // value*10 + d would pass INT_MAX: clamp now
                return sign == 1 ? INT_MAX : INT_MIN;
            value = value * 10 + d;
            i++;
        }
        return sign * value;                       // 4. no digits read -> 0
    }
};
func myAtoi(s string) int {
	i, n := 0, len(s)
	for i < n && s[i] == ' ' { // 1. skip leading spaces (only ' ')
		i++
	}
	sign := 1
	if i < n && (s[i] == '+' || s[i] == '-') { // 2. at most one sign
		if s[i] == '-' {
			sign = -1
		}
		i++
	}
	value := 0 // kept in [0, MaxInt32], the same check works for a 32-bit int
	for i < n && s[i] >= '0' && s[i] <= '9' { // 3. digits until a non-digit
		d := int(s[i] - '0')
		if value > (math.MaxInt32-d)/10 { // value*10 + d would pass MaxInt32: clamp now
			if sign == 1 {
				return math.MaxInt32
			}
			return math.MinInt32
		}
		value = value*10 + d
		i++
	}
	return sign * value // 4. no digits read -> 0
}

class Solution:
    def myAtoi(self, s: str) -> int:
        INT_MAX, INT_MIN = 2**31 - 1, -2**31
        i, n = 0, len(s)
        while i < n and s[i] == ' ':          # 1. skip leading spaces (only ' ')
            i += 1
        sign = 1
        if i < n and s[i] in '+-':            # 2. at most one sign
            sign = -1 if s[i] == '-' else 1
            i += 1
        value = 0
        while i < n and '0' <= s[i] <= '9':   # 3. digits until the first non-digit
            d = ord(s[i]) - ord('0')

                return INT_MAX if sign == 1 else INT_MIN

            i += 1
