func canCompleteCircuit(gas []int, cost []int) int {
	total := 0 // sum of all (gas - cost): decides IF a start exists
	tank := 0  // fuel left since the current candidate start
	start := 0 // current candidate start
	for i := range gas {
		net := gas[i] - cost[i]
		total += net
		tank += net
		if tank < 0 { // cannot reach i + 1 from start: no station in [start, i] works
			start = i + 1 // try the next station with an empty tank
			tank = 0
		}
	}
	if total < 0 {
		return -1
	}
	return start
}

        int tank = 0;    // fuel left since the current candidate start

                start = i + 1;   // try the next station with an empty tank
                tank = 0;
            }
        }
        return total >= 0 ? start : -1;
    }
};
