class Solution {
public:
    bool containsDuplicate(const vector<int>& nums) {
        unordered_set<int> seen;                 // values met so far
        seen.reserve(nums.size() * 2);           // no rehashing during the pass
        for (int x : nums) {
            if (!seen.insert(x).second) return true;  // insert failed: x was already there
        }
        return false;                            // every value was new
    }
};

func containsDuplicate(nums []int) bool {
	seen := make(map[int]struct{}, len(nums)) // a set: values met so far
	for _, x := range nums {
		if _, ok := seen[x]; ok {
			return true // second time we meet x
		}
		seen[x] = struct{}{}
	}
	return false // every value was new
}

static class Solution {
    public boolean containsDuplicate(int[] nums) {
        Set<Integer> seen = new HashSet<>(nums.length * 2);   // values met so far
        for (int x : nums) {
            if (!seen.add(x)) return true;    // add() returns false when x was already there
        }
        return false;                         // every value was new
    }
}
