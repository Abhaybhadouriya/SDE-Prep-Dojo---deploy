The templates (Go, C++, Java, Python)

One file per language holds all six moves. Each was run against the LeetCode examples plus edge cases (empty list, one node, removing the head, a self-loop, a k that divides the length and one that does not). All four files (Go 1.27, C++14, Java 21, Python 3.11) were compiled and run and pass these checks.

What is in each file (same names in every language; Go and Java use camelCase):

  • ListNode - the singly linked list node.
  • reverse_list - LC 206: in-place reversal with prev and curr.
  • middle_node - Second middle for even length.
  • detect_cycle - Floyd: node where the cycle starts, or None.
  • remove_nth_from_end - LC 19: dummy head plus a gap of n between two pointers.
  • merge_two_lists - LC 21: merge two sorted lists behind a dummy head.
  • reverse_k_group - LC 25: reverse every block of k nodes, leave a short tail as it is.
  • Node - node with next and random pointers.
  • copy_random_list - O(n) time, O(1) extra: weave copies in, set random, unweave.

type ListNode struct {
	Val  int
	Next *ListNode
}

func reverseList(head *ListNode) *ListNode {
	var prev *ListNode
	curr := head
	for curr != nil {
		next := curr.Next // 1. save the rest of the list
		curr.Next = prev  // 2. flip one arrow
		prev = curr       // 3. step both pointers forward
		curr = next
	}
	return prev
}

// middleNode returns the second middle for even length.
func middleNode(head *ListNode) *ListNode {
	slow, fast := head, head
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
	}
	return slow
}

// detectCycle (Floyd) returns the node where the cycle starts, or nil.
func detectCycle(head *ListNode) *ListNode {
	slow, fast := head, head
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
		if slow == fast { // phase 1: met inside the loop
			p := head // phase 2: both at speed 1
			for p != slow {
				p = p.Next
				slow = slow.Next
			}
			return p
		}
	}
	return nil
}

func removeNthFromEnd(head *ListNode, n int) *ListNode {
	dummy := &ListNode{Next: head}
	fast, slow := dummy, dummy
	for i := 0; i < n; i++ { // open a gap of n nodes
		fast = fast.Next
	}
	for fast.Next != nil {
		fast = fast.Next
		slow = slow.Next
	}
	slow.Next = slow.Next.Next // slow is just BEFORE the victim
	return dummy.Next
}

func mergeTwoLists(a, b *ListNode) *ListNode {
	dummy := &ListNode{}
	tail := dummy
	for a != nil && b != nil {
		if a.Val <= b.Val {
			tail.Next, a = a, a.Next
		} else {
			tail.Next, b = b, b.Next
		}
		tail = tail.Next
	}
	if a != nil {
		tail.Next = a
	} else {
		tail.Next = b
	}
	return dummy.Next
}

func reverseKGroup(head *ListNode, k int) *ListNode {
	dummy := &ListNode{Next: head}
	groupPrev := dummy
	for {
		kth := groupPrev
		for i := 0; i < k && kth != nil; i++ {
			kth = kth.Next
		}
		if kth == nil { // fewer than k left
			break
		}
		groupNext := kth.Next
		prev, curr := groupNext, groupPrev.Next
		for curr != groupNext {
			next := curr.Next
			curr.Next = prev
			prev = curr
			curr = next
		}
		first := groupPrev.Next // old first = new last of the block
		groupPrev.Next = kth
		groupPrev = first
	}
	return dummy.Next
}

type Node struct {
	Val    int
	Next   *Node
	Random *Node
}

// copyRandomList: O(n) time, O(1) extra. Weave copies in, set Random, unweave.
func copyRandomList(head *Node) *Node {
	if head == nil {
		return nil
	}
	for p := head; p != nil; p = p.Next.Next {
		p.Next = &Node{Val: p.Val, Next: p.Next}
	}
	for p := head; p != nil; p = p.Next.Next {
		if p.Random != nil {
			p.Next.Random = p.Random.Next
		}
	}
	copyHead := head.Next
	for p := head; p != nil; p = p.Next {
		c := p.Next
		p.Next = c.Next
		if c.Next != nil {
			c.Next = c.Next.Next
		}
	}
	return copyHead
}
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int v, ListNode* n = nullptr) : val(v), next(n) {}
};

// Reverse the whole list with three pointers. O(n) time, O(1) space.
ListNode* reverseList(ListNode* head) {
    ListNode* prev = nullptr;
    ListNode* curr = head;
    while (curr) {
        ListNode* next = curr->next;  // 1. save the rest of the list
        curr->next = prev;            // 2. flip one arrow
        prev = curr;                  // 3. step both pointers forward
        curr = next;
    }
    return prev;                      // prev is the new head
}

// Middle node; for even length returns the SECOND middle.
ListNode* middleNode(ListNode* head) {
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast && fast->next) {      // check fast first, then fast->next
        slow = slow->next;
        fast = fast->next->next;
    }
    return slow;
}

// Floyd: returns the node where the cycle starts, or nullptr.
ListNode* detectCycle(ListNode* head) {
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {           // phase 1: they met inside the loop
            ListNode* p = head;       // phase 2: one from head, one from meeting point
            while (p != slow) { p = p->next; slow = slow->next; }
            return p;
        }
    }
    return nullptr;
}

// Remove the n-th node from the end (1-based). Dummy head + gap of n.
ListNode* removeNthFromEnd(ListNode* head, int n) {
    ListNode dummy(0, head);
    ListNode* fast = &dummy;
    ListNode* slow = &dummy;
    for (int i = 0; i < n; ++i) fast = fast->next;   // open a gap of n nodes
    while (fast->next) { fast = fast->next; slow = slow->next; }
    ListNode* gone = slow->next;      // slow stops just BEFORE the victim
    slow->next = gone->next;
    delete gone;
    return dummy.next;                // works even when the head was removed
}

// Merge two sorted lists by splicing nodes (no new nodes).
ListNode* mergeTwoLists(ListNode* a, ListNode* b) {
    ListNode dummy(0);
    ListNode* tail = &dummy;
    while (a && b) {
        if (a->val <= b->val) { tail->next = a; a = a->next; }
        else                  { tail->next = b; b = b->next; }
        tail = tail->next;
    }
    tail->next = a ? a : b;           // attach whatever is left
    return dummy.next;
}

// Reverse every block of k nodes; a short last block stays as it is.
ListNode* reverseKGroup(ListNode* head, int k) {
    ListNode dummy(0, head);
    ListNode* groupPrev = &dummy;     // node just before the current block
    while (true) {
        ListNode* kth = groupPrev;
        for (int i = 0; i < k && kth; ++i) kth = kth->next;
        if (!kth) break;              // fewer than k nodes left
        ListNode* groupNext = kth->next;
        ListNode* prev = groupNext;   // the block's new tail must point onward
        ListNode* curr = groupPrev->next;
        while (curr != groupNext) {
            ListNode* next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        ListNode* first = groupPrev->next;  // old first = new last of the block
        groupPrev->next = kth;               // kth is the block's new first
        groupPrev = first;
    }
    return dummy.next;
}

// Deep copy of a list whose nodes also have a random pointer.
struct Node {
    int val;
    Node* next;
    Node* random;
    Node(int v) : val(v), next(nullptr), random(nullptr) {}
};

// O(n) time, O(1) extra space: weave copies in, set random, unweave.
Node* copyRandomList(Node* head) {
    if (!head) return nullptr;
    for (Node* p = head; p; p = p->next->next) {      // A -> A' -> B -> B' ...
        Node* c = new Node(p->val);
        c->next = p->next;
        p->next = c;
    }
    for (Node* p = head; p; p = p->next->next)        // copy of X.random is X.random->next
        p->next->random = p->random ? p->random->next : nullptr;
    Node* copyHead = head->next;
    for (Node* p = head; p; p = p->next) {            // restore the original, link the copies
        Node* c = p->next;
        p->next = c->next;
        c->next = c->next ? c->next->next : nullptr;
    }
    return copyHead;
}

static ListNode reverseList(ListNode head) {
    ListNode prev = null, curr = head;
    while (curr != null) {
        ListNode next = curr.next;   // 1. save the rest of the list
        curr.next = prev;            // 2. flip one arrow
        prev = curr;                 // 3. step both pointers forward
        curr = next;
    }
    return prev;
}

static ListNode reverseKGroup(ListNode head, int k) {

        for (int i = 0; i < k && kth != null; i++) kth = kth.next;

            ListNode next = curr.next;
            curr.next = prev;
            prev = curr;
            curr = next;

        ListNode first = groupPrev.next;          // old first = new last
        groupPrev.next = kth;
        groupPrev = first;
    }
    return dummy.next;

static class Node {
    int val;
    Node next, random;
    Node(int val) { this.val = val; }
}

// O(n) time, O(1) extra: weave copies in, set random, unweave.
static Node copyRandomList(Node head) {
    if (head == null) return null;
    for (Node p = head; p != null; p = p.next.next) {
        Node c = new Node(p.val);
        c.next = p.next;
        p.next = c;
    }
    for (Node p = head; p != null; p = p.next.next)
        p.next.random = (p.random != null) ? p.random.next : null;
    Node copyHead = head.next;
    for (Node p = head; p != null; p = p.next) {
        Node c = p.next;
        p.next = c.next;
        c.next = (c.next != null) ? c.next.next : null;
    }
    return copyHead;
}
class ListNode:
    def __init__(self, val: int = 0, next: Optional["ListNode"] = None):
        self.val = val
        self.next = next


def reverse_list(head: Optional[ListNode]) -> Optional[ListNode]:
    prev, curr = None, head
    while curr:
        nxt = curr.next        # 1. save the rest of the list
        curr.next = prev       # 2. flip one arrow
        prev, curr = curr, nxt # 3. step both pointers forward
    return prev                # new head


def middle_node(head: Optional[ListNode]) -> Optional[ListNode]:
    """Second middle for even length."""
    slow = fast = head
    while fast and fast.next:
        slow, fast = slow.next, fast.next.next
    return slow


def detect_cycle(head: Optional[ListNode]) -> Optional[ListNode]:
    """Floyd: node where the cycle starts, or None."""
    slow = fast = head
    while fast and fast.next:
        slow, fast = slow.next, fast.next.next
        if slow is fast:                 # phase 1: met inside the loop
            p = head                     # phase 2: walk both at speed 1
            while p is not slow:
                p, slow = p.next, slow.next
            return p
    return None


def remove_nth_from_end(head: Optional[ListNode], n: int) -> Optional[ListNode]:
    dummy = ListNode(0, head)
    fast = slow = dummy
    for _ in range(n):                   # open a gap of n nodes
        fast = fast.next
    while fast.next:
        fast, slow = fast.next, slow.next
    slow.next = slow.next.next           # slow stops just BEFORE the victim
    return dummy.next


def merge_two_lists(a: Optional[ListNode], b: Optional[ListNode]) -> Optional[ListNode]:
    dummy = tail = ListNode()
    while a and b:
        if a.val <= b.val:
            tail.next, a = a, a.next
        else:
            tail.next, b = b, b.next
        tail = tail.next
    tail.next = a or b                   # attach the leftover
    return dummy.next


def reverse_k_group(head: Optional[ListNode], k: int) -> Optional[ListNode]:
    dummy = ListNode(0, head)
    group_prev = dummy
    while True:
        kth = group_prev
        for _ in range(k):
            kth = kth.next
            if kth is None:
                return dummy.next        # fewer than k left: keep as is
        group_next = kth.next
        prev, curr = group_next, group_prev.next
        while curr is not group_next:
            nxt = curr.next
            curr.next = prev
            prev, curr = curr, nxt
        first = group_prev.next          # old first = new last of the block
        group_prev.next = kth
        group_prev = first


class Node:
    def __init__(self, val: int, next: "Node" = None, random: "Node" = None):
        self.val = val
        self.next = next
        self.random = random


def copy_random_list(head: Optional[Node]) -> Optional[Node]:
    """O(n) time, O(1) extra: weave copies in, set random, unweave."""
    if head is None:
        return None
    p = head
    while p:                             # A -> A' -> B -> B' ...
        p.next = Node(p.val, p.next)
        p = p.next.next
    p = head
    while p:                             # copy of X.random is X.random.next
        p.next.random = p.random.next if p.random else None
        p = p.next.next
    copy_head, p = head.next, head
    while p:                             # restore the original, link the copies
        c = p.next
        p.next = c.next
        c.next = c.next.next if c.next else None
        p = p.next
    return copy_head

Remember this

  • Loop guard for two-step moves: fast != null && fast.next != null, in that order.
  • Reversal order: save next, flip, move prev, move curr. Return prev.
  • Head may change? Dummy head, return dummy.next.
  • Gap of n between two pointers starting at the dummy leaves slow just before the n-th from the end.
