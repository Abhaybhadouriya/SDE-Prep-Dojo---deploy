class Solution {
public:
    int maximalSquare(vector<vector<char>>& matrix) {
        int cols = matrix[0].size();
        vector<int> side(cols + 1, 0);  // side[c + 1] = biggest square ending at (row, c)
        int best = 0;
        for (const auto& row : matrix) {
            int diag = 0;               // previous row's side[c]: the up-left neighbour
            for (int c = 0; c < cols; ++c) {
                int up = side[c + 1];   // still the previous row's value
                if (row[c] == '1') {
                    side[c + 1] = 1 + min({up, side[c], diag});  // up, left, up-left
                    best = max(best, side[c + 1]);
                } else {
                    side[c + 1] = 0;
                }
                diag = up;              // up-left for the next column
            }
        }
        return best * best;             // area, not side
    }
};

    def maximalSquare(self, matrix: List[List[str]]) -> int:

        side = [0] * (cols + 1)   # side[c + 1] = largest all-1 square with bottom-right at (row, c)

                    best = max(best, side[c + 1])
                else:
                    side[c + 1] = 0
                diag = up         # becomes the up-left neighbour of the next column
        return best * best        # the question asks for the area, not the side
