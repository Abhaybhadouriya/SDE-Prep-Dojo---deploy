func partition(s string) [][]string {
	n := len(s)
	// pal[i][j]: s[i..j] reads the same both ways. i goes down so the inside pal[i+1][j-1] is ready.
	pal := make([][]bool, n)
	for i := range pal {
		pal[i] = make([]bool, n)
	}
	for i := n - 1; i >= 0; i-- {
		for j := i; j < n; j++ {
			pal[i][j] = s[i] == s[j] && (j-i < 2 || pal[i+1][j-1])
		}
	}
	var out [][]string
	var path []string // the pieces chosen so far
	var explore func(start int)
	explore = func(start int) {
		if start == n { // the pieces cover s: one answer
			out = append(out, append([]string(nil), path...)) // copy, path keeps changing
			return
		}
		for end := start; end < n; end++ {
			if !pal[start][end] { // only a palindrome may be the next piece
				continue
			}
			path = append(path, s[start:end+1]) // choose
			explore(end + 1)                    // explore the rest
			path = path[:len(path)-1]           // undo
		}
	}
	explore(0)
	return out
}
class Solution {
    string s;
    int n = 0;
    vector<vector<char>> pal;     // pal[i][j]: s[i..j] reads the same both ways
    vector<string> path;          // the pieces chosen so far
    vector<vector<string>> out;

    void explore(int start) {
        if (start == n) { out.push_back(path); return; }   // the pieces cover s: one answer
        for (int end = start; end < n; end++) {
            if (!pal[start][end]) continue;                // only a palindrome may be the next piece
            path.push_back(s.substr(start, end - start + 1));  // choose
            explore(end + 1);                                  // explore the rest
            path.pop_back();                                   // undo
        }
    }

static class Solution {
    private String s;
    private boolean[][] pal;                       // pal[i][j]: s[i..j] reads the same both ways
    private final List<String> path = new ArrayList<>();  // the pieces chosen so far
    private List<List<String>> out;

    public List<List<String>> partition(String str) {
        s = str;
        int n = s.length();
        pal = new boolean[n][n];
        // i goes down so pal[i + 1][j - 1] (the inside) is ready before pal[i][j].
        for (int i = n - 1; i >= 0; i--)
            for (int j = i; j < n; j++)
                pal[i][j] = s.charAt(i) == s.charAt(j) && (j - i < 2 || pal[i + 1][j - 1]);
        out = new ArrayList<>();
        path.clear();
        explore(0);
        return out;
    }

    private void explore(int start) {
        if (start == s.length()) { out.add(new ArrayList<>(path)); return; }  // copy, path keeps changing
        for (int end = start; end < s.length(); end++) {
            if (!pal[start][end]) continue;               // only a palindrome may be the next piece
            path.add(s.substring(start, end + 1));        // choose
            explore(end + 1);                             // explore the rest
            path.remove(path.size() - 1);                 // undo
        }
    }
}
class Solution:
    def partition(self, s: str) -> List[List[str]]:
        n = len(s)
        # pal[i][j]: s[i..j] reads the same both ways. i goes down so the inside pal[i+1][j-1] is ready.
        pal = [[False] * n for _ in range(n)]
        for i in range(n - 1, -1, -1):
            for j in range(i, n):
                pal[i][j] = s[i] == s[j] and (j - i < 2 or pal[i + 1][j - 1])
        out, path = [], []

        def explore(start: int) -> None:
            if start == n:               # the pieces cover s: one answer
                out.append(path[:])      # copy, path keeps changing
                return
            for end in range(start, n):
                if pal[start][end]:      # only a palindrome may be the next piece
                    path.append(s[start:end + 1])  # choose
                    explore(end + 1)               # explore the rest
                    path.pop()                     # undo

        explore(0)
        return out
