class Solution {
public:
    int strStr(string haystack, string needle) {
        int n = haystack.size(), m = needle.size();
        // lps[i] = length of the longest proper prefix of needle[0..i] that is also its suffix.
        vector<int> lps(m, 0);
        for (int i = 1, k = 0; i < m; i++) {
            while (k > 0 && needle[i] != needle[k]) k = lps[k - 1];   // next shorter border
            if (needle[i] == needle[k]) k++;
            lps[i] = k;
        }
        for (int i = 0, j = 0; i < n; i++) {          // i never moves back
            while (j > 0 && haystack[i] != needle[j]) j = lps[j - 1];
            if (haystack[i] == needle[j] && ++j == m) return i - m + 1;
        }
        return -1;
    }
};

class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        m = len(needle)
        # lps[i] = length of the longest proper prefix of needle[0..i] that is also a suffix of it.
        lps = [0] * m
        k = 0                                   # length of the border we are extending
        for i in range(1, m):
            while k and needle[i] != needle[k]:
                k = lps[k - 1]                  # fall back to the next shorter border
            if needle[i] == needle[k]:
                k += 1
            lps[i] = k

        j = 0                                   # how many needle chars are matched right now
        for i, ch in enumerate(haystack):       # i only moves forward
            while j and ch != needle[j]:
                j = lps[j - 1]                  # keep the longest border, do not restart from 0
            if ch == needle[j]:
                j += 1
                if j == m:
                    return i - m + 1            # the match ends at i
        return -1
