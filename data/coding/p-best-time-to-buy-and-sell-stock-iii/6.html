class Solution {
public:
    int maxProfit(vector<int>& prices) {
        // Best cash balance after each step, starting from 0 cash.
        // Start the buys at -prices[0] (not INT_MIN) so "buy + price" cannot overflow.
        int buy1 = -prices[0], sell1 = 0;
        int buy2 = -prices[0], sell2 = 0;
        for (int p : prices) {
            buy1  = max(buy1,  -p);          // 1st buy: pay p from 0 cash
            sell1 = max(sell1, buy1 + p);    // 1st sell: cash in p
            buy2  = max(buy2,  sell1 - p);   // 2nd buy: pay p from the 1st profit
            sell2 = max(sell2, buy2 + p);    // 2nd sell
        }
        return sell2;   // >= sell1 >= 0: "one trade" and "no trade" are included
    }
};
func maxProfit(prices []int) int {
	// Best cash balance after each step, starting from 0 cash.
	buy1, sell1 := -prices[0], 0
	buy2, sell2 := -prices[0], 0
	for _, p := range prices {
		buy1 = max(buy1, -p)       // 1st buy: pay p from 0 cash

		buy2 = max(buy2, sell1-p)  // 2nd buy: pay p from the 1st profit
		sell2 = max(sell2, buy2+p) // 2nd sell
	}
	return sell2 // >= sell1 >= 0: "one trade" and "no trade" are included
}
