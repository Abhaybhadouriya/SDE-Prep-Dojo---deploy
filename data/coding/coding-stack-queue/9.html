// A Go slice is the stack: append = push, s[len(s)-1] = top, s = s[:len(s)-1] = pop.
func isValid(s string) bool {
	pair := map[byte]byte{')': '(', ']': '[', '}': '{'}
	st := []byte{}
	for i := 0; i < len(s); i++ {
		c := s[i]
		if c == '(' || c == '[' || c == '{' {
			st = append(st, c)
			continue
		}
		if len(st) == 0 || st[len(st)-1] != pair[c] { // empty check first
			return false
		}
		st = st[:len(st)-1]
	}
	return len(st) == 0
}

func removeDuplicates(s string) string {
	st := []byte{} // the answer IS the stack
	for i := 0; i < len(s); i++ {
		if n := len(st); n > 0 && st[n-1] == s[i] {
			st = st[:n-1]
		} else {
			st = append(st, s[i])
		}
	}
	return string(st)
}

func asteroidCollision(a []int) []int {
	st := []int{}
	for _, x := range a {
		alive := true
		for alive && x < 0 && len(st) > 0 && st[len(st)-1] > 0 {
			top := st[len(st)-1]
			if top < -x {
				st = st[:len(st)-1] // top explodes, keep checking
			} else {
				if top == -x {
					st = st[:len(st)-1]
				}
				alive = false
			}
		}
		if alive {
			st = append(st, x)
		}
	}
	return st
}

func evalRPN(tokens []string) int {
	st := []int{}
	for _, t := range tokens {
		if t == "+" || t == "-" || t == "*" || t == "/" {
			b, a := st[len(st)-1], st[len(st)-2] // b is the top
			st = st[:len(st)-2]
			switch t {
			case "+":
				st = append(st, a+b)
			case "-":
				st = append(st, a-b)
			case "*":
				st = append(st, a*b)
			default:
				st = append(st, a/b) // truncates toward zero
			}
		} else {
			v, _ := strconv.Atoi(t)
			st = append(st, v)
		}
	}
	return st[len(st)-1]
}

// calculate handles + - * / ( ) and spaces (LC 224, 227, 772).
func calculate(s string) int {
	i := 0
	var level func() int
	level = func() int {
		terms := []int{}
		num, op := 0, byte('+')
		for {
			c := byte(0)
			if i < len(s) {
				c = s[i]
			}
			i++
			switch {
			case c == ' ':
				continue
			case c >= '0' && c <= '9':
				num = num*10 + int(c-'0')
				continue
			case c == '(':
				num = level()
				continue
			}
			switch op {
			case '+':
				terms = append(terms, num)
			case '-':
				terms = append(terms, -num)
			case '*':
				terms[len(terms)-1] *= num
			default:
				terms[len(terms)-1] /= num
			}
			if c == ')' || c == 0 {
				break
			}
			op, num = c, 0
		}
		sum := 0
		for _, t := range terms {
			sum += t
		}
		return sum
	}
	return level()
}

func decodeString(s string) string {
	type frame struct {
		count  int
		prefix string
	}
	st := []frame{}
	var cur strings.Builder
	k := 0
	for i := 0; i < len(s); i++ {
		c := s[i]
		switch {
		case c >= '0' && c <= '9':
			k = k*10 + int(c-'0')
		case c == '[':
			st = append(st, frame{k, cur.String()})
			cur.Reset()
			k = 0
		case c == ']':
			top := st[len(st)-1]
			st = st[:len(st)-1]
			rep := strings.Repeat(cur.String(), top.count)
			cur.Reset()
			cur.WriteString(top.prefix + rep)
		default:
			cur.WriteByte(c)
		}
	}
	return cur.String()
}

func nextGreaterIndex(a []int) []int {
	ans := make([]int, len(a))
	st := []int{} // indices waiting for an answer
	for i := range ans {
		ans[i] = -1
	}
	for i, x := range a {
		for len(st) > 0 && a[st[len(st)-1]] < x {
			ans[st[len(st)-1]] = i
			st = st[:len(st)-1]
		}
		st = append(st, i)
	}
	return ans
}

func maxSlidingWindow(a []int, k int) []int {
	dq := []int{} // indices, values decreasing; dq[0] is the front
	out := []int{}
	for i, x := range a {
		if len(dq) > 0 && dq[0] <= i-k {
			dq = dq[1:] // fell out of the window
		}
		for len(dq) > 0 && a[dq[len(dq)-1]] <= x {
			dq = dq[:len(dq)-1] // can never be max again
		}
		dq = append(dq, i)
		if i >= k-1 {
			out = append(out, a[dq[0]])
		}
	}
	return out
}

type MyQueue struct{ in, out []int }

func (q *MyQueue) shift() {
	if len(q.out) == 0 { // move only when out is empty
		for len(q.in) > 0 {
			q.out = append(q.out, q.in[len(q.in)-1])
			q.in = q.in[:len(q.in)-1]
		}
	}
}
func (q *MyQueue) Push(x int) { q.in = append(q.in, x) }
func (q *MyQueue) Pop() int {
	q.shift()
	x := q.out[len(q.out)-1]
	q.out = q.out[:len(q.out)-1]
	return x
}
func (q *MyQueue) Peek() int   { q.shift(); return q.out[len(q.out)-1] }
func (q *MyQueue) Empty() bool { return len(q.in) == 0 && len(q.out) == 0 }
// 1. Matching brackets: push openers, pop on a closer and compare.
bool isValid(const string& s) {
    vector<char> st;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') { st.push_back(c); continue; }
        char want = c == ')' ? '(' : c == ']' ? '[' : '{';
        if (st.empty() || st.back() != want) return false;   // check empty BEFORE top
        st.pop_back();
    }
    return st.empty();                                       // leftovers = unclosed
}

// 2. Cancelling adjacent pairs: the answer string IS the stack.
string removeDuplicates(const string& s) {
    string st;
    for (char c : s) {
        if (!st.empty() && st.back() == c) st.pop_back();
        else st.push_back(c);
    }
    return st;
}

// 3. Collisions: a new item may destroy several on top before it settles.
vector<int> asteroidCollision(const vector<int>& a) {
    vector<int> st;
    for (int x : a) {
        bool alive = true;
        while (alive && x < 0 && !st.empty() && st.back() > 0) {
            if (st.back() < -x) st.pop_back();          // top explodes, keep checking
            else { if (st.back() == -x) st.pop_back(); alive = false; }
        }
        if (alive) st.push_back(x);
    }
    return st;
}

// 4. Reverse Polish notation: operands wait on the stack for their operator.
int evalRPN(const vector<string>& tokens) {
    vector<long long> st;
    for (const string& t : tokens) {
        if (t.size() == 1 && string("+-*/").find(t[0]) != string::npos) {
            long long b = st.back(); st.pop_back();     // pop order: b first, then a
            long long a = st.back(); st.pop_back();
            if (t[0] == '+') st.push_back(a + b);
            else if (t[0] == '-') st.push_back(a - b);
            else if (t[0] == '*') st.push_back(a * b);
            else st.push_back(a / b);                   // truncates toward zero
        } else {
            st.push_back(stoll(t));
        }
    }
    return (int)st.back();
}

// 5. Calculator for + - * / ( ) and spaces (covers LC 224, 227, 772).
//    Each level keeps a stack of signed terms; * and / fold into the top term.
//    '(' recurses; ')' or the end of the string closes the level.
long long calcLevel(const string& s, size_t& i) {
    vector<long long> terms;
    long long num = 0;
    char op = '+';                                       // operator BEFORE num
    while (true) {
        char c = i < s.size() ? s[i] : '\0';
        ++i;
        if (c == ' ') continue;
        if (isdigit((unsigned char)c)) { num = num * 10 + (c - '0'); continue; }
        if (c == '(') { num = calcLevel(s, i); continue; }
        // c is + - * / ) or end: apply the pending operator to num
        if (op == '+') terms.push_back(num);
        else if (op == '-') terms.push_back(-num);
        else if (op == '*') terms.back() *= num;
        else terms.back() /= num;
        if (c == ')' || c == '\0') break;
        op = c;
        num = 0;
    }
    long long sum = 0;
    for (long long t : terms) sum += t;
    return sum;
}
int calculate(const string& s) { size_t i = 0; return (int)calcLevel(s, i); }

// 6. Nested decode "3[a2[c]]" -> "accaccacc": push (count, text so far) on '['.
string decodeString(const string& s) {
    vector<pair<int, string>> st;
    string cur;
    int k = 0;
    for (char c : s) {
        if (isdigit((unsigned char)c)) k = k * 10 + (c - '0');
        else if (c == '[') { st.push_back({k, cur}); cur.clear(); k = 0; }
        else if (c == ']') {
            pair<int, string> top = st.back(); st.pop_back();
            string rep;
            for (int r = 0; r < top.first; ++r) rep += cur;
            cur = top.second + rep;
        } else cur += c;
    }
    return cur;
}

// 7. Monotonic stack: index of the next strictly greater element, or -1.
//    The stack holds indices still waiting for their answer; values decrease bottom->top.
vector<int> nextGreaterIndex(const vector<int>& a) {
    vector<int> ans(a.size(), -1), st;
    for (int i = 0; i < (int)a.size(); ++i) {
        while (!st.empty() && a[st.back()] < a[i]) { ans[st.back()] = i; st.pop_back(); }
        st.push_back(i);
    }
    return ans;
}

// 8. Monotonic deque: maximum of every window of size k.
//    Front = index of the current max; values decrease front->back.
vector<int> maxSlidingWindow(const vector<int>& a, int k) {
    deque<int> dq;
    vector<int> out;
    for (int i = 0; i < (int)a.size(); ++i) {
        if (!dq.empty() && dq.front() <= i - k) dq.pop_front();        // fell out of the window
        while (!dq.empty() && a[dq.back()] <= a[i]) dq.pop_back();     // can never be max again
        dq.push_back(i);
        if (i >= k - 1) out.push_back(a[dq.front()]);
    }
    return out;
}

// 9. Queue from two stacks: amortised O(1) per operation.
class MyQueue {
    vector<int> in, out;
    void shift() {                     // move only when out is empty
        if (out.empty()) while (!in.empty()) { out.push_back(in.back()); in.pop_back(); }
    }
public:
    void push(int x) { in.push_back(x); }
    int pop() { shift(); int x = out.back(); out.pop_back(); return x; }
    int peek() { shift(); return out.back(); }
    bool empty() const { return in.empty() && out.empty(); }
};
// Use ArrayDeque, not the old synchronized java.util.Stack.
static boolean isValid(String s) {
    Deque<Character> st = new ArrayDeque<>();
    for (char c : s.toCharArray()) {
        if (c == '(' || c == '[' || c == '{') { st.push(c); continue; }
        char want = c == ')' ? '(' : c == ']' ? '[' : '{';
        if (st.isEmpty() || st.pop() != want) return false;   // empty check first
    }
    return st.isEmpty();
}

static String removeDuplicates(String s) {
    StringBuilder st = new StringBuilder();                     // the answer IS the stack
    for (char c : s.toCharArray()) {
        int n = st.length();
        if (n > 0 && st.charAt(n - 1) == c) st.deleteCharAt(n - 1);
        else st.append(c);
    }
    return st.toString();
}

static int[] asteroidCollision(int[] a) {
    Deque<Integer> st = new ArrayDeque<>();
    for (int x : a) {
        boolean alive = true;
        while (alive && x < 0 && !st.isEmpty() && st.peek() > 0) {
            if (st.peek() < -x) st.pop();                       // top explodes, keep checking
            else { if (st.peek() == -x) st.pop(); alive = false; }
        }
        if (alive) st.push(x);
    }
    int[] out = new int[st.size()];
    for (int i = out.length - 1; i >= 0; i--) out[i] = st.pop();  // top is the LAST element
    return out;
}

static int evalRPN(String[] tokens) {
    Deque<Long> st = new ArrayDeque<>();
    for (String t : tokens) {
        if (t.length() == 1 && "+-*/".indexOf(t.charAt(0)) >= 0) {
            long b = st.pop(), a = st.pop();                    // b first
            switch (t.charAt(0)) {
                case '+': st.push(a + b); break;
                case '-': st.push(a - b); break;
                case '*': st.push(a * b); break;
                default:  st.push(a / b);                       // truncates toward zero
            }
        } else st.push(Long.parseLong(t));
    }
    return (int) (long) st.peek();
}

private static int pos;
static int calculate(String s) { pos = 0; return (int) calcLevel(s); }
private static long calcLevel(String s) {
    Deque<Long> terms = new ArrayDeque<>();
    long num = 0;
    char op = '+';
    while (true) {
        char c = pos < s.length() ? s.charAt(pos) : '\0';
        pos++;
        if (c == ' ') continue;
        if (Character.isDigit(c)) { num = num * 10 + (c - '0'); continue; }
        if (c == '(') { num = calcLevel(s); continue; }
        if (op == '+') terms.push(num);
        else if (op == '-') terms.push(-num);
        else if (op == '*') terms.push(terms.pop() * num);
        else terms.push(terms.pop() / num);
        if (c == ')' || c == '\0') break;
        op = c;
        num = 0;
    }
    long sum = 0;
    for (long t : terms) sum += t;
    return sum;
}

static String decodeString(String s) {
    Deque<Integer> counts = new ArrayDeque<>();
    Deque<StringBuilder> prefixes = new ArrayDeque<>();
    StringBuilder cur = new StringBuilder();
    int k = 0;
    for (char c : s.toCharArray()) {
        if (Character.isDigit(c)) k = k * 10 + (c - '0');
        else if (c == '[') { counts.push(k); prefixes.push(cur); cur = new StringBuilder(); k = 0; }
        else if (c == ']') {
            StringBuilder prev = prefixes.pop();
            prev.append(cur.toString().repeat(counts.pop()));
            cur = prev;
        } else cur.append(c);
    }
    return cur.toString();
}

static int[] nextGreaterIndex(int[] a) {
    int[] ans = new int[a.length];
    Arrays.fill(ans, -1);
    Deque<Integer> st = new ArrayDeque<>();                     // indices waiting for an answer
    for (int i = 0; i < a.length; i++) {
        while (!st.isEmpty() && a[st.peek()] < a[i]) ans[st.pop()] = i;
        st.push(i);
    }
    return ans;
}

static int[] maxSlidingWindow(int[] a, int k) {
    Deque<Integer> dq = new ArrayDeque<>();                     // indices, values decreasing
    int[] out = new int[a.length - k + 1];
    for (int i = 0; i < a.length; i++) {
        if (!dq.isEmpty() && dq.peekFirst() <= i - k) dq.pollFirst();
        while (!dq.isEmpty() && a[dq.peekLast()] <= a[i]) dq.pollLast();
        dq.offerLast(i);
        if (i >= k - 1) out[i - k + 1] = a[dq.peekFirst()];
    }
    return out;
}

static class MyQueue {
    private final Deque<Integer> in = new ArrayDeque<>(), out = new ArrayDeque<>();
    private void shift() { if (out.isEmpty()) while (!in.isEmpty()) out.push(in.pop()); }
    void push(int x) { in.push(x); }
    int pop() { shift(); return out.pop(); }
    int peek() { shift(); return out.peek(); }
    boolean empty() { return in.isEmpty() && out.isEmpty(); }
}
def is_valid(s: str) -> bool:
    pairs = {")": "(", "]": "[", "}": "{"}
    st = []
    for c in s:
        if c in "([{":
            st.append(c)
        elif not st or st.pop() != pairs[c]:     # empty check comes first
            return False
    return not st                                 # leftovers = unclosed


def remove_duplicates(s: str) -> str:
    st = []
    for c in s:
        if st and st[-1] == c:
            st.pop()
        else:
            st.append(c)
    return "".join(st)


def asteroid_collision(a: List[int]) -> List[int]:
    st = []
    for x in a:
        alive = True
        while alive and x < 0 and st and st[-1] > 0:
            if st[-1] < -x:
                st.pop()                          # top explodes, keep checking
            else:
                if st[-1] == -x:
                    st.pop()
                alive = False
        if alive:
            st.append(x)
    return st


def eval_rpn(tokens: List[str]) -> int:
    st = []
    for t in tokens:
        if t in ("+", "-", "*", "/"):
            b, a = st.pop(), st.pop()             # b is popped first
            if t == "+": st.append(a + b)
            elif t == "-": st.append(a - b)
            elif t == "*": st.append(a * b)
            else: st.append(int(a / b))           # truncate toward zero, NOT a // b
        else:
            st.append(int(t))
    return st[-1]


def calculate(s: str) -> int:
    """+ - * / ( ) and spaces (LC 224, 227, 772)."""
    i = 0

    def level() -> int:
        nonlocal i
        terms, num, op = [], 0, "+"
        while True:
            c = s[i] if i < len(s) else "\0"
            i += 1
            if c == " ":
                continue
            if c.isdigit():
                num = num * 10 + int(c)
                continue
            if c == "(":
                num = level()
                continue
            if op == "+": terms.append(num)
            elif op == "-": terms.append(-num)
            elif op == "*": terms[-1] *= num
            else: terms[-1] = int(terms[-1] / num)  # truncate toward zero
            if c in ")\0":
                return sum(terms)
            op, num = c, 0

    return level()


def decode_string(s: str) -> str:
    st, cur, k = [], "", 0
    for c in s:
        if c.isdigit():
            k = k * 10 + int(c)
        elif c == "[":
            st.append((k, cur))
            cur, k = "", 0
        elif c == "]":
            count, prev = st.pop()
            cur = prev + cur * count
        else:
            cur += c
    return cur


def next_greater_index(a: List[int]) -> List[int]:
    ans, st = [-1] * len(a), []                   # st: indices waiting for an answer
    for i, x in enumerate(a):
        while st and a[st[-1]] < x:
            ans[st.pop()] = i
        st.append(i)
    return ans


def max_sliding_window(a: List[int], k: int) -> List[int]:
    dq, out = deque(), []                         # indices, values decreasing
    for i, x in enumerate(a):
        if dq and dq[0] <= i - k:
            dq.popleft()                          # fell out of the window
        while dq and a[dq[-1]] <= x:
            dq.pop()                              # can never be max again
        dq.append(i)
        if i >= k - 1:
            out.append(a[dq[0]])
    return out


class MyQueue:
    def __init__(self):
        self._in, self._out = [], []

    def _shift(self):
        if not self._out:                         # move only when out is empty
            while self._in:
                self._out.append(self._in.pop())

    def push(self, x: int) -> None:
        self._in.append(x)

    def pop(self) -> int:
        self._shift()
        return self._out.pop()

    def peek(self) -> int:
        self._shift()
        return self._out[-1]

    def empty(self) -> bool:
        return not self._in and not self._out
