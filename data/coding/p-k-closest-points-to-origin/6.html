class Solution {
public:
    vector<vector<int>> kClosest(vector<vector<int>>& points, int k) {
        // Max-heap of (squared distance, index); the top is the worst of the k best so far.
        priority_queue<pair<int, int>> heap;
        for (int i = 0; i < (int)points.size(); i++) {
            int d = points[i][0] * points[i][0] + points[i][1] * points[i][1];  // <= 2e8, fits int
            if ((int)heap.size() < k) {
                heap.push({d, i});
            } else if (d < heap.top().first) {
                heap.pop();                        // evict the worst
                heap.push({d, i});
            }
        }
        vector<vector<int>> out;
        while (!heap.empty()) {
            out.push_back(points[heap.top().second]);
            heap.pop();
        }
        return out;
    }

    vector<vector<int>> kClosestQuickselect(vector<vector<int>> pts, int k) {
        auto dist = [](const vector<int>& p) { return p[0] * p[0] + p[1] * p[1]; };
        mt19937 rng(12345);
        int lo = 0, hi = pts.size() - 1;
        while (lo < hi) {
            int p = lo + rng() % (hi - lo + 1);    // random pivot
            swap(pts[p], pts[hi]);
            int pivot = dist(pts[hi]), store = lo;
            for (int i = lo; i < hi; i++)
                if (dist(pts[i]) < pivot) swap(pts[i], pts[store++]);
            swap(pts[store], pts[hi]);             // pivot at its final rank
            if (store == k - 1) break;
            if (store < k - 1) lo = store + 1; else hi = store - 1;
        }
        pts.resize(k);
        return pts;
    }
};
func dist(p []int) int { return p[0]*p[0] + p[1]*p[1] } // <= 2e8

// maxHeap of points by distance: the root is the worst of the k best so far.
type maxHeap [][]int

func (h maxHeap) Len() int            { return len(h) }
func (h maxHeap) Less(i, j int) bool  { return dist(h[i]) > dist(h[j]) }
func (h maxHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *maxHeap) Push(x interface{}) { *h = append(*h, x.([]int)) }
func (h *maxHeap) Pop() interface{} {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

func kClosest(points [][]int, k int) [][]int {
	h := &maxHeap{}
	for _, p := range points {
		if h.Len() < k {
			heap.Push(h, p)
		} else if dist(p) < dist((*h)[0]) {
			(*h)[0] = p // replace the worst, then sift down
			heap.Fix(h, 0)
		}
	}
	return *h
}

func kClosestQuickselect(points [][]int, k int) [][]int {
	pts := append([][]int(nil), points...)
	lo, hi := 0, len(pts)-1
	for lo < hi {
		p := lo + rand.Intn(hi-lo+1) // random pivot to the end
		pts[p], pts[hi] = pts[hi], pts[p]
		pivot, store := dist(pts[hi]), lo
		for i := lo; i < hi; i++ {
			if dist(pts[i]) < pivot {
				pts[i], pts[store] = pts[store], pts[i]
				store++
			}
		}
		pts[store], pts[hi] = pts[hi], pts[store] // pivot at its final rank
		if store == k-1 {
			break
		}
		if store < k-1 {
			lo = store + 1
		} else {
			hi = store - 1
		}
	}
	return pts[:k]
}

static class Solution {
    private static int dist(int[] p) { return p[0] * p[0] + p[1] * p[1]; }  // <= 2e8, fits int

    public int[][] kClosest(int[][] points, int k) {
        // Max-heap by distance: the head is the worst of the k best so far.
        PriorityQueue<int[]> heap = new PriorityQueue<>((a, b) -> Integer.compare(dist(b), dist(a)));
        for (int[] p : points) {
            if (heap.size() < k) {
                heap.offer(p);
            } else if (dist(p) < dist(heap.peek())) {
                heap.poll();                   // evict the worst
                heap.offer(p);
            }
        }
        return heap.toArray(new int[0][]);
    }

    public int[][] kClosestQuickselect(int[][] points, int k) {
        int[][] pts = points.clone();
        Random rnd = new Random(12345);
        int lo = 0, hi = pts.length - 1;
        while (lo < hi) {
            swap(pts, lo + rnd.nextInt(hi - lo + 1), hi);   // random pivot to the end
            int pivot = dist(pts[hi]), store = lo;
            for (int i = lo; i < hi; i++)
                if (dist(pts[i]) < pivot) swap(pts, i, store++);
            swap(pts, store, hi);              // pivot at its final rank
            if (store == k - 1) break;
            if (store < k - 1) lo = store + 1; else hi = store - 1;
        }
        return Arrays.copyOf(pts, k);
    }

    private static void swap(int[][] a, int i, int j) { int[] t = a[i]; a[i] = a[j]; a[j] = t; }
}
class Solution:
    def kClosest(self, points: List[List[int]], k: int) -> List[List[int]]:
        # Max-heap of size k (Python's heapq is a min-heap, so store -distance).
        # The root is the WORST of the k best so far: a new point only has to beat it.
        heap = []
        for x, y in points:
            d = x * x + y * y                   # squared distance: no sqrt, same order
            if len(heap) < k:
                heapq.heappush(heap, (-d, x, y))
            elif d < -heap[0][0]:
                heapq.heapreplace(heap, (-d, x, y))   # pop the worst, push the newcomer
        return [[x, y] for _, x, y in heap]

    def kClosestQuickselect(self, points: List[List[int]], k: int) -> List[List[int]]:
        # Partition until the k smallest distances sit in pts[0..k-1] (in any order).
        pts = points[:]
        dist = lambda p: p[0] * p[0] + p[1] * p[1]
        lo, hi = 0, len(pts) - 1
        while lo < hi:
            p = random.randint(lo, hi)          # random pivot: O(n) on average for any input
            pts[p], pts[hi] = pts[hi], pts[p]
            pivot, store = dist(pts[hi]), lo
            for i in range(lo, hi):             # Lomuto: closer-than-pivot points go left
                if dist(pts[i]) < pivot:
                    pts[i], pts[store] = pts[store], pts[i]
                    store += 1
            pts[store], pts[hi] = pts[hi], pts[store]   # the pivot lands at its final rank
            if store == k - 1:
                break
            if store < k - 1:
                lo = store + 1                  # the cut is to the right
            else:
                hi = store - 1                  # the cut is to the left
        return pts[:k]
