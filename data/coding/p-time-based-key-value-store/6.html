class TimeMap {
    // key -> (timestamp, value) pairs in increasing timestamp order (set times are strictly increasing)
    unordered_map<string, vector<pair<int, string>>> history;

public:
    TimeMap() {}

    void set(string key, string value, int timestamp) {
        history[key].emplace_back(timestamp, value);   // append keeps the vector sorted
    }

    string get(string key, int timestamp) {
        auto it = history.find(key);                   // find, not [], so no empty entry is created
        if (it == history.end()) return "";
        const vector<pair<int, string>>& h = it->second;
        int lo = 0, hi = h.size();                     // first index with time > timestamp
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (h[mid].first <= timestamp) lo = mid + 1;
            else hi = mid;
        }
        return lo == 0 ? "" : h[lo - 1].second;        // lo-1 = latest time <= timestamp
    }
};
type entry struct {
	time  int
	value string
}

type TimeMap struct {
	history map[string][]entry // key -> entries in increasing time order
}

func Constructor() TimeMap {
	return TimeMap{history: make(map[string][]entry)}
}

func (m *TimeMap) Set(key string, value string, timestamp int) {
	m.history[key] = append(m.history[key], entry{timestamp, value}) // append keeps it sorted
}

func (m *TimeMap) Get(key string, timestamp int) string {
	h := m.history[key] // a missing key gives a nil slice: no entry is created
	// first index whose time is > timestamp
	i := sort.Search(len(h), func(j int) bool { return h[j].time > timestamp })
	if i == 0 {
		return ""
	}
	return h[i-1].value // latest time <= timestamp
}

static class TimeMap {
    private static class Entry {
        final int time;
        final String value;
        Entry(int time, String value) { this.time = time; this.value = value; }
    }

    // key -> entries in increasing time order (set times are strictly increasing)
    private final Map<String, List<Entry>> history = new HashMap<>();

    public TimeMap() {}

    public void set(String key, String value, int timestamp) {
        history.computeIfAbsent(key, k -> new ArrayList<>()).add(new Entry(timestamp, value));
    }

    public String get(String key, int timestamp) {
        List<Entry> h = history.get(key);
        if (h == null) return "";
        int lo = 0, hi = h.size();                 // first index with time > timestamp
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (h.get(mid).time <= timestamp) lo = mid + 1;
            else hi = mid;
        }
        return lo == 0 ? "" : h.get(lo - 1).value; // lo-1 = latest time <= timestamp
    }
}
class TimeMap:
    def __init__(self):
        self.times = defaultdict(list)   # key -> timestamps, increasing (set times are strictly increasing)
        self.vals = defaultdict(list)    # key -> values, parallel to self.times[key]

    def set(self, key: str, value: str, timestamp: int) -> None:
        self.times[key].append(timestamp)   # appending keeps the list sorted: no insert cost
        self.vals[key].append(value)

    def get(self, key: str, timestamp: int) -> str:
        ts = self.times.get(key)            # .get does not create an empty entry
        if not ts:
            return ""
        i = bisect_right(ts, timestamp)     # first index whose time is > timestamp
        return self.vals[key][i - 1] if i > 0 else ""   # i-1 = latest time <= timestamp
