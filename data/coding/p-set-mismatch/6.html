class Solution {
public:
    // Index as hash: slot v-1 turns negative the first time value v is seen.
    vector<int> findErrorNums(vector<int>& nums) {
        int n = nums.size(), dup = -1, miss = -1;
        for (int i = 0; i < n; i++) {
            int v = abs(nums[i]);               // the slot may already carry a mark
            if (nums[v - 1] < 0) dup = v;       // slot already marked: second copy of v
            else nums[v - 1] = -nums[v - 1];
        }
        for (int i = 0; i < n; i++) {
            if (nums[i] > 0) miss = i + 1;      // nobody marked slot i: value i+1 never came
            else nums[i] = -nums[i];            // undo the mark, the caller gets its array back
        }
        return {dup, miss};
    }

    // Input untouched: d1 = dup - miss, d2 = dup^2 - miss^2, so d2 / d1 = dup + miss.
    vector<int> findErrorNumsMath(const vector<int>& nums) {
        long long n = nums.size(), d1 = 0, d2 = 0;
        for (long long i = 1; i <= n; i++) {
            long long x = nums[i - 1];
            d1 += x - i;                        // running differences stay small, no overflow
            d2 += x * x - i * i;
        }
        long long s = d2 / d1;                  // d1 != 0 because dup != miss
        return {(int)((s + d1) / 2), (int)((s - d1) / 2)};
    }
};

		}
		if nums[v-1] < 0 {
			dup = v // slot already marked: second copy of v
		} else {
			nums[v-1] = -nums[v-1]
		}
	}

		if nums[i] > 0 {
			miss = i + 1 // nobody marked slot i: value i+1 never came
		} else {
			nums[i] = -nums[i] // undo the mark, the caller gets its slice back
		}
	}

	for i, x := range nums {
		k := i + 1
		d1 += x - k // running differences stay small, no overflow
		d2 += x*x - k*k
	}
	s := d2 / d1 // d1 != 0 because dup != miss

class Solution:
    def findErrorNums(self, nums: List[int]) -> List[int]:
        # Index as hash: slot v-1 turns negative the first time value v is seen.
        dup = miss = -1
        for x in nums:
            v = abs(x)                   # the slot may already carry a mark
            if nums[v - 1] < 0:
                dup = v                  # slot already marked: second copy of v
            else:
                nums[v - 1] = -nums[v - 1]
        for i in range(len(nums)):
            if nums[i] > 0:
                miss = i + 1             # nobody marked slot i: value i+1 never came
            else:
                nums[i] = -nums[i]       # undo the mark, the caller gets its list back
        return [dup, miss]

    def findErrorNumsMath(self, nums: List[int]) -> List[int]:
        # Input untouched: d1 = dup - miss, d2 = dup^2 - miss^2, so d2 // d1 = dup + miss.
        d1 = d2 = 0
        for i, x in enumerate(nums, 1):
            d1 += x - i
            d2 += x * x - i * i
        s = d2 // d1                     # d1 != 0 because dup != miss
        return [(s + d1) // 2, (s - d1) // 2]
