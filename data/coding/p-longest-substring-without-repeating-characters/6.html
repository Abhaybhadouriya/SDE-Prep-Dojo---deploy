All four use the optimal approach: a 256-entry last-seen array (the input is ASCII), -1 meaning "never seen".

class Solution {
public:
    int lengthOfLongestSubstring(const string& s) {
        vector<int> last(256, -1);      // byte -> index of its most recent copy
        int left = 0, best = 0;         // invariant: s[left..right] has no repeat
        for (int right = 0; right < (int)s.size(); ++right) {
            unsigned char c = s[right];
            if (last[c] >= left)        // old copy is inside the window
                left = last[c] + 1;     // jump past it; left never moves back
            last[c] = right;
            best = max(best, right - left + 1);
        }
        return best;
    }
};
func lengthOfLongestSubstring(s string) int {
	var last [256]int // byte -> index of its most recent copy
	for i := range last {
		last[i] = -1
	}
	left, best := 0, 0 // invariant: s[left..right] has no repeat
	for right := 0; right < len(s); right++ {
		c := s[right]
		if last[c] >= left { // old copy is inside the window
			left = last[c] + 1 // jump past it; left never moves back
		}
		last[c] = right
		if right-left+1 > best {
			best = right - left + 1
		}
	}
	return best
}
static class Solution {
    public int lengthOfLongestSubstring(String s) {
        int[] last = new int[256];          // char -> index of its most recent copy
        Arrays.fill(last, -1);
        int left = 0, best = 0;             // invariant: s[left..right] has no repeat
        for (int right = 0; right < s.length(); right++) {
            char c = s.charAt(right);       // input is ASCII (letters, digits, symbols, space)
            if (last[c] >= left)            // old copy is inside the window
                left = last[c] + 1;         // jump past it; left never moves back
            last[c] = right;
            best = Math.max(best, right - left + 1);
        }
        return best;
    }
}
class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        last = {}            # char -> index of its most recent copy
        left = best = 0      # invariant: s[left..right] has no repeated char
        for right, ch in enumerate(s):
            if last.get(ch, -1) >= left:   # the old copy is inside the window
                left = last[ch] + 1        # jump just past it (left never moves back)
            last[ch] = right
            best = max(best, right - left + 1)
        return best
