Template: build your own plan from any lists

Suppose you want your own plan: a core sheet plus a company's frequent problems, minus what you already solved, ordered by priority, a few per day, and a revision date for each solved one. That is four small functions: an order-preserving merge (dedupe), a frequency filter, a stable sort and a spaced-repetition ladder. The same code works on the list and company files in /data/coding/data/.

package main

import (
	"fmt"
	"reflect"
	"sort"
)

var ladder = []int{1, 3, 7, 14, 30} // days until the next revision

// mergeLists is an order-preserving union: the first list that has a slug decides its place.
func mergeLists(lists ...[]string) []string {
	seen := map[string]bool{}
	out := []string{}
	for _, lst := range lists {
		for _, s := range lst {
			if !seen[s] {
				seen[s] = true
				out = append(out, s)
			}
		}
	}
	return out
}

type row struct {
	slug string
	freq int // 1-8 bar from the saved company page
}

// companyFilter keeps the frequent rows, in saved-page order.
func companyFilter(rows []row, minFreq int) []string {
	out := []string{}
	for _, r := range rows {
		if r.freq >= minFreq {
			out = append(out, r.slug)
		}
	}
	return out
}

// buildPlan drops solved problems, sorts by priority (stable), cuts into days.
func buildPlan(cand []string, prio map[string]float64, done map[string]bool, perDay int) [][]string {
	todo := []string{}
	for _, s := range cand {
		if !done[s] {
			todo = append(todo, s)
		}
	}
	sort.SliceStable(todo, func(i, j int) bool { return prio[todo[i]] > prio[todo[j]] })
	days := [][]string{}
	for i := 0; i < len(todo); i += perDay {
		end := i + perDay
		if end > len(todo) {
			end = len(todo)
		}
		days = append(days, todo[i:end])
	}
	return days
}

func nextReview(day, reps int) int {
	if reps > len(ladder)-1 {
		reps = len(ladder) - 1
	}
	return day + ladder[reps]
}

func main() {
	core := []string{"two-sum", "3sum", "merge-intervals"}
	company := []row{{"lru-cache", 6}, {"two-sum", 8}, {"number-of-islands", 3}, {"valid-word-abbreviation", 2}}
	prio := map[string]float64{"lru-cache": 42.87, "number-of-islands": 41.74,
		"merge-intervals": 41.65, "two-sum": 38.4, "3sum": 32.73}
	merged := mergeLists(core, companyFilter(company, 3))
	if !reflect.DeepEqual(merged, []string{"two-sum", "3sum", "merge-intervals", "lru-cache", "number-of-islands"}) {
		panic(fmt.Sprint(merged))
	}
	plan := buildPlan(merged, prio, map[string]bool{"two-sum": true}, 2)
	if !reflect.DeepEqual(plan, [][]string{{"lru-cache", "number-of-islands"}, {"merge-intervals", "3sum"}}) {
		panic(fmt.Sprint(plan))
	}
	if len(buildPlan(nil, prio, nil, 2)) != 0 {
		panic("empty plan expected")
	}
	if nextReview(10, 0) != 11 || nextReview(10, 1) != 13 || nextReview(10, 4) != 40 || nextReview(10, 9) != 40 {
		panic("ladder")
	}
	fmt.Println("all tests passed")
}
#include <bits/stdc++.h>
using namespace std;

const vector<int> LADDER = {1, 3, 7, 14, 30};  // days until the next revision

// Order-preserving union: the first list that has a slug decides its place.
vector<string> mergeLists(const vector<vector<string>>& lists) {
    unordered_set<string> seen;
    vector<string> out;
    for (auto& lst : lists)
        for (auto& s : lst)
            if (seen.insert(s).second) out.push_back(s);
    return out;
}

// rows = (slug, freq 1-8) in saved-page order; keep the frequent ones.
vector<string> companyFilter(const vector<pair<string, int>>& rows, int minFreq) {
    vector<string> out;
    for (auto& r : rows) if (r.second >= minFreq) out.push_back(r.first);
    return out;
}

// Drop solved problems, sort by priority (stable), cut into days.
vector<vector<string>> buildPlan(const vector<string>& cand, const map<string, double>& prio,
                                 const set<string>& done, int perDay) {
    vector<string> todo;
    for (auto& s : cand) if (!done.count(s)) todo.push_back(s);
    auto p = [&](const string& s) { auto it = prio.find(s); return it == prio.end() ? 0.0 : it->second; };
    stable_sort(todo.begin(), todo.end(), [&](const string& a, const string& b) { return p(a) > p(b); });
    vector<vector<string>> days;
    for (size_t i = 0; i < todo.size(); i += perDay)
        days.emplace_back(todo.begin() + i, todo.begin() + min(todo.size(), i + perDay));
    return days;
}

int nextReview(int day, int reps) { return day + LADDER[min(reps, (int)LADDER.size() - 1)]; }

int main() {
    vector<string> core = {"two-sum", "3sum", "merge-intervals"};
    vector<pair<string, int>> company = {{"lru-cache", 6}, {"two-sum", 8},
                                         {"number-of-islands", 3}, {"valid-word-abbreviation", 2}};
    map<string, double> prio = {{"lru-cache", 42.87}, {"number-of-islands", 41.74},
                                {"merge-intervals", 41.65}, {"two-sum", 38.4}, {"3sum", 32.73}};
    vector<string> merged = mergeLists({core, companyFilter(company, 3)});
    assert((merged == vector<string>{"two-sum", "3sum", "merge-intervals", "lru-cache", "number-of-islands"}));
    vector<vector<string>> plan = buildPlan(merged, prio, {"two-sum"}, 2);
    assert((plan == vector<vector<string>>{{"lru-cache", "number-of-islands"}, {"merge-intervals", "3sum"}}));
    assert(buildPlan({}, prio, {}, 2).empty());
    assert(nextReview(10, 0) == 11 && nextReview(10, 1) == 13 && nextReview(10, 4) == 40 && nextReview(10, 9) == 40);
    puts("all tests passed");
    return 0;
}
import java.util.*;

public class Templates {
    static final int[] LADDER = {1, 3, 7, 14, 30}; // days until the next revision

    // Order-preserving union: the first list that has a slug decides its place.
    @SafeVarargs
    static List<String> mergeLists(List<String>... lists) {
        LinkedHashSet<String> seen = new LinkedHashSet<>();
        for (List<String> lst : lists) seen.addAll(lst);
        return new ArrayList<>(seen);
    }

    // rows = {slug, freq 1-8} in saved-page order; keep the frequent ones.
    static List<String> companyFilter(String[][] rows, int minFreq) {
        List<String> out = new ArrayList<>();
        for (String[] r : rows) if (Integer.parseInt(r[1]) >= minFreq) out.add(r[0]);
        return out;
    }

    // Drop solved problems, sort by priority (List.sort is stable), cut into days.
    static List<List<String>> buildPlan(List<String> cand, Map<String, Double> prio,
                                        Set<String> done, int perDay) {
        List<String> todo = new ArrayList<>();
        for (String s : cand) if (!done.contains(s)) todo.add(s);
        todo.sort((a, b) -> Double.compare(prio.getOrDefault(b, 0.0), prio.getOrDefault(a, 0.0)));
        List<List<String>> days = new ArrayList<>();
        for (int i = 0; i < todo.size(); i += perDay)
            days.add(new ArrayList<>(todo.subList(i, Math.min(todo.size(), i + perDay))));
        return days;
    }

    static int nextReview(int day, int reps) { return day + LADDER[Math.min(reps, LADDER.length - 1)]; }

    public static void main(String[] args) {
        List<String> core = List.of("two-sum", "3sum", "merge-intervals");
        String[][] company = {{"lru-cache", "6"}, {"two-sum", "8"}, {"number-of-islands", "3"},
                              {"valid-word-abbreviation", "2"}};
        Map<String, Double> prio = Map.of("lru-cache", 42.87, "number-of-islands", 41.74,
                "merge-intervals", 41.65, "two-sum", 38.4, "3sum", 32.73);
        List<String> merged = mergeLists(core, companyFilter(company, 3));
        check(merged.equals(List.of("two-sum", "3sum", "merge-intervals", "lru-cache", "number-of-islands")));
        List<List<String>> plan = buildPlan(merged, prio, Set.of("two-sum"), 2);
        check(plan.equals(List.of(List.of("lru-cache", "number-of-islands"), List.of("merge-intervals", "3sum"))));
        check(buildPlan(List.of(), prio, Set.of(), 2).isEmpty());
        check(nextReview(10, 0) == 11 && nextReview(10, 1) == 13 && nextReview(10, 4) == 40 && nextReview(10, 9) == 40);
        System.out.println("all tests passed");
    }

    static void check(boolean ok) { if (!ok) throw new AssertionError(); }
}
LADDER = [1, 3, 7, 14, 30]  # days until the next revision, by successful reviews


def merge_lists(*lists):
    """Order-preserving union: the first list that has a slug decides its place."""
    seen, out = set(), []
    for lst in lists:
        for slug in lst:
            if slug not in seen:
                seen.add(slug)
                out.append(slug)
    return out


def company_filter(rows, min_freq):
    """rows = [(slug, freq 1-8)] in saved-page order; keep the frequent ones."""
    return [slug for slug, freq in rows if freq >= min_freq]


def build_plan(candidates, priority, done, per_day):
    """Drop solved problems, sort by priority (stable), cut into days."""
    todo = [s for s in candidates if s not in done]
    todo.sort(key=lambda s: -priority.get(s, 0.0))  # ties keep list order
    return [todo[i:i + per_day] for i in range(0, len(todo), per_day)]


def next_review(day, reps):
    """reps = successful reviews so far; returns the day of the next review."""
    return day + LADDER[min(reps, len(LADDER) - 1)]


def main():
    core = ["two-sum", "3sum", "merge-intervals"]
    company = [("lru-cache", 6), ("two-sum", 8), ("number-of-islands", 3),
               ("valid-word-abbreviation", 2)]
    priority = {"lru-cache": 42.87, "number-of-islands": 41.74,
                "merge-intervals": 41.65, "two-sum": 38.4, "3sum": 32.73}
    merged = merge_lists(core, company_filter(company, 3))
    assert merged == ["two-sum", "3sum", "merge-intervals", "lru-cache",
                      "number-of-islands"]
    plan = build_plan(merged, priority, {"two-sum"}, 2)
    assert plan == [["lru-cache", "number-of-islands"], ["merge-intervals", "3sum"]]
    assert build_plan([], priority, set(), 2) == []
    assert [next_review(10, r) for r in (0, 1, 4, 9)] == [11, 13, 40, 40]
    print("all tests passed")


if __name__ == "__main__":
    main()

All four versions (Go 1.27, C++14, Java 21, Python 3.11) were compiled and run and pass their checks. The priorities in the test are the real scores of those five problems.

From your world

The merge is the same as building a deduplicated route table from several feeds: first insertion wins, later duplicates are dropped, order is kept for determinism. The ladder is an exponential backoff timer, like TCP retransmission: each successful review doubles (roughly) the wait before the next one.
