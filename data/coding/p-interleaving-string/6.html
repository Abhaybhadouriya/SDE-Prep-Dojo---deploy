All four use the one-row DP. Inside the loop, ok[j] (before it is overwritten) is the cell above and ok[j-1] is the cell to the left.

class Solution {
public:
    bool isInterleave(string a, string b, string s) {
        int m = a.size(), n = b.size();
        if (m + n != (int)s.size()) return false;      // lengths must add up
        // ok[j] (while on row i) = can a[0..i) and b[0..j) build s[0..i+j) ?
        // One row of the (m+1) x (n+1) grid is enough: O(n) memory.
        vector<bool> ok(n + 1, false);
        for (int i = 0; i <= m; ++i) {
            for (int j = 0; j <= n; ++j) {
                if (i == 0 && j == 0) { ok[0] = true; continue; }
                bool fromA = i > 0 && ok[j] && a[i - 1] == s[i + j - 1];        // ok[j] = row above
                bool fromB = j > 0 && ok[j - 1] && b[j - 1] == s[i + j - 1];    // ok[j-1] = left
                ok[j] = fromA || fromB;
            }
        }
        return ok[n];
    }
};
func isInterleave(a, b, s string) bool {
	m, n := len(a), len(b)
	if m+n != len(s) { // lengths must add up
		return false
	}
	// ok[j] (while on row i) = can a[:i] and b[:j] build s[:i+j] ?
	// One row of the (m+1) x (n+1) grid is enough: O(n) memory.
	ok := make([]bool, n+1)
	for i := 0; i <= m; i++ {
		for j := 0; j <= n; j++ {
			if i == 0 && j == 0 {
				ok[0] = true
				continue
			}
			fromA := i > 0 && ok[j] && a[i-1] == s[i+j-1]   // ok[j] = row above
			fromB := j > 0 && ok[j-1] && b[j-1] == s[i+j-1] // ok[j-1] = left
			ok[j] = fromA || fromB
		}
	}
	return ok[n]
}

    def isInterleave(self, a: str, b: str, s: str) -> bool:

        if m + n != len(s):                  # lengths must add up
