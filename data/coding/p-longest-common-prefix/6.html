class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {
        const string& first = strs[0];
        for (size_t i = 0; i < first.size(); i++) {      // column i
            char c = first[i];
            for (size_t k = 1; k < strs.size(); k++) {
                // The first string that ends here, or differs here, ends the prefix.
                if (i == strs[k].size() || strs[k][i] != c) return first.substr(0, i);
            }
        }
        return first;                                    // the first string is itself the prefix
    }
};
func longestCommonPrefix(strs []string) string {
	first := strs[0]
	for i := 0; i < len(first); i++ { // column i
		c := first[i]
		for _, w := range strs[1:] {
			// The first string that ends here, or differs here, ends the prefix.
			if i == len(w) || w[i] != c {
				return first[:i]
			}
		}
	}
	return first // the first string is itself the prefix
}
