Kth smallest and the BST iterator: inorder you can pause

A recursive inorder walk is hard to stop in the middle and resume later. The iterative version keeps its whole state in a stack, so we can stop after every visit. The stack always holds the left spine that has not been visited yet:

  1. Start. Push the root and keep going left, pushing every node. The top of the stack is now the smallest key.
  2. next(). Pop the top; that is the answer. Before returning, push the left spine of its right child, because the next bigger key is the leftmost node of that right subtree (or, if there is no right subtree, an ancestor already waiting on the stack).
  3. hasNext(). The stack is not empty.

Click through all nine next() calls on the example tree. The blue boxes are the stack (top is highest), the green nodes are already returned:

  1. new BSTIterator(root): pushLeft(8) pushes 8, 3, 1
  2. next() = 1: pop 1; 1 has no right child, push nothing
  3. next() = 3: pop 3; pushLeft(6) pushes 6, 4
  4. next() = 4: pop 4; no right child
  5. next() = 6: pop 6; pushLeft(7) pushes 7
  6. next() = 7: pop 7
  7. next() = 8: pop 8; pushLeft(10) pushes 10 only
  8. next() = 10: pop 10; pushLeft(14) pushes 14, 13
  9. next() = 13: pop 13
  10. next() = 14: stack empty, hasNext() = false

Cost. One next() can push many nodes (step 3 pushes two), but every node is pushed once and popped once over the whole walk, so n calls cost O(n) total: O(1) amortized per call, O(h) memory. That is exactly what LeetCode 173 asks for. Kth smallest is "call next() k times", O(h + k). Validation is "call next() until the end and check each value is bigger than the last".

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

// BSTIterator is a paused inorder walk: the stack holds the left spine still
// to be visited. Next is O(1) amortized, memory O(h).
type BSTIterator struct{ st []*TreeNode }

func NewBSTIterator(root *TreeNode) *BSTIterator {
	it := &BSTIterator{}
	it.pushLeft(root)
	return it
}

func (it *BSTIterator) pushLeft(n *TreeNode) {
	for n != nil {
		it.st = append(it.st, n)
		n = n.Left
	}
}

func (it *BSTIterator) HasNext() bool { return len(it.st) > 0 }

func (it *BSTIterator) Next() int {
	n := it.st[len(it.st)-1]
	it.st = it.st[:len(it.st)-1]
	it.pushLeft(n.Right) // the successor lives in the right subtree's left spine
	return n.Val
}

// kthSmallest (1-based): stop the inorder walk at the kth visit. O(h + k).
func kthSmallest(root *TreeNode, k int) int {
	it := NewBSTIterator(root)
	v := 0
	for ; k > 0; k-- {
		v = it.Next()
	}
	return v
}

// isValidBST: inorder of a BST must be strictly increasing.
func isValidBST(root *TreeNode) bool {
	it := NewBSTIterator(root)
	first, prev := true, 0
	for it.HasNext() {
		v := it.Next()
		if !first && v <= prev {
			return false
		}
		prev, first = v, false
	}
	return true
}
// Paused inorder walk: the stack holds the "left spine" still to be visited.
// next() is O(1) amortized, memory O(h).
class BSTIterator {
    stack<TreeNode*> st;
    void pushLeft(TreeNode* n) { while (n) { st.push(n); n = n->left; } }
public:
    explicit BSTIterator(TreeNode* root) { pushLeft(root); }
    bool hasNext() const { return !st.empty(); }
    int next() {
        TreeNode* n = st.top(); st.pop();
        pushLeft(n->right);              // the successor lives in the right subtree's left spine
        return n->val;
    }
};

// Kth smallest (1-based): stop the inorder walk at the kth visit. O(h + k).
int kthSmallest(TreeNode* root, int k) {
    BSTIterator it(root);
    int v = 0;
    while (k-- > 0) v = it.next();
    return v;
}

// Validate with the same walk: inorder must be strictly increasing.
bool isValidBST(TreeNode* root) {
    BSTIterator it(root);
    bool first = true;
    long long prev = 0;
    while (it.hasNext()) {
        long long v = it.next();
        if (!first && v <= prev) return false;
        prev = v; first = false;
    }
    return true;
}
// Paused inorder walk: the stack holds the "left spine" still to be visited.
// next() is O(1) amortized, memory O(h).
static class BSTIterator {
    private final Deque<TreeNode> st = new ArrayDeque<>();
    BSTIterator(TreeNode root) { pushLeft(root); }
    private void pushLeft(TreeNode n) { while (n != null) { st.push(n); n = n.left; } }
    boolean hasNext() { return !st.isEmpty(); }
    int next() {
        TreeNode n = st.pop();
        pushLeft(n.right);              // the successor lives in the right subtree's left spine
        return n.val;
    }
}

class BSTIterator:
    """Paused inorder walk: the stack holds the left spine still to be visited.
    next() is O(1) amortized, memory O(h)."""

    def __init__(self, root: Optional[TreeNode]):
        self.st: List[TreeNode] = []
        self._push_left(root)

    def _push_left(self, n: Optional[TreeNode]) -> None:
        while n:
            self.st.append(n)
            n = n.left

    def has_next(self) -> bool:
        return bool(self.st)

    def next(self) -> int:
        n = self.st.pop()
        self._push_left(n.right)  # the successor lives in the right subtree's left spine
        return n.val


def kth_smallest(root: TreeNode, k: int) -> int:
    """Stop the inorder walk at the kth visit (1-based). O(h + k)."""
    it = BSTIterator(root)
    for _ in range(k - 1):
        it.next()
    return it.next()


def is_valid_bst(root: Optional[TreeNode]) -> bool:
    """Inorder of a BST must be strictly increasing."""
    it, prev = BSTIterator(root), None
    while it.has_next():
        v = it.next()
        if prev is not None and v <= prev:
            return False
        prev = v
    return True

Follow-up you should expect on kth smallest: "the tree is modified often and kth is asked often". Store a subtree size in each node, updated on insert and delete. At a node, if k equals size(left) + 1 you found it; if smaller go left; else go right with k reduced by size(left) + 1. That is O(h) per query (an order-statistic tree).
