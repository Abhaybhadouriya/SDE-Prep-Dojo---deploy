Build a balanced BST, rebalance one, find the closest value

From a sorted array (LeetCode 108): the middle element becomes the root, the left half builds the left subtree and the right half the right subtree. Every level halves the range, so the height is about log2 n and the tree is height-balanced. O(n) time, O(log n) recursion.

From a sorted linked list (LeetCode 109): either copy to an array, or build in inorder: count n, recurse on the left half first, then take the current list node as the root and advance the list pointer, then recurse on the right half. That consumes the list in order with no random access, O(n).

Rebalance an existing BST (LeetCode 1382): inorder into an array (sorted for free), then build from the array. O(n). No rotations needed in an interview.

Closest value to a target (LeetCode 270): one downward walk. At each node update the best answer, then go left if the target is smaller, right otherwise. The closest key is always on the search path of the target, because any key off the path is on the far side of some node on the path. Watch the tie rule (the smaller value wins) and the fact that the target is a real number.

Closest k values (LeetCode 272, Hard): inorder gives the keys sorted, so it becomes "k closest in a sorted array". Three answers in increasing quality: collect inorder and use a sliding window of size k (O(n)); keep a deque of size k during inorder and stop as soon as a new key is farther than the oldest (O(n) worst, early exit in practice); or use two stacks, a predecessor stack and a successor stack seeded by a search for the target, and merge like two pointers (O(h + k)).

// recoverTree: exactly two nodes were swapped; find the inversions in inorder.
func recoverTree(root *TreeNode) {
	var first, second, prev *TreeNode
	st := []*TreeNode{}
	cur := root
	for cur != nil || len(st) > 0 {
		for cur != nil {
			st = append(st, cur)
			cur = cur.Left
		}
		cur = st[len(st)-1]
		st = st[:len(st)-1]
		if prev != nil && prev.Val > cur.Val { // an inversion
			if first == nil {
				first = prev // 1st inversion: the big one is wrong
			}
			second = cur // last inversion: the small one is wrong
		}
		prev, cur = cur, cur.Right
	}
	first.Val, second.Val = second.Val, first.Val
}

// build a balanced BST from sorted a[lo..hi]: the middle becomes the root.
func build(a []int, lo, hi int) *TreeNode {
	if lo > hi {
		return nil
	}
	mid := lo + (hi-lo)/2
	return &TreeNode{Val: a[mid], Left: build(a, lo, mid-1), Right: build(a, mid+1, hi)}
}

// closestValue: one root-to-leaf walk, O(h). Ties go to the smaller value.
func closestValue(root *TreeNode, target float64) int {
	best := root.Val
	for root != nil {
		d := math.Abs(float64(root.Val) - target)
		bd := math.Abs(float64(best) - target)
		if d < bd || (d == bd && root.Val < best) {
			best = root.Val
		}
		if target < float64(root.Val) {
			root = root.Left
		} else {
			root = root.Right
		}
	}
	return best
}
// Recover a BST where exactly two nodes were swapped: find the inversions in inorder.
void recoverTree(TreeNode* root) {
    TreeNode *first = nullptr, *second = nullptr, *prev = nullptr;
    stack<TreeNode*> st;
    TreeNode* cur = root;
    while (cur || !st.empty()) {
        while (cur) { st.push(cur); cur = cur->left; }
        cur = st.top(); st.pop();
        if (prev && prev->val > cur->val) {   // an inversion
            if (!first) first = prev;         // 1st inversion: the big one is wrong
            second = cur;                     // last inversion: the small one is wrong
        }
        prev = cur;
        cur = cur->right;
    }
    swap(first->val, second->val);
}

// Balanced BST from a sorted array: the middle becomes the root, recurse on halves.
TreeNode* build(const vector<int>& a, int lo, int hi) {   // [lo, hi]
    if (lo > hi) return nullptr;
    int mid = lo + (hi - lo) / 2;
    TreeNode* n = new TreeNode(a[mid]);
    n->left = build(a, lo, mid - 1);
    n->right = build(a, mid + 1, hi);
    return n;
}

// Closest value to a real target: one root-to-leaf walk, O(h).
int closestValue(TreeNode* root, double target) {
    int best = root->val;
    while (root) {
        double d = fabs(root->val - target), bd = fabs(best - target);
        if (d < bd || (d == bd && root->val < best)) best = root->val;  // tie -> smaller
        root = target < root->val ? root->left : root->right;
    }
    return best;
}
// Recover a BST where exactly two nodes were swapped: find the inversions in inorder.
static void recoverTree(TreeNode root) {
    TreeNode first = null, second = null, prev = null, cur = root;
    Deque<TreeNode> st = new ArrayDeque<>();
    while (cur != null || !st.isEmpty()) {
        while (cur != null) { st.push(cur); cur = cur.left; }
        cur = st.pop();
        if (prev != null && prev.val > cur.val) {  // an inversion
            if (first == null) first = prev;       // 1st inversion: the big one is wrong
            second = cur;                          // last inversion: the small one is wrong
        }
        prev = cur;
        cur = cur.right;
    }
    int t = first.val; first.val = second.val; second.val = t;
}

// Balanced BST from a sorted array: the middle becomes the root, recurse on halves.
static TreeNode build(int[] a, int lo, int hi) {   // [lo, hi]
    if (lo > hi) return null;
    int mid = lo + (hi - lo) / 2;
    TreeNode n = new TreeNode(a[mid]);
    n.left = build(a, lo, mid - 1);
    n.right = build(a, mid + 1, hi);
    return n;
}

def recover_tree(root: TreeNode) -> None:
    """Exactly two nodes were swapped: find the inversions in inorder, swap back."""
    first = second = prev = None
    st, cur = [], root
    while cur or st:
        while cur:
            st.append(cur)
            cur = cur.left
        cur = st.pop()
        if prev and prev.val > cur.val:  # an inversion
            if not first:
                first = prev             # 1st inversion: the big one is wrong
            second = cur                 # last inversion: the small one is wrong
        prev, cur = cur, cur.right
    first.val, second.val = second.val, first.val


def build(a: List[int], lo: int, hi: int) -> Optional[TreeNode]:
    """Balanced BST from sorted a[lo..hi]: the middle becomes the root."""
    if lo > hi:
        return None
    mid = (lo + hi) // 2
    return TreeNode(a[mid], build(a, lo, mid - 1), build(a, mid + 1, hi))


def closest_value(root: TreeNode, target: float) -> int:
    """One root-to-leaf walk, O(h). Ties go to the smaller value."""
    best = root.val
    while root:
        if (abs(root.val - target), root.val) < (abs(best - target), best):
            best = root.val
        root = root.left if target < root.val else root.right
    return best
