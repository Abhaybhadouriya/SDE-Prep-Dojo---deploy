The counter-based backtracking in four languages, with one shared buffer. The drivers in coding-code/generate-parentheses/ check the exact lists for n = 1, 2, 3, and for every n up to 8 that the count equals the Catalan number, every string is valid, and there are no duplicates.

func generateParenthesis(n int) []string {
	var out []string
	cur := make([]byte, 0, 2*n) // one shared buffer: append, recurse, shrink
	var build func(open, close int)
	build = func(open, close int) {
		if len(cur) == 2*n { // every '(' is matched: a full answer
			out = append(out, string(cur))
			return
		}
		if open < n { // rule 1: still have '(' left to place
			cur = append(cur, '(')
			build(open+1, close)
			cur = cur[:len(cur)-1]
		}
		if close < open { // rule 2: a ')' needs an unmatched '(' before it
			cur = append(cur, ')')
			build(open, close+1)
			cur = cur[:len(cur)-1]
		}
	}
	build(0, 0)
	return out // already in lexicographic order ('(' < ')')
}
class Solution {
    vector<string> out;
    string cur;  // one shared buffer: push a char, recurse, pop it

    void build(int open, int close, int n) {
        if ((int)cur.size() == 2 * n) {          // every '(' is matched: a full answer
            out.push_back(cur);
            return;
        }
        if (open < n) {                          // rule 1: still have '(' left to place
            cur.push_back('(');
            build(open + 1, close, n);
            cur.pop_back();
        }
        if (close < open) {                      // rule 2: a ')' needs an unmatched '(' before it
            cur.push_back(')');
            build(open, close + 1, n);
            cur.pop_back();
        }
    }

public:
    vector<string> generateParenthesis(int n) {
        out.clear();
        cur.clear();
        build(0, 0, n);
        return out;                              // already in lexicographic order ('(' < ')')
    }
};

static class Solution {
    public List<String> generateParenthesis(int n) {
        List<String> out = new ArrayList<>();
        build(new StringBuilder(), 0, 0, n, out);
        return out;                              // already in lexicographic order ('(' < ')')
    }

    private void build(StringBuilder cur, int open, int close, int n, List<String> out) {
        if (cur.length() == 2 * n) {             // every '(' is matched: a full answer
            out.add(cur.toString());
            return;
        }
        if (open < n) {                          // rule 1: still have '(' left to place
            cur.append('(');
            build(cur, open + 1, close, n, out);
            cur.deleteCharAt(cur.length() - 1);
        }
        if (close < open) {                      // rule 2: a ')' needs an unmatched '(' before it
            cur.append(')');
            build(cur, open, close + 1, n, out);
            cur.deleteCharAt(cur.length() - 1);
        }
    }
}
class Solution:
    def generateParenthesis(self, n: int) -> List[str]:
        out, cur = [], []  # cur is one shared buffer: append, recurse, pop

        def build(open_: int, close: int) -> None:
            if len(cur) == 2 * n:          # every '(' is matched: a full answer
                out.append("".join(cur))
                return
            if open_ < n:                  # rule 1: still have '(' left to place
                cur.append("(")
                build(open_ + 1, close)
                cur.pop()
            if close < open_:              # rule 2: a ')' needs an unmatched '(' before it
                cur.append(")")
                build(open_, close + 1)
                cur.pop()

        build(0, 0)
        return out                         # already in lexicographic order ('(' < ')')
