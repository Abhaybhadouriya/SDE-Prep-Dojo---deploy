class Solution {
    // Greedy: load parcels in order, open a new day when the next one does not fit.
    static int daysNeeded(const vector<int>& weights, int cap) {
        int used = 1, load = 0;
        for (int w : weights) {
            if (load + w > cap) { ++used; load = 0; }
            load += w;
        }
        return used;
    }
public:
    int shipWithinDays(vector<int>& weights, int days) {
        int lo = 0, hi = 0;
        for (int w : weights) { lo = max(lo, w); hi += w; }   // sum <= 5e4 * 500 = 2.5e7
        while (lo < hi) {                                      // first cap that fits in `days`
            int mid = lo + (hi - lo) / 2;
            if (daysNeeded(weights, mid) <= days) hi = mid;    // feasible: try smaller
            else lo = mid + 1;                                 // too small: go larger
        }
        return lo;
    }
};

class Solution:
    def shipWithinDays(self, weights: List[int], days: int) -> int:
        def days_needed(cap: int) -> int:
            used, load = 1, 0
            for w in weights:
                if load + w > cap:      # this parcel does not fit today: start a new day
                    used += 1
                    load = 0
                load += w
            return used

        lo, hi = max(weights), sum(weights)   # lo: the heaviest parcel must fit; hi: one day
        while lo < hi:                        # find the first cap with days_needed(cap) <= days
            mid = (lo + hi) // 2
            if days_needed(mid) <= days:
                hi = mid                      # mid works: the answer is mid or smaller
            else:
                lo = mid + 1                  # mid too small: the answer is larger
        return lo
