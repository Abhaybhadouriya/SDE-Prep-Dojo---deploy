class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        // best[a] = fewest coins that sum to exactly a; amount + 1 means "not reachable"
        // (no answer can use more than `amount` coins, since every coin is >= 1).
        const int UNREACHED = amount + 1;
        vector<int> best(amount + 1, UNREACHED);
        best[0] = 0;                                   // zero coins make 0
        for (int a = 1; a <= amount; a++)
            for (int c : coins)
                if (c <= a && best[a - c] + 1 < best[a])
                    best[a] = best[a - c] + 1;         // last coin is c
        return best[amount] == UNREACHED ? -1 : best[amount];
    }
};
func coinChange(coins []int, amount int) int {
	// best[a] = fewest coins that sum to exactly a; amount+1 means "not reachable"
	// (no answer can use more than amount coins, since every coin is >= 1).
	unreached := amount + 1
	best := make([]int, amount+1)
	for a := 1; a <= amount; a++ {
		best[a] = unreached // best[0] stays 0: zero coins make 0
	}
	for a := 1; a <= amount; a++ {
		for _, c := range coins {
			if c <= a && best[a-c]+1 < best[a] {
				best[a] = best[a-c] + 1 // last coin is c
			}
		}
	}
	if best[amount] == unreached {
		return -1
	}
	return best[amount]
}

class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        # best[a] = fewest coins that sum to exactly a; amount + 1 means "not reachable"
        # (no answer can use more than `amount` coins, since every coin is >= 1).
        UNREACHED = amount + 1
        best = [0] + [UNREACHED] * amount          # zero coins make 0
        for a in range(1, amount + 1):
            for c in coins:
                if c <= a and best[a - c] + 1 < best[a]:
                    best[a] = best[a - c] + 1      # last coin is c
        return -1 if best[amount] == UNREACHED else best[amount]
