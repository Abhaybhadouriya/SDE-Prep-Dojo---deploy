Making it fast: path compression and union by rank

Path compression

While answering find(x), you visit every vertex between x and the root. Afterwards, point each of them directly at the root. The work you did once is never repeated for those vertices.

  1. Without any balancing, unions can build a chain: parent = [0, 0, 1, 2, 3]. A find(4) must climb four links, and a chain of n costs O(n) per find.
  2. First pass: follow parent pointers from 4 up to the vertex that is its own parent. The root is 0.
  3. Second pass: walk the same path again and point every vertex on it straight at the root. Vertices 2, 3 and 4 now have parent 0.
  4. The tree is flat. Any later find on these vertices is one hop. Combined with union by rank, the amortised cost per operation is O(α(n)), which is at most 4 for any input you will ever see.

There are three common ways to write it, all valid: the two-pass loop used in the templates below, the one-line recursive version parent[x] = find(parent[x]) (watch the recursion depth on long chains in Python), and path halving (parent[x] = parent[parent[x]]; x = parent[x] in the loop), which is a single pass and nearly as good.

Union by rank (or by size)

When merging two roots, hang the shorter tree under the taller one, so the combined height does not grow. Rank is an upper bound on height: it starts at 0, and only increases (by one) when two trees of equal rank are joined. Union by size (hang the smaller set under the bigger) gives the same guarantee and has a bonus: the size of every set is available at its root for free, which several problems need.

How fast is it?

VersionCost per operation
NaiveO(n) worst case
Union by rank or size onlyO(log n) worst case
Path compression onlyO(log n) amortised
BothO(α(n)) amortised, where α is the inverse Ackermann function

α(n) grows so slowly that it is at most 4 for any n that fits in the universe, so in an interview say "amortised nearly constant, O(α(n))" and move on. The proof (Tarjan, 1975) is not expected; knowing that you need both optimisations for that bound is.

// 1. Disjoint set union: path compression + union by rank, plus sizes and a component count.
type DSU struct {
	parent, rank, size []int
	count              int
}

func NewDSU(n int) *DSU {
	d := &DSU{parent: make([]int, n), rank: make([]int, n), size: make([]int, n), count: n}
	for i := range d.parent {
		d.parent[i], d.size[i] = i, 1 // every element is its own root
	}
	return d
}

func (d *DSU) Find(x int) int {
	root := x
	for d.parent[root] != root {
		root = d.parent[root]
	}
	for d.parent[x] != root { // second pass: point the whole path at the root
		d.parent[x], x = root, d.parent[x]
	}
	return root
}

// Union returns false if a and b were already in the same set.
func (d *DSU) Union(a, b int) bool {
	ra, rb := d.Find(a), d.Find(b)
	if ra == rb {
		return false
	}
	if d.rank[ra] < d.rank[rb] {
		ra, rb = rb, ra // shorter tree goes under the taller one
	}
	d.parent[rb] = ra
	d.size[ra] += d.size[rb]
	if d.rank[ra] == d.rank[rb] {
		d.rank[ra]++
	}
	d.count--
	return true
}
// 1. Disjoint set union: path compression + union by rank, plus sizes and a component count.
struct DSU {
    vector<int> parent, rnk, sz;
    int count;
    explicit DSU(int n) : parent(n), rnk(n, 0), sz(n, 1), count(n) {
        iota(parent.begin(), parent.end(), 0);       // every element is its own root
    }
    int find(int x) {
        int root = x;
        while (parent[root] != root) root = parent[root];
        while (parent[x] != root) { int next = parent[x]; parent[x] = root; x = next; }  // compress
        return root;
    }
    bool unite(int a, int b) {                       // false if already in the same set
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;
        if (rnk[ra] < rnk[rb]) swap(ra, rb);         // shorter tree goes under the taller one
        parent[rb] = ra;
        sz[ra] += sz[rb];
        if (rnk[ra] == rnk[rb]) rnk[ra]++;
        count--;
        return true;
    }
};

static class DSU {
    int[] parent, rank, size;
    int count;
    DSU(int n) {
        parent = new int[n]; rank = new int[n]; size = new int[n]; count = n;
        for (int i = 0; i < n; i++) { parent[i] = i; size[i] = 1; }   // each is its own root
    }
    int find(int x) {
        int root = x;
        while (parent[root] != root) root = parent[root];
        while (parent[x] != root) { int next = parent[x]; parent[x] = root; x = next; }  // compress
        return root;
    }
    boolean union(int a, int b) {                 // false if already in the same set
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;
        if (rank[ra] < rank[rb]) { int t = ra; ra = rb; rb = t; }   // shorter under taller
        parent[rb] = ra;
        size[ra] += size[rb];
        if (rank[ra] == rank[rb]) rank[ra]++;
        count--;
        return true;
    }
}
# 1. Disjoint set union: path compression + union by rank, plus sizes and a component count.
class DSU:
    def __init__(self, n):
        self.parent = list(range(n))   # every element starts as its own root
        self.rank = [0] * n            # upper bound on tree height
        self.size = [1] * n            # valid at roots only
        self.count = n                 # number of components

    def find(self, x):
        root = x
        while self.parent[root] != root:
            root = self.parent[root]
        while self.parent[x] != root:  # second pass: point the whole path at the root
            self.parent[x], x = root, self.parent[x]
        return root

    def union(self, a, b):
        """Merge the sets of a and b. Returns False if they were already together."""
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        if self.rank[ra] < self.rank[rb]:
            ra, rb = rb, ra            # hang the shorter tree under the taller one
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        if self.rank[ra] == self.rank[rb]:
            self.rank[ra] += 1
        self.count -= 1
        return True
