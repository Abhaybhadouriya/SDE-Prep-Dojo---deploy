Making a Large Island: size per root

The problem (restated). You get an n x n grid of 0s (water) and 1s (land). You may change at most one 0 into a 1. Return the size of the largest island possible afterwards, where an island is a group of 1s joined horizontally or vertically. Limits: 1 ≤ n ≤ 500. References: LeetCode 827, editorial, TUF G-52.

Trying every 0 and flood-filling is O(n4), too slow at n = 500. Instead, do the expensive work once:

  1. Union all adjacent land cells. Union by size means each root knows its island's size.
  2. For each 0, look at its four neighbours and collect the distinct roots among the land ones.
  3. Candidate = 1 + sum of those roots' sizes. Keep the maximum.
  4. If the grid has no 0 at all, the answer is n2: take the largest existing island size too, which the templates do by also scanning land cells.
// 5. Making a large island: size every island once, then try flipping each 0.
func largestIsland(grid [][]int) int {
	n := len(grid)
	d := NewDSU(n * n)
	for r := 0; r < n; r++ {
		for c := 0; c < n; c++ {
			if grid[r][c] == 0 {
				continue
			}
			if r+1 < n && grid[r+1][c] == 1 {
				d.Union(r*n+c, (r+1)*n+c)
			}
			if c+1 < n && grid[r][c+1] == 1 {
				d.Union(r*n+c, r*n+c+1)
			}
		}
	}
	dirs := [4][2]int{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
	best := 0
	for r := 0; r < n; r++ {
		for c := 0; c < n; c++ {
			if grid[r][c] == 1 {
				if s := d.size[d.Find(r*n+c)]; s > best {
					best = s
				}
				continue
			}
			seen := map[int]bool{} // distinct islands around this 0
			total := 1
			for _, dd := range dirs {
				nr, nc := r+dd[0], c+dd[1]
				if nr < 0 || nr >= n || nc < 0 || nc >= n || grid[nr][nc] == 0 {
					continue
				}
				root := d.Find(nr*n + nc)
				if !seen[root] {
					seen[root] = true
					total += d.size[root]
				}
			}
			if total > best {
				best = total
			}
		}
	}
	return best
}
// 5. Making a large island: size every island once, then try flipping each 0.
int largestIsland(const vector<vector<int>>& grid) {
    int n = grid.size(), best = 0;
    DSU dsu(n * n);
    for (int r = 0; r < n; r++)
        for (int c = 0; c < n; c++) {
            if (!grid[r][c]) continue;
            if (r + 1 < n && grid[r + 1][c]) dsu.unite(r * n + c, (r + 1) * n + c);
            if (c + 1 < n && grid[r][c + 1]) dsu.unite(r * n + c, r * n + c + 1);
        }
    const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    for (int r = 0; r < n; r++)
        for (int c = 0; c < n; c++) {
            if (grid[r][c]) { best = max(best, dsu.sz[dsu.find(r * n + c)]); continue; }
            int roots[4], k = 0, total = 1;          // distinct islands around this 0
            for (int d = 0; d < 4; d++) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr < 0 || nr >= n || nc < 0 || nc >= n || !grid[nr][nc]) continue;
                int root = dsu.find(nr * n + nc);
                if (find(roots, roots + k, root) == roots + k) { roots[k++] = root; total += dsu.sz[root]; }
            }
            best = max(best, total);
        }
    return best;
}
// 5. Making a large island: size every island once, then try flipping each 0.
static int largestIsland(int[][] grid) {
    int n = grid.length, best = 0;
    DSU dsu = new DSU(n * n);
    for (int r = 0; r < n; r++)
        for (int c = 0; c < n; c++) {
            if (grid[r][c] == 0) continue;
            if (r + 1 < n && grid[r + 1][c] == 1) dsu.union(r * n + c, (r + 1) * n + c);
            if (c + 1 < n && grid[r][c + 1] == 1) dsu.union(r * n + c, r * n + c + 1);
        }
    int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
    for (int r = 0; r < n; r++)
        for (int c = 0; c < n; c++) {
            if (grid[r][c] == 1) { best = Math.max(best, dsu.size[dsu.find(r * n + c)]); continue; }
            Set<Integer> roots = new HashSet<>();  // distinct islands around this 0
            int total = 1;
            for (int[] d : dirs) {
                int nr = r + d[0], nc = c + d[1];
                if (nr < 0 || nr >= n || nc < 0 || nc >= n || grid[nr][nc] == 0) continue;
                int root = dsu.find(nr * n + nc);
                if (roots.add(root)) total += dsu.size[root];
            }
            best = Math.max(best, total);
        }
    return best;
}
# 5. Making a large island: size every island once, then try flipping each 0.
def largest_island(grid):
    n = len(grid)
    dsu = DSU(n * n)
    for r in range(n):
        for c in range(n):
            if grid[r][c] == 1:
                if r + 1 < n and grid[r + 1][c] == 1:
                    dsu.union(r * n + c, (r + 1) * n + c)
                if c + 1 < n and grid[r][c + 1] == 1:
                    dsu.union(r * n + c, r * n + c + 1)
    best = 0
    for r in range(n):
        for c in range(n):
            if grid[r][c] == 1:
                best = max(best, dsu.size[dsu.find(r * n + c)])
                continue
            roots = set()              # distinct islands around this 0
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if 0 <= nr < n and 0 <= nc < n and grid[nr][nc] == 1:
                    roots.add(dsu.find(nr * n + nc))
            best = max(best, 1 + sum(dsu.size[x] for x in roots))
    return best

Complexity. O(n2 α(n2)) time and O(n2) space. The DFS alternative is just as good: paint each island with an id and record size[id], then do the same neighbour scan. Mentioning both, and that the de-duplication is the crux, is what interviewers look for.
