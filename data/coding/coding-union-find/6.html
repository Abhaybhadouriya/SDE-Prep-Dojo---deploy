Number of Islands II: components that change online

The problem (restated). A map of m x n cells starts as all water. You receive a list of positions [r, c]; each one turns that cell into land. After each addition, report the number of islands (groups of land cells joined horizontally or vertically). A position may repeat. Limits: m, n and the number of positions up to 104, with m x n ≤ 104. Reference: LeetCode 305 (Premium).

This is the problem that separates people who know DSU from people who know BFS. Re-counting islands with a flood fill after each addition is O(mn) per step, O(k x mn) in total. With DSU, each new land cell is a new island (count + 1), and each successful union with a land neighbour merges two islands (count - 1). Map cell (r, c) to the integer r x n + c.

// 4. Number of islands II: land appears one cell at a time; report the island count each time.
func numIslands2(m, n int, positions [][]int) []int {
	d := NewDSU(m * n)
	land := make([]bool, m*n)
	dirs := [4][2]int{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
	out := []int{}
	islands := 0
	for _, p := range positions {
		idx := p[0]*n + p[1]
		if !land[idx] { // a repeated position changes nothing
			land[idx] = true
			islands++ // a new island ...
			for _, dd := range dirs {
				nr, nc := p[0]+dd[0], p[1]+dd[1]
				if nr < 0 || nr >= m || nc < 0 || nc >= n || !land[nr*n+nc] {
					continue
				}
				if d.Union(idx, nr*n+nc) {
					islands-- // ... until it merges
				}
			}
		}
		out = append(out, islands)
	}
	return out
}
// 4. Number of islands II: land appears one cell at a time; report the island count each time.
vector<int> numIslands2(int m, int n, const vector<vector<int>>& positions) {
    DSU dsu(m * n);
    vector<char> land(m * n, 0);
    vector<int> out;
    int islands = 0;
    const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    for (const auto& p : positions) {
        int r = p[0], c = p[1], idx = r * n + c;
        if (!land[idx]) {                            // a repeated position changes nothing
            land[idx] = 1;
            islands++;                               // a new island ...
            for (int k = 0; k < 4; k++) {
                int nr = r + dr[k], nc = c + dc[k];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n || !land[nr * n + nc]) continue;
                if (dsu.unite(idx, nr * n + nc)) islands--;   // ... until it merges
            }
        }
        out.push_back(islands);
    }
    return out;
}

static List<Integer> numIslands2(int m, int n, int[][] positions) {
    DSU dsu = new DSU(m * n);
    boolean[] land = new boolean[m * n];
    List<Integer> out = new ArrayList<>();

        if (!land[idx]) {                         // a repeated position changes nothing
            land[idx] = true;

                int nr = p[0] + d[0], nc = p[1] + d[1];

}
# 4. Number of islands II: land appears one cell at a time; report the island count each time.
def num_islands2(m, n, positions):
    dsu = DSU(m * n)
    land = [False] * (m * n)
    islands, out = 0, []
    for r, c in positions:
        idx = r * n + c
        if not land[idx]:              # a repeated position changes nothing
            land[idx] = True
            islands += 1               # new cell is a new island ...
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                j = nr * n + nc
                if 0 <= nr < m and 0 <= nc < n and land[j] and dsu.union(idx, j):
                    islands -= 1       # ... until it merges with a neighbour
        out.append(islands)
    return out

Complexity. O(mn) to initialise, then O(α(mn)) per position: O(mn + k α(mn)) total. Two details cost people the problem: a repeated position must not add another island, and you must only union with neighbours that are already land. Counting with union's return value rather than comparing roots by hand keeps the count exact even when a new cell touches the same island from two sides.
