class Solution {
public:
    int calculate(string s) {
        long long total = 0;   // sum of all finished terms (64-bit: it may pass 2^31 midway)
        long long last = 0;    // the current term: * and / may still change it
        long long num = 0;     // number being read digit by digit
        char op = '+';         // operator that sits BEFORE num
        int n = s.size();
        for (int i = 0; i < n; i++) {
            char c = s[i];
            if (isdigit(c)) num = num * 10 + (c - '0');
            if (c == '+' || c == '-' || c == '*' || c == '/' || i == n - 1) {
                if (op == '+')      { total += last; last = num; }
                else if (op == '-') { total += last; last = -num; }
                else if (op == '*') { last *= num; }
                else                { last /= num; }   // C++ division truncates toward zero
                op = c;
                num = 0;
            }
            // spaces fall through and are skipped
        }
        return (int)(total + last);
    }
};
func calculate(s string) int {
	total, last, num := 0, 0, 0 // Go int is 64-bit on LeetCode, so total may pass 2^31 safely
	op := byte('+')             // operator that sits BEFORE num
	n := len(s)
	for i := 0; i < n; i++ {
		c := s[i]
		if c >= '0' && c <= '9' {
			num = num*10 + int(c-'0')
		}
		if c == '+' || c == '-' || c == '*' || c == '/' || i == n-1 {
			switch op {
			case '+':
				total += last
				last = num
			case '-':
				total += last
				last = -num
			case '*':
				last *= num
			case '/':
				last /= num // Go division truncates toward zero
			}
			op = c
			num = 0
		}
		// spaces fall through and are skipped
	}
	return total + last
}

class Solution:
    def calculate(self, s: str) -> int:
        total = 0          # sum of all terms that are finished
        last = 0           # the current term: * and / may still change it
        num = 0            # number being read digit by digit
        op = "+"           # operator that sits BEFORE num
        n = len(s)
        for i, c in enumerate(s):
            if c.isdigit():
                num = num * 10 + (ord(c) - ord("0"))
            if c in "+-*/" or i == n - 1:      # num is complete: apply the operator before it
                if op == "+":
                    total += last
                    last = num
                elif op == "-":
                    total += last
                    last = -num
                elif op == "*":
                    last *= num
                else:                           # '/' truncates toward zero; Python // floors
                    q = abs(last) // num
                    last = q if last >= 0 else -q
                op = c
                num = 0
            # spaces fall through and are skipped
        return total + last
