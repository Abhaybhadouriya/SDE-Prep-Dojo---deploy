Wildcard and regex matching

Problem (restated from LeetCode 44, Wildcard Matching). Given a string s and a pattern p, return whether p matches all of s. In p, ? matches any one character and * matches any run of characters, including an empty run. Example: s = "kernel", p = "k*n?l" is true (* = "er", ? = "e"); p = "*x" is false. Limits: lengths 0 to 2000; s has lowercase letters, p has lowercase letters, ? and *.

Problem (restated from LeetCode 10, Regular Expression Matching). Same idea with different symbols. . matches any one character, and * means "zero or more copies of the character just before it". The match must cover all of s. Example: s = "xxy", p = "z*x*y" is true (z* matches nothing, x* matches "xx"); s = "ab", p = ".*c" is false. Limits: 1 to 20 characters each; every * has a valid character before it.

For both, let dp[i][j] = "does the suffix s[i..) match the suffix p[j..)?", and fill from the end. Suffix form makes the star easy to read; the prefix form works just as well if you prefer it. dp[n][m] = true: empty matches empty.

  • Plain character, ? or .: it must match s[i], then move both: dp[i+1][j+1].
  • Wildcard *: match nothing (dp[i][j+1]) or eat one character and stay on the star (dp[i+1][j]).
  • Regex x*: read x* as one token. Skip it (dp[i][j+2]) or, if s[i] matches x, eat one character and stay (dp[i+1][j]).
// Regex with '.' and '*'. dp[i][j]: does s[i:] match p[j:]? Filled from the end.
func isMatch(s, p string) bool {
	n, m := len(s), len(p)
	dp := make([][]bool, n+1)
	for i := range dp {
		dp[i] = make([]bool, m+1)
	}
	dp[n][m] = true
	for i := n; i >= 0; i-- {
		for j := m - 1; j >= 0; j-- {
			first := i < n && (p[j] == s[i] || p[j] == '.')
			if j+1 < m && p[j+1] == '*' {
				dp[i][j] = dp[i][j+2] || (first && dp[i+1][j]) // zero times | eat one
			} else {
				dp[i][j] = first && dp[i+1][j+1]
			}
		}
	}
	return dp[0][0]
}

// Wildcard with '?' and '*' ('*' = any run, even empty).
func isMatchWild(s, p string) bool {
	n, m := len(s), len(p)
	dp := make([][]bool, n+1)
	for i := range dp {
		dp[i] = make([]bool, m+1)
	}
	dp[n][m] = true
	for i := n; i >= 0; i-- {
		for j := m - 1; j >= 0; j-- {
			if p[j] == '*' {
				dp[i][j] = dp[i][j+1] || (i < n && dp[i+1][j])
			} else {
				dp[i][j] = i < n && (p[j] == '?' || p[j] == s[i]) && dp[i+1][j+1]
			}
		}
	}
	return dp[0][0]
}
// Regex with '.' and '*'. dp[i][j]: does s[i..) match p[j..)? Filled from the end.
bool isMatch(const string& s, const string& p) {
    int n = s.size(), m = p.size();
    vector<vector<char>> dp(n + 1, vector<char>(m + 1, 0));
    dp[n][m] = 1;                                         // empty matches empty
    for (int i = n; i >= 0; i--)
        for (int j = m - 1; j >= 0; j--) {
            bool first = i < n && (p[j] == s[i] || p[j] == '.');
            if (j + 1 < m && p[j + 1] == '*')
                dp[i][j] = dp[i][j + 2]                   // use x* zero times
                        || (first && dp[i + 1][j]);       // eat one char, keep x*
            else
                dp[i][j] = first && dp[i + 1][j + 1];
        }
    return dp[0][0];
}

        }
    return dp[0][0];
}

// Wildcard with '?' and '*' ('*' = any run, even empty).
static boolean isMatchWild(String s, String p) {
    int n = s.length(), m = p.length();
    boolean[][] dp = new boolean[n + 1][m + 1];
    dp[n][m] = true;
    for (int i = n; i >= 0; i--)
        for (int j = m - 1; j >= 0; j--) {
            char c = p.charAt(j);
            if (c == '*') dp[i][j] = dp[i][j + 1] || (i < n && dp[i + 1][j]);
            else dp[i][j] = i < n && (c == '?' || c == s.charAt(i)) && dp[i + 1][j + 1];
        }
    return dp[0][0];
}
def is_match(s: str, p: str) -> bool:
    """Regex with '.' and '*'. dp[i][j]: does s[i:] match p[j:]? Filled from the end."""
    n, m = len(s), len(p)
    dp = [[False] * (m + 1) for _ in range(n + 1)]
    dp[n][m] = True
    for i in range(n, -1, -1):
        for j in range(m - 1, -1, -1):
            first = i < n and p[j] in (s[i], ".")
            if j + 1 < m and p[j + 1] == "*":
                dp[i][j] = dp[i][j + 2] or (first and dp[i + 1][j])  # zero times | eat one
            else:
                dp[i][j] = first and dp[i + 1][j + 1]
    return dp[0][0]

def is_match_wild(s: str, p: str) -> bool:
    """Wildcard with '?' and '*' ('*' = any run, even empty)."""
    n, m = len(s), len(p)
    dp = [[False] * (m + 1) for _ in range(n + 1)]
    dp[n][m] = True
    for i in range(n, -1, -1):
        for j in range(m - 1, -1, -1):
            if p[j] == "*":
                dp[i][j] = dp[i][j + 1] or (i < n and dp[i + 1][j])
            else:
                dp[i][j] = i < n and p[j] in (s[i], "?") and dp[i + 1][j + 1]
    return dp[0][0]

Note the loop bounds: i runs down from n (not n-1), because an empty rest of s can still match a pattern like a*b*. That row is where "empty string vs pattern" edge cases live, and it is the row most first attempts get wrong.

Wildcard matching also has a well-known greedy solution with O(1) extra memory: walk both strings; on * remember its position and the current s position; on a mismatch go back to the last star and let it eat one more character. It is O(n · m) in the worst case but fast in practice. Mention it as a follow-up; write the DP first, because it is easier to prove.
