Interleaving string

Problem (restated from LeetCode 97, Interleaving String). Given s1, s2 and s3, return whether s3 can be formed by weaving s1 and s2 together: every character of s3 comes from s1 or from s2, each of s1 and s2 is used fully, and each keeps its own order. Example: s1 = "ab", s2 = "xy", s3 = "axyb" is true; s3 = "ayxb" is false (y before x breaks s2's order). Limits: s1 and s2 have 0 to 100 characters, s3 up to 200. Follow-up: use only O(len(s2)) extra memory.

A greedy "take from whichever matches" fails when both match, so try both. dp[i][j] = can the first i characters of s1 and first j of s2 build the first i + j characters of s3? The next character of s3 is s3[i+j-1]; it came from s1 (arrow from above) if s1[i-1] equals it and dp[i-1][j] holds, or from s2 (arrow from the left) likewise. There is no diagonal: one step uses one character. Check n + m == len(s3) first, or the table answers a different question.

// Interleaving: dp[j] = can s1[:i] and s2[:j] build s3[:i+j]?
func isInterleave(s1, s2, s3 string) bool {
	n, m := len(s1), len(s2)
	if n+m != len(s3) {
		return false
	}
	dp := make([]bool, m+1)
	for i := 0; i <= n; i++ {
		for j := 0; j <= m; j++ {
			if i == 0 && j == 0 {
				dp[j] = true
				continue
			}
			fromS1 := i > 0 && dp[j] && s1[i-1] == s3[i+j-1]
			fromS2 := j > 0 && dp[j-1] && s2[j-1] == s3[i+j-1]
			dp[j] = fromS1 || fromS2
		}
	}
	return dp[m]
}
// Interleaving: dp[j] = can s1[0..i) and s2[0..j) build s3[0..i+j)?
bool isInterleave(const string& s1, const string& s2, const string& s3) {
    int n = s1.size(), m = s2.size();
    if (n + m != (int)s3.size()) return false;
    vector<char> dp(m + 1, 0);
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= m; j++) {
            if (i == 0 && j == 0) { dp[j] = 1; continue; }
            bool fromS1 = i > 0 && dp[j] && s1[i - 1] == s3[i + j - 1];        // dp[j] = row above
            bool fromS2 = j > 0 && dp[j - 1] && s2[j - 1] == s3[i + j - 1];    // dp[j-1] = left
            dp[j] = fromS1 || fromS2;
        }
    return dp[m];
}
// Interleaving: dp[j] = can s1[0..i) and s2[0..j) build s3[0..i+j)?
static boolean isInterleave(String s1, String s2, String s3) {
    int n = s1.length(), m = s2.length();
    if (n + m != s3.length()) return false;
    boolean[] dp = new boolean[m + 1];
    for (int i = 0; i <= n; i++)
        for (int j = 0; j <= m; j++) {
            if (i == 0 && j == 0) { dp[j] = true; continue; }
            boolean fromS1 = i > 0 && dp[j] && s1.charAt(i - 1) == s3.charAt(i + j - 1);
            boolean fromS2 = j > 0 && dp[j - 1] && s2.charAt(j - 1) == s3.charAt(i + j - 1);
            dp[j] = fromS1 || fromS2;
        }
    return dp[m];
}
def is_interleave(s1: str, s2: str, s3: str) -> bool:
    """dp[j] = can s1[:i] and s2[:j] build s3[:i+j]?"""
    n, m = len(s1), len(s2)
    if n + m != len(s3):
        return False
    dp = [False] * (m + 1)
    for i in range(n + 1):
        for j in range(m + 1):
            if i == 0 and j == 0:
                dp[j] = True
                continue
            from_s1 = i > 0 and dp[j] and s1[i - 1] == s3[i + j - 1]          # row above
            from_s2 = j > 0 and dp[j - 1] and s2[j - 1] == s3[i + j - 1]      # left
            dp[j] = from_s1 or from_s2
    return dp[m]

Another way to see it: a path from the top-left to the bottom-right corner of the grid, moving only right or down, where each move must agree with the next character of s3. That picture also explains why BFS or DFS with memo on (i, j) works, and why there are only (n+1)(m+1) states.
