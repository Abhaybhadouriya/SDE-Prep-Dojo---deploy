func longestCommonSubsequence(a string, b string) int {
	if len(a) < len(b) {
		a, b = b, a // keep the row as short as possible
	}
	n := len(b)
	// row[j] = LCS of (a's prefix so far, b[:j]); row[0] stays 0 (empty prefix of b)
	row := make([]int, n+1)
	for i := 0; i < len(a); i++ {
		diag := 0 // old row[j-1] = dp[i-1][j-1]
		for j := 1; j <= n; j++ {
			up := row[j] // dp[i-1][j], about to be overwritten
			if a[i] == b[j-1] {
				row[j] = diag + 1 // match: extend the LCS of both shorter prefixes
			} else if up > row[j-1] {
				row[j] = up // drop the last char of a
			} else {
				row[j] = row[j-1] // drop the last char of b
			}
			diag = up
		}
	}
	return row[n]
}

public:
    int longestCommonSubsequence(string a, string b) {
        if (a.size() < b.size()) swap(a, b);       // keep the row as short as possible
        int n = b.size();
        // row[j] = LCS of (a's prefix so far, b[0..j)); row[0] stays 0 (empty prefix of b)
        vector<int> row(n + 1, 0);
        for (char ca : a) {
            int diag = 0;                          // old row[j-1] = dp[i-1][j-1]
            for (int j = 1; j <= n; j++) {
                int up = row[j];                   // dp[i-1][j], about to be overwritten
                if (ca == b[j - 1])
                    row[j] = diag + 1;             // match: extend the LCS of both shorter prefixes
                else
                    row[j] = max(up, row[j - 1]);  // drop the last char of a or of b
                diag = up;
            }
        }
        return row[n];
    }
