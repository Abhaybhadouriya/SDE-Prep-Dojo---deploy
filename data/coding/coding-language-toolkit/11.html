The starter template: four idioms in four languages

This one program contains the idioms that appear most in interview code: counting with a map, sorting by a two-key comparator, a size-k heap for top-k, a monotonic deque, safe mid-point and modular multiplication. Type it out once in your chosen language from memory; if you can, you know the toolkit. All four versions were compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and pass their checks.

package main

import (
	"container/heap"
	"fmt"
	"math"
	"sort"
)

// Most frequent first; ties broken alphabetically. O(n log n).
func topKFrequent(words []string, k int) []string {
	cnt := map[string]int{} // missing key reads as 0
	for _, w := range words {
		cnt[w]++
	}
	keys := make([]string, 0, len(cnt))
	for w := range cnt { // map order is random: always sort
		keys = append(keys, w)
	}
	sort.Slice(keys, func(i, j int) bool {
		if cnt[keys[i]] != cnt[keys[j]] {
			return cnt[keys[i]] > cnt[keys[j]]
		}
		return keys[i] < keys[j]
	})
	if k > len(keys) {
		k = len(keys)
	}
	return keys[:k]
}

// MaxHeap implements heap.Interface; Less decides min or max.
type MaxHeap []int

func (h MaxHeap) Len() int           { return len(h) }
func (h MaxHeap) Less(i, j int) bool { return h[i] > h[j] } // ">" makes it a max-heap
func (h MaxHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *MaxHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *MaxHeap) Pop() any {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

// Max-heap of size k. O(n log k).
func kSmallest(a []int, k int) []int {
	h := &MaxHeap{}
	for _, x := range a {
		heap.Push(h, x)
		if h.Len() > k {
			heap.Pop(h) // drop the largest of the k+1
		}
	}
	out := append([]int(nil), (*h)...)
	sort.Ints(out)
	return out
}

// Monotonic deque of indices (a slice used as a deque), values decreasing. O(n).
func windowMax(a []int, k int) []int {
	dq := []int{}
	out := []int{}
	for i, x := range a {
		for len(dq) > 0 && a[dq[len(dq)-1]] <= x {
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, i)
		if dq[0] <= i-k {
			dq = dq[1:]
		}
		if i >= k-1 {
			out = append(out, a[dq[0]])
		}
	}
	return out
}

func mid(lo, hi int) int { return lo + (hi-lo)/2 }

using namespace std;

// Most frequent first; ties broken alphabetically. O(n log n).
vector<string> topKFrequent(const vector<string>& words, int k) {
    unordered_map<string, int> cnt;
    for (const string& w : words) cnt[w]++;          // operator[] inserts 0 first
    vector<pair<string, int>> v(cnt.begin(), cnt.end());
    sort(v.begin(), v.end(), [](const pair<string, int>& a, const pair<string, int>& b) {
        if (a.second != b.second) return a.second > b.second;   // strict: never >=
        return a.first < b.first;
    });
    vector<string> out;
    for (int i = 0; i < k && i < (int)v.size(); i++) out.push_back(v[i].first);
    return out;
}

// Max-heap of size k (priority_queue is a max-heap by default). O(n log k).
vector<int> kSmallest(const vector<int>& a, int k) {
    priority_queue<int> pq;
    for (int x : a) {
        pq.push(x);
        if ((int)pq.size() > k) pq.pop();            // drop the largest of the k+1
    }
    vector<int> out;
    while (!pq.empty()) { out.push_back(pq.top()); pq.pop(); }
    reverse(out.begin(), out.end());
    return out;
}

// Monotonic deque of indices, values decreasing. O(n).
vector<int> windowMax(const vector<int>& a, int k) {
    deque<int> dq;
    vector<int> out;
    for (int i = 0; i < (int)a.size(); i++) {
        while (!dq.empty() && a[dq.back()] <= a[i]) dq.pop_back();
        dq.push_back(i);
        if (dq.front() <= i - k) dq.pop_front();
        if (i >= k - 1) out.push_back(a[dq.front()]);
    }
    return out;
}

import java.util.*;

public class Toolkit {
    // Most frequent first; ties broken alphabetically. O(n log n).
    static List<String> topKFrequent(String[] words, int k) {
        Map<String, Integer> cnt = new HashMap<>();
        for (String w : words) cnt.merge(w, 1, Integer::sum);
        List<String> keys = new ArrayList<>(cnt.keySet());
        keys.sort((a, b) -> {
            int ca = cnt.get(a), cb = cnt.get(b);       // unbox to int before comparing
            if (ca != cb) return Integer.compare(cb, ca); // never "cb - ca" (overflow)
            return a.compareTo(b);
        });
        return keys.subList(0, Math.min(k, keys.size()));
    }

    // Max-heap of size k (PriorityQueue is a MIN-heap by default). O(n log k).
    static List<Integer> kSmallest(int[] a, int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>(Comparator.reverseOrder());
        for (int x : a) {
            pq.offer(x);
            if (pq.size() > k) pq.poll();               // drop the largest of the k+1
        }
        List<Integer> out = new ArrayList<>(pq);
        Collections.sort(out);
        return out;
    }

    // Monotonic deque of indices, values decreasing. O(n).
    static List<Integer> windowMax(int[] a, int k) {
        Deque<Integer> dq = new ArrayDeque<>();
        List<Integer> out = new ArrayList<>();
        for (int i = 0; i < a.length; i++) {
            while (!dq.isEmpty() && a[dq.peekLast()] <= a[i]) dq.pollLast();
            dq.offerLast(i);
            if (dq.peekFirst() <= i - k) dq.pollFirst();
            if (i >= k - 1) out.add(a[dq.peekFirst()]);
        }
        return out;
    }

    static int mid(int lo, int hi) { return lo + (hi - lo) / 2; }  // (lo + hi) / 2 can overflow
    static final long MOD = 1_000_000_007L;

    public static void main(String[] args) {
        String[] words = {"dns", "arp", "tcp", "dns", "arp", "dns", "udp"};
        System.out.println(topKFrequent(words, 2));
        System.out.println(kSmallest(new int[]{9, 4, 7, 1, 8, 2}, 3));
        System.out.println(windowMax(new int[]{1, 3, -1, -3, 5, 3, 6, 7}, 3));
        System.out.println(mid(0, Integer.MAX_VALUE) + " " + (123456789L * 987654321L) % MOD);
    }
}
import heapq
from collections import Counter, deque


def top_k_frequent(words, k):
    """Most frequent first; ties broken alphabetically. O(n log n)."""
    cnt = Counter(words)                       # dict word -> count
    order = sorted(cnt.items(), key=lambda kv: (-kv[1], kv[0]))
    return [w for w, _ in order[:k]]


def k_smallest(a, k):
    """Keep a max-heap of size k (negate: heapq is a min-heap). O(n log k)."""
    h = []
    for x in a:
        heapq.heappush(h, -x)
        if len(h) > k:
            heapq.heappop(h)                   # drop the largest of the k+1
    return sorted(-x for x in h)


def window_max(a, k):
    """Monotonic deque of indices, values decreasing. O(n): each index in/out once."""
    dq, out = deque(), []
    for i, x in enumerate(a):
        while dq and a[dq[-1]] <= x:
            dq.pop()                           # smaller values can never be a max again
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()                       # front slid out of the window
        if i >= k - 1:
            out.append(a[dq[0]])
    return out


def mid(lo, hi):
    return lo + (hi - lo) // 2                 # habit from C/Java; Python ints never overflow
