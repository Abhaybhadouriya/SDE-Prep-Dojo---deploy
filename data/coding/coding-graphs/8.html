Weighted shortest paths: Dijkstra and friends

When edges have different non-negative costs, BFS's "first discovery is best" argument fails: a path with more edges can be cheaper. Dijkstra's algorithm fixes this by always expanding the cheapest known vertex next, using a min-heap of (distance, vertex). When a vertex is popped for the first time, its distance is final, because every other route to it would pass through a vertex with a distance at least as large, and costs cannot go negative.

The version below is the "lazy" one: instead of a decrease-key operation, it pushes a new heap entry whenever a distance improves and skips stale entries on pop. That is how everyone writes it in Python, Java and C++ in interviews. Time O(E log E), which is O(E log V).

import (
    "container/heap"
    "math"
)

// min-heap of {dist, node} for Dijkstra
type item struct{ d, u int }
type minHeap []item

func (h minHeap) Len() int            { return len(h) }
func (h minHeap) Less(i, j int) bool  { return h[i].d < h[j].d }
func (h minHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x interface{}) { *h = append(*h, x.(item)) }
func (h *minHeap) Pop() interface{} {
    old := *h
    x := old[len(old)-1]
    *h = old[:len(old)-1]
    return x
}

// dijkstra: edges {u, v, w}, directed, w >= 0. math.MaxInt64 = unreachable.
func dijkstra(n int, edges [][3]int, src int) []int {
    type arc struct{ v, w int }
    adj := make([][]arc, n)
    for _, e := range edges {
        adj[e[0]] = append(adj[e[0]], arc{e[1], e[2]})
    }
    dist := make([]int, n)
    for i := range dist {
        dist[i] = math.MaxInt64
    }
    dist[src] = 0
    h := &minHeap{{0, src}}
    for h.Len() > 0 {
        top := heap.Pop(h).(item)
        if top.d > dist[top.u] {
            continue // stale entry
        }
        for _, a := range adj[top.u] {
            if nd := top.d + a.w; nd < dist[a.v] {
                dist[a.v] = nd
                heap.Push(h, item{nd, a.v})
            }
        }
    }
    return dist
}
// edges {u, v, w}, directed, w >= 0. LLONG_MAX = unreachable.
vector<long long> dijkstra(int n, const vector<array<int,3>>& edges, int src) {
    vector<vector<pair<int,int>>> adj(n);
    for (auto& e : edges) adj[e[0]].push_back({e[1], e[2]});
    vector<long long> dist(n, LLONG_MAX);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;  // min-heap
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        long long d = pq.top().first; int u = pq.top().second; pq.pop();
        if (d > dist[u]) continue;                  // stale entry
        for (const auto& e : adj[u]) {
            int v = e.first, w = e.second;
            if (d + w < dist[v]) { dist[v] = d + w; pq.push({dist[v], v}); }
        }
    }
    return dist;
}
// edges {u, v, w}, directed, w >= 0. Long.MAX_VALUE = unreachable.
static long[] dijkstra(int n, int[][] edges, int src) {
    List<List<int[]>> adj = new ArrayList<>();
    for (int i = 0; i < n; i++) adj.add(new ArrayList<>());
    for (int[] e : edges) adj.get(e[0]).add(new int[]{e[1], e[2]});
    long[] dist = new long[n];
    Arrays.fill(dist, Long.MAX_VALUE);
    PriorityQueue<long[]> pq = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
    dist[src] = 0;
    pq.add(new long[]{0, src});
    while (!pq.isEmpty()) {
        long[] top = pq.poll();
        long d = top[0];
        int u = (int) top[1];
        if (d > dist[u]) continue;                      // stale entry
        for (int[] e : adj.get(u))
            if (d + e[1] < dist[e[0]]) {
                dist[e[0]] = d + e[1];
                pq.add(new long[]{dist[e[0]], e[0]});
            }
    }
    return dist;
}
import heapq

def dijkstra(n, edges, src):
    """edges: (u, v, w) directed, w >= 0. Returns distances, inf = unreachable."""
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
    dist = [float('inf')] * n
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue                              # stale entry: u was already settled cheaper
        for v, w in adj[u]:
            if d + w < dist[v]:
                dist[v] = d + w
                heapq.heappush(pq, (dist[v], v))
    return dist

Two variants show up again and again. Minimax paths (Path With Minimum Effort, Swim in Rising Water): the cost of a path is its largest step, not the sum; Dijkstra still works if you replace d + w with max(d, w), and so does binary search on the answer plus BFS. Extra state (Cheapest Flights Within K Stops): the vertex alone is not enough, you need (vertex, stops used); a K-round Bellman-Ford is the cleanest answer there.
