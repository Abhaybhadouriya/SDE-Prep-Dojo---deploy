Union-find (disjoint set union)

Union-find keeps a collection of disjoint groups and answers two questions very fast: find(x) ("which group is x in?", answered by the group's root) and union(a, b) ("merge the groups of a and b"). Each group is a tree where every element points to a parent and the root points to itself.

Two small tricks make it nearly O(1) per operation:

  • Path compression (here: its cheap variant path halving): while walking up in find, point each visited node to its grandparent, so the trees stay flat.
  • Union by size (or rank): always hang the smaller tree under the root of the bigger one, so depth grows slowly.

With both, a sequence of m operations costs O(m α(n)), where α is the inverse Ackermann function: at most 4 for any n you will ever store. In an interview, say "amortised nearly constant".

type DSU struct {
    parent, size []int
    sets         int
}

func NewDSU(n int) *DSU {
    d := &DSU{parent: make([]int, n), size: make([]int, n), sets: n}
    for i := range d.parent {
        d.parent[i] = i
        d.size[i] = 1
    }
    return d
}

func (d *DSU) Find(x int) int {
    for d.parent[x] != x {
        d.parent[x] = d.parent[d.parent[x]] // path halving
        x = d.parent[x]
    }
    return x
}

func (d *DSU) Union(a, b int) bool {
    a, b = d.Find(a), d.Find(b)
    if a == b {
        return false // edge closes a cycle
    }
    if d.size[a] < d.size[b] {
        a, b = b, a
    }
    d.parent[b] = a
    d.size[a] += d.size[b]
    d.sets--
    return true
}
struct DSU {
    vector<int> parent, sz;
    int sets;
    DSU(int n) : parent(n), sz(n, 1), sets(n) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }  // path halving
        return x;
    }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;                   // already connected: edge closes a cycle
        if (sz[a] < sz[b]) swap(a, b);
        parent[b] = a;                              // smaller tree under the bigger
        sz[a] += sz[b];
        sets--;
        return true;
    }
};
static class DSU {
    int[] parent, size;
    int sets;
    DSU(int n) {
        parent = new int[n]; size = new int[n]; sets = n;
        for (int i = 0; i < n; i++) { parent[i] = i; size[i] = 1; }
    }
    int find(int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }  // path halving
        return x;
    }
    boolean union(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;                       // edge closes a cycle
        if (size[a] < size[b]) { int t = a; a = b; b = t; }
        parent[b] = a;
        size[a] += size[b];
        sets--;
        return true;
    }
}
class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n
        self.sets = n

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False                          # already connected: this edge closes a cycle
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra                      # hang the smaller tree under the bigger
        self.size[ra] += self.size[rb]
        self.sets -= 1
        return True

When to pick union-find over BFS/DFS: the edges arrive one at a time and you must answer connectivity after each one (Number of Islands II, "earliest moment everyone is connected"); you need to group items by shared keys (Accounts Merge: union each account with the first email it lists, then collect emails by root); or you need Kruskal's MST. For a fixed graph and a single "how many components" question, BFS/DFS is equally good. Union-find cannot split groups; if edges get deleted, think again (often: process the operations in reverse as additions).
