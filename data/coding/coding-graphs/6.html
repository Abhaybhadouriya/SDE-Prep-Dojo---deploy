Topological sort (dependencies)

A topological order of a directed graph lists the vertices so every edge goes from earlier to later: every course after its prerequisites. It exists exactly when the graph has no directed cycle (a DAG, "directed acyclic graph").

Kahn's algorithm is the interview favourite because it is a BFS you already know, plus one counter per vertex:

  1. Count the in-degree of every vertex (how many edges point into it).
  2. Push every vertex with in-degree 0: nothing blocks them.
  3. Pop a vertex, append it to the order, and "delete" its outgoing edges by decrementing each target's in-degree. Any target that reaches 0 is now free: push it.
  4. If the order holds all V vertices, it is valid. If fewer, the vertices left over all sit on or behind a cycle.
  1. Count in-degrees. Arrays and Recursion have no prerequisites, so both start in the queue.
  2. Pop 0 (Arrays) into the order. Its edge to Trees is removed, so Trees drops from 2 to 1: still blocked by Recursion.
  3. Pop 1 (Recursion). Trees and Backtracking both drop to 0 and join the queue.
  4. Pop 2 (Trees). Tree DP drops from 2 to 1: it still waits for Backtracking.
  5. Pop 4 (Backtracking). Tree DP reaches 0 and joins the queue.
  6. Pop 3 (Tree DP). All 5 vertices are placed, so this is a valid order. With a cycle, some in-degrees would never reach 0 and the order would come up short.
// topoSort: Kahn's algorithm. edge {u, v}: u before v. nil = cycle.
func topoSort(n int, edges [][2]int) []int {
    adj := make([][]int, n)
    indeg := make([]int, n)
    for _, e := range edges {
        adj[e[0]] = append(adj[e[0]], e[1])
        indeg[e[1]]++
    }
    q := []int{}
    for i := 0; i < n; i++ {
        if indeg[i] == 0 {
            q = append(q, i)
        }
    }
    order := []int{}
    for len(q) > 0 {
        u := q[0]
        q = q[1:]
        order = append(order, u)
        for _, v := range adj[u] {
            indeg[v]--
            if indeg[v] == 0 {
                q = append(q, v)
            }
        }
    }
    if len(order) < n {
        return nil // leftovers sit on a cycle
    }
    return order
}
// Kahn's algorithm. edge {u, v}: u before v. Empty result = cycle.
vector<int> topoSort(int n, const vector<pair<int,int>>& edges) {
    vector<vector<int>> adj(n);
    vector<int> indeg(n, 0), order;
    for (const auto& e : edges) { adj[e.first].push_back(e.second); indeg[e.second]++; }
    queue<int> q;
    for (int i = 0; i < n; i++) if (indeg[i] == 0) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u]) if (--indeg[v] == 0) q.push(v);
    }
    if ((int)order.size() < n) return {};           // leftovers sit on a cycle
    return order;
}
// Kahn's algorithm. edge {u, v}: u before v. Empty result = cycle.
static List<Integer> topoSort(int n, int[][] edges) {
    List<List<Integer>> adj = new ArrayList<>();
    for (int i = 0; i < n; i++) adj.add(new ArrayList<>());
    int[] indeg = new int[n];
    for (int[] e : edges) { adj.get(e[0]).add(e[1]); indeg[e[1]]++; }
    ArrayDeque<Integer> q = new ArrayDeque<>();
    for (int i = 0; i < n; i++) if (indeg[i] == 0) q.add(i);
    List<Integer> order = new ArrayList<>();
    while (!q.isEmpty()) {
        int u = q.poll();
        order.add(u);
        for (int v : adj.get(u)) if (--indeg[v] == 0) q.add(v);
    }
    return order.size() == n ? order : new ArrayList<>();  // leftovers sit on a cycle
}
from collections import deque

def topo_sort(n, edges):
    """Kahn's algorithm. edges: (u, v) means u must come before v. Returns [] if there is a cycle."""
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        adj[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return order if len(order) == n else []
