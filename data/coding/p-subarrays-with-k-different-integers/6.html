atMost(k) - atMost(k - 1) with a count array. Each file in coding-code/subarrays-with-k-different-integers/ runs the examples and edge cases and cross-checks 500 random arrays against the O(n2) brute force.

class Solution {
public:
    int subarraysWithKDistinct(vector<int>& nums, int k) {
        return atMost(nums, k) - atMost(nums, k - 1);
    }

private:
    // number of subarrays with at most k distinct values (values are 1..n)
    int atMost(const vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> freq(n + 1, 0);             // freq[v] = copies of v inside the window
        int distinct = 0, left = 0, count = 0;
        for (int right = 0; right < n; right++) {
            if (freq[nums[right]]++ == 0) distinct++;
            while (distinct > k) {              // one value too many: shrink from the left
                if (--freq[nums[left]] == 0) distinct--;
                left++;
            }
            count += right - left + 1;          // all starts in [left..right] are valid for this end
        }
        return count;
    }
};

func atMost(nums []int, k int) int {
	freq := make([]int, len(nums)+1) // freq[v] = copies of v inside the window
	distinct, left, count := 0, 0, 0

		if freq[x] == 0 {
			distinct++
		}
		freq[x]++
		for distinct > k { // one value too many: shrink from the left
			freq[nums[left]]--
			if freq[nums[left]] == 0 {
				distinct--
			}
			left++
		}
		count += right - left + 1 // all starts in [left..right] are valid for this end
	}
	return count
}

class Solution:
    def subarraysWithKDistinct(self, nums: List[int], k: int) -> int:
        return self.at_most(nums, k) - self.at_most(nums, k - 1)

    def at_most(self, nums: List[int], k: int) -> int:
        """Number of subarrays with at most k distinct values (values are 1..n)."""
        freq = [0] * (len(nums) + 1)    # freq[v] = copies of v inside the window
        distinct = 0
        left = 0
        count = 0
        for right, x in enumerate(nums):
            if freq[x] == 0:
                distinct += 1
            freq[x] += 1
            while distinct > k:         # one value too many: shrink from the left
                y = nums[left]
                freq[y] -= 1
                if freq[y] == 0:
                    distinct -= 1
                left += 1
            count += right - left + 1   # all starts in [left..right] are valid for this end
        return count
