All four use the reverse-all, then reverse-each-word pass with a trailing write pointer. Java, Python and Go copy the immutable string into one char/byte buffer first.

// Reverse everything, then reverse each word back while compacting spaces.
func reverseWords(s string) string {
	a := []byte(s) // Go strings are immutable: one byte buffer (input is ASCII)
	reverseBytes(a, 0, len(a)-1) // word order right, letters backwards
	n, w := len(a), 0            // w = end of the compacted result (w <= r always)
	for r := 0; r < n; {
		if a[r] == ' ' { // skip a run of spaces
			r++
			continue
		}
		if w > 0 { // exactly one space between words
			a[w] = ' '
			w++
		}
		start := w
		for r < n && a[r] != ' ' { // slide the word left
			a[w] = a[r]
			w++
			r++
		}
		reverseBytes(a, start, w-1) // fix the word's letters
	}
	return string(a[:w])
}

func reverseBytes(a []byte, i, j int) {
	for ; i < j; i, j = i+1, j-1 {
		a[i], a[j] = a[j], a[i]
	}
}
class Solution {
public:
    // In place: reverse everything, then reverse each word back while compacting spaces.
    string reverseWords(string s) {
        reverse(s.begin(), s.end());       // word order is now right, letters are backwards
        int n = s.size(), w = 0;           // w = end of the compacted result (w <= r always)
        for (int r = 0; r < n;) {
            if (s[r] == ' ') { r++; continue; }          // skip a run of spaces
            if (w > 0) s[w++] = ' ';                     // exactly one space between words
            int start = w;
            while (r < n && s[r] != ' ') s[w++] = s[r++]; // slide the word left
            reverse(s.begin() + start, s.begin() + w);   // fix the word's letters
        }
        s.resize(w);                       // drop the leftover tail
        return s;
    }
};
static class Solution {
    // Reverse everything, then reverse each word back while compacting spaces.
    public String reverseWords(String s) {
        char[] a = s.toCharArray();          // Java strings are immutable: one buffer
        int n = a.length, w = 0;             // w = end of the compacted result (w <= r)
        reverse(a, 0, n - 1);                // word order right, letters backwards
        for (int r = 0; r < n;) {
            if (a[r] == ' ') { r++; continue; }          // skip a run of spaces
            if (w > 0) a[w++] = ' ';                     // one space between words
            int start = w;
            while (r < n && a[r] != ' ') a[w++] = a[r++]; // slide the word left
            reverse(a, start, w - 1);                    // fix the word's letters
        }
        return new String(a, 0, w);
    }

    private static void reverse(char[] a, int i, int j) {
        while (i < j) { char t = a[i]; a[i++] = a[j]; a[j--] = t; }
    }
}
class Solution:
    def reverseWords(self, s: str) -> str:
        # Python strings are immutable, so work on a list of chars (the in-place idea).
        a = list(s)
        a.reverse()                     # word order right, letters backwards
        n, w, r = len(a), 0, 0          # w = end of the compacted result (w <= r always)
        while r < n:
            if a[r] == " ":             # skip a run of spaces
                r += 1
                continue
            if w > 0:                   # exactly one space between words
                a[w] = " "
                w += 1
            start = w
            while r < n and a[r] != " ":  # slide the word left
                a[w] = a[r]
                w += 1
                r += 1
            a[start:w] = a[start:w][::-1]  # fix the word's letters
        return "".join(a[:w])
