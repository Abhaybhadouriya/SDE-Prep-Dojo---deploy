Backtracking with a used[] mask in four languages. Each file in coding-code/permutations/ checks the three examples and, for every n from 1 to 6, that the output has n! distinct orderings that match std::next_permutation / itertools.permutations.

class Solution {
public:
    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>> out;
        vector<int> cur;                         // the order built so far
        vector<bool> used(nums.size(), false);   // used[i]: nums[i] is already in cur
        build(nums, used, cur, out);
        return out;
    }

private:
    // Slot number cur.size() is being filled. Any element not used yet may go
    // there, so the loop always starts at 0 (order matters, unlike subsets).
    void build(const vector<int>& nums, vector<bool>& used, vector<int>& cur,
               vector<vector<int>>& out) {
        if (cur.size() == nums.size()) {         // every slot filled: one permutation
            out.push_back(cur);
            return;
        }
        for (int i = 0; i < (int)nums.size(); i++) {
            if (used[i]) continue;               // each element appears once
            used[i] = true;  cur.push_back(nums[i]);    // choose
            build(nums, used, cur, out);
            cur.pop_back();  used[i] = false;           // un-choose
        }
    }
};
func permute(nums []int) [][]int {
	out := [][]int{}
	cur := []int{}                  // the order built so far
	used := make([]bool, len(nums)) // used[i]: nums[i] is already in cur

	// Slot number len(cur) is being filled. Any element not used yet may go
	// there, so the loop always starts at 0 (order matters, unlike subsets).
	var build func()
	build = func() {
		if len(cur) == len(nums) { // every slot filled: one permutation
			out = append(out, append([]int(nil), cur...)) // copy: cur's array is reused
			return
		}
		for i, x := range nums {
			if used[i] {
				continue // each element appears once
			}
			used[i] = true
			cur = append(cur, x) // choose
			build()
			cur = cur[:len(cur)-1] // un-choose
			used[i] = false
		}
	}
	build()
	return out
}

static class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> out = new ArrayList<>();
        build(nums, new boolean[nums.length], new ArrayList<>(), out);
        return out;
    }

    // Slot number cur.size() is being filled. Any element not used yet may go
    // there, so the loop always starts at 0 (order matters, unlike subsets).
    private void build(int[] nums, boolean[] used, List<Integer> cur, List<List<Integer>> out) {
        if (cur.size() == nums.length) {      // every slot filled: one permutation
            out.add(new ArrayList<>(cur));    // copy, cur keeps changing
            return;
        }
        for (int i = 0; i < nums.length; i++) {
            if (used[i]) continue;            // each element appears once
            used[i] = true;  cur.add(nums[i]);               // choose
            build(nums, used, cur, out);
            cur.remove(cur.size() - 1);  used[i] = false;    // un-choose (by index)
        }
    }
}
class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        out, cur = [], []
        used = [False] * len(nums)         # used[i]: nums[i] is already in cur

        # Slot number len(cur) is being filled. Any element not used yet may go
        # there, so the loop always starts at 0 (order matters, unlike subsets).
        def build() -> None:
            if len(cur) == len(nums):      # every slot filled: one permutation
                out.append(cur[:])         # copy, cur keeps changing
                return
            for i, x in enumerate(nums):
                if used[i]:
                    continue               # each element appears once
                used[i] = True
                cur.append(x)              # choose
                build()
                cur.pop()                  # un-choose
                used[i] = False

        build()
        return out
