Bottom-up returns: height, diameter, max path sum

The most important tree pattern for SDE-2 rounds. The recursive function returns one thing to its parent and, separately, updates a global answer with what it sees at its own node. The two are often different, and mixing them up is the classic bug.

Read the picture as two questions asked at every node:

  • What does my parent need? A path that continues upward can use only one of my children, so I return 1 + max(L, R).
  • What can I see that nobody above me can? A path that bends at me uses both children: L + R. I compare it with the global best.

Max path sum (LC 124, one of the most asked tree problems in the premium data: DoorDash, Nutanix, Amazon, Arista, Flipkart, Google) is the same skeleton with values: return val + max(L, R), update with val + L + R, and clamp negative child gains to 0 because you may simply not take that branch. On the tree [-7, 4, 12, null, null, 9, -3] the answer is 21, the path 9 → 12, which never touches the root; a single node with value -5 gives -5, so start the best at minus infinity, not 0.

// Return what the PARENT needs; update the answer with what THIS node sees.
func diameter(root *TreeNode) int {
	best := 0
	var depth func(*TreeNode) int // nodes on the longest downward chain
	depth = func(node *TreeNode) int {
		if node == nil {
			return 0
		}
		L, R := depth(node.Left), depth(node.Right)
		best = max(best, L+R) // a path that bends here (in edges)
		return 1 + max(L, R)  // only one side continues upward
	}
	depth(root)
	return best
}

func maxPathSum(root *TreeNode) int {
	best := math.MinInt
	var gain func(*TreeNode) int // best downward sum starting at node
	gain = func(node *TreeNode) int {
		if node == nil {
			return 0
		}
		L := max(0, gain(node.Left)) // a negative branch is dropped
		R := max(0, gain(node.Right))
		best = max(best, node.Val+L+R)
		return node.Val + max(L, R)
	}
	gain(root)
	return best
}
// Return what the PARENT needs; update the global answer with what THIS node sees.
int depth(TreeNode* node, int& diameter) {      // nodes on the longest downward chain
    if (!node) return 0;
    int L = depth(node->left, diameter);
    int R = depth(node->right, diameter);
    diameter = max(diameter, L + R);            // a path that bends here (in edges)
    return 1 + max(L, R);                       // only one side can continue upward
}

int gain(TreeNode* node, int& best) {           // best downward sum starting at node
    if (!node) return 0;
    int L = max(0, gain(node->left, best));     // a negative branch is dropped
    int R = max(0, gain(node->right, best));
    best = max(best, node->val + L + R);        // path that bends here
    return node->val + max(L, R);
}

static int diameter, best;

static int depth(TreeNode node) {              // nodes on the longest downward chain
    if (node == null) return 0;
    int L = depth(node.left), R = depth(node.right);
    diameter = Math.max(diameter, L + R);      // a path that bends here (in edges)
    return 1 + Math.max(L, R);                 // only one side continues upward
}

# Return what the PARENT needs; update the answer with what THIS node sees.
def diameter(root: Optional[TreeNode]) -> int:
    best = 0

    def depth(node):                    # nodes on the longest downward chain
        nonlocal best
        if not node:
            return 0
        L, R = depth(node.left), depth(node.right)
        best = max(best, L + R)         # a path that bends here (in edges)
        return 1 + max(L, R)            # only one side continues upward

    depth(root)
    return best


def max_path_sum(root: TreeNode) -> int:
    best = float("-inf")

    def gain(node):                     # best downward sum starting at node
        nonlocal best
        if not node:
            return 0
        L = max(0, gain(node.left))     # a negative branch is dropped
        R = max(0, gain(node.right))
        best = max(best, node.val + L + R)
        return node.val + max(L, R)

    gain(root)
    return best
