Build and serialize: one traversal is not enough (usually)

Preorder alone cannot rebuild a general binary tree: [1, 2] could be 2 as a left or a right child. You need either a second traversal or explicit null markers.

For serialize / deserialize (LC 297, one of the most frequent Hard tree problems across Amazon, LinkedIn, Nutanix and TikTok in the premium data), the simplest robust format is preorder with a marker for null: the example tree becomes 8,3,1,#,#,6,#,#,10,#,14,#,#,. Deserialize reads tokens in the same order, node then left then right, and a "#" ends a branch. Level-order with markers (LeetCode's own format) works too and is easier to eyeball; preorder is shorter to code.

// Preorder with "#" for null: "1,2,#,#,3,#,#,". Null markers make it unambiguous.
func serialize(root *TreeNode) string {
	var sb strings.Builder
	var ser func(*TreeNode)
	ser = func(n *TreeNode) {
		if n == nil {
			sb.WriteString("#,")
			return
		}
		sb.WriteString(strconv.Itoa(n.Val))
		sb.WriteByte(',')
		ser(n.Left)
		ser(n.Right)
	}
	ser(root)
	return sb.String()
}

func deserialize(s string) *TreeNode {
	toks := strings.Split(s, ",")
	i := 0
	var des func() *TreeNode
	des = func() *TreeNode {
		tok := toks[i]
		i++
		if tok == "#" {
			return nil
		}
		v, _ := strconv.Atoi(tok)
		n := &TreeNode{Val: v}
		n.Left = des() // same order as ser: node, left, right
		n.Right = des()
		return n
	}
	return des()
}

void ser(TreeNode* n, string& s) {
    if (!n) { s += "#,"; return; }
    s += to_string(n->val) + ",";
    ser(n->left, s);
    ser(n->right, s);
}
string serialize(TreeNode* root) { string s; ser(root, s); return s; }

TreeNode* des(const string& s, size_t& i) {
    size_t j = s.find(',', i);
    string tok = s.substr(i, j - i);
    i = j + 1;
    if (tok == "#") return nullptr;
    TreeNode* n = new TreeNode(stoi(tok));
    n->left = des(s, i);                        // same order as ser: node, left, right
    n->right = des(s, i);
    return n;
}

    ser(n.left, sb);

static TreeNode des(Iterator<String> it) {
    String tok = it.next();
    if (tok.equals("#")) return null;
    TreeNode n = new TreeNode(Integer.parseInt(tok));
    n.left = des(it);                          // same order as ser: node, left, right
    n.right = des(it);
    return n;
}
# Preorder with "#" for null: "1,2,#,#,3,#,#,". Null markers make it unambiguous.
def serialize(root: Optional[TreeNode]) -> str:
    parts = []

    def ser(n):
        if not n:
            parts.append("#")
            return
        parts.append(str(n.val))
        ser(n.left)
        ser(n.right)

    ser(root)
    return ",".join(parts) + ","


def deserialize(s: str) -> Optional[TreeNode]:
    it = iter(s.split(",")[:-1])

    def des():
        tok = next(it)
        if tok == "#":
            return None
        n = TreeNode(int(tok))
        n.left = des()                  # same order as ser: node, left, right
        n.right = des()
        return n

    return des()

Special cases worth knowing: a BST can be rebuilt from preorder alone (each value's bounds decide where it goes, O(n) with an upper-bound recursion), and a full binary tree from preorder + postorder. Flatten to linked list is a preorder rewire: for each node, hang its right subtree off the rightmost node of its left subtree, move left to right, continue (the same predecessor search as Morris).
