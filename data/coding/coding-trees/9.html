BST basics: the interval rule

A BST is not "left child smaller, right child larger". It is: every node in the left subtree is smaller and every node in the right subtree is larger. Equivalently, each node must fall inside an open interval inherited from all its ancestors.

// Every node must sit inside the open interval (lo, hi) inherited from its ancestors.
func validBST(n *TreeNode, lo, hi int64) bool {
	if n == nil {
		return true
	}
	v := int64(n.Val)
	if v <= lo || v >= hi {
		return false
	}
	return validBST(n.Left, lo, v) && validBST(n.Right, v, hi)
}

// call: validBST(root, math.MinInt64, math.MaxInt64)
// Every node must sit inside the open interval (lo, hi) inherited from its ancestors.
bool validBST(TreeNode* n, long long lo, long long hi) {
    if (!n) return true;
    if (n->val <= lo || n->val >= hi) return false;
    return validBST(n->left, lo, n->val) && validBST(n->right, n->val, hi);
}
// call: validBST(root, LLONG_MIN, LLONG_MAX)
// Every node must sit inside the open interval (lo, hi) inherited from its ancestors.
static boolean validBST(TreeNode n, long lo, long hi) {
    if (n == null) return true;
    if (n.val <= lo || n.val >= hi) return false;
    return validBST(n.left, lo, n.val) && validBST(n.right, n.val, hi);
}
// call: validBST(root, Long.MIN_VALUE, Long.MAX_VALUE)
# Every node must sit inside the open interval (lo, hi) inherited from its ancestors.
def valid_bst(n: Optional[TreeNode], lo=float("-inf"), hi=float("inf")) -> bool:
    if not n:
        return True
    if not (lo < n.val < hi):
        return False
    return valid_bst(n.left, lo, n.val) and valid_bst(n.right, n.val, hi)

From your world

Interview BSTs are not balanced unless you balance them; the kernel's rbtree is a BST that rebalances on every insert and erase so h stays O(log n). When an interviewer asks "what if the data arrives sorted?", the answer is "a plain BST degrades to a list, so use a self-balancing tree: std::map, Java's TreeMap, the kernel's rbtree."
