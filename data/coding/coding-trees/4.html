Morris traversal: O(1) extra space by borrowing null pointers

A tree with n nodes has n + 1 null child pointers. Morris traversal (1979) borrows some of them as temporary threads: before going into a node's left subtree, it makes that subtree's rightmost node (the inorder predecessor) point back up to the node. When the walk later arrives at the predecessor and follows its right pointer, it lands back on the node, sees the thread, removes it, visits the node and goes right. No stack is needed.

Each edge is walked a constant number of times (once down, at most twice while searching for predecessors), so the time is still O(n). The tree is modified while you walk it and restored at the end, so Morris is not safe if another thread reads the tree concurrently, and you must not stop the loop early (an early break leaves threads behind). Use it when the interviewer says "O(1) extra space": recover a BST with two swapped nodes, kth smallest without a stack, or a flatten-to-list variant.

From your world

The kernel's rb_node stores the node colour inside the low bits of the parent pointer (__rb_parent_color), because the pointer is aligned and those bits are always zero. Morris has the same spirit: storage that is "free" (a null pointer) temporarily carries state. The difference is that Morris must clean up after itself, like a lock you must release on every exit path.

// Iterative inorder: the stack holds ancestors whose left side is still in progress.
func inorder(root *TreeNode) []int {
	out := []int{}
	st := []*TreeNode{}
	cur := root
	for cur != nil || len(st) > 0 {
		for cur != nil { // dive left, remember the path
			st = append(st, cur)
			cur = cur.Left
		}
		cur = st[len(st)-1]
		st = st[:len(st)-1]
		out = append(out, cur.Val) // visit
		cur = cur.Right            // then the right subtree
	}
	return out
}

// Morris inorder: O(1) extra space via a temporary thread from the predecessor.
func morrisInorder(root *TreeNode) []int {
	out := []int{}
	cur := root
	for cur != nil {
		if cur.Left == nil {
			out = append(out, cur.Val)
			cur = cur.Right
			continue
		}
		pre := cur.Left // rightmost node of the left subtree
		for pre.Right != nil && pre.Right != cur {
			pre = pre.Right
		}
		if pre.Right == nil { // 1st visit: add the thread, go left
			pre.Right = cur
			cur = cur.Left
		} else { // 2nd visit: remove it, visit, go right
			pre.Right = nil
			out = append(out, cur.Val)
			cur = cur.Right
		}
	}
	return out
}
// Iterative inorder: the stack holds ancestors whose left side is still in progress.
vector<int> inorder(TreeNode* root) {
    vector<int> out;
    stack<TreeNode*> st;
    TreeNode* cur = root;
    while (cur || !st.empty()) {
        while (cur) { st.push(cur); cur = cur->left; }  // dive left, remember the path
        cur = st.top(); st.pop();
        out.push_back(cur->val);                        // visit
        cur = cur->right;                               // then do the right subtree
    }
    return out;
}

// Morris inorder: O(1) extra space. A temporary "thread" from the inorder
// predecessor back to cur replaces the stack; it is removed on the second visit.
vector<int> morrisInorder(TreeNode* root) {
    vector<int> out;
    TreeNode* cur = root;
    while (cur) {
        if (!cur->left) { out.push_back(cur->val); cur = cur->right; continue; }
        TreeNode* pre = cur->left;                      // rightmost node of the left subtree
        while (pre->right && pre->right != cur) pre = pre->right;
        if (!pre->right) { pre->right = cur; cur = cur->left; }   // 1st visit: add thread, go left
        else { pre->right = nullptr; out.push_back(cur->val); cur = cur->right; } // 2nd: remove it
    }
    return out;
}

static List<Integer> inorder(TreeNode root) {

def inorder(root: Optional[TreeNode]) -> List[int]:
    out, st, cur = [], [], root
    while cur or st:
        while cur:                      # dive left, remember the path
            st.append(cur)
            cur = cur.left
        cur = st.pop()
        out.append(cur.val)             # visit
        cur = cur.right                 # then the right subtree
    return out


# Morris inorder: O(1) extra space via a temporary thread from the predecessor.
def morris_inorder(root: Optional[TreeNode]) -> List[int]:
    out, cur = [], root
    while cur:
        if not cur.left:
            out.append(cur.val)
            cur = cur.right
            continue
        pre = cur.left                  # rightmost node of the left subtree
        while pre.right and pre.right is not cur:
            pre = pre.right
        if not pre.right:               # 1st visit: add the thread, go left
            pre.right = cur
            cur = cur.left
        else:                           # 2nd visit: remove it, visit, go right
            pre.right = None
            out.append(cur.val)
            cur = cur.right
    return out
