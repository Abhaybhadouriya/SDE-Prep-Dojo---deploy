Each file in coding-code/text-justification/ checks LeetCode's three published examples, the three examples on this page, a single word that fills the width, one-character width with several words, and a last line with trailing padding; Python also asserts every line has length maxWidth.

class Solution {
public:
    vector<string> fullJustify(vector<string>& words, int maxWidth) {
        vector<string> lines;
        int n = words.size(), i = 0;
        while (i < n) {
            // 1. Greedy: take words i..j-1 while letters + one space per gap still fits.
            int j = i, letters = 0;
            while (j < n && letters + (int)words[j].size() + (j - i) <= maxWidth)
                letters += words[j++].size();
            int gaps = j - i - 1;
            string line;
            if (j == n || gaps == 0) {
                // 2a. Last line or a single word: left-justify, pad the right end.
                for (int k = i; k < j; k++) {
                    if (k > i) line += ' ';
                    line += words[k];
                }
                line.append(maxWidth - line.size(), ' ');
            } else {
                // 2b. Full justify: spread the spaces, the leftmost `extra` gaps get one more.
                int each = (maxWidth - letters) / gaps, extra = (maxWidth - letters) % gaps;
                for (int k = i; k < j - 1; k++) {
                    line += words[k];
                    line.append(each + (k - i < extra ? 1 : 0), ' ');
                }
                line += words[j - 1];
            }
            lines.push_back(line);
            i = j;
        }
        return lines;
    }
};
func fullJustify(words []string, maxWidth int) []string {
	var lines []string
	n, i := len(words), 0
	for i < n {
		// 1. Greedy: take words i..j-1 while letters + one space per gap still fits.
		j, letters := i, 0
		for j < n && letters+len(words[j])+(j-i) <= maxWidth {
			letters += len(words[j])
			j++
		}
		gaps := j - i - 1
		var b strings.Builder
		if j == n || gaps == 0 {
			// 2a. Last line or a single word: left-justify, pad the right end.
			b.WriteString(strings.Join(words[i:j], " "))
			b.WriteString(strings.Repeat(" ", maxWidth-b.Len()))
		} else {
			// 2b. Full justify: spread the spaces, the leftmost `extra` gaps get one more.
			each, extra := (maxWidth-letters)/gaps, (maxWidth-letters)%gaps
			for k := i; k < j-1; k++ {
				b.WriteString(words[k])
				pad := each
				if k-i < extra {
					pad++
				}
				b.WriteString(strings.Repeat(" ", pad))
			}
			b.WriteString(words[j-1])
		}
		lines = append(lines, b.String())
		i = j
	}
	return lines
}

class Solution:
    def fullJustify(self, words: List[str], maxWidth: int) -> List[str]:
        lines, i, n = [], 0, len(words)
        while i < n:
            # 1. Greedy: take words i..j-1 while letters + one space per gap still fits.
            j, letters = i, 0
            while j < n and letters + len(words[j]) + (j - i) <= maxWidth:
                letters += len(words[j])
                j += 1
            gaps = j - i - 1
            if j == n or gaps == 0:
                # 2a. Last line or a single word: left-justify, pad the right end.
                line = ' '.join(words[i:j])
                line += ' ' * (maxWidth - len(line))
            else:
                # 2b. Full justify: spread the spaces, the leftmost `extra` gaps get one more.
                each, extra = divmod(maxWidth - letters, gaps)
                parts = []
                for k in range(i, j - 1):
                    parts.append(words[k])
                    parts.append(' ' * (each + (1 if k - i < extra else 0)))
                parts.append(words[j - 1])
                line = ''.join(parts)
            lines.append(line)
            i = j
        return lines
