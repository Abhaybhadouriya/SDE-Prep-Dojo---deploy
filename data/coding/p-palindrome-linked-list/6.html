Middle, reverse, compare, restore. The tests in coding-code/palindrome-linked-list/ also check that the list is unchanged after the call and run a 100,000-node list.

class Solution {
public:
    bool isPalindrome(ListNode* head) {
        // 1. slow stops at the last node of the front half (the middle node for odd lengths).
        ListNode *slow = head, *fast = head;
        while (fast->next && fast->next->next) {
            slow = slow->next;
            fast = fast->next->next;
        }
        // 2. Reverse the back half in place.
        ListNode* back = reverse(slow->next);
        // 3. Compare; the back half is never longer, so it decides when to stop.
        bool ok = true;
        for (ListNode *a = head, *b = back; b; a = a->next, b = b->next) {
            if (a->val != b->val) { ok = false; break; }
        }
        // 4. Restore the caller's list.
        slow->next = reverse(back);
        return ok;
    }

private:
    static ListNode* reverse(ListNode* node) {
        ListNode* prev = nullptr;
        while (node) {
            ListNode* nxt = node->next;
            node->next = prev;
            prev = node;
            node = nxt;
        }
        return prev;
    }
};
func isPalindrome(head *ListNode) bool {
	// 1. slow stops at the last node of the front half (the middle node for odd lengths).
	slow, fast := head, head
	for fast.Next != nil && fast.Next.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
	}
	// 2. Reverse the back half in place.
	back := reverseList(slow.Next)
	// 3. Compare; the back half is never longer, so it decides when to stop.
	ok := true
	for a, b := head, back; b != nil; a, b = a.Next, b.Next {
		if a.Val != b.Val {
			ok = false
			break
		}
	}
	// 4. Restore the caller's list.
	slow.Next = reverseList(back)
	return ok
}

func reverseList(node *ListNode) *ListNode {
	var prev *ListNode
	for node != nil {
		nxt := node.Next
		node.Next = prev
		prev, node = node, nxt
	}
	return prev
}

class Solution:
    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        # 1. slow stops at the last node of the front half (the middle node for odd lengths).
        slow, fast = head, head
        while fast.next and fast.next.next:
            slow = slow.next
            fast = fast.next.next

        # 2. Reverse the back half in place: it now reads from the tail toward the middle.
        back = self._reverse(slow.next)

        # 3. Walk both halves together. The back half is never longer than the front half,
        #    so it decides when to stop (an odd middle node is compared with nothing).
        ok, a, b = True, head, back
        while b:
            if a.val != b.val:
                ok = False
                break
            a, b = a.next, b.next

        # 4. Put the list back the way the caller gave it to us.
        slow.next = self._reverse(back)
        return ok

    @staticmethod
    def _reverse(node: Optional[ListNode]) -> Optional[ListNode]:
        prev = None
        while node:
            nxt = node.next
            node.next = prev
            prev, node = node, nxt
        return prev
