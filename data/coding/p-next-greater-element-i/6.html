class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        unordered_map<int, int> nextBigger;   // value -> its next greater element in nums2
        vector<int> st;                       // values still waiting, decreasing from bottom to top
        for (int x : nums2) {
            while (!st.empty() && st.back() < x) {
                nextBigger[st.back()] = x;    // x is the first bigger value to the right of st.back()
                st.pop_back();
            }
            st.push_back(x);
        }
        vector<int> ans;
        ans.reserve(nums1.size());
        for (int q : nums1) {
            auto it = nextBigger.find(q);
            ans.push_back(it == nextBigger.end() ? -1 : it->second);  // never popped: no bigger value
        }
        return ans;
    }
};
func nextGreaterElement(nums1 []int, nums2 []int) []int {
	nextBigger := make(map[int]int, len(nums2)) // value -> its next greater element in nums2
	var st []int                                // values still waiting, decreasing from bottom to top
	for _, x := range nums2 {
		for len(st) > 0 && st[len(st)-1] < x {
			nextBigger[st[len(st)-1]] = x // x is the first bigger value to the right of it
			st = st[:len(st)-1]
		}
		st = append(st, x)
	}
	ans := make([]int, len(nums1))
	for k, q := range nums1 {
		if v, ok := nextBigger[q]; ok {
			ans[k] = v
		} else {
			ans[k] = -1 // never popped: no bigger value
		}
	}
	return ans
}

static class Solution {
    public int[] nextGreaterElement(int[] nums1, int[] nums2) {
        Map<Integer, Integer> nextBigger = new HashMap<>();   // value -> next greater element in nums2
        ArrayDeque<Integer> st = new ArrayDeque<>();          // waiting values, decreasing to the top
        for (int x : nums2) {
            while (!st.isEmpty() && st.peek() < x)
                nextBigger.put(st.pop(), x);  // x is the first bigger value to the right of it
            st.push(x);
        }
        int[] ans = new int[nums1.length];
        for (int k = 0; k < nums1.length; k++)
            ans[k] = nextBigger.getOrDefault(nums1[k], -1);   // never popped: no bigger value
        return ans;
    }
}
class Solution:
    def nextGreaterElement(self, nums1: List[int], nums2: List[int]) -> List[int]:
        next_bigger = {}                      # value -> its next greater element in nums2
        stack = []                            # values still waiting, decreasing from bottom to top
        for x in nums2:
            while stack and stack[-1] < x:
                next_bigger[stack.pop()] = x  # x is the first bigger value to the right of it
            stack.append(x)
        return [next_bigger.get(q, -1) for q in nums1]   # never popped: no bigger value
