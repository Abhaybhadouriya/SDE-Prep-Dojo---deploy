class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        sort(intervals.begin(), intervals.end());        // by start (ties by end, harmless)
        vector<vector<int>> out;
        for (const auto& iv : intervals) {
            if (!out.empty() && iv[0] <= out.back()[1]) {
                out.back()[1] = max(out.back()[1], iv[1]); // overlap or touch: stretch the open block
            } else {
                out.push_back(iv);                          // gap: the open block is final, start a new one
            }
        }
        return out;
    }
};
func merge(intervals [][]int) [][]int {
	sort.Slice(intervals, func(i, j int) bool { return intervals[i][0] < intervals[j][0] }) // by start
	out := [][]int{}
	for _, iv := range intervals {
		if n := len(out); n > 0 && iv[0] <= out[n-1][1] {
			if iv[1] > out[n-1][1] { // overlap or touch: stretch the open block
				out[n-1][1] = iv[1]
			}
		} else {
			out = append(out, []int{iv[0], iv[1]}) // gap: start a new block
		}
	}
	return out
}

static class Solution {
    public int[][] merge(int[][] intervals) {
        Arrays.sort(intervals, (x, y) -> Integer.compare(x[0], y[0])); // by start
        List<int[]> out = new ArrayList<>();
        for (int[] iv : intervals) {
            int[] last = out.isEmpty() ? null : out.get(out.size() - 1);
            if (last != null && iv[0] <= last[1]) {
                last[1] = Math.max(last[1], iv[1]);   // overlap or touch: stretch the open block
            } else {
                out.add(new int[]{iv[0], iv[1]});     // gap: start a new block (copy, do not alias input)
            }
        }
        return out.toArray(new int[0][]);
    }
}
class Solution:
    def merge(self, intervals: List[List[int]]) -> List[List[int]]:
        intervals.sort(key=lambda iv: iv[0])           # by start
        out = []
        for start, end in intervals:
            if out and start <= out[-1][1]:
                out[-1][1] = max(out[-1][1], end)      # overlap or touch: stretch the open block
            else:
                out.append([start, end])               # gap: start a new block
        return out
