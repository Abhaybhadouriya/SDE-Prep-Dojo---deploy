class Solution {
public:
    void sortColors(vector<int>& nums) {
        // invariant: [0, lo) = 0s, [lo, mid) = 1s, [mid, hi] = not looked at yet, (hi, n) = 2s
        int lo = 0, mid = 0, hi = (int)nums.size() - 1;
        while (mid <= hi) {
            if (nums[mid] == 0) {
                swap(nums[lo++], nums[mid++]);   // nums[lo] was a 1 (or lo == mid): safe to step past
            } else if (nums[mid] == 1) {
                mid++;
            } else {
                swap(nums[mid], nums[hi--]);     // the value that came from hi is unseen: do NOT move mid
            }
        }
    }
};
func sortColors(nums []int) {
	// invariant: [0, lo) = 0s, [lo, mid) = 1s, [mid, hi] = not looked at yet, (hi, n) = 2s
	lo, mid, hi := 0, 0, len(nums)-1
	for mid <= hi {
		switch nums[mid] {
		case 0:
			nums[lo], nums[mid] = nums[mid], nums[lo] // nums[lo] was a 1 (or lo == mid)
			lo++
			mid++
		case 1:
			mid++
		default:
			nums[mid], nums[hi] = nums[hi], nums[mid] // the value from hi is unseen: keep mid
			hi--
		}
	}
}

static class Solution {
    public void sortColors(int[] nums) {
        // invariant: [0, lo) = 0s, [lo, mid) = 1s, [mid, hi] = not looked at yet, (hi, n) = 2s
        int lo = 0, mid = 0, hi = nums.length - 1;
        while (mid <= hi) {
            if (nums[mid] == 0) {
                swap(nums, lo++, mid++);   // nums[lo] was a 1 (or lo == mid): safe to step past
            } else if (nums[mid] == 1) {
                mid++;
            } else {
                swap(nums, mid, hi--);     // the value that came from hi is unseen: do NOT move mid
            }
        }
    }

    private static void swap(int[] a, int x, int y) {
        int t = a[x];
        a[x] = a[y];
        a[y] = t;
    }
}
