Backtracking with a start index in four languages. Each file in coding-code/subsets/ checks the examples and, for every n from 1 to 10, that the output has 2n distinct subsets equal to the bitmask enumeration.

func subsets(nums []int) [][]int {
	out := [][]int{}
	cur := []int{} // the subset being built (the recursion path)

	// Every node of the recursion is one subset: record it, then try each
	// later element as the next member. Only indexes >= start are allowed,
	// so {a, b} is built once (as a then b) and never again as b then a.
	var build func(start int)
	build = func(start int) {
		out = append(out, append([]int(nil), cur...)) // copy: cur's array is reused
		for i := start; i < len(nums); i++ {
			cur = append(cur, nums[i]) // choose
			build(i + 1)
			cur = cur[:len(cur)-1] // un-choose
		}
	}
	build(0)
	return out
}
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> out;
        vector<int> cur;                 // the subset being built (the recursion path)
        build(nums, 0, cur, out);
        return out;
    }

private:
    // Every node of the recursion is one subset: record it, then try each
    // later element as the next member. Only indexes >= start are allowed,
    // so {a, b} is built once (as a then b) and never again as b then a.
    void build(const vector<int>& nums, int start, vector<int>& cur, vector<vector<int>>& out) {
        out.push_back(cur);              // copy: O(len) per subset
        for (int i = start; i < (int)nums.size(); i++) {
            cur.push_back(nums[i]);      // choose
            build(nums, i + 1, cur, out);
            cur.pop_back();              // un-choose
        }
    }
};

static class Solution {
    public List<List<Integer>> subsets(int[] nums) {
        List<List<Integer>> out = new ArrayList<>();
        build(nums, 0, new ArrayList<>(), out);
        return out;
    }

    // Every node of the recursion is one subset: record it, then try each
    // later element as the next member. Only indexes >= start are allowed,
    // so {a, b} is built once (as a then b) and never again as b then a.
    private void build(int[] nums, int start, List<Integer> cur, List<List<Integer>> out) {
        out.add(new ArrayList<>(cur));        // copy: O(len) per subset
        for (int i = start; i < nums.length; i++) {
            cur.add(nums[i]);                 // choose
            build(nums, i + 1, cur, out);
            cur.remove(cur.size() - 1);       // un-choose (by index, not by value)
        }
    }
}
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        out, cur = [], []

        # Every node of the recursion is one subset: record it, then try each
        # later element as the next member. Only indexes >= start are allowed,
        # so {a, b} is built once (as a then b) and never again as b then a.
        def build(start: int) -> None:
            out.append(cur[:])                 # copy: O(len) per subset
            for i in range(start, len(nums)):
                cur.append(nums[i])            # choose
                build(i + 1)
                cur.pop()                      # un-choose

        build(0)
        return out
