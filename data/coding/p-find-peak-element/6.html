Binary search on the slope in four languages. Each file in coding-code/find-peak-element/ also runs 2,000 random arrays (with distinct neighbours) and checks that the returned index really is a peak.

class Solution {
public:
    int findPeakElement(vector<int>& nums) {
        int lo = 0, hi = (int)nums.size() - 1;
        // Invariant: [lo, hi] always holds a peak (outside the array counts as -infinity).
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;      // mid < hi, so mid + 1 is a valid index
            if (nums[mid] < nums[mid + 1])
                lo = mid + 1;                  // uphill to the right: a peak lies in (mid, hi]
            else
                hi = mid;                      // downhill: mid itself or something left of it is a peak
        }
        return lo;                             // lo == hi: the range shrank to one peak
    }
};
