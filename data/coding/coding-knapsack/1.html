0/1 knapsack: the table

The problem. You are given n items; item k has a positive integer weight w[k] and a value v[k]. A bag holds total weight at most W. Each item can be packed at most once. Return the largest total value that fits. (This is the classic GfG 0-1 Knapsack / TUF DP-19; typical limits n ≤ 1,000 and W ≤ 1,000.)

Apply the four-step recipe from the state design page:

  • State: best[i][c] = the most value using only the first i items with capacity c.
  • Choices (the last decision is about item i-1): skip it, getting best[i-1][c]; or, if it fits (c ≥ w), take it once, getting best[i-1][c-w] + v.
  • Base: best[0][c] = 0: with no items there is no value.
  • Answer: best[n][W]. Time O(n x W), space O(n x W) for the table.

Notice what the row index buys us. "Take item i-1" reads row i-1, which was computed without item i-1, so the item cannot be used twice. That one detail is the difference between 0/1 and unbounded, and it shows up again as the loop direction when we squeeze the table.

// 0/1 knapsack, full table: best[i][c] = most value using the first i items with capacity c.
func knap01Table(w, v []int, capacity int) int {
	n := len(w)
	best := make([][]int, n+1)
	for i := range best {
		best[i] = make([]int, capacity+1)
	}
	for i := 1; i <= n; i++ {
		for c := 0; c <= capacity; c++ {
			best[i][c] = best[i-1][c] // skip item i-1
			if c >= w[i-1] {
				best[i][c] = max(best[i][c], best[i-1][c-w[i-1]]+v[i-1]) // take it once
			}
		}
	}
	return best[n][capacity]
}

int knap01Table(const vector<int>& w, const vector<int>& v, int cap) {
    int n = w.size();
    vector<vector<int>> best(n + 1, vector<int>(cap + 1, 0));

        for (int c = 0; c <= cap; c++) {

            if (c >= w[i - 1])
                best[i][c] = max(best[i][c], best[i - 1][c - w[i - 1]] + v[i - 1]);   // take it once
        }
    return best[n][cap];
}

# 0/1 knapsack, full table: best[i][c] = most value using the first i items with capacity c.
def knap01_table(weights, values, cap):
    n = len(weights)
    best = [[0] * (cap + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        w, v = weights[i - 1], values[i - 1]
        for c in range(cap + 1):
            best[i][c] = best[i - 1][c]                      # skip item i-1
            if c >= w:
                best[i][c] = max(best[i][c], best[i - 1][c - w] + v)   # take it once
    return best[n][cap]
