Counting subsets: Target Sum and its relatives

Target Sum restated: you are given up to 20 non-negative integers and a target. Put a + or a - in front of every number; count how many sign patterns make the expression equal the target. Trying all 2n patterns works for n = 20, but the DP is the version interviewers want, because it generalises.

The trick is a line of algebra. Let P be the sum of the numbers with a plus and N the sum of those with a minus. Then P - N = target and P + N = total, so P = (total + target) / 2. A sign pattern is just a choice of which numbers go into the plus set, so the question becomes "how many subsets have sum P?".

Counting uses the same downward loop with += instead of or, and the base is ways[0] = 1 (the empty subset). Zeros need care: a 0 can go in either set, which doubles the count. The downward loop handles it on its own, because for x = 0 the update ways[s] += ways[s - 0] doubles every cell exactly once.

  • Count partitions with given difference is the same algebra: S1 - S2 = d gives S1 = (total + d) / 2.
  • Last Stone Weight II (smash stones until at most one is left; return the smallest possible weight) is "split into two groups with the smallest difference": find the largest reachable sum s ≤ total/2 and answer total - 2s.
  • Ones and Zeroes has two capacities (a budget of zeros and a budget of ones). The state gains a dimension, best[i][j], and both loops run downwards.
// 0/1 counting: signs +/- on every number so the total equals target. Plus-set sum P = (total+target)/2.
func findTargetSumWays(nums []int, target int) int {
	total := 0
	for _, x := range nums {
		total += x
	}
	if target > total || -target > total || (total+target)%2 != 0 {
		return 0
	}
	want := (total + target) / 2
	ways := make([]int, want+1)
	ways[0] = 1
	for _, x := range nums {
		for s := want; s >= x; s-- { // x = 0 doubles ways[s]
			ways[s] += ways[s-x]
		}
	}
	return ways[want]
}

// 0/1 with two capacities: largest subset of binary strings with at most m zeros and n ones.
func findMaxForm(strs []string, m, n int) int {
	best := make([][]int, m+1)
	for i := range best {
		best[i] = make([]int, n+1)
	}
	for _, s := range strs {
		z := strings.Count(s, "0")
		o := len(s) - z
		for i := m; i >= z; i-- {
			for j := n; j >= o; j-- {
				best[i][j] = max(best[i][j], best[i-z][j-o]+1)
			}
		}
	}
	return best[m][n]
}

int findTargetSumWays(const vector<int>& nums, int target) {
    int total = accumulate(nums.begin(), nums.end(), 0);
    if (abs(target) > total || (total + target) % 2) return 0;
    int want = (total + target) / 2;
    vector<int> ways(want + 1, 0);
    ways[0] = 1;
    for (int x : nums)
        for (int s = want; s >= x; s--)                                    // x = 0 doubles ways[s]
            ways[s] += ways[s - x];
    return ways[want];
}

// 0/1 with two capacities: largest subset of binary strings with at most m zeros and n ones.
static int findMaxForm(String[] strs, int m, int n) {
    int[][] best = new int[m + 1][n + 1];
    for (String s : strs) {
        int z = 0;
        for (char ch : s.toCharArray()) if (ch == '0') z++;
        int o = s.length() - z;
        for (int i = m; i >= z; i--)
            for (int j = n; j >= o; j--)
                best[i][j] = Math.max(best[i][j], best[i - z][j - o] + 1);
    }
    return best[m][n];
}
# 0/1 counting: signs +/- on every number so the total equals target. Plus-set sum P = (total+target)/2.
def find_target_sum_ways(nums, target):
    total = sum(nums)
    if abs(target) > total or (total + target) % 2:
        return 0
    want = (total + target) // 2
    ways = [1] + [0] * want                               # ways[s] = subsets summing to s
    for x in nums:
        for s in range(want, x - 1, -1):                  # x = 0 doubles ways[s]: +0 and -0
            ways[s] += ways[s - x]
    return ways[want]

# 0/1 with two capacities: largest subset of binary strings with at most m zeros and n ones.
def find_max_form(strs, m, n):
    best = [[0] * (n + 1) for _ in range(m + 1)]
    for s in strs:
        z, o = s.count('0'), s.count('1')
        for i in range(m, z - 1, -1):
            for j in range(n, o - 1, -1):
                best[i][j] = max(best[i][j], best[i - z][j - o] + 1)
    return best[m][n]
