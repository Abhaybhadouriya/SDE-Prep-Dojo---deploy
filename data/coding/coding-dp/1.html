Recursion, then memo: the same function, much faster

Write the stair idea as a recursive function. ways(i) = number of ways to stand on step i. You reach step i from i-1 or from i-2, so ways(i) = ways(i-1) + ways(i-2), with ways(0) = ways(1) = 1 (one way to be at the bottom: do nothing; one way to reach step 1). These are the Fibonacci numbers.

The plain recursion is correct but slow. Look at the call tree for ways(5):

Add a dictionary (or array) memo: before computing ways(i), check whether it is already there; after computing, store it. That is memoisation (top-down DP). The function is the same; the number of real computations drops from about 1.6n to n+1. Time O(n), space O(n) for the memo plus O(n) recursion depth.

Next, notice the order in which the memo fills: f(0), f(1), f(2) ... upward. So skip the recursion and fill an array in that order with a loop. That is tabulation (bottom-up DP). Finally, notice the loop reads only the last two entries, so keep two variables instead of the array. That is space optimisation. Here are all three steps as one reusable template:

Every version on this page, Go included, was compiled and run (Go 1.27, C++14, Java 21, Python 3.11) and passes its checks.

func climbMemo(n int) int {
    memo := make(map[int]int)
    var ways func(i int) int
    ways = func(i int) int { // ways to reach step i
        if i <= 1 {
            return 1
        }
        if v, ok := memo[i]; ok {
            return v
        }
        memo[i] = ways(i-1) + ways(i-2)
        return memo[i]
    }
    return ways(n)
}

func climbTab(n int) int {
    dp := make([]int, n+2)
    dp[0], dp[1] = 1, 1
    for i := 2; i <= n; i++ {
        dp[i] = dp[i-1] + dp[i-2]
    }
    return dp[n]
}

func climbO1(n int) int {
    prev2, prev1 := 1, 1 // dp[i-2], dp[i-1]
    for i := 2; i <= n; i++ {
        prev2, prev1 = prev1, prev1+prev2
    }
    return prev1
}

long long climbMemoRec(int i, vector<long long>& memo) {
    if (i <= 1) return 1;
    if (memo[i] != -1) return memo[i];
    return memo[i] = climbMemoRec(i - 1, memo) + climbMemoRec(i - 2, memo);
}

long long climbO1(int n) {
    long long prev2 = 1, prev1 = 1;                  // dp[i-2], dp[i-1]
    for (int i = 2; i <= n; i++) { long long cur = prev1 + prev2; prev2 = prev1; prev1 = cur; }
    return prev1;
}

def climb_memo(n):
    memo = {}
    def ways(i):                                  # ways to reach step i
        if i <= 1:
            return 1
        if i not in memo:
            memo[i] = ways(i - 1) + ways(i - 2)
        return memo[i]
    return ways(n)

def climb_tab(n):
    dp = [0] * (n + 1)
    dp[0] = 1
    if n >= 1:
        dp[1] = 1
    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]

def climb_o1(n):
    prev2, prev1 = 1, 1                           # dp[i-2], dp[i-1]
    for _ in range(2, n + 1):
        prev2, prev1 = prev1, prev1 + prev2
    return prev1
