class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        // seen[p] = how many prefixes so far had sum p.
        // sum(l..r) = pre[r+1] - pre[l], so a subarray ending here sums to k
        // exactly when an earlier prefix equals run - k.
        unordered_map<int, int> seen;
        seen[0] = 1;                    // the empty prefix: subarrays that start at index 0
        int run = 0, count = 0;         // |run| <= 2*10^4 * 1000 = 2*10^7: int is enough
        for (int x : nums) {
            run += x;
            auto it = seen.find(run - k);
            if (it != seen.end()) count += it->second;
            seen[run]++;                // add AFTER the lookup, so we never pair a prefix with itself
        }
        return count;
    }
};
func subarraySum(nums []int, k int) int {
	// seen[p] = how many prefixes so far had sum p.
	// sum(l..r) = pre[r+1] - pre[l], so a subarray ending here sums to k
	// exactly when an earlier prefix equals run - k.
	seen := map[int]int{0: 1} // the empty prefix: subarrays that start at index 0
	run, count := 0, 0
	for _, x := range nums {
		run += x
		count += seen[run-k] // a missing key reads as 0
		seen[run]++          // add AFTER the lookup, never pair a prefix with itself
	}
	return count
}

static class Solution {
    public int subarraySum(int[] nums, int k) {
        // seen[p] = how many prefixes so far had sum p.
        // sum(l..r) = pre[r+1] - pre[l], so a subarray ending here sums to k
        // exactly when an earlier prefix equals run - k.
        Map<Integer, Integer> seen = new HashMap<>();
        seen.put(0, 1);                 // the empty prefix: subarrays that start at index 0
        int run = 0, count = 0;
        for (int x : nums) {
            run += x;
            count += seen.getOrDefault(run - k, 0);
            seen.merge(run, 1, Integer::sum);   // add AFTER the lookup
        }
        return count;
    }
}
class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        # seen[p] = how many prefixes so far had sum p.
        # sum(l..r) = pre[r+1] - pre[l], so a subarray ending here sums to k
        # exactly when an earlier prefix equals run - k.
        seen = defaultdict(int)
        seen[0] = 1                  # the empty prefix: subarrays that start at index 0
        run = count = 0
        for x in nums:
            run += x
            count += seen.get(run - k, 0)   # .get: do not insert missing keys
            seen[run] += 1           # add AFTER the lookup, never pair a prefix with itself
        return count
