class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int> seen(nums.begin(), nums.end());   // O(1) "is x present?"
        int best = 0;
        for (int x : seen) {                 // loop over the SET: duplicates are gone
            if (seen.count(x - 1)) continue; // x - 1 exists, so x is not a run start
            int y = x + 1;                   // walk the run x, x+1, x+2, ...
            while (seen.count(y)) y++;
            best = max(best, y - x);         // run is [x, y-1]
        }
        return best;
    }
};

static class Solution {
    public int longestConsecutive(int[] nums) {
        Set<Integer> seen = new HashSet<>();      // O(1) "is x present?"
        for (int x : nums) seen.add(x);
        int best = 0;
        for (int x : seen) {                      // loop over the SET: duplicates are gone
            if (seen.contains(x - 1)) continue;   // x - 1 exists, so x is not a run start
            int y = x + 1;                        // walk the run x, x+1, x+2, ...
            while (seen.contains(y)) y++;
            best = Math.max(best, y - x);         // run is [x, y-1]
        }
        return best;
    }
}
class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:

        for x in seen:                   # loop over the SET: duplicates are gone

                continue
            y = x + 1                    # walk the run x, x+1, x+2, ...
            while y in seen:
                y += 1
            best = max(best, y - x)      # run is [x, y-1]
        return best
