func mergeTwoLists(a *ListNode, b *ListNode) *ListNode {
	dummy := &ListNode{} // fake node before the answer
	tail := dummy        // last node of the merged part
	for a != nil && b != nil {
		if a.Val <= b.Val { // <= takes from a on ties (stable)
			tail.Next = a
			a = a.Next
		} else {
			tail.Next = b
			b = b.Next
		}
		tail = tail.Next
	}
	if a != nil { // attach the leftover run in O(1)
		tail.Next = a
	} else {
		tail.Next = b
	}
	return dummy.Next
}

public:
    ListNode* mergeTwoLists(ListNode* a, ListNode* b) {
        ListNode dummy;              // fake node before the answer: no special case for the first node
        ListNode* tail = &dummy;     // last node of the merged part
        while (a && b) {
            if (a->val <= b->val) {  // <= takes from a on ties (stable)
                tail->next = a;
                a = a->next;
            } else {
                tail->next = b;
                b = b->next;
            }
            tail = tail->next;
        }
        tail->next = a ? a : b;      // one list is empty: attach the rest of the other in O(1)
        return dummy.next;
    }
};

def mergeTwoLists(self, a, b):
    if not a or not b:
        return a or b                 # one side is empty: the other is the answer
    if a.val <= b.val:
        a.next = self.mergeTwoLists(a.next, b)
        return a
    b.next = self.mergeTwoLists(a, b.next)
    return b
