All four use the left-to-right reach scan with an early exit.

class Solution {
public:
    bool canJump(const vector<int>& nums) {
        int last = (int)nums.size() - 1;
        int reach = 0;                          // farthest index reachable from [0..i]
        for (int i = 0; i <= last; ++i) {
            if (i > reach) return false;        // a gap: no earlier square can land on i
            reach = max(reach, i + nums[i]);    // every index in [0, reach] is reachable
            if (reach >= last) return true;     // early exit: the end is covered
        }
        return true;                            // not reached for valid input (n >= 1)
    }
};
func canJump(nums []int) bool {
	last := len(nums) - 1
	reach := 0 // farthest index reachable from squares 0..i
	for i, step := range nums {
		if i > reach { // a gap: no earlier square can land on i
			return false
		}
		if i+step > reach {
			reach = i + step // every index in [0, reach] is reachable
		}
		if reach >= last { // early exit: the end is covered
			return true
		}
	}
	return true // not reached for valid input (n >= 1)
}

class Solution:
    def canJump(self, nums: List[int]) -> bool:
        last = len(nums) - 1
        reach = 0                       # farthest index reachable from squares 0..i
        for i, step in enumerate(nums):
            if i > reach:               # a gap: no earlier square can land on i
                return False
            reach = max(reach, i + step)   # every index in [0, reach] is reachable
            if reach >= last:           # early exit: the end is covered
                return True
        return True                     # not reached for valid input (n >= 1)
