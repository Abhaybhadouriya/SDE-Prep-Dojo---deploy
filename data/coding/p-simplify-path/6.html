Split and stack in four languages. Each file in coding-code/simplify-path/ runs eight cases: trailing and doubled slashes, "." and "..", ".." at the root, the root itself, and names made of dots (..., .hidden, ..x). In Java, split("/") drops trailing empty strings but keeps the leading one; both are skipped, so it does not matter.

func simplifyPath(path string) string {
	dirs := []string{} // stack: the directories we are inside, root first
	for _, part := range strings.Split(path, "/") { // "//" gives empty parts, which we skip
		switch part {
		case "", ".": // stay in the same directory
		case "..": // go up, but never above the root
			if len(dirs) > 0 {
				dirs = dirs[:len(dirs)-1]
			}
		default:
			dirs = append(dirs, part) // a real name, even "..." or ".x"
		}
	}
	return "/" + strings.Join(dirs, "/")
}
class Solution {
public:
    string simplifyPath(string path) {
        vector<string> dirs;                     // stack: the directories we are inside, root first
        stringstream ss(path);
        string part;
        while (getline(ss, part, '/')) {         // "//" gives empty parts, which we skip
            if (part.empty() || part == ".") continue;      // stay in the same directory
            if (part == "..") {                             // go up, but never above the root
                if (!dirs.empty()) dirs.pop_back();
            } else {
                dirs.push_back(part);                       // a real name, even "..." or ".x"
            }
        }
        if (dirs.empty()) return "/";
        string out;
        for (const string& d : dirs) out += "/" + d;
        return out;
    }
};

static class Solution {
    public String simplifyPath(String path) {
        Deque<String> dirs = new ArrayDeque<>();      // stack: the directories we are inside
        for (String part : path.split("/")) {         // "//" gives empty parts, which we skip
            if (part.isEmpty() || part.equals(".")) continue;   // stay in the same directory
            if (part.equals("..")) {                            // go up, but never above the root
                if (!dirs.isEmpty()) dirs.pollLast();
            } else {
                dirs.addLast(part);                             // a real name, even "..." or ".x"
            }
        }
        if (dirs.isEmpty()) return "/";
        StringBuilder out = new StringBuilder();
        for (String d : dirs) out.append('/').append(d);        // iterates root first
        return out.toString();
    }
}
class Solution:
    def simplifyPath(self, path: str) -> str:
        dirs = []                                # stack: the directories we are inside, root first
        for part in path.split("/"):             # "//" gives empty parts, which we skip
            if part == "" or part == ".":
                continue                         # stay in the same directory
            if part == "..":
                if dirs:                         # go up, but never above the root
                    dirs.pop()
            else:
                dirs.append(part)                # a real name, even "..." or ".x"
        return "/" + "/".join(dirs)
