Adjacent pairs, a 26 x 26 edge matrix (dedupes for free) and Kahn's BFS; the tested files are in coding-code/alien-dictionary/. Seeding the queue in alphabetical order makes the output deterministic, which keeps tests simple; any valid order is accepted by the judge.

class Solution {
public:
    string alienOrder(vector<string>& words) {
        bool present[26] = {}, edge[26][26] = {};   // edge[a][b]: a comes before b
        int indeg[26] = {};
        for (const string& w : words)
            for (char c : w) present[c - 'a'] = true;   // letters with no rule still belong in the answer

        // Only ADJACENT words are compared; the first differing letter is the only fact a pair gives.
        for (size_t i = 0; i + 1 < words.size(); i++) {
            const string &a = words[i], &b = words[i + 1];
            size_t j = 0;
            while (j < a.size() && j < b.size() && a[j] == b[j]) j++;
            if (j == a.size() || j == b.size()) {       // one word is a prefix of the other
                if (a.size() > b.size()) return "";     // "abc" before "ab" can never be sorted
                continue;
            }
            int u = a[j] - 'a', v = b[j] - 'a';
            if (!edge[u][v]) { edge[u][v] = true; indeg[v]++; }   // count duplicate rules once
        }

        // Kahn: repeatedly output a letter that nothing still has to precede.
        queue<int> ready;
        int letters = 0;
        for (int c = 0; c < 26; c++)
            if (present[c]) { letters++; if (indeg[c] == 0) ready.push(c); }
        string order;
        while (!ready.empty()) {
            int u = ready.front(); ready.pop();
            order += char('a' + u);
            for (int v = 0; v < 26; v++)
                if (edge[u][v] && --indeg[v] == 0) ready.push(v);
        }
        return (int)order.size() == letters ? order : "";   // letters left over = a cycle
    }
};
func alienOrder(words []string) string {
	var present [26]bool
	var edge [26][26]bool // edge[a][b]: a comes before b
	var indeg [26]int
	for _, w := range words {
		for i := 0; i < len(w); i++ {
			present[w[i]-'a'] = true // letters with no rule still belong in the answer
		}
	}

	// Only ADJACENT words are compared; the first differing letter is the only fact a pair gives.
	for i := 0; i+1 < len(words); i++ {
		a, b := words[i], words[i+1]
		j := 0
		for j < len(a) && j < len(b) && a[j] == b[j] {
			j++
		}
		if j == len(a) || j == len(b) { // one word is a prefix of the other
			if len(a) > len(b) {
				return ""
			}
			continue
		}
		u, v := a[j]-'a', b[j]-'a'
		if !edge[u][v] { // count duplicate rules once
			edge[u][v] = true
			indeg[v]++
		}
	}

	// Kahn: repeatedly output a letter that nothing still has to precede.
	ready := []int{}
	letters := 0
	for c := 0; c < 26; c++ {
		if present[c] {
			letters++
			if indeg[c] == 0 {
				ready = append(ready, c)
			}
		}
	}
	order := []byte{}
	for len(ready) > 0 {
		u := ready[0]
		ready = ready[1:]
		order = append(order, byte('a'+u))
		for v := 0; v < 26; v++ {
			if edge[u][v] {
				indeg[v]--
				if indeg[v] == 0 {
					ready = append(ready, v)
				}
			}
		}
	}
	if len(order) != letters { // letters left over = a cycle
		return ""
	}
	return string(order)
}

class Solution:
    def alienOrder(self, words: List[str]) -> str:
        letters = sorted({c for w in words for c in w})   # letters with no rule still count
        after = {c: set() for c in letters}               # after[a] = letters that must follow a
        indeg = {c: 0 for c in letters}

        # Only ADJACENT words are compared; the first differing letter is the only fact a pair gives.
        for a, b in zip(words, words[1:]):
            for x, y in zip(a, b):
                if x != y:
                    if y not in after[x]:                 # count duplicate rules once
                        after[x].add(y)
                        indeg[y] += 1
                    break
            else:                                         # no difference: one is a prefix
                if len(a) > len(b):
                    return ""                             # "abc" before "ab" is impossible

        # Kahn: repeatedly output a letter that nothing still has to precede.
        ready = deque(c for c in letters if indeg[c] == 0)
        order = []
        while ready:
            u = ready.popleft()
            order.append(u)
            for v in sorted(after[u]):
                indeg[v] -= 1
                if indeg[v] == 0:
                    ready.append(v)
        return "".join(order) if len(order) == len(letters) else ""   # leftovers = a cycle
