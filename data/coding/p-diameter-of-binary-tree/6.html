class Solution {
    int best = 0;                        // longest path seen so far, in edges

    // height = number of nodes on the longest downward path from node (0 for null)
    int height(TreeNode* node) {
        if (node == nullptr) return 0;
        int l = height(node->left);
        int r = height(node->right);
        best = max(best, l + r);         // path that bends at node: l edges down-left + r edges down-right
        return 1 + max(l, r);            // the parent can extend only one branch
    }

public:
    int diameterOfBinaryTree(TreeNode* root) {
        best = 0;
        height(root);
        return best;
    }
};
func diameterOfBinaryTree(root *TreeNode) int {
	best := 0 // longest path seen so far, in edges

	// height = number of nodes on the longest downward path from node (0 for nil)
	var height func(node *TreeNode) int
	height = func(node *TreeNode) int {
		if node == nil {
			return 0
		}
		l := height(node.Left)
		r := height(node.Right)
		if l+r > best { // path that bends at node
			best = l + r
		}
		if l > r { // the parent can extend only one branch
			return 1 + l
		}
		return 1 + r
	}

	height(root)
	return best
}
