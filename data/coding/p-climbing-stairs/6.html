The two-variable loop in four languages; the tested files are in coding-code/climbing-stairs/.

class Solution {
public:
    int climbStairs(int n) {
        // prev2 = ways to reach step i-2, prev1 = ways to reach step i-1.
        // Start at i = 1: ways(0) = 1 (stand still), ways(1) = 1.
        int prev2 = 1, prev1 = 1;
        for (int i = 2; i <= n; i++) {
            int cur = prev1 + prev2;   // last move was a 1-step or a 2-step
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;                  // ways(45) = 1836311903 fits in int
    }
};
func climbStairs(n int) int {
	// prev2 = ways to reach step i-2, prev1 = ways to reach step i-1.
	// Start at i = 1: ways(0) = 1 (stand still), ways(1) = 1.
	prev2, prev1 := 1, 1
	for i := 2; i <= n; i++ {
		// the last move was a 1-step (from i-1) or a 2-step (from i-2)
		prev2, prev1 = prev1, prev1+prev2
	}
	return prev1
}

class Solution:
    def climbStairs(self, n: int) -> int:
        # prev2 = ways to reach step i-2, prev1 = ways to reach step i-1.
        # Start at i = 1: ways(0) = 1 (stand still), ways(1) = 1.
        prev2, prev1 = 1, 1
        for _ in range(2, n + 1):
            # the last move was a 1-step (from i-1) or a 2-step (from i-2)
            prev2, prev1 = prev1, prev1 + prev2
        return prev1
