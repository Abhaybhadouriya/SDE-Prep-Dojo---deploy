All four check the length, count with fixed arrays (Python uses Counter), compare the letter sets and then the sorted counts.

class Solution {
public:
    bool closeStrings(string word1, string word2) {
        if (word1.size() != word2.size()) return false;   // no operation changes the length
        vector<int> c1(26, 0), c2(26, 0);                 // c[x] = occurrences of letter x
        for (char ch : word1) c1[ch - 'a']++;
        for (char ch : word2) c2[ch - 'a']++;
        for (int x = 0; x < 26; x++)                      // operation 2 cannot create a new letter,
            if ((c1[x] == 0) != (c2[x] == 0)) return false; // so both must use the same letters
        sort(c1.begin(), c1.end());                       // operation 2 swaps counts between letters:
        sort(c2.begin(), c2.end());                       // only the multiset of counts must match
        return c1 == c2;                                  // operation 1 then fixes the order
    }
};
func closeStrings(word1 string, word2 string) bool {
	if len(word1) != len(word2) { // no operation changes the length
		return false
	}
	var c1, c2 [26]int // c[x] = occurrences of letter x
	for i := 0; i < len(word1); i++ {
		c1[word1[i]-'a']++
		c2[word2[i]-'a']++
	}
	for x := 0; x < 26; x++ { // operation 2 cannot create a new letter: same letters needed
		if (c1[x] == 0) != (c2[x] == 0) {
			return false
		}
	}
	sort.Ints(c1[:]) // operation 2 swaps counts between letters:
	sort.Ints(c2[:]) // only the multiset of counts must match
	return c1 == c2  // arrays compare element by element; operation 1 fixes the order
}

static class Solution {
    public boolean closeStrings(String word1, String word2) {
        if (word1.length() != word2.length()) return false;   // no operation changes the length
        int[] c1 = new int[26], c2 = new int[26];             // c[x] = occurrences of letter x
        for (int i = 0; i < word1.length(); i++) {
            c1[word1.charAt(i) - 'a']++;
            c2[word2.charAt(i) - 'a']++;
        }
        for (int x = 0; x < 26; x++)                          // operation 2 cannot create a new letter,
            if ((c1[x] == 0) != (c2[x] == 0)) return false;   // so both must use the same letters
        Arrays.sort(c1);                                      // operation 2 swaps counts between letters:
        Arrays.sort(c2);                                      // only the multiset of counts must match
        return Arrays.equals(c1, c2);                         // operation 1 then fixes the order
    }
}
class Solution:
    def closeStrings(self, word1: str, word2: str) -> bool:
        if len(word1) != len(word2):          # no operation changes the length
            return False
        c1, c2 = Counter(word1), Counter(word2)
        # operation 2 cannot create a new letter, so both must use the same letters;
        # it swaps counts between letters, so only the multiset of counts must match
        return c1.keys() == c2.keys() and sorted(c1.values()) == sorted(c2.values())
