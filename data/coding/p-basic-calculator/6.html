class Solution {
public:
    int calculate(string s) {
        long long total = 0;       // value of the current bracket level so far
        long long num = 0;         // number being read digit by digit
        int sign = 1;              // sign that applies to num (+1 or -1)
        vector<pair<long long, int>> saved;   // (total, sign before '(') for each open bracket

        for (char c : s) {
            if (isdigit((unsigned char)c)) {
                num = num * 10 + (c - '0');
            } else if (c == '+' || c == '-') {
                total += sign * num;           // the number before the operator is complete
                num = 0;
                sign = (c == '+') ? 1 : -1;
            } else if (c == '(') {
                saved.push_back({total, sign});   // park the outer level
                total = 0;                     // start a fresh level inside the bracket
                sign = 1;
            } else if (c == ')') {
                total += sign * num;           // close the last number inside the bracket
                num = 0;
                auto outer = saved.back();
                saved.pop_back();
                total = outer.first + outer.second * total;   // apply the sign that preceded '('
                sign = 1;
            }
            // spaces are skipped
        }
        return (int)(total + sign * num);
    }
};
func calculate(s string) int {
	total := 0 // value of the current bracket level so far (Go int is 64-bit)
	num := 0   // number being read digit by digit
	sign := 1  // sign that applies to num (+1 or -1)
	type level struct{ total, sign int }
	var saved []level // the outer level parked at each open bracket

	for i := 0; i < len(s); i++ {
		c := s[i]
		switch {
		case c >= '0' && c <= '9':
			num = num*10 + int(c-'0')
		case c == '+' || c == '-':
			total += sign * num // the number before the operator is complete
			num = 0
			sign = 1
			if c == '-' {
				sign = -1
			}
		case c == '(':
			saved = append(saved, level{total, sign}) // park the outer level
			total, sign = 0, 1                        // start a fresh level inside the bracket
		case c == ')':
			total += sign * num // close the last number inside the bracket
			num = 0
			outer := saved[len(saved)-1]
			saved = saved[:len(saved)-1]
			total = outer.total + outer.sign*total // apply the sign that preceded '('
			sign = 1
		}
		// spaces are skipped
	}
	return total + sign*num
}

class Solution:
    def calculate(self, s: str) -> int:
        total = 0          # value of the current bracket level so far
        num = 0            # number being read digit by digit
        sign = 1           # sign that applies to num (+1 or -1)
        saved = []         # (total, sign before '(') for each open bracket

        for c in s:
            if c.isdigit():
                num = num * 10 + int(c)
            elif c in "+-":
                total += sign * num            # the number before the operator is complete
                num = 0
                sign = 1 if c == "+" else -1
            elif c == "(":
                saved.append((total, sign))    # park the outer level
                total, sign = 0, 1             # start a fresh level inside the bracket
            elif c == ")":
                total += sign * num            # close the last number inside the bracket
                num = 0
                outer_total, outer_sign = saved.pop()
                total = outer_total + outer_sign * total   # apply the sign that preceded '('
                sign = 1
            # spaces are skipped
        return total + sign * num
