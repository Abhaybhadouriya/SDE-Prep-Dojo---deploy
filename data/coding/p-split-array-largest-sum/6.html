func splitArray(nums []int, k int) int {
	lo, hi := 0, 0
	for _, x := range nums {
		if x > lo {
			lo = x // some part must hold the biggest value
		}
		hi += x // one part holding everything (int is 64-bit on LeetCode)
	}
	// feasible(cap) is false...false true...true: find the first true
	for lo < hi {
		mid := lo + (hi-lo)/2
		if partsNeeded(nums, mid) <= k {
			hi = mid // mid works; the answer is mid or smaller
		} else {
			lo = mid + 1 // too many parts; the answer is larger
		}
	}
	return lo
}

// partsNeeded fills the current part greedily until the next value would overflow limit.
func partsNeeded(nums []int, limit int) int {
	parts, run := 1, 0
	for _, x := range nums {
		if run+x > limit { // x starts a new part
			parts++
			run = 0
		}
		run += x
	}
	return parts
}
class Solution {
    // Greedy check: if no part may exceed cap, how few parts do we need?
    // Keep adding to the current part; open a new part only when the next value would overflow cap.
    int partsNeeded(const vector<int>& nums, long long cap) {
        int parts = 1;
        long long run = 0;
        for (int x : nums) {
            if (run + x > cap) {   // x does not fit: close this part, x starts the next one
                parts++;
                run = 0;
            }
            run += x;
        }
        return parts;
    }

class Solution:
    def splitArray(self, nums: List[int], k: int) -> int:
        def parts_needed(cap: int) -> int:
            # greedy: fill the current part until the next value would overflow cap
            parts, run = 1, 0
            for x in nums:
                if run + x > cap:      # x starts a new part
                    parts += 1
                    run = 0
                run += x
            return parts

        lo, hi = max(nums), sum(nums)  # biggest single value .. everything in one part
        # feasible(cap) is False...False True...True: find the first True
        while lo < hi:
            mid = (lo + hi) // 2
            if parts_needed(mid) <= k:
                hi = mid               # mid works; the answer is mid or smaller
            else:
                lo = mid + 1           # too many parts; the answer is larger
        return lo

def splitArray(self, nums, k):
    n = len(nums)
    pre = [0] * (n + 1)
    for i, x in enumerate(nums):
        pre[i + 1] = pre[i] + x
    INF = float("inf")
    prev = [0] + [INF] * n                      # p = 0 pieces: only the empty prefix
    for p in range(1, k + 1):
        cur = [INF] * (n + 1)
        for i in range(p, n + 1):               # need at least p values for p pieces
            for j in range(p - 1, i):           # last piece is nums[j..i-1]
                cur[i] = min(cur[i], max(prev[j], pre[i] - pre[j]))
        prev = cur
    return prev[n]
