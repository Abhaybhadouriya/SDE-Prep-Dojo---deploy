class MedianFinder {
    priority_queue<int> low;                              // max-heap: the smaller half
    priority_queue<int, vector<int>, greater<int>> high;  // min-heap: the larger half
    // invariant: every value in low <= every value in high,
    //            low.size() == high.size() or low.size() == high.size() + 1
public:
    MedianFinder() {}

    void addNum(int num) {
        low.push(num);              // 1. enter through the low half
        high.push(low.top());       // 2. its largest value moves up, so the order rule holds
        low.pop();
        if (high.size() > low.size()) {   // 3. give one back if high got bigger
            low.push(high.top());
            high.pop();
        }
    }

    double findMedian() {
        if (low.size() > high.size()) return low.top();   // odd count: the middle is low's top
        return (low.top() + (double)high.top()) / 2.0;   // even count: mean of the two tops
    }
};
// minHeap of ints; the low half stores negated values so it acts as a max-heap.
type minHeap []int

func (h minHeap) Len() int           { return len(h) }
func (h minHeap) Less(i, j int) bool { return h[i] < h[j] }
func (h minHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *minHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *minHeap) Pop() any {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

// invariant: max(low) <= min(high), low.Len() == high.Len() or high.Len()+1
type MedianFinder struct {
	low  *minHeap // smaller half, negated
	high *minHeap // larger half
}

func Constructor() MedianFinder {
	return MedianFinder{low: &minHeap{}, high: &minHeap{}}
}

func (m *MedianFinder) AddNum(num int) {
	heap.Push(m.low, -num)                     // 1. enter through the low half
	heap.Push(m.high, -heap.Pop(m.low).(int))  // 2. low's largest value moves up
	if m.high.Len() > m.low.Len() {            // 3. give one back if high got bigger
		heap.Push(m.low, -heap.Pop(m.high).(int))
	}
}

func (m *MedianFinder) FindMedian() float64 {
	if m.low.Len() > m.high.Len() {
		return float64(-(*m.low)[0]) // odd count
	}
	return float64(-(*m.low)[0]+(*m.high)[0]) / 2.0 // even count
}
static class MedianFinder {
    // low = max-heap of the smaller half, high = min-heap of the larger half
    // invariant: every value in low <= every value in high, low.size() is high.size() or +1
    private final PriorityQueue<Integer> low = new PriorityQueue<>(Collections.reverseOrder());
    private final PriorityQueue<Integer> high = new PriorityQueue<>();

    public MedianFinder() {}

    public void addNum(int num) {
        low.offer(num);              // 1. enter through the low half
        high.offer(low.poll());      // 2. its largest value moves up, so the order rule holds
        if (high.size() > low.size()) {
            low.offer(high.poll());  // 3. give one back if high got bigger
        }
    }

    public double findMedian() {
        if (low.size() > high.size()) return low.peek();   // odd count
        return ((double) low.peek() + high.peek()) / 2.0;  // even count
    }
}
class MedianFinder:
    def __init__(self):
        # low is a max-heap of the smaller half (stored negated, heapq is a min-heap)
        # high is a min-heap of the larger half
        # invariant: max(low) <= min(high), len(low) == len(high) or len(high) + 1
        self.low = []
        self.high = []

    def addNum(self, num: int) -> None:
        # 1-2. enter through low, then move low's largest value up to high
        heapq.heappush(self.high, -heapq.heappushpop(self.low, -num))
        if len(self.high) > len(self.low):  # 3. give one back if high got bigger
            heapq.heappush(self.low, -heapq.heappop(self.high))

    def findMedian(self) -> float:
        if len(self.low) > len(self.high):  # odd count
            return float(-self.low[0])
        return (-self.low[0] + self.high[0]) / 2.0  # even count
