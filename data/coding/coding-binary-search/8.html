// Closed range [lo, hi]. Index of target or -1.
func findExact(nums []int, target int) int {
	lo, hi := 0, len(nums)-1
	for lo <= hi { // range not empty
		mid := lo + (hi-lo)/2
		if nums[mid] == target {
			return mid
		}
		if nums[mid] < target {
			lo = mid + 1 // mid is not it: drop it
		} else {
			hi = mid - 1
		}
	}
	return -1
}

// Half-open [lo, hi). First index with nums[i] >= x (len if none). Same as sort.SearchInts.
func lowerBound(nums []int, x int) int {
	lo, hi := 0, len(nums)
	for lo < hi {
		mid := lo + (hi-lo)/2
		if nums[mid] < x {
			lo = mid + 1 // too small: answer is right of mid
		} else {
			hi = mid // mid might be the answer: keep it
		}
	}
	return lo
}

// First index with nums[i] > x.
func upperBound(nums []int, x int) int {
	lo, hi := 0, len(nums)
	for lo < hi {
		mid := lo + (hi-lo)/2
		if nums[mid] <= x {
			lo = mid + 1
		} else {
			hi = mid
		}
	}
	return lo
}

// Smallest i in [lo, hi) with pred(i) true, or hi. pred must look like F..F T..T.
// The standard library version is sort.Search(n, pred) over [0, n).
func firstTrue(lo, hi int64, pred func(int64) bool) int64 {
	for lo < hi {
		mid := lo + (hi-lo)/2
		if pred(mid) {
			hi = mid
		} else {
			lo = mid + 1
		}
	}
	return lo
}

// floor(sqrt(n)) = (first x with x*x > n) - 1. int64 so x*x cannot overflow.
func intSqrt(n int) int {
	nn := int64(n)
	return int(firstTrue(0, nn+2, func(x int64) bool { return x*x > nn }) - 1)
}

// Distinct values, sorted then rotated.
func searchRotated(nums []int, target int) int {
	lo, hi := 0, len(nums)-1
	for lo <= hi {
		mid := lo + (hi-lo)/2
		if nums[mid] == target {
			return mid
		}
		if nums[lo] <= nums[mid] { // left half sorted
			if nums[lo] <= target && target < nums[mid] {
				hi = mid - 1
			} else {
				lo = mid + 1
			}
		} else { // right half sorted
			if nums[mid] < target && target <= nums[hi] {
				lo = mid + 1
			} else {
				hi = mid - 1
			}
		}
	}
	return -1
}

func findMinRotated(nums []int) int {
	lo, hi := 0, len(nums)-1
	for lo < hi {
		mid := lo + (hi-lo)/2
		if nums[mid] > nums[hi] {
			lo = mid + 1 // drop is right of mid
		} else {
			hi = mid
		}
	}
	return nums[lo]
}

func findPeak(nums []int) int {
	lo, hi := 0, len(nums)-1
	for lo < hi {
		mid := lo + (hi-lo)/2
		if nums[mid] < nums[mid+1] {
			lo = mid + 1 // uphill to the right
		} else {
			hi = mid
		}
	}
	return lo
}

// Rows sorted and each row starts after the previous row ends: one flat sorted array.
func searchMatrix(m [][]int, target int) bool {
	rows, cols := len(m), len(m[0])
	lo, hi := 0, rows*cols-1
	for lo <= hi {
		mid := lo + (hi-lo)/2
		v := m[mid/cols][mid%cols]
		if v == target {
			return true
		}
		if v < target {
			lo = mid + 1
		} else {
			hi = mid - 1
		}
	}
	return false
}

// Rows and columns sorted. Start top-right. O(rows + cols).
func searchMatrixStaircase(m [][]int, target int) bool {
	r, c := 0, len(m[0])-1
	for r < len(m) && c >= 0 {
		if m[r][c] == target {
			return true
		}
		if m[r][c] > target {
			c--
		} else {
			r++
		}
	}
	return false
}

type entry struct {
	t     int
	value string
}

// TimeMap: set(key, value, t) with increasing t per key; get = value at the latest t' <= t.
type TimeMap struct{ data map[string][]entry }

func NewTimeMap() *TimeMap { return &TimeMap{data: map[string][]entry{}} }

func (tm *TimeMap) Set(key, value string, t int) {
	tm.data[key] = append(tm.data[key], entry{t, value})
}

func (tm *TimeMap) Get(key string, t int) string {
	v := tm.data[key]
	i := sort.Search(len(v), func(i int) bool { return v[i].t > t }) // first time > t
	if i == 0 {
		return ""
	}
	return v[i-1].value
}

func check(ok bool, what string) {
	if !ok {
		panic(what)
	}
}

func main() {
	a := []int{-3, 0, 4, 8, 11, 15}
	check(findExact(a, 8) == 3 && findExact(a, 5) == -1 && findExact(nil, 1) == -1, "exact")
	b := []int{1, 3, 3, 3, 7, 9}
	for x := -1; x < 12; x++ {
		check(lowerBound(b, x) == sort.SearchInts(b, x), "lower")
		check(upperBound(b, x) == sort.SearchInts(b, x+1), "upper")
	}
	check(lowerBound(b, 3) == 1 && upperBound(b, 3) == 4, "bounds")
	check(intSqrt(0) == 0 && intSqrt(1) == 1 && intSqrt(8) == 2 && intSqrt(16) == 4, "sqrt")
	check(intSqrt(2147395599) == 46339 && intSqrt(2147483647) == 46340, "sqrt big")
	r := []int{4, 5, 6, 7, 0, 1, 2}
	check(searchRotated(r, 0) == 4 && searchRotated(r, 3) == -1, "rotated")
	check(searchRotated([]int{1}, 0) == -1 && searchRotated([]int{3, 1}, 1) == 1, "rotated small")
	for i, v := range r {
		check(searchRotated(r, v) == i, "rotated all")
	}
	check(findMinRotated([]int{3, 4, 5, 1, 2}) == 1 && findMinRotated([]int{11, 13, 15, 17}) == 11, "min")
	p := []int{1, 2, 1, 3, 5, 6, 4}
	i := findPeak(p)
	check((i == 0 || p[i] > p[i-1]) && (i == len(p)-1 || p[i] > p[i+1]), "peak")
	m := [][]int{{1, 3, 5, 7}, {10, 11, 16, 20}, {23, 30, 34, 60}}
	check(searchMatrix(m, 3) && !searchMatrix(m, 13) && searchMatrix(m, 60), "matrix")
	s := [][]int{{1, 4, 7, 11, 15}, {2, 5, 8, 12, 19}, {3, 6, 9, 16, 22}, {10, 13, 14, 17, 24}, {18, 21, 23, 26, 30}}
	check(searchMatrixStaircase(s, 5) && !searchMatrixStaircase(s, 20), "staircase")
	tm := NewTimeMap()
	tm.Set("foo", "bar", 1)
	check(tm.Get("foo", 1) == "bar" && tm.Get("foo", 3) == "bar", "tm1")
	tm.Set("foo", "bar2", 4)
	check(tm.Get("foo", 4) == "bar2" && tm.Get("foo", 5) == "bar2", "tm2")
	check(tm.Get("foo", 0) == "" && tm.Get("nope", 9) == "", "tm3")
	fmt.Println("ok")
}

// Closed range [lo, hi]. Index of target or -1.
int findExact(const vector<int>& nums, int target) {
    int lo = 0, hi = (int)nums.size() - 1;
    while (lo <= hi) {                         // range not empty
        int mid = lo + (hi - lo) / 2;          // no overflow
        if (nums[mid] == target) return mid;
        if (nums[mid] < target) lo = mid + 1;  // mid is not it: drop it
        else hi = mid - 1;
    }
    return -1;
}

// Half-open [lo, hi). First index with nums[i] >= x (size if none). Same as std::lower_bound.
int lowerBound(const vector<int>& nums, int x) {
    int lo = 0, hi = (int)nums.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] < x) lo = mid + 1;       // too small: answer is right of mid
        else hi = mid;                         // mid might be the answer: keep it
    }
    return lo;
}

// First index with nums[i] > x. Same as std::upper_bound.
int upperBound(const vector<int>& nums, int x) {
    int lo = 0, hi = (int)nums.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] <= x) lo = mid + 1;
        else hi = mid;
    }
    return lo;
}

// Smallest i in [lo, hi) with pred(i) true, or hi. pred must look like F..F T..T.
long long firstTrue(long long lo, long long hi, const function<bool(long long)>& pred) {
    while (lo < hi) {
        long long mid = lo + (hi - lo) / 2;
        if (pred(mid)) hi = mid;
        else lo = mid + 1;
    }
    return lo;
}

// floor(sqrt(n)) = (first x with x*x > n) - 1. long long so x*x cannot overflow.
int intSqrt(int n) {
    return (int)(firstTrue(0, (long long)n + 2, [n](long long x) { return x * x > n; }) - 1);
}

// Distinct values, sorted then rotated.
int searchRotated(const vector<int>& nums, int target) {

        if (nums[mid] == target) return mid;
        if (nums[lo] <= nums[mid]) {                       // left half sorted
            if (nums[lo] <= target && target < nums[mid]) hi = mid - 1;
            else lo = mid + 1;
        } else {                                           // right half sorted
            if (nums[mid] < target && target <= nums[hi]) lo = mid + 1;
            else hi = mid - 1;
        }
    }
    return -1;
}

int findMinRotated(const vector<int>& nums) {
    int lo = 0, hi = (int)nums.size() - 1;

        int mid = lo + (hi - lo) / 2;
        if (nums[mid] > nums[hi]) lo = mid + 1;            // drop is right of mid
        else hi = mid;
    }
    return nums[lo];
}

int findPeak(const vector<int>& nums) {
    int lo = 0, hi = (int)nums.size() - 1;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] < nums[mid + 1]) lo = mid + 1;       // uphill to the right
        else hi = mid;
    }
    return lo;
}

// Rows sorted and each row starts after the previous row ends: one flat sorted array.
bool searchMatrix(const vector<vector<int>>& m, int target) {
    int rows = (int)m.size(), cols = (int)m[0].size();
    int lo = 0, hi = rows * cols - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        int v = m[mid / cols][mid % cols];
        if (v == target) return true;
        if (v < target) lo = mid + 1; else hi = mid - 1;
    }
    return false;
}

// Rows and columns sorted. Start top-right. O(rows + cols).
bool searchMatrixStaircase(const vector<vector<int>>& m, int target) {
    int r = 0, c = (int)m[0].size() - 1;
    while (r < (int)m.size() && c >= 0) {
        if (m[r][c] == target) return true;
        if (m[r][c] > target) c--; else r++;
    }
    return false;
}

// set(key, value, t) with increasing t per key; get = value at the latest t' <= t.
class TimeMap {
    unordered_map<string, vector<pair<int, string>>> data;
public:
    void set(const string& key, const string& value, int t) { data[key].push_back({t, value}); }
    string get(const string& key, int t) {
        auto it = data.find(key);
        if (it == data.end()) return "";
        const auto& v = it->second;
        int lo = 0, hi = (int)v.size();               // first index with time > t
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (v[mid].first <= t) lo = mid + 1; else hi = mid;
        }
        return lo == 0 ? "" : v[lo - 1].second;
    }
};

import java.util.*;
import java.util.function.LongPredicate;

public class BinarySearch {

    // Closed range [lo, hi]. Index of target or -1.
    static int findExact(int[] nums, int target) {
        int lo = 0, hi = nums.length - 1;
        while (lo <= hi) {                          // range not empty
            int mid = lo + (hi - lo) / 2;           // or (lo + hi) >>> 1
            if (nums[mid] == target) return mid;
            if (nums[mid] < target) lo = mid + 1;   // mid is not it: drop it
            else hi = mid - 1;
        }
        return -1;
    }

    // Half-open [lo, hi). First index with nums[i] >= x (length if none).
    static int lowerBound(int[] nums, int x) {
        int lo = 0, hi = nums.length;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] < x) lo = mid + 1;        // too small: answer is right of mid
            else hi = mid;                          // mid might be the answer: keep it
        }
        return lo;
    }

    // First index with nums[i] > x.
    static int upperBound(int[] nums, int x) {
        int lo = 0, hi = nums.length;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] <= x) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }

    // Smallest i in [lo, hi) with pred(i) true, or hi. pred must look like F..F T..T.
    static long firstTrue(long lo, long hi, LongPredicate pred) {
        while (lo < hi) {
            long mid = lo + (hi - lo) / 2;
            if (pred.test(mid)) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }

    // floor(sqrt(n)) = (first x with x*x > n) - 1. long so x*x cannot overflow.
    static int intSqrt(int n) {
        return (int) (firstTrue(0, (long) n + 2, x -> x * x > n) - 1);
    }

    // Distinct values, sorted then rotated.
    static int searchRotated(int[] nums, int target) {
        int lo = 0, hi = nums.length - 1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] == target) return mid;
            if (nums[lo] <= nums[mid]) {                        // left half sorted
                if (nums[lo] <= target && target < nums[mid]) hi = mid - 1;
                else lo = mid + 1;
            } else {                                            // right half sorted
                if (nums[mid] < target && target <= nums[hi]) lo = mid + 1;
                else hi = mid - 1;
            }
        }
        return -1;
    }

    static int findMinRotated(int[] nums) {
        int lo = 0, hi = nums.length - 1;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] > nums[hi]) lo = mid + 1;             // drop is right of mid
            else hi = mid;
        }
        return nums[lo];
    }

    static int findPeak(int[] nums) {
        int lo = 0, hi = nums.length - 1;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] < nums[mid + 1]) lo = mid + 1;        // uphill to the right
            else hi = mid;
        }
        return lo;
    }

    // Rows sorted and each row starts after the previous row ends: one flat sorted array.
    static boolean searchMatrix(int[][] m, int target) {
        int rows = m.length, cols = m[0].length;
        int lo = 0, hi = rows * cols - 1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            int v = m[mid / cols][mid % cols];
            if (v == target) return true;
            if (v < target) lo = mid + 1; else hi = mid - 1;
        }
        return false;
    }

    // Rows and columns sorted. Start top-right. O(rows + cols).
    static boolean searchMatrixStaircase(int[][] m, int target) {
        int r = 0, c = m[0].length - 1;
        while (r < m.length && c >= 0) {
            if (m[r][c] == target) return true;
            if (m[r][c] > target) c--; else r++;
        }
        return false;
    }

    // set(key, value, t) with increasing t per key; get = value at the latest t' <= t.
    static class TimeMap {
        private final Map<String, TreeMap<Integer, String>> data = new HashMap<>();
        void set(String key, String value, int t) {
            data.computeIfAbsent(key, k -> new TreeMap<>()).put(t, value);
        }
        String get(String key, int t) {
            TreeMap<Integer, String> m = data.get(key);
            if (m == null) return "";
            Map.Entry<Integer, String> e = m.floorEntry(t);   // a balanced BST doing the binary search
            return e == null ? "" : e.getValue();
        }
    }

    static void check(boolean ok, String what) {
        if (!ok) throw new AssertionError(what);
    }

    public static void main(String[] args) {
        int[] a = {-3, 0, 4, 8, 11, 15};
        check(findExact(a, 8) == 3 && findExact(a, 5) == -1 && findExact(new int[0], 1) == -1, "exact");
        int[] b = {1, 3, 3, 3, 7, 9};
        check(lowerBound(b, 3) == 1 && upperBound(b, 3) == 4, "bounds");
        check(lowerBound(b, 0) == 0 && lowerBound(b, 10) == 6 && lowerBound(b, 4) == 4, "bounds edges");
        check(intSqrt(0) == 0 && intSqrt(1) == 1 && intSqrt(8) == 2 && intSqrt(16) == 4, "sqrt");
        check(intSqrt(2147395599) == 46339 && intSqrt(2147483647) == 46340, "sqrt big");
        int[] r = {4, 5, 6, 7, 0, 1, 2};
        check(searchRotated(r, 0) == 4 && searchRotated(r, 3) == -1, "rotated");
        check(searchRotated(new int[]{1}, 0) == -1 && searchRotated(new int[]{3, 1}, 1) == 1, "rotated small");
        for (int i = 0; i < r.length; i++) check(searchRotated(r, r[i]) == i, "rotated all");
        check(findMinRotated(new int[]{3, 4, 5, 1, 2}) == 1 && findMinRotated(new int[]{11, 13, 15, 17}) == 11, "min");
        int[] p = {1, 2, 1, 3, 5, 6, 4};
        int i = findPeak(p);
        check((i == 0 || p[i] > p[i - 1]) && (i == p.length - 1 || p[i] > p[i + 1]), "peak");
        int[][] m = {{1, 3, 5, 7}, {10, 11, 16, 20}, {23, 30, 34, 60}};
        check(searchMatrix(m, 3) && !searchMatrix(m, 13) && searchMatrix(m, 60), "matrix");
        int[][] s = {{1, 4, 7, 11, 15}, {2, 5, 8, 12, 19}, {3, 6, 9, 16, 22}, {10, 13, 14, 17, 24}, {18, 21, 23, 26, 30}};
        check(searchMatrixStaircase(s, 5) && !searchMatrixStaircase(s, 20), "staircase");
        TimeMap tm = new TimeMap();
        tm.set("foo", "bar", 1);
        check(tm.get("foo", 1).equals("bar") && tm.get("foo", 3).equals("bar"), "tm1");
        tm.set("foo", "bar2", 4);
        check(tm.get("foo", 4).equals("bar2") && tm.get("foo", 5).equals("bar2"), "tm2");
        check(tm.get("foo", 0).equals("") && tm.get("nope", 9).equals(""), "tm3");
        System.out.println("ok");
    }
}

import bisect


def find_exact(nums, target):
    """Closed range [lo, hi]. Returns an index of target or -1."""
    lo, hi = 0, len(nums) - 1
    while lo <= hi:                    # range not empty
        mid = lo + (hi - lo) // 2
        if nums[mid] == target:
            return mid
        if nums[mid] < target:
            lo = mid + 1               # mid is not it: drop it
        else:
            hi = mid - 1
    return -1


def lower_bound(nums, x):
    """Half-open [lo, hi). First index with nums[i] >= x (len(nums) if none)."""
    lo, hi = 0, len(nums)
    while lo < hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] < x:
            lo = mid + 1               # mid is too small: answer is right of it
        else:
            hi = mid                   # mid might be the answer: keep it
    return lo


def upper_bound(nums, x):
    """First index with nums[i] > x."""
    lo, hi = 0, len(nums)
    while lo < hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] <= x:
            lo = mid + 1
        else:
            hi = mid
    return lo


def first_true(lo, hi, pred):
    """Smallest i in [lo, hi) with pred(i) True, or hi. pred must be F..F T..T."""
    while lo < hi:
        mid = lo + (hi - lo) // 2
        if pred(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo


def int_sqrt(n):
    """floor(sqrt(n)) = (first x with x*x > n) - 1."""
    return first_true(0, n + 2, lambda x: x * x > n) - 1


def search_rotated(nums, target):
    """Distinct values, sorted then rotated. Index of target or -1."""
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] == target:
            return mid
        if nums[lo] <= nums[mid]:                  # left half [lo..mid] is sorted
            if nums[lo] <= target < nums[mid]:
                hi = mid - 1
            else:
                lo = mid + 1
        else:                                      # right half [mid..hi] is sorted
            if nums[mid] < target <= nums[hi]:
                lo = mid + 1
            else:
                hi = mid - 1
    return -1


def find_min_rotated(nums):
    """Distinct values. Compare with the right end: that side tells where the drop is."""
    lo, hi = 0, len(nums) - 1
    while lo < hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] > nums[hi]:
            lo = mid + 1               # drop is right of mid
        else:
            hi = mid                   # mid..hi is sorted, min is at mid or left
    return nums[lo]


def find_peak(nums):
    """Any index i with nums[i] > both neighbours (outside = -infinity). Neighbours differ."""
    lo, hi = 0, len(nums) - 1
    while lo < hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] < nums[mid + 1]:
            lo = mid + 1               # uphill to the right: a peak is there
        else:
            hi = mid                   # downhill: mid or something left is a peak
    return lo


def search_matrix(m, target):
    """Rows sorted, each row starts after the previous row ends: one flat sorted array."""
    rows, cols = len(m), len(m[0])
    lo, hi = 0, rows * cols - 1
    while lo <= hi:
        mid = lo + (hi - lo) // 2
        v = m[mid // cols][mid % cols]
        if v == target:
            return True
        if v < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return False


def search_matrix_staircase(m, target):
    """Rows and columns sorted (rows may overlap). Start top-right; O(rows + cols)."""
    r, c = 0, len(m[0]) - 1
    while r < len(m) and c >= 0:
        if m[r][c] == target:
            return True
        if m[r][c] > target:
            c -= 1                     # whole column below is even bigger
        else:
            r += 1                     # whole row to the left is even smaller
    return False


class TimeMap:
    """set(key, value, t) with increasing t per key; get(key, t) = value at the latest t' <= t."""

    def __init__(self):
        self.times = {}
        self.values = {}

    def set(self, key, value, t):
        self.times.setdefault(key, []).append(t)   # appended in time order: already sorted
        self.values.setdefault(key, []).append(value)

    def get(self, key, t):
        ts = self.times.get(key, [])
        i = upper_bound(ts, t) - 1                 # last time <= t
        return self.values[key][i] if i >= 0 else ""
