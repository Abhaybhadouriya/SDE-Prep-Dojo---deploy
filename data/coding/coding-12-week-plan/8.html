Template: turn a week's row into a daily list

You can build any week (or your own variant) with two small functions. split spreads a week's count over the study days, putting the extra problems late in the week when the lesson has sunk in. planWeek takes the top N problems of each group in Tier A order and deals them out Easy first. Change the quotas and you have your own plan.

package main

import (
	"fmt"
	"sort"
	"strings"
)

type Prob struct {
	Slug string
	Diff byte // 'E', 'M' or 'H'
}

type Quota struct {
	Group string
	Count int
}

func rankOf(d byte) int { // easy first inside a week
	switch d {
	case 'E':
		return 0
	case 'M':
		return 1
	}
	return 2
}

// split spreads n new problems over `days` study days; the extra ones go to the later days.
func split(n, days int) []int {
	out := make([]int, days)
	for i := range out {
		out[i] = n / days
		if i >= days-n%days {
			out[i]++
		}
	}
	return out
}

// planWeek takes the top of each group's queue (Tier A order) and deals them out easy first.
func planWeek(quotas []Quota, pool map[string][]Prob, days int) [][]string {
	var picked []Prob
	for _, q := range quotas {
		g := pool[q.Group]
		if q.Count < len(g) {
			g = g[:q.Count]
		}
		picked = append(picked, g...)
	}
	sort.SliceStable(picked, func(a, b int) bool { return rankOf(picked[a].Diff) < rankOf(picked[b].Diff) })
	var out [][]string
	i := 0
	for _, k := range split(len(picked), days) {
		var day []string
		for j := 0; j < k; j++ {
			day = append(day, picked[i].Slug)
			i++
		}
		out = append(out, day)
	}
	return out
}

func printSplit(n, days int) {
	parts := []string{}
	for _, k := range split(n, days) {
		parts = append(parts, fmt.Sprint(k))
	}
	fmt.Printf("split(%d, %d): %s\n", n, days, strings.Join(parts, " "))
}

func main() {
	printSplit(20, 6)
	printSplit(24, 6)
	pool := map[string][]Prob{
		"two-pointers": {{"3sum", 'M'}, {"valid-palindrome", 'E'},
			{"container-with-most-water", 'M'}, {"move-zeroes", 'E'}},
		"sliding-window": {{"minimum-window-substring", 'H'}, {"maximum-average-subarray-i", 'E'},
			{"longest-substring-without-repeating-characters", 'M'}, {"permutation-in-string", 'M'}},
	}
	week := planWeek([]Quota{{"two-pointers", 3}, {"sliding-window", 3}}, pool, 3)
	for d, slugs := range week {
		fmt.Printf("day %d: %s\n", d+1, strings.Join(slugs, ", "))
	}
}

struct Prob { string slug; char diff; };          // diff: 'E', 'M' or 'H'

int rankOf(char d) { return d == 'E' ? 0 : d == 'M' ? 1 : 2; }   // easy first inside a week

// n new problems over `days` study days; the extra ones go to the later days.
vector<int> split(int n, int days) {
    vector<int> out(days, n / days);
    for (int i = days - n % days; i < days; i++) out[i]++;
    return out;
}

// quotas = (group, count); pool[group] = problems in Tier A order.
vector<vector<string>> planWeek(const vector<pair<string, int>>& quotas,
                                map<string, vector<Prob>>& pool, int days) {
    vector<Prob> picked;
    for (auto& q : quotas) {
        auto& g = pool[q.first];
        for (int i = 0; i < q.second && i < (int)g.size(); i++) picked.push_back(g[i]);
    }
    stable_sort(picked.begin(), picked.end(),
                [](const Prob& a, const Prob& b) { return rankOf(a.diff) < rankOf(b.diff); });
    vector<vector<string>> out;
    size_t i = 0;
    for (int k : split((int)picked.size(), days)) {
        vector<string> day;
        for (int j = 0; j < k; j++) day.push_back(picked[i++].slug);
        out.push_back(day);
    }
    return out;
}

void printSplit(int n, int days) {
    printf("split(%d, %d):", n, days);
    for (int k : split(n, days)) printf(" %d", k);
    printf("\n");
}

import java.util.*;

public class Templates {
    record Prob(String slug, char diff) {}                // diff: 'E', 'M' or 'H'

    static int rankOf(char d) { return d == 'E' ? 0 : d == 'M' ? 1 : 2; }  // easy first

    // n new problems over `days` study days; the extra ones go to the later days.
    static int[] split(int n, int days) {
        int[] out = new int[days];
        Arrays.fill(out, n / days);
        for (int i = days - n % days; i < days; i++) out[i]++;
        return out;
    }

    // quotas = (group, count); pool.get(group) = problems in Tier A order.
    static List<List<String>> planWeek(List<Map.Entry<String, Integer>> quotas,
                                       Map<String, List<Prob>> pool, int days) {
        List<Prob> picked = new ArrayList<>();
        for (var q : quotas) {
            List<Prob> g = pool.get(q.getKey());
            picked.addAll(g.subList(0, Math.min(q.getValue(), g.size())));
        }
        picked.sort(Comparator.comparingInt(p -> rankOf(p.diff())));  // List.sort is stable
        List<List<String>> out = new ArrayList<>();
        int i = 0;
        for (int k : split(picked.size(), days)) {
            List<String> day = new ArrayList<>();
            for (int j = 0; j < k; j++) day.add(picked.get(i++).slug());
            out.add(day);
        }
        return out;
    }

    static void printSplit(int n, int days) {
        StringBuilder sb = new StringBuilder("split(" + n + ", " + days + "):");
        for (int k : split(n, days)) sb.append(' ').append(k);
        System.out.println(sb);
    }

    public static void main(String[] args) {
        printSplit(20, 6);
        printSplit(24, 6);
        Map<String, List<Prob>> pool = new HashMap<>();
        pool.put("two-pointers", List.of(new Prob("3sum", 'M'), new Prob("valid-palindrome", 'E'),
                new Prob("container-with-most-water", 'M'), new Prob("move-zeroes", 'E')));
        pool.put("sliding-window", List.of(new Prob("minimum-window-substring", 'H'),
                new Prob("maximum-average-subarray-i", 'E'),
                new Prob("longest-substring-without-repeating-characters", 'M'),
                new Prob("permutation-in-string", 'M')));
        var week = planWeek(List.of(Map.entry("two-pointers", 3), Map.entry("sliding-window", 3)), pool, 3);
        for (int d = 0; d < week.size(); d++)
            System.out.println("day " + (d + 1) + ": " + String.join(", ", week.get(d)));
    }
}
RANK = {"E": 0, "M": 1, "H": 2}                 # easy first inside a week


def split(n, days):
    """n new problems over `days` study days; the extra ones go to the later days."""
    base, extra = divmod(n, days)
    return [base] * (days - extra) + [base + 1] * extra


def plan_week(quotas, pool, days):
    """quotas = [(group, count)]; pool[group] = [(slug, diff)] in Tier A order."""
    picked = []
    for group, count in quotas:
        picked += pool[group][:count]            # top of each group's queue
    picked.sort(key=lambda p: RANK[p[1]])        # stable: keeps Tier A order on ties
    out, i = [], 0
    for k in split(len(picked), days):
        out.append([slug for slug, _ in picked[i:i + k]])
        i += k
    return out


def main():
    print("split(20, 6):", *split(20, 6))
    print("split(24, 6):", *split(24, 6))
    pool = {
        "two-pointers": [("3sum", "M"), ("valid-palindrome", "E"),
                         ("container-with-most-water", "M"), ("move-zeroes", "E")],
        "sliding-window": [("minimum-window-substring", "H"), ("maximum-average-subarray-i", "E"),
                           ("longest-substring-without-repeating-characters", "M"),
                           ("permutation-in-string", "M")],
    }
    week = plan_week([("two-pointers", 3), ("sliding-window", 3)], pool, 3)
    for d, slugs in enumerate(week, 1):
        print(f"day {d}:", ", ".join(slugs))


if __name__ == "__main__":
    main()

day 3: longest-substring-without-repeating-characters, minimum-window-substring

Why a stable sort: inside one difficulty the Tier A order is kept, and that order already puts the most-asked problems of a subgroup first. An unstable sort would shuffle them. The whole thing is O(n log n) for n picked problems, which for 26 problems a week does not matter; the point is that the rule is simple enough to apply by hand.
