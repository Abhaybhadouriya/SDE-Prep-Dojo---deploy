Recursive, in place, in four languages. Each file in coding-code/invert-binary-tree/ builds trees from level order, checks the inverted level order, checks the same root object is returned, and checks that inverting twice gives back the original.

class Solution {
public:
    TreeNode* invertTree(TreeNode* root) {
        if (root == nullptr) return nullptr;     // empty subtree mirrors to itself
        TreeNode* newLeft = invertTree(root->right);   // mirror both subtrees first...
        TreeNode* newRight = invertTree(root->left);
        root->left = newLeft;                    // ...then hang them on the opposite sides
        root->right = newRight;
        return root;
    }
};
func invertTree(root *TreeNode) *TreeNode {
	if root == nil {
		return nil // empty subtree mirrors to itself
	}
	newLeft := invertTree(root.Right) // mirror both subtrees first...
	newRight := invertTree(root.Left)
	root.Left, root.Right = newLeft, newRight // ...then hang them on the opposite sides
	return root
}
