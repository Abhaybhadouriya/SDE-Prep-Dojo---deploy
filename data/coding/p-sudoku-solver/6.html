Bitmask backtracking in four languages. Each file in coding-code/sudoku-solver/ solves five boards (a hard newspaper puzzle, the classic sample, one blank, an empty board and an already-solved board) and checks that every row, column and box holds 1-9 and every given digit is unchanged.

func solveSudoku(board [][]byte) {
	var rowUsed, colUsed, boxUsed [9]int // bit d set = digit d already placed
	var empty [][2]int                   // the cells we must fill, in scan order
	boxOf := func(r, c int) int { return (r/3)*3 + c/3 }
	toggle := func(r, c, d int) { // XOR: the same call places and un-places
		bit := 1 << d
		rowUsed[r] ^= bit
		colUsed[c] ^= bit
		boxUsed[boxOf(r, c)] ^= bit
	}
	for r := 0; r < 9; r++ {
		for c := 0; c < 9; c++ {
			if board[r][c] == '.' {
				empty = append(empty, [2]int{r, c})
			} else {
				toggle(r, c, int(board[r][c]-'0'))
			}
		}
	}
	var fill func(k int) bool
	fill = func(k int) bool {
		if k == len(empty) {
			return true // every blank has a digit: solved
		}
		r, c := empty[k][0], empty[k][1]
		used := rowUsed[r] | colUsed[c] | boxUsed[boxOf(r, c)]
		for d := 1; d <= 9; d++ {
			if used&(1<<d) != 0 {
				continue // clashes with the row, column or box
			}
			toggle(r, c, d)
			board[r][c] = byte('0' + d)
			if fill(k + 1) {
				return true // stop at the first full solution
			}
			toggle(r, c, d) // undo and try the next digit
			board[r][c] = '.'
		}
		return false // dead end: caller must change its choice
	}
	fill(0)
}
class Solution {
    int rowUsed[9] = {}, colUsed[9] = {}, boxUsed[9] = {};  // bit d set = digit d already placed
    vector<pair<int, int>> empty;                             // the cells we must fill, in scan order

    static int boxOf(int r, int c) { return (r / 3) * 3 + c / 3; }

    void place(vector<vector<char>>& b, int r, int c, int d, bool on) {
        int bit = 1 << d;
        rowUsed[r] ^= bit; colUsed[c] ^= bit; boxUsed[boxOf(r, c)] ^= bit;  // XOR toggles: same call undoes
        b[r][c] = on ? char('0' + d) : '.';
    }

    bool fill(vector<vector<char>>& b, size_t k) {
        if (k == empty.size()) return true;                   // every blank has a digit: solved
        int r = empty[k].first, c = empty[k].second;
        int used = rowUsed[r] | colUsed[c] | boxUsed[boxOf(r, c)];
        for (int d = 1; d <= 9; ++d) {
            if (used & (1 << d)) continue;                    // clashes with the row, column or box
            place(b, r, c, d, true);
            if (fill(b, k + 1)) return true;                  // stop at the first full solution
            place(b, r, c, d, false);                         // undo and try the next digit
        }
        return false;                                         // dead end: caller must change its choice
    }

public:
    void solveSudoku(vector<vector<char>>& board) {
        for (int r = 0; r < 9; ++r)
            for (int c = 0; c < 9; ++c) {
                if (board[r][c] == '.') empty.push_back({r, c});
                else place(board, r, c, board[r][c] - '0', true);
            }
        fill(board, 0);
    }
};

static class Solution {
    private final int[] rowUsed = new int[9], colUsed = new int[9], boxUsed = new int[9]; // bit d = digit d placed
    private final List<int[]> empty = new ArrayList<>();                                  // cells to fill, scan order

    private static int boxOf(int r, int c) { return (r / 3) * 3 + c / 3; }

    private void toggle(int r, int c, int d) {
        int bit = 1 << d;                      // XOR: the same call places and un-places
        rowUsed[r] ^= bit; colUsed[c] ^= bit; boxUsed[boxOf(r, c)] ^= bit;
    }

    private boolean fill(char[][] b, int k) {
        if (k == empty.size()) return true;    // every blank has a digit: solved
        int r = empty.get(k)[0], c = empty.get(k)[1];
        int used = rowUsed[r] | colUsed[c] | boxUsed[boxOf(r, c)];
        for (int d = 1; d <= 9; d++) {
            if ((used & (1 << d)) != 0) continue;   // clashes with the row, column or box
            toggle(r, c, d);
            b[r][c] = (char) ('0' + d);
            if (fill(b, k + 1)) return true;        // stop at the first full solution
            toggle(r, c, d);                        // undo and try the next digit
            b[r][c] = '.';
        }
        return false;                              // dead end: caller must change its choice
    }

    public void solveSudoku(char[][] board) {
        for (int r = 0; r < 9; r++)
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') empty.add(new int[]{r, c});
                else toggle(r, c, board[r][c] - '0');
            }
        fill(board, 0);
    }
}
class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        row_used = [0] * 9          # bit d set = digit d already in that row
        col_used = [0] * 9
        box_used = [0] * 9
        empty = []                  # the cells we must fill, in scan order

        def toggle(r: int, c: int, d: int) -> None:
            bit = 1 << d            # XOR: the same call places and un-places
            row_used[r] ^= bit
            col_used[c] ^= bit
            box_used[(r // 3) * 3 + c // 3] ^= bit

        for r in range(9):
            for c in range(9):
                if board[r][c] == ".":
                    empty.append((r, c))
                else:
                    toggle(r, c, int(board[r][c]))

        def fill(k: int) -> bool:
            if k == len(empty):
                return True                         # every blank has a digit: solved
            r, c = empty[k]
            used = row_used[r] | col_used[c] | box_used[(r // 3) * 3 + c // 3]
            for d in range(1, 10):
                if used & (1 << d):
                    continue                        # clashes with the row, column or box
                toggle(r, c, d)
                board[r][c] = str(d)
                if fill(k + 1):
                    return True                     # stop at the first full solution
                toggle(r, c, d)                     # undo and try the next digit
                board[r][c] = "."
            return False                            # dead end: caller must change its choice

        fill(0)
