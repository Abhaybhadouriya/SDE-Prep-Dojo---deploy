func canAttendMeetings(intervals [][]int) bool {
	sort.Slice(intervals, func(i, j int) bool { return intervals[i][0] < intervals[j][0] }) // by start
	for i := 1; i < len(intervals); i++ {
		if intervals[i][0] < intervals[i-1][1] { // starts before the previous one ends
			return false // (equal is fine: [s, e) is half-open)
		}
	}
	return true
}
class Solution {
public:
    bool canAttendMeetings(vector<vector<int>>& intervals) {
        sort(intervals.begin(), intervals.end(),
             [](const vector<int>& a, const vector<int>& b) { return a[0] < b[0]; });   // by start
        for (size_t i = 1; i < intervals.size(); i++)
            if (intervals[i][0] < intervals[i - 1][1])   // starts before the previous one ends
                return false;                            // (equal is fine: [s, e) is half-open)
        return true;
    }
};
