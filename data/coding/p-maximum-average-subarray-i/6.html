class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        long long sum = 0;
        for (int i = 0; i < k; i++) sum += nums[i];   // first window nums[0..k-1]
        long long best = sum;
        for (int i = k; i < (int)nums.size(); i++) {
            sum += nums[i] - nums[i - k];             // nums[i] enters, nums[i-k] leaves
            best = max(best, sum);
        }
        return (double)best / k;                       // compare sums, divide once
    }
};
func findMaxAverage(nums []int, k int) float64 {
	sum := 0
	for i := 0; i < k; i++ { // first window nums[0..k-1]
		sum += nums[i]
	}
	best := sum
	for i := k; i < len(nums); i++ {
		sum += nums[i] - nums[i-k] // nums[i] enters, nums[i-k] leaves
		if sum > best {
			best = sum
		}
	}
	return float64(best) / float64(k) // compare sums, divide once
}
