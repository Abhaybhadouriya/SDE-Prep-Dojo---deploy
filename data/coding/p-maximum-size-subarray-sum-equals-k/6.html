One pass with a 64-bit running sum and a first-index map; the tested files are in coding-code/maximum-size-subarray-sum-equals-k/ (each driver includes a 2·105-element case whose sum is 2·109, and the Python driver cross-checks 400 random arrays against the O(n2) version).

class Solution {
public:
    int maxSubArrayLen(vector<int>& nums, int k) {
        // sum(l+1..r) = pre[r] - pre[l]. For the end r, the longest subarray
        // with sum k starts right after the EARLIEST l with pre[l] == pre[r] - k.
        unordered_map<long long, int> first;
        first[0] = -1;                  // prefix 0 "before index 0": runs starting at index 0
        long long run = 0;              // |run| <= 2*10^5 * 10^4 = 2*10^9 > INT_MAX: 64-bit
        int best = 0;
        for (int i = 0; i < (int)nums.size(); i++) {
            run += nums[i];
            auto it = first.find(run - k);
            if (it != first.end()) best = max(best, i - it->second);
            first.emplace(run, i);      // emplace never overwrites: the first index stays
        }
        return best;
    }
};
func maxSubArrayLen(nums []int, k int) int {
	// sum(l+1..r) = pre[r] - pre[l]. For the end r, the longest subarray
	// with sum k starts right after the EARLIEST l with pre[l] == pre[r] - k.
	first := map[int]int{0: -1} // prefix 0 "before index 0": runs starting at index 0
	run, best := 0, 0           // Go int is 64-bit: |run| <= 2*10^9 is safe
	for i, x := range nums {
		run += x
		if j, ok := first[run-k]; ok && i-j > best {
			best = i - j
		}
		if _, seen := first[run]; !seen {
			first[run] = i // keep the first index only
		}
	}
	return best
}

static class Solution {
    public int maxSubArrayLen(int[] nums, int k) {
        // sum(l+1..r) = pre[r] - pre[l]. For the end r, the longest subarray
        // with sum k starts right after the EARLIEST l with pre[l] == pre[r] - k.
        Map<Long, Integer> first = new HashMap<>();
        first.put(0L, -1);              // prefix 0 "before index 0": runs starting at index 0
        long run = 0;                   // |run| <= 2*10^5 * 10^4 = 2*10^9 > Integer.MAX_VALUE
        int best = 0;
        for (int i = 0; i < nums.length; i++) {
            run += nums[i];
            Integer j = first.get(run - k);
            if (j != null) best = Math.max(best, i - j);
            first.putIfAbsent(run, i);  // keep the first index only
        }
        return best;
    }
}
class Solution:
    def maxSubArrayLen(self, nums: List[int], k: int) -> int:
        # sum(l+1..r) = pre[r] - pre[l]. For the end r, the longest subarray
        # with sum k starts right after the EARLIEST l with pre[l] == pre[r] - k.
        first = {0: -1}              # prefix 0 "before index 0": runs starting at index 0
        run = best = 0
        for i, x in enumerate(nums):
            run += x
            j = first.get(run - k)
            if j is not None:
                best = max(best, i - j)
            if run not in first:     # keep the first index only: later ones give shorter runs
                first[run] = i
        return best
