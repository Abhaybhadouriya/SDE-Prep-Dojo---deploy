class Solution {
public:
    int rob(vector<int>& nums) {
        int prev2 = 0;  // best total up to house i-2
        int prev1 = 0;  // best total up to house i-1
        for (int x : nums) {
            // skip house i (prev1) or take it (x + prev2, so house i-1 is untouched)
            int cur = max(prev1, prev2 + x);
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
};
func rob(nums []int) int {
	prev2, prev1 := 0, 0 // best totals up to house i-2 and house i-1
	for _, x := range nums {
		// skip house i (prev1) or take it (x + prev2, so house i-1 is untouched)
		cur := prev1
		if prev2+x > cur {
			cur = prev2 + x
		}
		prev2, prev1 = prev1, cur
	}
	return prev1
}
