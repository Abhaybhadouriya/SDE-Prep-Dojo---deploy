Both the rolling-row DP and the binomial version, in four languages. The test drivers (the examples, 1 × 1, 1 × n, 100 × 1, the largest answers near the cap, and every grid up to 14 × 14 checked against a memoised brute force) are in coding-code/unique-paths/.

class Solution {
public:
    int uniquePaths(int m, int n) {
        // ways[c] = number of paths to cell (current row, c)
        vector<int> ways(n, 1);  // row 0: only "all right" reaches each cell
        for (int r = 1; r < m; r++)
            for (int c = 1; c < n; c++)
                ways[c] += ways[c - 1];  // from above (old ways[c]) + from the left
        return ways[n - 1];
    }

    int uniquePathsMath(int m, int n) {
        // choose which of the m+n-2 moves are the m-1 "down" moves
        int k = min(m, n) - 1, total = m + n - 2;
        long long res = 1;
        for (int i = 1; i <= k; i++)
            res = res * (total - k + i) / i;  // exact: res becomes C(total-k+i, i)
        return (int)res;
    }
};
func uniquePaths(m int, n int) int {
	// ways[c] = number of paths to cell (current row, c)
	ways := make([]int, n)
	for c := range ways {
		ways[c] = 1 // row 0: only "all right" reaches each cell
	}
	for r := 1; r < m; r++ {
		for c := 1; c < n; c++ {
			ways[c] += ways[c-1] // from above (old ways[c]) + from the left
		}
	}
	return ways[n-1]
}

func uniquePathsMath(m int, n int) int {
	// choose which of the m+n-2 moves are the m-1 "down" moves
	k, total := m-1, m+n-2
	if n-1 < k {
		k = n - 1
	}
	res := int64(1)
	for i := 1; i <= k; i++ {
		res = res * int64(total-k+i) / int64(i) // exact: res becomes C(total-k+i, i)
	}
	return int(res)
}

class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        # ways[c] = number of paths to cell (current row, c)
        ways = [1] * n               # row 0: only "all right" reaches each cell
        for _ in range(1, m):
            for c in range(1, n):
                # from above (old ways[c]) + from the left (new ways[c-1])
                ways[c] += ways[c - 1]
        return ways[n - 1]

    def uniquePathsMath(self, m: int, n: int) -> int:
        # choose which of the m+n-2 moves are the m-1 "down" moves
        k = min(m, n) - 1
        total = m + n - 2
        res = 1
        for i in range(1, k + 1):
            res = res * (total - k + i) // i   # stays an integer: C(total-k+i, i)
        return res
