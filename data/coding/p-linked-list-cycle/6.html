func hasCycle(head *ListNode) bool {
	slow, fast := head, head // slow: 1 step per round, fast: 2 steps
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
		if slow == fast {
			return true // the hare caught the tortoise inside a loop
		}
	}
	return false // the hare fell off the end: no loop
}
class Solution {
public:
    bool hasCycle(ListNode* head) {
        ListNode* slow = head;              // 1 step per round
        ListNode* fast = head;              // 2 steps per round
        while (fast != nullptr && fast->next != nullptr) {
            slow = slow->next;
            fast = fast->next->next;
            if (slow == fast) return true;  // the hare caught the tortoise inside a loop
        }
        return false;                       // the hare fell off the end: no loop
    }
};
