1. Longest palindromic substring (expand around the centre)

// longestPalindrome expands around all 2n-1 centres. O(n^2) time, O(1) extra space.
func longestPalindrome(s string) string {
	bestStart, bestLen := 0, 0
	expand := func(lo, hi int) {
		for lo >= 0 && hi < len(s) && s[lo] == s[hi] {
			lo--
			hi++
		}
		if l := hi - lo - 1; l > bestLen {
			bestLen, bestStart = l, lo+1
		}
	}
	for c := 0; c < len(s); c++ {
		expand(c, c)   // odd length
		expand(c, c+1) // even length
	}
	return s[bestStart : bestStart+bestLen]
}
// Longest palindromic substring: expand around all 2n-1 centres. O(n^2) time, O(1) extra space.
string longestPalindrome(const string& s) {
    int n = s.size(), bestStart = 0, bestLen = 0;
    auto expand = [&](int lo, int hi) {
        while (lo >= 0 && hi < n && s[lo] == s[hi]) { --lo; ++hi; }
        int len = hi - lo - 1;                       // window is (lo, hi) exclusive
        if (len > bestLen) { bestLen = len; bestStart = lo + 1; }
    };
    for (int c = 0; c < n; ++c) {
        expand(c, c);        // odd length, centre on a character
        expand(c, c + 1);    // even length, centre between two characters
    }
    return s.substr(bestStart, bestLen);
}
// Longest palindromic substring: expand around all 2n-1 centres. O(n^2) time, O(1) extra space.
static String longestPalindrome(String s) {
    int bestStart = 0, bestLen = 0;
    for (int c = 0; c < s.length(); c++) {
        for (int odd = 0; odd < 2; odd++) {          // odd = 0: centre c; odd = 1: between c and c+1
            int lo = c, hi = c + odd;
            while (lo >= 0 && hi < s.length() && s.charAt(lo) == s.charAt(hi)) { lo--; hi++; }
            int len = hi - lo - 1;
            if (len > bestLen) { bestLen = len; bestStart = lo + 1; }
        }
    }
    return s.substring(bestStart, bestStart + bestLen);
}
def longest_palindrome(s: str) -> str:
    """Longest palindromic substring by expanding around all 2n-1 centres. O(n^2) time, O(1) space."""
    def expand(lo: int, hi: int) -> tuple:
        while lo >= 0 and hi < len(s) and s[lo] == s[hi]:
            lo -= 1
            hi += 1
        return lo + 1, hi - lo - 1          # start, length of the palindrome found

    best_start, best_len = 0, 0
    for c in range(len(s)):
        for lo, hi in ((c, c), (c, c + 1)):  # odd centre, even centre
            start, length = expand(lo, hi)
            if length > best_len:
                best_start, best_len = start, length
    return s[best_start:best_start + best_len]

2. String to integer with a 32-bit clamp

// myAtoi: skip spaces, read one sign, read digits, clamp to 32-bit.
func myAtoi(s string) int {
	i, n := 0, len(s)
	for i < n && s[i] == ' ' {
		i++
	}
	sign := 1
	if i < n && (s[i] == '+' || s[i] == '-') {
		if s[i] == '-' {
			sign = -1
		}
		i++
	}
	val := 0
	for i < n && s[i] >= '0' && s[i] <= '9' {
		d := int(s[i] - '0')
		if val > (math.MaxInt32-d)/10 { // val*10 + d would pass 2^31-1
			if sign == 1 {
				return math.MaxInt32
			}
			return math.MinInt32
		}
		val = val*10 + d
		i++
	}
	return sign * val
}
// Skip spaces, read one sign, read digits, clamp to 32-bit. One pass, no 64-bit tricks needed.
int myAtoi(const string& s) {
    size_t i = 0, n = s.size();
    while (i < n && s[i] == ' ') ++i;
    int sign = 1;
    if (i < n && (s[i] == '+' || s[i] == '-')) { sign = (s[i] == '-') ? -1 : 1; ++i; }
    int val = 0;                                     // kept non-negative, <= INT_MAX
    while (i < n && isdigit((unsigned char)s[i])) {
        int d = s[i] - '0';
        if (val > (INT_MAX - d) / 10)                // val*10 + d would overflow
            return sign == 1 ? INT_MAX : INT_MIN;    // also covers -2147483648 exactly
        val = val * 10 + d;
        ++i;
    }
    return sign * val;
}
// Skip spaces, read one sign, read digits, clamp to 32-bit.
static int myAtoi(String s) {
    int i = 0, n = s.length();
    while (i < n && s.charAt(i) == ' ') i++;
    int sign = 1;
    if (i < n && (s.charAt(i) == '+' || s.charAt(i) == '-')) { sign = s.charAt(i) == '-' ? -1 : 1; i++; }
    int val = 0;
    while (i < n && s.charAt(i) >= '0' && s.charAt(i) <= '9') {   // not Character.isDigit: it accepts non-ASCII digits
        int d = s.charAt(i) - '0';
        if (val > (Integer.MAX_VALUE - d) / 10)         // val*10 + d would overflow
            return sign == 1 ? Integer.MAX_VALUE : Integer.MIN_VALUE;
        val = val * 10 + d;
        i++;
    }
    return sign * val;
}
INT_MAX, INT_MIN = 2**31 - 1, -2**31

def my_atoi(s: str) -> int:
    """Skip spaces, read one sign, read digits, clamp to 32-bit. One pass, O(n)."""
    i, n = 0, len(s)
    while i < n and s[i] == ' ':
        i += 1
    sign = 1
    if i < n and s[i] in '+-':
        sign = -1 if s[i] == '-' else 1
        i += 1
    limit = INT_MAX if sign == 1 else -INT_MIN   # 2147483647 or 2147483648
    val = 0
    while i < n and '0' <= s[i] <= '9':          # not isdigit(): it accepts '²' and other Unicode digits
        d = ord(s[i]) - ord('0')
        if val > (limit - d) // 10:               # val*10 + d would pass the limit
            return INT_MAX if sign == 1 else INT_MIN
        val = val * 10 + d
        i += 1
    return sign * val

3. KMP: failure table and search

// buildLps: lps[i] = length of the longest proper prefix of p[:i+1] that is also its suffix.
func buildLps(p string) []int {
	lps := make([]int, len(p))
	for i, k := 1, 0; i < len(p); i++ {
		for k > 0 && p[i] != p[k] {
			k = lps[k-1]
		}
		if p[i] == p[k] {
			k++
		}
		lps[i] = k
	}
	return lps
}

// strStr returns the first index of pat in text, or -1. O(n + m).
func strStr(text, pat string) int {
	if len(pat) == 0 {
		return 0
	}
	lps := buildLps(pat)
	for i, k := 0, 0; i < len(text); i++ {
		for k > 0 && text[i] != pat[k] {
			k = lps[k-1]
		}
		if text[i] == pat[k] {
			k++
		}
		if k == len(pat) {
			return i - k + 1
		}
	}
	return -1
}
// lps[i] = length of the longest proper prefix of p[0..i] that is also a suffix of it.
vector<int> buildLps(const string& p) {
    vector<int> lps(p.size(), 0);
    for (size_t i = 1, k = 0; i < p.size(); ++i) {
        while (k > 0 && p[i] != p[k]) k = lps[k - 1];  // fall back to a shorter border
        if (p[i] == p[k]) ++k;
        lps[i] = k;
    }
    return lps;
}

// First index of pat in text, or -1. O(n + m).
int strStr(const string& text, const string& pat) {
    if (pat.empty()) return 0;
    vector<int> lps = buildLps(pat);
    for (size_t i = 0, k = 0; i < text.size(); ++i) {
        while (k > 0 && text[i] != pat[k]) k = lps[k - 1];
        if (text[i] == pat[k]) ++k;
        if (k == pat.size()) return (int)(i - k + 1);
    }
    return -1;
}
// lps[i] = length of the longest proper prefix of p[0..i] that is also a suffix of it.
static int[] buildLps(String p) {
    int[] lps = new int[p.length()];
    for (int i = 1, k = 0; i < p.length(); i++) {
        while (k > 0 && p.charAt(i) != p.charAt(k)) k = lps[k - 1];
        if (p.charAt(i) == p.charAt(k)) k++;
        lps[i] = k;
    }
    return lps;
}

// First index of pat in text, or -1. O(n + m).
static int strStr(String text, String pat) {
    if (pat.isEmpty()) return 0;
    int[] lps = buildLps(pat);
    for (int i = 0, k = 0; i < text.length(); i++) {
        while (k > 0 && text.charAt(i) != pat.charAt(k)) k = lps[k - 1];
        if (text.charAt(i) == pat.charAt(k)) k++;
        if (k == pat.length()) return i - k + 1;
    }
    return -1;
}
def build_lps(p: str) -> list:
    """lps[i] = length of the longest proper prefix of p[:i+1] that is also its suffix."""
    lps = [0] * len(p)
    k = 0                                   # length of the current matched prefix
    for i in range(1, len(p)):
        while k > 0 and p[i] != p[k]:
            k = lps[k - 1]                  # fall back to the next shorter border
        if p[i] == p[k]:
            k += 1
        lps[i] = k
    return lps

def str_str(text: str, pat: str) -> int:
    """Index of the first occurrence of pat in text, or -1. O(n + m)."""
    if not pat:
        return 0
    lps, k = build_lps(pat), 0
    for i, ch in enumerate(text):
        while k > 0 and ch != pat[k]:
            k = lps[k - 1]
        if ch == pat[k]:
            k += 1
        if k == len(pat):
            return i - k + 1
    return -1

The three KMP lines (while mismatch and k > 0: k = lps[k-1], if match: k++) are identical in the builder and the search. Learn them once; the only difference is that the builder starts at i = 1 and stores lps[i] = k.
