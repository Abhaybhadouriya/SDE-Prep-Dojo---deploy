class Solution {
public:
    // Stack of indices of '(' still waiting for a partner; mark removals with '*'.
    string minRemoveToMakeValid(string s) {
        vector<int> openIdx;
        for (int i = 0; i < (int)s.size(); i++) {
            if (s[i] == '(') {
                openIdx.push_back(i);
            } else if (s[i] == ')') {
                if (!openIdx.empty()) openIdx.pop_back();  // matched with the latest open
                else s[i] = '*';                           // nothing to close: drop it
            }
        }
        for (int i : openIdx) s[i] = '*';                  // '(' that were never closed
        s.erase(remove(s.begin(), s.end(), '*'), s.end()); // input has no '*' of its own
        return s;
    }

    // No stack: pass 1 drops bad ')', pass 2 drops the rightmost extra '('.
    string minRemoveTwoPass(const string& s) {
        string kept;
        int balance = 0, opens = 0;
        for (char c : s) {
            if (c == ')') {
                if (balance == 0) continue;                // unmatched ')'
                balance--;
            } else if (c == '(') {
                balance++;
                opens++;
            }
            kept += c;
        }
        int keepOpen = opens - balance;                    // how many '(' can stay
        string out;
        for (char c : kept) {
            if (c == '(') {
                if (keepOpen == 0) continue;               // one of the last extra opens
                keepOpen--;
            }
            out += c;
        }
        return out;
    }
};
// minRemoveToMakeValid: stack of indices of '(' still waiting for a partner.
func minRemoveToMakeValid(s string) string {
	drop := make([]bool, len(s))
	var openIdx []int
	for i := 0; i < len(s); i++ {
		switch s[i] {
		case '(':
			openIdx = append(openIdx, i)
		case ')':
			if len(openIdx) > 0 {
				openIdx = openIdx[:len(openIdx)-1] // matched with the latest open
			} else {
				drop[i] = true // nothing to close: drop it
			}
		}
	}
	for _, i := range openIdx { // '(' that were never closed
		drop[i] = true
	}
	var b strings.Builder
	b.Grow(len(s))
	for i := 0; i < len(s); i++ {
		if !drop[i] {
			b.WriteByte(s[i])
		}
	}
	return b.String()
}

// minRemoveTwoPass: no stack; pass 1 drops bad ')', pass 2 drops the rightmost extra '('.
func minRemoveTwoPass(s string) string {
	kept := make([]byte, 0, len(s))
	balance, opens := 0, 0
	for i := 0; i < len(s); i++ {
		c := s[i]
		if c == ')' {
			if balance == 0 {
				continue // unmatched ')'
			}
			balance--
		} else if c == '(' {
			balance++
			opens++
		}
		kept = append(kept, c)
	}
	keepOpen := opens - balance // how many '(' can stay
	out := make([]byte, 0, len(kept))
	for _, c := range kept {
		if c == '(' {
			if keepOpen == 0 {
				continue // one of the last extra opens
			}
			keepOpen--
		}
		out = append(out, c)
	}
	return string(out)
}

            if (a[i] == '(') {

                if (!openIdx.isEmpty()) openIdx.pop();   // matched with the latest open

        StringBuilder sb = new StringBuilder(a.length);
        for (int i = 0; i < a.length; i++)
            if (!drop[i]) sb.append(a[i]);

class Solution:
    def minRemoveToMakeValid(self, s: str) -> str:
        """Stack of indices of '(' still waiting for a partner."""
        chars = list(s)                        # strings are immutable: edit a list
        open_idx = []
        for i, c in enumerate(chars):
            if c == '(':
                open_idx.append(i)
            elif c == ')':
                if open_idx:
                    open_idx.pop()             # matched with the latest open
                else:
                    chars[i] = ''              # a ')' with nothing to close: drop it
        for i in open_idx:                     # '(' that were never closed
            chars[i] = ''
        return ''.join(chars)

    def minRemoveTwoPass(self, s: str) -> str:
        """No stack: pass 1 drops bad ')', pass 2 drops the rightmost extra '('."""
        kept, balance, opens = [], 0, 0
        for c in s:
            if c == ')':
                if balance == 0:
                    continue                   # unmatched ')'
                balance -= 1
            elif c == '(':
                balance += 1
                opens += 1
            kept.append(c)
        keep_open = opens - balance            # how many '(' can stay
        out = []
        for c in kept:
            if c == '(':
                if keep_open == 0:
                    continue                   # one of the last `balance` opens: drop it
                keep_open -= 1
            out.append(c)
        return ''.join(out)
