class Solution {
public:
    void solve(vector<vector<char>>& board) {
        rows = board.size();
        cols = board[0].size();
        for (int r = 0; r < rows; r++) { keep(board, r, 0); keep(board, r, cols - 1); }
        for (int c = 0; c < cols; c++) { keep(board, 0, c); keep(board, rows - 1, c); }
        while (!q.empty()) {                         // spread the safe tag inward
            auto cur = q.front(); q.pop();
            int r = cur.first, c = cur.second;
            keep(board, r + 1, c); keep(board, r - 1, c);
            keep(board, r, c + 1); keep(board, r, c - 1);
        }
        for (auto& row : board)                      // capture untagged 'O', restore '#'
            for (char& ch : row)
                ch = (ch == '#') ? 'O' : 'X';
    }

private:
    int rows = 0, cols = 0;
    queue<pair<int, int>> q;

    // a border-connected 'O' is safe: tag it '#' and queue it
    void keep(vector<vector<char>>& b, int r, int c) {
        if (r < 0 || c < 0 || r >= rows || c >= cols || b[r][c] != 'O') return;
        b[r][c] = '#';
        q.push({r, c});
    }
};
func solve(board [][]byte) {
	rows, cols := len(board), len(board[0])
	queue := [][2]int{}
	keep := func(r, c int) { // a border-connected 'O' is safe: tag it '#'
		if r < 0 || c < 0 || r >= rows || c >= cols || board[r][c] != 'O' {
			return
		}
		board[r][c] = '#'
		queue = append(queue, [2]int{r, c})
	}
	for r := 0; r < rows; r++ {
		keep(r, 0)
		keep(r, cols-1)
	}
	for c := 0; c < cols; c++ {
		keep(0, c)
		keep(rows-1, c)
	}
	for head := 0; head < len(queue); head++ { // spread the safe tag inward
		r, c := queue[head][0], queue[head][1]
		keep(r+1, c)
		keep(r-1, c)
		keep(r, c+1)
		keep(r, c-1)
	}
	for _, row := range board { // capture untagged 'O', restore '#'
		for c := range row {
			if row[c] == '#' {
				row[c] = 'O'
			} else {
				row[c] = 'X'
			}
		}
	}
}

static class Solution {
    private int rows, cols;
    private final ArrayDeque<int[]> q = new ArrayDeque<>();

    public void solve(char[][] board) {
        rows = board.length;
        cols = board[0].length;
        for (int r = 0; r < rows; r++) { keep(board, r, 0); keep(board, r, cols - 1); }
        for (int c = 0; c < cols; c++) { keep(board, 0, c); keep(board, rows - 1, c); }
        while (!q.isEmpty()) {                       // spread the safe tag inward
            int[] cur = q.poll();
            int r = cur[0], c = cur[1];
            keep(board, r + 1, c); keep(board, r - 1, c);
            keep(board, r, c + 1); keep(board, r, c - 1);
        }
        for (char[] row : board)                     // capture untagged 'O', restore '#'
            for (int c = 0; c < cols; c++)
                row[c] = (row[c] == '#') ? 'O' : 'X';
    }

    // a border-connected 'O' is safe: tag it '#' and queue it
    private void keep(char[][] b, int r, int c) {
        if (r < 0 || c < 0 || r >= rows || c >= cols || b[r][c] != 'O') return;
        b[r][c] = '#';
        q.add(new int[]{r, c});
    }
}
class Solution:
    def solve(self, board: List[List[str]]) -> None:
        rows, cols = len(board), len(board[0])
        q = deque()

        def keep(r, c):                      # a border-connected 'O' is safe: tag it '#'
            if 0 <= r < rows and 0 <= c < cols and board[r][c] == "O":
                board[r][c] = "#"
                q.append((r, c))

        for r in range(rows):                # seed: every 'O' on the left and right edge
            keep(r, 0)
            keep(r, cols - 1)
        for c in range(cols):                # ... and on the top and bottom edge
            keep(0, c)
            keep(rows - 1, c)
        while q:                             # BFS spreads the safe tag inward
            r, c = q.popleft()
            keep(r + 1, c); keep(r - 1, c); keep(r, c + 1); keep(r, c - 1)

        for r in range(rows):                # untagged 'O' cannot reach the border: capture
            for c in range(cols):
                if board[r][c] == "O":
                    board[r][c] = "X"
                elif board[r][c] == "#":
                    board[r][c] = "O"
