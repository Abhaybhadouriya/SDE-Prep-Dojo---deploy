class Solution {
public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        // greater-than comparator turns priority_queue into a min-heap on node value
        auto cmp = [](ListNode* a, ListNode* b) { return a->val > b->val; };
        priority_queue<ListNode*, vector<ListNode*>, decltype(cmp)> heap(cmp);
        for (ListNode* head : lists)
            if (head) heap.push(head);          // skip empty lists
        ListNode dummy;
        ListNode* tail = &dummy;
        while (!heap.empty()) {
            ListNode* node = heap.top();        // smallest head among all lists
            heap.pop();
            tail->next = node;
            tail = node;
            if (node->next) heap.push(node->next);  // successor becomes the new head of that list
        }
        return dummy.next;
    }
};
// nodeHeap is a min-heap of list nodes ordered by Val (container/heap interface).
type nodeHeap []*ListNode

func (h nodeHeap) Len() int            { return len(h) }
func (h nodeHeap) Less(i, j int) bool  { return h[i].Val < h[j].Val }
func (h nodeHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *nodeHeap) Push(x interface{}) { *h = append(*h, x.(*ListNode)) }
func (h *nodeHeap) Pop() interface{} {
	old := *h
	n := old[len(old)-1]
	*h = old[:len(old)-1]
	return n
}

func mergeKLists(lists []*ListNode) *ListNode {
	h := &nodeHeap{}
	for _, head := range lists {
		if head != nil { // skip empty lists
			*h = append(*h, head)
		}
	}
	heap.Init(h)
	dummy := &ListNode{}
	tail := dummy
	for h.Len() > 0 {
		node := heap.Pop(h).(*ListNode) // smallest head among all lists
		tail.Next = node
		tail = node
		if node.Next != nil { // successor becomes that list's new head
			heap.Push(h, node.Next)
		}
	}
	return dummy.Next
}

static class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        // min-heap ordered by node value (Integer.compare avoids subtraction overflow)
        PriorityQueue<ListNode> heap = new PriorityQueue<>((a, b) -> Integer.compare(a.val, b.val));
        for (ListNode head : lists)
            if (head != null) heap.offer(head);   // skip empty lists
        ListNode dummy = new ListNode();
        ListNode tail = dummy;
        while (!heap.isEmpty()) {
            ListNode node = heap.poll();          // smallest head among all lists
            tail.next = node;
            tail = node;
            if (node.next != null) heap.offer(node.next);  // successor becomes that list's head
        }
        return dummy.next;
    }
}
class Solution:
    def mergeKLists(self, lists: List[Optional[ListNode]]) -> Optional[ListNode]:
        # heap holds (value, list index, node): the index breaks ties so nodes are never compared
        heap = [(node.val, i, node) for i, node in enumerate(lists) if node]
        heapq.heapify(heap)
        dummy = tail = ListNode()
        while heap:
            _, i, node = heapq.heappop(heap)   # smallest head among all lists
            tail.next = node
            tail = node
            if node.next:                      # its successor becomes that list's new head
                heapq.heappush(heap, (node.next.val, i, node.next))
        return dummy.next
Divide and conquer version (Python) 
def mergeKLists(self, lists):
    if not lists:
        return None
    step = 1
    while step < len(lists):                 # log k rounds
        for i in range(0, len(lists) - step, 2 * step):
            lists[i] = merge_two(lists[i], lists[i + step])
        step *= 2
    return lists[0]

def merge_two(a, b):
    dummy = tail = ListNode()
    while a and b:
        if a.val <= b.val: tail.next, a = a, a.next
        else:             tail.next, b = b, b.next
        tail = tail.next
    tail.next = a or b
    return dummy.next
