Stack of survivors in four languages. Each file in coding-code/asteroid-collision/ also cross-checks 3,000 random rows against the O(n²) replay. The Java version uses a plain int[] as the stack, which avoids boxing and gives the answer with one Arrays.copyOf.

class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        vector<int> st;                          // survivors so far, left to right
        for (int a : asteroids) {
            bool alive = true;
            // Only "top moves right, new one moves left" can collide.
            while (alive && a < 0 && !st.empty() && st.back() > 0) {
                if (st.back() < -a) {            // top is smaller: it explodes, keep checking
                    st.pop_back();
                    continue;
                }
                if (st.back() == -a) st.pop_back();  // same size: both explode
                alive = false;                   // the incoming one is gone
            }
            if (alive) st.push_back(a);
        }
        return st;
    }
};
func asteroidCollision(asteroids []int) []int {
	st := make([]int, 0, len(asteroids)) // survivors so far, left to right
	for _, a := range asteroids {
		alive := true
		// Only "top moves right, new one moves left" can collide.
		for alive && a < 0 && len(st) > 0 && st[len(st)-1] > 0 {
			top := st[len(st)-1]
			if top < -a { // top is smaller: it explodes, keep checking
				st = st[:len(st)-1]
				continue
			}
			if top == -a { // same size: both explode
				st = st[:len(st)-1]
			}
			alive = false // the incoming one is gone
		}
		if alive {
			st = append(st, a)
		}
	}
	return st
}

static class Solution {
    public int[] asteroidCollision(int[] asteroids) {
        int[] st = new int[asteroids.length];  // array used as a stack: st[0..top-1]
        int top = 0;
        for (int a : asteroids) {
            boolean alive = true;
            // Only "top moves right, new one moves left" can collide.
            while (alive && a < 0 && top > 0 && st[top - 1] > 0) {
                if (st[top - 1] < -a) {         // top is smaller: it explodes, keep checking
                    top--;
                    continue;
                }
                if (st[top - 1] == -a) top--;   // same size: both explode
                alive = false;                  // the incoming one is gone
            }
            if (alive) st[top++] = a;
        }
        return Arrays.copyOf(st, top);
    }
}
class Solution:
    def asteroidCollision(self, asteroids: List[int]) -> List[int]:
        st: List[int] = []                   # survivors so far, left to right
        for a in asteroids:
            alive = True
            # Only "top moves right, new one moves left" can collide.
            while alive and a < 0 and st and st[-1] > 0:
                if st[-1] < -a:              # top is smaller: it explodes, keep checking
                    st.pop()
                    continue
                if st[-1] == -a:             # same size: both explode
                    st.pop()
                alive = False                # the incoming one is gone
            if alive:
                st.append(a)
        return st
