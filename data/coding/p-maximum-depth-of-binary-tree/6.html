func maxDepth(root *TreeNode) int {
	if root == nil { // empty subtree adds no level
		return 0
	}
	left := maxDepth(root.Left) // deepest path on the left, in nodes
	right := maxDepth(root.Right)
	if left > right { // this node + the deeper side
		return 1 + left
	}
	return 1 + right
}
class Solution {
public:
    int maxDepth(TreeNode* root) {
        if (root == nullptr) return 0;        // empty subtree adds no level
        int left = maxDepth(root->left);      // deepest path on the left, in nodes
        int right = maxDepth(root->right);
        return 1 + max(left, right);          // this node + the deeper side
    }
};

    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:                  # empty subtree adds no level
            return 0
        left = self.maxDepth(root.left)   # deepest path on the left, in nodes
        right = self.maxDepth(root.right)
        return 1 + max(left, right)       # this node + the deeper side
