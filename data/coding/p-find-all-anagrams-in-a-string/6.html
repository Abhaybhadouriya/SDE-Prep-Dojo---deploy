class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        int n = s.size(), m = p.size();
        vector<int> res;
        if (m > n) return res;                   // no window of length len(p) fits
        int diff[26] = {0};                      // count in p - count in the window
        for (int i = 0; i < m; i++) { diff[p[i] - 'a']++; diff[s[i] - 'a']--; }
        int matched = 0;                         // letters whose diff is 0 (out of 26)
        for (int d : diff) matched += (d == 0);
        auto update = [&](int c, int delta) {    // only a move to or from 0 changes matched
            if (diff[c] == 0) matched--;
            diff[c] += delta;
            if (diff[c] == 0) matched++;
        };
        for (int start = 0; ; start++) {         // window = s[start .. start+m-1]
            if (matched == 26) res.push_back(start);
            if (start + m == n) return res;
            update(s[start + m] - 'a', -1);      // next letter enters on the right
            update(s[start] - 'a', +1);          // first letter leaves on the left
        }
    }
};
func findAnagrams(s string, p string) []int {
	n, m := len(s), len(p)
	res := []int{}
	if m > n {
		return res // no window of length len(p) fits
	}
	var diff [26]int // diff[c] = count of c in p - count of c in the window
	for i := 0; i < m; i++ {
		diff[p[i]-'a']++
		diff[s[i]-'a']--
	}
	matched := 0 // letters whose diff is 0 (out of 26)
	for _, d := range diff {
		if d == 0 {
			matched++
		}
	}
	update := func(c byte, delta int) { // only a move to or from 0 changes matched
		if diff[c] == 0 {
			matched--
		}
		diff[c] += delta
		if diff[c] == 0 {
			matched++
		}
	}
	for start := 0; ; start++ { // window = s[start : start+m]
		if matched == 26 {
			res = append(res, start)
		}
		if start+m == n {
			return res
		}
		update(s[start+m]-'a', -1) // next letter enters on the right
		update(s[start]-'a', +1)   // first letter leaves on the left
	}

    private int matched;                     // letters whose diff is 0 (out of 26)

class Solution:
    def findAnagrams(self, s: str, p: str) -> List[int]:
        n, m = len(s), len(p)
        res: List[int] = []
        if m > n:
            return res                        # no window of length len(p) fits
        diff = [0] * 26                       # count in p - count in the window
        for a, b in zip(p, s):
            diff[ord(a) - 97] += 1
            diff[ord(b) - 97] -= 1
        matched = diff.count(0)               # letters whose diff is 0 (out of 26)

        def update(c: int, delta: int) -> None:
            nonlocal matched                  # only a move to or from 0 changes matched
            if diff[c] == 0:
                matched -= 1
            diff[c] += delta
            if diff[c] == 0:
                matched += 1

        for start in range(n - m + 1):        # window = s[start : start + m]
            if matched == 26:
                res.append(start)
            if start + m == n:
                break
            update(ord(s[start + m]) - 97, -1)  # next letter enters on the right
            update(ord(s[start]) - 97, +1)      # first letter leaves on the left
        return res
