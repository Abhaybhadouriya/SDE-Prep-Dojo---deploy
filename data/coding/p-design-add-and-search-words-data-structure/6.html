Trie with 26 child slots per node, and a recursive match. Each file in coding-code/design-add-and-search-words-data-structure/ runs the three call sequences above and 3,000 random add/search pairs against the list-scan version.

class WordDictionary {
    struct Node {
        unique_ptr<Node> next[26];   // one edge per letter; null = no word continues this way
        bool end = false;            // a stored word stops exactly here
    };
    Node root;

    // Can the rest of the pattern, starting at w[i], be matched from node n?
    bool match(const Node* n, const string& w, size_t i) const {
        if (i == w.size()) return n->end;           // pattern used up: need a word END, not a prefix
        if (w[i] != '.') {                          // a letter: exactly one edge to follow
            const Node* c = n->next[w[i] - 'a'].get();
            return c && match(c, w, i + 1);
        }
        for (const auto& c : n->next)               // '.': try every existing child
            if (c && match(c.get(), w, i + 1)) return true;   // stop at the first success
        return false;
    }

public:
    WordDictionary() {}

    void addWord(string word) {                     // plain trie insert, O(L)
        Node* n = &root;
        for (char ch : word) {
            auto& slot = n->next[ch - 'a'];
            if (!slot) slot.reset(new Node());
            n = slot.get();
        }
        n->end = true;
    }

    bool search(string word) { return match(&root, word, 0); }
};
type node struct {
	next [26]*node // one edge per letter; nil = no word continues this way
	end  bool      // a stored word stops exactly here
}

type WordDictionary struct {
	root *node
}

func Constructor() WordDictionary {
	return WordDictionary{root: &node{}}

func (d *WordDictionary) AddWord(word string) { // plain trie insert, O(L)
	n := d.root
	for i := 0; i < len(word); i++ {
		c := word[i] - 'a'
		if n.next[c] == nil {
			n.next[c] = &node{}
		}
		n = n.next[c]
	}
	n.end = true
}

func (d *WordDictionary) Search(word string) bool {
	// match: can the rest of the pattern, starting at word[i], be matched from n?
	var match func(n *node, i int) bool
	match = func(n *node, i int) bool {
		if i == len(word) {
			return n.end // pattern used up: need a word END, not a prefix
		}
		if word[i] != '.' { // a letter: exactly one edge to follow
			c := n.next[word[i]-'a']
			return c != nil && match(c, i+1)
		}
		for _, c := range n.next { // '.': try every existing child
			if c != nil && match(c, i+1) {
				return true // stop at the first success
			}
		}
		return false
	}
	return match(d.root, 0)
}

static class WordDictionary {
    private static class Node {
        Node[] next = new Node[26];   // one edge per letter; null = no word continues this way
        boolean end;                  // a stored word stops exactly here
    }
    private final Node root = new Node();

    public WordDictionary() {}

    public void addWord(String word) {               // plain trie insert, O(L)
        Node n = root;
        for (char ch : word.toCharArray()) {
            if (n.next[ch - 'a'] == null) n.next[ch - 'a'] = new Node();
            n = n.next[ch - 'a'];
        }
        n.end = true;
    }

    public boolean search(String word) { return match(root, word, 0); }

    // Can the rest of the pattern, starting at w[i], be matched from node n?
    private boolean match(Node n, String w, int i) {
        if (i == w.length()) return n.end;           // pattern used up: need a word END
        char ch = w.charAt(i);
        if (ch != '.') {                             // a letter: exactly one edge to follow
            Node c = n.next[ch - 'a'];
            return c != null && match(c, w, i + 1);
        }
        for (Node c : n.next)                        // '.': try every existing child
            if (c != null && match(c, w, i + 1)) return true;   // stop at the first success
        return false;
    }
}
class WordDictionary:
    def __init__(self):
        self.root = {}               # node = dict letter -> child; key '$' marks a word end

    def addWord(self, word: str) -> None:          # plain trie insert, O(L)
        node = self.root
        for ch in word:
            node = node.setdefault(ch, {})
        node['$'] = True

    def search(self, word: str) -> bool:
        # Can the rest of the pattern, starting at word[i], be matched from node?
        def match(node: dict, i: int) -> bool:
            if i == len(word):
                return '$' in node                  # need a word END, not just a prefix
            ch = word[i]
            if ch != '.':                           # a letter: exactly one edge to follow
                return ch in node and match(node[ch], i + 1)
            return any(match(child, i + 1)          # '.': try every child; any() stops early
                       for key, child in node.items() if key != '$')

        return match(self.root, 0)
