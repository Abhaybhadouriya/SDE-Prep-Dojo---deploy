class Solution {
public:
    int candy(vector<int>& ratings) {
        int n = ratings.size();
        vector<int> give(n, 1);                 // everyone starts with 1
        // Pass 1: higher than the LEFT neighbour -> one more than it.
        for (int i = 1; i < n; ++i)
            if (ratings[i] > ratings[i - 1]) give[i] = give[i - 1] + 1;
        // Pass 2: higher than the RIGHT neighbour -> one more than it, keeping pass 1's demand.
        for (int i = n - 2; i >= 0; --i)
            if (ratings[i] > ratings[i + 1]) give[i] = max(give[i], give[i + 1] + 1);
        int total = 0;                          // at most n(n+1)/2 ~ 1.25e9 < 2^31-1 for n = 5e4
        for (int g : give) total += g;
        return total;
    }
};
func candy(ratings []int) int {
	n := len(ratings)
	give := make([]int, n)
	for i := range give {
		give[i] = 1 // everyone starts with 1
	}
	// Pass 1: higher than the LEFT neighbour -> one more than it.
	for i := 1; i < n; i++ {
		if ratings[i] > ratings[i-1] {
			give[i] = give[i-1] + 1
		}
	}
	// Pass 2: higher than the RIGHT neighbour -> one more than it, keeping pass 1's demand.
	for i := n - 2; i >= 0; i-- {
		if ratings[i] > ratings[i+1] && give[i] <= give[i+1] {
			give[i] = give[i+1] + 1
		}
	}
	total := 0
	for _, g := range give {
		total += g
	}
	return total
}

        for (int i = n - 2; i >= 0; i--)
            if (ratings[i] > ratings[i + 1]) give[i] = Math.max(give[i], give[i + 1] + 1);
        int total = 0;
        for (int g : give) total += g;
        return total;
    }

class Solution:
    def candy(self, ratings: List[int]) -> int:
        n = len(ratings)
        give = [1] * n                      # everyone starts with the minimum of 1
        # Pass 1, left -> right: a higher rating than the LEFT neighbour needs one more than it.
        for i in range(1, n):
            if ratings[i] > ratings[i - 1]:
                give[i] = give[i - 1] + 1
        # Pass 2, right -> left: a higher rating than the RIGHT neighbour needs one more than it,
        # but keep what pass 1 demanded (max), so the left rule stays true.
        for i in range(n - 2, -1, -1):
            if ratings[i] > ratings[i + 1]:
                give[i] = max(give[i], give[i + 1] + 1)
        return sum(give)
