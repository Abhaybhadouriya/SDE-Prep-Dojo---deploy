Binary search on the answer in four languages. Each file in coding-code/sqrtx/ checks the examples, 0, 1, the largest perfect square below 231 (46340² = 2,147,395,600) and its neighbour, 231-1, and every x in 0..20,000 plus the top 5,000 values against a reference.

class Solution {
public:
    int mySqrt(int x) {
        long long lo = 0, hi = x;              // 64-bit: mid*mid can reach about 2^60
        while (lo < hi) {                      // invariant: lo*lo <= x, answer in [lo, hi]
            long long mid = lo + (hi - lo + 1) / 2;   // round UP so lo = mid makes progress
            if (mid * mid <= x) lo = mid;      // mid is a valid root: keep it
            else hi = mid - 1;                 // mid is too big, and so is everything above it
        }
        return (int)lo;
    }
};
func mySqrt(x int) int {
	lo, hi := int64(0), int64(x) // int64: mid*mid can reach about 2^60

}

class Solution:
    def mySqrt(self, x: int) -> int:
        lo, hi = 0, x                    # the answer is somewhere in [0, x]
        while lo < hi:                   # invariant: lo*lo <= x and the answer is in [lo, hi]
            mid = (lo + hi + 1) // 2     # round UP, so "lo = mid" always moves lo forward
            if mid * mid <= x:
                lo = mid                 # mid is a valid root (maybe not the largest): keep it
            else:
                hi = mid - 1             # mid is too big, and so is everything above it
        return lo
